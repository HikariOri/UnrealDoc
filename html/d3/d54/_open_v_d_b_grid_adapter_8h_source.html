109// Interface for sampling and getting other values from an openVDB grid, abstracting away the actual type/format of the underlying grid.
110class IOpenVDBGridAdapterBase
111{
112public:
113 virtual void IteratePhysical(TFunctionRef<void(const FIntVector3& Coord, uint32 NumComponents, float* VoxelValues)> OnVisit) const = 0;
114 virtual float Sample(const FIntVector3& VoxelCoord, uint32 ComponentIndex) const = 0;
115 virtual void GetMinMaxValue(uint32 ComponentIndex, float* OutMinVal, float* OutMaxVal) const = 0;
116 virtual float GetBackgroundValue(uint32 ComponentIndex) const = 0;
117 virtual ~IOpenVDBGridAdapterBase() = default;
118};

120template<typename GridType>
121class TOpenVDBGridAdapter : public IOpenVDBGridAdapterBase
122{
123 using ValueType = typename GridType::ValueType;
124public:
125 explicit TOpenVDBGridAdapter(openvdb::SharedPtr<GridType> Grid)
126 : Grid(Grid), Accessor(Grid->getConstAccessor())
127 {
128 }
129
130 void IteratePhysical(TFunctionRef<void(const FIntVector3& Coord, uint32 NumComponents, float* VoxelValues)> OnVisit) const override
131 {
132 constexpr uint32 NumComponents = GetOpenVDBValueNumComponents<ValueType>();
133 static_assert(NumComponents <= 4);
134
135 // Iterate over both voxels and tiles.
136 for (auto ValueIt = Grid->cbeginValueOn(); ValueIt; ++ValueIt)
137 {
138 const ValueType VoxelValue = ValueIt.getValue();
139 const openvdb::Coord CoordVDB = ValueIt.getCoord();
140
141 float VoxelValueComponents[4]{};
142 for (uint32 ComponentIdx = 0; ComponentIdx < NumComponents; ++ComponentIdx)
143 {
144 VoxelValueComponents[ComponentIdx] = GetOpenVDBValueComponent(VoxelValue, ComponentIdx);
145 }
146 const FIntVector3 Coord(CoordVDB[0], CoordVDB[1], CoordVDB[2]);
147
148 if (ValueIt.isVoxelValue())
149 {
150 // A single voxel
151 OnVisit(Coord, NumComponents, VoxelValueComponents);
152 }
153 else
154 {
155 // Tile, a higher level node specifying a single value for all finer level nodes/children without actually allocating memory for them.
156 openvdb::CoordBBox TileAABB;
157 ValueIt.getBoundingBox(TileAABB);
158 check(TileAABB.min().x() == Coord.X && TileAABB.min().y() == Coord.Y && TileAABB.min().z() == Coord.Z);
159 const openvdb::Coord AABBDimVDB = TileAABB.dim();
160 const FIntVector3 AABBDim = FIntVector3(AABBDimVDB.x(), AABBDimVDB.y(), AABBDimVDB.z());
161
162 // Iterate over tile AABB and invoke callback for every (virtual) voxel.
163 for (int32 Z = 0; Z < AABBDim.Z; ++Z)
164 {
165 for (int32 Y = 0; Y < AABBDim.Y; ++Y)
166 {
167 for (int32 X = 0; X < AABBDim.X; ++X)
168 {
169 const FIntVector3 TileLocalCoord(Coord.X + X, Coord.Y + Y, Coord.Z + Z);
170 OnVisit(TileLocalCoord, NumComponents, VoxelValueComponents);
171 }
172 }
173 }
174 }
175 }
176 }
177
178 float Sample(const FIntVector3& VoxelCoord, uint32 ComponentIndex) const override
179 {
180 ValueType VoxelValue = Accessor.getValue(openvdb::Coord(VoxelCoord.X, VoxelCoord.Y, VoxelCoord.Z));
181 return GetOpenVDBValueComponent(VoxelValue, ComponentIndex);
182 }
183
184 void GetMinMaxValue(uint32 ComponentIndex, float* OutMinVal, float* OutMaxVal) const override
185 {
186 *OutMinVal = FLT_MAX;
187 *OutMaxVal = -FLT_MAX;
188 if (auto Iter = Grid->tree().cbeginValueOn())
189 {
190 ValueType MinVal = *Iter;
191 ValueType MaxVal = MinVal;
192
193 for (++Iter; Iter; ++Iter)
194 {
195 ValueType Value = *Iter;
196 MinVal = GetOpenVDBMinComponent(MinVal, Value);
197 MaxVal = GetOpenVDBMaxComponent(MaxVal, Value);
198 }
199
200 *OutMinVal = GetOpenVDBValueComponent(MinVal, ComponentIndex);
201 *OutMaxVal = GetOpenVDBValueComponent(MaxVal, ComponentIndex);
202 }
203 }
204
205 float GetBackgroundValue(uint32 ComponentIndex) const override
206 {
207 return GetOpenVDBValueComponent(Grid->background(), ComponentIndex);
208 }
209
210private:
211 typename GridType::Ptr Grid;
212 typename GridType::ConstAccessor Accessor;
213};
