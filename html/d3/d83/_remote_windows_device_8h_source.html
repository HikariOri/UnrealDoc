16/* Remote Windows device support currently requires the Microsoft.Gaming.RemoteIterationClient_Microsoft WinGet package.
17* (The remote devices need the Microsoft.Gaming.RemoteIterationEndpoint_Microsoft WinGetpackage)
18*
19* It is recommended to use the Xbox PC Toolbox installer to configure host and remote devices : https://aka.ms/ToolboxInstaller
20* See https://aka.ms/GameRemoteDevtools for more details
21*
22* At present, the list of devices is read from the engine ini. These devices should be paired already.
23*
24* %LOCALAPPDATA%\Unreal Engine\Engine\Config\UserEngine.ini can be used for convenience
25*
26* [RemoteWin]
27* +DeviceNames=myfirstpc
28* +DeviceNames=mysecondpc

164 static bool RunWdRemote( const FString& Parameters, const FString& DeviceName )
165 {
166 FString CommandLineArgs = Parameters;
167 if (!DeviceName.IsEmpty())
168 {
169 CommandLineArgs += FString::Printf(TEXT(" /device:%s"), *DeviceName);
170 }
171
172 FString LocalAppData = FPlatformMisc::GetEnvironmentVariable(TEXT("LOCALAPPDATA"));
173 FString WdRemotePath = FPaths::Combine(LocalAppData, TEXT("Microsoft\\WinGet\\Links\\wdRemote.exe") );
174 if (!FPaths::FileExists(WdRemotePath))
175 {
176 UE_LOGF( LogRemoteWindowsTargetDevice, Error, "%ls not found!", *WdRemotePath);
177 return false;
178 }
179
180 int32 ReturnCode = 0;
181 FString StdOut;
182 FString StdErr;
183 FPlatformProcess::ExecProcess(*WdRemotePath, *CommandLineArgs, &ReturnCode, &StdOut, &StdErr);
184
185 UE_CLOGF( (ReturnCode != 0), LogRemoteWindowsTargetDevice, Error, "wdremote %ls - failed (0x%X)\n%ls\n%ls", *CommandLineArgs, ReturnCode, *StdOut, *StdErr);
186 return (ReturnCode == 0);
187 }
