364 void DeserializeSharedInputs(
365 FArchive& Ar,
366 const TArray<FShaderCompilerEnvironment>& SharedEnvironments,
367 const TArray<TUniquePtr<FShaderParametersMetadata>>& ShaderParameterStructures)
368 {
369 check(Ar.IsLoading());
370
371 int32 SharedEnvironmentIndex = 0;
372 Ar << SharedEnvironmentIndex;
373
374 if (SharedEnvironments.IsValidIndex(SharedEnvironmentIndex))
375 {
376 Environment.Merge(SharedEnvironments[SharedEnvironmentIndex]);
377 }
378
379 int32 ShaderParameterStructureIndex = INDEX_NONE;
380 Ar << ShaderParameterStructureIndex;
381 if (ShaderParameterStructureIndex != INDEX_NONE)
382 {
383 RootParametersStructure = ShaderParameterStructures[ShaderParameterStructureIndex].Get();
384 }

580 template<typename TValue>
581 void AddStatistic(const TCHAR* Name, TValue Value, FGenericShaderStat::EFlags Flags = FGenericShaderStat::EFlags::None, FName TagName = NAME_None)
582 {
583 // handle the case where a stat with the given name already exists; some backends compile iteratively and this is a simpler way to handle that
584 // than requiring each backend to manually avoid setting until the "final" compilation. linear search is in practice fine here; there are low
585 // single digit numbers of stats for all existing shader formats at the time of writing.
586 FName StatName(Name);
587 FGenericShaderStat* Stat = nullptr;
588 for (FGenericShaderStat& ExistingStat : ShaderStatistics)
589 {
590 if (ExistingStat.StatName == StatName)
591 {
592 Stat = &ExistingStat;
593 break;
594 }
595 }
596
597 if (!Stat)
598 {
599 Stat = &ShaderStatistics.AddZeroed_GetRef();
600 Stat->StatName = StatName;
601 }
602
603 Stat->Value = FShaderStatVariant(TInPlaceType<TValue>(), Value);
604 Stat->Flags = Flags;
605 Stat->TagName = TagName;
606 }

722struct FShaderPreprocessDependency
723{
724 FShaderHash PathInSourceHash; // PathInSourceHash doesn't include PathInSource's null terminator, so hash computation can use a string view
725 TArray<ANSICHAR> PathInSource; // Path as it appears in include directive in original shader source, allowing faster case sensitive hash
726 TArray<ANSICHAR> ParentPath; // For relative paths, ResultPath is dependent on the parent file the include directive is found in
727 TArray<ANSICHAR> ResultPath;
728 uint32 ResultPathHash; // Case insensitive hash of ResultPath (compatible with hash of corresponding FString)
729 uint32 ResultPathUniqueIndex; // Index of first instance of a given result path in Dependencies array
730 FShaderSharedAnsiStringPtr StrippedSource; // Source with comments stripped out, and converted to ANSICHAR (output of ShaderConvertAndStripComments)
731
732 inline bool EqualsPathInSource(const ANSICHAR* InPathInSource, int32 InPathInSourceLen, FShaderHash InPathInSourceHash, const ANSICHAR* InParentPath) const
733 {
734 // PathInSource is case sensitive, ParentPath is case insensitive.
735 // If the path is absolute (starts with '/'), then the parent path isn't relevant, and shouldn't be checked.
736 return
737 PathInSourceHash == InPathInSourceHash &&
738 (PathInSource[0] == '/' || !FCStringAnsi::Stricmp(ParentPath.GetData(), InParentPath)) &&
739 !FCStringAnsi::Strncmp(PathInSource.GetData(), InPathInSource, InPathInSourceLen);
740 }
741
742 inline bool EqualsResultPath(const FString& InResultPath, uint32 InResultPathHash) const
743 {
744 return (ResultPathHash == InResultPathHash) && InResultPath.Equals(ResultPath.GetData(), ESearchCase::IgnoreCase);
745 }
746
747 inline bool EqualsResultPath(const ANSICHAR* InResultPath, uint32 InResultPathHash) const
748 {
749 return (ResultPathHash == InResultPathHash) && !FCStringAnsi::Stricmp(ResultPath.GetData(), InResultPath);
750 }
751};

754// Structure that provides an array of #include dependencies for a given root shader file, including not just immediate
755// dependencies, but recursive dependencies from children as well. Not exhaustive, as it does not include platform
756// specific or generated files, although it does include children of "/Engine/Generated/Material.ush", as derived from
757// "/Engine/Private/MaterialTemplate.ush". Take the example of ClearUAV.usf:
758//
759// /Engine/Private/Tools/ClearUAV.usf #include "../Common.ush"
760// /Engine/Private/Common.ush #include "/Engine/Public/Platform.ush"
761// #include "PackUnpack.ush"
762// /Engine/Public/Platform.ush #include "FP16Math.ush"
763//
764// The above is a small subset, but the above (and many more) would all show up as elements in Dependencies:
765//
766// PathInSource ParentPath ResultPath
767// --------------------------------------------------------------------------------------------------------
768// ../Common.ush /Engine/Private/Tools/ClearUAV.usf /Engine/Private/Common.ush
769// /Engine/Public/Platform.ush /Engine/Private/Common.ush /Engine/Public/Platform.ush
770// PackUnpack.ush /Engine/Private/Common.ush /Engine/Private/PackUnpack.ush
771// FP16Math.ush /Engine/Public/Platform.ush /Engine/Public/FP16Math.ush
772//
773// The goal of this structure is to allow a shader preprocessor implementation to fetch most of the source dependencies in a
774// single query of the loaded shader cache, and then efficiently search for dependencies encountered in the shader source
775// code, without needing to do string operations to resolve paths (such as converting relative paths like "../Common.ush" to
776// "/Engine/Private/Common.ush"). Besides that, the array organization can be used to manage encountered source files
777// by index, rather than needing a map, and the "ResultPath" strings from this structure can be referenced by pointer,
778// rather than needing to dynamically allocate a copy of the resolved path. Lookups by PathInSource can use a much faster
779// case sensitive hash, because PathInSource has verbatim capitalization from the source code files. Altogether, this
780// utility structure saves a bunch of shader cache query, hash, map, string, and memory allocation overhead.
781//
782struct FShaderPreprocessDependencies
783{
784 // First item in array contains stripped source for root file, and is not in the hash tables
785 TArray<FShaderPreprocessDependency> Dependencies;
786 FHashTable BySource; // Hash table by PathInSource
787 FHashTable ByResult; // Hash table by ResultPath
788};
