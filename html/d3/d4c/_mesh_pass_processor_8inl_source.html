19static EVRSShadingRate GetShadingRateFromMaterial(EMaterialShadingRate MaterialShadingRate)
20{
21 switch (MaterialShadingRate)
22 {
23 case MSR_1x1:
24 return EVRSShadingRate::VRSSR_1x1;
25 case MSR_1x2:
26 return EVRSShadingRate::VRSSR_1x2;
27 case MSR_2x1:
28 return EVRSShadingRate::VRSSR_2x1;
29 case MSR_2x2:
30 return EVRSShadingRate::VRSSR_2x2;
31 }
32
33 if (GRHISupportsLargerVariableRateShadingSizes)
34 {
35 switch (MaterialShadingRate)
36 {
37 case MSR_4x2:
38 return EVRSShadingRate::VRSSR_4x2;
39 case MSR_2x4:
40 return EVRSShadingRate::VRSSR_2x4;
41 case MSR_4x4:
42 return EVRSShadingRate::VRSSR_4x4;
43 }
44 }
45
46 return EVRSShadingRate::VRSSR_1x1;
47}

49static bool ShouldMaterialAllowHardwareVariableRateShading(const FMaterial& RESTRICT MaterialResource, bool bUsesPixelDepthOffset)
50{
51 return MaterialResource.IsVariableRateShadingAllowed()
52 && !MaterialResource.IsMasked() // When using pixel discard, coarse shading causes the whole block to get discarded resulting in noticeable artifacts
53 && !bUsesPixelDepthOffset; // When writing depth from the pixel shader, coarse shading causes the whole block to output the same depth value resulting in noticeable artifacts
54}

56template<typename PassShadersType, typename ShaderElementDataType>
57void FMeshPassProcessor::BuildMeshDrawCommands(
58 const FMeshBatch& RESTRICT MeshBatch,
59 uint64 BatchElementMask,
60 const FPrimitiveSceneProxy* RESTRICT PrimitiveSceneProxy,
61 const FMaterialRenderProxy& RESTRICT MaterialRenderProxy,
62 const FMaterial& RESTRICT MaterialResource,
63 const FMeshPassProcessorRenderState& RESTRICT DrawRenderState,
64 const PassShadersType& PassShaders,
65 ERasterizerFillMode MeshFillMode,
66 ERasterizerCullMode MeshCullMode,
67 FMeshDrawCommandSortKey SortKey,
68 EMeshPassFeatures MeshPassFeatures,
69 const ShaderElementDataType& ShaderElementData)
70{
71 const FVertexFactory* RESTRICT VertexFactory = MeshBatch.VertexFactory;
72 const FPrimitiveSceneInfo* RESTRICT PrimitiveSceneInfo = PrimitiveSceneProxy ? PrimitiveSceneProxy->GetPrimitiveSceneInfo() : nullptr;
73
74 FMeshDrawCommand SharedMeshDrawCommand;
75 EFVisibleMeshDrawCommandFlags SharedFlags = EFVisibleMeshDrawCommandFlags::Default;
76
77 if (MaterialResource.MaterialUsesWorldPositionOffset_RenderThread())
78 {
79 SharedFlags |= EFVisibleMeshDrawCommandFlags::MaterialUsesWorldPositionOffset;
80
81 if (MaterialResource.ShouldAlwaysEvaluateWorldPositionOffset())
82 {
83 SharedFlags |= EFVisibleMeshDrawCommandFlags::MaterialAlwaysEvaluatesWorldPositionOffset;
84 }
85 }
86
87 SharedMeshDrawCommand.SetStencilRef(DrawRenderState.GetStencilRef());
88 SharedMeshDrawCommand.PrimitiveType = (EPrimitiveType)MeshBatch.Type;
89
90 FGraphicsMinimalPipelineStateInitializer PipelineState;
91 PipelineState.PrimitiveType = (EPrimitiveType)MeshBatch.Type;
92
93 EVertexInputStreamType InputStreamType = EVertexInputStreamType::Default;
94 if ((MeshPassFeatures & EMeshPassFeatures::PositionOnly) != EMeshPassFeatures::Default) InputStreamType = EVertexInputStreamType::PositionOnly;
95 if ((MeshPassFeatures & EMeshPassFeatures::PositionAndNormalOnly) != EMeshPassFeatures::Default) InputStreamType = EVertexInputStreamType::PositionAndNormalOnly;
96
97 check(VertexFactory && VertexFactory->IsInitialized());
98 FRHIVertexDeclaration* VertexDeclaration = VertexFactory->GetDeclaration(InputStreamType);
99
100 check(!VertexFactory->NeedsDeclaration() || VertexDeclaration);
101
102 const FMeshProcessorShaders MeshProcessorShaders = PassShaders.GetUntypedShaders();
103 PipelineState.SetupBoundShaderState(VertexDeclaration, MeshProcessorShaders);
104
105 SharedMeshDrawCommand.InitializeShaderBindings(MeshProcessorShaders);
106
107 PipelineState.RasterizerState = GetStaticRasterizerState<true>(MeshFillMode, MeshCullMode);
108
109 check(DrawRenderState.GetDepthStencilState());
110 check(DrawRenderState.GetBlendState());
111
112 PipelineState.BlendState = DrawRenderState.GetBlendState();
113 PipelineState.DepthStencilState = DrawRenderState.GetDepthStencilState();
114 PipelineState.DrawShadingRate = PipelineVariableRateShadingEnabled() ? GetShadingRateFromMaterial(MaterialResource.GetShadingRate()) : VRSSR_1x1;
115 PipelineState.bAllowVariableRateShading = ShouldMaterialAllowHardwareVariableRateShading(MaterialResource, MaterialResource.MaterialUsesPixelDepthOffset_RenderThread()) && HardwareVariableRateShadingSupportedByScene();
116
117 // PSO Precache hash only needed when PSO precaching is enabled
118 if (PipelineStateCache::IsPSOPrecachingEnabled())
119 {
120 PipelineState.ComputeStatePrecachePSOHash();
121 }
122
123 check(VertexFactory && VertexFactory->IsInitialized());
124 VertexFactory->GetStreams(FeatureLevel, InputStreamType, SharedMeshDrawCommand.VertexStreams);
125
126#if PSO_PRECACHING_VALIDATE
127 if (PSOCollectorStats::IsFullPrecachingValidationEnabled())
128 {
129 PSOCollectorStats::CheckShaderOnlyStateInCache(PipelineState, MaterialResource, VertexFactory->GetType(), PrimitiveSceneProxy, PSOCollectorIndex);
130 PSOCollectorStats::CheckMinimalPipelineStateInCache(PipelineState, MaterialResource, VertexFactory->GetType(), PrimitiveSceneProxy, PSOCollectorIndex);
131 }
132#endif // PSO_PRECACHING_VALIDATE
133
134 SharedMeshDrawCommand.PrimitiveIdStreamIndex = static_cast<int8>(VertexFactory->GetPrimitiveIdStreamIndex(FeatureLevel, InputStreamType));
135
136 if (SharedMeshDrawCommand.PrimitiveIdStreamIndex != INDEX_NONE)
137 {
138 SharedFlags |= EFVisibleMeshDrawCommandFlags::HasPrimitiveIdStreamIndex;
139 }
140
141 int32 DataOffset = 0;
142 if (PassShaders.VertexShader.IsValid())
143 {
144 FMeshDrawSingleShaderBindings ShaderBindings = SharedMeshDrawCommand.ShaderBindings.GetSingleShaderBindings(SF_Vertex, DataOffset);
145 PassShaders.VertexShader->GetShaderBindings(Scene, FeatureLevel, PrimitiveSceneProxy, MaterialRenderProxy, MaterialResource, ShaderElementData, ShaderBindings);
146 }
147
148 if (PassShaders.PixelShader.IsValid())
149 {
150 FMeshDrawSingleShaderBindings ShaderBindings = SharedMeshDrawCommand.ShaderBindings.GetSingleShaderBindings(SF_Pixel, DataOffset);
151 PassShaders.PixelShader->GetShaderBindings(Scene, FeatureLevel, PrimitiveSceneProxy, MaterialRenderProxy, MaterialResource, ShaderElementData, ShaderBindings);
152 }
153
154 if (PassShaders.GeometryShader.IsValid())
155 {
156 FMeshDrawSingleShaderBindings ShaderBindings = SharedMeshDrawCommand.ShaderBindings.GetSingleShaderBindings(SF_Geometry, DataOffset);
157 PassShaders.GeometryShader->GetShaderBindings(Scene, FeatureLevel, PrimitiveSceneProxy, MaterialRenderProxy, MaterialResource, ShaderElementData, ShaderBindings);
158 }
159
160 SharedMeshDrawCommand.SetDebugData(PrimitiveSceneProxy, &MaterialResource, &MaterialRenderProxy, MeshProcessorShaders, VertexFactory, MeshBatch, PSOCollectorIndex);
161 SharedMeshDrawCommand.SetStatsData(PrimitiveSceneProxy);
162
163 const int32 NumElements = ShouldSkipMeshDrawCommand(MeshBatch, PrimitiveSceneProxy) ? 0 : MeshBatch.Elements.Num();
164
165 for (int32 BatchElementIndex = 0; BatchElementIndex < NumElements; BatchElementIndex++)
166 {
167 if ((1ull << BatchElementIndex) & BatchElementMask)
168 {
169 const FMeshBatchElement& BatchElement = MeshBatch.Elements[BatchElementIndex];
170 FMeshDrawCommand& MeshDrawCommand = DrawListContext->AddCommand(SharedMeshDrawCommand, NumElements);
171
172 EFVisibleMeshDrawCommandFlags Flags = SharedFlags;
173 if (BatchElement.bForceInstanceCulling)
174 {
175 Flags |= EFVisibleMeshDrawCommandFlags::ForceInstanceCulling;
176 }
177 if (BatchElement.bPreserveInstanceOrder)
178 {
179 // TODO: add support for bPreserveInstanceOrder on mobile
180 if (ensureMsgf(FeatureLevel > ERHIFeatureLevel::ES3_1, TEXT("FMeshBatchElement::bPreserveInstanceOrder is currently only supported on non-mobile platforms.")))
181 {
182 Flags |= EFVisibleMeshDrawCommandFlags::PreserveInstanceOrder;
183 }
184 }
185 if (BatchElement.bFetchInstanceCountFromScene)
186 {
187 Flags |= EFVisibleMeshDrawCommandFlags::FetchInstanceCountFromScene;
188 }
189
190 DataOffset = 0;
191 if (PassShaders.VertexShader.IsValid())
192 {
193 FMeshDrawSingleShaderBindings VertexShaderBindings = MeshDrawCommand.ShaderBindings.GetSingleShaderBindings(SF_Vertex, DataOffset);
194 FMeshMaterialShader::GetElementShaderBindings(PassShaders.VertexShader, Scene, ViewIfDynamicMeshCommand, VertexFactory, InputStreamType, FeatureLevel, PrimitiveSceneProxy, MeshBatch, BatchElement, ShaderElementData, VertexShaderBindings, MeshDrawCommand.VertexStreams);
195 }
196
197 if (PassShaders.PixelShader.IsValid())
198 {
199 FMeshDrawSingleShaderBindings PixelShaderBindings = MeshDrawCommand.ShaderBindings.GetSingleShaderBindings(SF_Pixel, DataOffset);
200 FMeshMaterialShader::GetElementShaderBindings(PassShaders.PixelShader, Scene, ViewIfDynamicMeshCommand, VertexFactory, EVertexInputStreamType::Default, FeatureLevel, PrimitiveSceneProxy, MeshBatch, BatchElement, ShaderElementData, PixelShaderBindings, MeshDrawCommand.VertexStreams);
201 }
202
203 if (PassShaders.GeometryShader.IsValid())
204 {
205 FMeshDrawSingleShaderBindings GeometryShaderBindings = MeshDrawCommand.ShaderBindings.GetSingleShaderBindings(SF_Geometry, DataOffset);
206 FMeshMaterialShader::GetElementShaderBindings(PassShaders.GeometryShader, Scene, ViewIfDynamicMeshCommand, VertexFactory, EVertexInputStreamType::Default, FeatureLevel, PrimitiveSceneProxy, MeshBatch, BatchElement, ShaderElementData, GeometryShaderBindings, MeshDrawCommand.VertexStreams);
207 }
208
209 FMeshDrawCommandPrimitiveIdInfo IdInfo = GetDrawCommandPrimitiveId(PrimitiveSceneInfo, BatchElement);
210
211 DrawListContext->FinalizeCommand(MeshBatch, BatchElementIndex, IdInfo, MeshFillMode, MeshCullMode, SortKey, Flags, PipelineState, &MeshProcessorShaders, MeshDrawCommand);
212 }
213 }

247template<typename PassShadersType>
248void FMeshPassProcessor::AddGraphicsPipelineStateInitializer(
249 const FPSOPrecacheVertexFactoryData& VertexFactoryData,
250 const FMaterial& RESTRICT MaterialResource,
251 const FMeshPassProcessorRenderState& RESTRICT DrawRenderState,
252 const FGraphicsPipelineRenderTargetsInfo& RESTRICT RenderTargetsInfo,
253 const PassShadersType& PassShaders,
254 ERasterizerFillMode MeshFillMode,
255 ERasterizerCullMode MeshCullMode,
256 EPrimitiveType PrimitiveType,
257 EMeshPassFeatures MeshPassFeatures,
258 ESubpassHint SubpassHint,
259 uint8 SubpassIndex,
260 bool bRequired,
261 int32 InPSOCollectorIndex,
262 TArray<FPSOPrecacheData>& PSOInitializers)
263{
264 TRACE_CPUPROFILER_EVENT_SCOPE(FMeshPassProcessor::AddGraphicsPipelineStateInitializer);
265
266 FPSOPrecacheData PSOPrecacheData;
267
268 if (IsPSOShaderPreloadingEnabled())
269 {
270 PSOPrecacheData.ShaderPreloadData.Shaders.Append(PassShaders.GetUntypedShaders().GetValidShaders());
271 }
272 else
273 {
274 FGraphicsMinimalPipelineStateInitializer MinimalPipelineStateInitializer;
275 MinimalPipelineStateInitializer.PrimitiveType = PrimitiveType;
276
277 EVertexInputStreamType InputStreamType = EVertexInputStreamType::Default;
278 if ((MeshPassFeatures & EMeshPassFeatures::PositionOnly) != EMeshPassFeatures::Default) InputStreamType = EVertexInputStreamType::PositionOnly;
279 if ((MeshPassFeatures & EMeshPassFeatures::PositionAndNormalOnly) != EMeshPassFeatures::Default) InputStreamType = EVertexInputStreamType::PositionAndNormalOnly;
280
281 FRHIVertexDeclaration* VertexDeclaration = nullptr;
282 if (InputStreamType == EVertexInputStreamType::Default && VertexFactoryData.CustomDefaultVertexDeclaration)
283 {
284 VertexDeclaration = VertexFactoryData.CustomDefaultVertexDeclaration;
285 }
286 else
287 {
288 FVertexDeclarationElementList Elements;
289 VertexFactoryData.VertexFactoryType->GetShaderPSOPrecacheVertexFetchElements(InputStreamType, Elements);
290 VertexDeclaration = PipelineStateCache::GetOrCreateVertexDeclaration(Elements);
291 }
292 check(VertexDeclaration);
293
294 MinimalPipelineStateInitializer.SetupBoundShaderState(VertexDeclaration, PassShaders.GetUntypedShaders());
295 MinimalPipelineStateInitializer.RasterizerState = GetStaticRasterizerState<true>(MeshFillMode, MeshCullMode);
296
297 check(DrawRenderState.GetDepthStencilState());
298 check(DrawRenderState.GetBlendState());
299
300 MinimalPipelineStateInitializer.BlendState = DrawRenderState.GetBlendState();
301 MinimalPipelineStateInitializer.DepthStencilState = DrawRenderState.GetDepthStencilState();
302 MinimalPipelineStateInitializer.DrawShadingRate = GetShadingRateFromMaterial(MaterialResource.GetShadingRate());
303 MinimalPipelineStateInitializer.bAllowVariableRateShading = ShouldMaterialAllowHardwareVariableRateShading(MaterialResource, MaterialResource.MaterialUsesPixelDepthOffset_GameThread()) && HardwareVariableRateShadingSupportedByPlatform(GMaxRHIShaderPlatform);
304
305 // NOTE: AsGraphicsPipelineStateInitializer will create the RHIShaders internally if they are not cached yet
306 // Always pass false for shader creation - this function is only called during PSO precache collection.
307 // bRequired above is for FPSOPrecacheData::bRequired (PSO compilation priority) which is a separate concern.
308 // If any shaders fail to load, skip subsequent work and early out.
309
310 // FORT-1109249 : temporarily forcing required == true to ensure shaders are loaded when PSO Precache tasks are created.
311 // PSO precache misses will occur if shaders are not loaded as precache tasks will be silently dropped.
312 FGraphicsPipelineStateInitializer PipelineStateInitializer = MinimalPipelineStateInitializer.AsGraphicsPipelineStateInitializer(/*false*/);
313
314 if (!MinimalPipelineStateInitializer.BoundShaderState.IsShaderAllLoaded())
315 {
316 return;
317 }
318#if PSO_PRECACHING_VALIDATE
319 if (PSOCollectorStats::IsFullPrecachingValidationEnabled())
320 {
321 MinimalPipelineStateInitializer.StatePrecachePSOHash = PipelineStateInitializer.StatePrecachePSOHash;
322 FGraphicsMinimalPipelineStateInitializer ShadersOnlyInitializer = PSOCollectorStats::GetShadersOnlyInitializer(MinimalPipelineStateInitializer);
323 PSOCollectorStats::GetShadersOnlyPSOPrecacheStatsCollector().AddStateToCache(EPSOPrecacheType::MeshPass, ShadersOnlyInitializer, PSOCollectorStats::GetPSOPrecacheHash, &MaterialResource, InPSOCollectorIndex, VertexFactoryData.VertexFactoryType);
324 FGraphicsMinimalPipelineStateInitializer PatchedMinimalInitializer = PSOCollectorStats::PatchMinimalPipelineStateToCheck(MinimalPipelineStateInitializer);
325 PSOCollectorStats::GetMinimalPSOPrecacheStatsCollector().AddStateToCache(EPSOPrecacheType::MeshPass, PatchedMinimalInitializer, PSOCollectorStats::GetPSOPrecacheHash, &MaterialResource, InPSOCollectorIndex, VertexFactoryData.VertexFactoryType);
326 }
327#endif // PSO_PRECACHING_VALIDATE
328
329 ApplyTargetsInfo(PipelineStateInitializer, RenderTargetsInfo);
330 PipelineStateInitializer.SubpassHint = SubpassHint;
331 PipelineStateInitializer.SubpassIndex = SubpassIndex;
332 PSOPrecacheData.GraphicsPSOInitializer = PipelineStateInitializer;
333 }
334
335 PSOPrecacheData.bRequired = bRequired;
336 PSOPrecacheData.Type = FPSOPrecacheData::EType::Graphics;
337#if PSO_PRECACHING_VALIDATE
338 PSOPrecacheData.ResourceName = MaterialResource.GetAssetName();
339 PSOPrecacheData.PSOCollectorIndex = InPSOCollectorIndex;
340 PSOPrecacheData.VertexFactoryType = VertexFactoryData.VertexFactoryType;
341 if (PSOCollectorStats::IsFullPrecachingValidationEnabled())
342 {
343 PSOPrecacheData.bDefaultMaterial = MaterialResource.IsDefaultMaterial();
344 ConditionalBreakOnPSOPrecacheShader(PSOPrecacheData.GraphicsPSOInitializer);
345 }
346#endif // PSO_PRECACHING_VALIDATE
347 PSOInitializers.Add(MoveTemp(PSOPrecacheData));
348}
