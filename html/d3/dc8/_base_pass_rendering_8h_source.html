362 const bool bSLWForceVelocity = SLWForceVelocityCVar && SLWForceVelocityCVar->GetValueOnAnyThread() != 0;
363 if (bSLWForceVelocity && Parameters.MaterialParameters.ShadingModels.HasShadingModel(MSM_SingleLayerWater))
364 {
365 OutEnvironment.SetDefine(TEXT("GBUFFER_LAYOUT"), GBL_ForceVelocity);
366 }

386 {
387 static const auto SupportAllShaderPermutations = IConsoleManager::Get().FindTConsoleVariableDataInt(TEXT("r.SupportAllShaderPermutations"));
388 const bool bForceAllPermutations = SupportAllShaderPermutations && SupportAllShaderPermutations->GetValueOnAnyThread() != 0;
389 bool bShouldCache = Super::ShouldCompilePermutation(Parameters) || bForceAllPermutations;
390 return bShouldCache && (IsFeatureLevelSupported(Parameters.Platform, ERHIFeatureLevel::SM5));

393 static void ModifyCompilationEnvironment(const FMeshMaterialShaderPermutationParameters& Parameters, FShaderCompilerEnvironment& OutEnvironment)
394 {
395 Super::ModifyCompilationEnvironment(Parameters, OutEnvironment);
396
397 // @todo MetalMobile: Remove this hack and implement proper atmospheric-fog solution for Metal Mobile...
398 OutEnvironment.SetDefine(TEXT("BASEPASS_SKYATMOSPHERE_AERIALPERSPECTIVE"), !(IsMetalMobileSM5Platform(Parameters.Platform)) ? 1 : 0);
399
400 // GBUFFER_LAYOUT is used to derive GBUFFER_HAS_VELOCITY, which influences whether VelocityPrevScreenPosition is computed by the vertex shader.
401 // When SLW velocity outputs to the depth prepass, the base pass uses GBL_Default (no velocity RT). When velocity outputs to the base pass,
402 // GBL_ForceVelocity is used.
403 static const auto SLWVelocityOutputPass = IConsoleManager::Get().FindTConsoleVariableDataInt(TEXT("r.Water.SingleLayer.VelocityOutputPass"));
404 const int32 VelocityMode = SLWVelocityOutputPass ? SLWVelocityOutputPass->GetValueOnAnyThread() : 1;
405 if (Parameters.MaterialParameters.ShadingModels.HasShadingModel(MSM_SingleLayerWater))
406 {
407 // Must match GetSingleLayerWaterGBufferLayout: mode 0 = GBL_Default, mode 1 with prepass = GBL_Default,
408 // mode 1 without prepass or forward renderer, or mode 2 = GBL_ForceVelocity
409 // Forward+MSAA requires multisampled depth target while velocity RT stays as single sampled. Need to fallback to GBL_ForceVelocity.
410 const bool bDepthPrepass = IsSingleLayerWaterDepthPrepassEnabled(Parameters.Platform, GetMaxSupportedFeatureLevel(Parameters.Platform));
411 const bool bForwardMSAA = IsForwardShadingEnabled(Parameters.Platform);
412 const bool bUseForceVelocity = (VelocityMode == 2) || (VelocityMode == 1 && (!bDepthPrepass || bForwardMSAA));
413 OutEnvironment.SetDefine(TEXT("GBUFFER_LAYOUT"), bUseForceVelocity ? GBL_ForceVelocity : GBL_Default);
414 }
415 }
