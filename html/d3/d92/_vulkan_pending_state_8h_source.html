36 void SetComputePipeline(FVulkanComputePipeline* InComputePipeline)
37 {
38 if (InComputePipeline != CurrentPipeline)
39 {
40 CurrentPipeline = InComputePipeline;
41 FVulkanComputePipelineDescriptorState** Found = PipelineStates.Find(InComputePipeline);
42 if (Found)
43 {
44 CurrentState = *Found;
45 check(CurrentState->ComputePipeline == InComputePipeline);
46 }
47 else
48 {
49 CurrentState = new FVulkanComputePipelineDescriptorState(Device, InComputePipeline);
50 PipelineStates.Add(CurrentPipeline, CurrentState);
51 }
52
53 CurrentState->Reset();
54 }
55 }
56
57 void PrepareForDispatch(FVulkanCommandListContext& Context);

140 void SetViewport(float MinX, float MinY, float MinZ, float MaxX, float MaxY, float MaxZ)
141 {
142 Viewports.SetNumZeroed(1);
143
144 Viewports[0].x = MinX;
145 Viewports[0].y = MinY;
146 Viewports[0].width = MaxX - MinX;
147 Viewports[0].height = MaxY - MinY;
148 Viewports[0].minDepth = MinZ;
149 if (MinZ == MaxZ)
150 {
151 // Engine pases in some cases MaxZ as 0.0
152 Viewports[0].maxDepth = MinZ + 1.0f;
153 }
154 else
155 {
156 Viewports[0].maxDepth = MaxZ;
157 }
158
159 SetScissorRect((uint32)MinX, (uint32)MinY, (uint32)(MaxX - MinX), (uint32)(MaxY - MinY));
160 bScissorEnable = false;
161 }

163 void SetMultiViewport(const TArrayView<VkViewport>& InViewports)
164 {
165 Viewports = InViewports;
166
167 // Set the scissor rects appropriately.
168 Scissors.SetNumZeroed(Viewports.Num());
169 for (int32 Idx = 0; Idx < Scissors.Num(); ++Idx)
170 {
171 Scissors[Idx].offset.x = Viewports[Idx].x;
172 Scissors[Idx].offset.y = Viewports[Idx].y;
173 Scissors[Idx].extent.width = Viewports[Idx].width;
174 Scissors[Idx].extent.height = Viewports[Idx].height;
175 }
176 bScissorEnable = true;
177 }

179 void SetScissor(bool bInEnable, uint32 MinX, uint32 MinY, uint32 MaxX, uint32 MaxY)
180 {
181 if (bInEnable)
182 {
183 SetScissorRect(MinX, MinY, MaxX - MinX, MaxY - MinY);
184 }
185 else
186 {
187 checkf(Viewports.Num() > 0, TEXT("At least one Viewport is expected to be configured."));
188 SetScissorRect(Viewports[0].x, Viewports[0].y, Viewports[0].width, Viewports[0].height);
189 }
190
191 bScissorEnable = bInEnable;
192 }
