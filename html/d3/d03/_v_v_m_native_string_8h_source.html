30// Wraps a copy-on-write reference to a Utf8String to give it Verse semantics,
31// and since we copy objects around and don't modify them often in the VM gives
32// us a performance uplift in heavy string workloads.
33class FNativeString
34{
35 struct FCopyOnWriteContents final
36 {
37 FCopyOnWriteContents() = default;
38 FCopyOnWriteContents(const FUtf8String& InString)
39 : String(InString) {}
40 FCopyOnWriteContents(FUtf8String&& InString)
41 : String(MoveTemp(InString)) {}
42
43 FUtf8String String;
44 mutable int32 CachedHash = 0;
45
46 inline int32 Hash() const
47 {
48 if (0 == CachedHash)
49 {
50 // Do not forward to GetTypeHash(const FUtf8String&), which is case-insensitive.
51 CachedHash = FCrc::StrCrc32Len(*String, String.Len());
52
53 // 0 is technically a valid hash result, but since we are using it to mean the
54 // hash was not cached, if we get a hash result of 0 we modify the saved value.
55 if (UNLIKELY(0 == CachedHash))
56 {
57 CachedHash = -1;
58 }
59 }
60
61 return CachedHash;
62 }
63
64 UE_FORCEINLINE_HINT void ResetCachedHash()
65 {
66 CachedHash = 0;
67 }
68 };
69
70 struct FCopyOnWrite final
71 {
72 UE_FORCEINLINE_HINT FCopyOnWrite() = default;
73 UE_FORCEINLINE_HINT FCopyOnWrite(const FUtf8String& InString)
74 : Payload(MakeShared<FCopyOnWriteContents, ESPMode::ThreadSafe>(InString)) {}
75 UE_FORCEINLINE_HINT FCopyOnWrite(FUtf8String&& InString)
76 : Payload(MakeShared<FCopyOnWriteContents, ESPMode::ThreadSafe>(MoveTemp(InString))) {}
77
78 const FUtf8String& Read() const
79 {
80 const_cast<FCopyOnWrite*>(this)->SetIfNull();
81 return Payload->String;
82 }
83
84 FUtf8String& Write()
85 {
86 if (!SetIfNull())
87 {
88 if (!Payload.IsUnique())
89 {
90 Payload = MakeShared<FCopyOnWriteContents, ESPMode::ThreadSafe>(Payload->String);
91 }
92
93 // Since we are writing to the string we need to forget any previous cached hash we stored.
94 Payload->ResetCachedHash();
95 }
96
97 return Payload->String;
98 }
99
100 int32 Hash() const
101 {
102 const_cast<FCopyOnWrite*>(this)->SetIfNull();
103 return Payload->Hash();
104 }
105
106 private:
107 TSharedPtr<FCopyOnWriteContents, ESPMode::ThreadSafe> Payload;
108
109 static_assert(TIsZeroConstructType<decltype(Payload)>::Value);
110
111 inline bool SetIfNull()
112 {
113 if (UNLIKELY(!Payload.IsValid()))
114 {
115 Payload = MakeShared<FCopyOnWriteContents, ESPMode::ThreadSafe>();
116 return true;
117 }
118
119 return false;
120 }
121 };
