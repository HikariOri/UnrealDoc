131 FRealCurve* FindCurve(FName RowName, const FString& ContextString, bool bWarnIfNotFound = true) const
132 {
133 if (RowName.IsNone())
134 {
135 UE_CLOGF(bWarnIfNotFound, LogCurveTable, Warning, "UCurveTable::FindCurve : NAME_None is invalid row name for CurveTable '%ls' (%ls).", *GetPathName(), *ContextString);
136 return nullptr;
137 }
138
139 FRealCurve* const* FoundCurve = RowMap.Find(RowName);
140
141 if (FoundCurve == nullptr)
142 {
143 UE_CLOGF(bWarnIfNotFound, LogCurveTable, Warning, "UCurveTable::FindCurve : Row '%ls' not found in CurveTable '%ls' (%ls).", *RowName.ToString(), *GetPathName(), *ContextString);
144 return nullptr;
145 }
146
147 return *FoundCurve;
148 }
149
150 FRichCurve* FindRichCurve(FName RowName, const FString& ContextString, bool bWarnIfNotFound = true) const
151 {
152 if (CurveTableMode == ECurveTableMode::SimpleCurves)
153 {
154 UE_LOGF(LogCurveTable, Error, "UCurveTable::FindCurve : Using FindRichCurve on CurveTable '%ls' (%ls) that is storing simple curves.", *GetPathName(), *ContextString);
155 return nullptr;
156 }
157
158 return (FRichCurve*)FindCurve(RowName, ContextString, bWarnIfNotFound);
159 }
160
161 FSimpleCurve* FindSimpleCurve(FName RowName, const FString& ContextString, bool bWarnIfNotFound = true) const
162 {
163 if (CurveTableMode == ECurveTableMode::RichCurves)

165 UE_LOGF(LogCurveTable, Error, "UCurveTable::FindCurve : Using FindSimpleCurve on CurveTable '%ls' (%ls) that is storing rich curves.", *GetPathName(), *ContextString);
