421 // Conversion to other type.
422 template<typename FArg UE_REQUIRES(!std::is_same_v<T, FArg>)>
423 explicit TMatrix(const TMatrix<FArg>& From)
424 {
425#if PLATFORM_ENABLE_VECTORINTRINSICS
426 const FArg* RESTRICT Src = &(From.M[0][0]);
427 T* RESTRICT Dest = &(M[0][0]);
428
429 // Load src
430 TVectorRegisterType<FArg> SrcRow0 = VectorLoad(&Src[0]);
431 TVectorRegisterType<FArg> SrcRow1 = VectorLoad(&Src[4]);
432 TVectorRegisterType<FArg> SrcRow2 = VectorLoad(&Src[8]);
433 TVectorRegisterType<FArg> SrcRow3 = VectorLoad(&Src[12]);
434
435 // Coerce and store
436 if constexpr (std::is_same_v<T, float>)
437 {
438 VectorStore(MakeVectorRegisterFloatFromDouble(SrcRow0), &Dest[0]);
439 VectorStore(MakeVectorRegisterFloatFromDouble(SrcRow1), &Dest[4]);
440 VectorStore(MakeVectorRegisterFloatFromDouble(SrcRow2), &Dest[8]);
441 VectorStore(MakeVectorRegisterFloatFromDouble(SrcRow3), &Dest[12]);
442 }
443 else
444 {
445 VectorStore(TVectorRegisterType<T>(SrcRow0), &Dest[0]);
446 VectorStore(TVectorRegisterType<T>(SrcRow1), &Dest[4]);
447 VectorStore(TVectorRegisterType<T>(SrcRow2), &Dest[8]);
448 VectorStore(TVectorRegisterType<T>(SrcRow3), &Dest[12]);
449 }
450#else
451 M[0][0] = (T)From.M[0][0]; M[0][1] = (T)From.M[0][1]; M[0][2] = (T)From.M[0][2]; M[0][3] = (T)From.M[0][3];
452 M[1][0] = (T)From.M[1][0]; M[1][1] = (T)From.M[1][1]; M[1][2] = (T)From.M[1][2]; M[1][3] = (T)From.M[1][3];
453 M[2][0] = (T)From.M[2][0]; M[2][1] = (T)From.M[2][1]; M[2][2] = (T)From.M[2][2]; M[2][3] = (T)From.M[2][3];
454 M[3][0] = (T)From.M[3][0]; M[3][1] = (T)From.M[3][1]; M[3][2] = (T)From.M[3][2]; M[3][3] = (T)From.M[3][3];
455#endif
456
457 DiagnosticCheckNaN();
458 }

615// very high quality 4x4 matrix inverse
616// @todo: this is redundant with FMatrix44d::Inverse and should be removed ; seems to be unused
617template<typename FArg UE_REQUIRES(std::is_floating_point_v<FArg>)>
618static inline bool Inverse4x4( double* dst, const FArg* src )
619{
620 const double s0 = (double)(src[ 0]); const double s1 = (double)(src[ 1]); const double s2 = (double)(src[ 2]); const double s3 = (double)(src[ 3]);
621 const double s4 = (double)(src[ 4]); const double s5 = (double)(src[ 5]); const double s6 = (double)(src[ 6]); const double s7 = (double)(src[ 7]);
622 const double s8 = (double)(src[ 8]); const double s9 = (double)(src[ 9]); const double s10 = (double)(src[10]); const double s11 = (double)(src[11]);
623 const double s12 = (double)(src[12]); const double s13 = (double)(src[13]); const double s14 = (double)(src[14]); const double s15 = (double)(src[15]);
624
625 double inv[16];
626 inv[0] = s5 * s10 * s15 - s5 * s11 * s14 - s9 * s6 * s15 + s9 * s7 * s14 + s13 * s6 * s11 - s13 * s7 * s10;
627 inv[1] = -s1 * s10 * s15 + s1 * s11 * s14 + s9 * s2 * s15 - s9 * s3 * s14 - s13 * s2 * s11 + s13 * s3 * s10;
628 inv[2] = s1 * s6 * s15 - s1 * s7 * s14 - s5 * s2 * s15 + s5 * s3 * s14 + s13 * s2 * s7 - s13 * s3 * s6;
629 inv[3] = -s1 * s6 * s11 + s1 * s7 * s10 + s5 * s2 * s11 - s5 * s3 * s10 - s9 * s2 * s7 + s9 * s3 * s6;
630 inv[4] = -s4 * s10 * s15 + s4 * s11 * s14 + s8 * s6 * s15 - s8 * s7 * s14 - s12 * s6 * s11 + s12 * s7 * s10;
631 inv[5] = s0 * s10 * s15 - s0 * s11 * s14 - s8 * s2 * s15 + s8 * s3 * s14 + s12 * s2 * s11 - s12 * s3 * s10;
632 inv[6] = -s0 * s6 * s15 + s0 * s7 * s14 + s4 * s2 * s15 - s4 * s3 * s14 - s12 * s2 * s7 + s12 * s3 * s6;
633 inv[7] = s0 * s6 * s11 - s0 * s7 * s10 - s4 * s2 * s11 + s4 * s3 * s10 + s8 * s2 * s7 - s8 * s3 * s6;
634 inv[8] = s4 * s9 * s15 - s4 * s11 * s13 - s8 * s5 * s15 + s8 * s7 * s13 + s12 * s5 * s11 - s12 * s7 * s9;
635 inv[9] = -s0 * s9 * s15 + s0 * s11 * s13 + s8 * s1 * s15 - s8 * s3 * s13 - s12 * s1 * s11 + s12 * s3 * s9;
636 inv[10] = s0 * s5 * s15 - s0 * s7 * s13 - s4 * s1 * s15 + s4 * s3 * s13 + s12 * s1 * s7 - s12 * s3 * s5;
637 inv[11] = -s0 * s5 * s11 + s0 * s7 * s9 + s4 * s1 * s11 - s4 * s3 * s9 - s8 * s1 * s7 + s8 * s3 * s5;
638 inv[12] = -s4 * s9 * s14 + s4 * s10 * s13 + s8 * s5 * s14 - s8 * s6 * s13 - s12 * s5 * s10 + s12 * s6 * s9;
639 inv[13] = s0 * s9 * s14 - s0 * s10 * s13 - s8 * s1 * s14 + s8 * s2 * s13 + s12 * s1 * s10 - s12 * s2 * s9;
640 inv[14] = -s0 * s5 * s14 + s0 * s6 * s13 + s4 * s1 * s14 - s4 * s2 * s13 - s12 * s1 * s6 + s12 * s2 * s5;
641 inv[15] = s0 * s5 * s10 - s0 * s6 * s9 - s4 * s1 * s10 + s4 * s2 * s9 + s8 * s1 * s6 - s8 * s2 * s5;
642
643 double det = s0 * inv[0] + s1 * inv[4] + s2 * inv[8] + s3 * inv[12];
644 if( det == 0.0 || !FMath::IsFinite(det) )
645 {
646 memcpy(dst,&FMatrix44d::Identity,sizeof(FMatrix44d));
647 return false;
648 }
649
650 det = 1.0 / det;
651 for( int i = 0; i < 16; i++ )
652 {
653 dst[i] = inv[i] * det;
654 }
655 return true;
656}
