55 static TArray<float> ConvertPairToFlatKnots(const TArray<FKnot>& PairKnots)
56 {
57 TArray<float> FlatKnots;
58 for (const FKnot& Knot : PairKnots)
59 {
60 for (uint32 i = 0; i < Knot.Multiplicity; ++i)
61 {
62 FlatKnots.Add(Knot.Value);
63 }
64 }
65 return FlatKnots;
66 }
67
68 static bool IsValidKnotVector(const TArray<FKnot>& InKnots, int32 Degree)
69 {
70 TArray<float> FlatKnots = ConvertPairToFlatKnots(InKnots);
71 // Need at least 2*(degree+1) knots
72 if (FlatKnots.Num() < 2 * (Degree + 1))
73 {
74 return false;
75 }
76
77 // Check monotonicity
78 for (int32 i = 1; i < FlatKnots.Num(); ++i)
79 {
80 if (FlatKnots[i] < FlatKnots[i - 1])
81 {
82 return false;
83 }
84 }
85
86 return true;
87 }
88
89 bool operator==(const FKnot& Other) const
90 {
91 return FMath::IsNearlyEqual(Value, Other.Value) && Multiplicity == Other.Multiplicity;
92 }
93
94 friend FArchive& operator<<(FArchive& Ar, FKnot& Knot)
95 {
96 Ar << Knot.Value;
97 Ar << Knot.Multiplicity;
98 return Ar;
99 }

353 // Returns parameter value for a control point index using Greville Abscissae
354 virtual float GetParameter(int32 Index) const
355 {
356 UpdateFlatKnotsCache();
357 if (Index < 0 || Index + Degree >= FlatKnots.Num() - 1)
358 return 0.0f;
359
360 float Sum = 0.0f;
361 for (int j = 1; j <= Degree; ++j)
362 {
363 Sum += FlatKnots[Index + j];
364 }
365 return Sum / static_cast<float>(Degree);
366 }
367
368 // Find the control point index and local parameter for global parameter
369 virtual int32 FindIndexForParameter(float Parameter, float& OutLocalParam) const
370 {
371 UpdateFlatKnotsCache();
372 const TArray<float>& Knots = FlatKnots;
373
374 if (Knots.IsEmpty())
375 return INDEX_NONE;
376
377 // Clamp parameter to valid range
378 const float FirstValidKnot = Knots[Degree];
379 const float LastValidKnot = Knots[Knots.Num() - Degree - 1];
380
381 // Check for exact match with last knot (for insertion at the end)
382 if (FMath::IsNearlyEqual(Parameter, LastValidKnot))
383 {
384 OutLocalParam = 0.0f; // No interpolation needed for exact match
385 return NumKeys(); // Return index for appending
386 }
387 // Handle parameter near the end
388 else if (Parameter >= LastValidKnot - UE_SMALL_NUMBER)
389 {
390 OutLocalParam = 1.0f;
391 return NumKeys() - 2; // Last segment index
392 }
393
394 if (Parameter <= FirstValidKnot + UE_SMALL_NUMBER)
395 {
396 OutLocalParam = 0.0f;
397 return 0;
398 }
399
400
401 // Find the appropriate span
402 int32 Span = FindKnotSpan(Parameter);
403 if (Span < Degree || Span >= Knots.Num() - 1)
404 return INDEX_NONE;
405
406 int32 Index = Span - Degree;
407
408 // Calculate local parameter within span, with protection against near-zero denominators
409 float SpanStart = Knots[Span];
410 float SpanEnd = Knots[Span + 1];
411 float SpanLength = SpanEnd - SpanStart;
412
413 if (SpanLength < UE_SMALL_NUMBER)
414 {
415 // For nearly identical knots, use 0 or 1 based on which end is closer
416 OutLocalParam = (Parameter - SpanStart < SpanEnd - Parameter)
417 ? 0.0f
418 : 1.0f;
419 }
420 else
421 {
422 OutLocalParam = (Parameter - SpanStart) / SpanLength;
423 }
424
425 return Index;
426 }

455 bool ValidateKnotVector(const TArray<FKnot>& InKnots) const
456 {
457 const int32 RequiredKnots = GetExpectedNumKnots();
458 int32 TotalExpandedKnots = 0;
459
460 // Validate knot multiplicities and calculate total expanded knots
461 for (const FKnot& Knot : InKnots)
462 {
463 if (!ensureAlwaysMsgf(Knot.Multiplicity > 0,
464 TEXT("Invalid knot with zero multiplicity.")))
465 {
466 return false;
467 }
468
469 TotalExpandedKnots += Knot.Multiplicity;
470 }
471
472 if (!ensureAlwaysMsgf(TotalExpandedKnots >= RequiredKnots,
473 TEXT("Invalid knot vector size. Need %d knots, got %d."),
474 RequiredKnots,
475 TotalExpandedKnots))
476 {
477 return false;
478 }
479
480 // Verify knot values are non-decreasing (strict comparison, no epsilon)
481 for (int32 i = 1; i < InKnots.Num(); ++i)
482 {
483 if (!ensureAlwaysMsgf(InKnots[i].Value >= InKnots[i - 1].Value,
484 TEXT("Invalid knot vector: knot values must be non-decreasing.")))
485 {
486 return false;
487 }
488 }
489
490 return true;
491 }

525 virtual void Reparameterize(EParameterizationPolicy ParameterizationPolicy)
526 {
527 const TArray<ValueType>& Points = Values;
528 int32 NumValues = Points.Num();
529 if (NumValues < Degree + 1)
530 {
531 UE_LOGF(LogSpline, Warning, "Not enough points for B-spline: %d (need >= %d)",
532 NumValues, Degree + 1);
533 return;
534 }
535
536 const int32 KnotCount = NumValues + Degree + 1;
537 PairKnots.Reset();
538
539 switch (ParameterizationPolicy)
540 {
541 case EParameterizationPolicy::Uniform:
542 GenerateUniformKnots(KnotCount);
543 break;
544 case EParameterizationPolicy::ChordLength:
545 GenerateChordLengthKnots(KnotCount);
546 break;
547 case EParameterizationPolicy::Centripetal:
548 GenerateCentripetalKnots(KnotCount);
549 break;
550 }
551
552 // Mark the flat knots cache as dirty
553 MarkFlatKnotsCacheDirty();
554
555 UE_LOGF(LogSpline, Verbose, "Updated knot vector - Points: %d, Degree: %d, Knots: %d",
556 Points.Num(), Degree, PairKnots.Num());
557 }

648 int32 InsertKnot(FKnot InKnot)
649 {
650 // Binary search the knot vector:
651
652 int32 Low = 0; // This will be our insert index
653 int32 High = PairKnots.Num() - 1;
654
655 while (Low <= High)
656 {
657 const int32 Mid = Low + (High - Low) / 2;
658
659 if (PairKnots[Mid].Value == InKnot.Value)
660 {
661 UE_LOGF(LogSpline, Warning, "Knot insertion failed, knot already exists!");
662 return INDEX_NONE;
663 }
664 else if (PairKnots[Mid].Value < InKnot.Value)
665 {
666 Low = Mid + 1;
667 }
668 else
669 {
670 High = Mid - 1;
671 }
672 }
673
674 MarkFlatKnotsCacheDirty();
675 return PairKnots.Insert(InKnot, Low);
676 }

682 FValidKnotSearchParams(float InDesiredParameter)

692 struct FKnotSearchCache
693 {
694 explicit FKnotSearchCache(const TArray<FKnot>& PairKnots)
695 {
696 // Build a set of existing knot values (normalize zeros to avoid -0/+0 duplicates)
697 KnotValuesSet.Reserve(PairKnots.Num());
698 for (const FKnot& Knot : PairKnots)
699 {
700 KnotValuesSet.Add(Param::NormalizeKey(Knot.Value));
701 }
702 }
703
704 TSet<float> KnotValuesSet;
705 };
706
712 float GetNearestAvailableKnotValue(const FValidKnotSearchParams& InSearchParams, const FKnotSearchCache& SearchCache) const
713 {
714 const float DesiredParameter = InSearchParams.DesiredParameter;
715
716 if (!ensureAlwaysMsgf(!std::isnan(DesiredParameter), TEXT("DesiredParameter is NaN!")) ||
717 !ensureAlwaysMsgf(InSearchParams.bSearchLeft || InSearchParams.bSearchRight, TEXT("Invalid search params!")) ||
718 PairKnots.Num() == 0)
719 {
720 return DesiredParameter;
721 }
722
723 // Return the caller's exact value; NormalizeKey is only for set membership (folds -0.0f/+0.0f).
724 if (!SearchCache.KnotValuesSet.Contains(Param::NormalizeKey(DesiredParameter)))
725 {
726 return DesiredParameter;
727 }
728
729 // Walk outward alternating sides
730 float Left = DesiredParameter;
731 float Right = DesiredParameter;
732
733 // Copy so we can disable directions mid-search
734 bool bSearchLeft = InSearchParams.bSearchLeft;
735 bool bSearchRight = InSearchParams.bSearchRight;
736
737 // Tiny dead-man budget to catch regressions; normal exits happen via guard rails.
738 int32 StepBudget = 4096;
739 for (;;)
740 {
741 bool AnyProgress = false;
742
743 if (bSearchLeft)
744 {
745 const float NewLeft = Param::PrevDistinct(Left);
746 if (NewLeft == Left)
747 {
748 bSearchLeft = false;
749 } // Guard #1: no progress
750 else
751 {
752 Left = NewLeft;
753 AnyProgress = true;
754 if (!SearchCache.KnotValuesSet.Contains(Param::NormalizeKey(Left)))
755 {
756 return Left;
757 }
758 if (Left <= -std::numeric_limits<float>::max()) // Guard #2: saturated
759 {
760 bSearchLeft = false;
761 }
762 }
763 }
764
765 if (bSearchRight)
766 {
767 const float NewRight = Param::NextDistinct(Right);
768 if (NewRight == Right)
769 {
770 bSearchRight = false;
771 } // Guard #1: no progress
772 else
773 {
774 Right = NewRight;
775 AnyProgress = true;
776 if (!SearchCache.KnotValuesSet.Contains(Param::NormalizeKey(Right)))
777 {
778 return Right;
779 }
780 if (Right >= +std::numeric_limits<float>::max()) // Guard #2: saturated
781 {
782 bSearchRight = false;
783 }
784 }
785 }
786
787 if (!bSearchLeft && !bSearchRight) break; // Guard #3: nowhere left to search
788 if (!AnyProgress) break; // Paranoid: should be covered by #1
789
790 if (--StepBudget <= 0)
791 {
792 // Dead-man: catch unexpected spins
793 ensureMsgf(false, TEXT("Step budget exhausted in GetNearestAvailableKnotValue"));
794 break;
795 }
796 }
797
798 // Fallback: never return the conflicting original. Pick a deterministic, distinct nudge.
799 {
800 // Canonicalize -0/+0, then use its *bits* to choose a side deterministically.
801 const float Canon = Param::NormalizeKey(DesiredParameter);
802 const uint32 Bits = Internal::FloatToBits(Canon);
803
804 const bool bPreferRight =
805 (InSearchParams.bSearchRight && (!InSearchParams.bSearchLeft || ((Bits & 1u) != 0)));
806
807 // Step from the *original* DesiredParameter (not Canon) so we don’t alter the value unless needed.
808 float Fallback = Param::Step(
809 DesiredParameter, bPreferRight
810 ? Param::EDir::Right
811 : Param::EDir::Left);
812
813 float Probe = Param::NormalizeKey(Fallback);
814 if (SearchCache.KnotValuesSet.Contains(Probe))
815 {
816 // Try the opposite direction once
817 Fallback = Param::Step(
818 DesiredParameter, bPreferRight
819 ? Param::EDir::Left
820 : Param::EDir::Right);
821 Probe = Param::NormalizeKey(Fallback);
822 }
823
824 ensureMsgf(!SearchCache.KnotValuesSet.Contains(Probe), TEXT("Fallback still conflicts; check knot packing logic."));
825 return Fallback;
826 }
827 }

842 void GenerateUniformKnots(int32 KnotCount)
843 {
844 PairKnots.Reset();
845
846 int32 NumPoints = NumKeys();
847 if (this->IsClosedLoop())
848 {
849 // For a closed B-spline of degree d with n points:
850 // - Need additional d points wrapping from start
851 // - Need n+2d+1 knots total
852 // - Knot spacing should be uniform to maintain periodicity
853 const int32 n = NumPoints;
854 const int32 d = Degree;
855 const int32 NumKnots = n + 2 * d + 1;
856
857
858 // For periodic B-splines, use uniform knot spacing
859 for (int32 i = 0; i < NumKnots; ++i)
860 {
861 // Map to [0,n] range for n segments
862 PairKnots.Add(FKnot(static_cast<float>(i), 1));
863 }
864 }
865 else
866 {
867 for (int32 i = 0; i < KnotCount; ++i)
868 {
869 float Value;
870 if (bClampEnds)
871 {
872 if (i <= Degree)
873 {
874 // Multiple knots at start for endpoint interpolation
875 Value = 0.0f;
876 }
877 else if (i >= KnotCount - Degree - 1)
878 {
879 // Multiple knots at end for endpoint interpolation
880 Value = static_cast<float>(KnotCount - 2 * Degree);
881 }
882 else
883 {
884 // Uniform spacing for internal knots
885 Value = static_cast<float>(i - Degree);
886 }
887 }
888 else
889 {
890 // Simple uniform spacing when not clamped
891 Value = static_cast<float>(i);
892 }
893 PairKnots.Add(FKnot(Value, 1));
894 }
895 }
896 MarkFlatKnotsCacheDirty();
897 }

903 void GenerateChordLengthKnots(int32 KnotCount)
904 {
905 const TArray<ValueType>& Points = Values;
906 if (Points.Num() < 2)
907 {
908 GenerateUniformKnots(KnotCount);
909 return;
910 }
911
912 const int32 n = Points.Num();
913 const int32 d = Degree;
914 const int32 NumKnots = this->IsClosedLoop()
915 ? (n + 2 * d + 1)
916 : KnotCount;
917 PairKnots.Reset();
918
919 // Calculate chord lengths between consecutive points
920 TArray<float> Chords;
921 Chords.Reserve(Points.Num() - 1);
922 float TotalLength = 0.0f;
923
924 for (int32 i = 1; i < Points.Num(); ++i)
925 {
926 const float Length = static_cast<float>(Math::Distance(Points[i], Points[i - 1]));
927 Chords.Add(Length);
928 TotalLength += Length;
929 }
930
931 // For closed loop, add the closing segment length
932 if (this->IsClosedLoop() && Points.Num() > 0)
933 {
934 const float ClosureLength = static_cast<float>(Math::Distance(Points[0], Points.Last()));
935 Chords.Add(ClosureLength);
936 TotalLength += ClosureLength;
937 }
938
939 if (this->IsClosedLoop())
940 {
941 // For closed loop, generate periodic knot vector based on chord lengths
942 for (int32 i = 0; i < NumKnots; ++i)
943 {
944 PairKnots.Add(FKnot(static_cast<float>(i) * TotalLength / static_cast<float>(n + d), 1));
945 }
946 }
947 else if (bClampEnds)
948 {
949 // For clamped ends, we need:
950 // - First (Degree+1) knots equal to 0.0
951 // - Last (Degree+1) knots equal to 1.0
952 // - Middle knots based on chord lengths
953
954 // First set starting knots to 0
955 PairKnots.Add(FKnot(0.0f, Degree + 1));
956
957 // Calculate internal knots based on accumulated chord lengths
958 // We have (KnotCount - 2*(Degree+1)) internal knots to set
959 const int32 NumInternalKnots = KnotCount - 2 * (Degree + 1);
960 if (NumInternalKnots > 0)
961 {
962 // Compute normalized Knots at each control point
963 TArray<float> Params;
964 Params.SetNum(Points.Num());
965 Params[0] = 0.0f;
966
967 float AccumulatedLength = 0.0f;
968 for (int32 i = 1; i < Points.Num(); ++i)
969 {
970 AccumulatedLength += Chords[i - 1];
971 Params[i] = (TotalLength > UE_SMALL_NUMBER)
972 ? (AccumulatedLength / TotalLength)
973 : (static_cast<float>(i) / static_cast<float>(Points.Num() - 1));
974 }
975
976 // Map these Knots to internal knots
977 for (int32 i = 0; i < NumInternalKnots; ++i)
978 {
979 // Map i to appropriate parameter range
980 const float t = static_cast<float>(i + 1) / static_cast<float>(NumInternalKnots + 1);
981 // Map t to control point indices
982 const int32 LowIndex = FMath::FloorToInt(t * static_cast<float>(Points.Num() - 1));
983 const int32 HighIndex = FMath::Min(LowIndex + 1, Points.Num() - 1);
984 const float Alpha = t * static_cast<float>(Points.Num() - 1 - LowIndex);
985
986 // Interpolate Knots
987 const float KnotValue = FMath::Lerp(Params[LowIndex], Params[HighIndex], Alpha);
988 PairKnots.Add(FKnot(KnotValue, 1));
989 }
990 }
991 // Set the last (Degree+1) knots to 1.0
992 PairKnots.Add(FKnot(1.0f, Degree + 1));
993 }
994 else
995 {
996 // Unclamped knots - simple uniform distribution based on chord lengths
997 for (int32 i = 0; i < KnotCount; ++i)
998 {
999 const float Param = static_cast<float>(i) / static_cast<float>(KnotCount - 1);
1000 PairKnots.Add(FKnot(Param * TotalLength, 1));
1001 }
1002 }
1003 MarkFlatKnotsCacheDirty();
1004 }

1011 void GenerateCentripetalKnots(int32 KnotCount)
1012 {
1013 const TArray<ValueType>& Points = Values;
1014 if (Points.Num() < 2)
1015 {
1016 GenerateUniformKnots(KnotCount);
1017 return;
1018 }
1019
1020 const int32 n = Points.Num();
1021 const int32 d = Degree;
1022 const int32 NumKnots = this->IsClosedLoop()
1023 ? (n + 2 * d + 1)
1024 : KnotCount;
1025 PairKnots.Reset();

1028 Lengths.Reserve(Points.Num() - 1);
1029 float TotalLength = 0.0f;
1030
1031 for (int32 i = 1; i < Points.Num(); ++i)
1032 {
1033 // Square root of distance for centripetal parameterization
1034 const float Length = static_cast<float>(Math::CentripetalDistance(Points[i], Points[i - 1]));
1035 Lengths.Add(Length);
1036 TotalLength += Length;
1037 }
1038
1039 // For closed loop, add the closing segment
1040 if (this->IsClosedLoop() && Points.Num() > 0)

1043 Lengths.Add(ClosureLength);
1044 TotalLength += ClosureLength;
1045 }
1046
1047 if (this->IsClosedLoop())
1048 {
1049 // Generate periodic knot vector using centripetal parameterization
1050 for (int32 i = 0; i < NumKnots; ++i)
1051 {
1052 PairKnots.Add(FKnot(static_cast<float>(i) * TotalLength / static_cast<float>(n + d), 1));
1053 }
1054 }
1055 else if (bClampEnds)
1056 {
1057 // For clamped ends, same approach as chord length but with square root distances
1058
1059 // First set starting knots to 0
1060 PairKnots.Add(FKnot(0.0f, Degree + 1));

1063 const int32 NumInternalKnots = KnotCount - 2 * (Degree + 1);

1065 {
1066 // Compute normalized Knots at each control point
1067 TArray<float> Params;
1068 Params.SetNum(Points.Num());
1069 Params[0] = 0.0f;
1070
1071 float AccumulatedLength = 0.0f;
1072 for (int32 i = 1; i < Points.Num(); ++i)
1073 {
1074 AccumulatedLength += Lengths[i - 1];
1075 Params[i] = (TotalLength > UE_SMALL_NUMBER)
1076 ? (AccumulatedLength / TotalLength)
1077 : (static_cast<float>(i) / static_cast<float>(Points.Num() - 1));
1078 }
1079
1080 // Map these Knots to internal knots
1081 for (int32 i = 0; i < NumInternalKnots; ++i)
1082 {
1083 // Map i to appropriate parameter range
1084 const float t = static_cast<float>(i + 1) / static_cast<float>(NumInternalKnots + 1);

1093 }
1094 }
1095 // Set the last (Degree+1) knots to 1.0
1096 PairKnots.Add(FKnot(1.0f, Degree + 1));
1097 }

1099 {
1100 // Unclamped knots
1101 for (int32 i = 0; i < KnotCount; ++i)
1102 {
1103 const float Param = static_cast<float>(i) / static_cast<float>(KnotCount - 1);
1104 PairKnots.Add(FKnot(Param * TotalLength, 1));
1105 }
1106 }

1178 int32 FindKnotSpan(float Parameter) const
1179 {
1180 UpdateFlatKnotsCache();
1181 const TArray<float>& Knots = FlatKnots;
1182 // First, check that the knot vector is populated.
1183 if (Knots.Num() == 0)
1184 {
1185 return 0; // Return a safe fallback.
1186 }
1187
1188 const int32 NumValues = NumKeys();
1189 if (NumValues <= 0)
1190 {
1191 UE_LOGF(LogSpline, Error, "FindKnotSpan: Not enough knots to define a valid span.");
1192 return 0; // Safe fallback.
1193 }
1194
1195 // Handle the last knot explicitly for clamped endpoints
1196 if (Parameter >= Knots.Last() - UE_SMALL_NUMBER)
1197 {
1198 return NumValues - 1; // Last span index
1199 }
1200
1201 if (Parameter <= Knots[Degree] + UE_SMALL_NUMBER)
1202 {
1203 return Degree; // First valid span
1204 }
1205
1206 // Binary search for the span
1207 int32 Low = Degree;
1208 int32 High = Knots.Num() - Degree - 1;
1209
1210 while (Low <= High)
1211 {
1212 const int32 Mid = (Low + High) / 2;
1213
1214 if (Parameter >= Knots[Mid] && Parameter < Knots[Mid + 1])
1215 {
1216 return Mid;
1217 }
1218
1219 if (Parameter < Knots[Mid])
1220 {
1221 High = Mid - 1;
1222 }
1223 else
1224 {
1225 Low = Mid + 1;
1226 }
1227 }
1228
1229 // Shouldn't reach here; return fallback
1230 UE_LOGF(LogSpline, Warning, "Fallback in FindKnotSpan for t=%f", Parameter);
1231 return Degree;
1232 }

1234 virtual void FindWindow(float Parameter, FWindow& OutWindow, FWindowStore& Scratch) const
1235 {
1236 OutWindow = {};
1237 if (NumKeys() < Degree + 1)
1238 {
1239 return;
1240 }
1241
1242 int32 Span = FindKnotSpan(Parameter);
1243 // For closed loop, we need to ensure the span wraps properly
1244 if (this->IsClosedLoop())
1245 {
1246 // For closed loop:
1247 // - Span needs to wrap around number of control points
1248 // - Window indices wrap around array
1249 const int32 NumPoints = NumKeys();
1250 for (int32 i = 0; i <= Degree; ++i)
1251 {

1253 if (Index < 0) Index += NumPoints;

1258 {
1259 // Get control points for the window
1260 for (int32 i = 0; i <= Degree; ++i)
1261 {
1262 const int32 Index = Span - Degree + i;
1263 if (Index >= 0 && Index < NumKeys())
1264 {
1265 OutWindow[i] = &GetValue(Index);
1266 }
1267 }
