68 // returns empty TOptional if queue is empty
69 TOptional<ElementType> Dequeue()
70 {
71 UE_AUTORTFM_REPORT_HAZARD_IF_CLOSED(SpscQueue);
72
73 FNode* LocalTail = Tail.load(std::memory_order_relaxed);
74 FNode* LocalTailNext = LocalTail->Next.load(std::memory_order_acquire);
75 if (LocalTailNext == nullptr)
76 {
77 return {};
78 }
79
80 ElementType* TailNextValue = (ElementType*)&LocalTailNext->Value;
81 TOptional<ElementType> Value{ MoveTemp(*TailNextValue) };
82 DestructItem(TailNextValue);
83
84 Tail.store(LocalTailNext, std::memory_order_release);
85
86 if (Value.IsSet())
87 {
88 NumElems--;
89 }
90 return Value;
91 }

194 FNode* AllocNode()
195 {
196 // first tries to allocate node from internal node cache,
197 // if attempt fails, allocates node via ::operator new()
198
199 auto AllocFromCache = [this]()
200 {
201 FNode* Node = First;
202 First = First->Next;
203 Node->Next.store(nullptr, std::memory_order_relaxed);
204 return Node;
205 };
206
207 if (First != TailCopy)

213 if (First != TailCopy)

216 }

219 }
