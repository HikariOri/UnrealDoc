24// Abstract array of positions for running the KMeans algorithm in serial or parallel
25template<typename TVectorType, bool bUseAtomicType>
26struct TClusterCenterArray
27{
28private:
29 // Atomic values for parallel algorithm
30 TStaticArray<TArray<std::atomic<typename TVectorType::FReal>>, TVectorType::NumComponents> AtomicValues;
31
32 // Regular values for serial algorithm
33 TArray<TVectorType> Values;
34public:
35 void SetNumZeroed(uint32 Num)
36 {
37 #if PLATFORM_SUPPORT_FLOATING_INTERLOCKED_ADD
38 if (bUseAtomicType)
39 {
40 for (uint32 ComponentIndex = 0; ComponentIndex < TVectorType::NumComponents; ++ComponentIndex)
41 {
42 AtomicValues[ComponentIndex].SetNumZeroed(Num);
43 }
44 }
45 else
46 #endif
47 {
48 Values.SetNumZeroed(Num);
49 }
50 }

52 void Assign(int32 Index, const TVectorType& In)
53 {
54 #if PLATFORM_SUPPORT_FLOATING_INTERLOCKED_ADD
55 if (bUseAtomicType)
56 {
57 for (uint32 ComponentIndex = 0; ComponentIndex < TVectorType::NumComponents; ++ComponentIndex)
58 {
59 AtomicValues[ComponentIndex][Index] = In[ComponentIndex];
60 }
61 }
62 else
63 #endif
64 {
65 Values[Index] = In;

69 TVectorType DivideBy(int32 Index, int32 InDivider)
70 {
71 typedef typename TVectorType::FReal NumericType;
72
73 #if PLATFORM_SUPPORT_FLOATING_INTERLOCKED_ADD
74 if (bUseAtomicType)
75 {
76 TVectorType Out;
77 for (uint32 ComponentIndex = 0; ComponentIndex < TVectorType::NumComponents; ++ComponentIndex)
78 {
79 Out[ComponentIndex] = NumericType(AtomicValues[ComponentIndex][Index]) / NumericType(InDivider);
80 }
81 return Out;
82 }
83 else
84 #endif
85 {
86 return Values[Index] / NumericType(InDivider);
87 }
88 }

90 void Accumulate(int32 Index, const TVectorType& In)
91 {
92 #if PLATFORM_SUPPORT_FLOATING_INTERLOCKED_ADD
93 if (bUseAtomicType)
94 {
95 for (uint32 ComponentIndex = 0; ComponentIndex < TVectorType::NumComponents; ++ComponentIndex)
96 {
97 AtomicValues[ComponentIndex][Index].fetch_add(In[ComponentIndex]);
98 }
99 }
100 else
101 #endif
102 {
103 Values[Index] += In;
104 }
105 }

107 void RemoveAtSwap(size_t Index, EAllowShrinking AllowShrinking)
108 {
109 #if PLATFORM_SUPPORT_FLOATING_INTERLOCKED_ADD
110 if (bUseAtomicType)
111 {
112 for (uint32 ComponentIndex = 0; ComponentIndex < TVectorType::NumComponents; ++ComponentIndex)
113 {
114 AtomicValues[ComponentIndex].RemoveAtSwap(Index, AllowShrinking);
115 }
116 }
117 else
118 #endif
119 {
120 Values.RemoveAtSwap(Index, AllowShrinking);
121 }
122 }

129 // Max iterations of K-Means clustering. Will use fewer iterations if the clustering method converges.
130 int32 MaxIterations = 500;
131
132 // Random Seed used to initialize clustering (if InitialCenters are not provided)
133 int32 RandomSeed = 0;

137 // Mapping from input points to cluster IDs
138 TArray<int32> ClusterIDs;

141 TArray<std::atomic<int32>> ClusterSizes;

153 template<typename TVectorType = FVector, bool bRunParallel=false>
154 int32 ComputeClusters(
155 TArrayView<const TVectorType> PointsToCluster, int32 NumClusters,
156 TArrayView<const TVectorType> InitialCenters = TArrayView<const TVectorType>(),
157 TArray<TVectorType>* OutClusterCenters = nullptr)
158 {
159 constexpr bool bRunParallelIfCompatible = PLATFORM_SUPPORT_FLOATING_INTERLOCKED_ADD && bRunParallel && std::atomic<typename TVectorType::FReal>::is_always_lock_free;
160
161 if (PointsToCluster.IsEmpty())
162 {
163 // nothing to cluster
164 return 0;
165 }
166
167 // Initialize cluster centers
168 TArray<TVectorType> LocalCenters;
169 TArray<TVectorType>* UseCenters = &LocalCenters;
170 if (OutClusterCenters)
171 {
172 UseCenters = OutClusterCenters;
173 UseCenters->Reset();
174 }
175 if (InitialCenters.IsEmpty())
176 {
177 TArray<int32> Ordering;
178 const int32 OrderingNum = PointsToCluster.Num();
179 Ordering.SetNumUninitialized(OrderingNum);
180 for (int32 Idx = 0; Idx < OrderingNum; ++Idx)
181 {
182 Ordering[Idx] = Idx;
183 }
184 // Shuffle the first NumClusters indices and use them as the initial centers
185 NumClusters = FMath::Min(PointsToCluster.Num(), NumClusters);
186 UseCenters->Reserve(NumClusters);
187 FRandomStream RandomStream(RandomSeed);
188 for (int32 Idx = 0; Idx < NumClusters; ++Idx)
189 {
190 Swap(Ordering[Idx], Ordering[Idx + RandomStream.RandHelper(OrderingNum - Idx)]);
191 UseCenters->Add(PointsToCluster[Ordering[Idx]]);
192 }
193 }
194 else
195 {
196 // Note: We intentionally do not check if more centers were provided than points here.
197 // The excess centers will instead be removed below when no points are assigned to them.
198 UseCenters->Append(InitialCenters);
199 }
200
201 NumClusters = UseCenters->Num();
202 ClusterIDs.Init(-1, PointsToCluster.Num());
203 ClusterSizes.SetNumZeroed(NumClusters);
204
205 if (NumClusters == 0)
206 {
207 return NumClusters;
208 }
209 else if (NumClusters == 1)
210 {
211 for (int32 PointIdx = 0; PointIdx < PointsToCluster.Num(); ++PointIdx)
212 {
213 ClusterIDs[PointIdx] = 0;
214 }
215 ClusterSizes[0] = PointsToCluster.Num();
216 return NumClusters;
217 }
218
219
220 TClusterCenterArray<TVectorType, bRunParallelIfCompatible> NextCenters;
221 NextCenters.SetNumZeroed(NumClusters);
222
223 TArray<int32> ClusterIDRemap; // array to use if we need to remap cluster IDs due to an empty cluster
224
225 int32 UseMaxIterations = FMath::Max(1, MaxIterations); // always use at least one iteration to make sure the initial assignment happens
226 for (int32 Iterations = 0; Iterations < UseMaxIterations; ++Iterations)
227 {
228 bool bClustersChanged = Iterations == 0; // clusters always change on first iteration; otherwise we must check
229 if (bRunParallelIfCompatible)
230 {
231 ParallelFor(PointsToCluster.Num(),
232 [
233 this,
234 Iterations,
235 &PointsToCluster,
236 &NextCenters,
237 &UseCenters,
238 &bClustersChanged
239 ] (uint32 PointIdx)
240 {
241 const TVectorType& Point = PointsToCluster[PointIdx];
242 double ClosestDistSq = (double)UE::Geometry::DistanceSquared(Point, (*UseCenters)[0]);
243 int32 ClosestCenter = 0;
244 for (int32 CenterIdx = 1; CenterIdx < UseCenters->Num(); ++CenterIdx)
245 {
246 double DistSq = UE::Geometry::DistanceSquared(PointsToCluster[PointIdx], (*UseCenters)[CenterIdx]);
247 if (DistSq < ClosestDistSq)
248 {
249 ClosestDistSq = DistSq;
250 ClosestCenter = CenterIdx;
251 }
252 }
253
254 ClusterSizes[ClosestCenter]++;
255 if (Iterations > 0)
256 {
257 int32 OldClusterID = ClusterIDs[PointIdx];
258 ClusterSizes[OldClusterID]--;
259 if (OldClusterID != ClosestCenter)
260 {
261 bClustersChanged = true;
262 }
263 }
264
265 ClusterIDs[PointIdx] = ClosestCenter;
266 NextCenters.Accumulate(ClosestCenter, Point);
267 });
268 }
269 else
270 {
271 for (int32 PointIdx = 0; PointIdx < PointsToCluster.Num(); ++PointIdx)
272 {
273 const TVectorType& Point = PointsToCluster[PointIdx];
274 double ClosestDistSq = (double)DistanceSquared(Point, (*UseCenters)[0]);
275 int32 ClosestCenter = 0;
276 for (int32 CenterIdx = 1; CenterIdx < UseCenters->Num(); ++CenterIdx)
277 {
278 double DistSq = DistanceSquared(PointsToCluster[PointIdx], (*UseCenters)[CenterIdx]);
279 if (DistSq < ClosestDistSq)
280 {
281 ClosestDistSq = DistSq;
282 ClosestCenter = CenterIdx;
283 }
284 }
285
286 ClusterSizes[ClosestCenter]++;
287 if (Iterations > 0)
288 {
289 int32 OldClusterID = ClusterIDs[PointIdx];
290 ClusterSizes[OldClusterID]--;
291 if (OldClusterID != ClosestCenter)
292 {
293 bClustersChanged = true;
294 }
295 }
296
297 ClusterIDs[PointIdx] = ClosestCenter;
298 NextCenters.Accumulate(ClosestCenter, Point);
299 }
300 } // bParallel
301
302 // Stop iterating if clusters are unchanged
303 if (!bClustersChanged)
304 {
305 break;
306 }
307
308 // Update cluster centers and detect/delete any empty clusters
309 bool bDeletedClusters = false;
310 for (int32 ClusterIdx = 0; ClusterIdx < UseCenters->Num(); ++ClusterIdx)
311 {
312 if (ClusterSizes[ClusterIdx] > 0)
313 {
314 (*UseCenters)[ClusterIdx] = NextCenters.DivideBy(ClusterIdx, int32(ClusterSizes[ClusterIdx]));
315 NextCenters.Assign(ClusterIdx, TVectorType::ZeroVector);
316 }
317 else
318 {
319 if (!bDeletedClusters)
320 {
321 ClusterIDRemap.SetNumUninitialized(UseCenters->Num(), EAllowShrinking::No);
322 for (int32 Idx = 0; Idx < UseCenters->Num(); ++Idx)
323 {
324 ClusterIDRemap[Idx] = Idx;
325 }
326 }
327 bDeletedClusters = true;
328 ClusterIDRemap[ClusterSizes.Num() - 1] = ClusterIdx;
329 ClusterSizes.RemoveAtSwap(ClusterIdx, EAllowShrinking::No);
330 UseCenters->RemoveAtSwap(ClusterIdx, EAllowShrinking::No);
331 NextCenters.RemoveAtSwap(ClusterIdx, EAllowShrinking::No);
332 }
333 }
334 if (bDeletedClusters)
335 {
336 checkSlow(!ClusterSizes.IsEmpty());
337 checkSlow(ClusterSizes.Num() == UseCenters->Num());
338 for (int32& ClusterID : ClusterIDs)
339 {
340 ClusterID = ClusterIDRemap[ClusterID];
341 }
342 }
343 }
344
345 return ClusterSizes.Num();
346 }
