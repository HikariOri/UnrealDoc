103 // Helpers for Replay Support
104 static bool NetSerializeStateDataAsReplayData(DataStoreType* StateTokenStore, T& InOutStateData, FArchive& Ar, UPackageMapClient& PackageMapClient)
105 {
106 TSharedPtr<FNetFieldExportGroup> NetFieldExportGroup = PackageMapClient.GetNetFieldExportGroup(DataStoreType::GetTokenStoreName().ToString());
107 if (Ar.IsSaving())
108 {
109 if (!NetFieldExportGroup.IsValid())
110 {
111 SetupReplayExportGroup(StateTokenStore, PackageMapClient);
112 }
113 FNetToken StateToken = StateTokenStore->GetOrCreateToken(InOutStateData);
114 StateTokenStore->WriteNetToken(Ar, StateToken);
115
116 if (!NetFieldExportGroup->NetFieldExports.IsValidIndex(StateToken.GetIndex()))
117 {
118 NetFieldExportGroup->NetFieldExports.SetNum(StateToken.GetIndex() + 1, EAllowShrinking::No);
119 NetFieldExportGroup->bDirtyForReplay = true;
120 }
121 FNetFieldExport& NetFieldExport = NetFieldExportGroup->NetFieldExports[StateToken.GetIndex()];
122
123 AddStateToReplayExportGroup(StateTokenStore, InOutStateData, NetFieldExportGroup, PackageMapClient);
124 PackageMapClient.TrackNetFieldExport(NetFieldExportGroup.Get(), StateToken.GetIndex());
125
126 return true;
127 }
128 else if (Ar.IsLoading())
129 {
130 FNetToken StateToken = StateTokenStore->ReadNetToken(Ar);
131 if (Ar.IsError())
132 {
133 return false;
134 }
135 if (!NetFieldExportGroup.IsValid())
136 {
137 return false;
138 }
139
140 if (StateToken.IsValid())
141 {
142 if(GetStateFromReplayExportGroup(StateTokenStore, InOutStateData, StateToken, NetFieldExportGroup, PackageMapClient))
143 {
144 return true;
145 }
146 InOutStateData = DataStoreType::GetInvalidState();
147 return false;
148 }
149 }
150 return false;
151 }

153 static bool GetStateFromReplayExportGroup(DataStoreType* StateTokenStore, T& OutData, FNetToken& StateToken, TSharedPtr<FNetFieldExportGroup> NetFieldExportGroup, UPackageMapClient& Map)
154 {
155 if (!NetFieldExportGroup.IsValid())
156 {
157 return false;
158 }
159 const uint32 TagTokenIndex = StateToken.GetIndex();
160 if (static_cast<int32>(TagTokenIndex) >= NetFieldExportGroup->NetFieldExports.Num())
161 {
162 return false;
163 }
164 FNetFieldExport& NetFieldExport = NetFieldExportGroup->NetFieldExports[StateToken.GetIndex()];
165 FNetBitReader TempAr;
166 TArray<uint8>& Data = NetFieldExportGroup->NetFieldExports[TagTokenIndex].Blob;
167 TempAr.SetData(Data.GetData(),Data.NumBytes()*8);
168 StateTokenStore->ReadTokenData(TempAr, StateToken, &Map);
169 OutData = StateTokenStore->ResolveToken(StateToken);
170 return true;
171 }
172
173 static bool AddStateToReplayExportGroup(DataStoreType* StateTokenStore, T& InData, TSharedPtr<FNetFieldExportGroup> NetFieldExportGroup, UPackageMapClient& Map)
174 {
175 if (!NetFieldExportGroup.IsValid())
176 {
177 return false;
178 }
179 FNetToken StateToken = StateTokenStore->GetOrCreateToken(InData);
180 if (static_cast<int32>(StateToken.GetIndex()) >= NetFieldExportGroup->NetFieldExports.Num())
181 {
182 NetFieldExportGroup->NetFieldExports.SetNum(StateToken.GetIndex() + 1, EAllowShrinking::No);
183 }
184 FNetFieldExport& NetFieldExport = NetFieldExportGroup->NetFieldExports[StateToken.GetIndex()];
185 if (NetFieldExport.bExported)
186 {
187 return false;
188 }
189
190 FNetBitWriter TempAr;
191 StateTokenStore->WriteTokenData(TempAr, StateTokenStore->GetOrCreatePersistentState(InData), &Map);
192
193 NetFieldExport = FNetFieldExport(StateToken.GetIndex(),0,NAME_None,*TempAr.GetBuffer());
194 NetFieldExportGroup->bDirtyForReplay = true;
195 UE_LOGF(LogNetToken, Verbose, "Replay> Exported for %ls as NetFieldIndex %u", *DataStoreType::GetTokenStoreName().ToString(), StateToken.GetIndex());
196 return true;
197 }
198
199 static bool SetupReplayExportGroup(DataStoreType* StateTokenStore, UPackageMapClient& PackageMapClient)
200 {
201 const FString StoreName = DataStoreType::GetTokenStoreName().ToString();
202 TSharedPtr<FNetFieldExportGroup> NetFieldExportGroup = PackageMapClient.GetNetFieldExportGroup(StoreName);
203 if (NetFieldExportGroup.IsValid())
204 {
205 return true;
206 }
207 NetFieldExportGroup = MakeShared<FNetFieldExportGroup>();
208 NetFieldExportGroup->PathName = StoreName;
209 NetFieldExportGroup->NetFieldExports.SetNum(StateTokenStore->StoredStates.Num());
210 for (auto& Pair : StateTokenStore->StoredStates)
211 {
212 AddStateToReplayExportGroup(StateTokenStore, Pair.Value, NetFieldExportGroup, PackageMapClient);
213 }
214 PackageMapClient.AddNetFieldExportGroup(StoreName, NetFieldExportGroup);
215 return true;
216 }
