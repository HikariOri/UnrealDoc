98struct FTAAPassParameters
99{
100 // TAA pass to run.

102
103 // Whether to use the faster shader permutation.
104 ETAAQuality Quality = ETAAQuality::High;
105
106 // Whether output texture should be render targetable.

124
125 // Full resolution depth and velocity textures to reproject the history.
126 FRDGTexture* SceneDepthTexture = nullptr;

128
129 // Anti aliased scene color.
130 // Can have alpha channel, or CoC for DOF.
131 FRDGTexture* SceneColorInput = nullptr;
132
133 // Optional information that get anti aliased, such as separate CoC for DOF.
134 FRDGTexture* SceneMetadataInput = nullptr;
135
136 // Optional override for stencil source (when depth has been swapped but stencil lives elsewhere, e.g., translucency early velocity pass)
137 FRDGTexture* StencilDepthTexture = nullptr;
138
139 // If 1, a bilateral filter based on the circle-of-confusion for depth-of-field is used to reject history.
140 // If 0, this is disabled.
141 float CoCBilateralFilterStrength = 1.0;

143
144 FTAAPassParameters(const FViewInfo& View)
145 : InputViewRect(View.ViewRect)
146 , OutputViewRect(View.ViewRect)
147 { }
148
149
150 // Customizes the view rectangles for input and output.
151 FORCEINLINE void SetupViewRect(const FViewInfo& View, int32 InResolutionDivisor = 1)
152 {
153 ResolutionDivisor = InResolutionDivisor;
154
155 InputViewRect = View.ViewRect;
156
157 // When upsampling, always upsampling to top left corner to reuse same RT as before upsampling.
158 if (IsTAAUpsamplingConfig(Pass))
159 {
160 OutputViewRect.Min = FIntPoint(0, 0);
161 OutputViewRect.Max = View.GetSecondaryViewRectSize();
162 }
163 else
164 {
165 OutputViewRect = InputViewRect;
166 }
167 }

170 FIntPoint GetOutputExtent() const;

173 bool Validate() const;
