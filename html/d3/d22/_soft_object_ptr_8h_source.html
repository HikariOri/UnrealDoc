82 UObject* LoadSynchronous(ELoadFlags LoadFlags = LOAD_None) const
83 {
84 UObject* Asset = Get();
85 if (Asset == nullptr && !IsNull())
86 {
87 ToSoftObjectPath().TryLoad(nullptr, LoadFlags);
88
89 // TryLoad will have loaded this pointer if it is valid
90 Asset = Get();
91 }
92 return Asset;
