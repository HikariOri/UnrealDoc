86// Cache-state snapshot taken at the moment an install-cache error is produced.
87// Used as a plain field on errors that carry cache state directly
88// (InstallCachePurgeError, InstallCacheDefragError) and as the sole field of
89// the pushable InstallCacheErrorContext, so the same data can be attached to
90// errors that originate below the cache layer.
91struct FInstallCacheErrorState
92{
93 FIoStatus IoError = FIoStatus(EIoErrorCode::Unknown);
94 const TCHAR* CasType = nullptr;
95 uint64 MaxCacheSize = 0;
96 uint64 CacheSize = 0;
97 uint64 RequestedSize = 0;
98 uint64 DiskTotalBytes = 0;
99 uint64 DiskFreeBytes = 0;
100 uint32 LineNo = uint32(__builtin_LINE());
101 bool bDiskQuerySucceeded = false;
102};
