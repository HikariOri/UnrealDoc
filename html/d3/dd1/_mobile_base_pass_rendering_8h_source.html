357template< typename LightMapPolicyType, EMobileLocalLightSetting LocalLightSetting, EMobileTranslucentColorTransmittanceMode TranslucentColorTransmittanceFallback = EMobileTranslucentColorTransmittanceMode::DEFAULT>

363 static bool ShouldCompilePermutation(const FMeshMaterialShaderPermutationParameters& Parameters)
364 {
365 // We compile the point light shader combinations based on the project settings
366 const bool bIsLit = Parameters.MaterialParameters.ShadingModels.IsLit();
367 const bool bDeferredShadingEnabled = IsMobileDeferredShadingEnabled(Parameters.Platform);
368 const bool bIsTranslucent = IsTranslucentBlendMode(Parameters.MaterialParameters);
369 const bool bIsSLW = Parameters.MaterialParameters.ShadingModels.HasShadingModel(MSM_SingleLayerWater);
370 const bool bMaterialUsesForwardShading = bIsLit && (bIsTranslucent || bIsSLW);
371 // Translucent materials always support clustered shading on mobile deferred
372 const bool bForwardShading = !bDeferredShadingEnabled || bMaterialUsesForwardShading;
373
374 EMobileLocalLightSetting SupportedLocalLightsType = EMobileLocalLightSetting::LOCAL_LIGHTS_DISABLED;
375 if (bForwardShading && bIsLit)
376 {
377 SupportedLocalLightsType = GetMobileForwardLocalLightSetting(Parameters.Platform);
378 }
379 // Deferred shading does not need SkyLight and LocalLight permutations
380 // TODO: skip skylight permutations for deferred
381 bool bEnableLocalLights = LocalLightSetting != EMobileLocalLightSetting::LOCAL_LIGHTS_DISABLED;
382 const bool bShouldCacheByLocalLights = !bEnableLocalLights || (bIsLit && (SupportedLocalLightsType == LocalLightSetting));
383
384 return TMobileBasePassPSBaseType<LightMapPolicyType>::ShouldCompilePermutation(Parameters) &&
385 bShouldCacheByLocalLights &&
386 ShouldCacheShaderForColorTransmittanceFallback(Parameters, TranslucentColorTransmittanceFallback);
387 }
388
389 static void ModifyCompilationEnvironment(const FMaterialShaderPermutationParameters& Parameters, FShaderCompilerEnvironment& OutEnvironment)
390 {
391 MobileBasePassModifyCompilationEnvironment(Parameters, OutEnvironment);
392
393 const bool bMobileUsesShadowMaskTexture = MobileUsesShadowMaskTexture(Parameters.Platform);
394 const bool bEnableClusteredReflections = MobileForwardEnableClusteredReflections(Parameters.Platform);
395 const bool bDeferredShadingEnabled = IsMobileDeferredShadingEnabled(Parameters.Platform);
396 const bool bTranslucentMaterial = IsTranslucentBlendMode(Parameters.MaterialParameters);
397 const bool bIsSLW = Parameters.MaterialParameters.ShadingModels.HasShadingModel(MSM_SingleLayerWater);
398 const bool bTranslucentOrSLWMaterial = bTranslucentMaterial || bIsSLW;
399 const bool bIsLit = Parameters.MaterialParameters.ShadingModels.IsLit();
400 const bool bMaterialUsesForwardShading = bIsLit && bTranslucentOrSLWMaterial;
401 // Translucent materials always support clustered shading on mobile deferred
402 const bool bForwardShading = !bDeferredShadingEnabled || bMaterialUsesForwardShading;
403
404 // Only non-static skylights contribute into forward basepass
405 const bool bProjectSupportsNonStaticSkyLights = FReadOnlyCVARCache::EnableStationarySkylight() || !IsStaticLightingAllowed();
406 OutEnvironment.SetDefine(TEXT("ENABLE_SKY_LIGHT"), bIsLit && bForwardShading && bProjectSupportsNonStaticSkyLights);
407 OutEnvironment.SetDefine(TEXT("ENABLE_AMBIENT_OCCLUSION"), bForwardShading && IsMobileAmbientOcclusionEnabled(Parameters.Platform) ? 1u : 0u);
408
409 FForwardLightingParameters::ModifyCompilationEnvironment(Parameters.Platform, OutEnvironment);
410 OutEnvironment.SetDefine(TEXT("ENABLE_CLUSTERED_LIGHTS"), (LocalLightSetting == EMobileLocalLightSetting::LOCAL_LIGHTS_ENABLED) ? 1u : 0u);
411 OutEnvironment.SetDefine(TEXT("ENABLE_CLUSTERED_REFLECTION"), bEnableClusteredReflections ? 1u : 0u);
412 OutEnvironment.SetDefine(TEXT("ENABLE_SHADOWMASKTEXTURE"), bMobileUsesShadowMaskTexture && !bTranslucentOrSLWMaterial ? 1u : 0u);
413 OutEnvironment.SetDefine(TEXT("ENABLE_DBUFFER_TEXTURES"), Parameters.MaterialParameters.MaterialDomain == MD_Surface ? 1u : 0u);
414 EMobileTranslucentColorTransmittanceMode TranslucentColorTransmittanceMode = EMobileTranslucentColorTransmittanceMode::DEFAULT;
415 if (MaterialRequiresColorTransmittanceBlending(Parameters.MaterialParameters))
416 {
417 TranslucentColorTransmittanceMode = (TranslucentColorTransmittanceFallback == EMobileTranslucentColorTransmittanceMode::DEFAULT) ? MobileDefaultTranslucentColorTransmittanceMode(Parameters.Platform) : TranslucentColorTransmittanceFallback;
418 }
419 OutEnvironment.SetDefine(TEXT("MOBILE_TRANSLUCENT_COLOR_TRANSMITTANCE_DUAL_SRC_BLENDING"), TranslucentColorTransmittanceMode == EMobileTranslucentColorTransmittanceMode::DUAL_SRC_BLENDING ? 1u : 0u);
420 OutEnvironment.SetDefine(TEXT("MOBILE_TRANSLUCENT_COLOR_TRANSMITTANCE_PROGRAMMABLE_BLENDING"), TranslucentColorTransmittanceMode == EMobileTranslucentColorTransmittanceMode::PROGRAMMABLE_BLENDING ? 1u : 0u);
421 OutEnvironment.SetDefine(TEXT("MOBILE_TRANSLUCENT_COLOR_TRANSMITTANCE_SINGLE_SRC_BLENDING"), TranslucentColorTransmittanceMode == EMobileTranslucentColorTransmittanceMode::SINGLE_SRC_BLENDING ? 1u : 0u);
422 OutEnvironment.SetDefine(TEXT("MOBILE_SSR_ENABLED"), (!bDeferredShadingEnabled && AreMobileScreenSpaceReflectionsEnabled(Parameters.Platform)) ? 1u : 0u);
423
424 TMobileBasePassPSBaseType<LightMapPolicyType>::ModifyCompilationEnvironment(Parameters, OutEnvironment);
425 }

515template <EMobileLocalLightSetting LocalLightSetting, typename LightMapPolicyType>
516bool GetMobileBasePassShaders(
517 const LightMapPolicyType& RESTRICT LightMapPolicy,
518 const FMaterial& Material,
519 const FVertexFactoryType* VertexFactoryType,
520 EMobileTranslucentColorTransmittanceMode ColoredTransmittanceFallback,
521 TShaderRef<TMobileBasePassVSPolicyParamType<LightMapPolicyType>>& VertexShader,
522 TShaderRef<TMobileBasePassPSPolicyParamType<LightMapPolicyType>>& PixelShader
523)
524{
525 using FVertexShaderType = TMobileBasePassVSPolicyParamType<LightMapPolicyType>;
526 using FPixelShaderType = TMobileBasePassPSPolicyParamType<LightMapPolicyType>;
527
528 FMaterialShaderTypes ShaderTypes;
529 ShaderTypes.AddShaderType<TMobileBasePassVS<LightMapPolicyType>>();
530
531 switch (ColoredTransmittanceFallback)
532 {
533 default:
534 case EMobileTranslucentColorTransmittanceMode::DEFAULT:
535 ShaderTypes.AddShaderType<TMobileBasePassPS<LightMapPolicyType, LocalLightSetting, EMobileTranslucentColorTransmittanceMode::DEFAULT>>();
536 break;
537 case EMobileTranslucentColorTransmittanceMode::SINGLE_SRC_BLENDING:
538 ShaderTypes.AddShaderType<TMobileBasePassPS<LightMapPolicyType, LocalLightSetting, EMobileTranslucentColorTransmittanceMode::SINGLE_SRC_BLENDING>>();
539 break;
540 }
541
542 FMaterialShaders Shaders;
543 if (!Material.TryGetShaders(ShaderTypes, VertexFactoryType, Shaders))
544 {
545 return false;
546 }
547
548 Shaders.TryGetVertexShader(VertexShader);
549 Shaders.TryGetPixelShader(PixelShader);
550 return true;
551}
