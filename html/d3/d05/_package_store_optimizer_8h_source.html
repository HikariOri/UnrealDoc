30 {
31 NameMap.Add(DisplayId);

36 void MarkNamesAsReferenced(const TArray<FName>& Names, TArray<int32>& OutNameIndices)
37 {
38 for (FName Name : Names)
39 {
40 FDisplayNameEntryId DisplayId(Name);
41 int32& Index = NameIndices.FindOrAdd(DisplayId);
42 if (Index == 0)
43 {
44 NameMap.Add(DisplayId);
45 Index = NameMap.Num();
46 }
47
48 OutNameIndices.Add(Index - 1);
49 }
50 }
51
52 void MarkNameAsReferenced(FName Name)
53 {
54 FDisplayNameEntryId DisplayId(Name);
55 int32& Index = NameIndices.FindOrAdd(DisplayId);
56 if (Index == 0)
57 {
58 NameMap.Add(DisplayId);
59 Index = NameMap.Num();
60 }
61 }
62
63 FMappedName MapName(FName Name) const
64 {
65 int32 Index = NameIndices.FindChecked(FDisplayNameEntryId(Name));
66 return FMappedName::Create(Index - 1, Name.GetNumber(), NameMapType);
67 }
68
69 TConstArrayView<FDisplayNameEntryId> GetNameMap() const
70 {
71 return NameMap;
72 }
73
74 void Empty()
75 {
76 NameIndices.Empty();
77 NameMap.Empty();
78 }
79
80private:
81 TMap<FDisplayNameEntryId, int32> NameIndices;
82 TArray<FDisplayNameEntryId> NameMap;
83 FMappedName::EType NameMapType = FMappedName::EType::Package;
