29template <typename TLocationHit>
30void FillHitHelper(TLocationHit& Hit, const Chaos::FReal Distance, const FVector& WorldPosition, const FVector& WorldNormal, int32 FaceIdx, bool bComputeMTD)
31{
32 if constexpr (std::is_base_of_v<ChaosInterface::FLocationHit, TLocationHit> || std::is_base_of_v<ChaosInterface::FPTLocationHit, TLocationHit>)
33 {
34 Hit.Distance = Distance;
35 Hit.WorldPosition = WorldPosition;
36 Hit.WorldNormal = WorldNormal;
37 Hit.Flags = Distance > 0.f || bComputeMTD ? EHitFlags::Distance | EHitFlags::Normal | EHitFlags::Position : EHitFlags::Distance | EHitFlags::FaceIndex;
38 Hit.FaceIndex = FaceIdx;
39 }
40}

203 template <ESQType SQ>
204 bool Visit(const Chaos::TSpatialVisitorData<TPayload>& Instance, Chaos::FQueryFastData* CurData)
205 {
206 //QUICK_SCOPE_CYCLE_COUNTER(SQVisit);
207 TPayload Payload = Instance.Payload;
208
209 //todo: add a check to ensure hitbuffer matches SQ type
210 using namespace Chaos;
211 TGeometryType* GeometryParticle = GetPayloadForThread(Payload);
212
213 if (!GeometryParticle)
214 {
215 // This case handles particles created by the physics simulation without the main thread
216 // being made aware of their creation. We have a PT particle but no external particle
217 ensure(bGTData);
218
219 return true;
220 }
221
222 // Detect and fix any dangling handles on the game and physics threads
223 if constexpr (bGTData)
224 {
225 if (!(GeometryParticle->UniqueIdx() == Payload.UniqueIdx()))
226 {
227 UE_LOGF(LogChaos, Warning, "Query Dangling handle detected on game Thread. Payload Id: %d, Particle Id: %d", Payload.UniqueIdx().Idx, GeometryParticle->UniqueIdx().Idx);
228 ensureMsgf(false, TEXT("Query dangling handle detected on game Thread. Payload Id: %d, Particle Id: %d"), Payload.UniqueIdx().Idx, GeometryParticle->UniqueIdx().Idx);
229 return true;
230 }
231 }
232 else
233 {
234 if (GeometryParticle->GetHandleIdx() == INDEX_NONE || !(GeometryParticle->UniqueIdx() == Payload.UniqueIdx()))
235 {
236 UE_LOGF(LogChaos, Warning, "Query Dangling handle detected on Physics Thread. Payload Id: %d, Particle Id: %d", Payload.UniqueIdx().Idx, GeometryParticle->UniqueIdx().Idx);
237 ensureMsgf(false, TEXT("Query dangling handle detected on Physics Thread. Payload Id: %d, Particle Id: %d"), Payload.UniqueIdx().Idx, GeometryParticle->UniqueIdx().Idx);
238 return true;
239 }
240 }
241
242 const FShapesArray& Shapes = GeometryParticle->ShapesArray();
243
244 const bool bTestShapeBounds = Shapes.Num() > 1;
245 bool bContinue = true;
246
247 const FRigidTransform3 ActorTM(GeometryParticle->GetX(), GeometryParticle->GetR());
248
249#if CHAOS_DEBUG_DRAW
250 bool bAllShapesIgnoredInPrefilter = true;
251 bool bHitBufferIncreased = false;
252#endif
253
254 for (int32 ShapeIndex = 0; ShapeIndex < Shapes.Num(); ++ShapeIndex)
255 {
256 const TUniquePtr<FPerShapeData>& Shape = Shapes[ShapeIndex];
257
258 const FImplicitObject* Geom = Shape->GetGeometry();
259
260 CVD_TRACE_SCOPED_SCENE_QUERY_VISIT_HELPER(EChaosVDSceneQueryVisitorType::NarrowPhase, ActorTM, Payload.UniqueIdx().Idx, ShapeIndex, CurData);
261
262 if (bTestShapeBounds)
263 {
264 FAABB3 InflatedWorldBounds;
265 if (SQ == ESQType::Raycast)
266 {
267 InflatedWorldBounds = Shape->GetWorldSpaceShapeBounds();
268 }
269 else
270 {
271 // Transform to world bounds and get the proper half extent.
272 const FVec3 WorldHalfExtent = QueryGeomWorldBounds.Extents() * 0.5f;
273
274 InflatedWorldBounds = FAABB3(Shape->GetWorldSpaceShapeBounds().Min() - WorldHalfExtent, Shape->GetWorldSpaceShapeBounds().Max() + WorldHalfExtent);
275 }
276
277 if (SQ != ESQType::Overlap)
278 {
279 //todo: use fast raycast
280 Chaos::FReal TmpTime, TmpExitTime;
281 const FVec3 InflatedBoundsTraceStart = SQ == ESQType::Raycast ? StartPoint : QueryGeomWorldBounds.Center();
282 if (!InflatedWorldBounds.RaycastFast(InflatedBoundsTraceStart, CurData->Dir, CurData->InvDir, CurData->bParallel, CurData->CurrentLength, CurData->InvCurrentLength, TmpTime, TmpExitTime))
283 {
284 CVD_SET_SQ_SHAPE_REJECT_REASON(EChaosVDSQVisitRejectReason::FailedFastBoundTest);
285 continue;
286 }
287 }
288 else
289 {
290 if (!InflatedWorldBounds.Contains(QueryCenter))
291 {
292 CVD_SET_SQ_SHAPE_REJECT_REASON(EChaosVDSQVisitRejectReason::FailedFastBoundTest);
293 continue;
294 }
295 }
296 }
297
298 //TODO: use gt particles directly
299 ECollisionQueryHitType HitType = QueryFlags & FChaosQueryFlag::ePREFILTER ? QueryCallback.PreFilter(QueryFilterData, *Shape, *GeometryParticle) : ECollisionQueryHitType::Block;
300
301 if (HitType != ECollisionQueryHitType::None)
302 {
303#if CHAOS_DEBUG_DRAW
304 bAllShapesIgnoredInPrefilter = false;
305#endif
306
307 //QUICK_SCOPE_CYCLE_COUNTER(SQNarrow);
308 THitType Hit;
309 Hit.Actor = GeometryParticle;
310 Hit.Shape = Shape.Get();
311
312 bool bHit = false;
313
314 FVec3 WorldPosition{ 0.0f }, WorldNormal{0.0f};
315 Chaos::FReal Distance = 0; //not needed but fixes compiler warning for overlap
316 int32 FaceIdx = INDEX_NONE; //not needed but fixes compiler warning for overlap
317 FVec3 FaceNormal = FVec3::ZeroVector;
318 const bool bComputeMTD = !!((uint16)(OutputFlags.HitFlags & EHitFlags::MTD));
319
320 if (SQ == ESQType::Raycast)
321 {
322 FVec3 LocalNormal;
323 FVec3 LocalPosition;
324
325 const FVec3 DirLocal = ActorTM.InverseTransformVectorNoScale(Dir);
326 const FVec3 StartLocal = ActorTM.InverseTransformPositionNoScale(StartPoint);
327 bHit = Geom->Raycast(StartLocal, DirLocal, CurData->CurrentLength, /*Thickness=*/0.0, Distance, LocalPosition, LocalNormal, FaceIdx);
328 if (bHit)
329 {
330 WorldPosition = ActorTM.TransformPositionNoScale(LocalPosition);
331 WorldNormal = ActorTM.TransformVectorNoScale(LocalNormal);
332 }
333 }
334 else if (SQ == ESQType::Sweep && ensure(QueryGeom))
335 {
336 bHit = SweepQuery(*Geom, ActorTM, *QueryGeom, StartTM, CurData->Dir, CurData->CurrentLength, Distance, WorldPosition, WorldNormal, FaceIdx, FaceNormal, 0.f, bComputeMTD);
337 }
338 else if (SQ == ESQType::Overlap && ensure(QueryGeom))
339 {
340 if (bComputeMTD)
341 {
342 FMTDInfo MTDInfo;
343 bHit = OverlapQuery(*Geom, ActorTM, *QueryGeom, StartTM, /*Thickness=*/0, &MTDInfo);
344 if (bHit)
345 {
346 WorldNormal = MTDInfo.Normal;
347 WorldPosition = MTDInfo.Position;
348 Distance = -MTDInfo.Penetration;
349 }
350 }
351 else
352 {
353 bHit = OverlapQuery(*Geom, ActorTM, *QueryGeom, StartTM, /*Thickness=*/0);
354 }
355 }
356
357 if (bHit)
358 {
359 //QUICK_SCOPE_CYCLE_COUNTER(SQNarrowHit);
360
361 bool bAcceptHit = true;
362 if constexpr (SQ == ESQType::Sweep && std::is_same_v<THitType, ChaosInterface::FSweepHit>)
363 {
364 const THitType* CurrentHit = HitBuffer.GetCurrentHit();
365 bool bIsCoLocatedHit = false;
366 if (FaceIdx != INDEX_NONE || !FaceNormal.IsNearlyZero())
367 {
368 if (CurrentHit)
369 {
370 constexpr static FReal CoLocationEpsilon = 1e-6;
371 const FReal DistDelta = FMath::Abs(Distance - CurrentHit->Distance);
372
373 if (DistDelta < CoLocationEpsilon && !HitFaceNormal.IsNearlyZero())
374 {
375 bIsCoLocatedHit = true;
376 // We already have a face hit from another triangle mesh - see if this one is better (more opposing the sweep)
377 const FReal OldDot = FVec3::DotProduct(CurData->Dir, HitFaceNormal);
378 const FReal NewDot = FVec3::DotProduct(CurData->Dir, FaceNormal);
379
380 if (NewDot < OldDot)
381 {
382 // More opposing
383 HitFaceNormal = FaceNormal;
384 }
385 else
386 {
387 CVD_SET_SQ_SHAPE_REJECT_REASON(EChaosVDSQVisitRejectReason::ColocatedHitHasWorseNormal);
388 // This hit is co-located but has a worse normal
389 bAcceptHit = false;
390 }
391 }
392 }
393
394 if (bAcceptHit)
395 {
396 // Record the new face normal
397 HitFaceNormal = FaceNormal;
398 }
399 }
400 // If we're not in a co-located hit, we need to make sure the new hit is actually better than the old.
401 // For normal hits this is redundant due to clipping length.
402 // When there is an initial overlaps though, we cannot clip length < 0, so we must check the returned distance.
403 if (CurrentHit != nullptr && !bIsCoLocatedHit && CurrentHit->Distance < Distance)
404 {
405 bAcceptHit = false;
406 }
407 }
408
409 if (bAcceptHit)
410 {
411 FillHitHelper(Hit, Distance, WorldPosition, WorldNormal, FaceIdx, bComputeMTD);
412
413 if constexpr (std::is_base_of_v<ChaosInterface::FQueryHit, THitType> || std::is_base_of_v<ChaosInterface::FPTQueryHit, THitType>)
414 {
415 Hit.FaceNormal = HitFaceNormal;
416 }
417
418 HitType = QueryFlags & FChaosQueryFlag::ePOSTFILTER ? QueryCallback.PostFilter(QueryFilterData, Hit) : HitType;
419
420 if (HitType != ECollisionQueryHitType::None)
421 {
422 //overlap never blocks
423 const bool bBlocker = (HitType == ECollisionQueryHitType::Block || bAnyHit || HitBuffer.WantsSingleResult());
424 HitBuffer.InsertHit(Hit, bBlocker);
425
426 CVD_FILL_HIT_DATA_HELPER(Hit, HitType);
427
428#if CHAOS_DEBUG_DRAW
429 bHitBufferIncreased = true;
430#endif
431
432 if (bBlocker && SQ != ESQType::Overlap)
433 {
434 // When there is an initial overlap, MTD will return a negative distance. We need to clamp this to 0 as many checks don't work properly with a negative length.
435 CurData->SetLength(FMath::Max((FReal)0., Distance));
436 // If we have a zero length raycast, this is an initial overlap and there's no better result possible.
437 // Note: We do not do this check with sweeps as some initial overlap sweeps are better than others (more negative MTD).
438 if (CurData->CurrentLength == 0 && (SQ == ESQType::Raycast))
439 {
440 bContinue = false;
441 break;
442 }
443 }
444
445 if (bAnyHit)
446 {
447 bContinue = false;
448 break;
449 }
450 }
451 else
452 {
453 CVD_SET_SQ_SHAPE_REJECT_REASON(EChaosVDSQVisitRejectReason::PostFilter);
454 }
455 }
456 }
457 else
458 {
459 CVD_SET_SQ_SHAPE_REJECT_REASON(EChaosVDSQVisitRejectReason::NoHit);
460 }
461 }
462 else
463 {
464 CVD_SET_SQ_SHAPE_REJECT_REASON(EChaosVDSQVisitRejectReason::PreFilter);
465 }
466 }
467
468#if CHAOS_DEBUG_DRAW
469 if (DebugParams.IsDebugQuery() && ChaosSQDrawDebugVisitorQueries)
470 {
471 DebugDraw<SQ>(Instance, CurData, bAllShapesIgnoredInPrefilter, bHitBufferIncreased);
472 }
473#endif
474
475 return bContinue;
476 }

617 template <ESQType SQ>
618 bool Visit(const Chaos::TSpatialVisitorData<TPayload>& Instance, Chaos::FQueryFastData* CurData)
619 {
620 bool bContinue = true;
621 TPayload Payload = Instance.Payload;
622 //todo: add a check to ensure hitbuffer matches SQ type
623 using namespace Chaos;
624 TGeometryType* GeometryParticle = GetPayloadForThread(Payload);
625 if (!GeometryParticle)
626 {
627 // This case handles particles created by the physics simulation without the main thread
628 // being made aware of their creation. We have a PT particle but no external particle
629 ensure(bGTData);
630 return true;
631 }
632 const FShapesArray& Shapes = GeometryParticle->ShapesArray();
633 THitType Hit;
634 Hit.Actor = GeometryParticle;
635
636 for (int32 ShapeIndex = 0; ShapeIndex < Shapes.Num(); ++ShapeIndex)
637 {
638 const TUniquePtr<FPerShapeData>& Shape = Shapes[ShapeIndex];
639
640 CVD_TRACE_SCOPED_SCENE_QUERY_VISIT_HELPER(EChaosVDSceneQueryVisitorType::BroadPhase, FTransform(GeometryParticle->GetR(), GeometryParticle->GetX()), Payload.UniqueIdx().Idx, ShapeIndex, CurData);
641
642 ECollisionQueryHitType HitType = QueryFlags & FPhysicsQueryFlag::ePREFILTER ? QueryCallback.PreFilter(QueryFilterData, *Shape, *GeometryParticle) : ECollisionQueryHitType::Block;
643 if (HitType != ECollisionQueryHitType::None)
644 {
645 const bool bBlocker = (HitType == ECollisionQueryHitType::Block || bAnyHit || HitBuffer.WantsSingleResult());
646 Hit.Shape = Shape.Get();
647 HitBuffer.InsertHit(Hit, bBlocker);
648
649 CVD_FILL_HIT_DATA_HELPER(Hit, HitType);
650
651 if (bAnyHit)
652 {
653 bContinue = false;
654 }
655 break;
656 }
657 else
658 {
659 CVD_SET_SQ_SHAPE_REJECT_REASON(EChaosVDSQVisitRejectReason::PreFilter);
660 }
661 }
662 return bContinue;
663 }
664
665 const FVector StartPoint = FVector::ZeroVector;
666 const FVector Dir = FVector::ZeroVector;
667 const FVector HalfExtents = FVector::ZeroVector;
668 FHitFlags OutputFlags;
669 bool bAnyHit;
670 const ChaosInterface::FQueryDebugParams DebugParams;
671 ChaosInterface::FSQHitBuffer<THitType>& HitBuffer;
672 const FChaosQueryFlags QueryFlags;
673 const Chaos::Filter::FQueryFilterData QueryFilterData;
674 const QueryGeometryType* QueryGeom;

676 const FTransform StartTM;

681void SweepHelper(const QueryGeomType& QueryGeom, const Chaos::ISpatialAcceleration<TPayload, Chaos::FReal, 3>& SpatialAcceleration, const FTransform& StartTM, const FVector& Dir, const float DeltaMagnitude, ChaosInterface::FSQHitBuffer<TSweepHit>& HitBuffer, EHitFlags OutputFlags, const ChaosInterface::FSceneQueryCommonParams& InCommonParams)
682{
683 using namespace Chaos;
684 using namespace ChaosInterface;
685
686 const FAABB3 Bounds = QueryGeom.CalculateTransformedBounds(StartTM);
687 // There are cases where an invalid shape comes in which produces an invalid bounds. In this case, nan will enter causing the entire tree to be queried which will timeout.
688 const bool bIsValid = !Bounds.Min().ContainsNaN() && !Bounds.Max().ContainsNaN();
689 if (ensure(bIsValid))
690 {
691 TSQVisitor<QueryGeomType, TPayload, TSweepHit, std::is_same<TSweepHit, FSweepHit>::value> SweepVisitor(StartTM, Dir, HitBuffer, OutputFlags, QueryGeom, Bounds, InCommonParams);
692
693 HitBuffer.IncFlushCount();
694
695 // NOTE: Previously, we'd attempt to perform a sweep as an overlap if the DeltaMagnitude was 0.
696 // This caused various discrepancies as some of the out info couldn't be computed. Instead, sweep has been updated to handle 0 length sweeps.
697 const FVector HalfExtents = Bounds.Extents() * 0.5f;
698 SpatialAcceleration.Sweep(Bounds.GetCenter(), Dir, DeltaMagnitude, HalfExtents, SweepVisitor);
699
700 HitBuffer.DecFlushCount();
701 }
