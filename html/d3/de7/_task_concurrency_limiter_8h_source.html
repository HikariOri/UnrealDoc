16// a queue of free slots in range [0 .. max_concurrency). Initially contains all slots in the range.
17class FConcurrencySlots
18{
19public:
20 explicit FConcurrencySlots(uint32 MaxConcurrency)
21 : FreeSlots(MaxConcurrency)
22 {
23 for (uint32 Index = IndexOffset; Index < MaxConcurrency + IndexOffset; ++Index)
24 {
25 FreeSlots.push(Index);
26 }
27 }
28
29 bool Alloc(uint32& Slot)
30 {
31 if (FreeSlots.try_pop(Slot))
32 {
33 Slot -= IndexOffset;
34 return true;
35 }
36
37 return false;
38 }
39
40 void Release(uint32 Slot)
41 {
42 FreeSlots.push(Slot + IndexOffset);
43 }
44
45private:
46 // this queue uses 0 as a special "null" value. to work around this, slots are shifted by one for storage, thus ending up in
47 // [1 .. max_concurrency] range
48 static constexpr int32 IndexOffset = 1;
49 atomic_queue::AtomicQueueB<uint32> FreeSlots; // a bounded lock-free FIFO queue
50};

102template<typename TaskFunctionType>
103void FPimpl::Push(const TCHAR* DebugName, TaskFunctionType&& TaskFunction)
104{
105 FLimiterTask* LimiterTask = new FLimiterTask();
106 LowLevelTasks::FTask& Task = LimiterTask->Task;
107 Task.Init(
108 DebugName,
109 TaskPriority,
110 [
111 TaskFunction = MoveTemp(TaskFunction),
112 this,
113 // Assign Pimpl to keep the TaskConcurrenyLimiter alive as long as the Task we created is alive.
114 Pimpl = TSharedFromThis<FPimpl>::AsShared(),
115 // Assign FLimiterTask wrapper into a TRefCountPtr to keep the LimiterTask alive as long as its inner Task is alive.
116 LimiterTask = TRefCountPtr<FLimiterTask>(LimiterTask)
117 ]()
118 {
119 LowLevelTasks::FTask& Task = LimiterTask->Task;
120
121 // We can't pass the ConcurrencySlot in the lambda during creation as
122 // it's not actually acquired yet. The value is assigned during ProcessQueue
123 // after we have called Alloc and received a ConcurrencySlot for it.
124 uint32 ConcurrencySlot = (uint32)(UPTRINT)Task.GetUserData();
125
126 // This task is now executing, either on a taskthread that received the task we created, or on
127 // the thread that called Wait if the waitthread reached the TryExpedite call before the
128 // taskthread started the Task.
129 // In the case that we are in the taskthread, we need to implement our two contracts:
130 // 1. Remove Task from ScheduledTasks as soon as it starts.
131 // This prevent waitthread wastefully calling TryExpedite in most cases.
132 // Note a useless TryExpedite will still occur and be a noop if the threads are racing.
133 // Remove the pointer so that we can later safely free the Tasks's memory.
134 // 2. Call Release on Task when removing it from ScheduledTasks.
135 // This balances AddRef done in ProcessQueue.
136 // Note that if we reach here in the waitthread, then we have already removed the task from
137 // ScheduledTasks and this call to exchange is a noop and we set bOwnLimiterTask=false.
138 bool bOwnLimiterTask = ScheduledTasks[ConcurrencySlot].Task.exchange(nullptr, std::memory_order_release) != nullptr;
139 if (bOwnLimiterTask)
140 {
141 // Whomever obtains the task from ScheduledTasks must release the reference
142 LimiterTask->Release();
143 }
144
145 TaskFunction(ConcurrencySlot);
146
147 // Finish bookkeeping for the current task and look for more tasks to launch.
148 // If we find another task to launch, have it use the ConcurrencySlot we just finished
149 // with; note this means use of a ConcurrencySlot may shift around between threads.
150 // For the first task we find, bias it towards the current worker thread if possible.
151 // If we find more than one, kick off a new Task for the global scheduler to schedule,
152 // for every task after the first.
153 CompleteWorkItem(ConcurrencySlot);
154 }
155 );
156
157 AddWorkItem(LimiterTask);
158}
