117 void ProcessRegistration()
118 {
119 // Always take the UObjectDeleteListeners mutex first to preserve lock order in all case
120 // and avoid deadlock. This is important because we need to follow this lock flow :
121 // ConditionalFinishDestroy -> RemoveObjectFromDeleteListeners (UObjectDeleteListenersCritical) -> NotifyUObjectDeleted -> RemoveAnnotation (AnnotationMapCritical)
122 GUObjectArray.LockUObjectDeleteListeners();
123 ON_SCOPE_EXIT
124 {
125 GUObjectArray.UnlockUObjectDeleteListeners();
126 };
127
128 UE::TWriteScopeLock AnnotationMapLock(AnnotationMapCritical);
129 if (AnnotationMap.Num() == 0)
130 {
131 if (bRegistered)
132 {
133 GUObjectArray.RemoveUObjectDeleteListener(this);
134 bRegistered = false;
135 }
136 }
137 else
138 {
139 if (!bRegistered)
140 {
141 GUObjectArray.AddUObjectDeleteListener(this);
142 bRegistered = true;
143 }
144 }

374 {
375 UE::TScopeLock InverseAnnotationMapLock(InverseAnnotationMapCritical);

377 {
378 RemoveAnnotation(Object); // adding the default annotation is the same as removing an annotation

571template<typename TAnnotation, bool bAutoRemove, int32 NumAnnotationsPerChunk = 64 * 1024>
572class FUObjectAnnotationChunked : public FUObjectArray::FUObjectDeleteListener
573{
574 struct TAnnotationChunk
575 {
576 int32 Num;
577 TAnnotation* Items;
578
579 TAnnotationChunk()
580 : Num(0)
581 , Items(nullptr)
582 {}
583 };

587 TArray<TAnnotationChunk> Chunks;
589 int32 NumAnnotations;
591 int32 MaxAnnotations;
593 uint32 CurrentAllocatedMemory;
595 uint32 MaxAllocatedMemory;
596
598 FTransactionallySafeRWLock AnnotationArrayCritical;
599
603 void ExpandChunksToIndex(int32 Index) TSAN_SAFE
604 {
605 check(Index >= 0);
606 int32 ChunkIndex = Index / NumAnnotationsPerChunk;
607 if (ChunkIndex >= Chunks.Num())
608 {
609 Chunks.AddZeroed(ChunkIndex + 1 - Chunks.Num());
610 }
611 check(ChunkIndex < Chunks.Num());
612 MaxAnnotations = Chunks.Num() * NumAnnotationsPerChunk;
613 }
614
618 TAnnotation& AllocateAnnotation(int32 Index) TSAN_SAFE
619 {
620 ExpandChunksToIndex(Index);
621
622 const int32 ChunkIndex = Index / NumAnnotationsPerChunk;
623 const int32 WithinChunkIndex = Index % NumAnnotationsPerChunk;
624
625 TAnnotationChunk& Chunk = Chunks[ChunkIndex];
626 if (!Chunk.Items)
627 {
628 Chunk.Items = new TAnnotation[NumAnnotationsPerChunk];
629 CurrentAllocatedMemory += NumAnnotationsPerChunk * sizeof(TAnnotation);
630 MaxAllocatedMemory = FMath::Max(CurrentAllocatedMemory, MaxAllocatedMemory);
631 }
632 if (Chunk.Items[WithinChunkIndex].IsDefault())
633 {
634 Chunk.Num++;
635 check(Chunk.Num <= NumAnnotationsPerChunk);
636 NumAnnotations++;
637 }
638
639 return Chunk.Items[WithinChunkIndex];
640 }
641
645 void FreeAnnotation(int32 Index) TSAN_SAFE
646 {
647 const int32 ChunkIndex = Index / NumAnnotationsPerChunk;
648 const int32 WithinChunkIndex = Index % NumAnnotationsPerChunk;
649
650 if (ChunkIndex >= Chunks.Num())
651 {
652 return;
653 }
654
655 TAnnotationChunk& Chunk = Chunks[ChunkIndex];
656 if (!Chunk.Items)
657 {
658 return;
659 }
660
661 if (Chunk.Items[WithinChunkIndex].IsDefault())
662 {
663 return;
664 }
665
666 Chunk.Items[WithinChunkIndex] = TAnnotation();
667 Chunk.Num--;
668 check(Chunk.Num >= 0);
669 if (Chunk.Num == 0)
670 {
671 delete[] Chunk.Items;
672 Chunk.Items = nullptr;
673 const uint32 ChunkMemory = NumAnnotationsPerChunk * sizeof(TAnnotation);
674 check(CurrentAllocatedMemory >= ChunkMemory);
675 CurrentAllocatedMemory -= ChunkMemory;
676 }
677 NumAnnotations--;
678 check(NumAnnotations >= 0);
679 }
680
684 void FreeAllAnnotations() TSAN_SAFE
685 {
686 for (TAnnotationChunk& Chunk : Chunks)
687 {
688 delete[] Chunk.Items;
689 }
690 Chunks.Empty();
691 NumAnnotations = 0;
692 MaxAnnotations = 0;
693 CurrentAllocatedMemory = 0;
694 MaxAllocatedMemory = 0;
695 }
