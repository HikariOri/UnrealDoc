65template< typename FGetBounds >
66uint32 FBVHCluster::Split( uint32 Offset, uint32 Num, const FGetBounds& GetBounds )
67{
68 FVector3f LeastCost( MAX_flt );
69 FIntVector LeastCostSplit( -1 );
70
71 uint32* Unsorted = &Indexes[ Offset ];
72 uint32* Sorted = &SortedTo[ Offset ];
73
74 for( int32 Dim = 0; Dim < 3; Dim++ )
75 {
76 Sort( Sorted, Unsorted, Num, Dim, GetBounds );
77
78 FBounds3f Bounds;
79
80 // SAH sweep forward
81 for( uint32 i = 0; i < Num; i++ )
82 {
83 Bounds += GetBounds( Sorted[i] );
84
85 FVector3f Size = Bounds.Max - Bounds.Min;
86 float HalfSurfaceArea = Size.X * Size.Y + Size.X * Size.Z + Size.Y * Size.Z;
87
88 int32 Count = i + 1;
89 //CostLeft[ Offset + i ] = HalfSurfaceArea * (float)Count;
90 CostLeft[ Offset + i ] = (float)Count * Size.SizeSquared();
91 }
92
93 Bounds = FBounds3f();
94
95 // SAH sweep back
96 for( int32 i = Num - 1; i >= 0; i-- )
97 {
98 Bounds += GetBounds( Sorted[i] );
99
100 FVector3f Size = Bounds.Max - Bounds.Min;
101 float HalfSurfaceArea = Size.X * Size.Y + Size.X * Size.Z + Size.Y * Size.Z;
102
103 int32 Count = Num - i - 1;
104 //CostRight[ Offset + i ] = HalfSurfaceArea * (float)Count;
105 CostRight[ Offset + i ] = (float)Count * Size.SizeSquared();
106 }
107
108 uint32 NumPartitions = FMath::DivideAndRoundUp( Num, MaxPartitionSize );
109
110 // Find least SAH cost
111 for( uint32 i = 0; i < Num - 1; i++ )
112 {
113 uint32 Num0 = i + 1;
114 uint32 Num1 = Num - Num0;
115
116 uint32 NumPartitions0 = FMath::DivideAndRoundUp( Num0, MaxPartitionSize );
117 uint32 NumPartitions1 = FMath::DivideAndRoundUp( Num1, MaxPartitionSize );
118
119 if( NumPartitions0 + NumPartitions1 == NumPartitions )
120 {
121 float Cost = CostLeft[ Offset + i ] + CostRight[ Offset + i + 1 ];
122 if( Cost < LeastCost[ Dim ] )
123 {
124 LeastCost[ Dim ] = Cost;
125 LeastCostSplit[ Dim ] = i;
126 }
127 }
128 }
129
130 Swap( Sorted, Unsorted );
131 }
132
133 uint32 BestDim = FMath::Min3Index( LeastCost[0], LeastCost[1], LeastCost[2] );
134 check( LeastCost[ BestDim ] > 0.0f );
135
136 Sort( Sorted, Unsorted, Num, BestDim, GetBounds );
137
138 // Even number of sorts means final sorted values are in the original array.
139
140 return LeastCostSplit[ BestDim ];
141}
