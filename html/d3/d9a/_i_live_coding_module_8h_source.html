35 // Enable by default tracks the value in the live coding settings.
36 // If enabled by default and the startup mode is one of the automatic settings, then
37 // live coding console will be started when the module is started.
38 virtual void EnableByDefault(bool bEnabled) = 0;
39 virtual bool IsEnabledByDefault() const = 0;
40
41 // The enabled for session state and started state are NOT independent states.
42 // If a request is made to enabled live coding (automatically or manually) and live coding
43 // hasn't already been started, then it is started. If and only if the console starts
44 // will both the started state and the enabled for session state will be true.
45 // If enable for session is set to false, the console isn't stopped so the started state
46 // will remain true.
47 //
48 // Started = false, Enabled = false => no request to enable has ever been made or it failed to start
49 // Started = true, Enabled = true => a request to enable was made and it started properly
50 // Started = true, Enabled = false => has been started but later disabled
51 // Started = false, Enabled = true => impossible state.
52 virtual bool HasStarted() const = 0;
53 virtual void EnableForSession(bool bEnabled) = 0;
