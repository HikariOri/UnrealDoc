343inline uint32 GetAdvanceAmount(EInstructionType InstructionType)
344{
345 switch (InstructionType)
346 {
347 case EInstructionType::SOP2: return 1u;
348 case EInstructionType::SOPK: return 1u;
349 case EInstructionType::SOP1: return 1u;
350 case EInstructionType::SOPC: return 1u;
351 case EInstructionType::SOPP: return 1u;
352 case EInstructionType::SMEM: return 2u;
353 case EInstructionType::VOP2: return 1u;
354 case EInstructionType::VOP1: return 1u;
355 case EInstructionType::VOP3: return 2u;
356 case EInstructionType::VOPC: return 1u;
357 case EInstructionType::VOP3P: return 2u;
358 case EInstructionType::VINTERP: return 1u;
359 case EInstructionType::LDSGDS: return 2u;
360 case EInstructionType::MUBUF: return 2u;
361 case EInstructionType::MTBUF: return 2u;
362 case EInstructionType::MIMG: return 2u;
363 case EInstructionType::EXPORT: return 2u;
364 case EInstructionType::FSG: return 2u;
365 default:
366 break;
367 }
368
369 return 0xffffffffu; // Force termination
370}

372inline bool IsSOP2(const uint32* Code, uint32& AdvanceAmount)
373{
374 const FInstSOP2& SOP2 = *reinterpret_cast<const FInstSOP2*>(Code);
375 const bool bMatch = (SOP2.ID == SOP2Header);
376 if (bMatch)
377 {
378 AdvanceAmount = GetAdvanceAmount(EInstructionType::SOP2);
379 if (SOP2.SSRC0 == Operand_Literal || SOP2.SSRC1 == Operand_Literal)
380 {
381 // Skip 32bit literal next in the instruction stream
382 AdvanceAmount++;
383 }
384 }
385 return bMatch;
386}
