157 void Construct( const FArguments& InArgs )
158 {
159 OnSocketChosen = InArgs._OnSocketChosen;
160 SceneComponent = InArgs._SceneComponent;
161
162 // Add in 'No Socket' selection if required
163 if (InArgs._ProvideNoSocketOption)
164 {
165 Sockets.Add( SSocketChooserPopup::FSocketInfo::Make(FComponentSocketDescription(NAME_None, EComponentSocketType::Invalid)) );
166 }
167
168 TextFilterPtr = MakeShareable(new FTextFilterExpressionEvaluator(ETextFilterExpressionEvaluatorMode::BasicString));
169
170 // Build set of sockets
171 if (SceneComponent != NULL)
172 {
173 TArray<FComponentSocketDescription> Descriptions;
174 SceneComponent->QuerySupportedSockets(/*out*/ Descriptions);
175 for (auto DescriptionIt = Descriptions.CreateConstIterator(); DescriptionIt; ++DescriptionIt)
176 {
177 Sockets.Add( SSocketChooserPopup::FSocketInfo::Make(*DescriptionIt) );
178 }
179
180 FilteredSockets.Append(Sockets);
181 }
182
183 // Then make widget
184 this->ChildSlot
185 [
186 SNew(SBorder)
187 .BorderImage(FAppStyle::GetBrush(TEXT("Menu.Background")))
188 .Padding(5.f)
189 .Content()
190 [
191 SNew(SVerticalBox)
192 +SVerticalBox::Slot()
193 .AutoHeight()
194 .Padding(0.0f, 1.0f)
195 [
196 SNew(STextBlock)
197 .Font( FAppStyle::GetFontStyle(TEXT("SocketChooser.TitleFont")) )
198 .Text( NSLOCTEXT("SocketChooser", "ChooseSocketOrBoneLabel", "Choose Socket or Bone") )
199 ]
200 +SVerticalBox::Slot()
201 .AutoHeight()
202 .Padding(0.0f, 1.0f)
203 [
204 SAssignNew(SearchBox, SSearchBox)
205 .OnTextChanged(this, &SSocketChooserPopup::HandleSearchTextChanged)
206 ]
207 +SVerticalBox::Slot()
208 .AutoHeight()
209 .MaxHeight(512.f)
210 [
211 SNew(SBox)

227 virtual void Tick( const FGeometry& AllottedGeometry, const double InCurrentTime, const float InDeltaTime ) override
228 {
229 // Make sure that my window is on-screen
230 TSharedPtr<SWindow> Window = CheckAndGetWindowPtr();
231 if(Window.IsValid())
232 {
233 FVector2f WindowLocation = Window->IsMorphing() ? Window->GetMorphTargetPosition() : Window->GetPositionInScreen();
234 FVector2f WindowSize = Window->GetDesiredSize();
235 FSlateRect Anchor(WindowLocation.X, WindowLocation.Y, WindowLocation.X, WindowLocation.Y);
236 WindowLocation = FSlateApplication::Get().CalculatePopupWindowPosition( Anchor, WindowSize, false );
237
238 // Update the window's position!
239 if( Window->IsMorphing() )
240 {
241 if( WindowLocation != Window->GetMorphTargetPosition() )
242 {
243 Window->UpdateMorphTargetShape( FSlateRect(WindowLocation.X, WindowLocation.Y, WindowLocation.X + WindowSize.X, WindowLocation.Y + WindowSize.Y) );
244 }
245 }
246 else
247 {
248 if( WindowLocation != Window->GetPositionInScreen() )
249 {
250 Window->MoveWindowTo(WindowLocation);
251 }
252 }
253 }
254 }

256 void HandleSearchTextChanged(const FText& InText)
257 {
258 TextFilterPtr->SetFilterText(InText);
259
260 FilteredSockets.Reset();
261
262 if (InText.IsEmpty())
263 {
264 FilteredSockets.Append(Sockets);
265 }
266 else
267 {
268 for (TSharedPtr<FSocketInfo>& SocketInfo : Sockets)
269 {
270 if (TextFilterPtr->TestTextFilter(SocketInfo->FilterContext))
271 {
272 FilteredSockets.Add(SocketInfo);
273 }
274 }
275 }
276
277 SocketListView->RequestListRefresh();
278 }

280 virtual FReply OnKeyDown(const FGeometry& MyGeometry, const FKeyEvent& InKeyEvent) override
281 {
282 if(InKeyEvent.GetKey() == EKeys::Enter)
283 {
284 TArray<TSharedPtr<FSocketInfo>> SelectedItems = SocketListView->GetSelectedItems();
285 if(SelectedItems.Num() > 0)
286 {
287 ChooseSocket(SelectedItems[0]);
288 return FReply::Handled();
289 }
290 }
291
292 return FReply::Unhandled();
293 }
