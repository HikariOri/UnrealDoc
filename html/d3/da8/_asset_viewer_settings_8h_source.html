186 void LoadEnvironmentMap()
187 {
188 if (EnvironmentCubeMap == nullptr)
189 {
190 if (!EnvironmentCubeMapPath.IsEmpty())
191 {
192 // Load cube map from stored path
193 UObject* LoadedObject = LoadObject<UObject>(nullptr, *EnvironmentCubeMapPath);
194 while (UObjectRedirector* Redirector = Cast<UObjectRedirector>(LoadedObject))
195 {
196 LoadedObject = Redirector->DestinationObject;
197 }
198
199 EnvironmentCubeMap = Cast<UTextureCube>(LoadedObject);
200 }
201 }

204 void SetShowFlags(FEngineShowFlags& ShowFlags) const
205 {
206 // for reasons I haven't been able to discern exactly, this must be called prior to EnableAdvancedFeatures()
207 // to prevent a crash in the renderer caused by unallocated or missing resources
208 ShowFlags.DisableAdvancedFeatures();
209
210 if (bPostProcessingEnabled)
211 {
212 ShowFlags.EnableAdvancedFeatures();
213 ShowFlags.SetBloom(true); // bloom not included in EnableAdvancedFeatures() for thumbnails (see func comments)
214 }
215 else
216 {
217 ShowFlags.DisableAdvancedFeatures();
218 ShowFlags.SetBloom(false);
219 }
220
221 ShowFlags.SetTonemapper(bEnableToneMapping);
222 ShowFlags.SetMeshEdges(bShowMeshEdges);
223 }
