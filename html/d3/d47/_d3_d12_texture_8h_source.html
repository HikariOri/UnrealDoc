187inline FD3D12RenderTargetView* FD3D12Texture::GetRenderTargetView(int32 MipIndex, int32 ArraySliceIndex) const
188{
189 int32 ArrayIndex = MipIndex;
190
191 if (bCreatedRTVsPerSlice)
192 {
193 check(ArraySliceIndex >= 0);
194 ArrayIndex = MipIndex * RTVArraySizePerMip + ArraySliceIndex;
195 check(ArrayIndex < RenderTargetViews.Num());
196 }
197 else
198 {
199 // Catch attempts to use a specific slice without having created the texture to support it
200 check(ArraySliceIndex == -1 || ArraySliceIndex == 0);
201 }
202
203 if (ArrayIndex < RenderTargetViews.Num())
204 {
205 return RenderTargetViews[ArrayIndex].Get();
206 }
207 return 0;
208}
