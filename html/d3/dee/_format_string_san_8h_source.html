130 static constexpr FResult Check(bool bInsideFormatSpec, int CurArgPos, const CharType* Fmt)
131 {
132 using ArgPointeeType = std::remove_pointer_t<Arg>;
133
134 if (!bInsideFormatSpec)
135 {
136 while (*Fmt != (CharType)'\0' && *Fmt != (CharType)'%')
137 {
138 ++Fmt;
139 }
140
141 if (Fmt[0] == (CharType)'\0')
142 {
143 return { EFormatStringSanStatus::NotEnoughSpecifiers, CurArgPos };
144 }
145
146 if (*Fmt == (CharType)'%')
147 {
148 ++Fmt;
149 }
150 }
151
152 if (*Fmt == (CharType)'\0')
153 {
154 if (Fmt[-1] == (CharType)'%' || bInsideFormatSpec) // nb: Fmt[-1] is safe here because zero-length fmt strings are addressed above
155 {
156 return {EFormatStringSanStatus::IncompleteFormatSpecifierOrUnescapedPercent, CurArgPos};
157 }
158 else
159 {
160 return {EFormatStringSanStatus::Ok, 0};
161 }
162 }
163
164 while (*Fmt == (CharType)'+' || *Fmt == (CharType)'#' || *Fmt == (CharType)' ' || *Fmt == (CharType)'0')
165 {
166 ++Fmt;
167 }
168
169 if (*Fmt == (CharType)'-')
170 {
171 ++Fmt;
172 }
173 if (*Fmt == (CharType)'*')
174 {
175 return HandleDynamicLengthSpecifier(CurArgPos, Fmt);
176 }
177 else if (CharIsDigit(*Fmt))
178 {
179 Fmt = SkipInteger(Fmt);
180 }
181
182 if (*Fmt == (CharType)'.')
183 {
184 ++Fmt;
185 }
186 if (*Fmt == (CharType)'*')
187 {
188 return HandleDynamicLengthSpecifier(CurArgPos, Fmt);
189 }
190 else if (CharIsDigit(*Fmt))
191 {
192 Fmt = SkipInteger(Fmt);
193 }
194
195 if (Fmt[0] == (CharType)'l' && Fmt[1] == (CharType)'s')
196 {
197 if constexpr (TIsTString_V<Arg>)
198 {
199 return {EFormatStringSanStatus::LSNeedsDereferencedWideString, CurArgPos};
200 }
201 else if constexpr (TIsCharType_V<Arg>)
202 {
203 return {EFormatStringSanStatus::LSNeedsPtrButGotChar, CurArgPos};
204 }
205 else if constexpr (!std::is_pointer_v<Arg> || !TIsCharType_V<ArgPointeeType>)
206 {
207 return {EFormatStringSanStatus::LSNeedsWideCharPtrArg, CurArgPos};
208 }
209 else if constexpr (sizeof(ArgPointeeType) != sizeof(WIDECHAR))
210 {
211 return {(sizeof(CharType) == 1) ? EFormatStringSanStatus::LSNeedsWideCharPtrArgButGotNarrowOnNarrowString : EFormatStringSanStatus::LSNeedsWideCharPtrArgButGotNarrowOnWideString, CurArgPos};
212 }
213 else
214 {
215 return TCheckFormatString<CharType, Args...>::Check(false, CurArgPos + 1, Fmt + 2);
216 }
217 }
218 else if (Fmt[0] == (CharType)'h' && Fmt[1] == (CharType)'s')
219 {
220 if constexpr (TIsTString_V<Arg>)
221 {
222 return {EFormatStringSanStatus::HSNeedsDereferencedNarrowString, CurArgPos};
223 }
224 else if constexpr (TIsCharType_V<Arg>)
225 {
226 return {EFormatStringSanStatus::HSNeedsPtrButGotChar, CurArgPos};
227 }
228 else if constexpr (!std::is_pointer_v<Arg> || !TIsCharType_V<ArgPointeeType>)
229 {
230 return {EFormatStringSanStatus::HSNeedsNarrowCharPtrArg, CurArgPos};
231 }
232 else if constexpr (sizeof(ArgPointeeType) != sizeof(ANSICHAR))
233 {
234 return {(sizeof(CharType) == 1) ? EFormatStringSanStatus::HSNeedsNarrowCharPtrArgButGotWideOnNarrowString : EFormatStringSanStatus::HSNeedsNarrowCharPtrArgButGotWideOnWideString, CurArgPos};
235 }
236 else
237 {
238 return TCheckFormatString<CharType, Args...>::Check(false, CurArgPos + 1, Fmt + 2);
239 }
240 }
241 else if (Fmt[0] == (CharType)'s')
242 {
243 if constexpr (TIsTString_V<Arg>)
244 {
245 return {(sizeof(CharType) == 1) ? EFormatStringSanStatus::SNeedsDereferencedNarrowString : EFormatStringSanStatus::SNeedsDereferencedWideString, CurArgPos};
246 }
247 else if constexpr (TIsCharType_V<Arg>)
248 {
249 return {EFormatStringSanStatus::SNeedsPtrButGotChar, CurArgPos};
250 }
251 else if constexpr (!std::is_pointer_v<Arg> || !TIsCharType_V<ArgPointeeType>)
252 {
253 return {(sizeof(CharType) == 1) ? EFormatStringSanStatus::SNeedsNarrowCharPtrArg : EFormatStringSanStatus::SNeedsWideCharPtrArg, CurArgPos};
254 }

258 }
259 else

262 }
263 }
264 else if (Fmt[0] == (CharType)'S')
265 {
266 if constexpr (TIsTString_V<Arg>)
267 {
268 return {(sizeof(CharType) == 1) ? EFormatStringSanStatus::CapitalSNeedsDereferencedWideString : EFormatStringSanStatus::CapitalSNeedsDereferencedNarrowString, CurArgPos};
269 }
270 else if constexpr (TIsCharType_V<Arg>)
271 {
272 return {EFormatStringSanStatus::CapitalSNeedsPtrButGotChar, CurArgPos};
273 }
274 else if constexpr (!std::is_pointer_v<Arg> || !TIsCharType_V<ArgPointeeType>)
275 {
276 return {(sizeof(CharType) == 1) ? EFormatStringSanStatus::CapitalSNeedsWideCharPtrArg : EFormatStringSanStatus::CapitalSNeedsNarrowCharPtrArg, CurArgPos};
277 }
278 else if constexpr (sizeof(ArgPointeeType) == sizeof(CharType))

281 }

285 }
286 }

289 {

292 case (CharType)'c':

303 case (CharType)'X':

305 case (CharType)'u':

307 {
308 return {EFormatStringSanStatus::DNeedsIntegerArg, CurArgPos};
309 }

311 {
312 return {EFormatStringSanStatus::DNeedsIntegerArg, CurArgPos};

320 {

322 }

324 {

328 {
329 return TCheckFormatString<CharType, Args...>::Check(false, CurArgPos + 1, Fmt + 1);

331 case (CharType)'p':

333 {

335 }
336 else
337 {
338 return TCheckFormatString<CharType, Args...>::Check(false, CurArgPos + 1, Fmt + 1);
339 }
340 case (CharType)'I':
341 if (!(Fmt[1] == (CharType)'6' && Fmt[2] == (CharType)'4'))
342 {
343 return {EFormatStringSanStatus::I64BadSpec, CurArgPos};
344 }
345 else if (!CharIsIntegerFormatSpecifier(Fmt[3]))
346 {
347 return {EFormatStringSanStatus::I64BadSpec, CurArgPos};
348 }
349 else if constexpr (!(TIsIntegralEnum_V<Arg> || std::is_integral_v<Arg>))

352 }
353 else
354 {
355 return TCheckFormatString<CharType, Args...>::Check(false, CurArgPos + 1, Fmt + 4);
356 }
357
358 case (CharType)'l':
359 if (CharIsIntegerFormatSpecifier(Fmt[1]))
360 {
361 if constexpr (!(TIsIntegralEnum_V<Arg> || std::is_integral_v<Arg>))
362 {
363 return {EFormatStringSanStatus::LNeedsIntegerArg, CurArgPos};
364 }
365 else
366 {
367 return TCheckFormatString<CharType, Args...>::Check(false, CurArgPos + 1, Fmt + 2);
368 }
369 }
370 else if (Fmt[1] == (CharType)'f')
371 {

380 }
381 else if (Fmt[1] != (CharType)'l')
382 {
383 return {EFormatStringSanStatus::InvalidFormatSpec, CurArgPos};
384 }
385 else if (!CharIsIntegerFormatSpecifier(Fmt[2]))
386 {
387 return {EFormatStringSanStatus::LLNeedsIntegerSpec, CurArgPos};
388 }
389 else if constexpr (!(TIsIntegralEnum_V<Arg> || std::is_integral_v<Arg> || std::is_pointer_v<Arg>))
390 {

392 }
393 else

395 return TCheckFormatString<CharType, Args...>::Check(false, CurArgPos + 1, Fmt + 3);
396 }
397
398 case (CharType)'h':
399 if (CharIsIntegerFormatSpecifier(Fmt[1]))
400 {
401 if constexpr (!(TIsIntegralEnum_V<Arg> || std::is_integral_v<Arg>))
402 {
403 return {EFormatStringSanStatus::HNeedsIntegerArg, CurArgPos};
404 }
405 return TCheckFormatString<CharType, Args...>::Check(false, CurArgPos + 1, Fmt + 2);
406 }
407 else if (Fmt[1] != (CharType)'h')
408 {
409 return {EFormatStringSanStatus::InvalidFormatSpec, CurArgPos};
410 }
411 else if (!CharIsIntegerFormatSpecifier(Fmt[2]))
412 {
413 return {EFormatStringSanStatus::HHNeedsIntegerSpec, CurArgPos};

415 else if constexpr (!(TIsIntegralEnum_V<Arg> || std::is_integral_v<Arg>))

419 else

425 case (CharType)'e':
426 case (CharType)'g':

445 template <typename CharType>
446 struct TCheckFormatString<CharType>
447 {
448 static constexpr FResult Check(bool bInsideFormatSpec, int CurArgPos, const CharType* Fmt)
449 {
450 if (bInsideFormatSpec)
451 {
452 return {EFormatStringSanStatus::IncompleteFormatSpecifierOrUnescapedPercent, CurArgPos};
453 }
454
455 while (*Fmt)
456 {
457 if (Fmt[0] != (CharType)'%')
458 {
459 ++Fmt;
460 continue;
461 }
462
463 if (Fmt[1] == (CharType)'%')
464 {
465 Fmt += 2;
466 continue;
467 }
468
469 if (Fmt[1] == (CharType)'\0')
470 {
471 return {EFormatStringSanStatus::IncompleteFormatSpecifierOrUnescapedPercent, CurArgPos};
472 }
473
474 return {EFormatStringSanStatus::NotEnoughArguments, CurArgPos};
475 }
476
477 return {EFormatStringSanStatus::Ok, 0};
478 }
