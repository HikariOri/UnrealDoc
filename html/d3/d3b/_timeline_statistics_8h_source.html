96 template<
97 typename TimelineType,
98 typename BucketMappingFunc,
99 typename BucketKeyType
100 >
101 static void CreateAggregation(
102 const TArray<const TimelineType*>& Timelines,
103 BucketMappingFunc BucketMapper,
104 double IntervalStart,
105 double IntervalEnd,
106 TSharedPtr<FCancellationToken> CancellationToken,
107 TAggregationResult<BucketKeyType, FAggregatedTimingStats>& Result)
108 {
109 TAggregationResult<BucketKeyType, FInternalAggregationEntry> InternalResult;
110
111 // Compute instance count and total/min/max inclusive/exclusive times for each timer.
112 for (const TimelineType* Timeline : Timelines)
113 {
114 if (CancellationToken.IsValid() && CancellationToken->ShouldCancel())
115 {
116 return;
117 }
118 ProcessTimeline(Timeline, BucketMapper, UpdateTotalMinMaxTimerAggregationStats, IntervalStart, IntervalEnd, InternalResult);
119 }
120
121 // Now, as we know min/max inclusive/exclusive times for each timer, we can compute histogram and median values.
122 const bool bComputeMedian = true;
123 if (bComputeMedian)
124 {
125 // Update bucket size (DT) for computing histogram.
126 for (auto& KV : InternalResult.GetMap())
127 {
128 InitHistogram(KV.Value, InternalResult.GetAllocator());
129 }
130
131 // Compute histogram.
132 for (const TimelineType* Timeline : Timelines)
133 {
134 if (CancellationToken.IsValid() && CancellationToken->ShouldCancel())
135 {
136 return;
137 }
138 ProcessTimeline(Timeline, BucketMapper, UpdateHistogram, IntervalStart, IntervalEnd, InternalResult);
139 }
140 }
141
142 // Compute average and median inclusive/exclusive times.
143 for (auto& KV : InternalResult.GetMap())
144 {
145 PostProcessTimerStats(KV.Value, bComputeMedian);
146 Result.GetMap().Add(KV.Key, KV.Value.Stats);
147 }
148 };

150 template<
151 typename TimelineType,
152 typename BucketMappingFunc,
153 typename BucketKeyType
154 >
155 static void CreateFrameStatsAggregation(
156 const TArray<const TimelineType*>& Timelines,
157 BucketMappingFunc BucketMapper,
158 const TArray<FFrameData>& Frames,
159 TSharedPtr<FCancellationToken> CancellationToken,
160 TAggregationResult<BucketKeyType, FAggregatedTimingStats>& Result,
161 double RatioOfThreadsToUse)
162 {
163 int32 FramesNum = Frames.Num();
164 if (FramesNum == 0)
165 {
166 return;
167 }
168
169 TAggregationResult<BucketKeyType, FInternalFrameAggregationEntry> GlobalResult;
170 TAggregationResult<BucketKeyType, FAggregatedTimingStats> InitialTimerMap;
171
172 double StartTime = Frames[0].StartTime;
173 double EndTime = Frames[FramesNum - 1].EndTime;
174
175 // Gather the keys for all timers so we have a stable map (no insertions necessary) for the next step.
176 // This is to guarantee that GlobalResult and FrameResult have the same iteration order.
177 for (const TimelineType* Timeline : Timelines)
178 {
179 GatherKeysFromTimeline(Timeline, BucketMapper, StartTime, EndTime, InitialTimerMap);
180 }
181
182 // For very large numbers of timers or frames, an out of memory is possible so don't compute the median.
183 constexpr int64 MaxSize = 5 * 100 * 1000 * 1000;
184 bool bComputeMedian = InitialTimerMap.GetMap().Num() * (int64)FramesNum < MaxSize;
185
186 for (auto& KV : InitialTimerMap.GetMap())
187 {
188 FInternalFrameAggregationEntry Entry;
189 if (bComputeMedian)
190 {
191 Entry.FrameInclusiveTimes.AddUninitialized(Frames.Num());
192 Entry.FrameExclusiveTimes.AddUninitialized(Frames.Num());
193 }
194 Entry.Inner.Stats = KV.Value;
195 GlobalResult.GetMap().Add(KV.Key, MoveTemp(Entry));
196 }
197
198 TQueue<TSharedPtr<TAggregationResult<BucketKeyType, FAggregatedTimingStats>>, EQueueMode::Mpsc> FrameResultsQueue;
199
200 int32 NumTasks = FMath::Max(1, (int32)((float)GThreadPool->GetNumThreads() * RatioOfThreadsToUse));
201 NumTasks = FMath::Min((int32)NumTasks, Frames.Num());
202 int32 FramesPerTask = Frames.Num() / NumTasks;
203 int32 ExtraFrameTasks = Frames.Num() - NumTasks * FramesPerTask;
204
205 int32 StartIndex = 0;
206 int32 EndIndex = FramesPerTask + (ExtraFrameTasks > 0 ? 1 : 0);
207 --ExtraFrameTasks;
208
209 TArray<UE::Tasks::TTask<void>> AsyncTasks;
210 for (int32 Index = 0; Index < NumTasks; ++Index)
211 {
212 AsyncTasks.Add(UE::Tasks::Launch(UE_SOURCE_LOCATION,
213 [StartIndex, EndIndex, &Timelines, bComputeMedian, BucketMapper, &Frames, &FrameResultsQueue, &InitialTimerMap, &GlobalResult, CancellationToken]()
214 {
215 if (CancellationToken.IsValid() && CancellationToken->ShouldCancel())
216 {
217 return;
218 }
219
220 TSharedPtr<TAggregationResult<BucketKeyType, FAggregatedTimingStats>> TaskResult = MakeShared<TAggregationResult<BucketKeyType, FAggregatedTimingStats>>(InitialTimerMap.GetMap());
221 TAggregationResult<BucketKeyType, FAggregatedTimingStats> FrameResult(InitialTimerMap.GetMap());
222
223 for (int FrameIndex = StartIndex; FrameIndex < EndIndex; ++FrameIndex)
224 {
225 if (CancellationToken.IsValid() && CancellationToken->ShouldCancel())
226 {
227 return;
228 }
229
230 // Compute instance count and total/min/max inclusive/exclusive times for each timer.
231 for (const TimelineType* Timeline : Timelines)
232 {
233 ProcessTimelineForFrameStats(Timeline, BucketMapper, UpdateTotalMinMaxTimerStats, Frames[FrameIndex].StartTime, Frames[FrameIndex].EndTime, FrameResult);
234 }
235
236 typename TSparseMap<BucketKeyType, FAggregatedTimingStats>::TIterator FrameResultIterator(FrameResult.GetMap());
237 typename TSparseMap<BucketKeyType, FAggregatedTimingStats>::TIterator TaskResultIterator(TaskResult->GetMap());
238 typename TSparseMap<BucketKeyType, FInternalFrameAggregationEntry>::TIterator GlobalResultIterator(GlobalResult.GetMap());
239 while (FrameResultIterator)
240 {
241 check(FrameResultIterator->Key == TaskResultIterator->Key);
242 check(GlobalResultIterator->Key == TaskResultIterator->Key);
243
244 FAggregatedTimingStats& FrameStats = FrameResultIterator->Value;
245 FAggregatedTimingStats& TaskResultStats = TaskResultIterator->Value;
246
247 TaskResultStats.InstanceCount += FrameStats.InstanceCount;
248
249 TaskResultStats.TotalInclusiveTime += FrameStats.TotalInclusiveTime;
250 TaskResultStats.MinInclusiveTime = FMath::Min(TaskResultStats.MinInclusiveTime, FrameStats.TotalInclusiveTime);
251 TaskResultStats.MaxInclusiveTime = FMath::Max(TaskResultStats.MaxInclusiveTime, FrameStats.TotalInclusiveTime);
252
253 TaskResultStats.TotalExclusiveTime += FrameStats.TotalExclusiveTime;
254 TaskResultStats.MinExclusiveTime = FMath::Min(TaskResultStats.MinExclusiveTime, FrameStats.TotalExclusiveTime);
255 TaskResultStats.MaxExclusiveTime = FMath::Max(TaskResultStats.MaxExclusiveTime, FrameStats.TotalExclusiveTime);
256
257 if (bComputeMedian)
258 {
259 FInternalFrameAggregationEntry& GlobalResultValue = GlobalResultIterator->Value;
260 GlobalResultValue.FrameInclusiveTimes[FrameIndex] = FrameStats.TotalInclusiveTime;
261 GlobalResultValue.FrameExclusiveTimes[FrameIndex] = FrameStats.TotalExclusiveTime;
262 }
263
264 // Reset the per frame stats.
265 FrameStats = FAggregatedTimingStats();
266
267 ++FrameResultIterator;
268 ++TaskResultIterator;
269 ++GlobalResultIterator;
270 }
271 }
272
273 FrameResultsQueue.Enqueue(MoveTemp(TaskResult));
274 }));
275
276 StartIndex = EndIndex;
277 EndIndex = StartIndex + FramesPerTask + (ExtraFrameTasks > 0 ? 1 : 0);
278 --ExtraFrameTasks;
279 }
280
281 check(StartIndex == Frames.Num());
282
283 int ProcessedResults = 0;
284 while (ProcessedResults < NumTasks)
285 {
286 if (CancellationToken.IsValid() && CancellationToken->ShouldCancel())
287 {
288 UE::Tasks::Wait(AsyncTasks);
289 return;
290 }
291
292 if (FrameResultsQueue.IsEmpty())
293 {
294 FPlatformProcess::SleepNoStats(0.1f);
295 continue;
296 }
297
298 TSharedPtr<TAggregationResult<BucketKeyType, FAggregatedTimingStats>> TaskResult;
299 ensure(FrameResultsQueue.Dequeue(TaskResult));
300 ++ProcessedResults;
301
302 typename TSparseMap<BucketKeyType, FAggregatedTimingStats>::TIterator TaskResultIterator(TaskResult->GetMap());
303 typename TSparseMap<BucketKeyType, FInternalFrameAggregationEntry>::TIterator ResultIterator(GlobalResult.GetMap());
304 while (ResultIterator)
305 {
306 check(ResultIterator->Key == TaskResultIterator->Key);
307
308 FAggregatedTimingStats& TaskStats = TaskResultIterator->Value;
309 FAggregatedTimingStats& ResultStats = ResultIterator->Value.Inner.Stats;
310
311 ResultStats.InstanceCount += TaskStats.InstanceCount;
312
313 ResultStats.TotalInclusiveTime += TaskStats.TotalInclusiveTime;
314 ResultStats.MinInclusiveTime = FMath::Min(ResultStats.MinInclusiveTime, TaskStats.MinInclusiveTime);
315 ResultStats.MaxInclusiveTime = FMath::Max(ResultStats.MaxInclusiveTime, TaskStats.MaxInclusiveTime);
316
317 ResultStats.TotalExclusiveTime += TaskStats.TotalExclusiveTime;
318 ResultStats.MinExclusiveTime = FMath::Min(ResultStats.MinExclusiveTime, TaskStats.MinExclusiveTime);
319 ResultStats.MaxExclusiveTime = FMath::Max(ResultStats.MaxExclusiveTime, TaskStats.MaxExclusiveTime);
320
321 ++TaskResultIterator;
322 ++ResultIterator;
323 }
324 }
325
326 if (bComputeMedian)
327 {
328 // Compute the median inclusive and exclusive time.
329 for (auto& KV : GlobalResult.GetMap())
330 {
331 FInternalFrameAggregationEntry& FrameEntry = KV.Value;
332 FInternalAggregationEntry& AggregationEntry = FrameEntry.Inner;
333
334 if (AggregationEntry.Stats.InstanceCount == 0)
335 {
336 // Some timers might have 0 instances because the gathering phase also searched between frames.
337 continue;
338 }
339
340 InitHistogram(AggregationEntry, GlobalResult.GetAllocator());
341
342 for (int32 Index = 0; Index < FramesNum; ++Index)
343 {
344 UpdateHistogram(AggregationEntry, FrameEntry.FrameInclusiveTimes[Index], FrameEntry.FrameExclusiveTimes[Index]);
345 }
346
347 ComputeMedianFromHistogram(AggregationEntry);
348 }
349 }
350
351 for (auto& KV : GlobalResult.GetMap())
352 {
353 FAggregatedTimingStats& Stats = KV.Value.Inner.Stats;
354
355 Stats.AverageInclusiveTime = Stats.TotalInclusiveTime / FramesNum;
356 Stats.AverageExclusiveTime = Stats.TotalExclusiveTime / FramesNum;
357 Stats.AverageInstanceCount = static_cast<double>(Stats.InstanceCount) / FramesNum;
358
359 Result.GetMap().Add(KV.Key, Stats);
360 }
361 }

434 double EventInclusiveTime = Time - StackEntry.StartTime;
435 check(EventInclusiveTime >= 0.0);
436 double EventExclusiveTime = StackEntry.ExclusiveTime;
437 check(EventExclusiveTime >= 0.0 && EventExclusiveTime <= EventInclusiveTime);
438 Stack.Pop(EAllowShrinking::No);
