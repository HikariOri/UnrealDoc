69 template <typename DestEncoding, typename SourceEncoding>
70 static constexpr bool CanConvertCodepoint(SourceEncoding Codepoint)
71 {
72 // It is assumed that the incoming codepoint is already valid and we're only testing if it can be converted to DestEncoding.
73
74 static_assert(TIsCharType<SourceEncoding>::Value, "Source encoding is not a char type");
75 static_assert(TIsCharType<DestEncoding >::Value, "Destination encoding is not a char type");
76
77 // This is only defined for fixed-width encodings, because codepoints cannot be represented in a single variable-width code unit.
78 static_assert(TIsFixedWidthCharEncoding_V<SourceEncoding>, "Source encoding is not fixed-width");
79
80 if constexpr (TIsCharEncodingSimplyConvertibleTo_V<SourceEncoding, DestEncoding>)
81 {
82 // Simple conversions mean conversion is always possible
83 return true;
84 }
85 else if constexpr (!TIsFixedWidthCharEncoding_V<DestEncoding>)
86 {
87 // Converting all codepoints to a variable-width encoding should always be possible
88 return true;
89 }
90 else if constexpr (std::is_same_v<DestEncoding, ANSICHAR>)
91 {
92 return (uint32)Codepoint <= 0x7F;
93 }
94 else
95 {
96 // The logic above should hopefully mean this branch is only taken for UTF32CHAR->UCS2CHAR.
97 // There's a variety of '16-bit' char types between platforms though, so let's just test sizes.
98 static_assert(sizeof(SourceEncoding) == 4 && sizeof(DestEncoding) == 2, "Unimplemented conversion");
99
100 // Can't encode more than 16-bit in UCS-2
101 return (uint32)Codepoint <= 0xFFFF;
102 }
103 }

134 template <typename SourceEncoding, typename DestEncoding>
135 static inline DestEncoding* Convert(DestEncoding* Dest, int32 DestSize, const SourceEncoding* Src)
136 {
137 if constexpr (TIsCharEncodingSimplyConvertibleTo_V<SourceEncoding, DestEncoding>)
138 {
139 for (;;)
140 {
141 if (DestSize == 0)
142 {
143 return nullptr;
144 }
145
146 if (!(*Dest++ = (DestEncoding)*Src++))
147 {
148 return Dest;
149 }
150
151 --DestSize;
152 }
153 }
154 else if constexpr (TIsFixedWidthCharEncoding_V<SourceEncoding> && TIsFixedWidthCharEncoding_V<DestEncoding>)
155 {
156 DestEncoding* DestCopy = Dest;
157 const SourceEncoding* SrcCopy = Src;
158 int32 DestSizeCopy = DestSize;
159
160 bool bInvalidChars = false;
161 for (;;)
162 {
163 if (DestSize == 0)
164 {
165 Dest = nullptr;
166 break;
167 }
168
169 SourceEncoding SrcCh = *Src++;
170 *Dest++ = (DestEncoding)SrcCh;
171 if (!SrcCh)
172 {
173 break;
174 }
175 bInvalidChars |= !CanConvertCodepoint<DestEncoding>(SrcCh);
176
177 --DestSize;
178 }
179
180 if (bInvalidChars)
181 {
182 for (;;)
183 {
184 if (DestSizeCopy == 0)
185 {
186 break;
187 }
188
189 SourceEncoding SrcCh = *SrcCopy++;
190 if (!SrcCh)
191 {
192 break;
193 }
194 if (!CanConvertCodepoint<DestEncoding>(SrcCh))
195 {
196 *DestCopy = UNICODE_BOGUS_CHAR_CODEPOINT;
197 }
198 ++DestCopy;
199
200 --DestSizeCopy;
201 }
202 }
203
204 return Dest;
205 }
206 else
207 {
208 DestEncoding* Result = UE::Core::Private::Convert(Dest, DestSize, Src);
209 if (Result)
210 {
211 *Result++ = (DestEncoding)0;
212 }
213 return Result;
214 }

228 template <typename SourceEncoding, typename DestEncoding>
229 static inline DestEncoding* Convert(DestEncoding* Dest, int32 DestSize, const SourceEncoding* Src, int32 SrcSize)
230 {
231 if constexpr (TIsCharEncodingCompatibleWith_V<SourceEncoding, DestEncoding>)
232 {
233 if (DestSize < SrcSize)
234 {
235 return nullptr;
236 }
237
238 return (DestEncoding*)Memcpy(Dest, Src, SrcSize * sizeof(SourceEncoding)) + SrcSize;
239 }
240 else if constexpr (TIsCharEncodingSimplyConvertibleTo_V<SourceEncoding, DestEncoding>)
241 {
242 const int32 Size = DestSize <= SrcSize ? DestSize : SrcSize;
243 for (int I = 0; I < Size; ++I)
244 {
245 SourceEncoding SrcCh = Src[I];
246 Dest[I] = (DestEncoding)SrcCh;
247 }
248
249 return DestSize < SrcSize ? nullptr : Dest + Size;
250 }
251 else if constexpr (TIsFixedWidthCharEncoding_V<SourceEncoding> && TIsFixedWidthCharEncoding_V<DestEncoding>)
252 {
253 const int32 Size = DestSize <= SrcSize ? DestSize : SrcSize;
254 bool bInvalidChars = false;
255 for (int I = 0; I < Size; ++I)
256 {
257 SourceEncoding SrcCh = Src[I];
258 Dest[I] = (DestEncoding)SrcCh;
259 bInvalidChars |= !CanConvertCodepoint<DestEncoding>(SrcCh);
260 }
261
262 if (bInvalidChars)
263 {
264 for (int I = 0; I < Size; ++I)
265 {
266 if (!CanConvertCodepoint<DestEncoding>(Src[I]))
267 {
268 Dest[I] = UNICODE_BOGUS_CHAR_CODEPOINT;
269 }
270 }
271 }
272
273 return DestSize < SrcSize ? nullptr : Dest + Size;
274 }
275 else
276 {
277 return UE::Core::Private::Convert(Dest, DestSize, Src, SrcSize);
278 }
