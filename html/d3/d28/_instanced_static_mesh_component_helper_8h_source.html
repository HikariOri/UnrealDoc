52{

93template<class T>
94void FInstancedStaticMeshComponentHelper::GetNavigationData(const T& Component, FNavigationRelevantData& Data, const FNavDataPerInstanceTransformDelegate& Delegate)
95{
96 FPrimitiveComponentHelper::AddNavigationModifier(Component, Data);
97
98 // Navigation data will get refreshed once async compilation finishes
99 UStaticMesh* StaticMesh = Component.GetStaticMesh();
100 if (StaticMesh && !StaticMesh->IsCompiling() && StaticMesh->GetNavCollision())
101 {
102 UNavCollisionBase* NavCollision = StaticMesh->GetNavCollision();
103 if (Component.ShouldExportAsObstacle(*NavCollision))
104 {
105 Data.Modifiers.MarkAsPerInstanceModifier();
106 NavCollision->GetNavigationModifier(Data.Modifiers, FTransform::Identity);
107
108 // Hook per instance transform delegate
109 Data.NavDataPerInstanceTransformDelegate = Delegate;
110 }
111 }

114template<class T>
115bool FInstancedStaticMeshComponentHelper::DoCustomNavigableGeometryExport(const T& Component, FNavigableGeometryExport& GeomExport, const FNavDataPerInstanceTransformDelegate& Delegate)
116{
117 UStaticMesh* StaticMesh = Component.GetStaticMesh();
118 if (StaticMesh && StaticMesh->GetNavCollision())
119 {
120 UNavCollisionBase* NavCollision = StaticMesh->GetNavCollision();
121 if (Component.ShouldExportAsObstacle(*NavCollision))
122 {
123 return false;
124 }
125
126 if (NavCollision->HasConvexGeometry())
127 {
128 NavCollision->ExportGeometry(FTransform::Identity, GeomExport);
129 }
130 else
131 {
132 UBodySetup* BodySetup = StaticMesh->GetBodySetup();
133 if (BodySetup)
134 {
135 GeomExport.ExportRigidBodySetup(*BodySetup, FTransform::Identity);
136 }
137 }
138
139 // Hook per instance transform delegate
140 GeomExport.SetNavDataPerInstanceTransformDelegate(Delegate);
141 }
142
143 // we don't want "regular" collision export for this component
144 return false;
145}

147template<class T>
148void FInstancedStaticMeshComponentHelper::GetNavigationPerInstanceTransforms(const T& Component, const FBox& AreaBox, TArray<FTransform>& InstanceData)
149{
150 const FBox InstanceBounds = FInstancedStaticMeshComponentHelper::GetInstanceNavigationBounds(Component);
151 if (InstanceBounds.IsValid)
152 {
153 const FBox LocalAreaBox = AreaBox.InverseTransformBy(Component.GetComponentTransform());
154 Component.ForEachInstanceMatrix([&Component, &InstanceBounds, &LocalAreaBox, &InstanceData](const FMatrix& InstanceTransform)
155 {
156 const FTransform InstanceToComponent(InstanceTransform);
157 if (!InstanceToComponent.GetScale3D().IsZero())
158 {
159 const FBoxSphereBounds TransformedInstanceBounds = InstanceBounds.TransformBy(InstanceTransform);
160 if (LocalAreaBox.Intersect(TransformedInstanceBounds.GetBox()))
161 {
162 InstanceData.Add(InstanceToComponent * Component.GetComponentTransform());
163 }
164 }
165 });
166 }
167}

169template<class T>
170FBox FInstancedStaticMeshComponentHelper::GetInstanceNavigationBounds(const T& Component)
171{
172 if (const UStaticMesh* Mesh = Component.GetStaticMesh())
173 {
174 const UNavCollisionBase* NavCollision = Mesh->GetNavCollision();
175 const FBox NavBounds = NavCollision ? NavCollision->GetBounds() : FBox();
176 return NavBounds.IsValid ? NavBounds : Mesh->GetBounds().GetBox();
177 }
178
179 return FBox();
