216 TSharedPtr<IAutomationLatentCommand> BuildQueue(TArray<TSharedPtr<IAutomationLatentCommand>>& Queue)
217 {
218 TSharedPtr<IAutomationLatentCommand> Result = nullptr;
219 if (Queue.Num() == 0)
220 {
221 return Result;
222 }
223 else if (Queue.Num() == 1)
224 {
225 Result = Queue[0];
226 }
227 else
228 {
229 Result = MakeShared<FRunSequence>(Queue);
230 }
231
232 Queue.Empty();
233 return Result;
234 }
