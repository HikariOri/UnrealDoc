97template<typename TProperties>
98class TLinuxTargetPlatformSettings
99 : public TTargetPlatformSettingsBase<TProperties>
100{
101public:
102
103 typedef TTargetPlatformSettingsBase<TProperties> TSuper;
104
108 TLinuxTargetPlatformSettings(const TCHAR* CookFlavor = nullptr, const TCHAR* OverrideIniPlatformName = nullptr)
109 : TSuper(CookFlavor, OverrideIniPlatformName)
110 {
111#if WITH_ENGINE
112 TextureLODSettings = nullptr;
113 StaticMeshLODSettings.Initialize(this);
114
115 // Get the Target RHIs for this platform, we do not always want all those that are supported.
116 TArray<FName> TargetedShaderFormats;
117 TLinuxTargetPlatformSettings::GetAllTargetedShaderFormats(TargetedShaderFormats);
118
119 // If we are targeting ES 2.0/3.1, we also must cook encoded HDR reflection captures
120 static FName NAME_SF_VULKAN_ES31(TEXT("SF_VULKAN_ES31"));
121 static FName NAME_OPENGL_150_ES3_1(TEXT("GLSL_150_ES31"));
122 bRequiresEncodedHDRReflectionCaptures = TargetedShaderFormats.Contains(NAME_SF_VULKAN_ES31)
123 || TargetedShaderFormats.Contains(NAME_OPENGL_150_ES3_1);
124#endif // WITH_ENGINE
125 }
126
127
128public:
129
130 //~ Begin ITargetPlatform Interface
131
132 virtual bool SupportsFeature(ETargetPlatformFeatures Feature) const override
133 {
134 if (Feature == ETargetPlatformFeatures::UserCredentials || Feature == ETargetPlatformFeatures::Packaging)
135 {
136 return true;
137 }
138
139 if (Feature == ETargetPlatformFeatures::CanCookPackages)
140 {
141 return TProperties::HasEditorOnlyData();
142 }
143
144 return TSuper::SupportsFeature(Feature);
145 }
146
147 virtual void GetAllPossibleShaderFormats( TArray<FName>& OutFormats ) const override
148 {
149 if (!TProperties::IsServerOnly())
150 {
151 static FName NAME_VULKAN_SM5(TEXT("SF_VULKAN_SM5"));
152 static FName NAME_VULKAN_SM6(TEXT("SF_VULKAN_SM6"));
153 static FName NAME_VULKAN_ES31(TEXT("SF_VULKAN_ES31"));
154
155 OutFormats.AddUnique(NAME_VULKAN_SM5);
156 OutFormats.AddUnique(NAME_VULKAN_SM6);
157 OutFormats.AddUnique(NAME_VULKAN_ES31);
158 }
159 }
160
161 virtual void GetAllTargetedShaderFormats( TArray<FName>& OutFormats ) const override
162 {
163 // Get the Target RHIs for this platform, we do not always want all those that are supported. (reload in case user changed in the editor)
164 TArray<FString>TargetedShaderFormats;
165 GConfig->GetArray(LINUX_SECTION_TEXT, TEXT("TargetedRHIs"), TargetedShaderFormats, GEngineIni);
166
167 // Gather the list of Target RHIs and filter out any that may be invalid.
168 TArray<FName> PossibleShaderFormats;
169 GetAllPossibleShaderFormats(PossibleShaderFormats);
170
171 for (int32 ShaderFormatIdx = TargetedShaderFormats.Num() - 1; ShaderFormatIdx >= 0; ShaderFormatIdx--)
172 {
173 FString ShaderFormat = TargetedShaderFormats[ShaderFormatIdx];
174 if (PossibleShaderFormats.Contains(FName(*ShaderFormat)) == false)
175 {
176 TargetedShaderFormats.RemoveAt(ShaderFormatIdx);
177 }
178 }
179
180 for(const FString& ShaderFormat : TargetedShaderFormats)
181 {
182 OutFormats.AddUnique(FName(*ShaderFormat));
183 }
184 }
185
186 virtual bool IsRunningPlatform() const override
187 {
188 // Must be Linux platform as editor for this to be considered a running platform
189 return PLATFORM_LINUX && !UE_SERVER && !UE_GAME && WITH_EDITOR && TProperties::HasEditorOnlyData();
190 }
191
192#if WITH_ENGINE
193 virtual void GetReflectionCaptureFormats(TArray<FName>& OutFormats) const override
194 {
195 if (bRequiresEncodedHDRReflectionCaptures)
196 {
197 OutFormats.Add(FName(TEXT("EncodedHDR")));
198 }
199
200 OutFormats.Add(FName(TEXT("FullHDR")));
201 }
202
203 virtual const class FStaticMeshLODSettings& GetStaticMeshLODSettings( ) const override
204 {
205 return StaticMeshLODSettings;
206 }
207
208 virtual const UTextureLODSettings& GetTextureLODSettings() const override
209 {
210 return *TextureLODSettings;
211 }
212
213 virtual void RegisterTextureLODSettings(const UTextureLODSettings* InTextureLODSettings) override
214 {
215 TextureLODSettings = InTextureLODSettings;
216 }
217
218#endif //WITH_ENGINE
219
220 virtual bool ShouldStripNaniteFallbackMeshes() const override
221 {
222 bool bGenerateNaniteFallbackMeshes = true;
223 GConfig->GetBool(TEXT("/Script/LinuxTargetPlatform.LinuxTargetSettings"), TEXT("bGenerateNaniteFallbackMeshes"), bGenerateNaniteFallbackMeshes, GEngineIni);
224
225 return !bGenerateNaniteFallbackMeshes;
226 }
227
228 virtual bool UsesRayTracing() const override
229 {
230 return GetRayTracingMode() != ERayTracingRuntimeMode::Disabled;
231 }
232
233 virtual ERayTracingRuntimeMode GetRayTracingMode() const override
234 {
235 if (!TTargetPlatformSettingsBase<TProperties>::UsesRayTracing())
236 {
237 return ERayTracingRuntimeMode::Disabled;
238 }
239
240 return TTargetPlatformSettingsBase<TProperties>::ParseRuntimeRayTracingMode(LINUX_SECTION_TEXT);
241 }
242
243 //~ End ITargetPlatform Interface
244
245protected:
246
247
248#if WITH_ENGINE
249 // Holds the texture LOD settings.
250 const UTextureLODSettings* TextureLODSettings;
251
252 // Holds static mesh LOD settings.
253 FStaticMeshLODSettings StaticMeshLODSettings;
254
255 // True if the project requires encoded HDR reflection captures
256 bool bRequiresEncodedHDRReflectionCaptures;
257#endif // WITH_ENGINE
258};
