386 OutValue = (*CustomAccessor.Functions.Getter)(InObject, MetaData);

388 static void GetObjectPropertyValue(const UObject* InObject, FBoolMetaData MetaData, uint16 PropertyOffset, bool& OutValue)
389 {
390 const void* PropertyAddress = reinterpret_cast<const uint8*>(InObject) + PropertyOffset;
391 switch(MetaData.BitFieldSize)
392 {
393 case 0: OutValue = (*reinterpret_cast<const bool *>(PropertyAddress)); return; // 0 means no bitfield
394 case 1: OutValue = (*reinterpret_cast<const uint8 *>(PropertyAddress) & (uint8 (1u) << MetaData.BitIndex)); return;
395 case 2: OutValue = (*reinterpret_cast<const uint16*>(PropertyAddress) & (uint16(1u) << MetaData.BitIndex)); return;
396 case 4: OutValue = (*reinterpret_cast<const uint32*>(PropertyAddress) & (uint32(1u) << MetaData.BitIndex)); return;
397 case 8: OutValue = (*reinterpret_cast<const uint64*>(PropertyAddress) & (uint64(1u) << MetaData.BitIndex)); return;
398 }

403 }

405 {
406 TOptional<bool> Property = FTrackInstancePropertyBindings::StaticValue<bool>(InObject, *PropertyPath.ToString());
407 if (Property)

414 static void SetObjectPropertyValue(UObject* InObject, FBoolMetaData MetaData, const FCustomPropertyAccessor& BaseCustomAccessor, bool InValue)
415 {
416 const TCustomPropertyAccessor<FBoolPropertyTraits>& CustomAccessor = static_cast<const TCustomPropertyAccessor<FBoolPropertyTraits>&>(BaseCustomAccessor);
417 (*CustomAccessor.Functions.Setter)(InObject, MetaData, InValue);
418 }
419 static void SetObjectPropertyValue(UObject* InObject, FBoolMetaData MetaData, uint16 PropertyOffset, bool InValue)
420 {
421 void* PropertyAddress = reinterpret_cast<uint8*>(InObject) + PropertyOffset;
422
423 // Perform a branchless set by getting the current value, removing the bit, then a bitwise or with InValue in the bit's position.
424 // For example, bit index 4 of a uint8 field:
425 // f(true) = (Value & 0b11101111) | 0b00010000
426 // f(false) = (Value & 0b11101111) | 0b00000000
427 switch(MetaData.BitFieldSize)
428 {
429 case 0: *reinterpret_cast<bool *>(PropertyAddress) = InValue; return; // 0 means no bitfield
430 case 1: *reinterpret_cast<uint8 *>(PropertyAddress) = (*reinterpret_cast<uint8 *>(PropertyAddress) & ~(uint8 (1u) << MetaData.BitIndex)) | (uint8 (InValue) << MetaData.BitIndex); return;
431 case 2: *reinterpret_cast<uint16*>(PropertyAddress) = (*reinterpret_cast<uint16*>(PropertyAddress) & ~(uint16(1u) << MetaData.BitIndex)) | (uint16(InValue) << MetaData.BitIndex); return;
432 case 4: *reinterpret_cast<uint32*>(PropertyAddress) = (*reinterpret_cast<uint32*>(PropertyAddress) & ~(uint32(1u) << MetaData.BitIndex)) | (uint32(InValue) << MetaData.BitIndex); return;
433 case 8: *reinterpret_cast<uint64*>(PropertyAddress) = (*reinterpret_cast<uint64*>(PropertyAddress) & ~(uint64(1u) << MetaData.BitIndex)) | (uint64(InValue) << MetaData.BitIndex); return;
434 }
435 }
436 static void SetObjectPropertyValue(UObject* InObject, FBoolMetaData MetaData, FTrackInstancePropertyBindings* PropertyBindings, bool InValue)
437 {
438 PropertyBindings->CallFunction<bool>(*InObject, InValue);
439 }
