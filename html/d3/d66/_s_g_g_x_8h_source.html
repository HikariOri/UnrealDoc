128template< typename T >
129FORCEINLINE T TSGGX<T>::ProjectedArea( const TVector<T>& Direction ) const
130{
131 // Projected area
132 // alpha = sqrt( w^T S w )
133 // alpha = sqrt( w^T n n^T w )
134 // alpha = abs( dot( n, w ) )
135
136 T x = Direction | TVector<T>( xx, xy, xz );
137 T y = Direction | TVector<T>( xy, yy, yz );
138 T z = Direction | TVector<T>( xz, yz, zz );
139 T wSw = Direction | TVector<T>( x, y, z );
140 return FMath::Sqrt( FMath::Max( 0.0f, wSw ) );
141}

143template< typename T >
144void TSGGX<T>::EigenSolve( TVector<T>& Values, TVector<T> Vectors[3] ) const
145{
146 // Diagonalize matrix
147 // A = V S V^T
148 T A[] =
149 {
150 xx, xy, xz,
151 xy, yy, yz,
152 xz, yz, zz
153 };
154 T S[3];
155 T V[9];
156
157 JacobiSVD::EigenSolver3( A, S, V, 1e-8f );
158
159 for( uint32 i = 0; i < 3; i++ )
160 {
161 Values[i] = S[i];
162 Vectors[i][0] = V[i+0];
163 Vectors[i][1] = V[i+3];
164 Vectors[i][2] = V[i+6];
165 }
166}

168template< typename T >
169void TSGGX<T>::FitIsotropic( TVector<T>& Center, TVector2<T>& Alpha ) const
170{
171 TVector<T> Values;
172 TVector<T> Vectors[3];
173 EigenSolve( Values, Vectors );
174
175 T Scale[3];
176 for( uint32 k = 0; k < 3; k++ )
177 Scale[k] = FMath::Sqrt( FMath::Abs( Values[k] ) ); // Should be positive already
178
179 T MaxRatio = 0.0;
180 int32 MaxIndex = 0;
181 for( uint32 k = 0; k < 3; k++ )
182 {
183 const uint32 k0 = k;
184 const uint32 k1 = (1 << k0) & 3;
185
186 T Ratio = FMath::Min( Scale[k0], Scale[k1] )
187 / FMath::Max( Scale[k0], Scale[k1] );
188 if( MaxRatio < Ratio )
189 {
190 MaxRatio = Ratio;
191 MaxIndex = k;
192 }
193 }
194
195 const uint32 k0 = MaxIndex;
196 const uint32 k1 = (1 << k0) & 3;
197 const uint32 k2 = (1 << k1) & 3;
198
199 Center = Vectors[k2];
200
201 Alpha[0] = 0.5f * ( Scale[k0] + Scale[k1] );
202 Alpha[1] = Scale[k2];
203}

205// Approx surface area of ellipsoid
206FORCEINLINE float SurfaceAreaEllipsoid( float a, float b, float c )
207{
208#if 1
209 float a2 = a*a;
210 float b2 = b*b;
211 float c2 = c*c;
212
213 //area ~= pi * ( (1-1/sqrt3) * (ab + ac + bc) + (1+1/sqrt3) * sqrt( a^2*b^2 + a^2*c^2 + b^2*c^2 ) )
214 return 1.32779329f * ( a*b + a*c + b*c ) +
215 4.95539202f * FMath::Sqrt( a2 * b2 + a2 * c2 + b2 * c2 );
216#else
217 const float p = 1.6f; // or p=ln(3)/ln(2)
218 float ap = FMath::Pow( a, p );
219 float bp = FMath::Pow( b, p );
220 float cp = FMath::Pow( c, p );
221 return (2.0f * PI) * FMath::Pow( ap * bp + ap * cp + bp * cp, 1.0f / p );
222#endif
223}

225template< typename T >
226T TSGGX<T>::SurfaceArea( const TVector<T>& Values )
227{
228 /*
229 integral sqrt(wSw) = pi * SurfaceArea
230 Average ProjectedArea() = 1/4 * SurfaceArea
231
232 Ellipsoid scale
233 a = ( S1 * S2 / S0 )^0.25 / sqrt(pi);
234 b = ( S2 * S0 / S1 )^0.25 / sqrt(pi);
235 c = ( S0 * S1 / S2 )^0.25 / sqrt(pi);
236
237 area ~= pi * ( (1-1/sqrt3) * (ab + ac + bc) + (1+1/sqrt3) * sqrt( a^2*b^2 + a^2*c^2 + b^2*c^2 ) )
238 */
239 const float sqrt3 = 1.73205080756887729353f;
240
241 return (1.0f - 1.0f / sqrt3) * ( FMath::Sqrt( Values[0] ) + FMath::Sqrt( Values[1] ) + FMath::Sqrt( Values[2] ) ) +
242 (1.0f + 1.0f / sqrt3) * FMath::Sqrt( Values[0] + Values[1] + Values[2] );
243}
