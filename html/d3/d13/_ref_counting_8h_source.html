45
46 void CheckAtLeast(uint32 N) const
47 {
48 // It's harmless to check if your refcount is at least a certain amount. Be aware
49 // that inside an AutoRTFM transaction, Release() is deferred until commit, so an
50 // object's refcount may be higher than you expected. In other words, when inside
51 // of a transaction, this check may not trigger even when the number of active
52 // reference holders is lower than the passed-in value.
53 checkSlow(RefCount >= N);
54 }
55
56private:
57 uint32 RefCount = 0;
58};

352 if (Reference)

358 OldReference->Release();
359 }
360 }

362 }
