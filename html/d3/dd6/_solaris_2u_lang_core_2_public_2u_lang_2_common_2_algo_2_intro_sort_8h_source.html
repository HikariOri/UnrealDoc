26 template <typename T, typename ProjectionType, typename PredicateType>
27 void IntroSortInternal(T* First, size_t Num, ProjectionType Projection, PredicateType Predicate)
28 {
29 struct SStack
30 {
31 T* Min;
32 T* Max;
33 uint32_t MaxDepth;
34 };
35
36 if( Num < 2 )
37 {
38 return;
39 }
40
41 SStack RecursionStack[32]={{First, First+Num-1, (uint32_t)(CMath::Loge((float)Num) * 2.f)}}, Current, Inner;
42 for( SStack* StackTop=RecursionStack; StackTop>=RecursionStack; --StackTop ) //-V625
43 {
44 Current = *StackTop;
45
46 Loop:
47 int32_t Count = int32_t(Current.Max - Current.Min + 1);
48
49 if ( Current.MaxDepth == 0 )
50 {
51 // We're too deep into quick sort, switch to heap sort
52 HeapSortInternal( Current.Min, Count, Projection, Predicate );
53 continue;
54 }
55
56 if( Count <= 8 )
57 {
58 // Use simple bubble-sort.
59 while( Current.Max > Current.Min )
60 {
61 T *Max, *Item;
62 for( Max=Current.Min, Item=Current.Min+1; Item<=Current.Max; Item++ )
63 {
64 if( Predicate( uLang::Invoke( Projection, *Max ), uLang::Invoke( Projection, *Item ) ) )
65 {
66 Max = Item;
67 }
68 }
69 uLang::Swap( *Max, *Current.Max-- );
70 }
71 }
72 else
73 {
74 // Grab middle element so sort doesn't exhibit worst-cast behavior with presorted lists.
75 uLang::Swap( Current.Min[Count/2], Current.Min[0] );
76
77 // Divide list into two halves, one with items <=Current.Min, the other with items >Current.Max.
78 Inner.Min = Current.Min;
79 Inner.Max = Current.Max+1;
80 for( ; ; )
81 {
82 while( ++Inner.Min<=Current.Max && !Predicate( uLang::Invoke( Projection, *Current.Min ), uLang::Invoke( Projection, *Inner.Min ) ) );
83 while( --Inner.Max> Current.Min && !Predicate( uLang::Invoke( Projection, *Inner.Max ), uLang::Invoke( Projection, *Current.Min ) ) );
84 if( Inner.Min>Inner.Max )
85 {
86 break;
87 }
88 uLang::Swap( *Inner.Min, *Inner.Max );
89 }
90 uLang::Swap( *Current.Min, *Inner.Max );
91
92 --Current.MaxDepth;
93
94 // Save big half and recurse with small half.
95 if( Inner.Max-1-Current.Min >= Current.Max-Inner.Min )
96 {
97 if( Current.Min+1 < Inner.Max )
98 {
99 StackTop->Min = Current.Min;
100 StackTop->Max = Inner.Max - 1;
101 StackTop->MaxDepth = Current.MaxDepth;
102 StackTop++;
103 }
104 if( Current.Max>Inner.Min )
105 {
106 Current.Min = Inner.Min;
107 goto Loop;
108 }
109 }
110 else
111 {
112 if( Current.Max>Inner.Min )
113 {
114 StackTop->Min = Inner .Min;
115 StackTop->Max = Current.Max;
116 StackTop->MaxDepth = Current.MaxDepth;
117 StackTop++;
118 }
119 if( Current.Min+1<Inner.Max )
120 {
121 Current.Max = Inner.Max - 1;
122 goto Loop;
123 }
124 }
125 }
126 }
127 }
128}
