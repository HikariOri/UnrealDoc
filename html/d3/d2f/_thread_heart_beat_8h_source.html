268 bool bSuspendedAllThreads;

270public:
271 inline FSlowHeartBeatScope(bool bAllThreads = false)
272 : bSuspendedAllThreads(bAllThreads)

274 {
275 if (FThreadHeartBeat* HB = FThreadHeartBeat::GetNoInit())
276 {
277 bSuspended = true;
