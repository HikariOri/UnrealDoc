34 struct FBuildPatchInstallerAction
35 {
36 public:
37 FBuildPatchInstallerAction(const FInstallerAction& InInstallerAction)
38 : InstallerAction(InInstallerAction)
39 , CurrentManifest(StaticCastSharedPtr<FBuildPatchAppManifest>(InstallerAction.TryGetCurrentManifest()))
40 , InstallManifest(StaticCastSharedPtr<FBuildPatchAppManifest>(InstallerAction.TryGetInstallManifest()))
41 {
42 // Pre-process install tags. Doing this logic here means it doesn't need repeating around lower level code.
43 // No tags means full installation.
44 InstallTags = InstallerAction.GetInstallTags();
45 if (InstallTags.Num() == 0)
46 {
47 GetInstallOrCurrentManifest().GetFileTagList(InstallTags);
48 }
49 // Always require the empty tag.
50 InstallTags.Add(TEXT(""));
51 // Cache tagged files.
52 if (InstallManifest.IsValid())
53 {
54 InstallManifest->GetTaggedFileList(InstallTags, TaggedFiles);
55 }
56 }
57
58 bool IsInstall() const { return InstallerAction.IsInstall(); }

60 bool IsRepair() const { return InstallerAction.IsRepair(); }
