71 inline bool NeedsResolving() const
72 {
73 if (ResolvedField)
74 {
75#if WITH_EDITORONLY_DATA
76 UStruct* Owner = ResolvedOwner.Get();
77 // In uncooked builds we also need to check if the serial number on the owner struct is identical
78 // It will change if the struct has been recompiled or its properties have been destroyed

83#else

86#endif // WITH_EDITORONLY_DATA
87 }

181 inline void ResolveField(FFieldClass* ExpectedClass = FField::StaticClass(), UStruct* InCurrentStruct = nullptr, EPathResolveType InResolveType = FFieldPath::UseStructIfOuterNotFound) const
182 {
183 FField* FoundField = TryToResolvePath(InCurrentStruct, InResolveType);
184 if (FoundField && FoundField->IsA(ExpectedClass)
185#if WITH_EDITORONLY_DATA
186 && (!InitialFieldClass || FoundField->IsA(InitialFieldClass))
187#endif // WITH_EDITORONLY_DATA
188 )
189 {
190 ResolvedField = FoundField;
191#if WITH_EDITORONLY_DATA
192 if (!InitialFieldClass)
193 {
194 InitialFieldClass = FoundField->GetClass();
195 }
196 UStruct* Owner = ResolvedOwner.Get();
197 check(Owner);
198 FieldPathSerialNumber = GetFieldPathSerialNumber(Owner);
199#endif // WITH_EDITORONLY_DATA
200 }
201 else if (ResolvedField)
202 {
203 // In case this field has been previously resolved, clear the owner as well as it's impossible the original field
204 // will ever come back (it's most likely been deleted) and we don't want to resolve to a newly created one even if its name and class match
205 ResolvedOwner.Reset();
206 ResolvedField = nullptr;
207 }
208 }
209
216 inline FField* GetTyped(FFieldClass* ExpectedType, UStruct* InCurrentStruct = nullptr) const
217 {
218 if (NeedsResolving() && Path.Num())
219 {
220 ResolveField(ExpectedType, InCurrentStruct, FFieldPath::UseStructIfOuterNotFound);
221 }
222 return ResolvedField;
223 }
224
229 inline bool IsPathToFieldEmpty() const
230 {
231 return !Path.Num();
232 }
233
238 inline bool IsStale() const
239 {
240 return ResolvedField && (!ResolvedOwner.IsValid()
241#if WITH_EDITORONLY_DATA
242 || !IsFieldPathSerialNumberIdentical(ResolvedOwner.Get())
243#endif // WITH_EDITORONLY_DATA
244 );
245 }
246
250 inline void Reset()
251 {
252 ClearCachedField();
253 ResolvedOwner.Reset();
254 Path.Empty();
255 }
256
257 UE_FORCEINLINE_HINT bool UEOpEquals(const FFieldPath& InOther) const
258 {
259 return ResolvedOwner == InOther.ResolvedOwner && Path == InOther.Path;
260 }

262 COREUOBJECT_API FString ToString() const;

267 [[nodiscard]] inline friend uint32 GetTypeHash(const FFieldPath& InPropertyPath)
268 {
269 uint32 HashValue = 0;
270 for (const FName& PathSegment : InPropertyPath.Path)
271 {
272 HashValue = HashCombine(HashValue, GetTypeHash(PathSegment));
273 }
