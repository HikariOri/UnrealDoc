74class FRHICommandListIterator
75{
76public:
77 FRHICommandListIterator(FRHICommandListBase& CmdList)
78 {
79 CmdPtr = CmdList.Root;
80#if DO_CHECK
81 NumCommands = 0;
82 CmdListNumCommands = CmdList.NumCommands;
83#endif
84 }
85 ~FRHICommandListIterator()
86 {
87#if DO_CHECK
88 checkf(CmdListNumCommands == NumCommands, TEXT("Missed %d Commands!"), CmdListNumCommands - NumCommands);
89#endif
90 }
91
92 inline bool HasCommandsLeft() const
93 {
94 return !!CmdPtr;
95 }
96
97 inline FRHICommandBase* NextCommand()
98 {
99 FRHICommandBase* RHICmd = CmdPtr;
100 CmdPtr = RHICmd->Next;
101#if DO_CHECK
102 NumCommands++;
103#endif
104 return RHICmd;
105 }
106
107private:
108 FRHICommandBase* CmdPtr;
109
110#if DO_CHECK
111 uint32 NumCommands;
112 uint32 CmdListNumCommands;
113#endif
114};

183 template <typename TDesc, typename... TValues>
184 inline FRHIBreadcrumbNode* FRHIBreadcrumbAllocator::AllocBreadcrumb(TRHIBreadcrumbInitializer<TDesc, TValues...> const& Args)
185 {
186 TDesc const* Desc = std::get<0>(Args);
187 if (!Desc)
188 {
189 return nullptr;
190 }
191
192 UE::RHI::GPUProfiler::FGPUStat* GPUStat = std::get<2>(Args);
193
194 return std::apply([&](auto&&... Values)
195 {
196 using TType = UE::RHI::Breadcrumbs::Private::TRHIBreadcrumb<TDesc>;
197 static_assert(std::is_trivially_destructible<TType>::value, "Only trivially destructable types may be used with the RHI breadcrumb allocator.");
198
199 void* Addr = Inner.Alloc(sizeof(TType), alignof(TType));
200 return new (Addr) TType(*this, *Desc, GPUStat, std::forward<decltype(Values)>(Values)...);
201 }, std::get<1>(Args));
202 }
