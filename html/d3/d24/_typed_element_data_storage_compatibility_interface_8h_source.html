257 template<typename ObjectType>
258 RowHandle ICompatibilityProvider::AddCompatibleObject(ObjectType&& Object)
259 {
260 auto RawPointer = GetRawPointer(Forward<ObjectType>(Object));
261 using BaseType = std::remove_cv_t<std::remove_pointer_t<decltype(RawPointer)>>;
262
263 if constexpr (std::is_base_of_v<UObject, BaseType>)
264 {
265 return AddCompatibleObjectExplicit(RawPointer);
266 }
267 else
268 {
269 return AddCompatibleObjectExplicit(RawPointer, BaseType::StaticStruct());
270 }
271 }
