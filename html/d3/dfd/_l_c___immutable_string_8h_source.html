167 // layout of a short string:
168 // the actual string data (including the null terminator), 1 byte storing the length of the short string, 4 byte hash
169 struct ShortString

172 static const uint8_t MSB_CLEAR_MASK = 0x7Fu; // mask for clearing the MSB

174 char str[sizeof(const char*) + sizeof(uint32_t) - 1];
