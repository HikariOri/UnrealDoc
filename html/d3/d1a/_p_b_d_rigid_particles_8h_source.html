120 void SetSleeping(int32 Index, bool bSleeping)
121 {
122 if (Sleeping(Index) && bSleeping == false)
123 {
124 SetPreV(Index, this->GetV(Index));
125 SetPreW(Index, this->GetW(Index));
126 }
127 else if (bSleeping)
128 {
129 //being put to sleep, so zero out velocities
130 this->SetV(Index, FVec3(0));
131 this->SetW(Index, FVec3(0));
132 }
133
134 bool CurrentlySleeping = this->ObjectState(Index) == EObjectStateType::Sleeping;
135 if (CurrentlySleeping != bSleeping)
136 {
137 TGeometryParticleHandle<T, d>* Particle = reinterpret_cast<TGeometryParticleHandle<T, d>*>(this->Handle(Index));
138 this->AddSleepData(Particle, bSleeping);
139 }
140
141 // Dynamic -> Sleeping or Sleeping -> Dynamic
142 if (this->ObjectState(Index) == EObjectStateType::Dynamic || this->ObjectState(Index) == EObjectStateType::Sleeping)
143 {
144 this->ObjectState(Index) = bSleeping ? EObjectStateType::Sleeping : EObjectStateType::Dynamic;
145 }
146
147 // Possible for code to set a Dynamic to Kinematic State then to Sleeping State
148 if (this->ObjectState(Index) == EObjectStateType::Kinematic && bSleeping)
149 {
150 this->ObjectState(Index) = EObjectStateType::Sleeping;
151 }
152
153 if (bSleeping)
154 {
155 EnsureSleepingObjectState(this->ObjectState(Index));
156 }
157 }
158
159 void SetObjectState(int32 Index, EObjectStateType InObjectState)
160 {
161 const EObjectStateType CurrentState = this->ObjectState(Index);
162
163 if (CurrentState == EObjectStateType::Uninitialized)
164 {
165 // When the state is first initialized, treat it like a static.
166 this->InvM(Index) = 0.0f;
167 this->InvI(Index) = TVec3<FRealSingle>(0);
168 }
169
170 if ((CurrentState == EObjectStateType::Dynamic || CurrentState == EObjectStateType::Sleeping) && (InObjectState == EObjectStateType::Kinematic || InObjectState == EObjectStateType::Static))
171 {
172 // Transitioning from dynamic to static or kinematic, set inverse mass and inertia tensor to zero.
173 this->InvM(Index) = 0.0f;
174 this->InvI(Index) = TVec3<FRealSingle>(0);
175 }
176 else if ((CurrentState == EObjectStateType::Kinematic || CurrentState == EObjectStateType::Static || CurrentState == EObjectStateType::Uninitialized) && (InObjectState == EObjectStateType::Dynamic || InObjectState == EObjectStateType::Sleeping))
177 {
178 // Transitioning from kinematic or static to dynamic, compute the inverses.
179 this->InvM(Index) = FMath::IsNearlyZero(this->M(Index)) ? 0.0f : 1.f / this->M(Index);
180 this->InvI(Index) = (this->I(Index).IsNearlyZero() || FMath::IsNearlyZero(this->M(Index))) ? TVec3<FRealSingle>::ZeroVector :
181 TVec3<FRealSingle>(
182 1.f / this->I(Index)[0],
183 1.f / this->I(Index)[1],
184 1.f / this->I(Index)[2]);
185
186 this->SetP(Index, this->GetX(Index));
187 this->SetQf(Index, this->GetRf(Index));
188 }
189 else if (InObjectState == EObjectStateType::Sleeping)
190 {
191 SetSleeping(Index, true);
192 return;
193 }
194
195 const bool bCurrentSleeping = this->ObjectState(Index) == EObjectStateType::Sleeping;
196 const bool bNewSleeping = InObjectState == EObjectStateType::Sleeping;
197 if(bCurrentSleeping != bNewSleeping)
198 {
199 TGeometryParticleHandle<T, d>* Particle = reinterpret_cast<TGeometryParticleHandle<T, d>*>(this->Handle(Index));
200 this->AddSleepData(Particle, bNewSleeping);
201 }
202
203 this->ObjectState(Index) = InObjectState;
204 }

222 virtual void Serialize(FChaosArchive& Ar) override
223 {
224 TRigidParticles<T, d>::Serialize(Ar);
225
226 Ar << MP;
227
228 Ar.UsingCustomVersion(FUE5ReleaseStreamObjectVersion::GUID);
229 if (Ar.CustomVer(FUE5ReleaseStreamObjectVersion::GUID) >= FUE5ReleaseStreamObjectVersion::SinglePrecisonParticleDataPT)
230 {
231 Ar << MQ << MPreV << MPreW;
232 }
233 else
234 {
235 TArrayCollectionArray<TRotation<FReal, d>> QDouble;
236 QDouble.Resize(MQ.Num());
237 for (int32 Index = 0; Index < MQ.Num(); ++Index)
238 {
239 QDouble[Index] = TRotation<FReal, d >(MQ[Index]);
240 }
241 TArrayCollectionArray<TVector<FReal, d>> PreVDouble;
242 PreVDouble.Resize(MPreV.Num());
243 for (int32 Index = 0; Index < MPreV.Num(); ++Index)
244 {
245 PreVDouble[Index] = TVector<FReal, d >(MPreV[Index]);
246 }
247 TArrayCollectionArray<TVector<FReal, d>> PreWDouble;
248 PreWDouble.Resize(MPreW.Num());
249 for (int32 Index = 0; Index < MPreW.Num(); ++Index)
250 {
251 PreWDouble[Index] = TVector<FReal, d >(MPreW[Index]);
252 }
253
254 Ar << QDouble << PreVDouble << PreWDouble;
255
256 MQ.Resize(QDouble.Num());
257 for (int32 Index = 0; Index < QDouble.Num(); ++Index)
258 {
259 MQ[Index] = TRotation<FRealSingle, d >(QDouble[Index]);
260 }
261 MPreV.Resize(PreVDouble.Num());
262 for (int32 Index = 0; Index < PreVDouble.Num(); ++Index)
263 {
264 MPreV[Index] = TVector<FRealSingle, d >(PreVDouble[Index]);
265 }
266 MPreW.Resize(PreWDouble.Num());
267 for (int32 Index = 0; Index < PreWDouble.Num(); ++Index)
268 {
269 MPreW[Index] = TVector<FRealSingle, d >(PreWDouble[Index]);
270 }
271 }
