170 static bool IsValidChildType(const Type InParentType, const Type InChildType)
171 {
172 if (InParentType == CST_Local)
173 {
174 // Local collections can only contain other local collections
175 return InChildType == CST_Local;
176 }
177
178 if (InParentType == CST_Private)
179 {
180 // Private collections can only contain local or private collections
181 return InChildType == CST_Local || InChildType == CST_Private;
182 }
183
184 if (InParentType == CST_Shared)
185 {
186 // Shared collections can contain any kind of collection
187 return InChildType == CST_Local || InChildType == CST_Private || InChildType == CST_Shared;
188 }
189
190 return false;
191 }
