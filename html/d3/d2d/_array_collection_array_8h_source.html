18template<typename RemoveIndexFunc>
19FORCEINLINE void RemoveAtSwapLambda(const int32 ArraySize, const TArrayView<int32>& Indices, const RemoveIndexFunc& RemoveFunc)
20{
21 if (Indices.IsEmpty() || (Indices.Num() > ArraySize))
22 {
23 return;
24 }
25#if !(UE_BUILD_SHIPPING || UE_BUILD_TEST)
26 TBitArray<> ValidIndices;
27 ValidIndices.Init(false,ArraySize);
28 for (int32 RemovedIndex : Indices)
29 {
30 if ((RemovedIndex < ArraySize) && (RemovedIndex >= 0))
31 {
32 // Check if the list of indices to remove is unique or not
33 check(!ValidIndices[RemovedIndex]);
34 ValidIndices[RemovedIndex] = true;
35 }
36 }
37#endif
38 // The mapping array is twice the size M of the indices to be removed among the array of size N elements
39 // [forward indices : array index position of the last M element position | reverse indices : element index value of the last M array positions]
40 TArray<int32> IndexMapping;
41 IndexMapping.Init(INDEX_NONE, 2 * Indices.Num());
42
43 // Since we are storing only that last M elements we need to store the forward/reverse offset to come back to global indices
44 const int32 ForwardOffset = ArraySize - Indices.Num();
45 const int32 ReverseOffset = ForwardOffset - Indices.Num();
46
47 // The goal here is to perform successive RemoveAtSwap on a list of indices without sorting them
48 // since doing naively sequential calls could leads to wrong results if not sorted correctly
49 int32 EndIndex = ArraySize - 1;
50 for (int32 RemovedIndex : Indices)
51 {
52 if ((RemovedIndex < ArraySize) && (RemovedIndex >= 0) && IndexMapping.IsValidIndex(EndIndex - ReverseOffset))
53 {
54 const int32 ForwardMapping = (RemovedIndex < ForwardOffset) ? RemovedIndex : IndexMapping[RemovedIndex - ForwardOffset];
55 const int32 ReverseMapping = IndexMapping[EndIndex - ReverseOffset];
56
57 const int32 ForwardIndex = (ForwardMapping != INDEX_NONE) ? ForwardMapping : RemovedIndex;
58 const int32 ReverseIndex = (ReverseMapping != INDEX_NONE) ? ReverseMapping : EndIndex;
59
60 if (ReverseIndex >= ForwardOffset)
61 {
62 IndexMapping[ReverseIndex - ForwardOffset] = ForwardIndex;
63 }
64
65 // If ForwardIndex >= ForwardOffset then ForwardIndex >= ReverseOffset and the IndexMapping access is valid
66 if (ForwardIndex >= ForwardOffset)
67 {
68 IndexMapping[ForwardIndex - ReverseOffset] = ReverseIndex;
69 }
70 RemoveFunc(ForwardIndex);
71 --EndIndex;
72 }
73 }
74}

188 {

190 OtherTArray.Reserve(OtherTArray.Num() + Indices.Num());

192 // Add elements to the new collection arrays

195 if (TArray<T>::IsValidIndex(Idx))

197 OtherTArray.Emplace(MoveTemp(TArray<T>::operator [](Idx)));
198 }
199 }
200
201 // Remove elements from the old collection arrays
202 if(bRemoveElements)
203 {
204 RemoveAtSwap(Indices);
205 }
