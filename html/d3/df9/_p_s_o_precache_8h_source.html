244
245 // Precache priority mode
246 EPSOPrecachePriority Priority = EPSOPrecachePriority::Medium;
247
248 // Keep the PSO in memory until the owning material is request deleted
249 EPSOPrecacheInMemoryMode InMemoryMode = EPSOPrecacheInMemoryMode::Cached;
250
251 FGraphicsPipelineStateInitializer GraphicsPSOInitializer;
252 FComputePipelineStateInitializer ComputeInitializer;
253
254 FShaderPreloadData ShaderPreloadData;
255
256 bool SetComputeShader(const TShaderRef<FShader>& InComputeShader)
257 {
258 if (GetPSOPrecacheMode() == EPSOPrecacheMode::PreloadShader)
259 {
260 ShaderPreloadData.Shaders.Emplace(InComputeShader);
261 }
262 else
263 {
264 FRHIComputeShader* ComputeShaderRHI = InComputeShader.GetComputeShader(false);
265 if (!ComputeShaderRHI)
266 {
267 return false;
268 }
269 ComputeInitializer = FComputePipelineStateInitializer(ComputeShaderRHI, InComputeShader.GetSpecializationConstants());
270 }
271 return true;
272 }
273

275 FString ResourceName;
