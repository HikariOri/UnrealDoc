97 virtual bool Serialize(FArchive& Archive) override
98 {
99 int32 Version = LAUNCHERSERVICES_SIMPLEPROFILEVERSION;
100
101 Archive << Version;
102
103 if (Version != LAUNCHERSERVICES_SIMPLEPROFILEVERSION)
104 {
105 return false;
106 }
107
108 // IMPORTANT: make sure to bump LAUNCHERSERVICES_SIMPLEPROFILEVERSION when modifying this!
109 Archive << DeviceName
110 << Variant
111 << BuildConfiguration
112 << CookMode;
113
114 return true;
115 }
116
117 virtual void Save(TJsonWriter<>& Writer) override
118 {
119 int32 Version = LAUNCHERSERVICES_SIMPLEFILEFORMATCHANGE;
120
121 Writer.WriteObjectStart();
122 Writer.WriteValue(TEXT("Version"), Version);
123 Writer.WriteValue(TEXT("DeviceName"), DeviceName);
124 Writer.WriteValue(TEXT("Variant"), Variant.ToString());
125 Writer.WriteValue(TEXT("BuildConfiguration"), (int32)BuildConfiguration);
126 Writer.WriteValue(TEXT("CookMode"), CookMode);
127 Writer.WriteObjectEnd();
128 }
129
130 virtual bool Load(const FJsonObject& Object) override
131 {
132 int32 Version = (int32)Object.GetNumberField(TEXT("Version"));
133 if (Version < LAUNCHERSERVICES_SIMPLEFILEFORMATCHANGE)
134 {
135 return false;
136 }
137
138 DeviceName = Object.GetStringField(TEXT("DeviceName"));
139 Variant = *(Object.GetStringField(TEXT("Variant")));
140 BuildConfiguration = (EBuildConfiguration)((int32)Object.GetNumberField(TEXT("BuildConfiguration")));
141 CookMode = (TEnumAsByte<ELauncherProfileCookModes::Type>)((int32)Object.GetNumberField(TEXT("CookMode")));
142
143 return true;
144 }
145
146 virtual void SetDefaults() override
147 {
148 // None will mean the preferred Variant for the device is used.
149 Variant = NAME_None;
150
151 // I don't use FApp::GetBuildConfiguration() because i don't want the act of running in debug the first time to cause the simple
152 // profiles created for your persistent devices to be in debug. The user might not see this if they don't expand the Advanced options.
153 BuildConfiguration = EBuildConfiguration::Development;
154
155 CookMode = ELauncherProfileCookModes::OnTheFly;
156 }
157
158private:
159
160 // Holds the name of the device this simple profile is for
161 FString DeviceName;
162
163 // Holds the name of the device variant.
164 FName Variant;
165
166 // Holds the desired build configuration (only used if creating new builds).
167 EBuildConfiguration BuildConfiguration;
168
169 // Holds the cooking mode.
170 TEnumAsByte<ELauncherProfileCookModes::Type> CookMode;
171};

1390 TMap<FString, FString> ParseCommands(FString CommandLine)
1391 {
1392 TMap<FString, FString> RoleCommands;
1393 FString Left;
1394 FString Right;
1395 while (CommandLine.Split(TEXT(" "), &Left, &Right))
1396 {
1397 if(Left.Len() > 0)
1398 {
1399 FString Key;
1400 FString Value;
1401 if (!Left.Split(TEXT("="), &Key, &Value))
1402 {
1403 Key = Left;
1404 Value = TEXT("true");
1405 }
1406 Key.RemoveFromStart(TEXT("-"));
1407 RoleCommands.Add(Key, Value);
1408 }
1409 CommandLine = Right;
1410 }
1411 if (CommandLine.Len() > 0)
1412 {
1413 if (!CommandLine.Split(TEXT("="), &Left, &Right))
1414 {
1415 Right = TEXT("true");
1416 }
1417 Left.RemoveFromStart(TEXT("-"));
1418 RoleCommands.Add(Left, Right);
1419 }
1420
1421 return RoleCommands;
1422 }

1466 bool FindAllPlatforms(TArray<FString>& ServerPlatforms, TArray<FString>& ClientPlatforms, FString& OptionalParams)
1467 {

1471 {
1472 // Platform info for the given platform
1473 const PlatformInfo::FTargetPlatformInfo* PlatformInfo = PlatformInfo::FindPlatformInfo(FName(*InPlatforms[PlatformIndex]));
1474 if (PlatformInfo == nullptr)
1475 {
1476 return false;
1477 }
1478
1479 // add the UBT platform name to the appropriate list
1480 TArray<FString>& Platforms = PlatformInfo->PlatformType == EBuildTargetType::Server ? ServerPlatforms : ClientPlatforms;
1481 Platforms.Add(PlatformInfo->DataDrivenPlatformInfo->UBTPlatformName.ToString());
1482
1483 // Append any extra UAT flags specified for this platform flavor
1484 if (!PlatformInfo->UATCommandLine.IsEmpty())
1485 {
1486 OptionalParams += PlatformInfo->UATCommandLine;
1487 }
1488
1489 bUATClosesAfterLaunch |= PlatformInfo->DataDrivenPlatformInfo->bUATClosesAfterLaunch;

1492 // If both Client and Server are desired to be built avoid Server causing clients to not be built PlatformInfo wise
1493 if (OptionalParams.Contains(TEXT("-client")) && OptionalParams.Contains(TEXT("-noclient")))
1494 {
1495 OptionalParams = OptionalParams.Replace(TEXT("-noclient"), TEXT(""));
1496 }
1497
1498 return bUATClosesAfterLaunch;
