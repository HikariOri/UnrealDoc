177 template<typename Func>
178 void ForEach(Func&& Callback) const
179 {
180 // This could maybe be smarter and leverage intrinsics to directly
181 // scan for the next set bit but I just want to make it work for
182 // now so let's keep it simple until it shows up in a profile
183
184 uint64 IterMask = Mask;
185 int Ordinal = 0;
186
187 while(IterMask)
188 {
189 if (IterMask & 1)
190 {
191 Callback(ITargetPlatform::GetPlatformFromOrdinal(Ordinal));
192 }
193
194 IterMask >>= 1;
195 ++Ordinal;
196 }
197 }
198
199 // TODO: support for ranged for? It's easy to make mistakes when refactoring
200 // regular for loops into ForEach type constructs since the semantics of any
201 // return statements does change when you put them inside a lambda.
