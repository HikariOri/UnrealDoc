254 template<typename ChannelType>
255 void Optimize(ChannelType* InChannel, const FKeyDataOptimizationParams& Params)
256 {
257 using namespace UE::MovieScene;
258
259 auto ChannelInterface = InChannel->GetData();
260 if (ChannelInterface.GetTimes().Num() > 1)
261 {
262 int32 StartIndex = 0;
263 int32 EndIndex = 0;
264
265 {
266 TArrayView<const FFrameNumber> Times = ChannelInterface.GetTimes();
267 StartIndex = Params.Range.GetLowerBound().IsClosed() ? Algo::LowerBound(Times, Params.Range.GetLowerBoundValue()) : 0;
268 EndIndex = Params.Range.GetUpperBound().IsClosed() ? Algo::UpperBound(Times, Params.Range.GetUpperBoundValue()) : Times.Num();
269 }
270
271 for (int32 Index = StartIndex; Index < EndIndex && Index < ChannelInterface.GetTimes().Num(); ++Index)
272 {
273 // Reget times and values as they may be reallocated
274 FFrameNumber Time = ChannelInterface.GetTimes()[Index];
275 auto OriginalValue = ChannelInterface.GetValues()[Index];
276
277 // If the channel evaluates the same with this key removed, we can leave it out
278 ChannelInterface.RemoveKey(Index);
279 if (ValueExistsAtTime(InChannel, Time, OriginalValue))
280 {
281 Index--;
282 }
283 else
284 {
285 ChannelInterface.AddKey(Time, MoveTemp(OriginalValue));
286 }
287 }
288 }
