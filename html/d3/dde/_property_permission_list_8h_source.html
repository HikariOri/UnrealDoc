36enum class EPropertyPermissionListRules : uint8
37{
38 // If no PermissionList is manually defined for this Struct, AllowList all properties from this Struct and its subclasses
39 AllowListAllProperties,
40 // If a PermissionList is manually defined for this Struct, AllowList all properties from this Struct's subclasses.
41 // If this functionality is needed without any properties to AllowList, a fake property must be added to AllowList instead.
42 AllowListAllSubclassProperties,
43 // If a PermissionList is manually defined for this struct, PermissionList those properties. Otherwise, use the parent Struct's rule.
44 UseExistingPermissionList
45};
