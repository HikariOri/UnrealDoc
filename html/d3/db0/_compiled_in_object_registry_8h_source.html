21 // Matches list markers in TList noting which objects have been constructed to which step

29 template<typename T>
30 struct TList
31 {
32 // Linked list of objects to construct. New items are added here, before items that may have
33 // already been partially processed which are indicated by the following pointers.
34 T* Head = nullptr;
35 // First block of objects that have already been processed by AddAndHashObjects
36 T* AlreadyAdded = nullptr;
37 // First block of objects that have already been processed by FinishConstructingObjects
38 T* AlreadyConstructed = nullptr;
39 // First block of objects that have already been processed by CreateClassDefaultObjects
40 T* AlreadyConstructedDefaultObjects = nullptr;

43 void EmptyObjects();

45 [[nodiscard]] T* AdvanceConstruction(EConstructionStep Step)
46 {
47 T** Link = nullptr;
48 switch (Step)
49 {
50 case EConstructionStep::AddObjects:
51 Link = &AlreadyAdded;
52 break;
53 case EConstructionStep::ConstructObjects:
54 Link = &AlreadyConstructed;
55 break;
56 case EConstructionStep::ConstructDefaultObjects:
57 Link = &AlreadyConstructedDefaultObjects;
58 break;
59 default: check(false);
60 }
61
62 T* Ret = *Link;
63 *Link = Head;
64 return Ret;
65 }
66
68 [[nodiscard]] bool HasObjectsPendingConstruction() const
69 {
70 return Head && (Head != AlreadyAdded || Head != AlreadyConstructed);
71 }
72
74 [[nodiscard]] bool HasPendingObjects() const
75 {
76 return Head && (Head != AlreadyAdded || Head != AlreadyConstructed || Head != AlreadyConstructedDefaultObjects);
77 }
78 };
