185 EVisibility SetPinnedItemVisibility(const int32 IndexInList, const int32 NumPinnedItems) const
186 {
187 // If the hierarchy is not collapsed (i.e all items are visible)
188 if (!OwnerListView->bIsHierarchyCollapsed)
189 {
190 return EVisibility::Visible;
191 }
192

199 }

218
219 // If the hierarchy is collapsed, the last item before the collapsed items gets the ellipses
220 return IndexInList == CurrentMaxPinnedItems - 2 ? EVisibility::Visible : EVisibility::Collapsed;
221 }

587protected:
588 FReply OnKeyDown_Internal(const FGeometry& MyGeometry, const FKeyEvent& InKeyEvent)
589 {
590 const TArrayView<const ItemType> ItemsSourceRef = GetItems();
591
592 // Don't respond to key-presses containing "Alt" as a modifier
593 if ( ItemsSourceRef.Num() > 0 && !InKeyEvent.IsAltDown() )
594 {
595 bool bWasHandled = false;
596 NullableItemType ItemNavigatedTo = TListTypeTraits<ItemType>::MakeNullPtr();
597
598 // Check for selection manipulation keys (Up, Down, Home, End, PageUp, PageDown)
599 if ( InKeyEvent.GetKey() == EKeys::Home )
600 {
601 // Select the first item
602 ItemNavigatedTo = ItemsSourceRef[0];
603 bWasHandled = true;
604 }
605 else if ( InKeyEvent.GetKey() == EKeys::End )
606 {
607 // Select the last item
608 ItemNavigatedTo = ItemsSourceRef.Last();
609 bWasHandled = true;
610 }
611 else if ( InKeyEvent.GetKey() == EKeys::PageUp )
612 {
613 int32 SelectionIndex = 0;
614 if( TListTypeTraits<ItemType>::IsPtrValid(SelectorItem) )
615 {
616 SelectionIndex = ItemsSourceRef.Find( TListTypeTraits<ItemType>::NullableItemTypeConvertToItemType( SelectorItem ) );
617 }
618
619 int32 NumItemsInAPage = FMath::TruncToInt(GetNumLiveWidgets());
620 int32 Remainder = NumItemsInAPage % GetNumItemsPerLine();
621 NumItemsInAPage -= Remainder;
622
623 if ( SelectionIndex >= NumItemsInAPage )
624 {
625 // Select an item on the previous page
626 ItemNavigatedTo = ItemsSourceRef[SelectionIndex - NumItemsInAPage];
627 }
628 else
629 {
630 ItemNavigatedTo = ItemsSourceRef[0];
631 }
632
633 bWasHandled = true;
634 }
635 else if ( InKeyEvent.GetKey() == EKeys::PageDown )
636 {
637 int32 SelectionIndex = 0;
638 if( TListTypeTraits<ItemType>::IsPtrValid(SelectorItem) )
639 {
640 SelectionIndex = ItemsSourceRef.Find( TListTypeTraits<ItemType>::NullableItemTypeConvertToItemType( SelectorItem ) );
641 }
642
643 int32 NumItemsInAPage = FMath::TruncToInt(GetNumLiveWidgets());
644 int32 Remainder = NumItemsInAPage % GetNumItemsPerLine();
645 NumItemsInAPage -= Remainder;
646
647 if ( SelectionIndex < ItemsSourceRef.Num() - NumItemsInAPage )
648 {
649 // Select an item on the next page
650 ItemNavigatedTo = ItemsSourceRef[SelectionIndex + NumItemsInAPage];
651 }
652 else
653 {
654 ItemNavigatedTo = ItemsSourceRef.Last();
655 }
656
657 bWasHandled = true;
658 }
659
660 if( TListTypeTraits<ItemType>::IsPtrValid(ItemNavigatedTo) )
661 {
662 ItemType ItemToSelect( TListTypeTraits<ItemType>::NullableItemTypeConvertToItemType( ItemNavigatedTo ) );
663 NavigationSelect( ItemToSelect, InKeyEvent );
664 }
665 else
666 {
667 // Change selected status of item.
668 if (bHandleSpacebarSelection && TListTypeTraits<ItemType>::IsPtrValid(SelectorItem) && InKeyEvent.GetKey() == EKeys::SpaceBar)
669 {
670 ItemType SelectorItemDereference(TListTypeTraits<ItemType>::NullableItemTypeConvertToItemType(SelectorItem));
671
672 // Deselect.
673 if (InKeyEvent.IsControlDown() || GetSelectionMode() == ESelectionMode::SingleToggle)
674 {
675 this->Private_SetItemSelection(SelectorItemDereference, !(this->Private_IsItemSelected(SelectorItemDereference)), true);
676 this->Private_SignalSelectionChanged(ESelectInfo::OnKeyPress);
677 bWasHandled = true;
678 }
679 else
680 {
681 // Already selected, don't handle.
682 if (this->Private_IsItemSelected(SelectorItemDereference))
683 {
684 bWasHandled = false;
685 }
686 // Select.
687 else
688 {
689 this->Private_SetItemSelection(SelectorItemDereference, true, true);
690 this->Private_SignalSelectionChanged(ESelectInfo::OnKeyPress);
691 bWasHandled = true;
692 }
693 }
694
695 RangeSelectionStart = SelectorItem;
696
697 // If the selector is not in the view, scroll it into view.
698 TSharedPtr<ITableRow> WidgetForItem = this->WidgetGenerator.GetWidgetForItem(SelectorItemDereference);
699 if (!WidgetForItem.IsValid())
700 {
701 this->RequestScrollIntoView(SelectorItemDereference, InKeyEvent.GetUserIndex());
702 }
703 }
704 // Select all items
705 else if ( (!InKeyEvent.IsShiftDown() && !InKeyEvent.IsAltDown() && InKeyEvent.IsControlDown() && InKeyEvent.GetKey() == EKeys::A) && GetSelectionMode() == ESelectionMode::Multi )
706 {
707 this->Private_ClearSelection();
708
709 for ( int32 ItemIdx = 0; ItemIdx < ItemsSourceRef.Num(); ++ItemIdx )
710 {
711 this->Private_SetItemSelection( ItemsSourceRef[ItemIdx], true );
712 }
713
714 this->Private_SignalSelectionChanged(ESelectInfo::OnKeyPress);
715
716 bWasHandled = true;
717 }
718 }
719
720 if (bWasHandled)
721 {
722 return FReply::Handled();
723 }
724 }
725
726 return STableViewBase::OnKeyDown(MyGeometry, InKeyEvent);
727 }

730 virtual FNavigationReply OnNavigation(const FGeometry& MyGeometry, const FNavigationEvent& InNavigationEvent) override
731 {
732 NavigationTypeOnScrollIntoView = EUINavigation::Invalid;
733
734 if (this->HasValidItemsSource() && this->bHandleDirectionalNavigation && (this->bHandleGamepadEvents || InNavigationEvent.GetNavigationGenesis() != ENavigationGenesis::Controller))
735 {
736 const TArrayView<const ItemType> ItemsSourceRef = this->GetItems();
737 const int32 NumItems = ItemsSourceRef.Num();
738
739 const int32 NumItemsPerLine = GetNumItemsPerLine();
740 const int32 CurSelectionIndex = (!TListTypeTraits<ItemType>::IsPtrValid(SelectorItem)) ? -1 : ItemsSourceRef.Find(TListTypeTraits<ItemType>::NullableItemTypeConvertToItemType(SelectorItem));
741 int32 AttemptSelectIndex = -1;
742
743 const EUINavigation NavType = InNavigationEvent.GetNavigationType();
744 const bool bNavBackward = (Orientation == Orient_Vertical && NavType == EUINavigation::Up) ||
745 (Orientation == Orient_Horizontal && NavType == EUINavigation::Left);
746 const bool bNavForward = (Orientation == Orient_Vertical && NavType == EUINavigation::Down) ||
747 (Orientation == Orient_Horizontal && NavType == EUINavigation::Right);
748 // Any of the four directional nav types - includes cross-axis directions (e.g. Down on a
749 // horizontal list) that leave bNavForward and bNavBackward both false.
750 const bool bIsAnyDirectionalNav = NavType == EUINavigation::Up || NavType == EUINavigation::Down ||
751 NavType == EUINavigation::Left || NavType == EUINavigation::Right;
752
753 if (bNavBackward)
754 {
755 // Nav backward by a line
756 AttemptSelectIndex = CurSelectionIndex - NumItemsPerLine;
757 }
758 else if (bNavForward)
759 {
760 AttemptSelectIndex = CurSelectionIndex + NumItemsPerLine;
761
762 // The list might be jagged so attempt to determine if there's a partially filled row we can move to
763 if (!ItemsSourceRef.IsValidIndex(AttemptSelectIndex))
764 {
765 if (NumItems > 0)
766 {
767 // NumItemsWide should never be 0, ensuring for sanity
768 ensure(NumItemsPerLine > 0);
769
770 // calculate total number of rows and row of current index (1 index)
771 const int32 NumLines = FMath::CeilToInt((float)NumItems / (float)NumItemsPerLine);
772 const int32 CurLine = FMath::CeilToInt((float)(CurSelectionIndex + 1) / (float)NumItemsPerLine);
773
774 // if not on final row, assume a jagged list and select the final item
775 if (CurLine < NumLines)
776 {
777 AttemptSelectIndex = NumItems - 1;
778 }
779 }
780 }
781 }
782
783 // Cross-list entry with no current selection: select the proximate boundary item.
784 if (bEnableProximateEntryNavigation && CurSelectionIndex == -1 && bIsAnyDirectionalNav && NumItems > 0)
785 {
786 ensure(NumItemsPerLine > 0);
787 const TSharedPtr<UE::Slate::Private::FListViewNavigationContext> NavContext = FSlateApplication::Get().GetPendingNavigationContext<UE::Slate::Private::FListViewNavigationContext>(InNavigationEvent.GetUserIndex());
788 if (NavContext.IsValid())
789 {
790 const bool bNavForwardForBoundary = NavType == EUINavigation::Down || NavType == EUINavigation::Right;
791 AttemptSelectIndex = FindProximateBoundaryItem(NavContext->SourceWidget.Pin(), NavContext->NavigationType, ItemsSourceRef, NumItems, NumItemsPerLine, bNavForwardForBoundary);
792 FSlateApplication::Get().ClearPendingNavigationContext(InNavigationEvent.GetUserIndex());
793 }
794 else if (NavType == EUINavigation::Up || NavType == EUINavigation::Left)
795 {
796 // Backward entry with no context: land on the first item of the last row.
797 AttemptSelectIndex = ((NumItems - 1) / NumItemsPerLine) * NumItemsPerLine;
798 }
799 // Forward without context: falls through to CurSelectionIndex + NumItemsPerLine
800 // (computed above), which equals NumItemsPerLine - 1 when CurSelectionIndex is -1.
801 }
802
803 // If it's valid we'll scroll it into view and return an explicit widget in the FNavigationReply.
804 if (ItemsSourceRef.IsValidIndex(AttemptSelectIndex) && this->bIsGamepadScrollingEnabled)
805 {
806 TOptional<ItemType> ItemToSelect = Private_FindNextSelectableOrNavigableWithIndexAndDirection(ItemsSourceRef[AttemptSelectIndex], AttemptSelectIndex, AttemptSelectIndex >= CurSelectionIndex);
807 if (ItemToSelect.IsSet())
808 {
809 NavigationSelect(ItemToSelect.GetValue(), InNavigationEvent);
810 return FNavigationReply::Explicit(nullptr);
811 }
812 }
813
814 // Escaping this list's boundary: write the navigation context for the receiving list.
815 if (bEnableProximateEntryNavigation && bIsAnyDirectionalNav)
816 {
817 TSharedRef<UE::Slate::Private::FListViewNavigationContext> NavContext = MakeShared<UE::Slate::Private::FListViewNavigationContext>();
818 NavContext->NavigationType = NavType;
819
820 // Capture the source widget for cross-axis proximity calculations.
821 if (TListTypeTraits<ItemType>::IsPtrValid(SelectorItem))
822 {
823 const ItemType CurrentItem = TListTypeTraits<ItemType>::NullableItemTypeConvertToItemType(SelectorItem);
824 if (const TSharedPtr<ITableRow> SelectedRow = this->WidgetFromItem(CurrentItem))
825 {
826 NavContext->SourceWidget = SelectedRow->AsWidget();
827 }
828 }
829 FSlateApplication::Get().SetPendingNavigationContext(InNavigationEvent.GetUserIndex(), NavContext);
830 }
831 }
832
833 return STableViewBase::OnNavigation(MyGeometry, InNavigationEvent);
834 }

1056 void ProcessItemCleanUp()
1057 {
1058
1059 for (int32 ItemIndex = 0; ItemIndex < ItemsToBeCleanedUp.Num(); ++ItemIndex)
1060 {
1061 ItemType ItemToBeCleanedUp = ItemsToBeCleanedUp[ItemIndex];
1062 const TSharedRef<ITableRow>* FindResult = ItemToWidgetMap.Find(ItemToBeCleanedUp);
1063 if (FindResult != nullptr)
1064 {
1065 const TSharedRef<ITableRow> WidgetToCleanUp = *FindResult;
1066 ItemToWidgetMap.Remove(ItemToBeCleanedUp);
1067 WidgetMapToItem.Remove(&WidgetToCleanUp.Get());
1068
1069 if (ensureMsgf(OwnerList, TEXT("OwnerList is null, something is wrong.")))
1070 {
1071 WidgetToCleanUp->ResetRow();
1072 OwnerList->OnRowReleased.ExecuteIfBound(WidgetToCleanUp);
1073 }
1074 }
1075 else if(!TListTypeTraits<ItemType>::IsPtrValid(ItemToBeCleanedUp))
1076 {
1077 // If we get here, it means we have an invalid object. We will need to remove that object from both maps.
1078 // This may happen for example when ItemType is a UObject* and the object is garbage collected.
1079 auto Widget = WidgetMapToItem.FindKey(ItemToBeCleanedUp);
1080 if (Widget != nullptr)
1081 {
1082 for (auto WidgetItemPair = ItemToWidgetMap.CreateIterator(); WidgetItemPair; ++WidgetItemPair)
1083 {
1084 const ITableRow* Item = &(WidgetItemPair.Value().Get());
1085 if (Item == *Widget)
1086 {
1087 WidgetItemPair.RemoveCurrent();
1088 break;
1089 }
1090 }
1091
1092 WidgetMapToItem.Remove(*Widget);
1093 }
1094 }
1095 }
1096
1097 ItemsToBeCleanedUp.Reset();

1514 virtual FReGenerateResults ReGenerateItems( const FGeometry& MyGeometry ) override
1515 {
1516 auto DoubleFractional = [](double Value) -> double
1517 {
1518 return Value - FMath::TruncToDouble(Value);
1519 };
1520
1521 // Clear all the items from our panel. We will re-add them in the correct order momentarily.
1522 this->ClearWidgets();
1523
1524 // Ensure that we always begin and clean up a generation pass.
1525 FGenerationPassGuard GenerationPassGuard(WidgetGenerator);
1526
1527 const TArrayView<const ItemType> Items = GetItems();
1528 if (Items.Num() > 0)
1529 {
1530 // Items in view, including fractional items
1531 float ItemsInView = 0.0f;
1532
1533 // Total length of widgets generated so far (height for vertical lists, width for horizontal)
1534 float LengthGeneratedSoFar = 0.0f;
1535
1536 // Length of generated widgets that in the bounds of the view.
1537 float ViewLengthUsedSoFar = 0.0f;
1538
1539 // Index of the item at which we start generating based on how far scrolled down we are
1540 // Note that we must generate at LEAST one item.
1541 int32 StartIndex = FMath::Clamp( (int32)(FMath::FloorToDouble(CurrentScrollOffset)), 0, Items.Num() - 1 );
1542
1543 // Length of the first item that is generated. This item is at the location where the user requested we scroll
1544 float FirstItemLength = 0.0f;
1545
1546 // Generate widgets assuming scenario a.
1547 bool bHasFilledAvailableArea = false;
1548 bool bAtEndOfList = false;
1549
1550 const float LayoutScaleMultiplier = MyGeometry.GetAccumulatedLayoutTransform().GetScale();
1551 FTableViewDimensions MyDimensions(this->Orientation, MyGeometry.GetLocalSize());
1552
1553 for( int32 ItemIndex = StartIndex; !bHasFilledAvailableArea && ItemIndex < Items.Num(); ++ItemIndex )
1554 {
1555 const ItemType& CurItem = Items[ItemIndex];
1556
1557 if (!TListTypeTraits<ItemType>::IsPtrValid(CurItem))
1558 {
1559 // Don't bother generating widgets for invalid items
1560 continue;
1561 }
1562
1563 const float ItemLength = GenerateWidgetForItem(CurItem, ItemIndex, StartIndex, LayoutScaleMultiplier);
1564
1565 const bool bIsFirstItem = ItemIndex == StartIndex;
1566
1567 if (bIsFirstItem)
1568 {
1569 FirstItemLength = ItemLength;
1570 }
1571
1572 // Track the number of items in the view, including fractions.
1573 if (bIsFirstItem)
1574 {
1575 // The first item may not be fully visible (but cannot exceed 1)
1576 // FirstItemFractionScrolledIntoView is the fraction of the item that is visible after taking into account anything that may be scrolled off the top/left of the list view
1577 const float FirstItemFractionScrolledIntoView = 1.0f - (float)FMath::Max(DoubleFractional(CurrentScrollOffset), 0.0);
1578
1579 // FirstItemLengthScrolledIntoView is the length of the item, ignoring anything that is scrolled off the top/left of the list view
1580 const float FirstItemLengthScrolledIntoView = ItemLength * FirstItemFractionScrolledIntoView;
1581
1582 // FirstItemVisibleFraction is either: The visible item length as a fraction of the available list view length (if the item size is larger than the available size, otherwise this will be >1), or just FirstItemFractionScrolledIntoView (which can never be >1)
1583 const float FirstItemVisibleFraction = FMath::Min(MyDimensions.ScrollAxis / FirstItemLengthScrolledIntoView, FirstItemFractionScrolledIntoView);
1584
1585 ItemsInView += FirstItemVisibleFraction;
1586 }
1587 else if (ViewLengthUsedSoFar + ItemLength > MyDimensions.ScrollAxis)
1588 {
1589 // The last item may not be fully visible either
1590 ItemsInView += (MyDimensions.ScrollAxis - ViewLengthUsedSoFar) / ItemLength;
1591 }
1592 else
1593 {
1594 ItemsInView += 1;
1595 }
1596
1597 LengthGeneratedSoFar += ItemLength;
1598
1599 ViewLengthUsedSoFar += (bIsFirstItem)
1600 ? ItemLength * ItemsInView // For the first item, ItemsInView <= 1.0f
1601 : ItemLength;
1602
1603 bAtEndOfList = ItemIndex >= Items.Num() - 1;
1604
1605 if (bIsFirstItem && ViewLengthUsedSoFar >= MyDimensions.ScrollAxis)
1606 {
1607 // Since there was no sum of floating points, make sure we correctly detect the case where one element
1608 // fills up the space.
1609 bHasFilledAvailableArea = true;
1610 }
1611 else
1612 {
1613 // Note: To account for accrued error from floating point truncation and addition in our sum of dimensions used,
1614 // we pad the allotted axis just a little to be sure we have filled the available space.
1615 const float FloatPrecisionOffset = 0.001f;
1616 bHasFilledAvailableArea = ViewLengthUsedSoFar >= MyDimensions.ScrollAxis + FloatPrecisionOffset;
1617 }
1618 }
1619
1620 // Handle scenario b.
1621 // We may have stopped because we got to the end of the items, but we may still have space to fill!
1622 if (bAtEndOfList && !bHasFilledAvailableArea)
1623 {
1624 double NewScrollOffsetForBackfill = static_cast<double>(StartIndex) + (LengthGeneratedSoFar - MyDimensions.ScrollAxis) / FirstItemLength;
1625
1626 for (int32 ItemIndex = StartIndex - 1; LengthGeneratedSoFar < MyDimensions.ScrollAxis && ItemIndex >= 0; --ItemIndex)
1627 {
1628 const ItemType& CurItem = Items[ItemIndex];
1629 if (TListTypeTraits<ItemType>::IsPtrValid(CurItem))
1630 {
1631 const float ItemLength = GenerateWidgetForItem(CurItem, ItemIndex, StartIndex, LayoutScaleMultiplier);
1632
1633 if (LengthGeneratedSoFar + ItemLength > MyDimensions.ScrollAxis && ItemLength > 0.f)
1634 {
1635 // Generated the item that puts us over the top.
1636 // Count the fraction of this item that will stick out beyond the list
1637 NewScrollOffsetForBackfill = static_cast<double>(ItemIndex) + (LengthGeneratedSoFar + ItemLength - MyDimensions.ScrollAxis) / ItemLength;
1638 }
1639
1640 // The widget used up some of the available vertical space.
1641 LengthGeneratedSoFar += ItemLength;
1642 }
1643 }
1644
1645 return FReGenerateResults(NewScrollOffsetForBackfill, LengthGeneratedSoFar, Items.Num() - NewScrollOffsetForBackfill, true);
1646 }
1647
1648 return FReGenerateResults(CurrentScrollOffset, LengthGeneratedSoFar, ItemsInView, false);
1649 }
1650
1651 return FReGenerateResults(0.0f, 0.0f, 0.0f, false);
1652 }

1654 float GenerateWidgetForItem( const ItemType& CurItem, int32 ItemIndex, int32 StartIndex, float LayoutScaleMultiplier )
1655 {
1656 ensure(TListTypeTraits<ItemType>::IsPtrValid(CurItem));
1657 // Find a previously generated Widget for this item, if one exists.
1658 TSharedPtr<ITableRow> WidgetForItem = WidgetGenerator.GetWidgetForItem( CurItem );
1659 if ( !WidgetForItem.IsValid() )
1660 {
1661 // We couldn't find an existing widgets, meaning that this data item was not visible before.
1662 // Make a new widget for it.
1663 WidgetForItem = this->GenerateNewWidget(CurItem);
1664 }
1665 else
1666 {
1667 if (OnRefreshRow.IsBound())
1668 {
1669 OnRefreshRow.Execute(CurItem);
1670 }
1671 }
1672
1673 // It is useful to know the item's index that the widget was generated from.
1674 // Helps with even/odd coloring
1675 WidgetForItem->SetIndexInList(ItemIndex);
1676
1677 // Let the item generator know that we encountered the current Item and associated Widget.
1678 WidgetGenerator.OnItemSeen( CurItem, WidgetForItem.ToSharedRef() );
1679
1680 // We rely on the widgets desired size in order to determine how many will fit on screen.
1681 const TSharedRef<SWidget> NewlyGeneratedWidget = WidgetForItem->AsWidget();
1682 NewlyGeneratedWidget->MarkPrepassAsDirty();
1683 NewlyGeneratedWidget->SlatePrepass(LayoutScaleMultiplier);
1684
1685 // We have a widget for this item; add it to the panel so that it is part of the UI.
1686 if (ItemIndex >= StartIndex)
1687 {
1688 // Generating widgets downward
1689 this->AppendWidget( WidgetForItem.ToSharedRef() );
1690 }
1691 else
1692 {
1693 // Backfilling widgets; going upward
1694 this->InsertWidget( WidgetForItem.ToSharedRef() );
1695 }
1696
1697 const bool bIsVisible = NewlyGeneratedWidget->GetVisibility().IsVisible();
1698 FTableViewDimensions GeneratedWidgetDimensions(this->Orientation, bIsVisible ? NewlyGeneratedWidget->GetDesiredSize() : FVector2D::ZeroVector);
1699 return GeneratedWidgetDimensions.ScrollAxis;

2533 virtual float ScrollBy( const FGeometry& MyGeometry, float ScrollByAmountInSlateUnits, EAllowOverscroll InAllowOverscroll ) override
2534 {
2535 auto DoubleFractional = [](double Value) -> double
2536 {
2537 return Value - FMath::TruncToDouble(Value);
2538 };
2539
2540 if (InAllowOverscroll == EAllowOverscroll::No)
2541 {
2542 //check if we are on the top of the list and want to scroll up
2543 if (DesiredScrollOffset < UE_KINDA_SMALL_NUMBER && ScrollByAmountInSlateUnits < 0)
2544 {
2545 return 0.0f;
2546 }
2547
2548 //check if we are on the bottom of the list and want to scroll down
2549 if (bWasAtEndOfList && ScrollByAmountInSlateUnits > 0)
2550 {
2551 return 0.0f;
2552 }
2553 }
2554
2555 float AbsScrollByAmount = FMath::Abs( ScrollByAmountInSlateUnits );
2556 int32 StartingItemIndex = (int32)CurrentScrollOffset;
2557 double NewScrollOffset = DesiredScrollOffset;
2558
2559 const bool bWholeListVisible = DesiredScrollOffset == 0.0 && bWasAtEndOfList;
2560 if ( InAllowOverscroll == EAllowOverscroll::Yes && Overscroll.ShouldApplyOverscroll(DesiredScrollOffset == 0.0, bWasAtEndOfList, ScrollByAmountInSlateUnits ) )
2561 {
2562 const float UnclampedScrollDelta = FMath::Sign(ScrollByAmountInSlateUnits) * AbsScrollByAmount;
2563 const float ActuallyScrolledBy = Overscroll.ScrollBy(MyGeometry, UnclampedScrollDelta);
2564 if (ActuallyScrolledBy != 0.0f)
2565 {
2566 this->RequestLayoutRefresh();
2567 }
2568 return ActuallyScrolledBy;
2569 }
2570 else if (!bWholeListVisible)
2571 {
2572 // We know how far we want to scroll in SlateUnits, but we store scroll offset in "number of widgets".
2573 // Challenge: each widget can be a different height/width.
2574 // Strategy:
2575 // Scroll "one widget's length" at a time until we've scrolled as far as the user asked us to.
2576 // Generate widgets on demand so we can figure out how big they are.
2577
2578 const TArrayView<const ItemType> Items = GetItems();
2579 if (Items.Num() > 0)
2580 {
2581 int32 ItemIndex = StartingItemIndex;
2582 const float LayoutScaleMultiplier = MyGeometry.GetAccumulatedLayoutTransform().GetScale();
2583 while( AbsScrollByAmount != 0 && ItemIndex < Items.Num() && ItemIndex >= 0 )
2584 {
2585 const ItemType& CurItem = Items[ItemIndex];
2586 if (!TListTypeTraits<ItemType>::IsPtrValid(CurItem))
2587 {
2588 // If the CurItem is not valid, we do not generate a new widget for it, we skip it.
2589 ++ItemIndex;
2590 continue;
2591 }
2592
2593 TSharedPtr<ITableRow> RowWidget = WidgetGenerator.GetWidgetForItem( CurItem );
2594 if (!RowWidget.IsValid())
2595 {
2596 // We couldn't find an existing widgets, meaning that this data item was not visible before.
2597 // Make a new widget for it.
2598 RowWidget = this->GenerateNewWidget( CurItem );
2599
2600 // It is useful to know the item's index that the widget was generated from.
2601 // Helps with even/odd coloring
2602 RowWidget->SetIndexInList(ItemIndex);
2603
2604 // Let the item generator know that we encountered the current Item and associated Widget.
2605 WidgetGenerator.OnItemSeen( CurItem, RowWidget.ToSharedRef() );
2606
2607 RowWidget->AsWidget()->SlatePrepass(LayoutScaleMultiplier);
2608 }
2609
2610 const FTableViewDimensions WidgetDimensions(this->Orientation, RowWidget->AsWidget()->GetDesiredSize());
2611 if (ScrollByAmountInSlateUnits > 0)
2612 {
2613 const float RemainingDistance = WidgetDimensions.ScrollAxis * (float)(1.0 - DoubleFractional(NewScrollOffset));
2614
2615 if (AbsScrollByAmount > RemainingDistance)
2616 {
2617 if (ItemIndex != Items.Num())
2618 {
2619 AbsScrollByAmount -= RemainingDistance;
2620 NewScrollOffset = 1.0 + (int32)NewScrollOffset;
2621 ++ItemIndex;
2622 }
2623 else
2624 {
2625 NewScrollOffset = Items.Num();
2626 break;
2627 }
2628 }
2629 else if ( AbsScrollByAmount == RemainingDistance)
2630 {
2631 NewScrollOffset = 1.0 + (int32)NewScrollOffset;
2632 break;
2633 }
2634 else
2635 {
2636 NewScrollOffset = (int32)NewScrollOffset + (1.0 - ((RemainingDistance - AbsScrollByAmount) / WidgetDimensions.ScrollAxis));
2637 break;
2638 }
2639 }
2640 else
2641 {
2642 float Fractional = FMath::Fractional( (float)NewScrollOffset );
2643 if ( FMath::IsNearlyEqual(Fractional, 0.f) )
2644 {
2645 Fractional = 1.0f;
2646 --NewScrollOffset;
2647 }
2648
2649 const float PrecedingDistance = WidgetDimensions.ScrollAxis * Fractional;
2650
2651 if ( AbsScrollByAmount > PrecedingDistance)
2652 {
2653 if ( ItemIndex != 0 )
2654 {
2655 AbsScrollByAmount -= PrecedingDistance;
2656 NewScrollOffset -= DoubleFractional( NewScrollOffset );
2657 --ItemIndex;
2658 }
2659 else
2660 {
2661 NewScrollOffset = 0.0;
2662 break;
2663 }
2664 }
2665 else if ( AbsScrollByAmount == PrecedingDistance)
2666 {
2667 NewScrollOffset -= DoubleFractional( NewScrollOffset );
2668 break;
2669 }
2670 else
2671 {
2672 NewScrollOffset = float(FMath::TruncToInt32(NewScrollOffset)) + ((PrecedingDistance - AbsScrollByAmount) / WidgetDimensions.ScrollAxis);
2673 break;
2674 }
2675 }
2676 }
2677 }
2678
2679
2680 return ScrollTo( (float)NewScrollOffset );
2681 }
2682
2683 return 0;
2684 }

2861 int32 FindProximateBoundaryItem(TSharedPtr<SWidget> SourceWidget, EUINavigation InNavigationType, const TArrayView<const ItemType>& ItemsSourceRef, int32 NumItems, int32 NumItemsPerLine, bool bNavForward) const
2862 {
2863 int32 EffectiveNumItems = NumItems;
2864 if (!bNavForward)
2865 {
2866 for (int32 i = NumItems - 1; i >= 0; --i)
2867 {
2868 if (this->WidgetFromItem(ItemsSourceRef[i]).IsValid())
2869 {
2870 EffectiveNumItems = i + 1;
2871 break;
2872 }
2873 }
2874 }
2875
2876 if (!SourceWidget.IsValid())
2877 {
2878 return bNavForward ? 0 : ((EffectiveNumItems - 1) / NumItemsPerLine) * NumItemsPerLine;
2879 }
2880
2881 // The navigation direction determines which axis we compare for proximity - perpendicular
2882 // to the direction of travel. Down/Up -> preserve X (horizontal position).
2883 // Left/Right -> preserve Y (vertical position).
2884 const EUINavigation NavType = InNavigationType;
2885 const bool bCompareOnXAxis = (NavType == EUINavigation::Down || NavType == EUINavigation::Up);
2886
2887 // For cross-axis navigation (e.g. Down into a horizontal list), all items share the same
2888 // position on the nav axis, so all items are candidates. For primary-axis navigation, only
2889 // the boundary row/column is searched.
2890 const bool bNavIsCrossAxis =
2891 (Orientation == Orient_Vertical && (NavType == EUINavigation::Left || NavType == EUINavigation::Right)) ||
2892 (Orientation == Orient_Horizontal && (NavType == EUINavigation::Up || NavType == EUINavigation::Down));
2893
2894 const int32 BoundaryLineStart = bNavIsCrossAxis ? 0
2895 : bNavForward ? 0
2896 : ((EffectiveNumItems - 1) / NumItemsPerLine) * NumItemsPerLine;
2897 const int32 BoundaryLineEnd = bNavIsCrossAxis ? EffectiveNumItems - 1
2898 : bNavForward ? FMath::Min(NumItemsPerLine, EffectiveNumItems) - 1
2899 : EffectiveNumItems - 1;
2900
2901 const FSlateRect SourceBounds = SourceWidget->GetCachedGeometry().GetRenderBoundingRect();
2902 const float SourceCrossCenter = bCompareOnXAxis
2903 ? (SourceBounds.Left + SourceBounds.Right) * 0.5f
2904 : (SourceBounds.Top + SourceBounds.Bottom) * 0.5f;
2905
2906 // Pass 1: search rendered boundary items
2907 int32 BestIndex = BoundaryLineStart;
2908 float BestDist = TNumericLimits<float>::Max();
2909
2910 for (int32 Idx = BoundaryLineStart; Idx <= BoundaryLineEnd; ++Idx)
2911 {
2912 const TSharedPtr<ITableRow> Row = this->WidgetFromItem(ItemsSourceRef[Idx]);
2913 if (!Row.IsValid())
2914 {
2915 continue;
2916 }
2917 const FSlateRect RowBounds = Row->AsWidget()->GetCachedGeometry().GetRenderBoundingRect();
2918 const float ItemCrossCenter = bCompareOnXAxis
2919 ? (RowBounds.Left + RowBounds.Right) * 0.5f
2920 : (RowBounds.Top + RowBounds.Bottom) * 0.5f;
2921 const float Dist = FMath::Abs(ItemCrossCenter - SourceCrossCenter);
2922 if (Dist < BestDist)
2923 {
2924 BestDist = Dist;
2925 BestIndex = Idx;
2926 }
2927 }
2928
2929 if (BestDist < TNumericLimits<float>::Max())
2930 {
2931 return BestIndex;
2932 }
2933
2934 // Pass 2: off-screen fallback - map source position to a slot index
2935 if (NumItemsPerLine > 1)
2936 {
2937 const FSlateRect ListBounds = this->GetCachedGeometry().GetRenderBoundingRect();
2938 const float ListExtent = bCompareOnXAxis
2939 ? FMath::Max(ListBounds.Right - ListBounds.Left, 1.0f)
2940 : FMath::Max(ListBounds.Bottom - ListBounds.Top, 1.0f);
2941 const float ListOrigin = bCompareOnXAxis ? ListBounds.Left : ListBounds.Top;
2942 const float NormalizedPos = FMath::Clamp((SourceCrossCenter - ListOrigin) / ListExtent, 0.0f, 1.0f);
2943 const int32 ColumnIdx = FMath::Clamp(FMath::FloorToInt(NormalizedPos * static_cast<float>(NumItemsPerLine)), 0, NumItemsPerLine - 1);
2944 const int32 CandidateIdx = BoundaryLineStart + ColumnIdx;
2945 if (ItemsSourceRef.IsValidIndex(CandidateIdx))
2946 {
2947 return CandidateIdx;
2948 }
2949 }
2950
