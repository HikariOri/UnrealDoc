50UE_AUTORTFM_FORCEINLINE void FContext::DidFree(void* LogicalAddress)
51{
52 // We can do free's in the open within a transaction *during* when the
53 // transaction itself is being destroyed, so we need to check for that case.
54 FTransaction* Transaction = GetCurrentTransaction();
55 if (AUTORTFM_LIKELY(Transaction))

61UE_AUTORTFM_FORCEINLINE void FContext::DeferUntilComplete(TTask<void()>&& Callback)
62{
63 AUTORTFM_ASSERT(Status == EContextStatus::OnTrack);
64
65 // We explicitly must copy the function here because the original was allocated
66 // within a transactional context, and thus the memory is allocating under
67 // transactionalized conditions. By copying, we create an open copy of the callback.
68 TTask<void()> Copy(Callback);
69 CompletionTasks.Add(std::move(Copy));
70}
