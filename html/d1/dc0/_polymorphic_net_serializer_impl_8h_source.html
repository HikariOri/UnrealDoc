283void TPolymorphicStructNetSerializerImpl<ExternalSourceType, ExternalSourceItemType, GetItem>::SerializeDelta(FNetSerializationContext& Context, const FNetSerializeDeltaArgs& Args)
284{
285 FNetBitStreamWriter& Writer = *Context.GetBitStreamWriter();
286 const QuantizedType& Value = *reinterpret_cast<const QuantizedType*>(Args.Source);
287 const QuantizedType& PrevValue = *reinterpret_cast<const QuantizedType*>(Args.Prev);
288 const ConfigType& Config = *static_cast<const ConfigType*>(Args.NetSerializerConfig);
289
290 // If prev has the same type we can delta-compress
291 if (Writer.WriteBool(Value.StructName == PrevValue.StructName))
292 {
293 const FTypeInfo* TypeInfo = Config.RegisteredTypes.GetTypeInfo(Value.StructName);
294 if (TypeInfo != nullptr)
295 {
296 UE::Net::FReplicationStateOperations::SerializeDelta(Context, static_cast<const uint8*>(Value.StructData), static_cast<const uint8*>(PrevValue.StructData), TypeInfo->Descriptor);
297 }
298 }
299 else
300 {
301 const FTypeInfo* TypeInfo = Config.RegisteredTypes.GetTypeInfo(Value.StructName);
302 if (Writer.WriteBool(TypeInfo != nullptr))
303 {
304 CA_ASSUME(TypeInfo != nullptr);
305 FNetTokenStore* TokenStore = Context.GetNetTokenStore();
306 FNameTokenStore* NameStore = TokenStore->GetDataStore<FNameTokenStore>();
307 const FNetToken NameToken = NameStore->GetOrCreateToken(Value.StructName);
308 TokenStore->WriteNetTokenWithKnownType<FNameTokenStore>(Context, NameToken);
309 FNetTokenStore::AppendExport(Context, NameToken);
310 UE::Net::FReplicationStateOperations::Serialize(Context, static_cast<const uint8*>(Value.StructData), TypeInfo->Descriptor);
311 }
312 }

377 QuantizedType& TargetValue = *reinterpret_cast<QuantizedType*>(Args.Target);
378 const ConfigType& Config = *static_cast<const ConfigType*>(Args.NetSerializerConfig);

380 InternalFreeItem(Context, Config, TargetValue);
381
382 const TSharedPtr<SourceItemType>& Item = GetItem(SourceValue);
383 const UScriptStruct* ScriptStruct = Item.IsValid() ? Item->GetScriptStruct() : nullptr;

386 // Quantize polymorphic data

388 if (TypeInfo != nullptr)

392 TempValue.StructName = TypeInfo->StructName;

397 }
398 else
399 {
400 if (ScriptStruct)
401 {
402 Context.SetError(NetError_PolymorphicStructNetSerializer_InvalidStructType);
403 UE_LOGF(LogIris, Warning, "TPolymorphicStructNetSerializerImpl::Quantize Trying to quantize unregistered ScriptStruct type %ls.", ToCStr(ScriptStruct->GetName()));
404 }
405 }
406
407 TargetValue = TempValue;

410template <typename ExternalSourceType, typename ExternalSourceItemType, TSharedPtr<ExternalSourceItemType>&(*GetItem)(ExternalSourceType&)>
411void TPolymorphicStructNetSerializerImpl<ExternalSourceType, ExternalSourceItemType, GetItem>::Dequantize(FNetSerializationContext& Context, const FNetDequantizeArgs& Args)
412{
413 const QuantizedType& SourceValue = *reinterpret_cast<const QuantizedType*>(Args.Source);
414 SourceType& TargetValue = *reinterpret_cast<SourceType*>(Args.Target);
415 const ConfigType& Config = *static_cast<const ConfigType*>(Args.NetSerializerConfig);
416
417 // Dequantize polymorphic data
418 if (const FTypeInfo* TypeInfo = Config.RegisteredTypes.GetTypeInfo(SourceValue.StructName))
419 {
420 const FReplicationStateDescriptor* Descriptor = TypeInfo->Descriptor;
421 const UScriptStruct* ScriptStruct = TypeInfo->ScriptStruct;
422
423 // NOTE: We always allocate new memory in order to behave like the code we are trying to mimic expects, see GameplayEffectContextHandle
424 // this should really be a policy of this class as it is far from optimal.
425
426 // Allocate external struct, owned by external state therefore using global allocator
427 SourceItemType* NewData = static_cast<SourceItemType*>(FMemory::Malloc(ScriptStruct->GetStructureSize(), ScriptStruct->GetMinAlignment()));
428 ScriptStruct->InitializeStruct(NewData);
429
430 FReplicationStateOperations::Dequantize(Context, reinterpret_cast<uint8*>(NewData), static_cast<const uint8*>(SourceValue.StructData), Descriptor);
431
432 GetItem(TargetValue) = MakeShareable(NewData, FSourceItemTypeDeleter());
433 }
434 else
435 {
436 GetItem(TargetValue).Reset();
437 }

440template <typename ExternalSourceType, typename ExternalSourceItemType, TSharedPtr<ExternalSourceItemType>&(*GetItem)(ExternalSourceType&)>
441bool TPolymorphicStructNetSerializerImpl<ExternalSourceType, ExternalSourceItemType, GetItem>::IsEqual(FNetSerializationContext& Context, const FNetIsEqualArgs& Args)
442{
443 if (Args.bStateIsQuantized)
444 {
445 const ConfigType& Config = *static_cast<const ConfigType*>(Args.NetSerializerConfig);
446 const QuantizedType& ValueA = *reinterpret_cast<const QuantizedType*>(Args.Source0);
447 const QuantizedType& ValueB = *reinterpret_cast<const QuantizedType*>(Args.Source1);
448
449 if (ValueA.StructName != ValueB.StructName)
450 {
451 return false;
452 }
453
454 const FTypeInfo* TypeInfo = Config.RegisteredTypes.GetTypeInfo(ValueA.StructName);
455 if (TypeInfo != nullptr && !UE::Net::FReplicationStateOperations::IsEqualQuantizedState(Context, static_cast<const uint8*>(ValueA.StructData), static_cast<const uint8*>(ValueB.StructData), TypeInfo->Descriptor))
456 {
457 return false;
458 }
459 }
460 else
461 {
462 const SourceType& ValueA = *reinterpret_cast<const SourceType*>(Args.Source0);
463 const SourceType& ValueB = *reinterpret_cast<const SourceType*>(Args.Source1);
464
465 // Assuming there's a custom operator== because if there's not we would be hitting TSharedRef== which checks if the reference is identical, not the instance data.
466 return ValueA == ValueB;
467 }
468
469 return true;
470}

569template <typename ExternalSourceType, typename ExternalSourceArrayItemType, TArrayView<TSharedPtr<ExternalSourceArrayItemType>>(*GetArray)(ExternalSourceType& Source), void(*SetArrayNum)(ExternalSourceType& Source, SIZE_T Num)>
570void TPolymorphicArrayStructNetSerializerImpl<ExternalSourceType, ExternalSourceArrayItemType, GetArray, SetArrayNum>::Serialize(FNetSerializationContext& Context, const FNetSerializeArgs& Args)
571{
572 FNetBitStreamWriter& Writer = *Context.GetBitStreamWriter();
573 const QuantizedType& SourceValue = *reinterpret_cast<const QuantizedType*>(Args.Source);
574 const ConfigType& Config = *static_cast<const ConfigType*>(Args.NetSerializerConfig);
575
576 // Serialize quantized data
577 Writer.WriteBits(SourceValue.NumItems, ArrayItemBits);
578 for (uint32 It = 0, EndIt = SourceValue.NumItems; It != EndIt; ++It)
579 {
580 UE_NET_TRACE_SCOPE(Element, Writer, Context.GetTraceCollector(), ENetTraceVerbosity::VeryVerbose);
581
582 const FQuantizedItem& Item = SourceValue.Items[It];
583 const FTypeInfo* TypeInfo = Config.RegisteredTypes.GetTypeInfo(Item.StructName);
584 if (Writer.WriteBool(TypeInfo != nullptr))
585 {
586 CA_ASSUME(TypeInfo != nullptr);
