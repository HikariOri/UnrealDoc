460 virtual FMatrix44f GetSampleToRGBMatrix() const
461 {
462 FMatrix Pre = FMatrix::Identity;
463 FVector Off;
464 switch (GetFormat())
465 {
466 case EMediaTextureSampleFormat::R4FL: Off = MediaShaders::YUVOffsetFloat; break;
467 case EMediaTextureSampleFormat::Y416:
468 case EMediaTextureSampleFormat::P010:
469 case EMediaTextureSampleFormat::ShortYUY2:
470 case EMediaTextureSampleFormat::YUVv216: Off = MediaShaders::YUVOffset16bits; break;
471 case EMediaTextureSampleFormat::YUVv210: Off = MediaShaders::YUVOffset10bits; break;
472 default: Off = MediaShaders::YUVOffset8bits; break;
473 }
474 Pre.M[0][3] = -Off.X;
475 Pre.M[1][3] = -Off.Y;
476 Pre.M[2][3] = -Off.Z;
477 return FMatrix44f(MediaShaders::YuvToRgbRec709Scaled * Pre); // assumes sRGB & video range
