60 // Sets the current preview controller for the scene (handles uninitializing and initializing controllers)
61 bool SetPreviewController(UClass* PreviewControllerClass, class IPersonaPreviewScene* PreviewScene)
62 {
63 if (!PreviewControllerClass->HasAnyClassFlags(CLASS_Abstract) &&
64 PreviewControllerClass->IsChildOf(UPersonaPreviewSceneController::StaticClass()) &&
65 (!PreviewControllerInstance || PreviewControllerClass != PreviewControllerInstance->GetClass()))
66 {
67 PreviewController = PreviewControllerClass;
68 if (PreviewControllerInstance)
69 {
70 PreviewControllerInstance->UninitializeView(this, PreviewScene);
71 }
72 PreviewControllerInstance = GetControllerForClass(PreviewControllerClass);
73 PreviewControllerInstance->InitializeView(this, PreviewScene);
74 return true;
75 }
76 return false;
77 }
