16/* Class used in shader compilation pipelines which wraps source code and ensures sufficient padding such that 16-byte wide SIMD
17 * operations on the source are guaranteed to read valid memory even if starting from the last character.
18 */
19class FShaderSource
20{
21private:
22 inline void SetLen(int32 Num, EAllowShrinking AllowShrinking = EAllowShrinking::Default)
23 {
24 Source.SetNumUninitialized(Num + ShaderSourceSimdPadding, AllowShrinking);
25 FMemory::Memzero(Source.GetData() + Num, sizeof(CharType) * ShaderSourceSimdPadding);
26
27 SourceCompressed.Empty();
28 DecompressedCharCount = 0;
29 }
30
31public:
32 typedef ANSICHAR CharType;
33 typedef FAnsiStringView FViewType;
34 typedef FAnsiString FStringType;
35 typedef FCStringAnsi FCStringType;
36
37 template <int NumChars>
38 using TStringBuilder = TAnsiStringBuilder<NumChars>;

40 /* Construct an empty shader source object; will still contain padding */
41 FShaderSource() { SetLen(0); };
42
43 /* Given a string view construct a shader source object containing the contents of that string.
44 * Note that this will incur a memcpy of the string contents.
45 * @param InSrc The source string to be copied
46 * @param AdditionalSlack optional additional space to allocate; this is on top of the automatic padding.
47 */
48 RENDERCORE_API explicit FShaderSource(FViewType InSrc, int32 AdditionalSlack = 0);
49
50 /* Set the given string as the contents of this shader source object. The inner allocation will grow to fit
51 * the string contents as needed.
52 * Note that this will incur a memcpy of the string contents.
53 * @param InSrc The source string to be copied
54 * @param AdditionalSlack optional additional space to allocate; this is on top of the automatic padding.
55 */
56 RENDERCORE_API void Set(FViewType InSrc, int32 AdditionalSlack = 0);
57
58 /* Returns the array which contains the compressed data for external population.
59 * Note that this is only intended to be used in a very specific situation (loading UBA-virtualized files
60 * in SCW) and will only work properly if DecompressedCharCount is correctly set.
61 */
62 inline TArray<uint8>& EditCompressed() { return SourceCompressed; }
63
64 /* Move assignment operator accepting a string object. This will append padding bytes to the existing string, as such it's
65 * best if there's sufficient extra capacity in the string storage to avoid incurring a realloc-and-copy here.
66 * @param InSrc The source string whose data this object will take ownership of.
67 */
68 RENDERCORE_API FShaderSource& operator=(FStringType&& InSrc);
69
70 /* Reduces the set size of the stored string length, optionally shrinking the allocation.
71 * @param Num the desired allocation size (padding bytes will be added on top of this)
72 * @param AllowShrinking whether to reallocate or keep the existing larger size allocation
73 */
74 inline void ShrinkToLen(int32 Num, EAllowShrinking AllowShrinking = EAllowShrinking::Default)
75 {
76 checkf(Num <= Len(), TEXT("Trying to shrink to %d characters but existing allocation is smaller (%d characters)"), Num, Len());
77 SetLen(Num, AllowShrinking);
78 }
79
80 /* String view accessor. Will decompress data if it was previously compressed.
81 * @return a string view pointing to the source contents, excluding padding.
82 */
83 inline FViewType GetView() const
84 {
85 checkf(!IsCompressed(), TEXT("FShaderSource is compressed; must decompress prior to calling GetView"));
86 return { Source.GetData(), Len() };
87 }
88
89 /* Direct data pointer accessor. Will decompress data if it was previously compressed.
90 * @return a pointer to the source data; will be null terminated by the SIMD padding.
91 */
92 inline CharType* GetData()
93 {
94 checkf(!IsCompressed(), TEXT("FShaderSource is compressed; must decompress prior to calling GetData"));
95 return Source.GetData();
96 }
97
98 /* IsEmpty predicate
99 * @return true if this source object is empty excluding the SIMD padding, false otherwise.
100 */
101 inline bool IsEmpty() const { return Len() == 0; }
102
103 /* Length accessor.
104 * @return the non-padded length of the source (also excluding null terminator)
105 */
106 inline int32 Len() const
107 {
108 checkf(!IsCompressed(), TEXT("Len should not be called on compressed FShaderSource."))
109 return Source.Num() - ShaderSourceSimdPadding;
110 };
111
112 inline bool IsCompressed() const
113 {
114 return DecompressedCharCount != 0; // DecompressedCharCount member is only set when compression occurs
115 }
116
117 inline int32 GetDecompressedSize() const
118 {
119 return DecompressedCharCount * sizeof(FShaderSource::CharType);
120 }
121
122 /* FArchive serialization operator. Note this currently serializes padding for simplicity's sake.
123 * @param Ar The archive to serialize from/to
124 * @param ShaderSource the source object to serialize
125 */
126 friend FArchive& operator<<(FArchive& Ar, FShaderSource& ShaderSource);
127
128 RENDERCORE_API void Compress();
129 RENDERCORE_API void Decompress();
130
131 inline FMemoryView GetCompressed() const
132 {
133 checkf(IsCompressed(), TEXT("Requesting compressed shader source but source has not been compressed"));
134 return FMemoryView(SourceCompressed.GetData(), (uint64)SourceCompressed.Num());
135 }
136
137private:
138 friend class FShaderCompileJob;
139
140 static_assert(sizeof(CharType) == 1);
141 static constexpr int32 ShaderSourceSimdPadding = 15;
142 TArray<CharType> Source;
143 TArray<uint8> SourceCompressed;
144 int32 DecompressedCharCount = 0;
145};
