36 static EventEmitterFlag ComputeEmitterFlag(bool LocalEmitter, bool GlobalEmitter)
37 {
38 EventEmitterFlag EmitterFlagOut = EventEmitterFlag::EmptyDispatcher;
39 if (LocalEmitter)
