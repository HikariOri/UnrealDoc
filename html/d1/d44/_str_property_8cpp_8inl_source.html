46EConvertFromTypeResult UE_STRPROPERTY_CLASS::ConvertFromType(const FPropertyTag& Tag, FStructuredArchive::FSlot Slot, uint8* Data, const UStruct* DefaultsStruct, const uint8* Defaults)
47{
48 if (GetClass()->GetFName() == Tag.Type)
49 {
50 return EConvertFromTypeResult::UseSerializeItem;
51 }
52
53 // Convert serialized text to string.
54 if (Tag.Type==NAME_TextProperty)
55 {
56 FText Text;
57 Slot << Text;
58 const UE_STRPROPERTY_STRINGTYPE String(FTextInspector::GetSourceString(Text) ? *FTextInspector::GetSourceString(Text) : TEXT(""));
59 SetPropertyValue_InContainer(Data, String, Tag.ArrayIndex);
60 }
61 else if (Tag.Type==NAME_StrProperty)
62 {
63 FString SavedStr;
64 Slot << SavedStr;
65 const UE_STRPROPERTY_STRINGTYPE ConvertedString(SavedStr);
66 SetPropertyValue_InContainer(Data, ConvertedString, Tag.ArrayIndex);
67 }
68 else if (Tag.Type==NAME_Utf8StrProperty)
69 {
70 FUtf8String SavedStr;
71 Slot << SavedStr;
72 const UE_STRPROPERTY_STRINGTYPE ConvertedString(SavedStr);
73 SetPropertyValue_InContainer(Data, ConvertedString, Tag.ArrayIndex);
74 }
75 else if (Tag.Type==NAME_AnsiStrProperty)
76 {
77 FAnsiString SavedStr;
78 Slot << SavedStr;
79 const UE_STRPROPERTY_STRINGTYPE ConvertedString(SavedStr);
80 SetPropertyValue_InContainer(Data, ConvertedString, Tag.ArrayIndex);
81 }
82 else
83 {
84 return EConvertFromTypeResult::CannotConvert;
85 }
86
87 return EConvertFromTypeResult::Converted;
88}

115const TCHAR* UE_STRPROPERTY_CLASS::ImportText_Internal(const TCHAR* Buffer, TNotNull<void*> ContainerOrPropertyPtr, EPropertyPointerType PropertyPointerType, UObject* Parent, int32 PortFlags, FOutputDevice* ErrorText) const
116{
117 UE_STRPROPERTY_STRINGTYPE ImportedText;
118 if( !(PortFlags & PPF_Delimited) )
119 {
120 ImportedText = Buffer;
121
122 // in order to indicate that the value was successfully imported, advance the buffer past the last character that was imported
123 Buffer += FCString::Strlen(Buffer);
124 }
125 else
126 {
127 // require quoted string here
128 if (*Buffer != TCHAR('"'))
129 {
130 ErrorText->Logf(TEXT("Missing opening '\"' in string property value: %s"), Buffer);
131 return NULL;
132 }
133 const TCHAR* Start = Buffer;
134 FString Temp;
135 Buffer = FPropertyHelpers::ReadToken(Buffer, Temp);
136 if (Buffer == NULL)
137 {
138 return NULL;
139 }
140 if (Buffer > Start && Buffer[-1] != TCHAR('"'))
141 {
142 ErrorText->Logf(TEXT("Missing terminating '\"' in string property value: %s"), Start);
143 return NULL;
144 }
145 ImportedText = UE_STRPROPERTY_STRINGTYPE(MoveTemp(Temp));
146 }
147 if (PropertyPointerType == EPropertyPointerType::Container && HasSetter())
148 {
149 SetValue_InContainer(ContainerOrPropertyPtr, ImportedText);
150 }
151 else
152 {
153 *(UE_STRPROPERTY_STRINGTYPE*)PointerToValuePtr(ContainerOrPropertyPtr, PropertyPointerType) = MoveTemp(ImportedText);
154 }
155 return Buffer;
156}
