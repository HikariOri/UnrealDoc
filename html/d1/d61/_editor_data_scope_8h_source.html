91// ============================================================================
92// TLS convenience overloads
93// ============================================================================
94// These free functions wrap ICoreProvider scope methods using the thread-local
95// current scope (GetCurrentScope()) as the row argument. They exist so callers
96// that operate in the ambient scope don't need to pass GetCurrentScope()
97// explicitly at every call site.
98
100template<TColumnType T>
101const T* GetScopeData(ICoreProvider& Storage)
102{
103 return Storage.GetScopeData<T>(GetCurrentScope());
104}
105
107template<TColumnType T>
108FScopeDataVersion GetScopeDataVersion(ICoreProvider& Storage)
109{
110 return Storage.GetScopeDataVersion<T>(GetCurrentScope());
111}
112
114template<TColumnType T>
115FScopeDataVersion SetScopeData(ICoreProvider& Storage, T&& Data)
116{
117 return Storage.SetScopeData<T>(GetCurrentScope(), Forward<T>(Data));
118}
119
121template<TColumnType T>
122bool RemoveScopeData(ICoreProvider& Storage)
123{
124 return Storage.RemoveScopeData<T>(GetCurrentScope());
125}
