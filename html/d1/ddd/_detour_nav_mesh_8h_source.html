208struct dtTileVert
209{
210 float x, y, z;
211
213 inline void setFromWorld(const dtReal* bmin, const dtReal* worldPos)
214 {
215 const dtReal dx = worldPos[0] - bmin[0];
216 const dtReal dy = worldPos[1] - bmin[1];
217 const dtReal dz = worldPos[2] - bmin[2];
218
219 // Verify that tile-local offsets are reasonable for float storage.
220
221 // Maximum allowed tile-local vertex offset before precision becomes problematic.
222 // Typical tiles are <10,000 units. This threshold (100x that) catches bugs like:
223 // - Vertex assigned to wrong tile
224 // - Wrong bmin passed to setFromWorld()
225 // - Coordinate system errors
226 // Values this large indicate real bugs, not just edge cases.
227 constexpr dtReal DT_TILE_VERT_MAX_OFFSET = 1000000.0;
228 dtAssert((dx < 0 ? -dx : dx) < DT_TILE_VERT_MAX_OFFSET && (dy < 0 ? -dy : dy) < DT_TILE_VERT_MAX_OFFSET && (dz < 0 ? -dz : dz) < DT_TILE_VERT_MAX_OFFSET);
229
230 x = (float)dx;
231 y = (float)dy;
232 z = (float)dz;
233 }
234
236 inline void toWorld(const dtReal* bmin, dtReal* out) const
237 {
238 out[0] = (dtReal)x + bmin[0];
239 out[1] = (dtReal)y + bmin[1];
240 out[2] = (dtReal)z + bmin[2];
241 }
242};

794 inline dtPolyRef encodePolyId(unsigned int salt, unsigned int it, unsigned int ip) const
795 {
796 return ((dtPolyRef)salt << (m_polyBits+m_tileBits)) | ((dtPolyRef)it << m_polyBits) | (dtPolyRef)ip;
797 }
798
806 inline void decodePolyId(dtPolyRef ref, unsigned int& salt, unsigned int& it, unsigned int& ip) const
807 {
808 const dtPolyRef saltMask = ((dtPolyRef)1<<m_saltBits)-1;
809 const dtPolyRef tileMask = ((dtPolyRef)1<<m_tileBits)-1;
810 const dtPolyRef polyMask = ((dtPolyRef)1<<m_polyBits)-1;
811 salt = (unsigned int)((ref >> (m_polyBits+m_tileBits)) & saltMask);
812 it = (unsigned int)((ref >> m_polyBits) & tileMask);
813 ip = (unsigned int)(ref & polyMask);
814 }
815
820 inline unsigned int decodePolyIdSalt(dtPolyRef ref) const
821 {
822 const dtPolyRef saltMask = ((dtPolyRef)1<<m_saltBits)-1;
823 return (unsigned int)((ref >> (m_polyBits+m_tileBits)) & saltMask);
824 }
825
830 inline unsigned int decodePolyIdTile(dtPolyRef ref) const
831 {
832 const dtPolyRef tileMask = ((dtPolyRef)1<<m_tileBits)-1;
833 return (unsigned int)((ref >> m_polyBits) & tileMask);
834 }
835
840 inline unsigned int decodePolyIdPoly(dtPolyRef ref) const
841 {
842 const dtPolyRef polyMask = ((dtPolyRef)1<<m_polyBits)-1;

844 }
