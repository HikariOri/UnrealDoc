582 template <typename InitKeyType>
583 inline ElementType* AllocateMemoryForEmplace(InitKeyType&& InKey)
584 {
585 int32 InsertIndex = Algo::LowerBoundBy(Pairs, InKey, FKeyForward(), SortPredicate());
586 check(InsertIndex >= 0 && InsertIndex <= Pairs.Num());
587
588 ElementType* DataPtr = nullptr;
589 // Since we returned lower bound we already know InKey <= InsertIndex key. So if InKey is not < InsertIndex key, they must be equal
590 if (Pairs.IsValidIndex(InsertIndex) && !SortPredicate()(InKey, Pairs[InsertIndex].Key))
591 {
592 // Replacing element, delete old one
593 DataPtr = Pairs.GetData() + InsertIndex;
594 DestructItems(DataPtr, 1);
595 }
596 else
597 {
598 // Adding new one, this may reallocate Pairs
599 Pairs.InsertUninitialized(InsertIndex, 1);
600 DataPtr = Pairs.GetData() + InsertIndex;
601 }
602
603 return DataPtr;
604 }
