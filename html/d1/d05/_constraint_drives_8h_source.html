21// Specifies how the constraint responds when the angular drive target would violate the joint's angular limits.
22// Without a response, the drive and the limit constraint compete, producing spurious torques.
23UENUM(BlueprintType)
24enum class EAngularDriveLimitViolationResponse : uint8
25{
26 // Drive target rotation is left unchanged. The drive and the limit constraint may compete.
27 None = 0,
28 // Project the drive target onto the joint limits using a linear swing-angle approximation.
29 // Accurate for small angles; exact for pure Swing1 or pure Swing2 rotations.
30 ClampLinear = 1,
31 // Project the drive target onto the joint limits using the exact Chaos 4*atan2 formula.
32 // More accurate near the limit boundary but slightly more expensive than ClampLinear.
33 ClampExact = 2,
34 // Widen the joint's angular limits to accommodate the drive target instead of clipping the
35 // target. Limits are never narrowed below their authored defaults; they are restored each
36 // frame when the target moves back inside. Has no effect on locked axes.
37 WidenLimits = 3,
38};
