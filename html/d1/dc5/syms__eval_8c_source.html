40SYMS_API void
41syms_eval_op_push(SYMS_Arena *arena, SYMS_EvalOpList *list, SYMS_EvalOp op, SYMS_EvalOpParams params){
42 SYMS_ASSERT(op < SYMS_EvalOp_COUNT);
43
44 //- setup new op
45 SYMS_EvalOpNode *loose_op = syms_push_array(arena, SYMS_EvalOpNode, 1);
46 loose_op->op = op;
47 loose_op->params = params;
48
49 //- main op push
50 // determine encoded byte count
51 SYMS_U8 ctrlbits = syms_eval_opcode_ctrlbits[op];
52 SYMS_U64 extra_byte_count = (ctrlbits >> SYMS_EvalOpCtrlBits_DecodeShft)&SYMS_EvalOpCtrlBits_DecodeMask;
53
54 // push onto list
55 SYMS_QueuePush(list->first, list->last, loose_op);
56 SYMS_U64 encoded_byte_count = 1 + extra_byte_count;
57 list->byte_count += encoded_byte_count;
58}

64 loose_op->op = SYMS_EvalIRExtKind_Bytecode;

139SYMS_API SYMS_B32
140syms_eval_opcode_type_compatible(SYMS_EvalOp op, SYMS_EvalTypeGroup group){
141 SYMS_B32 result = syms_false;
142 switch (op){
143 case SYMS_EvalOp_Neg:case SYMS_EvalOp_Add:case SYMS_EvalOp_Sub:
144 case SYMS_EvalOp_Mul:case SYMS_EvalOp_Div:
145 case SYMS_EvalOp_EqEq:case SYMS_EvalOp_NtEq:
146 case SYMS_EvalOp_LsEq:case SYMS_EvalOp_GrEq:case SYMS_EvalOp_Less:case SYMS_EvalOp_Grtr:
147 {
148 if (group != SYMS_EvalTypeGroup_Other){
149 result = syms_true;
150 }
151 }break;
152 case SYMS_EvalOp_Mod:case SYMS_EvalOp_LShift:case SYMS_EvalOp_RShift:
153 case SYMS_EvalOp_BitNot:case SYMS_EvalOp_BitAnd:case SYMS_EvalOp_BitXor:case SYMS_EvalOp_BitOr:
154 case SYMS_EvalOp_LogNot:case SYMS_EvalOp_LogAnd:case SYMS_EvalOp_LogOr:
155 {
156 if (group == SYMS_EvalTypeGroup_S || group == SYMS_EvalTypeGroup_U){
157 result = syms_true;
158 }
159 }break;
160 }
161 return(result);
162}

166SYMS_API SYMS_String8
167syms_eval_bytecode_from_op_list(SYMS_Arena *arena, SYMS_EvalOpList *list){
168 // allocate output
169 SYMS_String8 result = {0};
170 result.size = list->byte_count;
171 result.str = syms_push_array(arena, SYMS_U8, result.size);
172
173 // iterate loose op nodes
174 SYMS_U8 *ptr = result.str;
175 SYMS_U8 *opl = result.str + result.size;
176 for (SYMS_EvalOpNode *node = list->first;
177 node != 0;
178 node = node->next){
179 SYMS_U32 op = node->op;
180
181 switch (op){
182 default:
183 {
184 // compute bytecode advance
185 SYMS_U8 ctrlbits = syms_eval_opcode_ctrlbits[op];
186 SYMS_U64 extra_byte_count = (ctrlbits >> SYMS_EvalOpCtrlBits_DecodeShft)&SYMS_EvalOpCtrlBits_DecodeMask;
187
188 SYMS_U8 *next_ptr = ptr + 1 + extra_byte_count;
189 SYMS_ASSERT(next_ptr <= opl);
190
191 // fill bytecode
192 ptr[0] = op;
193 syms_memmove(ptr + 1, &node->params, extra_byte_count);
194
195 // advance output pointer
196 ptr = next_ptr;
197 }break;
198
199 case SYMS_EvalIRExtKind_Bytecode:
200 {
201 // compute bytecode advance
202 SYMS_U64 size = node->params.data.size;
203 SYMS_U8 *next_ptr = ptr + size;
204 SYMS_ASSERT(next_ptr <= opl);
205
206 // fill bytecode
207 syms_memmove(ptr, node->params.data.str, size);
208
209 // advance output pointer
210 ptr = next_ptr;
211 }break;
212 }
213 }
214
215 return(result);
216}
