20inline D3D12_DESCRIPTOR_HEAP_TYPE Translate(ERHIDescriptorHeapType InHeapType)
21{
22 switch (InHeapType)
23 {
24 default: checkNoEntry();
25 case ERHIDescriptorHeapType::Standard: return D3D12_DESCRIPTOR_HEAP_TYPE_CBV_SRV_UAV;
26 case ERHIDescriptorHeapType::RenderTarget: return D3D12_DESCRIPTOR_HEAP_TYPE_RTV;
27 case ERHIDescriptorHeapType::DepthStencil: return D3D12_DESCRIPTOR_HEAP_TYPE_DSV;
28 case ERHIDescriptorHeapType::Sampler: return FD3D12_DESCRIPTOR_HEAP_TYPE_SAMPLER;
29 }
30}

32inline D3D12_DESCRIPTOR_HEAP_TYPE Translate(ERHIDescriptorType InDescriptorType)
33{
34 const ERHIDescriptorTypeMask TypeMask = RHIDescriptorTypeMaskFromType(InDescriptorType);
35 if (EnumHasAnyFlags(TypeMask, ERHIDescriptorTypeMask::CBV | ERHIDescriptorTypeMask::SRV | ERHIDescriptorTypeMask::UAV))
36 {
37 return D3D12_DESCRIPTOR_HEAP_TYPE_CBV_SRV_UAV;
38 }
39
40 if (EnumHasAnyFlags(TypeMask, ERHIDescriptorTypeMask::Sampler))
41 {
42 return FD3D12_DESCRIPTOR_HEAP_TYPE_SAMPLER;
43 }
44
45 checkNoEntry();
46 return D3D12_DESCRIPTOR_HEAP_TYPE_CBV_SRV_UAV;
47}

49inline ERHIDescriptorTypeMask D3D12DescriptorTypeMaskFromHeapType(ERHIDescriptorHeapType InHeapType)
50{
51 if (InHeapType == ERHIDescriptorHeapType::Standard)
52 {
53 return ERHIDescriptorTypeMask::CBV | ERHIDescriptorTypeMask::SRV | ERHIDescriptorTypeMask::UAV;
54 }
55
56 if (InHeapType == ERHIDescriptorHeapType::Sampler)
57 {
58 return ERHIDescriptorTypeMask::Sampler;
59 }
60
61 return ERHIDescriptorTypeMask::None;
62}
