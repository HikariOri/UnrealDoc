108 void SetEvents(const TArray<FAutomationExecutionEntry>& InEvents, int32 InWarningTotal, int32 InErrorTotal)
109 {
110 Entries = InEvents;
111 WarningTotal = InWarningTotal;
112 ErrorTotal = InErrorTotal;
113 }
114
115 void AddEvent(const FAutomationExecutionEntry& Event)
116 {
117 switch ( Event.Event.Type )
118 {
119 case EAutomationEventType::Warning:
120 WarningTotal++;
121 break;
122 case EAutomationEventType::Error:
123 ErrorTotal++;
124 break;
125 }
126
127 Entries.Add(Event);
128 }
129
130 const TArray<FAutomationExecutionEntry>& GetEntries() const { return Entries; }
131
132 int32 GetLogTotal() const { return Entries.Num() - (WarningTotal + ErrorTotal); }
133 int32 GetWarningTotal() const { return WarningTotal; }
134 int32 GetErrorTotal() const { return ErrorTotal; }
135
136public:
137 /* The current state of this test */
