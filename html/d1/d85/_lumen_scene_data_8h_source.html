672private:
673
674 // Data structure overview
675 // -----------------------
676 // * The atlas is divided in to pages
677 // * Each page is 128x128
678 // * Each page can be divided into sub-allocation to hold smaller element size (e..g 8x8, 8x16, 8x32, ...)
679 //
680 // * Card elements are allocated into these pages.
681 // * Card elements are allocated into page with the correct sub-allocation size
682 //
683 // Data structures:
684 // * FPageBin - Holds reference to all the page allocations for a given element size. Due to this, there is at max. 64 FPageBin E.g.:
685 // * 1 FPageBin for 8x8 allocation,
686 // * 1 FPageBin for 8x16 allocation,
687 // * 1 FPageBin for 8x32 allocation,
688 // * ...
689 // A PageBin holds reference to several PageBinAllocation, one per physical page. There can be a large quantity of FPageBinAllocation
690 // * FPageBinAllocation - Tracks the sub-allocation within a single physical page. A physical page (128x128) will be broken into 32x32 sub-allocation for
691 // an element size of 8x8. The sub-allocation is tracked with a bitfield to indicate which slot is available or not
692 // * FPageBinLookup - Lookup table to fast retrivable of FPageBin based on the desired element size. The lookup is a 8x8 table, so there is at max 64 FPageBin
693 struct FPageBinAllocation
694 {
695 public:
696 void Init(const FIntPoint& InPageCoord, const FIntPoint& InPageSizeInElements)
697 {

849 // Bin lookups are stored as 2D mapping (8x8 - [1-128]x[1-128])
850 // This mapping indexes PageX dim. and PageY dim.
851 // As an example, a 8x16 SubPage allocator will be stored at [3,4] (i.e., [log2(8),log2(16)] )
852 // 0 1 2 3 4 5 6 7
853 // --------------------
854 // 1 2 4 8 16 32 64 128
855 // 0 | 1
856 // 1 | 2 X
857 // 2 | 4
858 // 3 | 8 X
859 // 4 | 16
860 // 5 | 32 X
861 // 6 | 64 X
862 // 7 | 128
863 static const uint8 InvalidPageBinIndex = 0xFF;
864 typedef TStaticArray<uint8, 64u> FPageBinLookup;
865 FPageBinLookup PageBinLookup;
866 bool bInitPageBinLookup = true;
867
868 uint8 GetLookupIndex(const FIntPoint& InRes) const
869 {
870 checkSlow(FMath::IsPowerOfTwo(InRes.X) && FMath::IsPowerOfTwo(InRes.Y));
871 checkSlow(InRes.X <= Lumen::PhysicalPageSize && InRes.Y <= Lumen::PhysicalPageSize);
872 const uint32 OutIndex = FMath::FloorLog2(InRes.X) + FMath::FloorLog2(InRes.Y) * 8u;
873 //check(OutIndex < 64u);
874 return (uint8)OutIndex;
875 }
876
877 const FPageBin* GetBin(const FIntPoint& InRes) const
878 {
879 const uint8 LookupIndex = GetLookupIndex(InRes);
880 const uint8 BinIndex = PageBinLookup[LookupIndex];
881 if (BinIndex != InvalidPageBinIndex)
882 {
883 return &PageBins[BinIndex];
884 }
885 return nullptr;
886 }
887
888 FPageBin* GetBin(const FIntPoint& InRes)
889 {
890 const uint8 LookupIndex = GetLookupIndex(InRes);
891 const uint8 BinIndex = PageBinLookup[LookupIndex];
892 if (BinIndex != InvalidPageBinIndex)
893 {
894 return &PageBins[BinIndex];
895 }
896 return nullptr;
897 }
898
899 FPageBin* GetOrAddBin(const FIntPoint& InRes)
900 {
901 const uint8 LookupIndex = GetLookupIndex(InRes);
902 const uint8 BinIndex = PageBinLookup[LookupIndex];
903 if (BinIndex == InvalidPageBinIndex)
904 {
905 PageBinLookup[LookupIndex] = PageBins.Num();
906 PageBins.Add(FPageBin(InRes));
907
908 // There can't be more than 64 PageBins, as the sub-allocation resolution within
909 // a 128x128 physical page is bound to 64 (8x8, 8x16, 8x32, ..., 128x64, 128x128).
910 check(PageBins.Num() <= 64);
911 return &PageBins.Last();
912 }
913 else
914 {
915 return &PageBins[BinIndex];
916 }
917 }

931// The advantage of sharing an origin is that Lumen scene data can be shared and updated once. In the future, we could
932// allow origins to be shared for other use cases, such as nDisplay inner frustums, or imagine a sim with a wide angle

963 //
964 // The first is the "GetDeferredLightParameters" utility function, which uses a bunch of data from the FViewInfo structure, which

972 // View.OneOverPreExposure ""

982 // things so there is a formal concept of shared origin views (FViewSharedOrigin?) at a higher level in the scene renderer
983 // itself. Then we could pull all of the above into that structure. But that goes well beyond the scope of adding Lumen support
984 // for cube maps, which is the immediate goal.
985 //
986 // There may be rendering artifacts with forward lighting, volumetric cloud shadows, and ray tracing, given that the code that
987 // generates those may not be completely shared origin view friendly. Forward lighting pulls in lights from the frustum, so that
988 // definitely seems like it should be modified to take into account the frustums of all shared origin views. It's less clear if
