123struct UE_INTERNAL FRegistryEvents
124{
125 // Maps the old handle removed from the registry to the path string the old handle was registered to
126 TMap<FPathHandle, FString> RemovedPaths;
127
128 // Holds all new handles added to the registry. The added path strings can be looked up in the registry using these handles.
129 TArray<FPathHandle> AddedPaths;
130
131 // Maps the old handles removed from the registry that have been replaced by a new handle.
132 // The path strings for the updated paths can be looked up in the registry using the new handles (i.e. Pair.Value).
133 // This map contains the parent chain of paths to the path that was updated.
134 // e.g. When updating /Module/Class/Data: /Module, /Module/Class and /Module/Class/Data will be marked as updated.
135 // This map does not include paths whose FPathData FPathHandle members have changed.
136 TMap<FPathHandle, FPathHandle> UpdatedPaths;
137

139 ASSETREGISTRY_API void Reset();

557inline FPathHandle FVersePathRegistry::AllocateHandleNoLock()
558{
559 uint32 Index;
560 if (!FreeList.IsEmpty())
561 {
562 Index = FreeList.Last();
563 FreeList.Pop();
564 }
565 else
566 {
567 Index = PathDatas.Num();

570 return { Index, PathDatas[Index].Generation};

573inline void FVersePathRegistry::FreeHandleNoLock(const uint32 Index)
574{
575 check(Index < (uint32)PathDatas.Num());
576 check((PathDatas[Index].Generation & FPathHandle::GenerationMask) + 1 < FPathHandle::GenerationMask);
577 PathDatas[Index].Generation++;
578 FreeList.Push(Index);
579}

581inline bool FVersePathRegistry::IsValidHandleNoLock(FPathHandle Handle) const
582{
583 const uint32 Index = Handle.Index;
584 if (Index == FPathHandle::Null.Index || Index >= (uint32)PathDatas.Num())
585 {
586 return false;
587 }
588
589 const uint32 Generation = Handle.Generation & FPathHandle::GenerationMask;
590 const uint32 ActualGeneration = PathDatas[Index].Generation;
591 if (Generation != ActualGeneration)
592 {
593 return false;
594 }
595 return true;
596}
