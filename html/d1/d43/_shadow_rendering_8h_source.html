1180 void Set(FRHIBatchedShaderParameters& BatchedParameters, const FSceneView& View, const FProjectedShadowInfo* ShadowInfo, bool bModulatedShadows, bool bUseFadePlane, bool SubPixelShadow)
1181 {
1182 const FVector PreViewTranslation = View.ViewMatrices.GetPreViewTranslation();
1183 const FIntPoint ShadowBufferResolution = ShadowInfo->GetShadowBufferResolution();
1184
1185 if (ShadowTileOffsetAndSizeParam.IsBound())
1186 {
1187 FVector2D InverseShadowBufferResolution(1.0f / ShadowBufferResolution.X, 1.0f / ShadowBufferResolution.Y);
1188 FVector4f ShadowTileOffsetAndSize(
1189 (ShadowInfo->BorderSize + ShadowInfo->X) * InverseShadowBufferResolution.X,
1190 (ShadowInfo->BorderSize + ShadowInfo->Y) * InverseShadowBufferResolution.Y,
1191 ShadowInfo->ResolutionX * InverseShadowBufferResolution.X,
1192 ShadowInfo->ResolutionY * InverseShadowBufferResolution.Y);
1193 SetShaderValue(BatchedParameters, ShadowTileOffsetAndSizeParam, ShadowTileOffsetAndSize);
1194 }
1195
1196 // Set the transform from screen coordinates to shadow depth texture coordinates.
1197 if (bModulatedShadows)
1198 {
1199 // UE-29083 : work around precision issues with ScreenToShadowMatrix on low end devices.
1200 const FMatrix44f ScreenToShadow = FMatrix44f(ShadowInfo->GetScreenToShadowMatrix(View, 0, 0, ShadowBufferResolution.X, ShadowBufferResolution.Y)); // LWC_TODO: Precision loss
1201 SetShaderValue(BatchedParameters, ScreenToShadowMatrix, ScreenToShadow);
1202 }
1203 else
1204 {
1205 const FMatrix44f ScreenToShadow = FMatrix44f(ShadowInfo->GetScreenToShadowMatrix(View)); // LWC_TODO: Precision loss
1206 SetShaderValue(BatchedParameters, ScreenToShadowMatrix, ScreenToShadow);
1207 }
1208
1209 if (SoftTransitionScale.IsBound())
1210 {
1211 const float TransitionSize = ShadowInfo->ComputeTransitionSize();
1212
1213 SetShaderValue(BatchedParameters, SoftTransitionScale, FVector3f(0, 0, 1.0f / TransitionSize));
1214 }
1215
1216 if (ShadowBufferSize.IsBound())
1217 {
1218 FVector2D ShadowBufferSizeValue(ShadowBufferResolution.X, ShadowBufferResolution.Y);
1219
1220 SetShaderValue(BatchedParameters, ShadowBufferSize,
1221 FVector4f(ShadowBufferSizeValue.X, ShadowBufferSizeValue.Y, 1.0f / ShadowBufferSizeValue.X, 1.0f / ShadowBufferSizeValue.Y));
1222 }
1223
1224 FRHITexture* ShadowDepthTextureValue;
1225
1226 // Translucency shadow projection has no depth target
1227 if (ShadowInfo->RenderTargets.DepthTarget)
1228 {
1229 ShadowDepthTextureValue = ShadowInfo->RenderTargets.DepthTarget->GetRHI();
1230 }
1231 else
1232 {
1233 ShadowDepthTextureValue = GSystemTextures.BlackDummy->GetRHI();
1234 }
1235
1236 FRHISamplerState* DepthSamplerState = TStaticSamplerState<SF_Point,AM_Clamp,AM_Clamp,AM_Clamp>::GetRHI();
1237
1238 SetTextureParameter(BatchedParameters, ShadowDepthTexture, ShadowDepthTextureSampler, DepthSamplerState, ShadowDepthTextureValue);
1239
1240 if (ShadowDepthTextureSampler.IsBound())
1241 {
1242 SetSamplerParameter(BatchedParameters, ShadowDepthTextureSampler, DepthSamplerState);
1243 }
1244
1245 SetShaderValue(BatchedParameters, ProjectionDepthBias, FVector4f(ShadowInfo->GetShaderDepthBias(), ShadowInfo->GetShaderSlopeDepthBias(), ShadowInfo->GetShaderReceiverDepthBias(), ShadowInfo->MaxSubjectZ - ShadowInfo->MinSubjectZ));
1246 SetShaderValue(BatchedParameters, FadePlaneOffset, ShadowInfo->CascadeSettings.FadePlaneOffset);
1247
1248 if(InvFadePlaneLength.IsBound() && bUseFadePlane)
1249 {
1250 check(ShadowInfo->CascadeSettings.FadePlaneLength > 0);
1251 SetShaderValue(BatchedParameters, InvFadePlaneLength, 1.0f / ShadowInfo->CascadeSettings.FadePlaneLength);
1252 }
1253
1254 if (LightPositionOrDirection.IsBound())
1255 {
1256 const FVector LightDirection = ShadowInfo->GetLightSceneInfo().Proxy->GetDirection();
1257 const FVector LightPosition = ShadowInfo->GetLightSceneInfo().Proxy->GetPosition() + PreViewTranslation;
1258 const bool bIsDirectional = ShadowInfo->GetLightSceneInfo().Proxy->GetLightType() == LightType_Directional;
1259 SetShaderValue(BatchedParameters, LightPositionOrDirection, bIsDirectional ? FVector4f((FVector3f)LightDirection,0) : FVector4f((FVector3f)LightPosition,1));
1260 }
1261
1262 if (SubPixelShadow)
1263 {
1264 float DeviceZNear = 1;
1265 float DeviceZFar = 0;
1266 const bool bIsCascadedShadow = ShadowInfo->bDirectionalLight && !(ShadowInfo->bPerObjectOpaqueShadow || ShadowInfo->bPreShadow);
1267 if (bIsCascadedShadow)
1268 {
1269 FVector4 Near = View.ViewMatrices.GetViewToClip().TransformFVector4(FVector4(0, 0, ShadowInfo->CascadeSettings.SplitNear));
1270 FVector4 Far = View.ViewMatrices.GetViewToClip().TransformFVector4(FVector4(0, 0, ShadowInfo->CascadeSettings.SplitFar));
1271 // LWC_TODO: precision loss?
1272 DeviceZNear = (float)(Near.Z / Near.W);
1273 DeviceZFar = (float)(Far.Z / Far.W);
1274 }
1275
1276 FVector2f SliceNearAndFarDepth;
1277 SliceNearAndFarDepth.X = DeviceZNear;
1278 SliceNearAndFarDepth.Y = DeviceZFar;
1279 SetShaderValue(BatchedParameters, ShadowNearAndFarDepth, SliceNearAndFarDepth);
1280 SetShaderValue(BatchedParameters, bCascadeUseFadePlane, bUseFadePlane ? 1 : 0);
1281 }
1282
1283 SetShaderValue(BatchedParameters, PerObjectShadowFadeStart, ShadowInfo->PerObjectShadowFadeStart);
1284 SetShaderValue(BatchedParameters, InvPerObjectShadowFadeLength, ShadowInfo->InvPerObjectShadowFadeLength);
1285 }
