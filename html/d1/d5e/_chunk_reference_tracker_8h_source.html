127 FORCEINLINE TArray<FGuid> OrderedUniqueReferences(const FBuildPatchAppManifestRef& InstallManifest)
128 {
129 TArray<FGuid> ChunkReferences;
130 // Create our full list of chunks, no dupes, just one reference per chunk in the correct order.
131 TArray<FString> AllFiles;
132 TSet<FGuid> AllChunks;
133 InstallManifest->GetFileList(AllFiles);
134 for (const FString& File : AllFiles)
135 {
136 const FFileManifest* NewFileManifest = InstallManifest->GetFileManifest(File);
137 if (NewFileManifest != nullptr)
138 {
139 for (const FChunkPart& ChunkPart : NewFileManifest->ChunkParts)
140 {
141 bool bWasAlreadyInSet = false;
142 AllChunks.Add(ChunkPart.Guid, &bWasAlreadyInSet);
143 if (!bWasAlreadyInSet)
144 {
145 ChunkReferences.Add(ChunkPart.Guid);
146 }
147 }
148 }
149 }
150 return ChunkReferences;
151 }
