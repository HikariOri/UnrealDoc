433 struct FSharedData
434 {
435 UE::FManualResetEvent Event;
436 std::atomic<int32> CompletedTaskIndex{ 0 };
437 };
438
439 // Shared data usage is important to avoid the variable to go out of scope
440 // before all the task have been run even if we exit after the first event
441 // is triggered.
442 TSharedRef<FSharedData> SharedData = MakeShared<FSharedData>();
443
444 for (int32 Index = 0; Index < Tasks.Num(); ++Index)
445 {
446 Launch(UE_SOURCE_LOCATION,
447 [SharedData, Index]
448 {
449 SharedData->CompletedTaskIndex.store(Index, std::memory_order_relaxed);
450 SharedData->Event.Notify();
451 },
452 Prerequisites(Tasks[Index]),
453 ETaskPriority::Default,
454 EExtendedTaskPriority::Inline
455 );
456 }
457
458 if (SharedData->Event.WaitFor(UE::FMonotonicTimeSpan::FromMilliseconds(Timeout.GetTotalMilliseconds())))
459 {
460 return SharedData->CompletedTaskIndex.load(std::memory_order_relaxed);
461 }
462
463 return INDEX_NONE;
464 }

475 struct FSharedData
476 {
477 explicit FSharedData(uint32 InitRefCount)
478 : RefCount(InitRefCount)
479 {
480 }
481
482 FTaskEvent Event{ UE_SOURCE_LOCATION };
483 std::atomic<uint32> RefCount;
484 };
485
486 FSharedData* SharedData = new FSharedData(Tasks.Num());
487 // `SharedData` can be destroyed before leaving the scope, cache the result locally
488 FTaskEvent Result = SharedData->Event;
489
490 for (const FTask& Task : Tasks)
491 {
492 Launch(UE_SOURCE_LOCATION,
493 [SharedData, Num = Tasks.Num()]
494 {
495 // cache the local copy as `SharedData` can be concurrently deleted right after decrementing the ref counter
496 FTaskEvent Event = SharedData->Event;
497 uint32 PrevRefCount = SharedData->RefCount.fetch_sub(1, std::memory_order_acq_rel); // acq_rel to sync between tasks
498
499 if (UNLIKELY(PrevRefCount == Num))
500 { // the first completed task
501 Event.Trigger();
502 }
503
504 if (UNLIKELY(PrevRefCount == 1))
505 { // the last completed task
506 delete SharedData;
507 }
508 },
509 Prerequisites(Task),
510 ETaskPriority::Default,
511 EExtendedTaskPriority::Inline
512 );
513 }
514
515 return Result;
516 }
