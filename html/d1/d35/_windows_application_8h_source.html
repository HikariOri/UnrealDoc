35 enum Type
36 {
37 //Stops displaying progress and returns the button to its normal state.
38 NoProgress = 0x0,
39
40 //The progress indicator does not grow in size, but cycles repeatedly along the
41 //length of the task bar button. This indicates activity without specifying what
42 //proportion of the progress is complete. Progress is taking place, but there is
43 //no prediction as to how long the operation will take.
44 Indeterminate = 0x1,
45
46 //The progress indicator grows in size from left to right in proportion to the
47 //estimated amount of the operation completed. This is a determinate progress
48 //indicator; a prediction is being made as to the duration of the operation.
49 Normal = 0x2,
50
51 //The progress indicator turns red to show that an error has occurred in one of
52 //the windows that is broadcasting progress. This is a determinate state. If the
53 //progress indicator is in the indeterminate state, it switches to a red determinate
54 //display of a generic percentage not indicative of actual progress.
55 Error = 0x4,
56
57 //The progress indicator turns yellow to show that progress is currently stopped in
58 //one of the windows but can be resumed by the user. No error condition exists and
59 //nothing is preventing the progress from continuing. This is a determinate state.
60 //If the progress indicator is in the indeterminate state, it switches to a yellow
61 //determinate display of a generic percentage not indicative of actual progress.
62 Paused = 0x8,
63 };
