629 virtual bool Raycast(const FVec3& StartPoint, const FVec3& Dir, const FReal Length, const FReal Thickness, FReal& OutTime, FVec3& OutPosition, FVec3& OutNormal, int32& OutFaceIndex) const override
630 {
631 ensure(Length > 0);
632 ensure(FMath::IsNearlyEqual(Dir.SizeSquared(), (FReal)1, (FReal)UE_KINDA_SMALL_NUMBER));
633 ensure(Thickness == 0 || (FMath::IsNearlyEqual(MScale[0], MScale[1]) && FMath::IsNearlyEqual(MScale[0], MScale[2]))); //non uniform turns sphere into an ellipsoid so no longer a raycast and requires a more expensive sweep
634
635 const FVec3 UnscaledStart = MInvScale * StartPoint;
636 const FVec3 UnscaledDirDenorm = MInvScale * Dir;
637 const FReal LengthScale = UnscaledDirDenorm.Size();
638 if (ensure(LengthScale > TNumericLimits<FReal>::Min()))
639 {
640 const FReal LengthScaleInv = FReal(1) / LengthScale;
641 const FReal UnscaledLength = Length * LengthScale;
642 const FVec3 UnscaledDir = UnscaledDirDenorm * LengthScaleInv;
643
644 FVec3 UnscaledPosition;
645 FVec3 UnscaledNormal;
646 FReal UnscaledTime;
647
648 if (MObject->Raycast(UnscaledStart, UnscaledDir, UnscaledLength, Thickness * MInvScale[0], UnscaledTime, UnscaledPosition, UnscaledNormal, OutFaceIndex))
649 {
650 //We double check that NewTime < Length because of potential precision issues. When that happens we always keep the shortest hit first
651 const FReal NewTime = LengthScaleInv * UnscaledTime;
652 if (NewTime == 0) // Normal/Position output may be uninitialized with TOI 0 so we use ray information for that as the ray origin is likely inside the shape
653 {
654 OutPosition = StartPoint;
655 OutNormal = -Dir;
656 OutTime = NewTime;
657 return true;
658 }
659 else if (NewTime < Length)
660 {
661 OutPosition = MScale * UnscaledPosition;
662 OutNormal = (MInvScale * UnscaledNormal).GetSafeNormal(TNumericLimits<FRealSingle>::Min());
663 OutTime = NewTime;

673 template <typename QueryGeomType>
674 bool LowLevelSweepGeom(const QueryGeomType& B, const TRigidTransform<T, d>& BToATM, const TVector<T, d>& LocalDir, const T Length, T& OutTime, TVector<T, d>& LocalPosition, TVector<T, d>& LocalNormal, int32& OutFaceIndex, TVector<T, d>& OutFaceNormal, T Thickness = 0, bool bComputeMTD = false) const
675 {
676 ensure(Length >= 0);
677 ensure(FMath::IsNearlyEqual(LocalDir.SizeSquared(), (T)1, (T)UE_KINDA_SMALL_NUMBER));
678 ensure(Thickness == 0 || (FMath::IsNearlyEqual(MScale[0], MScale[1]) && FMath::IsNearlyEqual(MScale[0], MScale[2])));
679
680 const TVector<T, d> UnscaledDirDenorm = MInvScale * LocalDir;
681 const T LengthScale = UnscaledDirDenorm.Size();
682 if (ensure(LengthScale > TNumericLimits<T>::Min()))
683 {
684 const T LengthScaleInv = 1.f / LengthScale;
685 const T UnscaledLength = Length * LengthScale;
686 const TVector<T, d> UnscaledDir = UnscaledDirDenorm * LengthScaleInv;
687
688
689
690 TRigidTransform<T, d> BToATMNoScale(BToATM.GetLocation() * FVec3(MInvScale), BToATM.GetRotation());
691
692 constexpr bool bIncludeScaleInResults = std::is_same_v<TConcrete, FTriangleMeshImplicitObject>;
693 if constexpr (!bIncludeScaleInResults)
694 {
695 TVector<T, d> UnscaledPosition;
696 TVector<T, d> UnscaledNormal;
697 T UnscaledTime;
698 if (MObject->SweepGeom(B, BToATMNoScale, UnscaledDir, UnscaledLength, UnscaledTime, UnscaledPosition, UnscaledNormal, OutFaceIndex, OutFaceNormal, Thickness, bComputeMTD, MScale))
699 {
700 const T NewTime = LengthScaleInv * UnscaledTime;
701 //We double check that NewTime < Length because of potential precision issues. When that happens we always keep the shortest hit first
702 if (NewTime < Length)
703 {
704 OutTime = NewTime;
705 LocalPosition = MScale * UnscaledPosition;
706 LocalNormal = (MInvScale * UnscaledNormal).GetSafeNormal();

708 }
709 }
710 }
711 else
712 {
713 TVector<T, d> OutScaledPosition;
714 TVector<T, d> OutScaledNormal;
715 T OutScaledTime;
716 if (MObject->SweepGeom(B, BToATMNoScale, UnscaledDir, UnscaledLength, OutScaledTime, OutScaledPosition, OutScaledNormal, OutFaceIndex, OutFaceNormal, Thickness, bComputeMTD, MScale, bIncludeScaleInResults))
717 {
718 //We double check that OutScaledTime < Length because of potential precision issues. When that happens we always keep the shortest hit first
719 if (OutScaledTime < Length)
720 {
721 OutTime = OutScaledTime;
722 LocalPosition = OutScaledPosition;
723 LocalNormal = OutScaledNormal.GetSafeNormal();
724 if (LocalNormal.IsNearlyZero())
725 {
726 // Fallback: use negated sweep direction when GJK produces degenerate normal
727 UE_LOGF(LogChaos, Warning, "LowLevelSweepGeom: OutScaledNormal produced zero normal after GetSafeNormal. OutScaledNormal: (%f, %f, %f), MScale: (%f, %f, %f)",
728 OutScaledNormal[0], OutScaledNormal[1], OutScaledNormal[2], MScale[0], MScale[1], MScale[2]);

732 }

734 }
735 }

994 FORCEINLINE_DEBUGGABLE FVec3 Support(const FVec3& Direction, const FReal Thickness, int32& VertexIndex) const
995 {
996 // Support_obj(dir) = pt => for all x in obj, pt \dot dir >= x \dot dir
997 // We want Support_objScaled(dir) = Support_obj(dir') where dir' is some modification of dir so we can use the unscaled support function
998 // If objScaled = Aobj where A is a transform, then we can say Support_objScaled(dir) = pt => for all x in obj, pt \dot dir >= Ax \dot dir
999 // But this is the same as pt \dot dir >= dir^T Ax = (dir^TA) x = (A^T dir)^T x
1000 //So let dir' = A^T dir.
1001 //Since we only support scaling on the principal axes A is a diagonal (and therefore symmetric) matrix and so a simple component wise multiplication is sufficient
1002 const FVec3 UnthickenedPt = MObject->Support(Direction * MScale, 0.0f, VertexIndex) * MScale;
1003 return Thickness > 0 ? FVec3(UnthickenedPt + Direction.GetSafeNormal() * Thickness) : UnthickenedPt;
1004 }
