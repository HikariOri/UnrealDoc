22class FCachedFileHandle : public IFileHandle
23{
24public:
25 FCachedFileHandle(IFileHandle* InFileHandle, bool bInReadable, bool bInWritable)
26 : FileHandle(InFileHandle)
27 , FilePos(InFileHandle->Tell())
28 , TellPos(FilePos)
29 , FileSize(InFileHandle->Size())
30 , bWritable(bInWritable)
31 , bReadable(bInReadable)
32 , CurrentCache(0)
33 {
34 FlushCache();
35 }
36
37 virtual ~FCachedFileHandle() = default;
38
39
40 virtual int64 Tell() override
41 {
42 return FilePos;
43 }
44
45 virtual bool Seek(int64 NewPosition) override
46 {
47 if (NewPosition < 0 || NewPosition > FileSize)
48 {
49 return false;
50 }
51 FilePos=NewPosition;
52 return true;
53 }
54
55 virtual bool SeekFromEnd(int64 NewPositionRelativeToEnd = 0) override
56 {
57 return Seek(FileSize - NewPositionRelativeToEnd);
58 }
59
60 virtual bool Read(uint8* Destination, int64 BytesToRead) override
61 {
62 if (!bReadable || BytesToRead < 0 || (BytesToRead + FilePos > FileSize))
63 {
64 return false;
65 }
66
67 if (BytesToRead == 0)
68 {
69 return true;
70 }
71
72 bool Result = false;
73 if (BytesToRead > BufferCacheSize) // reading more than we cache
74 {
75 // if the file position is within the cache, copy out the remainder of the cache
76 int32 CacheIndex=GetCacheIndex(FilePos);
77 if (CacheIndex < CacheCount)
78 {
79 int64 CopyBytes = CacheEnd[CacheIndex]-FilePos;
80 FMemory::Memcpy(Destination, BufferCache[CacheIndex]+(FilePos-CacheStart[CacheIndex]), CopyBytes);
81 FilePos += CopyBytes;
82 BytesToRead -= CopyBytes;
83 Destination += CopyBytes;
84 }
85
86 if (InnerSeek(FilePos))
87 {
88 Result = InnerRead(Destination, BytesToRead);
89 }
90 if (Result)
91 {
92 FilePos += BytesToRead;
93 }
94 }
95 else
96 {
97 Result = true;
98
99 while (BytesToRead && Result)
100 {
101 uint32 CacheIndex=GetCacheIndex(FilePos);
102 if (CacheIndex > CacheCount)
103 {
104 // need to update the cache
105 uint64 AlignedFilePos=FilePos&BufferSizeMask; // Aligned Version
106 uint64 SizeToRead=FMath::Min<uint64>(BufferCacheSize, FileSize-AlignedFilePos);
107 InnerSeek(AlignedFilePos);
108 Result = InnerRead(BufferCache[CurrentCache], SizeToRead);
109
110 if (Result)
111 {
112 CacheStart[CurrentCache] = AlignedFilePos;
113 CacheEnd[CurrentCache] = AlignedFilePos+SizeToRead;
114 CacheIndex = CurrentCache;
115 // move to next cache for update
116 CurrentCache++;
117 CurrentCache%=CacheCount;
118 }
119 }
120
121 // copy from the cache to the destination
122 if (Result)
123 {
124 // Analyzer doesn't see this - if this code ever changes make sure there are no buffer overruns!
125 CA_ASSUME(CacheIndex < CacheCount);
126 uint64 CorrectedBytesToRead=FMath::Min<uint64>(BytesToRead, CacheEnd[CacheIndex]-FilePos);
127 FMemory::Memcpy(Destination, BufferCache[CacheIndex]+(FilePos-CacheStart[CacheIndex]), CorrectedBytesToRead);
128 FilePos += CorrectedBytesToRead;
129 Destination += CorrectedBytesToRead;
130 BytesToRead -= CorrectedBytesToRead;
131 }
132 }
133 }
134 return Result;
135 }
136
137 virtual bool ReadAt(uint8* Destination, int64 BytesToRead, int64 Offset) override
138 {
139 // concurrent reads won't be cached yet
140 return FileHandle->ReadAt(Destination, BytesToRead, Offset);
141 }
142
143 virtual bool Write(const uint8* Source, int64 BytesToWrite) override
144 {
145 if (!bWritable || BytesToWrite < 0)
146 {
147 return false;
148 }
149
150 if (BytesToWrite == 0)
151 {
152 return true;
153 }
154
155 InnerSeek(FilePos);
156 bool Result = FileHandle->Write(Source, BytesToWrite);
157 if (Result)
158 {
159 FilePos += BytesToWrite;
160 FileSize = FMath::Max<int64>(FilePos, FileSize);
161 FlushCache();
162 TellPos = FilePos;
163 }
164 return Result;
165 }
166
167 virtual int64 Size() override
168 {
169 return FileSize;
170 }
171
172 virtual bool Flush(const bool bFullFlush = false) override
173 {
174 if (bWritable)
175 {
176 return FileHandle->Flush(bFullFlush);
177 }
178 return false;
179 }
180
181 virtual bool Truncate(int64 NewSize) override
182 {
183 if (bWritable)
184 {
185 if (FileHandle->Truncate(NewSize))
186 {
187 FlushCache();
188 FilePos = TellPos = FileHandle->Tell();
189 FileSize = FileHandle->Size();
190 return true;
191 }
192 }
193 return false;
194 }
195
196 virtual void ShrinkBuffers() override
197 {
198 FileHandle->ShrinkBuffers();
199 }
200
201private:
202
203 static constexpr uint32 BufferCacheSize = 64 * 1024; // Seems to be the magic number for best perf
204 static constexpr uint64 BufferSizeMask = ~((uint64)BufferCacheSize-1);
205 static constexpr uint32 CacheCount = 2;
206
207 bool InnerSeek(uint64 Pos)
208 {
209 if (Pos==TellPos)
210 {
211 return true;
212 }
213 bool bOk=FileHandle->Seek(Pos);
214 if (bOk)
215 {
216 TellPos=Pos;
217 }
218 return bOk;
219 }
220 bool InnerRead(uint8* Dest, uint64 BytesToRead)
221 {
222 if (FileHandle->Read(Dest, BytesToRead))
223 {
224 TellPos += BytesToRead;
225 return true;
226 }
227 return false;
228 }
229 int32 GetCacheIndex(int64 Pos) const
230 {
231 for (uint32 i=0; i<UE_ARRAY_COUNT(CacheStart); ++i)
232 {
233 if (Pos >= CacheStart[i] && Pos < CacheEnd[i])
234 {
235 return i;
236 }
237 }
238 return CacheCount+1;
239 }
240 void FlushCache()
241 {
242 for (uint32 i=0; i<CacheCount; ++i)
243 {
244 CacheStart[i] = CacheEnd[i] = -1;
245 }
246 }
247
248 TUniquePtr<IFileHandle> FileHandle;
249 int64 FilePos; /* Desired position in the file stream, this can be different to FilePos due to the cache */
250 int64 TellPos; /* Actual position in the file, this can be different to FilePos */
251 int64 FileSize;
252 bool bWritable;
253 bool bReadable;
254 uint8 BufferCache[CacheCount][BufferCacheSize];
255 int64 CacheStart[CacheCount];
256 int64 CacheEnd[CacheCount];
257 int32 CurrentCache;
258};
