168 FScopedHighPriorityWaitTracker(std::atomic<int32>* InCounter, Private::ELockPriority InPriority)
169 : Counter((InPriority == Private::PriorityHigh) ? InCounter : nullptr)
170 {
171 if (Counter)
172 {
173 UE_AUTORTFM_OPEN
174 {
175 Counter->fetch_add(1, std::memory_order_relaxed);
176 };
177 AutoRTFM::PushOnAbortHandler(this, [this]
178 {
179 Counter->fetch_sub(1, std::memory_order_relaxed);
180 });
181 }
182 }
183
184 ~FScopedHighPriorityWaitTracker()
185 {
186 if (Counter)
187 {
188 UE_AUTORTFM_OPEN
189 {
190 Counter->fetch_sub(1, std::memory_order_relaxed);
191 };
192 AutoRTFM::PopOnAbortHandler(this);
193 }
194 }
195
196private:
197 std::atomic<int32>* Counter = nullptr;

229 : Lock(InLockObject)

234 GuardWrapper.Emplace(Lock, LockType);
235 }
236
237 // NOTE: As the name suggests, this function should be used with caution.
238 // It releases the read lock _before_ acquiring a new write lock. This is not an atomic operation and the caller should
239 // not treat it as such.
240 // E.g. Pointers read from protected data structures prior to this call may be invalid after the function is called.
241 void ReleaseReadOnlyLockAndAcquireWriteLock_USE_WITH_CAUTION()
242 {
243 if (LockType == SLT_ReadOnly)
244 {
245 GuardWrapper.Reset();
246 {
247 FScopedHighPriorityWaitTracker Tracker(&Lock.HighPriorityWaitersCount, Priority);
248 GuardWrapper.Emplace(Lock, SLT_Write);
249 }
250 LockType = SLT_Write;
251 }
252 }
253
254 Private::FRWLockWithPriority& Lock;
255 TOptional<TRWScopeLock<Private::FRWLockWithPriority>> GuardWrapper;
256 Private::ELockPriority Priority;
257 FRWScopeLockType LockType;
