61void FRecvStats::UpdateRecv(uint32 NowMs, uint32 RecvSizeKiB)
62{
63 if (RecvSizeKiB == 0)
64 {
65 return;
66 }
67
68 TotalRecvKiB.fetch_add(RecvSizeKiB, std::memory_order_relaxed);
69
70 // Quantize time for a modicum of stability.
71 uint32 NowQuantMs = NowMs - (NowMs % BucketMs);
72
73 uint32 DeltaMs = NowQuantMs - LastMs;
74 LastMs = NowQuantMs;
75
76 Accumulator.TimeMs += DeltaMs;
77 if (Accumulator.TimeMs < BucketMs)
78 {
79 Accumulator.RecvKiB += RecvSizeKiB;
80 return;
81 }
82
83 uint32 RecvKiB = Accumulator.RecvKiB;
84 uint32 TimeMs = NowQuantMs - DeltaMs;
85 if (Accumulator.TimeMs > BucketMs)
86 {
87 Accumulator = {
88 .RecvKiB = RecvSizeKiB,
89 .TimeMs = 0,
90 };
91 }
92 else
93 {
94 uint32 Partial = ((NowMs - NowQuantMs) * RecvSizeKiB) / BucketMs;
95
96 RecvKiB += RecvSizeKiB - Partial,
97 Accumulator = {
98 .RecvKiB = Partial,
99 .TimeMs = BucketMs - Accumulator.TimeMs,
100 };
101 TimeMs = NowQuantMs - Accumulator.TimeMs;
102 }
103
104 uint64 Value = (uint64(TimeMs) << 32u) + RecvKiB;
105
106 Cursor += 1;
107 uint32 Index = (Cursor - 1) % UE_ARRAY_COUNT(Buckets);
108 Buckets[Index].Value.store(Value, std::memory_order_release);
109}

112void FRecvStats::UpdateTtfb(uint32 TtfbMs)

116 return;

125void FRecvStats::Flush()
126{
127 if (Accumulator.RecvKiB == 0)
128 {
129 return;
130 }
131
132 uint64 Value = (uint64(LastMs) << 32u) + Accumulator.RecvKiB;
133 uint32 Index = Cursor % UE_ARRAY_COUNT(Buckets);
134 Buckets[Index].Value.store(Value, std::memory_order_release);

138void FRecvStats::Sample(uint32 NowMs, FLoopPerf& Out) const
139{
140 static_assert((BucketMs * UE_ARRAY_COUNT(Buckets)) / 1000 == UE_ARRAY_COUNT(Out.RecvKiBps));
141
142 uint32 NowQuantMs = NowMs - (NowMs % BucketMs);
143
144 union FData
145 {
146 FBucket::FData Data;
147 uint64 Value;
148 };
149
150 // Sample all the buckets
151 FData Datas[UE_ARRAY_COUNT(Buckets)];
152 for (uint32 i = UE_ARRAY_COUNT(Buckets); i--;)
153 {
154 Datas[i].Value = Buckets[i].Value.load(std::memory_order_acquire);
155 }
156
157 // Find the newest bucket
158 int32 Index = -1;
159 for (uint32 i = 0, PrevMs = 0u; const FData& Data : Datas)
160 {
161 if (uint32 TimeMs = Data.Data.TimeMs; TimeMs >= PrevMs)
162 {
163 Index = int32(i);
164 PrevMs = TimeMs;
165 }
166 ++i;
167 }
168
169 // Gather samples for the last N seconds.
170 for (uint32 i = 0, n = UE_ARRAY_COUNT(Out.RecvKiBps), TimeMs = NowQuantMs; i < n; ++i)
171 {
172 const FBucket::FData& Data = Datas[Index].Data;
173
174 TimeMs -= BucketMs;
175 if (TimeMs > Data.TimeMs)
176 {
177 Out.RecvKiBps[i] = 0;
178 continue;
179 }
180
181 uint32 Value = Data.RecvKiB;
182 Value = (Value > 0xffff) ? 0xffff : Value;
183 Out.RecvKiBps[i] = uint16(Value);
184
185 Index = (Index - 1) % UE_ARRAY_COUNT(Buckets);
186 }
187
188 Out.RecvKiB = TotalRecvKiB.load(std::memory_order_relaxed);
189
190 // Time to first byte
191 Out.TtfbSamples = TtfbSamples.load(std::memory_order_relaxed);
192}

233 uint32 Numer = 0;
234 for (uint32 i = RecvBucketCount; uint32 Sample : RecvKiBps)
235 {

248uint32 FLoopPerf::GetTimeToFirstByteMs() const
249{
250 uint64 Samples = TtfbSamples;
251 uint32 Accumulator = 0;
252 uint32 Prev = 0;
253 for (uint32 i = 0, n = 8; i < n; ++i, Samples >>= 8)
254 {
255 uint32 Value = Samples & 0xff;
256 if (Value == 0)
257 {
258 Value = Prev;
259 }
260 Accumulator += Value;
261 Prev = Value;
262 }
263 return Accumulator >> 3;
264}

270/*
271 * - Activities (requests send with a loop) are managed in singly-linked lists
272 * - Each activity has an associated host it is talking to.
273 * - Hosts are ephemeral, or represented externally via a FConnectionPool object
274 * - Loop has a group for each host, and each host-group has a bunch of socket-groups
275 * - Host-group has a list of work; pending activities waiting to start
276 * - Socket-groups own up to two activities; one sending, one receiving
277 * - As it recvs, a socket-group will, if possible, fetch more work from the host
278 *
279 * Loop:
280 * FHostGroup[HostPtr]:
281 * Work: Act0 -> Act1 -> Act2 -> Act3 -> ...
282 * FPeerGroup[0...HostMaxConnections]:
283 * Act.Send
284 * Act.Recv
285 */

317FActivityNode* FActivityNode::Create(FParams& Params, FAnsiStringView Url, FCertRootsRef VerifyCert)
318{
319 static_assert(alignof(FActivityNode) <= 16);
320 static_assert(alignof(FHost) <= alignof(FActivityNode));
321
322 check(!Params.Method.IsEmpty());
323
324 if (Url.IsEmpty())
325 {
326 uint32 BufferSize = Params.BufferSize;
327 BufferSize = (BufferSize >= 128) ? BufferSize : 128;
328 BufferSize = (BufferSize + 15) & ~15;
329
330 uint32 AllocSize = sizeof(FActivityNode) + BufferSize;
331 auto* Ptr = (FActivityNode*)FMemory::Malloc(AllocSize, alignof(FActivityNode));
332
333 check(Params.Host != nullptr);
334 check(Params.Host->IsPooled());
335
336 Params.Buffer = (char*)(Ptr + 1);
337 Params.BufferSize = uint16(BufferSize);
338 Params.bIsKeepAlive = true;
339 return new (Ptr) FActivityNode(Params);
340 }
341
342 // Parse the URL into its components
343 FUrlOffsets UrlOffsets;
344 if (ParseUrl(Url, UrlOffsets) < 0)
345 {
346 return nullptr;
347 }
348
349 FAnsiStringView HostName = UrlOffsets.HostName.Get(Url);
350
351 uint32 Port = 0;
352 if (UrlOffsets.Port)
353 {
354 FAnsiStringView PortView = UrlOffsets.Port.Get(Url);
355 Port = uint32(CrudeToInt(PortView));
356 }
357
358 FAnsiStringView Path;
359 if (UrlOffsets.Path > 0)
360 {
361 Path = Url.Mid(UrlOffsets.Path);
362 }
363
364 uint32 BufferSize = Params.BufferSize;
365 BufferSize = (BufferSize >= 128) ? BufferSize : 128;
366 BufferSize += sizeof(FHost);
367 BufferSize += HostName.Len() + 1;
368 BufferSize = (BufferSize + 15) & ~15;
369
370 uint32 AllocSize = sizeof(FActivityNode) + BufferSize;
371 auto* Ptr = (FActivityNode*)FMemory::Malloc(AllocSize, alignof(FActivityNode));
372
373 // Create an emphemeral host
374 if (UrlOffsets.SchemeLength == 5)
375 {
376 if (VerifyCert == ECertRootsRefType::None)
377 {
378 VerifyCert = FCertRoots::Default();
379 }
380 check(VerifyCert != ECertRootsRefType::None);
381 }
382 else
383 {
384 VerifyCert = FCertRoots::NoTls();
385 }
386
387 FHost* Host = (FHost*)(Ptr + 1);
388
389 char* HostNamePtr = (char*)(Host + 1);
390 uint32 HostNameLength = HostName.Len();
391 memcpy(HostNamePtr, HostName.GetData(), HostNameLength);
392 HostNamePtr[HostNameLength] = '\0';
393
394 EHttpVersion Version = Params.HttpVersion;
395 Version = (Version != EHttpVersion::Two) ? EHttpVersion::One : Version;
396
397 new (Host) FHost({
398 .HostName = HostNamePtr,
399 .Port = Port,
400 .HttpVersion= Version,
401 .VerifyCert = VerifyCert,
402 });
403
404 HostNameLength = (HostNameLength + 8) & ~7;
405
406 check(Params.Host == nullptr);
407 Params.Path = Path;
408 Params.Host = Host;
409 Params.Buffer = HostNamePtr + HostNameLength;
410 Params.BufferSize = uint16(AllocSize - ptrdiff_t(Params.Buffer - (char*)Ptr));
411 Params.bIsKeepAlive = false;
412 return new (Ptr) (FActivityNode)(Params);
413}

463{

622 {
623 return nullptr;

634void FWorkQueue::TickCancels(FTickState& State)
635{
636 if (State.Cancels == 0 || (State.Cancels & ActiveSlots) == 0)
637 {
638 return;
639 }
640
641 // We are going to rebuild the list of activities to maintain order as the
642 // activity list is singular.
643
644 check(!List.IsEmpty());
645 ActiveSlots = 0;
646
647 FActivityNode* Activity = List.Detach();
648 for (FActivityNode* Next; Activity != nullptr; Activity = Next)
649 {
650 Next = Activity->Next;
651 Activity->Next = nullptr;
652
653 if (uint64 Slot = (1ull << Activity->Slot); (State.Cancels & Slot) == 0)
654 {
655 AddActivity(Activity);
656 continue;
657 }
658
659 Activity->Cancel();
660
661 State.DoneList.Prepend(Activity);
662 }

762bool FPeerGroup::MaybeStartNewWork(FTickState& State)
763{
764 if (InflightNum >= InflightMax) return false;
765 if (IsKeepAlive == 0) return false;
766
767 if (!State.Work->HasWork())
768 {
769 if (InflightNum == 0)
770 {
771 LastUseMs = State.NowMs;
772 }
773 return false;
774 }
775
776 FActivityNode* Activity = State.Work->PopActivity();
777 do
778 {
779 Trace(Activity, ETrace::StartWork);
780
781 check(Activity->GetHost()->IsPooled());
782
783 Send.Append(Activity);
784 InflightNum++;
785 if (InflightNum == InflightMax)
786 {
787 break;
788 }
789
790 Activity = State.Work->PopActivity();
791 }
792 while (Activity != nullptr);
793
794 return true;
795}

823void FPeerGroup::RecvInternal(FTickState& State)
824{
825 check(bNegotiating == false);
826 check(!Recv.IsEmpty());
827
828 FActivityNode* Activity = nullptr;
829
830 FOutcome Outcome = Peer.GetPendingTransactId();
831 if (Outcome.IsWaiting())
832 {
833 bWaiting |= true;
834 return;
835 }
836
837 while (Outcome.IsOk())
838 {
839 uint32 TransactId = Outcome.GetResult();
840 Activity = Recv.MoveToHead(
841 [TransactId] (FActivityNode* Activity)
842 {
843 return Activity->GetTransactId() == TransactId;
844 }
845 );
846
847 if (Activity == nullptr)
848 {
849 // When an activity is cancelled, its sink is called and the activity retired. If
850 // supported then the peer may also be informed about the cancel and it will not
851 // send any further related frames. Some frames however may already be on the wire
852 // and when received, have no corresponding activity. So here we have to skip over
853 // transactions we are unable to match to an activity.
854 if (GInflightCancels)
855 {
856 Peer.SkipPendingTransactId();
857 return RecvInternal(State);
858 }
859
860 Outcome = FOutcome::Error("could not find work for transact id", TransactId);
861 break;
862 }
863
864 Outcome = Activity->Tick(Peer, &State.RecvAllowance);
865 break;
866 }
867
868 // Any sort of error here is unrecoverable
869 if (Outcome.IsError())
870 {
871 Fail(State, Outcome);
872 return;
873 }
874
875 check(Activity != nullptr);
876
877 IsKeepAlive &= uint8(Activity->GetTransaction()->IsKeepAlive() == true);
878 bWaiting |= (Outcome.IsWaiting() && Outcome.IsWaitData());
879
880 // If there was no data available this is far as receiving can go
881 if (Outcome.IsWaiting())
882 {
883 return;
884 }
885
886 // If we have a time here we are timing time-to-first-byte
887 if (LastUseMs != 0)
888 {
889 uint32 TtfbMs = GetNowMs() - LastUseMs;
890 TtfbMs = (TtfbMs <= 0xff) ? TtfbMs : 0xff;
891 State.TtfbMs = (State.TtfbMs << 8) | TtfbMs;
892 LastUseMs = 0;
893 }
894
895 // If an okay result is -1, then message recv is done and content is next
896 // and it is likely that we have it.
897 if (Outcome.GetResult() == uint32(FActivity::EStage::Response))
898 {
899 return RecvInternal(State);
900 }
901
902 check(Outcome.GetResult() == uint32(FActivity::EStage::Content));
903
904 check(Recv.GetHead() == Activity);
905 Recv.PopPrepend(State.DoneList);
906
907 check(InflightNum > 0);
908 InflightNum--;
909
910 if (MaybeStartNewWork(State))
911 {
912 SendInternal(State);
913 }
914
915 // If the server wants the connection closed then we can drop our peer
916 if (IsKeepAlive == 0)
917 {
918 Peer = FHttpPeer();
919 }
920}

923void FPeerGroup::SendInternal(FTickState& State)
924{
925 check(bNegotiating == false);
926 check(IsKeepAlive == 1);
927 check(!Send.IsEmpty());
928
929 FActivityNode* Activity = Send.GetHead();
930
931 FOutcome Outcome = Activity->Tick(Peer);
932
933 if (Outcome.IsWaiting())
934 {
935 bWaiting = true;
936 return;
937 }
938
939 if (Outcome.IsError())
940 {
941 Fail(State, Outcome);
942 return;
943 }
944
945 check(Outcome.GetResult() == uint32(FActivity::EStage::Request));
946
947 Send.PopAppend(Recv);
948
949 if (!Send.IsEmpty())
950 {
951 return SendInternal(State);
952 }

956bool FPeerGroup::Tick(FTickState& State)
957{
958 if (bNegotiating)
959 {
960 Negotiate(State);
961 }
962
963 else if (!Send.IsEmpty())
964 {

966 }

982void FPeerGroup::TickSend(FTickState& State, FHost& Host, FPoller& Poller)
983{
984 // This path is only for those that have spare capacity
985 if (InflightNum >= InflightMax)
986 {
987 return;
988 }
989
990 // Failing will try and recover work which we don't want to happen yet
991 FActivityNode* Pending = State.Work->PopActivity();
992 check(Pending != nullptr);
993
994 // Close idle sockets
995 if (uint32 TimeoutMs = LastUseMs + GIdleMs; LastUseMs && TimeoutMs < State.NowMs)
996 {
997 if (InflightNum == 0)
998 {
999 LastUseMs = 0;
1000 Peer = FHttpPeer();
1001 }
1002 }
1003
1004 // We don't have a connected socket on first use, or if a keep-alive:close
1005 // was received from the server. So we connect here.
1006 bool bWillBlock = false;
1007 if (!Peer.IsValid())
1008 {
1009 FOutcome Outcome = FOutcome::None();
1010
1011 FSocket Socket;
1012 if (Socket.Create())
1013 {
1014 FWaitable Waitable = Socket.GetWaitable();
1015 Poller.Register(Waitable);
1016
1017 Outcome = Host.Connect(Socket);
1018 }
1019 else
1020 {
1021 Outcome = FOutcome::Error("Failed to create socket");
1022 }
1023
1024 if (Outcome.IsError())
1025 {
1026 // We failed to connect, let's bail.
1027 Recv.Prepend(Pending);
1028 Fail(State, Outcome);
1029 return;
1030 }
1031
1032 IsKeepAlive = 1;
1033 bNegotiating = true;
1034 bWillBlock = Outcome.IsWaiting();
1035 InflightNum = 0;
1036 LastRecvSize = 0;
1037
1038 FHttpPeer::FParams Params = {
1039 .Socket = MoveTemp(Socket),
1040 .Certs = Host.GetVerifyCert(),
1041 .HostName = Host.GetHostName().GetData(),
1042 .HttpVersion = Host.GetHttpVersion(),
1043 };
1044 Peer = FHttpPeer(MoveTemp(Params));
1045 }
1046
1047 // If we are idle we can time the time-to-first-byte
1048 if (!bNegotiating && (InflightNum == 0) && Recv.IsEmpty())
1049 {
1050 LastUseMs = GetNowMs();
1051 }
1052
1053 Send.Append(Pending);
1054 InflightNum++;
1055
1056 MaybeStartNewWork(State);
1057
1058 if (!bWillBlock)
1059 {
1060 if (bNegotiating)
1061 {
1062 return Negotiate(State);
1063 }
1064 return SendInternal(State);
1065 }
1066
1067 // Non-blocking connect
1068 bWaiting = true;
1069}

1072void FPeerGroup::TickCancels(FTickState& State)
1073{
1074 if (State.Cancels == 0 || !GInflightCancels)
1075 {
1076 return;
1077 }
1078
1079 auto Impl = [this, &State] (FActivityList& List)
1080 {
1081 if (List.IsEmpty())
1082 {
1083 return;
1084 }
1085
1086 for (FActivityNode* Next, *Activity = List.Detach(); Activity != nullptr; Activity = Next)
1087 {
1088 Next = Activity->Next;
1089 Activity->Next = nullptr;
1090
1091 bool bCancelled = false;
1092 if (uint64 Slot = (1ull << Activity->Slot); (State.Cancels & Slot) != 0)
1093 {
1094 bCancelled = Activity->CancelTransaction(Peer);
1095 }
1096
1097 if (!bCancelled)
1098 {
1099 List.Prepend(Activity);
1100 continue;
1101 }
1102
1103 Activity->Cancel();
1104 State.DoneList.Prepend(Activity);
1105 --InflightNum;
1106 }
1107
1108 List.Reverse();
1109 };
1110
1111 Impl(Send);
1112 Impl(Recv);
1113}

1155int32 FHostGroup::Wait(const FTickState& State)
1156{
1157 // Collect groups that are waiting on something
1158 TArray<FWaiter, TFixedAllocator<64>> Waiters;
1159 for (uint32 i = 0, n = PeerGroups.Num(); i < n; ++i)
1160 {
1161 FWaiter Waiter = PeerGroups[i].GetWaiter();
1162 if (!Waiter.IsValid())
1163 {
1164 continue;
1165 }
1166
1167 Waiter.SetIndex(i);
1168 Waiters.Add(MoveTemp(Waiter));
1169 }
1170
1171 if (Waiters.IsEmpty())
1172 {
1173 return 0;
1174 }
1175
1176 Trace(ETrace::Wait);
1177 ON_SCOPE_EXIT { Trace(ETrace::Unwait); };
1178
1179 // If the poll timeout is negative then treat that as a fatal timeout
1180 check(State.FailTimeoutMs);
1181 int32 PollTimeoutMs = State.PollTimeoutMs;
1182 if (PollTimeoutMs < 0)
1183 {
1184 PollTimeoutMs = State.FailTimeoutMs;
1185 }
1186
1187 // Actually do the wait
1188 int32 Result = FWaiter::Wait(Waiters, Poller, PollTimeoutMs);
1189 if (Result <= 0)
1190 {
1191 // If the user opts to not block then we don't accumulate wait time and
1192 // leave it to them to manage time a fail timoue
1193 WaitTimeAccum += PollTimeoutMs;
1194
1195 if (State.PollTimeoutMs < 0 || WaitTimeAccum >= State.FailTimeoutMs)
1196 {
1197 return MIN_int32;
1198 }
1199
1200 return Result;
1201 }
1202
1203 WaitTimeAccum = 0;
1204
1205 // For each waiter that's ready, find the associated group "unwait" them.
1206 int32 Count = 0;
1207 for (int32 i = 0, n = Waiters.Num(); i < n; ++i)
1208 {
1209 if (!Waiters[i].IsReady())
1210 {
1211 continue;
1212 }
1213
1214 uint32 Index = Waiters[i].GetIndex();
1215 check(Index < uint32(PeerGroups.Num()));
1216 PeerGroups[Index].Unwait();
1217
1218 Waiters.RemoveAtSwap(i, EAllowShrinking::No);
1219 --n, --i, ++Count;
1220 }
1221 check(Count == Result);
1222
1223 return Result;
1224}

1227void FHostGroup::Tick(FTickState& State)
1228{
1229 State.Work = &Work;
1230
1231 if (BusyCount = Work.HasWork(); BusyCount)
1232 {
1233 Work.TickCancels(State);
1234
1235 // Get available work out on idle sockets as soon as possible
1236 for (FPeerGroup& Group : PeerGroups)
1237 {
1238 if (!Work.HasWork())
1239 {
1240 break;
1241 }
1242
1243 Group.TickSend(State, Host, Poller);
1244 }
1245 }
1246
1247 // Wait on the groups that are
1248 if (int32 Result = Wait(State); Result < 0)
1249 {
1250 FOutcome Reason = (Result == MIN_int32)
1251 ? FOutcome::Error("FailTimeout hit")
1252 : FOutcome::Error("poll() returned an unexpected error");
1253
1254 for (FPeerGroup& Group : PeerGroups)
1255 {
1256 Group.Fail(State, Reason);
1257 }
1258
1259 return;
1260 }
1261
1262 // Tick everything, starting with groups that are maybe closest to finishing
1263 for (FPeerGroup& Group : PeerGroups)
1264 {
1265 BusyCount += (Group.Tick(State) == true);
1266 }
1267
1268 // Gather how much each peer has recv'd since the last time
1269 for (FPeerGroup& Group : PeerGroups)
1270 {
1271 State.RecvSizeKiB += Group.CollectRecvSizeKiB();
1272 }

1408 if (FreeSlotsLoad == PrevFreeSlots)
1409 {
1410 return;
1411 }
1412 PrevFreeSlots = FreeSlotsLoad;
1413
1414 // Fetch the pending activities from out in the wild
1415 FActivityList Collected;
1416 {
1417 FScopeLock _(&Lock);
1418 Swap(Collected, Pending);
1419 }
1420 Collected.Reverse();
1421
1422 // Group activities by their host.
1423 FActivityNode* Activity = Collected.Detach();

1425 {

1431 FHostGroup* Group = Groups.FindByPredicate(Pred);
1432 if (Group == nullptr)

1434 Group = &(Groups.Emplace_GetRef(Host));
1435 }

1439 }

1443uint32 FEventLoop::FImpl::Tick(int32 PollTimeoutMs)
1444{
1445 TRACE_CPUPROFILER_EVENT_SCOPE(IasHttp::Tick);
1446
1447 ReceiveWork();
1448
1449 // We limit recv sizes as a way to control bandwidth use.
1450 int32 RecvAllowance = Throttler.GetAllowance();
1451 if (RecvAllowance <= 0)
1452 {
1453 if (PollTimeoutMs == 0)
1454 {
1455 return BusyCount;
1456 }
1457
1458 int32 ThrottleWaitMs = -RecvAllowance;
1459 if (PollTimeoutMs > 0)
1460 {
1461 ThrottleWaitMs = FMath::Min(ThrottleWaitMs, PollTimeoutMs);
1462 }
1463 FPlatformProcess::SleepNoStats(float(ThrottleWaitMs) / 1000.0f);
1464
1465 RecvAllowance = Throttler.GetAllowance();
1466 if (RecvAllowance <= 0)
1467 {
1468 return BusyCount;
1469 }
1470 }
1471
1472 uint64 CancelsLoad = Cancels.load(std::memory_order_relaxed);
1473
1474 // Tick groups
1475 FTickState TickState = {
1476 .Cancels = CancelsLoad,
1477 .RecvAllowance = RecvAllowance,
1478 .PollTimeoutMs = PollTimeoutMs,
1479 .FailTimeoutMs = FailTimeoutMs,
1480 .NowMs = GetNowMs(),
1481 };
1482 for (FHostGroup& Group : Groups)
1483 {
1484 Group.Tick(TickState);
1485 }
1486
1487 // Track some stats
1488 RecvStats.Update(TickState.NowMs, TickState.RecvSizeKiB, TickState.TtfbMs);
1489
1490 // Remove groups that are idle
1491 for (uint32 i = 0, n = Groups.Num(); i < n; ++i)
1492 {
1493 FHostGroup& Group = Groups[i];
1494 if (Group.IsBusy())
1495 {
1496 continue;
1497 }
1498
1499 Groups.RemoveAtSwap(i, EAllowShrinking::No);
1500 --n, --i;
1501 }
1502
1503 Throttler.ReturnUnused(RecvAllowance);
1504
1505 uint64 ReturnedSlots = 0;
1506 for (FActivityNode* Activity = TickState.DoneList.Detach(); Activity != nullptr;)
1507 {
1508 FActivityNode* Next = Activity->Next;
1509 ReturnedSlots |= (1ull << Activity->Slot);
1510
1511 FActivityNode::Destroy(Activity);
1512
1513 --BusyCount;
1514 Activity = Next;
1515 }
1516
1517 uint32 BusyBias = 0;
1518 if (ReturnedSlots)
1519 {
1520 uint64 LatestFree = FreeSlots.fetch_add(ReturnedSlots, std::memory_order_relaxed);
1521 BusyBias += (LatestFree != PrevFreeSlots);
1522 PrevFreeSlots += ReturnedSlots;
1523 }
1524
1525 // To cover the scenario where one of the completed slots gets cancelled
1526 // after the earlier load but before now, we will unset returned slots
1527 // from the cancel mask too.
1528 if (uint64 Mask = CancelsLoad | ReturnedSlots; Mask)
1529 {
1530 Cancels.fetch_and(~Mask, std::memory_order_relaxed);
1531 }
1532
1533 // There is still some accounting in the stats object but there is no reason
1534 // to get any further ticks. Make sure the accounting is externally visible.
1535 uint32 Result = BusyCount + BusyBias;
1536 if (Result == 0)
1537 {
1538 RecvStats.Flush();
1539 }
1540
1541 return Result;
1542}
