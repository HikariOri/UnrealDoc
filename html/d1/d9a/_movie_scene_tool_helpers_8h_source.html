697 /*
698 * Get the channel values at the specified time with the specified sections, will get
699 * the channels from within the specified start and end indices
700 * @param StartIndex Start Index to get values from
701 * @param EndIndex End Index to get values from, should be no greater than the number of channels -1
702 * @param Sections List of non-absolute(additive and override) sections to evaluate. Note this may not be all of the sections in the track,
703 * but just a subset, which allows us to evaluate up to a certain point in the track.
704 * @param AbsoluteSections List of absolute channels to evaluate
705 * @param FrameTime Frame to evaluate
706 * @return Returns the total channel value of those sections for each channel. Remember if OverrideChannelIndex is set this will just be one channel.
707 */
708 template<typename ChannelType, typename CurveValueType>
709 static TArray<CurveValueType> GetChannelValues(const int32 StartIndex, const int32 EndIndex, const TArray<UMovieSceneSection*>& Sections, const TArray<UMovieSceneSection*>& AbsoluteSections, const FFrameNumber& FrameTime)
710 {
711 TArray<CurveValueType> Values;
712 int32 NumChannels = 0;
713 if (Sections.Num() > 0)
714 {
715 TArrayView<ChannelType*>Channels = Sections[0]->GetChannelProxy().GetChannels<ChannelType>();
716 NumChannels = Channels.Num();
717 }
718 else if(AbsoluteSections.Num() > 0)
719 {
720 TArrayView<ChannelType*>Channels = AbsoluteSections[0]->GetChannelProxy().GetChannels<ChannelType>();
721 NumChannels = Channels.Num();
722 }
723 else
724 {
725 UE_LOGF(LogMovieScene, Warning, "GetChannelValues:: Invalid number of channels");
726 return Values;
727 }
728 if (StartIndex < 0 || EndIndex >= NumChannels || EndIndex < StartIndex)
729 {
730 UE_LOGF(LogMovieScene, Warning, "GetChannelValues:: Invalid Start/End indices");
731 return Values;
732 }
733
734 for (int32 ChannelIndex = StartIndex; ChannelIndex <= EndIndex; ++ChannelIndex)
735 {
736 CurveValueType Value = 0.0;
737 if (AbsoluteSections.Num() > 0)
738 {
739 for (UMovieSceneSection* AbsoluteSection : AbsoluteSections)
740 {
741 TArrayView<ChannelType*>Channels = AbsoluteSection->GetChannelProxy().GetChannels<ChannelType>();
742 float Weight = AbsoluteSection->GetTotalWeightValue(FrameTime);
743 CurveValueType WeightedValue = 0.0;
744 ChannelType* Channel = Channels[ChannelIndex];
745 Channel->Evaluate(FrameTime, WeightedValue);
746 WeightedValue *= ((double)Weight);
747 Value += WeightedValue;
748 }
749
750 Value /= (double)AbsoluteSections.Num();
751 }
752
753 for (UMovieSceneSection* Section : Sections)
754 {
755 TArrayView<ChannelType*> Channels = Section->GetChannelProxy().GetChannels<ChannelType>();
756 float Weight = Section->GetTotalWeightValue(FrameTime);
757 CurveValueType WeightedValue = 0.0;
758 ChannelType* Channel = Channels[ChannelIndex];
759 Channel->Evaluate(FrameTime, WeightedValue);
760 if (Section->GetBlendType().Get() == EMovieSceneBlendType::Additive)
761 {
762 WeightedValue *= Weight;
763 Value += WeightedValue;
764 }
765 else if (Section->GetBlendType().Get() == EMovieSceneBlendType::Override)
766 {
767 Value = (Value * (1.0 - Weight)) +

786 * @param OtherSections Set of non-absolute sections for each channel specified.
787 * @param OverrideChanneIndex ChannelIndex for an override merge.
788 * @param MergeAlgorithm The algorithm to use to blend the channels together.
789 * @return Returns true if successful.

804 CurveValueType WeightedValue = 0.0;

807 WeightedValue *= ((double)Weight);
808 InOutValue += WeightedValue;
809 DNumChannels += 1.0;
810 }
811 }

814 InOutValue /= DNumChannels;
815 }
816 }

820 {

822 {
823 float Weight = Sections[WeightIndex]->GetTotalWeightValue(FrameTime);
824 if (Sections[WeightIndex]->GetBlendType().IsValid() == false ||
825 Sections[WeightIndex]->GetBlendType().Get() != EMovieSceneBlendType::Additive)
826 {
827 Weight = 1.0;
828 }
829 CurveValueType WeightedValue = 0.0;
830 ChannelType* EachChannel = Channels[WeightIndex];
831 EachChannel->Evaluate(FrameTime, WeightedValue);
832 WeightedValue *= Weight;
833 InOutValue += WeightedValue;
834 }
835 }
836 }
837 else if (MergeAlgorithm == FChannelMergeAlgorithm::Override)
838 {
839 //when doing an override merge we need to just get the full value since the new layer will also be
840 //an override layer
841 if (Sections[1]->GetRange().Contains(Frame))
842 {
843 TArray<CurveValueType> ChannelValues = MovieSceneToolHelpers::GetChannelValues<ChannelType,
844 CurveValueType>(OverrideChannelIndex, OverrideChannelIndex, OtherSections, AbsoluteSections, Frame);
845 if (ChannelValues.Num() == 1)
846 {
847 InOutValue = ChannelValues[0];
848 }
849 }
850 }
851 else
852 {
853 return false;
854 }

974 {
975 if (Sections[0]->HasEndFrame() && Sections[0]->GetExclusiveEndFrame() <= Frame)
976 {
977 if (Sections[0]->GetExclusiveEndFrame() != Frame)
978 {
979 Sections[0]->SetEndFrame(Frame);
980 }
981 }
982 else
983 {
984 Sections[0]->SetStartFrame(Frame);
985 }
986 }
987 const FFrameTime FrameTime(Frame);
988 int32 KeyIndex = Channel->GetData().GetIndex(Handles[FrameIndex]);
989 ChannelValueType Value = Channel->GetData().GetValues()[KeyIndex];
990 //got value with tangents and times, now we perform the operation
991 Value.Value = 0.0; //zero out the value we calculate it
992
993 MergeValue(Value.Value, Channels, Frame, SectionChannelIndex, Sections, AbsoluteSections,
994 OtherSections, OverrideChannelIndex, MergeAlgorithm);
995 KeysToSet.Add(TPair<FFrameNumber, ChannelValueType>(Frame, Value));
996 }
