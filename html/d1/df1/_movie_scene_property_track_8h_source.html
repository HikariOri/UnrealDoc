352 void Commit(const UMovieSceneSection* InSection, const UE::MovieScene::FEntityImportParams& Params, UE::MovieScene::FImportedEntity* OutImportedEntity)
353 {
354 const FGuid ObjectBindingID = Params.GetObjectBindingID();
355 const FMovieSceneEvaluationFieldEntityMetaData* MetaData = Params.EntityMetaData;
356 const FBuiltInComponentTypes* BuiltInComponents = FBuiltInComponentTypes::Get();
357 const FMovieSceneTracksComponentTypes* TracksComponents = FMovieSceneTracksComponentTypes::Get();
358
359 if (UMovieScenePropertyTrack* PropertyTrack = Cast<UMovieScenePropertyTrack>(InSection->GetOuter()))
360 {
361 if (FMovieScenePropertyTrackEntityImportHelper::IsPropertyValueID(Params))
362 {
363 const FName NotifyFunctionName = (MetaData ? MetaData->NotifyFunctionName : NAME_None);
364
365 OutImportedEntity->AddBuilder(
366 Builder
367 .Add(BuiltInComponents->PropertyBinding, PropertyTrack->GetPropertyBinding())
368 .AddConditional(BuiltInComponents->GenericObjectBinding, ObjectBindingID, ObjectBindingID.IsValid())
369 .AddConditional(TracksComponents->PropertyNotify, FPropertyNotifyComponentData{ NotifyFunctionName }, !NotifyFunctionName.IsNone()));
370 }
371 else if (FMovieScenePropertyTrackEntityImportHelper::IsEditConditionToggleID(Params))
372 {
373 // We effectively discard the builder we've been setting up, because we just
374 // need to import the edit condition toggle entity.
375 FMovieScenePropertyTrackEntityImportHelper::ImportEditConditionToggleEntity(Params, OutImportedEntity);
376 }
377 else if (ensure(OverrideInfo.Registry))
378 {
379 // This should be an overriden channel, which goes into a different entity.
380 if (ensure(OverrideInfo.RegistryProvider))
381 {
382 auto BaseBuilder = FEntityBuilder()
383 .Add(BuiltInComponents->PropertyBinding, PropertyTrack->GetPropertyBinding())
384 .AddConditional(BuiltInComponents->GenericObjectBinding, ObjectBindingID, ObjectBindingID.IsValid())
385 .AddTag(PropertyTag);
386 OutImportedEntity->AddBuilder(BaseBuilder);
387
388 FChannelOverrideProviderTraitsHandle Traits = OverrideInfo.RegistryProvider->GetChannelOverrideProviderTraits();
389 check(Traits.IsValid());
390 const FName ChannelOverrideName = Traits->GetChannelOverrideName(Params.EntityID);
391 auto* ChannelOverrideParams = OverrideInfo.ImportParams.FindByPredicate(
392 [=](const FChannelOverrideEntityImportParams& CurParams)
393 {
394 return CurParams.ChannelName == ChannelOverrideName;
395 });
396 if (ensure(ChannelOverrideParams))
397 {
398 OverrideInfo.Registry->ImportEntityImpl(*ChannelOverrideParams, Params, OutImportedEntity);
399 }
400 }
401 }
402 }
403 else
404 {
405 OutImportedEntity->AddBuilder(
406 Builder
407 .AddConditional(BuiltInComponents->GenericObjectBinding, ObjectBindingID, ObjectBindingID.IsValid()));
408 }
409 }
