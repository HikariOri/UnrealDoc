177 static TArray<FSolverReal> CalculateMapValues(const TConstArrayView<FRealSingle>& Multipliers, int32 ParticleCount, const FSolverVec2& MultiplierClampRange)
178 {
179 TArray<FSolverReal> Result;
180 if (Multipliers.Num() == ParticleCount && ParticleCount > 0)
181 {
182 // Copy multipliers with clamp
183 Result.SetNumUninitialized(ParticleCount);
184 for (int32 Index = 0; Index < ParticleCount; ++Index)
185 {
186 Result[Index] = FMath::Clamp(Multipliers[Index], MultiplierClampRange[0], MultiplierClampRange[1]);
187 }
188 }
189 return Result;

192 template<int32 Valence>
193 static TArray<FSolverReal> CalculateMapValues(
194 const TConstArrayView<FRealSingle>& Multipliers,
195 const TConstArrayView<TVector<int32, Valence>>& Constraints,
196 int32 ParticleOffset,
197 int32 ParticleCount)
198 {
199 TArray<FSolverReal> Result;
200 const int32 ConstraintCount = Constraints.Num();
201 if (Multipliers.Num() == ParticleCount && ParticleCount > 0 && ConstraintCount > 0)
202 {
203 // Average vertex multipliers to constraints.
204 Result.SetNumUninitialized(ConstraintCount);
205
206 for (int32 ConstraintIndex = 0; ConstraintIndex < ConstraintCount; ++ConstraintIndex)
207 {
208 const TVector<int32, Valence>& Constraint = Constraints[ConstraintIndex];
209
210 FSolverReal Weight = 0.f;
211 for (int32 Index = 0; Index < Valence; ++Index)
212 {
213 Weight += FMath::Clamp((FSolverReal)Multipliers[Constraint[Index] - ParticleOffset], (FSolverReal)0.f, (FSolverReal)1.f);
214 }
215 Weight /= (FSolverReal)Valence;
216
217 Result[ConstraintIndex] = Weight;
218 }
219 }
220 return Result;
221 }

257inline void FPBDFlatWeightMap::ReorderIndicesAndShrink(const TArray<int32>& OrigToReorderedConstraintIndices, int32 NewNum)
258{
259 if (MapValues.Num() == OrigToReorderedConstraintIndices.Num())
260 {
261 TArray<FSolverReal> ReorderedValues;
262 ReorderedValues.SetNumUninitialized(NewNum);
263 for (int32 OrigConstraintIndex = 0; OrigConstraintIndex < OrigToReorderedConstraintIndices.Num(); ++OrigConstraintIndex)
264 {
265 const int32 ReorderedConstraintIndex = OrigToReorderedConstraintIndices[OrigConstraintIndex];
266 if (ReorderedConstraintIndex != INDEX_NONE)
267 {
268 ReorderedValues[ReorderedConstraintIndex] = MapValues[OrigConstraintIndex];
269 }
270 }
271 MapValues = MoveTemp(ReorderedValues);
272 MapView.SetMapValues(TConstArrayView<FSolverReal>(MapValues));
273 }
274 else
275 {
276 check(MapValues.Num() == 0);
277 }
