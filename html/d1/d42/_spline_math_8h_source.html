38 inline uint32 ToOrdered(float f)
39 {
40 const uint32 u = FloatToBits(f);
41 const bool neg = (u & 0x80000000u) != 0;
42 return neg
43 ? ~u
44 : (u ^ 0x80000000u);

48 {
49 const uint32 u = (ordered & 0x80000000u)
50 ? (ordered ^ 0x80000000u)

58 inline float NextafterNoSubnormal(float from, bool bStepRight)
59 {
60 if (FMath::IsNaN(from) || !FMath::IsFinite(from))
61 {
62 return from; // keep NaN/Inf unchanged
63 }
64 if (from == 0.0f)
65 {
66 const float minNorm = std::numeric_limits<float>::min();
67 return bStepRight
68 ? +minNorm
69 : -minNorm;
70 }
71
72 uint32 ordered = ToOrdered(from);
73 if (bStepRight)
74 {
75 ++ordered;
76 }
77 else
78 {
79 --ordered;
80 }
81 float y = FromOrdered(ordered);
82
83 const float minNorm = std::numeric_limits<float>::min();
84 const float ay = FMath::Abs(y);
85 if (ay > 0.0f && ay < minNorm)
86 {
87 // Preserve the computed result's sign to maintain monotonicity
88 const float snapped = std::copysign(minNorm, y);
89
90 if (snapped == from)
91 {
92 // We're at ±FLT_MIN stepping toward zero; make monotone progress to signed zero
93 y = std::copysign(0.0f, y);
94 }
95 else
96 {
97 y = snapped;
98 }
99 }
100 if (!FMath::IsFinite(y))
101 {
102 y = bStepRight
103 ? +std::numeric_limits<float>::max()
104 : -std::numeric_limits<float>::max();
105 }
106 return y;
107 }

110 inline float NormalizeZeroForKey(float v)
111 {
112 return (v == 0.0f)
113 ? 0.0f
114 : v;
115 }

348 // Bezier basis computation
349 template <int32 Order>
350 static void ComputeBezierBasis(float t, TArray<float>& OutBasis)
351 {
352 OutBasis.SetNumZeroed(4); // Cubic Bezier
353
354 const float mt = 1.0f - t;
355
356 if constexpr (Order == 0)
357 {
358 // Position basis
359 OutBasis[0] = mt * mt * mt;
360 OutBasis[1] = 3.0f * mt * mt * t;
361 OutBasis[2] = 3.0f * mt * t * t;
362 OutBasis[3] = t * t * t;
363 }
364 else if constexpr (Order == 1)
365 {
366 // First derivative basis
367 OutBasis[0] = -3.0f * mt * mt;
368 OutBasis[1] = 3.0f * mt * (1.0f - 3.0f * t);
369 OutBasis[2] = 3.0f * t * (2.0f - 3.0f * t);
370 OutBasis[3] = 3.0f * t * t;
371 }
372 else if constexpr (Order == 2)
373 {
374 // Second derivative basis
375 OutBasis[0] = 6.0f * mt;
376 OutBasis[1] = -12.0f + 18.0f * t;
377 OutBasis[2] = 6.0f - 18.0f * t;
378 OutBasis[3] = 6.0f * t;
379 }
380 else if constexpr (Order == 3)
381 {
382 // Third derivative basis (constant)
383 OutBasis[0] = -6.0f;
384 OutBasis[1] = 18.0f;
385 OutBasis[2] = -18.0f;
386 OutBasis[3] = 6.0f;
387 }
388 }

390 // compute basis functions for a given degree
391 static void ComputeBSplineBasisFunctions(
392 const TArray<float>& KnotVector,
393 int32 Span,
394 float Parameter,
395 int32 Degree,
396 TArray<float>& OutBasis)
397 {
398 OutBasis.SetNum(Degree + 1);
399 TArray<float> Left;
400 TArray<float> Right;
401 Left.SetNum(Degree + 1);
402 Right.SetNum(Degree + 1);
403
404 OutBasis[0] = 1.0f;
405
406 for (int32 j = 1; j <= Degree; ++j)
407 {
408 const int32 LeftIndex = Span + 1 - j;
409 const int32 RightIndex = Span + j;
410
411 if (LeftIndex < 0 || LeftIndex >= KnotVector.Num() ||
412 RightIndex < 0 || RightIndex >= KnotVector.Num())
413 {
414 UE_LOGF(LogTemp, Warning, "Invalid knot indices in ComputeBasisFunctions: LeftIndex=%d, RightIndex=%d",
415 LeftIndex, RightIndex);
416 // Instead of continuing, we should zero out the basis and return.
417 OutBasis.SetNumZeroed(Degree + 1);
418 return;
419 }
420
421 Left[j] = Parameter - KnotVector[LeftIndex];
422 Right[j] = KnotVector[RightIndex] - Parameter;
423
424 float Saved = 0.0f;
425
426 for (int32 r = 0; r < j; ++r)
427 {
428 float Temp;
429 const float Denominator = Right[r + 1] + Left[j - r];
430 // Correctly handle zero denominator
431 if (FMath::IsNearlyZero(Denominator))
432 {
433 Temp = 0.0f;
434 }
435 else
436 {
437 Temp = OutBasis[r] / Denominator;
438 }
439
440 OutBasis[r] = Saved + Right[r + 1] * Temp;
441 Saved = Left[j - r] * Temp;
442 }
443
444 OutBasis[j] = Saved;
445 }
446 }

450 // Helper struct for shared derivative implementation
451 template <typename T, int32 Order>
452 struct TGenericDerivativeHelper
453 {
454 static T Compute(TArrayView<const T* const> Window, float Parameter)
455 {
456 // Basic validation
457 if (!FSplineValidation::IsValidWindow<T, Order+1>(Window))
458 {
459 return FSplineValidation::GetDefaultValue<T>();
460 }
461
462 if (!FSplineValidation::IsValidParameter(Parameter))
463 {
464#if ENABLE_NAN_DIAGNOSTIC
465 logOrEnsureNanError(TEXT("Invalid parameter in derivative calculation"));
466#endif
467 return FSplineValidation::GetDefaultValue<T>();
468 }
469
470 // Get pre-computed binomial coefficients
471 static const TArray<TArray<double>>& BinomialTable = GetBinomialTable<20>();
472 static const TArray<double>& FactorialTable = GetFactorialTable<20>();
473
474 const int32 WindowSize = Window.Num();
475 if (Order >= WindowSize || Order >= BinomialTable.Num())
476 {
477 return FSplineValidation::GetDefaultValue<T>();
478 }
479
480 // Optional geometry validation for higher derivatives
481 if (Order > 1 && !FSplineValidation::IsValidGeometry(Window))
482 {
483 return FSplineValidation::GetDefaultValue<T>();
484 }
485
486 T Result = FSplineValidation::GetDefaultValue<T>();
487 const float t = Parameter;
488 const float mt = 1.0f - t;
489
490 // Compute derivative using binomial expansion
491 for (int32 i = 0; i <= WindowSize - Order - 1; ++i)
492 {
493 double Coeff = BinomialTable[WindowSize - 1][i];
494
495 // Protect against overflow
496 if (Coeff > DBL_MAX / (WindowSize - 1))
497 {
498 return FSplineValidation::GetDefaultValue<T>();
499 }
500
501 // Apply derivative scaling
502 Coeff *= FactorialTable[WindowSize - 1] / FactorialTable[WindowSize - Order - 1];
503
504 // Apply parameter terms with underflow protection
505 if (mt > 0.0f)
506 {
507 const float PowMT = FMath::Pow(mt, WindowSize - Order - 1 - i);
508 if (FPlatformMath::IsFinite(PowMT))
509 {
510 Coeff *= PowMT;
511 }
512 }
513 if (t > 0.0f)
514 {
515 const float PowT = FMath::Pow(t, i);
516 if (FPlatformMath::IsFinite(PowT))
517 {
518 Coeff *= PowT;
519 }
520 }
521
522 Result += *Window[i] * Coeff;
523 }
524 return Result;
525 }
526 };

528 // Computes the nth derivative (where n == DerivOrder) of a BSpline curve.
529 // Note: This implementation assumes that DerivOrder <= Degree.
530 template <typename T, int32 DerivOrder>
531 struct TBSplineDerivativeCalculator
532 {
545 static T Compute(
546 TArrayView<const T* const> Window,
547 const TArray<float>& Knots,
548 int32 Span,
549 float Parameter,
550 int32 Degree)
551 {
552 // Basic validation
553 if (Window.Num() < Degree + 1 || DerivOrder > Degree)
554 {
555 return T();
556 }
557
558 TArray<T> DerivControlPoints;
559 DerivControlPoints.SetNum(Degree);
560 const int32 KnotStart = Span - Degree;
561
562 // Determine if knots are uniformly spaced
563 bool bIsUniform = true;
564 const float h = Knots[1] - Knots[0];
565 for (int32 i = 2; i < Knots.Num(); ++i)
566 {
567 if (!FMath::IsNearlyEqual(Knots[i] - Knots[i - 1], h, UE_KINDA_SMALL_NUMBER))
568 {
569 bIsUniform = false;
570 break;
571 }
572 }
573
574 // Compute derivative control points
575 for (int32 i = 0; i < Degree; ++i)
576 {
577 const int32 KnotDenom = KnotStart + i + Degree - DerivOrder + 2; //Knot index for the denominator
578 const int32 KnotNum = KnotStart + i + 1; //Knot index in the numerator
579
580 if (KnotDenom >= Knots.Num() || KnotNum >= Knots.Num() ||
581 KnotDenom < 0 || KnotNum < 0)
582 {
583 return T();
584 }
585
586 const float Denom = Knots[KnotDenom] - Knots[KnotNum];
587
588 // Handle uniform knots differently
589 if (bIsUniform)
590 {
591 // For uniform knots, use the fixed interval h instead of Denom
592 DerivControlPoints[i] = ((*Window[i + 1] - *Window[i])) * (static_cast<float>(Degree) / h);
593 //Correct scaling for uniform knots
594 }
595 else
596 {
597 DerivControlPoints[i] = ((*Window[i + 1] - *Window[i])) * (static_cast<float>(Degree) / Denom);
598 //Correct scaling for non-uniform knots
599 }
600 }
601
602 // Adjust scale factor for uniform knots
603 //Removing the scale factor
604 float ScaleFactor = 1.0;
605
606 // Recurse for higher derivatives with adjusted knots and parameters
607 if constexpr (DerivOrder > 1)
608 {
609 //No change here
610 TArray<const T*> DerivWindow;
611 DerivWindow.SetNum(DerivControlPoints.Num());
612 for (int32 i = 0; i < DerivControlPoints.Num(); ++i)
613 {
614 DerivWindow[i] = &DerivControlPoints[i];
615 }
616
617 // Trim the knot vector for uniform case
618 TArray<float> TrimmedKnots;
619 if (bIsUniform)
620 {
621 TrimmedKnots.Reserve(Knots.Num() - 2);
622 for (int32 i = 1; i < Knots.Num() - 1; ++i)
623 {
624 TrimmedKnots.Add(Knots[i]);
625 }
626 }
627 else
628 {
629 TrimmedKnots = Knots;
630 }
631
632 return TBSplineDerivativeCalculator<T, DerivOrder - 1>::Compute(
633 DerivWindow, //Reduced degree control points
634 TrimmedKnots, //Remove the first and the last knot as the degree is now p-1
635 Span, // Keep Span the same for uniform knots
636 Parameter, //Same parameter
637 Degree - 1 //Reduce the degree
638 ) * ScaleFactor; //Multiply with scale factor
639 }
640 else
641 {
642 // Base case: First derivative
643 TArray<float> Basis;
644 Math::ComputeBSplineBasisFunctions(Knots, Span, Parameter, Degree - 1, Basis);
645
646 T Result = T();
647 for (int32 i = 0; i < Degree; ++i)
648 {
649 Result += DerivControlPoints[i] * Basis[i];
650 }
651 return Result * ScaleFactor;
652 }
653 }
654 };

699 template <typename T, int32 Order>
700 struct TBezierDerivativeCalculator
701 {
702 static T Compute(TArrayView<const T* const> Window, float Parameter, float SegmentScale)
703 {
704 if constexpr (Order > 3)
705 {
706 return FSplineValidation::GetDefaultValue<T>();
707 }
708
709 if (!FSplineValidation::IsValidWindow<T, 4>(Window) || // Cubic Bezier needs 4 points
710 !FSplineValidation::IsValidParameter(Parameter))
711 {
712 return FSplineValidation::GetDefaultValue<T>();
713 }
714
715 // Use existing basis computation
716 TArray<float> Basis;
717 ComputeBezierBasis<Order>(Parameter, Basis);
718
719 // Apply basis to control points
720 T Result = FSplineValidation::GetDefaultValue<T>();
721 for (int32 i = 0; i < Window.Num(); ++i)
722 {
723 Result += *Window[i] * Basis[i];
724 }
725
726 // Apply segment scaling based on derivative order
727 // For derivatives, we need to divide by SegmentScale^Order because
728 // derivatives scale inversely with parameter range
729 if constexpr (Order > 0)
730 {
731 // If L == 0, the segment is degenerate in parameter space
732 // return the zero derivative (well-defined limit)
733 if (SegmentScale == 0.0f)
734 {
735 return FSplineValidation::GetDefaultValue<T>();
736 }
737
738 // Calculate scaling factor: 1 / SegmentScale^Order
739 float ScalingFactor = FMath::Pow(1.0f / SegmentScale, static_cast<float>(Order));
740 return Result * ScalingFactor;
741 }
742 else
743 {
744 return Result; // No scaling needed for position (Order = 0)
745 }
746 }
747 };

759 template <typename ValueType>
760 float FindNearestPoint_Cubic(
761 const TArrayView<ValueType> Coeffs,
762 float StartParam,
763 float EndParam,
764 float& OutSquaredDistance)
765 {
766 // Test endpoints first
767 const ValueType& P0 = Coeffs[0]; // Already relative to target
768 const ValueType At1 = P0 + Coeffs[1] + Coeffs[2] + Coeffs[3]; // Curve at t=1
769
770 float BestDistSq = static_cast<float>(Math::SizeSquared(P0));
771 float BestParam = StartParam;
772
773 const float EndDistSq = static_cast<float>(Math::SizeSquared(At1));
774 if (EndDistSq < BestDistSq)
775 {
776 BestDistSq = EndDistSq;
777 BestParam = EndParam;
778 }
779
780 // Find roots of (C(t) - P).Dot(C'(t))
781 const ValueType& P1 = Coeffs[1];
782 const ValueType& P2 = Coeffs[2];
783 const ValueType& P3 = Coeffs[3];
784
785 TStaticArray<double, 6> RootCoeffs;
786 RootCoeffs[5] = 3.0 * Dot(P3, P3);
787 RootCoeffs[4] = 5.0 * Dot(P3, P2);
788 RootCoeffs[3] = 4.0 * Dot(P3, P1) + 2.0 * Dot(P2, P2);
789 RootCoeffs[2] = 3.0 * Dot(P2, P1) + 3.0 * Dot(P3, P0);
790 RootCoeffs[1] = Dot(P1, P1) + 2.0 * Dot(P2, P0);
791 RootCoeffs[0] = Dot(P1, P0);
792
793 // Find roots in [0,1]
794 UE::Math::TPolynomialRootSolver<double, 5> RootSolver(RootCoeffs, 0, 1);
795
796 // Check each root
797 for (int32 RootIdx = 0; RootIdx < RootSolver.Roots.Num(); ++RootIdx)
798 {
799 const float t = static_cast<float>(RootSolver.Roots[RootIdx]);
800 const float Param = FMath::Lerp(StartParam, EndParam, t);
801
802 const ValueType Point = P0 +
803 P1 * t +
804 P2 * (t * t) +
805 P3 * (t * t * t);
806
807 const float DistSq = static_cast<float>(Math::SizeSquared(Point));
808 if (DistSq < BestDistSq)
809 {
810 BestDistSq = DistSq;
811 BestParam = Param;
812 }
813 }
814
815 OutSquaredDistance = BestDistSq;
816 return BestParam;
817 }

819 template <typename ValueType>
820 ValueType ComputeFirstDerivativeCentralDifference(const TSplineInterface<ValueType>& Spline, double Param,
821 double Step = UE_KINDA_SMALL_NUMBER)
822 {
823 FInterval1f ParameterSpace = Spline.GetParameterSpace();
824 Step = FMath::Max(Step, UE_KINDA_SMALL_NUMBER);
825
826 const double ParamRight = FMath::Clamp(Param + Step, ParameterSpace.Min, ParameterSpace.Max);
827 const double ParamLeft = FMath::Clamp(Param - Step, ParameterSpace.Min, ParameterSpace.Max);
828
829 const ValueType ValueRight = Spline.Evaluate(ParamRight);
830 const ValueType ValueLeft = Spline.Evaluate(ParamLeft);
831
832 return (ValueRight - ValueLeft) / (2.0 * Step);

835 template <typename ValueType>
836 ValueType ComputeSecondDerivativeCentralDifference(const TSplineInterface<ValueType>& Spline, double Param, double Step = 0.1)
837 {
838 FInterval1f ParameterSpace = Spline.GetParameterSpace();
839 Step = FMath::Max(Step, UE_KINDA_SMALL_NUMBER);
840
841 const double ParamRight = FMath::Clamp(Param + Step, ParameterSpace.Min, ParameterSpace.Max);
842 Param = FMath::Clamp(Param, ParameterSpace.Min, ParameterSpace.Max);
843 const double ParamLeft = FMath::Clamp(Param - Step, ParameterSpace.Min, ParameterSpace.Max);
844
845 const ValueType ValueRight = Spline.Evaluate(ParamRight);
846 const ValueType Value = Spline.Evaluate(Param);
847 const ValueType ValueLeft = Spline.Evaluate(ParamLeft);
848
849 return (ValueRight - 2.0 * Value + ValueLeft) / (Step * Step);
