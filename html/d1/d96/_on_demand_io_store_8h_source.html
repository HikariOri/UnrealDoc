165FOnDemandChunkEntryReferences& FOnDemandContainer::FindOrAddChunkEntryReferences(const FOnDemandInternalContentHandle& ContentHandle)
166{
167 const UPTRINT ContentHandleId = ContentHandle.HandleId();
168 check(ContentHandleId != 0);
169
170 for (FOnDemandChunkEntryReferences& Refs : ChunkEntryReferences)
171 {
172 if (Refs.ContentHandleId == ContentHandleId)
173 {
174 return Refs;
175 }
176 }
177
178 FOnDemandChunkEntryReferences& NewRef = ChunkEntryReferences.AddDefaulted_GetRef();
179 NewRef.ContentHandleId = ContentHandleId;
180 NewRef.Indices.SetNum(ChunkEntries.Num(), false);
181 return NewRef;
182}
