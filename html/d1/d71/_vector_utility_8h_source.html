212 {

231 FORCEINLINE VectorRegister4Float VectorCrossNoFMA(const VectorRegister4Float& Vec1, const VectorRegister4Float& Vec2)
232 {
233#if PLATFORM_ENABLE_VECTORINTRINSICS
234 // YZX
235 VectorRegister4Float A = VectorSwizzle(Vec2, 1, 2, 0, 3);
236 VectorRegister4Float B = VectorSwizzle(Vec1, 1, 2, 0, 3);
237 // XY, YZ, ZX
238 // This is the only way found to avoid the compiler on XSX using FMA.
239 // By forcing two FMA in a row, the subtract cannot be a FMA at the end.
240 // This allow to have a symmetric and reliable cross product.
241 A = VectorMultiplyAdd(A, Vec1, VectorZero());
242 B = VectorMultiplyAdd(B, Vec2, VectorZero());
243 // XY-YX, YZ-ZY, ZX-XZ
244 A = VectorSubtract(A, B);
245 // YZ-ZY, ZX-XZ, XY-YX
246 return VectorSwizzle(A, 1, 2, 0, 3);
247#else
248 return VectorCross(Vec1, Vec2);
249#endif
250 }

260 FORCEINLINE VectorRegister4Double VectorCrossNoFMA(const VectorRegister4Double& Vec1, const VectorRegister4Double& Vec2)
261 {
262#if PLATFORM_ENABLE_VECTORINTRINSICS_NEON
263 VectorRegister4Double C = VectorMultiply(Vec1, VectorSwizzle(Vec2, 1, 2, 0, 3));
264 C = VectorSubtract(C, VectorMultiply(VectorSwizzle(Vec1, 1, 2, 0, 3), Vec2));
265 return VectorSwizzle(C, 1, 2, 0, 3);
266#elif PLATFORM_ENABLE_VECTORINTRINSICS
267 // YZX
268 VectorRegister4Double A = VectorSwizzle(Vec2, 1, 2, 0, 3);
269 VectorRegister4Double B = VectorSwizzle(Vec1, 1, 2, 0, 3);
270 // XY, YZ, ZX
271 A = VectorMultiply(A, Vec1);
272 // XY-YX, YZ-ZY, ZX-XZ
273 A = VectorSubtract(A, VectorMultiply(B, Vec2));
274 // YZ-ZY, ZX-XZ, XY-YX
275 return VectorSwizzle(A, 1, 2, 0, 3);
276#else
277 return VectorCross(Vec1, Vec2);
278#endif
279 }

305{
306 VectorRegister4Double Result;

316#endif
317#else
318 Result = MakeVectorRegisterDouble(
319 uint64(~((uint64*)(A.V))[0] & ((uint64*)(B.V))[0]),
320 uint64(~((uint64*)(A.V))[1] & ((uint64*)(B.V))[1]),
321 uint64(~((uint64*)(A.V))[2] & ((uint64*)(B.V))[2]),
