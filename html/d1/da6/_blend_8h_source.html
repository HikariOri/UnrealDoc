36 // exponent used in blend; controls shape (larger number == sharper blend)
37 double BlendPower = 2.0;

39 // blend distance
40 double BlendFalloff = 5.0;

42 // size of the cells used when sampling the distance field
43 double GridCellSize = 1.0;
44
45 // size of the cells used when meshing the output (marching cubes' cube size)
46 double MeshCellSize = 1.0;
47
48 // if true, perform a smooth subtraction instead of a smooth union
49 bool bSubtract = false;
50
51 // Set cell sizes to hit the target voxel counts along the max dimension of the bounds
52 void SetCellSizesAndFalloff(FAxisAlignedBox3d Bounds, double BlendFalloffIn, int TargetInputVoxelCount, int TargetOutputVoxelCount)
53 {
54 BlendFalloff = BlendFalloffIn;
55 GridCellSize = (Bounds.MaxDim() + BlendFalloff * 2.0) / double(TargetInputVoxelCount);
56 MeshCellSize = (Bounds.MaxDim() + BlendFalloff * 2.0) / double(TargetOutputVoxelCount);
57 }

78 bool Validate()
79 {
80 bool bHasSourcesWithBounds = Sources.Num() > 0 && Sources.Num() == SourceBounds.Num();
81 for (int SourceIdx = 0; SourceIdx < Sources.Num(); SourceIdx++)
82 {
83 if (Sources[SourceIdx] == nullptr)
84 {
85 return false;
86 }
87 }
88
89 bool bValidParams = BlendPower > 0 && BlendFalloff > 0 && GridCellSize > 0 && MeshCellSize > 0 && FMath::IsFinite(GridCellSize) && FMath::IsFinite(MeshCellSize);
90 return bHasSourcesWithBounds && bValidParams;
91 }

96 const FMeshShapeGenerator& Generate(bool bReuseComputed = false)
97 {
98 MarchingCubes.Reset();
99 if (!ensure(Validate()))
100 {
101 // give up and return and empty result on invalid parameters
102 return MarchingCubes;
103 }
104
105 ComputeBounds();
106 GenerateBlendAnalytic(bReuseComputed);
107 return MarchingCubes;
108 }

110protected:
111
112 void ComputeBounds()
113 {
114 CombinedBounds = FAxisAlignedBox3d::Empty();
115 for (FAxisAlignedBox3d Box : SourceBounds)
116 {
117 CombinedBounds.Contain(Box);
118 }
119 }
120
121 void ComputeSpatials(bool bReuseComputed)
122 {
123 ComputedSpatials.SetNum(Sources.Num());
124 ParallelFor(Sources.Num(), [this, bReuseComputed](int SourceIdx)
125 {
126 if (!bReuseComputed || ComputedSpatials[SourceIdx].GetMesh() != Sources[SourceIdx] || !ComputedSpatials[SourceIdx].IsValid(false))
127 {
128 ComputedSpatials[SourceIdx].SetMesh(Sources[SourceIdx], true);
129 }
130 }
131 );
132 }
133
134 void ComputeLazySDFs(bool bReuseComputed)
135 {
136 ComputeSpatials(bReuseComputed);
137
138 if (!bReuseComputed || ComputedSDFs.Num() != Sources.Num())
139 {
140 ComputedSDFs.Reserve(Sources.Num());
141 SDFMaxDistances.Reserve(Sources.Num());
142 for (int i = 0; i < Sources.Num(); i++)
143 {
144 ComputedSDFs.Emplace(Sources[i], GridCellSize, &ComputedSpatials[i], false); // not auto-building here, to share code w/ need-rebuild path below
145 SDFMaxDistances.Add(0.0);
146 }
147 }
148
149 double NeedDistance = BlendFalloff;
150 ParallelFor(Sources.Num(), [this, bReuseComputed, NeedDistance](int SourceIdx)
151 {
152 // previously computed sdf has signs computed out to (at least) the required distance, no need to recompute
153 if (bReuseComputed && NeedDistance <= SDFMaxDistances[SourceIdx])
154 {
155 return;
156 }
157
158 // TODO: if we do have a previously computed sdf, and want to reuse computed, but need more distance
159 // we could expand the sdf here instead of throwing it out and fully recomputing.
160
161 float UseMaxOffset = (float)BlendFalloff;
162
163 ComputedSDFs[SourceIdx].MaxOffsetDistance = UseMaxOffset;
164 ComputedSDFs[SourceIdx].CellSize = GridCellSize;
165
166 ComputedSDFs[SourceIdx].CancelF = CancelF;
167 ComputedSDFs[SourceIdx].Initialize();
168
169 SDFMaxDistances[SourceIdx] = UseMaxOffset;
170 });
171 }
172
173 void GenerateBlendAnalytic(bool bReuseComputed)
174 {
175 if (CancelF())
176 {
177 return;
178 }
179
180 ComputeSpatials(bReuseComputed);
181
182 if (CancelF())
183 {
184 return;
185 }
186
187 ComputeLazySDFs(bReuseComputed);
188
189 if (CancelF())
190 {
191 return;
192 }
193
194 TSkeletalRicciNaryBlend3<TDistanceFieldToSkeletalField<TTriLinearGridInterpolant<TCachingMeshSDF<TriangleMeshType>>, double>, double> Blend;
195 Blend.BlendPower = BlendPower;
196 Blend.bSubtract = bSubtract;
197 Blend.Children.Reserve(Sources.Num());
198 TArray<TTriLinearGridInterpolant<TCachingMeshSDF<TriangleMeshType>>> Interpolants; Interpolants.Reserve(Sources.Num());
199 TArray<TDistanceFieldToSkeletalField<TTriLinearGridInterpolant<TCachingMeshSDF<TriangleMeshType>>, double>> SkeletalFields; SkeletalFields.Reserve(Sources.Num());
200 for (int SourceIdx = 0; SourceIdx < Sources.Num(); SourceIdx++)
201 {
202 Interpolants.Add(ComputedSDFs[SourceIdx].MakeInterpolant());
203 SkeletalFields.Emplace(&Interpolants[SourceIdx], BlendFalloff);
204 Blend.Children.Add(&SkeletalFields[SourceIdx]);
205 }
206
207 MarchingCubes.CancelF = CancelF;
208
209 MarchingCubes.CubeSize = MeshCellSize;
210
211 MarchingCubes.IsoValue = TDistanceFieldToSkeletalField<TBoundedImplicitFunction3<double>, double>::ZeroIsocontour;
212 MarchingCubes.Bounds = CombinedBounds;
213 MarchingCubes.Bounds.Expand(BlendFalloff);
214 MarchingCubes.RootMode = ERootfindingModes::LerpSteps;
215 MarchingCubes.RootModeSteps = 3;
216
217 if (CancelF())
218 {
219 return;
220 }
221
222 TArray<FVector3d> Seeds;
223 for (const TriangleMeshType* Source : Sources)
224 {
225 for (int VID = 0; VID < Source->MaxVertexID(); VID++)
226 {
227 if (!Source->IsVertex(VID))
228 {
229 continue;
230 }
231
232 FVector3d Seed = Source->GetVertex(VID);
233 // Only add vertices that are inside the spatial bounds (only vertices that are not on any triangles will be outside)
234 if (MarchingCubes.Bounds.Contains(Seed))
235 {
236 Seeds.Add(Seed);
237 }
238 }
239 }
240
241 MarchingCubes.Implicit = [&Blend](const FVector3d& Pt) {return Blend.Value(Pt);}; //-V1047 - This lambda is cleared before routine exit
242
243 MarchingCubes.GenerateContinuation(Seeds);
244
245 if (Seeds.Num() > 0 && MarchingCubes.Triangles.Num() == 0)
246 {
247 // fall back to full generation; seeds failed!
248 MarchingCubes.Generate();
249 }
250
251 // TODO: refactor FMarchingCubes to not retain the implicit function, or refactor this function so the implicit function isn't invalid after returning,
253 MarchingCubes.Implicit = nullptr;
254 }
255};
