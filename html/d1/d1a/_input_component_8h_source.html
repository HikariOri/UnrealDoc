15template<class DelegateType, class DynamicDelegateType>
16struct TInputUnifiedDelegate
17{
18 TInputUnifiedDelegate() {};
19 TInputUnifiedDelegate(DelegateType D) : FuncDelegate(MakeShared<DelegateType>(MoveTemp(D))) {};
20 TInputUnifiedDelegate(DynamicDelegateType D) : FuncDynDelegate(MakeShared<DynamicDelegateType>(MoveTemp(D))) {};
21
23 inline bool IsBound() const
24 {
25 return ((FuncDelegate.IsValid() && FuncDelegate->IsBound()) || (FuncDynDelegate.IsValid() && FuncDynDelegate->IsBound()));
26 }
27
29 inline UObject* GetUObject() const
30 {
31 if (FuncDelegate.IsValid())
32 {
33 if (FuncDelegate->IsBound())
34 {
35 return FuncDelegate->GetUObject();
36 }
37 }
38 else if (FuncDynDelegate.IsValid())
39 {
40 if (FuncDynDelegate->IsBound())
41 {
42 return FuncDynDelegate->GetUObject();
43 }
44 }
45
46 return nullptr;
47 }
48
50 inline bool IsBoundToObject(void const* Object) const
51 {
52 if (FuncDelegate.IsValid())
53 {
54 if (FuncDelegate->IsBound())
55 {
56 return FuncDelegate->IsBoundToObject(Object);
57 }
58 }
59 else if (FuncDynDelegate.IsValid())
60 {
61 if (FuncDynDelegate->IsBound())
62 {
63 return FuncDynDelegate->IsBoundToObject(Object);
64 }
65 }
66
67 return false;
68 }

71 template< class UserClass >
72 inline void BindDelegate(UserClass* Object, typename DelegateType::template TMethodPtr< UserClass > Func)
73 {
74 FuncDynDelegate.Reset();;
75 FuncDelegate = MakeShared<DelegateType>(DelegateType::CreateUObject(Object, Func));
76 }
77
79 inline void BindDelegate(UObject* Object, const FName FuncName)
80 {
81 FuncDelegate.Reset();
82 FuncDynDelegate = MakeShared<DynamicDelegateType>();
83 FuncDynDelegate->BindUFunction(Object, FuncName);
84 }
85
87 DelegateType& GetDelegateForManualSet()
88 {
89 FuncDynDelegate.Reset();
90 FuncDelegate = MakeShared<DelegateType>();
91 return *FuncDelegate;
92 }
93
95 inline void Unbind()
96 {
97 FuncDelegate.Reset();
98 FuncDynDelegate.Reset();
99 }

102 inline const DelegateType& GetDelegate() const
103 {
104 return (FuncDelegate.IsValid() ? *FuncDelegate : UnboundDelegate);
105 }
106
108 inline const DynamicDelegateType& GetDynamicDelegate() const
109 {
110 return (FuncDynDelegate.IsValid() ? *FuncDynDelegate : UnboundDynamicDelegate);
111 }
112
113protected:
115 TSharedPtr<DelegateType> FuncDelegate;
117 TSharedPtr<DynamicDelegateType> FuncDynDelegate;
118
119 static const DelegateType UnboundDelegate;
120 static const DynamicDelegateType UnboundDynamicDelegate;
121};

282 inline void Unbind()
283 {
284 switch(BoundDelegateType)
285 {
286 case EBoundDelegate::Delegate:
287 FuncDelegate->Unbind();
288 break;
289
290 case EBoundDelegate::DelegateWithKey:
291 FuncDelegateWithKey->Unbind();
292 break;
293
294 case EBoundDelegate::DynamicDelegate:
295 FuncDynDelegate->Unbind();
296 break;
297 }
298 BoundDelegateType = EBoundDelegate::Unbound;
299 }

302 inline void Execute(const FKey Key) const
303 {
304 switch(BoundDelegateType)
305 {
306 case EBoundDelegate::Delegate:
307 if (FuncDelegate->IsBound())
308 {
309 FuncDelegate->Execute();
310 }
311 break;
312
313 case EBoundDelegate::DelegateWithKey:
314 if (FuncDelegateWithKey->IsBound())
315 {
316 FuncDelegateWithKey->Execute(Key);
317 }
318 break;
319
320 case EBoundDelegate::DynamicDelegate:
321 if (FuncDynDelegate->IsBound())
322 {
323 FuncDynDelegate->Execute(Key);
324 }
325 break;
326 }
327 }
