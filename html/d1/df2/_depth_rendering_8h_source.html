81 static bool ShouldCompilePermutation(const FMeshMaterialShaderPermutationParameters& Parameters)
82 {
83 // Only the local vertex factory supports the position-only stream
84 if (bUsePositionOnlyStream)
85 {
86 return Parameters.VertexFactoryType->SupportsPositionOnly() && Parameters.MaterialParameters.bIsSpecialEngineMaterial;
87 }
88

98 Parameters.MaterialParameters.bMaterialMayModifyMeshPosition)
99 && !Parameters.VertexFactoryType->SupportsNaniteRendering();

123class FDepthOnlyPS : public FMeshMaterialShader
124{
125 DECLARE_SHADER_TYPE(FDepthOnlyPS,MeshMaterial);
126public:
127 static bool ShouldCompilePermutation(const FMeshMaterialShaderPermutationParameters& Parameters)
128 {
129 if (IsTranslucentBlendMode(Parameters.MaterialParameters))
130 {
131 return Parameters.MaterialParameters.bIsTranslucencyWritingCustomDepth;
132 }
133
134 return
135 // Compile for materials that are masked
136 (!Parameters.MaterialParameters.bWritesEveryPixel || Parameters.MaterialParameters.bHasPixelDepthOffsetConnected)
137 && !Parameters.VertexFactoryType->SupportsNaniteRendering();
138 }
139
140 FDepthOnlyPS(const ShaderMetaType::CompiledShaderInitializerType& Initializer):
141 FMeshMaterialShader(Initializer)
142 {
143 }
144
145 static void ModifyCompilationEnvironment(const FMaterialShaderPermutationParameters& Parameters, FShaderCompilerEnvironment& OutEnvironment)
146 {
147 FMeshMaterialShader::ModifyCompilationEnvironment(Parameters, OutEnvironment);
148
149 OutEnvironment.SetDefine(TEXT("ALLOW_DEBUG_VIEW_MODES"), AllowDebugViewmodes(Parameters.Platform));
150
151 // Regular opaque materials do not support scene textures, but SLW does support reading custom depth/stencil,
152 // so we need to make an exception here for the water depth prepass.
153 const bool bIsSingleLayerWater = Parameters.MaterialParameters.ShadingModels.HasShadingModel(MSM_SingleLayerWater);
154 OutEnvironment.SetDefine(TEXT("SCENE_TEXTURES_DISABLED"), bIsSingleLayerWater ? 0u : 1u);
155 }

157 FDepthOnlyPS() {}
