643template<typename T>
644inline UE::Math::TVector<T> FMath::RayPlaneIntersection(const UE::Math::TVector<T>& RayOrigin, const UE::Math::TVector<T>& RayDirection, const UE::Math::TPlane<T>& Plane)
645{
646 using TVector = UE::Math::TVector<T>;
647 const TVector PlaneNormal = TVector(Plane.X, Plane.Y, Plane.Z);
648 const TVector PlaneOrigin = PlaneNormal * Plane.W;
649
650 const T Distance = TVector::DotProduct((PlaneOrigin - RayOrigin), PlaneNormal) / TVector::DotProduct(RayDirection, PlaneNormal);
651 return RayOrigin + RayDirection * Distance;
652}

678template<typename T>
679inline bool FMath::IntersectPlanes3(UE::Math::TVector<T>& I, const UE::Math::TPlane<T>& P1, const UE::Math::TPlane<T>& P2, const UE::Math::TPlane<T>& P3)
680{
681 // Compute determinant, the triple product P1|(P2^P3)==(P1^P2)|P3.
682 const T Det = (P1 ^ P2) | P3;
683 if (Square(Det) < Square(0.001f))
684 {
685 // Degenerate.
686 I = UE::Math::TVector<T>::ZeroVector;
687 return 0;
688 }
689 else
690 {
691 // Compute the intersection point, guaranteed valid if determinant is nonzero.
692 I = (P1.W * (P2 ^ P3) + P2.W * (P3 ^ P1) + P3.W * (P1 ^ P2)) / Det;
693 }
694 return 1;
695}

697template<typename T>
698inline bool FMath::IntersectPlanes2(UE::Math::TVector<T>& I, UE::Math::TVector<T>& D, const UE::Math::TPlane<T>& P1, const UE::Math::TPlane<T>& P2)
699{
700 // Compute line direction, perpendicular to both plane normals.
701 D = P1 ^ P2;
702 const T DD = D.SizeSquared();
703 if (DD < Square(0.001f))
704 {
705 // Parallel or nearly parallel planes.
706 D = I = UE::Math::TVector<T>::ZeroVector;
707 return 0;
708 }
709 else
710 {
711 // Compute intersection.
712 I = (P1.W * (P2 ^ D) + P2.W * (D ^ P1)) / DD;
713 D.Normalize();
714 return 1;
715 }
