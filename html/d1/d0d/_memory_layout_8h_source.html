248 // Override for types that need access to a PointerTable in order to destroy frozen data
249 // WARNING! Frozen data may have deduplicated portions, so it's possible that a given object may have multiple references
250 // Any attempt to cleanup/destroy frozen data must take care to safely allow DestroyObject to be called multiple times on the same frozen object
251 // Generally speaking, it's *NOT* safe to invoke the built-in destructor multiple times
252 template<typename T>
253 inline void DestroyObject(T* Object, const FPointerTableBase* PtrTable, bool bIsFrozen)
254 {
255 // Only call destructor for non-frozen objects
256 if (!bIsFrozen)
257 {
258 Object->~T();
259 }
260 // Wipe destroyed memory to a common pattern
261 FMemory::Memset(Object, 0xfe, sizeof(T));
262 }
