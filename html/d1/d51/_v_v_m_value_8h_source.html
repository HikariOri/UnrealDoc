320public:
321 // VValue is a NaN-boxed 64-bit payload representing the fundamental value type in the
322 // Verse VM. NaN-boxing allows us to represent immediates like int32s and floats,
323 // and also pointers like VCell, all in a 64-bit payload. We do this by encoding
324 // non-floats in the IEEE754 NaN space. E.g, all non-double VValues will be NaN when
325 // you subtract FloatOffset from them. This means we need to be careful with the
326 // actual set of values we use to store NaN. The NaNs that are safe to be boxed
327 // as a float we call PureNaNs.
328 //
329 // On x86-64 and arm64, doing math on PureNaNs has the following properties:
330 // - When doing binary operations involving one PureNaN, the output is again a PureNaN.
331 // - The exact value you get on newly indefinite results (e.g, Inf - Inf) may vary in the sign, but is a PureNaN.
332 // This means that when doing arithmetic on VFloats, which can't be an impure NaN,
333 // we don't need to purify the result of the arithmetic operation when boxing them
334 // as a VValue. However, when boxing an unknown double value as a VValue, we need to
335 // purify any incoming NaNs.
336
337 // Encoding space by top 16 bits:
338 // 0x0000... Cell/Placeholder/UObject/Root (see below)
339 // 0x0001... \
340 // ... float
341 // 0xfffc... /
342 // 0xfffd... unused
343 // 0xfffe... unused
344 // 0xffff... int32
345
346 // If the top 16 bits are 0000, the lower 4 bits encode as follows:
347 // 0b0000... Cell
348 // 0b0001... Placeholder
349 // 0b0010... Root
350 // 0bX011... UObject
351 // 0b0100... char8
352 // 0b0101... char32
353 // 0b0110... transparent VRef
354 // 0b0111... unused
355 // 0b1XXX... unused (except for 0xb1011)
356
357 // VValue assumes a 48-bit address space for pointers.
358
359 // Special code which means this VValue is uninitialized
360 static constexpr uint64 UninitializedValue = 0;
361
362 // Number-related constants
363 static constexpr uint64 NumberTagMask = 0xffff'0000'0000'0000ull;
364 static constexpr uint64 Int32Tag = 0xffff'0000'0000'0000ull;
365 static constexpr uint64 FloatOffset = 0x0001'0000'0000'0000ull;
366 static constexpr uint64 MaxPureNaN = 0xfffb'ffff'ffff'ffffull;
367 static constexpr uint64 MaxFloatTag = 0xfffc'0000'0000'0000ull;
368
369 // Non-number related constants
370 static constexpr uint64 NumLowerEncodingBits = 4;
371 static constexpr uint64 NonCellTagMask = NumberTagMask | 0xfull; // Used for Placeholder, Root
372 static constexpr uint64 UObjectTagMask = NumberTagMask | 0x7ull; // Used for UObject
373 static constexpr uint64 PlaceholderTag = 0x1ull;
374 static constexpr uint64 RootTag = 0x2ull;
375 static constexpr uint64 UObjectTag = 0x3ull;
376 static constexpr uint64 CharTag = 0x4ull;
377 static constexpr uint64 Char32Tag = 0x5ull;
