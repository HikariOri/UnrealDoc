83 template <FTransaction::EAbortMode AbortMode = FTransaction::EAbortMode::Default>
84 void Fail(FAccessContext Context)
85 {
86 // You can't have two sibling transactions that have started. The successor child
87 // transaction can only begin after the predecessor child finishes.
88 // For example, in this Verse code:
89 // if (<ParentTransaction>):
90 // if (<Child1>): ...
91 // if (<Child2>): ...
92 // Child2 can only start after Child1 commits or aborts. Therefore, the list of active
93 // transactions forms a linked list, not a tree (even though we need to track the
94 // tree of failure contexts).
95 //
96 // And because of how transactions work, we need to Abort beginning at the innermost
97 // transaction and moving upwards towards the parent. Since we're just talking about
98 // a linked list here, we count the number of started transactions and just abort the
99 // Context's CurrentTransaction that many times.
100 //
101 // (If for some reason we found that we needed to call Abort on the entire failure context
102 // tree, and needed to maintain that children needed to be aborted before parents, we could
103 // just do a post order traversal to achieve that.)
104 uint32 NumActiveTransactions = 0;
105 TArray<FTrail*, TInlineAllocator<4>> PreOrder;
106 ForEachInPreOrder([&](VFailureContext& FailureContext) {
107 PreOrder.Push(&FailureContext.Trail);
108 if (FailureContext.Transaction.IsActive())
109 {
110 ++NumActiveTransactions;
111 }
112 FailureContext.bFailed = true;
113 });
114
115 for (uint32 I = 0; I < NumActiveTransactions; ++I)
116 {
117 Context.CurrentTransaction()->Abort<AbortMode>(Context);
118 }
119
120 // Backtrack for all children before their parent.
121 for (FTrail* T : ReverseIterate(PreOrder))
122 {
123 T->Log.Backtrack(Context);
124 T->Log.Empty();
125 }
126 }
