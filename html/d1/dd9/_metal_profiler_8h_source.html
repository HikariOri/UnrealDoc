110 inline TRefCountPtr<FMetalBreadcrumbInstance> GetBreadcrumbEvent(FRHIBreadcrumbNode* Node, bool bWithinRenderPass)
111 {
112 FScopeLock Lock(&Mutex);
113 TRefCountPtr<FMetalBreadcrumbInstance>& InstanceRef = CreatedBreadcrumbs.FindOrAdd(Node);
114 if (!InstanceRef.IsValid())
115 {
116 InstanceRef = new FMetalBreadcrumbInstance;
117 InstanceRef->Node = Node;
118 }
119 InstanceRef->bWithinRenderPass |= bWithinRenderPass;
120 return InstanceRef;
121 }

123 inline void AddSample(FMetalCounterSamplePtr Sample)

126 {
127 ActiveInstance->Samples.Add(Sample);
128 }

131 inline void OnBreadcrumbBegin(FMetalBreadcrumbInstance* InInstance)

136 inline void ResolveTopForCmdBuffer(FMetalCommandBuffer* CmdBuffer)
137 {
138 for (const TRefCountPtr<FMetalBreadcrumbInstance>& ActiveInstance : ActiveBreadcrumbs)
139 {
140 if (ActiveInstance->BeginCmdBuffer != CmdBuffer || ActiveInstance->bTopResolved)
141 {
142 continue;
143 }
144
145 if (uint64_t* Start = ActiveInstance->TimestampTOP)
146 {
147 if (ActiveInstance->bWithinRenderPass || ActiveInstance->Samples.Num() == 0)
148 {
149 *Start = 0;
150 }
151 else
152 {
153 uint64_t MinStart = 0;
154 for (const FMetalCounterSamplePtr& Sample : ActiveInstance->Samples)
155 {
156 uint64_t StartTime, EndTime;
157 Sample->ResolveStageCounters(StartTime, EndTime);
158
159 MinStart = MinStart > 0 ? FMath::Min(StartTime, MinStart) : StartTime;
160 }
161 *Start = MinStart;
162 }
163 }
164 ActiveInstance->bTopResolved = true;
165 }
166 }
167
168 inline void OnBreadcrumbEnd(FMetalBreadcrumbInstance* InInstance)
169 {
170 uint64_t* Start = InInstance->bTopResolved ? nullptr : InInstance->TimestampTOP;
171 uint64_t* End = InInstance->TimestampBOP;
172
173 if (InInstance->bWithinRenderPass || InInstance->Samples.Num() == 0)
174 {
175 if (Start)
176 {
177 *Start = 0;
178 }
179 if (End)
180 {
181 *End = 0;
182 }
183 }
184 else
185 {
186 for (const FMetalCounterSamplePtr& Sample : InInstance->Samples)
187 {
188 uint64_t StartTime, EndTime;
189 Sample->ResolveStageCounters(StartTime, EndTime);
190 if (Start)
191 {
192 *Start = *Start > 0 ? FMath::Min(StartTime, *Start) : StartTime;
193 }
194 if (End)
195 {
196 *End = *End > 0 ? FMath::Max(EndTime, *End) : EndTime;
197 }
198 }
199 }
200 InInstance->bTopResolved = true;
201
202 ActiveBreadcrumbs.Remove(InInstance);
203 CreatedBreadcrumbs.Remove(InInstance->Node);
204 }
