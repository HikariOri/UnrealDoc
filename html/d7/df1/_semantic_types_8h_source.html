293class CGlobalType : public CNormalType
294{
295public:
296 static const ETypeKind StaticTypeKind = Kind;

298 // CTypeBase interface.
299 virtual CUTF8String AsCodeRecursive(ETypeSyntaxPrecedence OuterPrecedence, TArray<const CFlowType*>& VisitedFlowTypes, bool bLinkable, ETypeStringFlag Flag) const override
300 {
301 if (Kind == ETypeKind::False) { return "false"; }
302 else if (Kind == ETypeKind::True) { return "true"; }
303 else if (Kind == ETypeKind::Void) { return "void"; }
304 else if (Kind == ETypeKind::Any) { return "any"; }
305 else if (Kind == ETypeKind::Comparable) { return "comparable"; }
306 else if (Kind == ETypeKind::Logic) { return "logic"; }
307 else if (Kind == ETypeKind::Rational) { return "rational"; }
308 else if (Kind == ETypeKind::Char8) { return "char"; }
309 else if (Kind == ETypeKind::Char32) { return "char32"; }
310 else if (Kind == ETypeKind::Path) { return "path"; }
311 else if (Kind == ETypeKind::Range) { return "$range"; }
312 else if (Kind == ETypeKind::Persistable) { return "persistable"; }
313 else { ULANG_UNREACHABLE(); }
314 }
315
316 // CNormalType interface.
317 virtual EComparability GetComparability() const override
318 {
319 if (Kind == ETypeKind::Comparable
320 || Kind == ETypeKind::Logic
321 || Kind == ETypeKind::Rational
322 || Kind == ETypeKind::Char8
323 || Kind == ETypeKind::Char32
324 || Kind == ETypeKind::False)
325 {
326 return EComparability::ComparableAndHashable;
327 }
328 else if (Kind == ETypeKind::True || Kind == ETypeKind::Void)
329 {
330 return EComparability::Comparable;
331 }
332 else
333 {
334 return EComparability::Incomparable;
335 }
336 }
337
338 virtual bool IsPersistable() const override
339 {
340 return Kind == Cases<
341 ETypeKind::Void,
342 ETypeKind::Logic,
343 ETypeKind::Char8,
344 ETypeKind::Char32,
345 ETypeKind::Persistable>;
346 }
347
348 virtual bool IsExplicitlyCastable() const override
349 {
350 return Kind == ETypeKind::False;
351 }
352
353 virtual bool IsExplicitlyConcrete() const override
354 {
355 return Kind == ETypeKind::False;
356 }
357
358 virtual bool CanBeCustomAccessorDataType() const override
359 {
360 return Kind == Cases<
361 ETypeKind::Logic,
362 ETypeKind::Rational,
363 ETypeKind::Char8,
364 ETypeKind::Char32
365 >;
366 }
367
368 virtual bool CanBePredictsVarDataType() const override
369 {
370 return Kind == Cases<
371 ETypeKind::Logic
372 >;
373 }

375private:
