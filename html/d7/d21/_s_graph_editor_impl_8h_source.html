55struct FAlignmentData
56{
57 FAlignmentData(UEdGraphNode* InNode, int32& InTargetProperty, float InTargetOffset)
58 : Node(InNode), TargetProperty(InTargetProperty), TargetOffset(InTargetOffset)
59 {}

81struct FAlignmentHelper
82{
84 FAlignmentHelper(TSharedRef<SGraphEditor> InGraphEditor, EOrientation InOrientation, EAlignType InAlignType)
85 : GraphEditor(MoveTemp(InGraphEditor))
86 {
87 // We align to the node that was clicked on, if available (not when invoked from a key shortcut)
88 CardinalNode = GraphEditor->GetGraphNodeForMenu();
89
90 Orientation = InOrientation;
91 AlignType = InAlignType;
92
93 // Collect all the alignment data for all the selected nodes
94 for (UObject* It : GraphEditor->GetSelectedNodes())
95 {
96 if (UEdGraphNode* Node = Cast<UEdGraphNode>(It))
97 {
98 AlignmentData.Add(GetAlignmentDataForNode(Node));
99 }
100 }
101
102 // Sort the data based on target - important for future algorithms
103 AlignmentData.Sort([](const FAlignmentData& A, const FAlignmentData& B) {
104 return A.GetTarget() < B.GetTarget();
105 });
106 }
107
109 void Align()
110 {
111 if (AlignmentData.Num() > 1)
112 {
113 UEdGraph* Graph = AlignmentData[0].Node->GetGraph();
114
115 if (Graph)
116 {
117 const UEdGraphSchema* Schema = Graph->GetSchema();
118
119 if (Schema)
120 {
121 float Target = DetermineAlignmentTarget();
122
123 for (FAlignmentData& Entry : AlignmentData)
124 {
125 float TargetProperty = Target - Entry.TargetOffset;
126 FVector2f TargetPosition = Entry.Node->GetPosition();
127
128 if (Orientation == EOrientation::Orient_Horizontal)
129 {
130 TargetPosition.X = TargetProperty;
131 }
132 else
133 {
134 TargetPosition.Y = TargetProperty;
135 }
136
137 Schema->SetNodePosition(Entry.Node, TargetPosition);
138 }
139 }
140 }
141 }
142 }
143
144private:
145
147 FAlignmentData GetAlignmentDataForNode(UEdGraphNode* Node);
148
150 float DetermineAlignmentTarget()
151 {
152 if (CardinalNode)
153 {
154 return GetAlignmentDataForNode(CardinalNode).GetTarget();
155 }
156
157 if (AlignType == EAlignType::Minimum)
158 {
159 float Target = TNumericLimits<float>::Max();
160 for (const FAlignmentData& Entry : AlignmentData)
161 {
162 Target = FMath::Min(Target, Entry.GetTarget());
163 }
164 return Target;
165 }
166 else if (AlignType == EAlignType::Maximum)
167 {
168 float Target = TNumericLimits<float>::Lowest();
169 for (const FAlignmentData& Entry : AlignmentData)
170 {
171 Target = FMath::Max(Target, Entry.GetTarget());
172 }
173 return Target;
174 }
175 else
176 {
177 // Use the mean
178 float SumTotal = 0.f;
179 for (const FAlignmentData& Entry : AlignmentData)
180 {
181 SumTotal += Entry.GetTarget();
182 }
183 return SumTotal / static_cast<float>(AlignmentData.Num());
184 }
185 }
186
188 TSharedRef<SGraphEditor> GraphEditor;
190 EOrientation Orientation;
192 EAlignType AlignType;
194 UEdGraphNode* CardinalNode;
196 TArray<FAlignmentData> AlignmentData;
