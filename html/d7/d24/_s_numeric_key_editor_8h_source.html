88 {
89 return InKeyEditor->GetMetaData(Key);
90 });

92 TNumericPropertyParams<NumericType> NumericPropertyParams(Property, MetaDataGetter);

104 .SliderExponent(NumericPropertyParams.SliderExponent)

107 .LinearDeltaSensitivity(NumericPropertyParams.GetLinearDeltaSensitivityAttribute())

149 void OnValueChanged(NumericType Value)
150 {
151 // Update the key editor only when spinning/sliding the value since the value hasn't been committed yet.
152 // This should only happen in this case because OnValueChanged is called aggressively (ie. on focus changing)
153 if (bSliding)
154 {
155 KeyEditor->SetValueWithNotify(Value, EMovieSceneDataChangeType::TrackValueChanged);
156 }
157 }
158
159 void OnValueCommitted(NumericType Value, ETextCommit::Type CommitInfo)
160 {
161 if (CommitInfo == ETextCommit::OnEnter || CommitInfo == ETextCommit::OnUserMovedFocus
162#if !UE_BUILD_SHIPPING
163 || CommitInfo == ETextCommit::Default
164#endif // !UE_BUILD_SHIPPING
165 )
166 {
167 const FScopedTransaction Transaction( LOCTEXT("SetNumericKey", "Set Key Value") );
168 KeyEditor->SetValueWithNotify(Value, EMovieSceneDataChangeType::TrackValueChangedRefreshImmediately);
169 }
170
171 KeyEditor->EndEditing();
172 }
