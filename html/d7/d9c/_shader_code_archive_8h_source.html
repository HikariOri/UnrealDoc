70struct FShaderCodeEntry
71{
72 // If the corresponding shader is part of the library,
73 // it is stored uncompressed and assigned an index within
74 // the library.
75 static constexpr uint8 HasLibraryIndexTag = 1u << SF_NumBits;
76 static constexpr uint32 FrequencyPayloadMask = HasLibraryIndexTag - 1u;
77
78 uint64 Offset = 0;
79 uint32 Size = 0;
80
81 // The active member is specified by the tag stored
82 // in the TagAndFrequency field. Rely on access methods.
83 struct
84 {
85 union
86 {
87 uint32 UncompressedSize;
88 uint32 LibraryIndex;
89 };
90 } Payload {};
91 uint8 TagAndFrequency = 0;
92
93 EShaderFrequency GetFrequency() const
94 {
95 return (EShaderFrequency)(TagAndFrequency & FrequencyPayloadMask);
96 }
97
98 void SetFrequency(EShaderFrequency Frequency)
99 {
100 check(Frequency <= FrequencyPayloadMask);
101 TagAndFrequency = (TagAndFrequency & HasLibraryIndexTag) | Frequency;
102 }
103
104 uint32 GetUncompressedSize() const
105 {
106 return IsPartOfLibrary() ? Size : Payload.UncompressedSize;
107 }
108
109 void SetUncompressedSize(uint32 InUncompressedSize)
110 {
111 check(!IsPartOfLibrary());
112 Payload.UncompressedSize = InUncompressedSize;
113 }
114
115 int32 GetLibraryIndex() const
116 {
117 return IsPartOfLibrary() ? Payload.LibraryIndex : INDEX_NONE;
118 }
119
120 void SetLibraryIndex(int32 Index)
121 {
122 check(Index >= 0);
123 Payload.LibraryIndex = Index;
124 TagAndFrequency |= HasLibraryIndexTag;
125 }
126
127 bool IsCompressed() const
128 {
129 return !IsPartOfLibrary() && Size != GetUncompressedSize();
130 }
131
132 bool IsPartOfLibrary() const
133 {
134 return (TagAndFrequency & HasLibraryIndexTag) != 0;
135 }
136
137 friend FArchive& operator <<(FArchive& Ar, FShaderCodeEntry& Ref)
138 {
139 return Ar << Ref.Offset << Ref.Size << (Ref.IsPartOfLibrary() ? Ref.Payload.LibraryIndex : Ref.Payload.UncompressedSize) << Ref.TagAndFrequency;
140 }
141};
