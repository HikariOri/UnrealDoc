130 Generator->Update(&_VerseScope, sizeof(_VerseScope), "SSettings::_VerseScope");

158

161 Generator->Update(Dependency->AsCString(), Dependency->ByteLen(), "SSettings::_DependencyPackages[]");
162 }
163

165 {
166 const CUTF8String& VniDestDir = *_VniDestDir;
167 Generator->Update(VniDestDir.AsCString(), VniDestDir.ByteLen(), "SSettings::_VniDestDir");

170 Generator->Update(&_bEnableSceneGraph, sizeof(_bEnableSceneGraph), "SSettings::_bEnableSceneGraph");

203 });
204

207 Generator->Update(Dependency->AsCString(), Dependency->ByteLen(), "SVersionedDigest::_DependencyPackages[]");
