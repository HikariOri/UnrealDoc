1011 { \

1013 { \
1014 extern CORE_API void RegisterSigningKeyCallback(void (*)(TArray<uint8>&, TArray<uint8>&)); \
1015 RegisterSigningKeyCallback(&Callback); \

1017 static void Callback(TArray<uint8>& OutExponent, TArray<uint8>& OutModulus) \
1018 { \
1019 const uint8 Exponent[] = { ExponentValue }; \
1020 const uint8 Modulus[] = { ModulusValue }; \
1021 OutExponent.SetNum(UE_ARRAY_COUNT(Exponent)); \
1022 OutModulus.SetNum(UE_ARRAY_COUNT(Modulus)); \
1023 for(int ByteIdx = 0; ByteIdx < UE_ARRAY_COUNT(Exponent); ByteIdx++) \
1024 { \
1025 OutExponent[ByteIdx] = Exponent[ByteIdx]; \
1026 } \
1027 for(int ByteIdx = 0; ByteIdx < UE_ARRAY_COUNT(Modulus); ByteIdx++) \
1028 { \
1029 OutModulus[ByteIdx] = Modulus[ByteIdx]; \
1030 } \
