41 T2DIndexer(TConstArrayView<ValueType> InValues)
42 {
43 FIntRect MinMaxKeys(FIntPoint(MAX_int32, MAX_int32), FIntPoint(MIN_int32, MIN_int32));
44
45 TArray<FIntPoint> AllKeys;
46 const int32 NumValues = InValues.Num();
47 AllKeys.AddDefaulted(NumValues);
48
49 // First, we need the bound of all values so that we can properly size the linear array :
50 for (int32 Index = 0; Index < NumValues; ++Index)
51 {
52 const ValueType& Value = InValues[Index];
53 FIntPoint Key = KeyFuncs::GetKey(Value);
54 AllKeys[Index] = Key;
55 MinMaxKeys.Include(Key);
56 }
57 if (!AllKeys.IsEmpty())
58 {
59 KeyExclusiveBounds = FIntRect(MinMaxKeys.Min, MinMaxKeys.Max + 1);
60 }
61
62 // Now we can properly size the array and fill it :
63 int32 NumEntries = KeyExclusiveBounds.Area();
64 AllValues.AddDefaulted(NumEntries);
65 ValidValueBitIndices.Init(false, NumEntries);
66 for (int32 Index = 0; Index < NumValues; ++Index)
67 {
68 int32 ValueIndex = GetValueIndexForKey(AllKeys[Index]);
69 AllValues[ValueIndex] = InValues[Index];
70 ValidValueBitIndices[ValueIndex] = true;
71 }

171 {
172 FIntRect LocalBounds = InBounds;
173 // Convert to exclusive bounds if necessary :
174 if (bInInclusiveBounds)
175 {
176 LocalBounds.Max += FIntPoint(1, 1);
177 }
178 // Intersect with the current key bounds in order to limit the test to only the tracked area :
179 LocalBounds.Clip(KeyExclusiveBounds);

228 TArray<ValueType> GetValidValuesForBitIndices(const TBitArray<>& InBitIndices)
229 {
230 // Intersect the list and the list of valid elements :
231 check(InBitIndices.Num() == ValidValueBitIndices.Num());
232 TBitArray<> LocalBitIndices = TBitArray<>::BitwiseAND(ValidValueBitIndices, InBitIndices, EBitwiseOperatorFlags::MinSize);
233
234 TArray<ValueType> Result;
235 Result.Reserve(InBitIndices.CountSetBits());
236 for (TConstSetBitIterator It(LocalBitIndices); It; ++It)
237 {
238 Result.Add(AllValues[It.GetIndex()]);
239 }
240 return Result;

243 FIntRect GetValidValuesBoundsForBitIndices(const TBitArray<>& InBitIndices, bool bInInclusiveBounds)
244 {
245 // Intersect the list and the list of valid elements :
246 check(InBitIndices.Num() == ValidValueBitIndices.Num());
247 TBitArray<> LocalBitIndices = TBitArray<>::BitwiseAND(ValidValueBitIndices, InBitIndices, EBitwiseOperatorFlags::MinSize);
248 if (LocalBitIndices.IsEmpty())
249 {
250 return FIntRect();
251 }
252
253 const int32 FirstSetBitIndex = InBitIndices.Find(true);
254 const int32 LastSetBitIndex = InBitIndices.FindLast(true);
255 if (FirstSetBitIndex == INDEX_NONE)
256 {
257 check(LastSetBitIndex == INDEX_NONE);
258 return FIntRect();
259 }
260
261 bool bIsValid = false;
262 const int32 Stride = KeyExclusiveBounds.Width();
263 FIntRect Bounds(FIntPoint(MAX_int32, MAX_int32), FIntPoint(MIN_int32, MIN_int32));
264 const int32 YMax = FMath::DivideAndRoundDown(LastSetBitIndex, Stride) + 1;
265 for (int32 Y = FirstSetBitIndex / Stride; Y < YMax; ++Y)
266 {
267 const int32 LineFirstSetBitIndex = LocalBitIndices.FindFrom(true, Y * Stride);
268 if (LineFirstSetBitIndex != INDEX_NONE)
269 {
270 const int32 LineLastSetBitIndex = LocalBitIndices.FindLastFrom(true, (Y + 1) * Stride - 1);
271 check(LineLastSetBitIndex != INDEX_NONE);
272 FIntPoint LineMinKey = GetValueKeyForIndex(LineFirstSetBitIndex);
273 FIntPoint LineMaxKey = GetValueKeyForIndex(LineLastSetBitIndex);
274 Bounds.Min = Bounds.Min.ComponentMin(LineMinKey);
275 Bounds.Max = Bounds.Max.ComponentMax(LineMaxKey);
276 bIsValid = true;
277 }
278 }
279
280 if (!bInInclusiveBounds)
281 {
282 Bounds.Max += FIntPoint(1, 1);
283 }
284
285 check(bIsValid);
286 return Bounds;
287 }
