135 FORCEINLINE_DEBUGGABLE void Raycast(const FVec3& Start, const FVec3& Dir, const FReal Length, SQVisitor& Visitor) const
136 {
137 FRealSingle CurrentLength = static_cast<FRealSingle>(Length);
138 const VectorRegister4Float StartSimd = MakeVectorRegisterFloatFromDouble(MakeVectorRegisterDouble(Start.X, Start.Y, Start.Z, 0.0));
139 const VectorRegister4Float DirSimd = MakeVectorRegisterFloatFromDouble(MakeVectorRegisterDouble(Dir.X, Dir.Y, Dir.Z, 0.0));
140 const VectorRegister4Float Parallel = VectorCompareGT(GlobalVectorConstants::SmallNumber, VectorAbs(DirSimd));
141 const VectorRegister4Float InvDirSimd = VectorBitwiseNotAnd(Parallel, VectorDivide(VectorOne(), DirSimd));
142
143 const auto BoundsFilter = [&StartSimd, &InvDirSimd, &Parallel, &CurrentLength](const FAABBVectorized& Bounds) -> EFilterResult
144 {
145 const VectorRegister4Float CurDataLength = VectorLoadFloat1(&CurrentLength);
146 const bool bHit = Bounds.RaycastFast(StartSimd, InvDirSimd, Parallel, CurDataLength);
147
148 return bHit ? EFilterResult::Keep : EFilterResult::Skip;
149 };
150
151 const auto FaceVisitor = [&Visitor, &CurrentLength](int32 FaceIndex)
152 {
153 const bool bContinueVisiting = Visitor.VisitRaycast(FaceIndex, CurrentLength);
154 return bContinueVisiting? EVisitorResult::Continue: EVisitorResult::Stop;
155 };
156
157 VisitTree(BoundsFilter, FaceVisitor);
158 }

232 FORCEINLINE_DEBUGGABLE EVisitorResult VisitFaces(int32 StartIndex, int32 IndexCount, BoundsFilterType& BoundsFilter, FaceVisitorType& FaceVisitor) const
233 {
234 const int32 EndIndex = (StartIndex + IndexCount);
235 for (int32 FaceIndex = StartIndex; FaceIndex < EndIndex; ++FaceIndex)

237 if (BoundsFilter(FaceBounds[FaceIndex]) == EFilterResult::Keep)
238 {
239 if (FaceVisitor(FaceIndex) == EVisitorResult::Stop)
240 {
241 return EVisitorResult::Stop;
242 }
243 }
244 }
245 return EVisitorResult::Continue;

249 FORCEINLINE_DEBUGGABLE void VisitTree(BoundsFilterType& BoundsFilter, FaceVisitorType& FaceVisitor) const
250 {
251 if (Nodes.Num() == 0)
252 {
253 return;
254 }
255 if (!FaceBounds.IsEmpty())
256 {
257 VisitTreeImpl<BoundsFilterType, FaceVisitorType, false>(BoundsFilter, FaceVisitor);
258 }
259 else
260 {
261 VisitTreeImpl<BoundsFilterType, FaceVisitorType, true>(BoundsFilter, FaceVisitor);

265 template <typename BoundsFilterType, typename FaceVisitorType, bool bUseQuantizedBounds>
266 FORCEINLINE_DEBUGGABLE void VisitTreeImpl(BoundsFilterType& BoundsFilter, FaceVisitorType& FaceVisitor) const
267 {
268 TArray<int32, TInlineAllocator<16>> NodeIndexStack;
269 NodeIndexStack.Push(0);
270 while (NodeIndexStack.Num())
271 {
272 const int32 NodeIndex = NodeIndexStack.Pop(EAllowShrinking::No);
273 check(Nodes.IsValidIndex(NodeIndex));
274 const FNode& Node = Nodes[NodeIndex];
275
276 for (int32 ChildIndex = 0; ChildIndex < 2; ++ChildIndex)
277 {
278 const FChildData& ChildData = Node.Children;
279 const int32 FaceIndex = ChildData.GetChildOrFaceIndex(ChildIndex);
280 if (FaceIndex != INDEX_NONE)
281 {
282 if (BoundsFilter(ChildData.GetBounds(ChildIndex)) == EFilterResult::Keep)
283 {
284 const uint32 FaceCount = ChildData.GetFaceCount(ChildIndex);
285 if (FaceCount > 0)
286 {
287 if constexpr (!bUseQuantizedBounds)
288 {
289 if (EVisitorResult::Stop == VisitFaces(FaceIndex, FaceCount, BoundsFilter, FaceVisitor))
290 {
291 return;
292 }
293 }
294 else
295 {
296 if (EVisitorResult::Stop == VisitFacesQuantized(ChildData.GetBounds(ChildIndex), FaceIndex, FaceCount, BoundsFilter, FaceVisitor))
297 {
298 return;
299 }
300 }
301 }
302 else
303 {
304 NodeIndexStack.Push(FaceIndex);
305 }
306 }
307 }
308 }
309 }

338 // alternate visit faces method, using quantized face bounds
339 template <typename BoundsFilterType, typename FaceVisitorType>
340 FORCEINLINE_DEBUGGABLE EVisitorResult VisitFacesQuantized(const FAABBVectorized& NodeBounds, int32 StartIndex, int32 IndexCount, BoundsFilterType& BoundsFilter, FaceVisitorType& FaceVisitor) const
341 {
342 // compute the scale factor to convert quantized uint8 values to relative positions within the node
343 constexpr VectorRegister4Float DeQuantFracScale = MakeVectorRegisterFloatConstant(1.f / 255.f, 1.f / 255.f, 1.f / 255.f, 1.f);
344 const VectorRegister4Float DeQuantScale = VectorMultiply(VectorSubtract(NodeBounds.GetMax(), NodeBounds.GetMin()), DeQuantFracScale);
345
346 const int32 EndIndex = (StartIndex + IndexCount);
347 int32 FaceIndex = StartIndex;
348 // Note EndIndex*6 is 1 element *past* the actual quantized bounds data, but should still be valid because we store an extra buffer byte to allow VectorLoadByte4 to be safely used
349 checkSlow(IndexCount > 0 && (QuantizedFaceBoundData.IsValidIndex(FaceIndex * 6) && QuantizedFaceBoundData.IsValidIndex(EndIndex * 6)));
350 for (const uint8* QBoundPtr = QuantizedFaceBoundData.GetData() + StartIndex * 6; FaceIndex < EndIndex; ++FaceIndex, QBoundPtr += 6)
351 {
352 if (BoundsFilter(FAABBVectorized(
353 VectorMultiplyAdd(DeQuantScale, VectorLoadByte4(&QBoundPtr[0]), NodeBounds.GetMin()),
354 VectorMultiplyAdd(DeQuantScale, VectorLoadByte4(&QBoundPtr[3]), NodeBounds.GetMin()))) == EFilterResult::Keep)
355 {
356 if (FaceVisitor(FaceIndex) == EVisitorResult::Stop)
357 {
358 return EVisitorResult::Stop;
359 }
360 }
361 }
362 return EVisitorResult::Continue;
363 }

721 void SerializeImp(FChaosArchive& Ar)
722 {
723 Ar.UsingCustomVersion(FExternalPhysicsCustomObjectVersion::GUID);
724 Ar.UsingCustomVersion(FUE5MainStreamObjectVersion::GUID);
725
726 FImplicitObject::SerializeImp(Ar);
727 Ar << MParticles;
728 Ar << MElements;
729 TBox<FReal, 3>::SerializeAsAABB(Ar, MLocalBoundingBox);
730
731 if (Ar.CustomVer(FExternalPhysicsCustomObjectVersion::GUID) < FExternalPhysicsCustomObjectVersion::RemovedConvexHullsFromTriangleMeshImplicitObject)
732 {
733 TUniquePtr<TGeometryParticles<FReal, 3>> ConvexHulls;
734 Ar << ConvexHulls;
735 }
736
737 if (Ar.CustomVer(FExternalPhysicsCustomObjectVersion::GUID) < FExternalPhysicsCustomObjectVersion::TrimeshSerializesBV)
738 {
739 // Should now only hit when loading older trimeshes
740 RebuildFastBVH();
741 }
742 else if (Ar.CustomVer(FExternalPhysicsCustomObjectVersion::GUID) < FExternalPhysicsCustomObjectVersion::TrimeshSerializesAABBTree)
743 {
744 TBoundingVolume<int32> Dummy;
745 Ar << Dummy;
746 RebuildFastBVH();
747 }
748 else if (Ar.CustomVer(FUE5MainStreamObjectVersion::GUID) < FUE5MainStreamObjectVersion::UseTriangleMeshBVH)
749 {
750 // Serialize acceleration
751 BVHType BVH;
752 Ar << BVH;
753 if (BVH.GetNodes().IsEmpty())
754 {
755 RebuildFastBVH();
756 }
757 else
758 {
759 RebuildFastBVHFromTree(BVH);
760 }
761 }
762 else
763 {
764 Ar << FastBVH;
765 }
766
767 if (Ar.CustomVer(FUE5MainStreamObjectVersion::GUID) < FUE5MainStreamObjectVersion::RemoveTriangleMeshBVHFaces)
768 {
769 // Force to rebuild the BVH
770 RebuildFastBVH();
771 }
772
773 if (Ar.CustomVer(FExternalPhysicsCustomObjectVersion::GUID) >= FExternalPhysicsCustomObjectVersion::AddTrimeshMaterialIndices)
774 {
775 Ar << MaterialIndices;
776 }
777
778 if (Ar.CustomVer(FExternalPhysicsCustomObjectVersion::GUID) >= FExternalPhysicsCustomObjectVersion::TriangleMeshHasFaceIndexMap)
779 {
780 // TODO: This data is only needed in editor unless project configuration requests this for gameplay. We should not serialize this when cooking
781 // unless it is required for gameplay, as we are wasting disk space.
782 if (Ar.IsLoading())
783 {
784 TUniquePtr<TArray<int32>> ExternalFaceIndexMapTemp = MakeUnique<TArray<int32>>(TArray<int32>());
785 Ar << *ExternalFaceIndexMapTemp;
786 if (!ExternalFaceIndexMapTemp->IsEmpty())
787 {
788 ExternalFaceIndexMap = MoveTemp(ExternalFaceIndexMapTemp);
789 }
790 }
791 else
792 {
793 if (ExternalFaceIndexMap == nullptr)
794 {
795 TArray<int32> EmptyArray;
796 Ar << EmptyArray;
797 }
798 else
799 {
800 Ar << *ExternalFaceIndexMap;
801 }
802 }
803 }
804
805 Ar.UsingCustomVersion(FPhysicsObjectVersion::GUID);
806 if (Ar.CustomVer(FPhysicsObjectVersion::GUID) >= FPhysicsObjectVersion::TriangleMeshHasVertexIndexMap)
807 {
808 if (Ar.IsLoading())
809 {
810 ExternalVertexIndexMap = MakeUnique<TArray<int32>>(TArray<int32>());
811 Ar << *ExternalVertexIndexMap;
812 }
813 else
814 {
815 if (ExternalVertexIndexMap == nullptr)
816 {
817 TArray<int32> EmptyArray;
818 Ar << EmptyArray;
819 }
820 else
821 {
822 Ar << *ExternalVertexIndexMap;
823 }
824 }
825 }

828 if (Ar.IsLoading() && FastBVH.NeedsQuantizedBounds())
829 {
830 MeshSettingsFlags |= ETriangleMeshSettingsFlags::QuantizePerFaceBounds;
831 FastBVH.InitializeQuantizedBounds(MParticles, MElements);
832 }

884 template<typename TriangleVisitor>
885 void VisitTriangles(const FAABB3& QueryBounds, const FRigidTransform3& QueryTransform, const TriangleVisitor& Visitor) const
886 {
887 TArray<int32> OverlapIndices;
888 FindOverlappingTriangles(QueryBounds, OverlapIndices);
889
890 const bool bStandardWinding = ((QueryTransform.GetScale3D().X * QueryTransform.GetScale3D().Y * QueryTransform.GetScale3D().Z) >= FReal(0));
891
892 for (int32 OverlapIndex = 0; OverlapIndex < OverlapIndices.Num(); ++OverlapIndex)
893 {
894 const int32 TriangleIndex = OverlapIndices[OverlapIndex];
895 FTriangle Triangle;
896 int32 VertexIndex0, VertexIndex1, VertexIndex2;
897 GetTransformedTriangle(TriangleIndex, QueryTransform, Triangle, VertexIndex0, VertexIndex1, VertexIndex2);
898
899 if (!bStandardWinding)
900 {
901 Triangle.ReverseWinding();
902 Swap(VertexIndex1, VertexIndex2);
903 }
904
905 Visitor(Triangle, TriangleIndex, VertexIndex0, VertexIndex1, VertexIndex2);
906 }
