312 template<typename TaskImpl, typename... TaskConstructionArgs>
313 FGraphEventRef Dispatch(FEntityManager* EntityManager, const FSystemTaskPrerequisites& Prerequisites, FSystemSubsequentTasks* Subsequents, TaskConstructionArgs&&... InArgs) const
314 {
315 static_assert(sizeof...(T) == 0, "Dispatch() is only for non-structured logic, which means that any call to Read or Write (or their variants) won't do anything -- please remove them");
316
317 checkfSlow(IsInGameThread(), TEXT("Tasks can only be dispatched from the game thread."));
318
319 const bool bRunInline = !ensure(EntityManager->IsLockedDown()) || EntityManager->GetThreadingModel() == EEntityThreadingModel::NoThreading;
320 if (bRunInline)
321 {
322 TaskImpl Task{ Forward<TaskConstructionArgs>(InArgs)... };
323 Task.Run();
324 return nullptr;
325 }
326 else
327 {
328 FGraphEventArray GatheredPrereqs;
329 this->PopulatePrerequisites(Prerequisites, &GatheredPrereqs);
330
331 ENamedThreads::Type ThisThread = EntityManager->GetDispatchThread();
332 checkSlow(ThisThread != ENamedThreads::AnyThread);
333
334 FGraphEventRef NewTask = TGraphTask< TUnstructuredTask<TaskImpl> >::CreateTask(GatheredPrereqs.Num() != 0 ? &GatheredPrereqs : nullptr, ThisThread)
335 .ConstructAndDispatchWhenReady( this->CommonParams, Forward<TaskConstructionArgs>(InArgs)... );
336
337 if (Subsequents)
338 {
339 this->PopulateSubsequents(NewTask, *Subsequents);
340 }
341
342 return NewTask;
343 }
344 }

371 template<typename TaskImpl, typename... TaskConstructionArgs>
372 FGraphEventRef Dispatch_PerAllocation(FEntityManager* EntityManager, const FSystemTaskPrerequisites& Prerequisites, FSystemSubsequentTasks* Subsequents, TaskConstructionArgs&&... InArgs) const
373 {
374 checkfSlow(IsInGameThread(), TEXT("Tasks can only be dispatched from the game thread."));
375
376 if (!this->IsValid())
377 {
378 return nullptr;
379 }
380
381 // Quick check to prevent dispatching a task that might not have any work to do. We only check the accessors
382 // here, so the task could still early-return with no work done if some custom filters end up matching
383 // nothing. Callers should in this case do their best to prevent useless tasks being dispatched.
384 if (!this->HasAnyWork(EntityManager))
385 {
386 return nullptr;
387 }
388
389 // If this ensure triggers, we are not in the evaluation phase - the callee should be using RunInline_ or Iterate_ variants
390 const bool bRunInline = !ensure(EntityManager->IsLockedDown()) || EntityManager->GetThreadingModel() == EEntityThreadingModel::NoThreading;
391 if (bRunInline)
392 {
393 TaskImpl Task{ Forward<TaskConstructionArgs>(InArgs)... };
394 TEntityAllocationTaskBase<TaskImpl, T...>(EntityManager, *this).Run(Task);
395 return nullptr;
396 }
397 else
398 {
399 FGraphEventArray GatheredPrereqs;
400 this->PopulatePrerequisites(Prerequisites, &GatheredPrereqs);
401
402 ENamedThreads::Type ThisThread = EntityManager->GetDispatchThread();
403 checkSlow(ThisThread != ENamedThreads::AnyThread);
404
405 FGraphEventRef NewTask = TGraphTask< TEntityAllocationTask<TaskImpl, T...> >::CreateTask(GatheredPrereqs.Num() != 0 ? &GatheredPrereqs : nullptr, ThisThread)
406 .ConstructAndDispatchWhenReady( EntityManager, *this, Forward<TaskConstructionArgs>(InArgs)... );
407
408 if (Subsequents)
409 {
410 this->PopulateSubsequents(NewTask, *Subsequents);
411 }
412
413 return NewTask;
414 }

591 template<typename TaskImpl, typename... TaskConstructionArgs>
592 FTaskID ScheduleImpl(FEntityManager* EntityManager, IEntitySystemScheduler* InScheduler, const FTaskParams& TaskParams, TaskFunctionPtr InFunction, TaskConstructionArgs&&... InArgs) const
593 {
594 if (!this->IsValid())
595 {
596 return FTaskID::None();
597 }
598
599 if (!this->HasAnyWork(EntityManager))
600 {
601 return FTaskID::None();
602 }
603
604 using TaskType = TScheduledEntityTask<TaskImpl, T...>;
605 TSharedPtr<TaskType> SharedTask = MakeShared<TaskType>(*this, Forward<TaskConstructionArgs>(InArgs)...);
606
607 FComponentMask ReadDependencies, WriteDependencies;
608 this->PopulateReadWriteDependencies(ReadDependencies, WriteDependencies);
609
610 FEntityComponentFilter Filter;
611 this->PopulateFilter(&Filter);
612
613 auto PrelockComponentData = [this](FEntityAllocationIteratorItem Allocation, TArray<FPreLockedDataPtr>& OutComponentHeaders)
614 {
615 this->PreLockComponentHeaders(Allocation, OutComponentHeaders);
616 };
617 FTaskID TaskID = InScheduler->CreateForkedAllocationTask(TaskParams, SharedTask, InFunction, PrelockComponentData, Filter, ReadDependencies, WriteDependencies);
618
619 if (!TaskID && TaskParams.bForcePrePostTask)
620 {
621 TaskID = InScheduler->AddNullTask();
622 }
623
624 if (TaskID)
625 {
626 FTaskID PreTask = SchedulePreTask(InScheduler, TaskParams, SharedTask, (TaskImpl*)0);
627 InScheduler->AddChildFront(TaskID, PreTask);
628
629 FTaskID PostTask = SchedulePostTask(InScheduler, TaskParams, SharedTask, (TaskImpl*)0);
630 InScheduler->AddChildBack(TaskID, PostTask);
631 }
632
633 return TaskID;
634 }
