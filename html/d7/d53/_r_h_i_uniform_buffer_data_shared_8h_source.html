23 inline FRHIDescriptorHandle GetBindlessResourceHandle(FUniformDataReader Reader, const FRHIUniformBufferResource& Resource)
24 {
25 switch (Resource.MemberType)
26 {
27 case UBMT_TEXTURE:
28 {
29 FRHITexture* Texture = Reader.Read<FRHITexture*>(Resource);
30 return Texture ? Texture->GetDefaultBindlessHandle() : FRHIDescriptorHandle();
31 }
32 break;
33 case UBMT_SRV:
34 {
35 FRHIShaderResourceView* ShaderResourceView = Reader.Read<FRHIShaderResourceView*>(Resource);
36 return ShaderResourceView ? ShaderResourceView->GetBindlessHandle() : FRHIDescriptorHandle();
37 }
38 break;
39 case UBMT_SAMPLER:
40 {
41 FRHISamplerState* SamplerState = Reader.Read<FRHISamplerState*>(Resource);
42 return SamplerState ? SamplerState->GetBindlessHandle() : FRHIDescriptorHandle();
43 }
44 break;
45 case UBMT_RESOURCE_COLLECTION:
46 {
47 FRHIResourceCollection* ResourceCollection = Reader.Read<FRHIResourceCollection*>(Resource);
48 return ResourceCollection ? ResourceCollection->GetBindlessHandle() : FRHIDescriptorHandle();
49 }
50 break;
51 case UBMT_RDG_TEXTURE:
52 {
53 FRDGTexture* RDGTexture = Reader.Read<FRDGTexture*>(Resource);
54 FRHITexture* Texture = RDGTexture ? RDGTexture->GetRHI() : nullptr;
55 return Texture ? Texture->GetDefaultBindlessHandle() : FRHIDescriptorHandle();
56 }
57 break;
58 case UBMT_RDG_TEXTURE_SRV:
59 case UBMT_RDG_TEXTURE_NON_PIXEL_SRV:
60 case UBMT_RDG_BUFFER_SRV:
61 {
62 FRDGShaderResourceView* RDGShaderResourceView = Reader.Read<FRDGShaderResourceView*>(Resource);
63 if (RDGShaderResourceView)
64 RDGShaderResourceView->MarkResourceAsUsed();
65
66 FRHIShaderResourceView* ShaderResourceView = RDGShaderResourceView ? RDGShaderResourceView->GetRHI() : nullptr;
67 return ShaderResourceView ? ShaderResourceView->GetBindlessHandle() : FRHIDescriptorHandle();
68 }
69 break;
70 case UBMT_UAV:
71 {
72 FRHIUnorderedAccessView* UnorderedAccessView = Reader.Read<FRHIUnorderedAccessView*>(Resource);
73 return UnorderedAccessView ? UnorderedAccessView->GetBindlessHandle() : FRHIDescriptorHandle();
74 }
75 break;
76 case UBMT_RDG_TEXTURE_UAV:
77 case UBMT_RDG_BUFFER_UAV:
78 {
79 FRDGUnorderedAccessView* RDGUnorderedAccessView = Reader.Read<FRDGUnorderedAccessView*>(Resource);
80 FRHIUnorderedAccessView* UnorderedAccessView = RDGUnorderedAccessView ? RDGUnorderedAccessView->GetRHI() : nullptr;
81 return UnorderedAccessView ? UnorderedAccessView->GetBindlessHandle() : FRHIDescriptorHandle();
82 }
83 break;
84
85 default:
86 // Do nothing.
87 break;
88 }
89 return FRHIDescriptorHandle();
90 }

92 inline void UpdateUniformBufferConstants(void* DestinationData, const void* SourceData, const FRHIUniformBufferLayout& Layout, bool bAllowBindless = true)
93 {
94 check(DestinationData != nullptr);
95 check(SourceData != nullptr);
96
97 // First copy wholesale
98 FMemory::Memcpy(DestinationData, SourceData, Layout.ConstantBufferSize);
99
100#if PLATFORM_SUPPORTS_BINDLESS_RENDERING
101 if (bAllowBindless)
102 {
103 FUniformDataReader Reader(SourceData);
104
105 // Then copy indices over
106 for (const FRHIUniformBufferResource& Resource : Layout.Resources)
107 {
108 if (Resource.MemberType == UBMT_DESCRIPTOR_RANGE)
109 {
110 if (FRHIDescriptorRange* DescriptorRange = Reader.Read<FRHIDescriptorRange*>(Resource))
111 {
112 FRHIDescriptorRangeShaderData ShaderData = DescriptorRange->GetShaderData();
113 FMemory::Memcpy(reinterpret_cast<uint8*>(DestinationData) + Resource.MemberOffset, &ShaderData, sizeof(ShaderData));
114 }
115 }
116 else
117 {
118 const FRHIDescriptorHandle Handle = GetBindlessResourceHandle(Reader, Resource);
119 if (Handle.IsValid())
120 {
121 const uint32 BindlessIndex = Handle.GetIndex();
122 FMemory::Memcpy(reinterpret_cast<uint8*>(DestinationData) + Resource.MemberOffset, &BindlessIndex, sizeof(BindlessIndex));
123 }
124 }
125 }
126 }
127#endif
128 }
