111 void PreVisitChildren(const FViewModelPtr& ViewModel) override
112 {
113 FlagsType ThisModelFlags = ComputeFlagsForModel(ViewModel);
114 IndividualItemFlags.Emplace(ThisModelFlags);
115 AccumulatedChildFlags.Emplace(FlagsType::None);
116 }
117 void PostVisitChildren(const FViewModelPtr& ViewModel) override
118 {
119 FlagsType ThisModelFlags = IndividualItemFlags.Pop();
120 FlagsType FlagsPropagatedFromChildren = AccumulatedChildFlags.Pop();
121
122 ThisModelFlags = CombinePropagatedChildFlags(ThisModelFlags, FlagsPropagatedFromChildren);
123 PostComputeChildrenFlags(ViewModel, ThisModelFlags, AccumulatedChildFlags.Last());
124
125 AccumulatedChildFlags.Last() |= (ThisModelFlags & FlagsType::InheritedFromChildren);
126
127 if (ThisModelFlags != FlagsType::None)
128 {
129 CachedFlagsFromNodeID.Add(ViewModel->GetModelID(), ThisModelFlags);
130 }
131 }
132 void EndUpdate() override
133 {
134 RootFlags = AccumulatedChildFlags.Last();
135 AccumulatedChildFlags.Empty();
136 IndividualItemFlags.Empty();
137
138 CachedFlagsFromNodeID.Shrink();
139 }
140
141protected:
