210 virtual FReply OnDrop( const FGeometry& MyGeometry, const FDragDropEvent& DragDropEvent ) override
211 {
212 bool bHandled = false;
213 TArray<TWeakObjectPtr<AActor>> ActorsToDrop;
214
215 TSharedPtr<FActorDragDropOp> ActorDragOp = nullptr;
216 TSharedPtr<FFolderDragDropOp> FolderDragOp = nullptr;
217
218 if (const TSharedPtr<FCompositeDragDropOp> CompositeDragOp = DragDropEvent.GetOperationAs<FCompositeDragDropOp>())
219 {
220 ActorDragOp = CompositeDragOp->GetSubOp<FActorDragDropOp>();
221 FolderDragOp = CompositeDragOp->GetSubOp<FFolderDragDropOp>();
222 }
223 else
224 {
225 ActorDragOp = DragDropEvent.GetOperationAs<FActorDragDropOp>();
226 FolderDragOp = DragDropEvent.GetOperationAs<FFolderDragDropOp>();
227 }
228
229 if (ActorDragOp.IsValid())
230 {
231 ActorsToDrop = ActorDragOp->Actors;
232 bHandled = true;
233 }
234
235 if (FolderDragOp.IsValid())
236 {
237 if (UWorld* World = FolderDragOp->World.Get())
238 {
239 FActorFolders::GetWeakActorsFromFolders(*World, FolderDragOp->Folders, ActorsToDrop, FolderDragOp->RootObject);
240
241 bHandled = true;
242 }
243 }
244
245 if (ActorsToDrop.Num() > 0)
246 {
247 ViewModel->AddActorsToNewLayer(ActorsToDrop);
248 }
249
250 return bHandled ? FReply::Handled() : FReply::Unhandled();
251 }
252
253 FReply OnDragRow(const FGeometry& MyGeometry, const FPointerEvent& MouseEvent)
254 {
255 if (MouseEvent.IsMouseButtonDown(EKeys::LeftMouseButton) && ViewModel->GetSelectedLayers().Num() > 0)
256 {
257 TSharedRef<FLayersDragDropOp> DragDropOp = MakeShared<FLayersDragDropOp>();
258
259 for (TSharedPtr<FLayerViewModel> Layer : ViewModel->GetSelectedLayers())
260 {
261 FName LayerName = Layer->GetFName();
262 if (LayerName != NAME_None)
263 {
264 DragDropOp->Layers.Add(LayerName);
265 }
266 }
267
268 DragDropOp->Construct();
269 return FReply::Handled().BeginDragDrop(DragDropOp);
270 }
271
272 return FReply::Unhandled();
273 }

307 void UpdateSelection()

310 {
311 return;

314 bUpdatingSelection = true;
315 const auto& SelectedLayers = ViewModel->GetSelectedLayers();
316 ListView->ClearSelection();
317 for( auto LayerIter = SelectedLayers.CreateConstIterator(); LayerIter; ++LayerIter )
318 {
319 ListView->SetItemSelection( *LayerIter, true );
320 }
321
322 if( SelectedLayers.Num() == 1 )
323 {
324 ListView->RequestScrollIntoView( SelectedLayers[ 0 ] );
325 }
326 bUpdatingSelection = false;
327 }
