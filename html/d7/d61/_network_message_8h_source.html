173struct FNFSMessageHeader
174{
176 uint32 Magic;
178 uint32 PayloadSize;
180 uint32 PayloadCrc;
181
183 FNFSMessageHeader(const FSimpleAbstractSocket& InSocket)
184 : Magic(InSocket.GetMagic())
185 , PayloadSize(0)
186 , PayloadCrc(0)
187 {
188 }
189
191 FNFSMessageHeader(const FSimpleAbstractSocket& InSocket, const TArray<uint8>& Payload)
192 : Magic(InSocket.GetMagic())
193 {
194 // make a header for the given payload
195 PayloadSize = Payload.Num();
196 check(PayloadSize);
197 PayloadCrc = FCrc::MemCrc_DEPRECATED(Payload.GetData(), Payload.Num());
198 }
199
201 friend FArchive& operator<<(FArchive& Ar, FNFSMessageHeader& Header)
202 {
203 uint32 DesiredMagic = Header.Magic;
204 Ar << Header.Magic;
205 if (Ar.IsLoading())
206 {
207 check(DesiredMagic);
208 if (Header.Magic != DesiredMagic)
209 {
210 uint32 DesiredMagicSwapped = BYTESWAP_ORDER32(DesiredMagic);
211 check(DesiredMagic != DesiredMagicSwapped);
212 if (Header.Magic == DesiredMagicSwapped)
213 {
214 Ar.SetByteSwapping(!Ar.ForceByteSwapping());
215 Header.Magic = DesiredMagic;
216 }
217 }
218 }
219 if (Header.Magic == DesiredMagic)
220 {
221 Ar << Header.PayloadSize;
222 Ar << Header.PayloadCrc;
223 }
224 return Ar;
225 }
226
227
237 static SOCKETS_API bool WrapAndSendPayload(const TArray<uint8>& Payload, const FSimpleAbstractSocket& Socket);
238
247 static SOCKETS_API bool ReceivePayload(FArrayReader& OutPayload, const FSimpleAbstractSocket& Socket);
