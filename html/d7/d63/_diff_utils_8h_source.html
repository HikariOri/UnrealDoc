93 bool IsSubPropertyMatch(const FPropertySoftPath& PotentialBasePropertyPath) const
94 {
95 if (PropertyChain.Num() <= PotentialBasePropertyPath.PropertyChain.Num())
96 {
97 return false;
98 }
99
100 for (int32 CurChainElement = 0; CurChainElement < PotentialBasePropertyPath.PropertyChain.Num(); CurChainElement++)
101 {
102 if (PotentialBasePropertyPath.PropertyChain[CurChainElement] != PropertyChain[CurChainElement])
103 {
104 return false;
105 }
106 }
107
108 return true;
109 }
110
116 bool IsSubPropertyMatch(TConstArrayView<FName> PotentialBasePropertyNameArray) const
117 {
118 if (PropertyChain.Num() <= PotentialBasePropertyNameArray.Num())
119 {
120 return false;
121 }
122
123 for (int32 CurChainElement = 0; CurChainElement < PotentialBasePropertyNameArray.Num(); CurChainElement++)
124 {
125 if (PotentialBasePropertyNameArray[CurChainElement] != PropertyChain[CurChainElement].PropertyName)
126 {
127 return false;
128 }
129 }
130
131 return true;
132 }
