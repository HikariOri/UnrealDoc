29 struct FWorkGroupInfo
30 {
31 FUint32Point WorkGroupWorkBoundary;
32 uint32 FirstItem;

38 FWorkGroupInfo PackWorkGroupInfo(uint32 FirstItem, uint32 NumItems, uint32 Payload, uint32 CarryOverStartOffset, uint64 WorkGroupWorkBoundary)
39 {
40 checkSlow(NumItems > 0);
41 checkSlow(NumItems - 1 < (1U << NumItemBits));
42 checkSlow(FirstItem < (1U << (32U - NumItemBits)));

51 struct FPackedItem
52 {
53 uint32 Packed;
54 };

56 FPackedItem PackItem(bool bHasCarryOver, uint32 NumChildren, uint32 Payload, uint32 BatchPrefixSum)
57 {
58 checkSlow(NumChildren > 0);
59 checkSlow(NumChildren - 1 < (1U << NumItemBits));
60 //checkSlow(ParentWorkItemOffset < (1U << (32U - NumItemBits)));
61 checkSlow(BatchPrefixSum < (1U << PrefixBits));
62 // arbitrary per-item payload encoded in as many bits as we happen to have left over
63 checkSlow(Payload < (1U << PayLoadBits));
64 uint32 Packed = (bHasCarryOver ? 1u : 0u)
65 | ((NumChildren - 1u) << 1u)
66 | (BatchPrefixSum << (1u + NumItemBits))
67 | (Payload << (1u + NumItemBits + PrefixBits));
68 return FPackedItem { Packed };
69 }

87 template <typename PayloadGeneratorType>
88 void Add(PayloadGeneratorType& PayloadGenerator, uint32 NumChildren)
89 {
90 uint32 NumAdded = 0u;
91 while (NumAdded < NumChildren)
92 {
93 uint32 MaxChildrenThisBatch = ThreadGroupSize - CurrentWorkGroupPrefixSum;
94
95 if (MaxChildrenThisBatch > 0)
96 {
97 const uint32 NumChildrenThisItem = FMath::Min(MaxChildrenThisBatch, NumChildren - NumAdded);
98
99 Items.Add(PackItem(NumAdded != 0u, NumChildrenThisItem, PayloadGenerator.GetItemPayload(), CurrentWorkGroupPrefixSum));
100 CurrentWorkGroupNumItems += 1U;
101 NumAdded += NumChildrenThisItem;
102 CurrentWorkGroupPrefixSum += NumChildrenThisItem;
103 CurrentWorkGroupWorkBoundary |= 1ull << (CurrentWorkGroupPrefixSum - 1u);
104 }
105
106 // Flush batch if it is not possible to add any more items (for one of the reasons)
107 if (MaxChildrenThisBatch <= 0U || CurrentWorkGroupPrefixSum >= ThreadGroupSize)
108 {
109 WorkGroupInfos.Add(PackWorkGroupInfo(CurrentWorkGroupFirstItem, CurrentWorkGroupNumItems, PayloadGenerator.GetWorkGroupPayload(), CurrentWorkGroupCarryOver, CurrentWorkGroupWorkBoundary));
110 CurrentWorkGroupFirstItem = uint32(Items.Num());
111 CurrentWorkGroupPrefixSum = 0u;
112 CurrentWorkGroupNumItems = 0U;
113 CurrentWorkGroupCarryOver = NumAdded;
114 CurrentWorkGroupWorkBoundary = 0ull;
115 }
116 }
117 TotalChildren += NumAdded;
118 }
119
120 bool IsEmpty() const
121 {
122 return Items.IsEmpty();
123 }

129 void FinalizeBatches(PayloadGeneratorType& PayloadGenerator)

134 CurrentWorkGroupNumItems = 0;
135 }
