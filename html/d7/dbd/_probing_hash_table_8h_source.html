132 FSlotId Add(uint32 Hash, uint32 Index)
133 {
134 if (UNLIKELY(FreeSlotCount == 0))
135 {
136 ExpandCapacity(1);
137 }
138
139 --FreeSlotCount;
140 for (uint32 BlockIndex = Hash & BlockCountMinusOne;; BlockIndex = (BlockIndex + 1) & BlockCountMinusOne)
141 {
142 FBlock& Block = Blocks[BlockIndex];
143 if (const uint64 AvailableSlots = ~Block.MaskAndHashes & FBlock::SlotMask)
144 {
145 const uint32 SlotIndexPlusOne = FMath::CountTrailingZeros((uint32)AvailableSlots);
146 Block.MaskAndHashes |= (uint64(1) << SlotIndexPlusOne) | (uint64(uint8(Hash >> 24)) << (SlotIndexPlusOne * 8));
147 Block.Slots[SlotIndexPlusOne - 1] = {Hash, Index};
148 return FSlotId::Create(BlockIndex, SlotIndexPlusOne);
149 }
150 Block.MaskAndHashes |= FBlock::ProbeMask;
151 }
152 }

159 inline void Remove(FSlotId Id)
160 {
161 checkSlow(Id.BlockIndex <= BlockCountMinusOne);
162 checkSlow(Id.SlotIndexPlusOne >= 1 && Id.SlotIndexPlusOne <= SlotsPerBlock);
163 checkSlow(Blocks && (Blocks[Id.BlockIndex].MaskAndHashes & (uint64(1) << Id.SlotIndexPlusOne)));
164 FBlock& Block = Blocks[Id.BlockIndex];
165 Block.MaskAndHashes &= ~((uint64(1) << Id.SlotIndexPlusOne) | (uint64(0xff) << (Id.SlotIndexPlusOne * 8)));
166 ++FreeSlotCount;
167 }

189 [[nodiscard]] FSlotId FindNext(FSlotId Id, uint32 Hash, uint32& OutIndex) const
190 {
191 checkSlow(Id.BlockIndex <= BlockCountMinusOne);
192
193 // Splat the most significant byte of Hash across a uint64.
194 uint64 HashMsb = uint8(Hash >> 24);
195 HashMsb |= HashMsb << 32;
196 HashMsb |= HashMsb << 16;
197 HashMsb |= HashMsb << 8;
198
199 uint32 BlockIndex = Id.BlockIndex;
200 uint32 SlotIndexPlusOne = Id.SlotIndexPlusOne;
201 uint32 SlotMask = (uint32(0xff) << SlotIndexPlusOne) & 0xff;
202
203 // Advance to the next slot.
204 ++SlotIndexPlusOne;
205 SlotMask &= SlotMask - 1;
206
207 for (;;)
208 {
209 const FBlock& Block = Blocks[BlockIndex];
210 if (Block.MaskAndHashes & SlotMask)
211 {
212 // Compare by XOR which leaves a 0 byte for each matching hash.
213 uint64 MatchingSlotMask = Block.MaskAndHashes ^ HashMsb;
214 // Propagate 1 bits in each byte into the least significant bit.
215 MatchingSlotMask |= MatchingSlotMask >> 4;
216 MatchingSlotMask |= MatchingSlotMask >> 2;
217 MatchingSlotMask |= MatchingSlotMask >> 1;
218 // Matching bytes have LSB=0. Negate the bits and mask out everything else.
219 MatchingSlotMask = ~MatchingSlotMask & 0x0101010101010101;
220 // Move the bits for matching slots from one byte each into one bit each in the least significant byte.
221 MatchingSlotMask = (MatchingSlotMask * 0x0102040810204080) >> 56;
222 // Mask out slots based on the starting slot index.
223 MatchingSlotMask &= uint64(0xff) << SlotIndexPlusOne;
224 // Mask out unoccupied slots.
225 MatchingSlotMask &= Block.MaskAndHashes;
226
227 // Compare the entire hash for any slots that matched the MSB of Hash.
228 for (; MatchingSlotMask; MatchingSlotMask &= MatchingSlotMask - 1)
229 {
230 SlotIndexPlusOne = FMath::CountTrailingZeros((uint32)MatchingSlotMask);
231 const FSlot& Slot = Block.Slots[SlotIndexPlusOne - 1];
232 if (Slot.Hash == Hash)
233 {
234 OutIndex = Slot.Index;
235 return FSlotId::Create(BlockIndex, SlotIndexPlusOne);
236 }
237 }
238 }
239
240 // Advance to the next block only if the probe flag has been set.
241 if (Block.MaskAndHashes & FBlock::ProbeMask)
242 {
243 BlockIndex = (BlockIndex + 1) & BlockCountMinusOne;
244 if (BlockIndex != Id.BlockIndex)
245 {
246 SlotIndexPlusOne = 1;
247 SlotMask = FBlock::SlotMask;
248 continue;
249 }
250 }
251
252 return {};
253 }
254 }

354 FORCENOINLINE void ExpandCapacity(const uint32 SlotCount)
355 {
356 FConstIterator It(*this);
357 FBlock* OldBlocks = Blocks;
358 ON_SCOPE_EXIT { delete[] OldBlocks; };
359
360 const uint32 MinCapacity = CalculateCapacity(BlockCountMinusOne + 1) + SlotCount;
361 const uint32 MinBlockCount = FMath::DivideAndRoundUp(MinCapacity + MinCapacity * (MaxLoadFactorDen - MaxLoadFactorNum) / MaxLoadFactorNum, SlotsPerBlock);
362 // Allocate at least 16 blocks and always round up to the next power of two to optimize calculations.
363 const uint32 BlockCount = FMath::Max(16u, FMath::RoundUpToPowerOfTwo(MinBlockCount));
364 BlockCountMinusOne = BlockCount - 1;
365 Blocks = new FBlock[BlockCount];
366
367 FreeSlotCount = CalculateCapacity(BlockCount);
368
369 // Copy everything from the old blocks to the new blocks.
370 for (; It; ++It)
371 {
372 Add(It.GetHash(), It.GetIndex());
373 }
374 }

377 constexpr static uint32 CalculateCapacity(uint32 BlockCount)
378 {
379 constexpr uint32_t Multiplier = SlotsPerBlock * MaxLoadFactorNum / MaxLoadFactorDen;
380 constexpr uint32_t Remainder = SlotsPerBlock * MaxLoadFactorNum - Multiplier * MaxLoadFactorDen;
381 return BlockCount * Multiplier + BlockCount * Remainder / MaxLoadFactorDen;
382 }
