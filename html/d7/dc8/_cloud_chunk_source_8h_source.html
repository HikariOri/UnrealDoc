108 struct FCloudSourceConfig
109 {
110 // An array of cloud root paths, supporting HTTP(s) and file access. HTTP(s) roots must begin with the protocol.
111 TArray<FString> CloudRoots;
112 // The maximum number of times that a single chunk should retry, before registering a fatal error.
113 // Infinite can be specified as < 0.
114 int32 MaxRetryCount = 6;
115 // Number of retries to enable logging of http warnings.
116 int32 RetriesToEnableVerbose;
117 // The minimum number of chunks to request ahead of what is required, depending on store slack.
118 int32 PreFetchMinimum = 16;
119 // The maximum number of chunks to request ahead of what is required, depending on store slack.
120 // This shouldn't be too much because we'll queue too many on the same CDN which might not be good and we have to chew
121 // though a ton of requests before we swap over, and we only really want there to be one queued + one ready to go per connection anyway.
122 int32 PreFetchMaximum = 32;
123 // Maximum download speed in bytes per second.
124 // Infinite can be specified as 0.
125 uint64 DownloadRateLimitBps = 0;
126 // Array of times in seconds, representing the time between each retry upon failure. The last entry will be used
127 // indefinitely once it is reached.
128 TArray<float> RetryDelayTimes;
129 // Configures what success rate values apply to which EBuildPatchDownloadHealth value, there should be
130 // EBuildPatchDownloadHealth::NUM_Values entries in this array.
131 TArray<float> HealthPercentages;
132 // When all requests are failing, how many seconds before a success until we determine the state as disconnected.
133 float DisconnectedDelay = 5.0f;
134 // If true, the downloads will not begin until the first Get request is made. It is fairly fundamental to stop downloads of chunks until resume
135 // data is processed, but can be special case disabled.
136 bool bBeginDownloadsOnFirstGet = true;
137 // The minimum time to allow a http download before assessing it as affected by TCP zero window issue.
138 float TcpZeroWindowMinimumSeconds = 20.0f;
139 // The context for allocating shared resources.
140 IBuildInstallerSharedContext* SharedContext = nullptr;
141 // Whether cloud source should run its own thread. If true, SharedContext is required.
142 bool bRunOwnThread = true;
143
148 FCloudSourceConfig(const TArray<FString>& InCloudRoots)
149 : CloudRoots(InCloudRoots)
150 , RetriesToEnableVerbose(MaxRetryCount - 1)
151 {
152 const float RetryFloats[] = {0.5f, 1.0f, 1.0f, 3.0f, 3.0f, 10.0f, 10.0f, 20.0f, 20.0f, 30.0f};
153 RetryDelayTimes.Empty(UE_ARRAY_COUNT(RetryFloats));
154 RetryDelayTimes.Append(RetryFloats, UE_ARRAY_COUNT(RetryFloats));
155 const float HealthFloats[] = {0.0f, 0.0f, 0.9f, 0.99f, 1.0f};
156 check((int32)EBuildPatchDownloadHealth::NUM_Values == UE_ARRAY_COUNT(HealthFloats));
157 HealthPercentages.Empty(UE_ARRAY_COUNT(HealthFloats));
158 HealthPercentages.Append(HealthFloats, UE_ARRAY_COUNT(HealthFloats));
159 }
