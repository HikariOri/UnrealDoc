409 inline float ProcessSample(float InSample)
410 {
411 float NormSample = NormalizeSample(InSample);
412
413 float Diff = Value - NormSample;
414 if (Value <= NormSample)
415 {
416 Value = (GetAttackTimeSamples() * Diff) + NormSample;
417 }
418 else
419 {
420 Value = (GetReleaseTimeSamples() * Diff) + NormSample;
421 }
422
423 return Value;

426 inline void ProcessBuffer(const float* InSamples, const int32 InNumSamples, float* OutSamples)
427 {
428 check(InSamples != OutSamples);
429
430 NormalizeSamples(InSamples, InNumSamples, OutSamples);
431
432 for (int32 SampleIndex = 0; SampleIndex < InNumSamples; ++SampleIndex)
433 {
434 float NormSample = OutSamples[SampleIndex];
435 float Diff = Value - NormSample;
436 if (Value <= NormSample)
437 {
438 Value = (GetAttackTimeSamples() * Diff) + NormSample;
439 }
440 else
441 {
442 Value = (GetReleaseTimeSamples() * Diff) + NormSample;
443 }
444 OutSamples[SampleIndex] = Value;
445 }

460 inline float NormalizeSample(float InSample)
461 {
462 switch (Mode)
463 {
464 case EPeakMode::Peak:
465 return FMath::Abs(InSample);
466
467 case EPeakMode::MeanSquared:
468
469 {
470 float SampleSquared = InSample * InSample;
471 AnalysisValue = AnalysisFilterBeta * SampleSquared + AnalysisFilterAlpha * AnalysisValue;
472 return AnalysisValue;
473 }
474
475 case EPeakMode::RootMeanSquared:
476
477 {
478 float SampleSquared = InSample * InSample;
479 AnalysisValue = AnalysisFilterBeta * SampleSquared + AnalysisFilterAlpha * AnalysisValue;
480 return FMath::Sqrt(AnalysisValue);
481 }
482
483 case EPeakMode::TruePeak:
484 default:
485 {
486 checkNoEntry();
487 }
488 }
489
490 return InSample;
491 }

493 inline void NormalizeSamples(const float* InSamples, const int32 InNumSamples, float* OutSamples)
494 {
495 TArrayView<const float> In = TArrayView<const float>(InSamples, InNumSamples);
496 TArrayView<float> Out = TArrayView<float>(OutSamples, InNumSamples);
497 switch (Mode)
498 {
499 case EPeakMode::Peak:
500 ArrayAbs(In, Out);
501 return;
502
503 case EPeakMode::MeanSquared:
504 {
505 ArraySquare(In, Out);
506 for (int32 SampleIndex = 0; SampleIndex < InNumSamples; ++SampleIndex)
507 {
508 AnalysisValue = AnalysisFilterBeta * OutSamples[SampleIndex] + AnalysisFilterAlpha * AnalysisValue;
509 OutSamples[SampleIndex] = AnalysisValue;
510 }
511 return;
512 }
513
514 case EPeakMode::RootMeanSquared:
515
516 {
517 ArraySquare(In, Out);
518 for (int32 SampleIndex = 0; SampleIndex < InNumSamples; ++SampleIndex)
519 {
520 AnalysisValue = AnalysisFilterBeta * OutSamples[SampleIndex] + AnalysisFilterAlpha * AnalysisValue;
521 OutSamples[SampleIndex] = AnalysisValue;
522 }
523 ArraySqrtInPlace(Out);
524 return;
525 }
526
527 case EPeakMode::TruePeak:
528 default:
529 {
530 checkNoEntry();
531 }
532 }
533 }

538 {

540 AnalysisFilterBeta = 1.f - AnalysisFilterAlpha;
541 }
542 }
