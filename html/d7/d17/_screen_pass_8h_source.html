473void DrawScreenPass(
474 FRHICommandList& RHICmdList,
475 FScreenPassViewInfo ViewInfo,
476 const FScreenPassTextureViewport& OutputViewport,
477 const FScreenPassTextureViewport& InputViewport,

492 DrawScreenPass_PostSetup(RHICmdList, ViewInfo, OutputViewport, InputViewport, PipelineState, Flags);
