809// 1d, 31 bit (uses the sign bit for internal use), O(n) where n is the amount of elements stored
810// does not enforce any alignment
811// unoccupied regions get compacted but occupied don't get compacted
812class FRangeAllocator
813{
814public:
815
816 struct FRange
817 {
818 // not valid
819 FRange()
820 : Start(0)
821 , Size(0)
822 {
823 check(!IsValid());
824 }
825
826 void SetOccupied(int32 InStart, int32 InSize)
827 {
828 check(InStart >= 0);
829 check(InSize > 0);
830
831 Start = InStart;
832 Size = InSize;
833 check(IsOccupied());
834 }
835
836 void SetUnOccupied(int32 InStart, int32 InSize)
837 {
838 check(InStart >= 0);
839 check(InSize > 0);
840
841 Start = InStart;
842 Size = -InSize;
843 check(!IsOccupied());
844 }
845
846 bool IsValid() { return Size != 0; }
847
848 bool IsOccupied() const { return Size > 0; }
849 uint32 ComputeSize() const { return (Size > 0) ? Size : -Size; }
850
851 // @apram InSize can be <0 to remove from the size
852 void ExtendUnoccupied(int32 InSize) { check(!IsOccupied()); Size -= InSize; }
853
854 void MakeOccupied(int32 InSize) { check(InSize > 0); check(!IsOccupied()); Size = InSize; }
855 void MakeUnOccupied() { check(IsOccupied()); Size = -Size; }
856
857 bool operator==(const FRange& rhs) const { return Start == rhs.Start && Size == rhs.Size; }
858
859 int32 GetStart() { return Start; }
860 int32 GetEnd() { return Start + ComputeSize(); }
861
862 private:
863 // in bytes
864 int32 Start;
865 // in bytes, 0:not valid, <0:unoccupied, >0:occupied
866 int32 Size;
867 };
868public:
869
870 // constructor
871 FRangeAllocator(uint32 TotalSize)
872 {
873 FRange NewRange;
874
875 NewRange.SetUnOccupied(0, TotalSize);
876
877 Entries.Add(NewRange);
878 }
879
880 // specified range must be non occupied
881 void OccupyRange(FRange InRange)
882 {
883 check(InRange.IsValid());
884 check(InRange.IsOccupied());
885
886 for(uint32 i = 0, Num = Entries.Num(); i < Num; ++i)
887 {
888 FRange& ref = Entries[i];
889
890 if(!ref.IsOccupied())
891 {
892 int32 OverlapSize = ref.GetEnd() - InRange.GetStart();
893
894 if(OverlapSize > 0)
895 {
896 int32 FrontCutSize = InRange.GetStart() - ref.GetStart();
897
898 // there is some front part we cut off
899 if(FrontCutSize > 0)
900 {
901 FRange NewFrontRange;
902
903 NewFrontRange.SetUnOccupied(InRange.GetStart(), ref.ComputeSize() - FrontCutSize);
904
905 ref.SetUnOccupied(ref.GetStart(), FrontCutSize);
906
907 ++i;
908
909 // remaining is added behind the found element
910 Entries.Insert(NewFrontRange, i);
911
912 // don't access ref or Num any more - Entries[] might be reallocated
913 }
914
915 check(Entries[i].GetStart() == InRange.GetStart());
916
917 int32 BackCutSize = Entries[i].ComputeSize() - InRange.ComputeSize();
918
919 // otherwise the range was already occupied or not enough space was left (internal error)
920 check(BackCutSize >= 0);
921
922 // there is some back part we cut off
923 if(BackCutSize > 0)
924 {
925 FRange NewBackRange;
926
927 NewBackRange.SetUnOccupied(Entries[i].GetStart() + InRange.ComputeSize(), BackCutSize);
928
929 Entries.Insert(NewBackRange, i + 1);
930 }
931
932 Entries[i] = InRange;
933 return;
934 }
935 }
936 }
937 }
938
939 // @param InSize >0
940 FRange AllocRange(uint32 InSize)//, uint32 Alignment)
941 {
942 check(InSize > 0);
943
944 for(uint32 i = 0, Num = Entries.Num(); i < Num; ++i)
945 {
946 FRange& ref = Entries[i];
947
948 if(!ref.IsOccupied())
949 {
950 uint32 RefSize = ref.ComputeSize();
951
952 // take the first fitting one - later we could optimize for minimal fragmentation
953 if(RefSize >= InSize)
954 {
955 ref.MakeOccupied(InSize);
956
957 FRange Ret = ref;
958
959 if(RefSize > InSize)
960 {
961 FRange NewRange;
962
963 NewRange.SetUnOccupied(ref.GetEnd(), RefSize - InSize);
964
965 // remaining is added behind the found element
966 Entries.Insert(NewRange, i + 1);
967 }
968 return Ret;
969 }
970 }
971 }
972
973 // nothing found
974 return FRange();
975 }
976
977 // @param In needs to be what was returned by AllocRange()
978 void ReleaseRange(FRange In)
979 {
980 int32 Index = Entries.Find(In);
981
982 check(Index != INDEX_NONE);
983
984 FRange& refIndex = Entries[Index];
985
986 refIndex.MakeUnOccupied();
987
988 Compacten(Index);
989 }
990
991 // for debugging
992 uint32 GetNumEntries() const { return Entries.Num(); }
993
994 // for debugging
995 uint32 ComputeUnoccupiedSize() const
996 {
997 uint32 Ret = 0;
998
999 for(uint32 i = 0, Num = Entries.Num(); i < Num; ++i)
1000 {
1001 const FRange& ref = Entries[i];
1002
1003 if(!ref.IsOccupied())
1004 {
1005 uint32 RefSize = ref.ComputeSize();
1006
1007 Ret += RefSize;
1008 }
1009 }
1010
1011 return Ret;
1012 }

1014private:
1015 // compact unoccupied ranges
1016 void Compacten(uint32 StartIndex)
1017 {
1018 check(!Entries[StartIndex].IsOccupied());
1019
1020 if(StartIndex && !Entries[StartIndex-1].IsOccupied())
1021 {
1022 // Seems we can combine with the element before,
1023 // searching further is not needed as we assume the buffer was compact before the last change.
1024 --StartIndex;
1025 }
1026
1027 uint32 ElementsToRemove = 0;
1028 uint32 SizeGained = 0;
1029
1030 for(uint32 i = StartIndex + 1, Num = Entries.Num(); i < Num; ++i)
1031 {
1032 FRange& ref = Entries[i];
1033
1034 if(!ref.IsOccupied())
1035 {
1036 ++ElementsToRemove;
1037 SizeGained += ref.ComputeSize();
1038 }
1039 else
1040 {
1041 break;
1042 }
1043 }
1044
1045 if(ElementsToRemove)
1046 {
1047 Entries.RemoveAt(StartIndex + 1, ElementsToRemove, EAllowShrinking::No);
1048 Entries[StartIndex].ExtendUnoccupied(SizeGained);
1049 }
1050 }
