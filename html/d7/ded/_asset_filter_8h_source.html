93 virtual bool PassesFilter( FilterType InItem ) const override
94 {
95 // The Conversion Function is preferred if it is specified
96 if(OnConvertItem.IsBound())
97 {
98 // Convert the item to an FAssetData
99 FAssetData AssetData;
100 bool bWasConverted = OnConvertItem.Execute(InItem, AssetData);
101
102 // Skip the current item if it wasn't converted
103 if(!bWasConverted)
104 {
105 /* If the current item does not want to be compared with Asset Type Filters, return true if there are no
106 * asset type filters active, since we want to show it then. Return false if there are any filters active
107 * since we only want to show items that pass the filters if so */
108 return CompiledBackendFilter.IsEmpty();
109 }
110
111 // Use the AssetRegistry to check if the Item passes the compiled filter
112 return IAssetRegistry::Get()->IsAssetIncludedByFilter(AssetData, CompiledBackendFilter);
113 }
114 // If not, try to use the comparison function
115 else if(OnCompareItem.IsBound())
116 {
117 // If there are no filters active just return true
118 if(CompiledBackendFilter.IsEmpty())
119 {
120 return true;
121 }
122
123 return OnCompareItem.Execute(InItem, CompiledBackendFilter.ClassPaths);
124 }
125
126 return false;
127 }
