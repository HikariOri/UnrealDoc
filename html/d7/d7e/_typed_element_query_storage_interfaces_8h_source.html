564 {
565 const UScriptStruct* TypeInfo = ColumnType::StaticStruct();
566
567 if constexpr (std::is_move_constructible_v<ColumnType>)
568 {
569 void* Address = AddColumnUninitialized(Row, TypeInfo,

660 template <typename T>
661 void ICommonQueryContext::PushCommand(T CommandContext)
662 {
663 if constexpr(std::is_empty_v<T>)
664 {
665 // The command object doesn't hold any arguments so use a cheaper path that only stores the callback.
666 PushCommand(
667 [](void*)
668 {
669 T Instance;
670 Instance.operator()();
671 }, nullptr);
672 }
673 else
674 {
675 // Wrapper lambda to store the type and call operator().
676 auto CommandFunction = [](void* InInstanceOfT)
677 {
678 T* Instance = static_cast<T*>(InInstanceOfT);
679 Instance->operator()();
680 };
681
682 FEmplaceObjectParams Params;
683
684 Params.ObjectSize = sizeof(T);
685 Params.Alignment = alignof(T);
686 Params.Construct = [](void* Destination, void* SourceCommandContext)
687 {
688 T& SourceCommand = *static_cast<T*>(SourceCommandContext);
689 new (Destination) T(MoveTemp(SourceCommand));
690 };
691 if (std::is_trivially_destructible_v<T>)
692 {
693 Params.Destroy = nullptr;
694 }
695 else
696 {
697 Params.Destroy = [](void* EmplacedObject)
698 {
699 static_cast<T*>(EmplacedObject)->~T();
700 };
701 }
702
703 Params.SourceObject = &CommandContext;
704
705 void* EmplacedCommandContext = EmplaceObjectInScratch(Params);
706 PushCommand(CommandFunction, EmplacedCommandContext);
707 }
708 }
