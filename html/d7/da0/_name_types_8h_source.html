240template <typename CharType>
241static constexpr uint32 ParseNumberFromName(const CharType* Name, int32& InOutLen)
242{
243 int32 Digits = 0;
244 for (const CharType* It = Name + InOutLen - 1; It >= Name && *It >= '0' && *It <= '9'; --It)
245 {
246 ++Digits;
247 }
248
249 const CharType* FirstDigit = Name + InOutLen - Digits;
250 constexpr int32 MaxDigitsInt32 = 10;
251 if (Digits && Digits < InOutLen && FirstDigit[-1] == '_' && Digits <= MaxDigitsInt32 && (Digits == 1 || *FirstDigit != '0'))
252 {
253 int64 Number = 0;
254 for (int32 Index = 0; Index < Digits; ++Index)
255 {
256 Number = 10 * Number + (FirstDigit[Index] - '0');
257 }
258 if (Number < MAX_int32)
259 {
260 InOutLen -= 1 + Digits;
261 return static_cast<uint32>(NAME_EXTERNAL_TO_INTERNAL(Number));
262 }
263 }
264
265 return NAME_NO_NUMBER_INTERNAL;
266}
