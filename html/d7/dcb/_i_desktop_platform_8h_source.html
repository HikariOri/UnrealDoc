27 EnableItalic = 0x4, // Whether the font should be generated in italics or not
28 EnableUnderline = 0x8, // Whether the font should be generated with an underline or not
29 AlphaOnly = 0x10, // Forces PF_G8 and only maintains Alpha value and discards color
30 CreatePrintableOnly = 0x20, // Skips generation of glyphs for any characters that are not considered 'printable'
31 IncludeASCIIRange = 0x40, // When specifying a range of characters and this is enabled, forces ASCII characters (0 thru 255) to be included as well
32 EnableDropShadow = 0x80, // Enables a very simple, 1-pixel, black colored drop shadow for the generated font
33 EnableLegacyMode = 0x100, // Enables legacy font import mode. This results in lower quality antialiasing and larger glyph bounds, but may be useful when debugging problems
34 UseDistanceFieldAlpha = 0x200 // Alpha channel of the font textures will store a distance field instead of a color mask
35};
