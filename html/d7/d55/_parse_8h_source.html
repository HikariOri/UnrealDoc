145 //
146 // Parse a hex digit.
147 //
148 static inline int32 HexDigit(TCHAR c)
149 {
150 int32 Result = 0;
151
152 if (c >= '0' && c <= '9')
153 {
154 Result = c - '0';
155 }
156 else if (c >= 'a' && c <= 'f')
157 {
158 Result = c + 10 - 'a';
159 }
160 else if (c >= 'A' && c <= 'F')
161 {
162 Result = c + 10 - 'A';
163 }
164 else
165 {
166 Result = 0;
167 }
168
169 return Result;
170 }
