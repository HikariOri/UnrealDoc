27// Affine arithmetic
28//
29// [ Thonat et al. 2021, "Tessellation-Free Displacement Mapping for Ray Tracing" ]
30// [ de Figueiredo and Stolf 2004, "Affine Arithmetic: Concepts and Applications" ]
31// [ Rump and Kashiwagi 2015, "Implementation and improvements of affine arithmetic" ]
32
33template< typename T, uint32 Num >
34struct TAffine
35{
36 T c;
37 T K;
38 T e[ Num ];
39
40 TAffine() {}
41 FORCEINLINE TAffine( T Constant )
42 : c( Constant )
43 , K( 0.0f )
44 {
45 for( uint32 i = 0; i < Num; i++ )
46 e[i] = T( 0.0f );
47 }
48
49 FORCEINLINE TAffine( T Min, T Max )
50 : c( 0.5f * ( Min + Max ) )
51 , K( 0.5f * ( Max - Min ) )
52 {
53 for( uint32 i = 0; i < Num; i++ )
54 e[i] = T( 0.0f );
55 }
56
57 FORCEINLINE TAffine( T Min, T Max, uint32 Index )
58 : c( 0.5f * ( Min + Max ) )
59 , K( 0.0f )
60 {
61 for( uint32 i = 0; i < Num; i++ )
62 e[i] = T( 0.0f );
63 e[ Index ] = 0.5f * ( Max - Min );
64 }
65
66 FORCEINLINE TAffine< T, Num >& operator+=( const TAffine< T, Num >& Other )
67 {
68 c += Other.c;
69 K += Other.K;
70 for( uint32 i = 0; i < Num; i++ )
71 e[i] += Other.e[i];
72
73 return *this;
74 }
75
76 FORCEINLINE TAffine< T, Num > operator+( const TAffine< T, Num >& Other ) const
77 {
78 return TAffine< T, Num >(*this) += Other;
79 }
80
81 FORCEINLINE TAffine< T, Num >& operator-=( const TAffine< T, Num >& Other )
82 {
83 c -= Other.c;
84 K += Other.K;
85 for( uint32 i = 0; i < Num; i++ )
86 e[i] -= Other.e[i];
87
88 return *this;
89 }
90
91 FORCEINLINE TAffine< T, Num > operator-( const TAffine< T, Num >& Other ) const
92 {
93 return TAffine< T, Num >(*this) -= Other;
94 }
95
96 FORCEINLINE T GetMin() const
97 {
98 T Result = c - Affine::Abs(K);
99 for( uint32 i = 0; i < Num; i++ )
100 Result -= Affine::Abs( e[i] );
101
102 return Result;
103 }
104
105 FORCEINLINE T GetMax() const
106 {
107 T Result = c + Affine::Abs(K);
108 for( uint32 i = 0; i < Num; i++ )
109 Result += Affine::Abs( e[i] );
110
111 return Result;
112 }
113
114 // Smaller than (v|v)
115 // require that T has SizeSquared and declares floating point type FReal
116 template <typename Q = T>
117 requires (UE::CFloatingPoint<typename Q::FReal> && std::is_member_function_pointer_v<decltype(&Q::SizeSquared)>)
118 FORCEINLINE TAffine<typename Q::FReal, Num> SizeSquared() const
119 {
120 using FReal = typename Q::FReal;
121
122 TAffine< FReal, Num > Result;
123 Result.c = c.SizeSquared();
124 Result.K = FReal(2.0) * Affine::Abs( c | K );
125
126 T Extent = K;
127 for( uint32 i = 0; i < Num; i++ )
128 {
129 Extent += Affine::Abs( e[i] );
130 Result.e[i] = FReal(2.0) * ( c | e[i] );
131 }
132
133 Result.c += FReal(0.5) * Extent.SizeSquared();
134 Result.K += FReal(0.5) * Extent.SizeSquared();
135
136 return Result;
137 }
138};

140template< typename T, typename U, uint32 Num >
141FORCEINLINE TAffine< T, Num > operator*( const TAffine< T, Num >& A, const TAffine< U, Num >& B )
142{
143 TAffine< T, Num > Result;
144 Result.c = A.c * B.c;
145 Result.K =
146 Affine::Abs( A.K * B.c ) +
147 Affine::Abs( A.c * B.K );
148
149 T AK = A.K;
150 U BK = B.K;
151 for( uint32 i = 0; i < Num; i++ )
152 {
153 Result.e[i] = A.e[i] * B.c + A.c * B.e[i];
154 AK += Affine::Abs( A.e[i] );
155 BK += Affine::Abs( B.e[i] );
156 }
157 Result.K += AK * BK;
158
159 return Result;
160}

162template< typename T, uint32 Num >
163FORCEINLINE TAffine< float, Num > operator|( const TAffine< T, Num >& A, const TAffine< T, Num >& B )
164{
165 TAffine< float, Num > Result;
166 Result.c = A.c | B.c;
167 Result.K =
168 Affine::Abs( A.K | B.c ) +
169 Affine::Abs( A.c | B.K );
170
171 T AK = A.K;
172 T BK = B.K;
173 for( uint32 i = 0; i < Num; i++ )
174 {
175 Result.e[i] = ( A.e[i] | B.c ) + ( A.c | B.e[i] );
176 AK += Affine::Abs( A.e[i] );
177 BK += Affine::Abs( B.e[i] );
178 }
179 Result.K += AK | BK;
180
181 return Result;
182}

184template <UE::CFloatingPoint T, uint32 Num>
185FORCEINLINE TAffine<T, Num> Clamp( const TAffine< T, Num >& x, T Min, T Max )
186{
187 // Using Chebyshev approximation
188 T xMin = x.GetMin();
189 T xMax = x.GetMax();
190 T FuncMin = FMath::Clamp( xMin, Min, Max );
191 T FuncMax = FMath::Clamp( xMax, Min, Max );
192
193 if( Min <= xMin && xMax <= Max )
194 return x;
195 if( xMax <= Min )
196 return TAffine< T, Num >( Min );
197 if( xMin >= Max )
198 return TAffine< T, Num >( Max );
199
200 T Alpha = ( FuncMax - FuncMin ) / ( xMax - xMin );
201 T Gamma = T(0.5) * ( T(1.) - Alpha ) * ( FuncMax + FuncMin );
202 T Delta = ( T(1.) - Alpha ) * FuncMax - Gamma;
203
204 TAffine<T, Num > Result;
205 Result.c = Alpha * x.c + Gamma;
206 Result.K = FMath::Abs( Alpha * x.K ) + Delta;
207 for( uint32 i = 0; i < Num; i++ )
208 Result.e[i] = Alpha * x.e[i];
209
210 return Result;
211}

213template <UE::CFloatingPoint T, uint32 Num>
214FORCEINLINE TAffine<T, Num> InvSqrt( const TAffine< T, Num >& x )
215{
216 // Using min range approximation
217 T xMin = FMath::Max( T(1.e-4), x.GetMin() );
218 T xMax = FMath::Max( T(1.e-4), x.GetMax() );
219 T FuncMin = FMath::InvSqrt( xMin );
220 T FuncMax = FMath::InvSqrt( xMax );
221
222 T Alpha = -T(0.5) * FuncMax * FuncMax * FuncMax;
223 T Gamma = T(0.5) * ( FuncMin + FuncMax - Alpha * ( xMin + xMax ) );
224 T Delta = FMath::Abs( T(0.5) * ( FuncMin - FuncMax - Alpha * ( xMin - xMax ) ) );
225
226 TAffine< T, Num > Result;
227 Result.c = Alpha * x.c + Gamma;
228 Result.K = FMath::Abs( Alpha * x.K ) + Delta;
229
230 for( uint32 i = 0; i < Num; i++ )
231 Result.e[i] = Alpha * x.e[i];
232
233 return Result;
234}

236template< typename T, uint32 Num >
237FORCEINLINE TAffine< T, Num > Normalize( const TAffine< T, Num >& x )
238{
239 using FReal = typename T::FReal;
240 return x * InvSqrt( Clamp<FReal, Num>( x.SizeSquared(), FReal(1.e-4), FReal(1.0) ) );
