94typedef enum OodleTexRT_BC7PrepDecodeFlags
95{
96 OodleTexRT_BC7PrepDecodeFlags_None = 0, // No special processing.
97 OodleTexRT_BC7PrepDecodeFlags_DestinationIsCached = 1, // Destination is in write-back cached memory.
98 OodleTexRT_BC7PrepDecodeFlags_AvoidWideVectors = 2, // Avoid using wide (>=256b) vector instructions.
99
100 OodleTexRT_BC7PrepDecodeFlags_Force32 = 0x40000000
101} OodleTexRT_BC7PrepDecodeFlags;
102/* OodleTexRT_BC7PrepDecodeFlags controls BC7Prep decoding.
103
104 Several of these flag values may be ORed together.
105
106 `OodleTexRT_BC7PrepDecodeFlags_DestinationIsCached` may be used when the destination address
107 is in cached memory, i.e. not uncached or write combined. This enables slightly faster
108 decoding, up to about 10%. However, setting this flag when the destination is actually in
109 write-combined memory will frequently result in a 2x slowdown instead, so tread carefully.
110
111 `OodleTexRT_BC7PrepDecodeFlags_AvoidWideVectors` disables usage of wide (>=256-bit) vector
112 instructions even on CPUs that support it. On certain CPUs (mainly Intel CPUs derived
113 from the Broadwell or Skylake microarchitectures) any use of a large class of 256-bit vector
114 instructions will reduce the maximum clock rate by up to 20%; furthermore, for most non-server
115 SKUs, this reduction affects all cores in the system, even when only a single core uses wide
116 vector instructions. BC7Prep decoding generally only runs in short bursts; you can pass this
117 flag to avoid use of 256-bit vectors in the decoder if nothing else in your app uses them, to
118 avoid penalizing all other threads in the system.
119*/
120

364OOFUNC1 OodleTexRT_Err OOFUNC2 OodleTexRT_BC7Prep_ReadHeader(const OodleTexRT_BC7PrepHeader * header,
365 OO_SINTa * out_num_blocks, OO_SINTa * out_payload_size);
366/* Reads and validates the header and determines the size of the decoded and coded BC7 blocks.
367
368 $:header Header for a BC7Prep data chunk.
369 $:out_num_blocks If non-null, receives number of coded BC7 blocks in this blob.
370 $:out_payload_size If non-null, receives size of corresponding BC7Prep chunk in bytes.
371 $:return OodleTexRT_OK on success, an error code if the header is malformed.
372
373 These numbers are useful for memory management. The block count determines how large the
374 data will be after decoding (and also how much scratch memory is required as per
375 $OodleTexRT_BC7Prep_MinDecodeScratchSize) and the payload size determines the size of the
376 BC7Prep data itself, in case you're not storing it anywhere else.
377
378 This function is provided for convenience. If you use a fixed limit on the number of blocks in
379 a single BC7Prep chunk and store their sizes elsewhere, you can just allocate all memory buffers
380 once up front and don't need to call this function at all.
381*/

383OOFUNC1 OO_SINTa OOFUNC2 OodleTexRT_BC7Prep_MinDecodeScratchSize(OO_SINTa nblocks);
384/* Returns how much decoder scratch memory (in bytes) is required for the given number of BC7 blocks.
385
386 BC7Prep decoding temporarily needs some memory during decoding, the scratch working buffer.
387 Its required size depends on the number of blocks encoded. On error (negative result), returns
388 an $OodleTexRT_Err.
389
390 You can either decide on a global maximum _nblocks_ and pre-allocate BC7Prep decoder scratch
391 memory once for that many blocks, or manage everything more dynamically and use
392 $OodleTexRT_BC7Prep_ReadHeader to figure out how many blocks a given BC7Prep chunk decodes to.
393
394 If you need to decode multiple BC7Prep blocks in sequence, it is recommended to allocate
395 the scratch buffer once (for the maximum _nblocks_ required) and reuse it between calls. Note
396 that if you're concurrently decoding BC7Prep blocks on multiple threads, each thread neads
397 its own scratch buffer.
398*/

400OOFUNC1 OO_SINTa OOFUNC2 OodleTexRT_BC7Prep_Decode(void * output_buf, OO_SINTa output_size,
401 const void * bc7prep_data, OO_SINTa bc7prep_data_size,
402 const OodleTexRT_BC7PrepHeader * header, OodleTexRT_BC7PrepDecodeFlags flags,
403 void * scratch_buf, OO_SINTa scratch_size);
404/* Decodes a BC7Prep data stream.
405
406 $:output_buf Where the decoded BC7 blocks get written to.
407 $:output_size Size of _output_buf_ in bytes.
408 $:bc7prep_data The encoded BC7Prep data chunk.
409 $:bc7prep_data_size Size of _bc7prep_data_ in bytes.
410 $:header The header for the data blob.
411 $:flags Decode flags. See notes below. Pass $OodleTexRT_BC7PrepDecodeFlags_None by default.
412 $:scratch_buf Pointer to a scratch working buffer.
413 $:scratch_size Size of _scratch_buf_ in bytes.
414 $:return Number of bytes written to _output_buf_ on success, a negative number ($OodleTexRT_Err) on error.
415
416 Decodes the given BC7Prep data chunk to _output_buf_; _output_size_ must be at least 16 times the number of
417 BC7 blocks encoded in the given BC7Prep data chunk.
418
419 Returns number of bytes written; on success this equals the BC7 block size (16 bytes) times the number
420 of blocks in the given BC7Prep chunk.
421
422 You can determine the number of blocks and _bc7prep_data_size_ from the header using
423 $OodleTexRT_BC7Prep_ReadHeader.
424
425 If _output_buf_ is known to be in cached memory (i.e. not write-combined memory like GPU memory
426 on consoles or sometimes GPU upload heaps on PC), you may set _flags_ to
427 $OodleTexRT_BC7PrepDecodeFlags_DestinationIsCached. This is up to 20% faster when the destination is
428 in cached memory, but sometimes takes 2x as long when it's not. When in doubt, leave it off.
429
430 The scratch buffer is used for all working memory during the decode process. BC7Prep decoding
431 does not perform any memory allocations. You can reuse the scratch memory between subsequent
432 BC7Prep_Decode calls (as long as it's large enough), but two concurrent BC7Prep_Decodes running
433 on different threads need to have their private scratch buffers.
434
435 BC7Prep _encoding_ is performed by $OodleTex_BC7Prep_Encode and lives in the main Oodle Texture
436 library, not the runtime.
