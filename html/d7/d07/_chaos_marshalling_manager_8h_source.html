81 template< typename TProxiesArray>
82 void AddMultipleUnsafe(TProxiesArray& ProxiesArray)
83 {
84 if(ProxiesArray.Num())
85 {
86 FDirtyProxiesBucket& Bucket = DirtyProxyBuckets[(uint32)ProxiesArray[0]->GetType()];
87 int32 Idx = Bucket.ProxiesData.Num();
88 Bucket.ProxiesData.Append(ProxiesArray);
89
90 for (IPhysicsProxyBase* Proxy : ProxiesArray)
91 {
92 Proxy->SetDirtyIdx(Idx++);
93 ensure(ProxiesArray[0]->GetType() == Proxy->GetType());
94 }
95
96 DirtyProxyBucketInfo.Num[(uint32)ProxiesArray[0]->GetType()] += ProxiesArray.Num();
97 DirtyProxyBucketInfo.TotalNum += ProxiesArray.Num();
98 }

100 }

102 // Forcefully removes the proxy from being dirty.
103 void Remove(IPhysicsProxyBase* Base)
104 {
105 const int32 Idx = Base->GetDirtyIdx();
106 if(Idx != INDEX_NONE)
107 {
108 FDirtyProxiesBucket& Bucket = DirtyProxyBuckets[(uint32)Base->GetType()];
109 if(Idx == Bucket.ProxiesData.Num() - 1)
110 {
111 //last element so just pop
112 Bucket.ProxiesData.Pop(EAllowShrinking::No);
113 --DirtyProxyBucketInfo.Num[(uint32)Base->GetType()];
114 --DirtyProxyBucketInfo.TotalNum;
115 }
116 else if(Bucket.ProxiesData.IsValidIndex(Idx))
117 {
118 //update other proxy's idx
119 Bucket.ProxiesData.RemoveAtSwap(Idx);
120 Bucket.ProxiesData[Idx].SetDirtyIdx(Idx);
121 --DirtyProxyBucketInfo.Num[(uint32)Base->GetType()];
122 --DirtyProxyBucketInfo.TotalNum;
123 }
124
125 Base->ResetDirtyIdx();
126 }
127 }
128
131 void RemoveWithoutDirtyIdx(IPhysicsProxyBase* Base, FDirtyPropertiesManager& Manager)
132 {
133 FDirtyProxiesBucket& Bucket = DirtyProxyBuckets[(uint32)Base->GetType()];
134 for (int32 Idx = Bucket.ProxiesData.Num() - 1; Idx >= 0; Idx--)
135 {
136 if (Bucket.ProxiesData[Idx].Proxy == Base)
137 {
138 const int32 LastIdx = Bucket.ProxiesData.Num() - 1;
139
140 // Clear data from manager
141 Bucket.ProxiesData[Idx].Clear(Manager, Idx, GetShapesDirtyData());
142
143 if (Idx == LastIdx)
144 {
145 // Last element, pop to remove it
146 Bucket.ProxiesData.Pop(EAllowShrinking::No);
147 }
148 else
149 {
150 // Remove and swap with last element, update moved proxy's idx
151 Bucket.ProxiesData.RemoveAtSwap(Idx);
152
153 // Move data in manager for the swapped element
154 Bucket.ProxiesData[Idx].MoveData(Manager, /*FromIdx*/LastIdx, /*ToIdx*/Idx);
155 }
156
157 --DirtyProxyBucketInfo.Num[(uint32)Base->GetType()];
158 --DirtyProxyBucketInfo.TotalNum;
159
160 // There is only one entry per proxy
161 break;
162 }
163 }
164 }
