15// this is needed because most of the COM VS automation interfaces only can be called from a single thread.

31 // helper function that runs any given function in the internal COM thread and waits until execution has finished
32 // BEGIN EPIC MOD
33 template <typename F, typename... Args>
34 inline std::invoke_result_t<F, Args...> CallInThread(F ptrToFunction, Args&&... args)
35 // END EPIC MOD
36 {
37 // BEGIN EPIC MOD
38 typedef std::invoke_result_t<F, Args...> ReturnValue;
39 // END EPIC MOD
40
41 // a helper lambda that calls the given functions with the provided arguments.
42 // because this lambda needs to capture, it cannot be converted to a function pointer implicitly,
43 // and therefore cannot be used as a COM function directly.
44 // however, we can solve this by another indirection.
45 auto captureLambda = [ptrToFunction, args...]()
46 {
47 return (*ptrToFunction)(args...);
48 };
49
50 // helper to make the following easier to read
51 typedef decltype(captureLambda) CaptureLambdaType;
52
53 // here's the trick: we generate another capture-less lambda that has the same signature as our COM function,
54 // and internally cast the given object to its original type, calling the lambda with captures from within
55 // this capture-less lambda.
56 // it is OK to leave both the lambda as well as the return value on the stack. this function returns only
57 // after the COM function has been called by the internal thread.
58 ReturnValue result = {};
59 m_function.function = [](void* captureLambda, void* returnValueAddr)
60 {
61 CaptureLambdaType* realCaptureLambda = static_cast<CaptureLambdaType*>(captureLambda);
62 ReturnValue* realReturnValueAddr = static_cast<ReturnValue*>(returnValueAddr);
63 *realReturnValueAddr = (*realCaptureLambda)();
64 };
65
66 m_function.context = &captureLambda;
67 m_function.returnValueAddr = &result;
68
69 // signal event that function in thread should now be called
70 m_functionAvailableEvent.Signal();
71
72 // wait for signal that the function has finished executing
73 m_functionFinishedExecutingEvent.Wait();
74
75 return result;
76 }
