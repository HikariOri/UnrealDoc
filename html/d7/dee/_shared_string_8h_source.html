188template <typename CharType>
189[[nodiscard]] TSharedString<CharType>::TSharedString(const TStringView<CharType> String)
190{
191 if (const int32 Length = String.Len())
192 {
193 static_assert(alignof(int32) <= MIN_ALIGNMENT);
194 static_assert(alignof(int32) >= alignof(CharType));
195 const SIZE_T Size = sizeof(int32) + sizeof(int32) + sizeof(CharType) + sizeof(CharType) * Length;
196 int32* const Header = static_cast<int32*>(FMemory::Malloc(Size));
197 // RefCount needs to be at the same address as the allocation to make FMemory::Free work as the DeleteFn.
198 new((void*)&Header[0]) FRefCountType(1);
199 new((void*)&Header[1]) int32(Length);
200 Chars = reinterpret_cast<CharType*>(&Header[2]);
201 String.CopyString(Chars, Length);
202 Chars[Length] = CharType(0);
203 }
204}
