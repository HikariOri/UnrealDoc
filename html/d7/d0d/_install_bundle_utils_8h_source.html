111 template<typename EnumType, typename StringArrType, EnumType EnumCount = EnumType::Count>
112 const TCHAR* TLexToString(EnumType E, const StringArrType& Strings)
113 {
114 constexpr auto Count = InstallBundleUtil::CastToUnderlying(EnumCount);
115 static_assert(Count == UE_ARRAY_COUNT(Strings), "");
116
117 auto Idx = InstallBundleUtil::CastToUnderlying(E);
118 if (Idx >= 0 && Idx < Count)
119 {
120 return Strings[Idx];
121 }
122 else
123 {
124 return TEXT("");
125 }
