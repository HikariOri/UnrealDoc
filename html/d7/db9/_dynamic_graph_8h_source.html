297 int FindEdge(int VA, int VB) const
298 {
299 int vMax = FMath::Max(VA, VB);
300 for (int EID : vertex_edges.Values(FMath::Min(VA, VB)))

302 //if (edge_has_v(EID, vMax)) // (slower option; does not use the fact that edges always have larger vertex as B)

311 EMeshResult RemoveEdge(int EID, bool bRemoveIsolatedVertices)
312 {
313 if (!edges_refcount.IsValid(EID))
314 {
315 check(false);
316 return EMeshResult::Failed_NotAnEdge;
317 }
318
319 FIndex2i ev(edges[EID].A, edges[EID].B);
320 vertex_edges.Remove(ev.A, EID);
321 vertex_edges.Remove(ev.B, EID);
322
323 edges_refcount.Decrement(EID);
324
325 // Decrement vertex refcounts. If any hit 1 and we got remove-isolated flag,
326 // we need to remove that vertex
327 for (int j = 0; j < 2; ++j)
328 {
329 int VID = ev[j];
330 vertices_refcount.Decrement(VID);
331 if (bRemoveIsolatedVertices && vertices_refcount.GetRefCount(VID) == 1)
332 {
333 vertices_refcount.Decrement(VID);
334 check(vertices_refcount.IsValid(VID) == false);
335 vertex_edges.Clear(VID);
336 }
337 }
338
339 updateTimeStamp(true);
340 return EMeshResult::Ok;
341 }

368 EMeshResult SplitEdge(int EAB, FEdgeSplitInfo& Split)
369 {
370 if (!IsEdge(EAB))
371 {
372 return EMeshResult::Failed_NotAnEdge;
373 }
374
375 // look up primary edge
376 int eab_i = 3 * EAB;
377 int A = edges[EAB].A, B = edges[EAB].B;
378 int GID = edges[EAB].Group;
379 int f = append_new_split_vertex(A, B);
380
381 // rewrite edge bc, create edge af
382 int eaf = EAB;
383 replace_edge_vertex(eaf, B, f);
384 vertex_edges.Remove(B, EAB);
385 vertex_edges.Insert(f, eaf);
386
387 // create new edge fb
388 int efb = add_edge(f, B, GID);
389
390 // update vertex refcounts
391 vertices_refcount.Increment(f, 2);
392
393 Split.VNew = f;
394 Split.ENewBN = efb;
395
396 updateTimeStamp(true);
397 return EMeshResult::Ok;
398 }

400 EMeshResult SplitEdgeWithExistingVertex(int EAB, int ExistingMidVert, FEdgeSplitInfo& Split)
401 {
402 if (!IsEdge(EAB))
403 {
404 return EMeshResult::Failed_NotAnEdge;
405 }
406
407 int f = ExistingMidVert;
408 if (!IsVertex(f))
409 {
410 return EMeshResult::Failed_NotAVertex;
411 }
412
413 // look up primary edge
414 int eab_i = 3 * EAB;
415 int A = edges[EAB].A, B = edges[EAB].B;
416 int GID = edges[EAB].Group;
417
418 int EAf = FindEdge(A, f);
419 int FIncr = 0; // track how much to increment the reference count of middle vertex, f
420 int BIncr = 0;
421 if (EAf != InvalidID)
422 {
423 RemoveEdge(EAB, false); // this will handle changes to refcounts, so no need to touch bIncr
424 edges[EAf].Group = GID;
425 }
426 else
427 {
428 // rewrite edge ab to create edge af
429 EAf = EAB;
430 replace_edge_vertex(EAf, B, f);
431 vertex_edges.Remove(B, EAB);
432 vertex_edges.Insert(f, EAf);
433 FIncr++;
434 BIncr--;
435 }
436
437 int EfB = FindEdge(f, B);
438 if (EfB == InvalidID)
439 {
440 EfB = add_edge(f, B, GID);
441 FIncr++;
442 BIncr++;
443 }
444 else
445 {
446 edges[EfB].Group = GID;
447 }
448
449 // update middle vertex refcounts
450 vertices_refcount.Increment(f, (unsigned short) FIncr);
451 if (BIncr > 0)
452 {
453 vertices_refcount.Increment(B, (unsigned short) BIncr);
454 }
455 if (BIncr < 0)
456 {
457 unsigned short BDecr = (unsigned short) (-BIncr);
458 vertices_refcount.Decrement(B, BDecr);
459 }
460
461 Split.VNew = f;
462 Split.ENewBN = EfB;
463
464 updateTimeStamp(true);
465 return EMeshResult::Ok;
466 }

484 EMeshResult CollapseEdge(int VKeep, int VRemove, FEdgeCollapseInfo& Collapse)
485 {
486 bool DiscardIsolatedVertices = true;
487
488 if (IsVertex(VKeep) == false || IsVertex(VRemove) == false)
489 {
490 return EMeshResult::Failed_NotAnEdge;
491 }
492
493 int B = VKeep; // renaming for sanity. We remove A and keep B
494 int A = VRemove;
495
496 int EAB = FindEdge(A, B);
497 if (EAB == InvalidID)
498 {
499 return EMeshResult::Failed_NotAnEdge;
500 }
501
502 // get rid of any edges that will be duplicates
503 bool done = false;
504 while (!done)
505 {
506 done = true;
507 for (int eax : vertex_edges.Values(A))
508 {
509 int o = edge_other_v(eax, A);
510 if (o != B && FindEdge(B, o) != InvalidID)
511 {
512 RemoveEdge(eax, DiscardIsolatedVertices);
513 done = false;
514 break;
515 }
516 }
517 }
518
519 vertex_edges.Remove(B, EAB);
520 for (int eax : vertex_edges.Values(A))
521 {
522 int o = edge_other_v(eax, A);
523 if (o == B)
524 {
525 continue; // discard this edge
526 }
527 replace_edge_vertex(eax, A, B);
528 vertices_refcount.Decrement(A);
529 vertex_edges.Insert(B, eax);
530 vertices_refcount.Increment(B);
531 }
532
533 edges_refcount.Decrement(EAB);
534 vertices_refcount.Decrement(B);
535 vertices_refcount.Decrement(A);
536 if (DiscardIsolatedVertices)
537 {
538 vertices_refcount.Decrement(A); // second Decrement discards isolated vtx
539 check(!IsVertex(A));
540 }
541
542 vertex_edges.Clear(A);
543
544 Collapse.VKept = VKeep;
545 Collapse.VRemoved = VRemove;
546 Collapse.ECollapsed = EAB;
547
548 updateTimeStamp(true);
549 return EMeshResult::Ok;
550 }

611 virtual bool CheckValidity(EValidityCheckFailMode FailMode = EValidityCheckFailMode::Check) const
612 {
613 bool is_ok = true;
614 TFunction<void(bool)> CheckOrFailF = [&](bool b)
615 {
616 is_ok = is_ok && b;
617 };
618 if (FailMode == EValidityCheckFailMode::Check)
619 {
620 CheckOrFailF = [&](bool b)
621 {
622 checkf(b, TEXT("FEdgeLoop::CheckValidity failed!"));
623 is_ok = is_ok && b;
624 };
625 }
626 else if (FailMode == EValidityCheckFailMode::Ensure)
627 {
628 CheckOrFailF = [&](bool b)
629 {
630 ensureMsgf(b, TEXT("FEdgeLoop::CheckValidity failed!"));
631 is_ok = is_ok && b;
632 };
633 }
634
635 // edge verts/tris must exist
636 for (int EID : EdgeIndices())
637 {
638 CheckOrFailF(IsEdge(EID));
639 CheckOrFailF(edges_refcount.GetRefCount(EID) == 1);
640 FIndex2i ev = GetEdgeV(EID);
641 CheckOrFailF(IsVertex(ev[0]));
642 CheckOrFailF(IsVertex(ev[1]));
643 CheckOrFailF(ev[0] < ev[1]);
644 }
645
646 // verify compact check
647 bool is_compact = vertices_refcount.IsDense();
648 if (is_compact)
649 {
650 for (int VID = 0; VID < VertexCount(); ++VID)
651 {
652 CheckOrFailF(vertices_refcount.IsValid(VID));
653 }
654 }
655
656 // vertex edges must exist and reference this vert
657 for (int VID : VertexIndices())
658 {
659 CheckOrFailF(IsVertex(VID));
660
661 //FVector3d V = GetVertex(VID);
662 //CheckOrFailF(double.IsNaN(V.SquaredLength()) == false);
663 //CheckOrFailF(double.IsInfinity(V.SquaredLength()) == false);
664
665 for (int edgeid : vertex_edges.Values(VID))
666 {
667 CheckOrFailF(IsEdge(edgeid));
668 CheckOrFailF(edge_has_v(edgeid, VID));
669
670 int otherV = edge_other_v(edgeid, VID);
671 int e2 = FindEdge(VID, otherV);
672 CheckOrFailF(e2 != InvalidID);
673 CheckOrFailF(e2 == edgeid);
674 e2 = FindEdge(otherV, VID);
675 CheckOrFailF(e2 != InvalidID);
676 CheckOrFailF(e2 == edgeid);
677 }
678
679 CheckOrFailF(vertices_refcount.GetRefCount(VID) == vertex_edges.GetCount(VID) + 1);
680 }
681
682 subclass_validity_checks(CheckOrFailF);
683
684 return is_ok;
685 }
