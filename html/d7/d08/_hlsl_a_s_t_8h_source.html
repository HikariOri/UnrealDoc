32 // Wrapper for identifier strings.
33 // Provides an indirection that can be used for efficient symbol renaming, etc.
34 class FIdentifier
35 {
36 private:
37 FIdentifier(const TCHAR* InString, int32 InLength)
38 : StringView(InString, InLength)
39 {
40 }
41 ~FIdentifier()
42 {
43 // We don't expect this to ever get destroyed, as we only allocate everything from the linear allocator
44 checkNoEntry();
45 }
46
47 public:
48
49 // NOTE: All memory is owned by the liner allocator and is released when it is destroyed. No need to delete it explicitly.
50 static FIdentifier* New(FLinearAllocator* Allocator, const TCHAR* InString, int32 InLength = -1)
51 {
52 if (InLength < 0)
53 {
54 InLength = FCString::Strlen(InString);
55 }
56 const TCHAR* DuplicatedString = Allocator->Strdup(InString, InLength);
57 void* IdentifierMemory = Allocator->Alloc(sizeof(FIdentifier), alignof(FIdentifier));
58 return new(IdentifierMemory) FIdentifier(DuplicatedString, InLength);
59 }
60
61 static FIdentifier* New(FLinearAllocator* Allocator, FStringView InString)
62 {
63 return FIdentifier::New(Allocator, InString.GetData(), InString.Len());
64 }
65
66 FStringView ToStringView() const
67 {
68 return StringView;
69 };
70
71 FString ToString() const
72 {
73 return FString(StringView);
74 };
75
76 const TCHAR* GetData() const
77 {
78 return StringView.GetData();
79 }
80
81 int32 Len() const
82 {
83 return StringView.Len();
84 }
85
86 bool Equals(FStringView Other) const
87 {
88 return Other.Equals(StringView, ESearchCase::CaseSensitive);
89 }
90
91 void Rename(FLinearAllocator* Allocator, FStringView InString)
92 {
93 int32 NewLength = InString.Len();
94 const TCHAR* NewData = Allocator->Strdup(InString.GetData(), InString.Len());
95 StringView = FStringView(NewData, NewLength);
96 }
97
98 private:
99 FStringView StringView;
100 };

151 inline FASTWriter& operator << (uint32 N)
152 {
153 union
154 {
155 uint32 U;
156 int32 I;
157 } Alias;
158 Alias.U = N;
159 if (Alias.I < 0)
160 {
161 (*this) << *FString::Printf(TEXT("%uu"), N);
162 }
163 else
164 {
165 (*this) << *FString::Printf(TEXT("%u"), N);
166 }
167 return *this;
168 }
169
170 inline FASTWriter& operator << (float F)
171 {
172 if (F == 0)
173 {
174 (*this) << TEXT("0.0");
175 }
176 else
177 {
178 float Abs = FMath::Abs(F);
179 if (Abs <= 1e-6 || Abs >= 1e6)
180 {
181 (*this) << *FString::Printf(TEXT("%g"), F);
182 }
183 else
184 {
185 (*this) << *FString::Printf(TEXT("%f"), F);
186 }
187 }
188 return *this;
189 }

362 inline EOperators TokenToASTOperator(EHlslToken Token)
363 {
364 switch (Token)
365 {
366 case EHlslToken::Equal:
367 return AST::EOperators::Assign;
368
369 case EHlslToken::PlusEqual:
370 return AST::EOperators::AddAssign;
371
372 case EHlslToken::MinusEqual:
373 return AST::EOperators::SubAssign;
374
375 case EHlslToken::TimesEqual:
376 return AST::EOperators::MulAssign;
377
378 case EHlslToken::DivEqual:
379 return AST::EOperators::DivAssign;
380
381 case EHlslToken::ModEqual:
382 return AST::EOperators::ModAssign;
383
384 case EHlslToken::GreaterGreaterEqual:
385 return AST::EOperators::RSAssign;
386
387 case EHlslToken::LowerLowerEqual:
388 return AST::EOperators::LSAssign;
389
390 case EHlslToken::AndEqual:
391 return AST::EOperators::AndAssign;
392
393 case EHlslToken::OrEqual:
394 return AST::EOperators::OrAssign;
395
396 case EHlslToken::XorEqual:
397 return AST::EOperators::XorAssign;
398
399 case EHlslToken::Question:
400 return AST::EOperators::Conditional;
401
402 case EHlslToken::OrOr:
403 return AST::EOperators::LogicOr;
404
405 case EHlslToken::AndAnd:
406 return AST::EOperators::LogicAnd;
407
408 case EHlslToken::Or:
409 return AST::EOperators::BitOr;
410
411 case EHlslToken::Xor:
412 return AST::EOperators::BitXor;
413
414 case EHlslToken::And:
415 return AST::EOperators::BitAnd;
416
417 case EHlslToken::EqualEqual:
418 return AST::EOperators::Equal;

420 case EHlslToken::NotEqual:
421 return AST::EOperators::NEqual;
422
423 case EHlslToken::Lower:
424 return AST::EOperators::Less;
425
426 case EHlslToken::Greater:
427 return AST::EOperators::Greater;
428
429 case EHlslToken::LowerEqual:
430 return AST::EOperators::LEqual;
431
432 case EHlslToken::GreaterEqual:
433 return AST::EOperators::GEqual;
434
435 case EHlslToken::LowerLower:
436 return AST::EOperators::LShift;
437
438 case EHlslToken::GreaterGreater:
439 return AST::EOperators::RShift;
