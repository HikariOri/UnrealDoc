104static bool D3D12RHI_IsRenderDocPresent(ID3D12Device* Device)
105{
106 IID RenderDocID;
107 if (SUCCEEDED(IIDFromString(L"{A7AA6116-9C8D-4BBA-9083-B4D816B71B78}", &RenderDocID)))
108 {
109 TRefCountPtr<IUnknown> RenderDoc;
110 if (SUCCEEDED(Device->QueryInterface(RenderDocID, (void**)RenderDoc.GetInitReference())))
111 {
112 return true;
113 }
114 }
115
116 return false;
117}

735class FScopedResourceBarrier
736{
737private:
738 FD3D12ContextCommon& Context;
739 const FD3D12Resource* pResource;
740 ED3D12Access BeforeD3D12Access;
741 const ED3D12Access DesiredD3D12Access;
742 const uint32 Subresource;
743 bool bRestoreState;
744
745public:
746 FScopedResourceBarrier(
747 FD3D12ContextCommon& InContext,
748 const FD3D12Resource* InResource,
749 ED3D12Access InBeforeD3D12Access,
750 ED3D12Access InDesiredD3D12Access,
751 uint32 InSubresource)
752 : Context(InContext)
753 , pResource(InResource)
754 , BeforeD3D12Access(InBeforeD3D12Access)
755 , DesiredD3D12Access(InDesiredD3D12Access)
756 , Subresource (InSubresource)
757 , bRestoreState(false)
758 {
759 check(IsValidAccess(DesiredD3D12Access));
760 check(DesiredD3D12Access != ED3D12Access::Unknown);
761
762 const bool bBeforED3D12AccessIsKnown = BeforeD3D12Access != ED3D12Access::Unknown;
763 if (!pResource->RequiresResourceStateTracking() || bBeforED3D12AccessIsKnown)
764 {
765 BeforeD3D12Access = bBeforED3D12AccessIsKnown ? BeforeD3D12Access : pResource->GetDefaultAccess();
766 check(IsValidAccess(BeforeD3D12Access));
767 check(BeforeD3D12Access != ED3D12Access::Unknown);
768
769 Context.AddBarrier(pResource, BeforeD3D12Access, DesiredD3D12Access, Subresource);
770 bRestoreState = true;
771 }
772 }
773
774 FScopedResourceBarrier(
775 FD3D12ContextCommon& InContext,
776 const FD3D12Resource* InResource,
777 ED3D12Access InDesiredD3D12Access,
778 uint32 InSubresource)
779 : FScopedResourceBarrier(
780 InContext,
781 InResource,
782 ED3D12Access::Unknown,
783 InDesiredD3D12Access,
784 InSubresource)
785 {}
786
787 ~FScopedResourceBarrier()
788 {
789 if (bRestoreState)
790 {
791 Context.AddBarrier(pResource, DesiredD3D12Access, BeforeD3D12Access, Subresource);
792 }
793 }
