49 int32 getCoeffIntraNoAsm(int32& run, int32& last, M4Bitstream& bs)
50 {
51 uint32 bits = bs.show(12);
52 M4CHECK(bits >= 8);
53
54 M4_VLC vlc = getIntraVlcTab(bits);
55 M4CHECK(vlc != VLC_ERROR);
56
57 bs.skip(VLC_LEN(vlc));
58
59 int32 level;
60 uint32 code = VLC_CODE(vlc);
61 if (code != VLC_ESCAPE)
62 {
63 run = (code >> 8) & 255;
64 level = code & 255;
65 last = (code >> 16) & 1;
66 int32 ret = bs.getBit() ? -level : level;
67 return ret;
68 }
69
70 uint32 escMode = bs.show(2);
71 if (escMode < 3)
72 {
73 bs.skip((escMode == 2) ? 2 : 1);
74
75 bits = bs.show(12);
76 M4CHECK(bits >= 8);
77
78 vlc = getIntraVlcTab(bits);
79 M4CHECK(vlc != VLC_ERROR);
80
81 bs.skip(VLC_LEN(vlc));
82
83 code = VLC_CODE(vlc);
84 run = (code >> 8) & 255;
85 level = code & 255;
86 last = (code >> 16) & 1;
87
88 if (escMode < 2) // first escape mode, level is offset
89 {
90 level += mMaxLevel[last][run]; // need to add back the max level
91 }
92 else if (escMode == 2) // second escape mode, run is offset
93 {
94 run += mMaxRun[last][level] + 1;
95 }
96 else

99 }
100

102 return ret;

106 bs.skip(2);

108 run = (int32) bs.getBits(6);

112 bs.skip(1); // marker
113
