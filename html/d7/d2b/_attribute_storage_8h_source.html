354 enum class EAttributeStorageResult : uint64
355 {
356 None = 0x0,
357 //Success result.
358 Operation_Success = 0x1,
359
360
361 //Operation error results from here.
362
363 //The type of the value was not matching the existing type. We cannot change the type of an existing attribute.
364 Operation_Error_WrongType = (0x1 << 20),
365 //The size of the value is different from the existing size. We cannot have a different size.
366 Operation_Error_WrongSize = (0x1 << 21),
367 //The AttributeAllocationTable has an attribute whose offset is not valid in the storage.
368 Operation_Error_AttributeAllocationCorrupted = (0x1 << 22),
369 //We cannot find the specified key.
370 Operation_Error_CannotFoundKey = (0x1 << 23),
371 //There was an error when removing an attribute from the AttributeAllocationTable. The TArray remove has failed.
372 Operation_Error_CannotRemoveAttribute = (0x1 << 24),
373 //We try to override an attribute but the specified options do not allow override.
374 Operation_Error_CannotOverrideAttribute = (0x1 << 25),
375 //The storage is invalid (nullptr).
376 Operation_Error_InvalidStorage = (0x1 << 26),
377 //Cannot get a valid value data pointer.
378 Operation_Error_InvalidMultiSizeValueData = (0x1 << 27),
379 };
