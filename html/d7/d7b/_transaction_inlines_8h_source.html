37UE_AUTORTFM_FORCEINLINE bool FTransaction::IsOnStack(const void* LogicalAddress) const
38{
39 if (StackRange.Contains(LogicalAddress))
40 {
41 return true;
42 }
43
44#if AUTORTFM_CHECK_ASAN_FAKE_STACKS
45 if (void* FakeStack = __asan_get_current_fake_stack())
46 {
47 void* Beg = nullptr;
48 void* End = nullptr;
49 void* RealAddress = __asan_addr_is_in_fake_stack(FakeStack, const_cast<void*>(LogicalAddress), &Beg, &End);
50 return RealAddress && StackRange.Contains(RealAddress);
51 }
52#endif // AUTORTFM_CHECK_ASAN_FAKE_STACKS
53
54 return false;
55}

57UE_AUTORTFM_FORCEINLINE bool FTransaction::ShouldRecordWrite(void* LogicalAddress) const
58{
59 // We cannot record writes to stack memory used within the transaction, as
60 // undoing the writes may corrupt stack memory that has been unwound or
61 // is now being used for a different variable from the one the write was
62 // made.
63 return !IsOnStack(LogicalAddress);
64}

66AUTORTFM_NO_ASAN UE_AUTORTFM_FORCEINLINE void FTransaction::RecordWrite(
67 void* LogicalAddress, size_t Size, EWriteFlags Flags /* = EWriteFlags::Default */)
68{
69 if (AUTORTFM_UNLIKELY(0 == Size))
70 {
71 return;
72 }
73
74 if (!ShouldRecordWrite(LogicalAddress))
75 {
76 return;
77 }
78
79 if (Size <= FHitSet::MaxSize)
80 {
81 FHitSetEntry HitSetEntry{};
82 HitSetEntry.Address = reinterpret_cast<uintptr_t>(LogicalAddress);
83 HitSetEntry.Size = static_cast<uint16_t>(Size);
84 HitSetEntry.Flags = static_cast<uintptr_t>(Flags);
85
86 if (HitSet.FindOrTryInsert(HitSetEntry))
87 {
88 return;
89 }
90 }
91
92 if (NewMemoryTracker.Contains(LogicalAddress, Size))
93 {
94 return;
95 }
96
97 FWriteLogEntry Entry{.LogicalAddress = static_cast<std::byte*>(LogicalAddress),
98 .Data = static_cast<std::byte*>(LogicalAddress),
99 .Size = Size,
100 .Flags = Flags};
101
102 AutoRTFM::TScopedGuard<bool> RecursionGuard(bIsInAllocateFn, true);
103
104 switch (Entry.Size)
105 {
106 case 8:
107 WriteLog.PushSmall<8>(Entry.LogicalAddress, Entry.Data, Entry.Flags);
108 break;
109 case 4:
110 WriteLog.PushSmall<4>(Entry.LogicalAddress, Entry.Data, Entry.Flags);
111 break;
112 case 2:
113 WriteLog.PushSmall<2>(Entry.LogicalAddress, Entry.Data, Entry.Flags);
114 break;
115 default:
116 WriteLog.Push(Entry);
117 break;
118 }
119
120#if AUTORTFM_SANITIZER
121 if ((Flags & EWriteFlags::NoSanitize) != EWriteFlags::NoSanitize)
122 {
123 Sanitizer::ClosedWrite(ID, LogicalAddress, Size, __builtin_return_address(0));
124 }
125#endif
126}

128template <unsigned SIZE>
129AUTORTFM_NO_ASAN UE_AUTORTFM_FORCEINLINE void FTransaction::RecordWrite(void* LogicalAddress)
130{
131 static_assert(SIZE <= 8);
132 if (!ShouldRecordWrite(LogicalAddress))
133 {
134 return;
135 }
136
137 FHitSetEntry Entry{};
138 Entry.Address = reinterpret_cast<uintptr_t>(LogicalAddress);
139 Entry.Size = static_cast<uint16_t>(SIZE);
140
141 switch (HitSet.FindOrTryInsertNoResize(Entry))
142 {
143 case FHitSet::EInsertResult::Exists:
144 return;
145 case FHitSet::EInsertResult::Inserted:
146 AUTORTFM_MUST_TAIL return FTransaction::RecordWriteInsertedSlow<SIZE>(LogicalAddress);
147 case FHitSet::EInsertResult::NotInserted:
148 AUTORTFM_MUST_TAIL return FTransaction::RecordWriteNotInsertedSlow<SIZE>(LogicalAddress);
149 }
150}

167template <unsigned SIZE>
168AUTORTFM_NO_ASAN UE_AUTORTFM_FORCENOINLINE void FTransaction::RecordWriteInsertedSlow(void* LogicalAddress)
169{
170 if (NewMemoryTracker.Contains(LogicalAddress, SIZE))
171 {
172 return;
173 }
174
175 AutoRTFM::TScopedGuard<bool> RecursionGuard(bIsInAllocateFn, true);
176
177 WriteLog.PushSmall<SIZE>(static_cast<std::byte*>(LogicalAddress), static_cast<std::byte*>(LogicalAddress));
178
179 AUTORTFM_SANITIZER_ONLY(Sanitizer::ClosedWrite(ID, LogicalAddress, SIZE, __builtin_return_address(0)));
180}

194UE_AUTORTFM_FORCEINLINE void FTransaction::DidFree(void* LogicalAddress)
195{
196 AUTORTFM_ASSERT(bTrackAllocationLocations);
197
198 // Checking if one byte is in the interval map is enough to ascertain if it
199 // is new memory and we should be worried.
200 if (!bIsInAllocateFn)
201 {
202 if (NewMemoryTracker.Contains(LogicalAddress, 1))
203 {
204 AutoRTFM::TScopedGuard<bool> RecursionGuard(bIsInAllocateFn, true);
205 ForTheRuntime::ReportAutoRTFMHazard(ForTheRuntime::EHazardType::OpenFree);
206 AUTORTFM_REPORT_ERROR("Transactional memory was freed from the open.");
207 }
208 }
