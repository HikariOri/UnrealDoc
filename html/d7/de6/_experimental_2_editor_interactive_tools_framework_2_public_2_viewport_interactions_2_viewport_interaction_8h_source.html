33/*
34 * Binding complexity is chosen so that "tool" behaviors win out over viewport interactions on the same complexity footing.
35 * Tool priority in this diagram is assumed to have a grandfathered single point of complexity.
36 *
37 * These statements should be true:
38 * - A tool beats an unmodified click.
39 * - A modified click can beat a tool.
40 * - An unmodified drag loses to a click until it is confirmed.
41 * - A modified, unconfirmed drag can beat an unmodified click.
42 *
43 * Offsetting clicks by 1/3rd and unconfirmed drags by another 1/3rd from that allows this:
44 *
45 * |-- Tool
46 * |-- Click
47 * |--|-- Modified Click
48 * |-- Drag (unconfirmed)
49 * |--|-- Drag confirmed (or modified unconfirmed drag)
50 * |--|--|-- Modified drag confirmed
51 */
52static constexpr int VIEWPORT_INTERACTION_PARTIAL_STEP = static_cast<int>(FInputCapturePriority::BINDING_COMPLEXITY_PRIORITY_STEP * (1.f/3.f));
53static_assert(VIEWPORT_INTERACTION_PARTIAL_STEP >= 1);
54
55static constexpr int CLICK_PRIORITY = FInputCapturePriority::DEFAULT_TOOL_PRIORITY + FInputCapturePriority::BINDING_COMPLEXITY_PRIORITY_STEP + VIEWPORT_INTERACTION_PARTIAL_STEP;
56static constexpr int DRAG_PRIORITY = FInputCapturePriority::DEFAULT_TOOL_PRIORITY + FInputCapturePriority::BINDING_COMPLEXITY_PRIORITY_STEP + VIEWPORT_INTERACTION_PARTIAL_STEP * 2;
57static constexpr int DRAG_CONFIRMED_PRIORITY = DRAG_PRIORITY - FInputCapturePriority::BINDING_COMPLEXITY_PRIORITY_STEP;
