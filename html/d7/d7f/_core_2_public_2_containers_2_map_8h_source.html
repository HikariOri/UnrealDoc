170 {
171 FScriptMapLayout Result;

306 [[nodiscard]] int32 FindPairIndex(const void* Key, const FScriptMapLayout& MapLayout, TFunctionRef<uint32(const void*)> GetKeyHash, TFunctionRef<bool(const void*, const void*)> KeyEqualityFn) const
307 {
308 if (Pairs.Num())
309 {
310 // !unsafe! 'Pairs' is mostly treated as a set of TPair<Key, Value>, so anything in
311 // FScriptSet could assume that Key is actually a TPair<Key, Value>, we can hide this
312 // complexity from our caller, at least (so their GetKeyHash/EqualityFn is unaware):
313 return Pairs.FindIndex(
314 Key,
315 MapLayout.SetLayout,
316 GetKeyHash, // We 'know' that the implementation of Find doesn't call GetKeyHash on anything except Key
317 [KeyEqualityFn](const void* InKey, const void* InPair )
318 {
319 return KeyEqualityFn(InKey, (uint8*)InPair);
320 }
321 );
322 }
323
324 return INDEX_NONE;
325 }

328 [[nodiscard]] uint8* FindValue(const void* Key, const FScriptMapLayout& MapLayout, TFunctionRef<uint32(const void*)> GetKeyHash, TFunctionRef<bool(const void*, const void*)> KeyEqualityFn)
329 {
330 int32 FoundIndex = FindPairIndex(Key, MapLayout, GetKeyHash, KeyEqualityFn);
331 if (FoundIndex != INDEX_NONE)
332 {
333 uint8* Result = (uint8*)GetData(FoundIndex, MapLayout) + MapLayout.ValueOffset;
334 return Result;
335 }
336
337 return nullptr;
338 }

345 TFunctionRef<uint32(const void*)> GetKeyHash,

381 TFunctionRef<uint32(const void*)> GetKeyHash,
382 TFunctionRef<bool(const void*, const void*)> KeyEqualityFn,

402 // This function isn't intended to be called, just to be compiled to validate the correctness of the type.
403 static void CheckConstraints()
404 {
405 typedef TScriptMap ScriptType;
406 typedef TMap<int32, int8> RealType;
407
408 // Check that the class footprint is the same
409 static_assert(sizeof(ScriptType) == sizeof(RealType), "TScriptMap's size doesn't match TMap");
410 static_assert(alignof(ScriptType) == alignof(RealType), "TScriptMap's alignment doesn't match TMap");
411
412 // Check member sizes
413 static_assert(sizeof(DeclVal<ScriptType>().Pairs) == sizeof(DeclVal<RealType>().Pairs), "TScriptMap's Pairs member size does not match TMap's");
414
415 // Check member offsets
416 static_assert(STRUCT_OFFSET(ScriptType, Pairs) == STRUCT_OFFSET(RealType, Pairs), "TScriptMap's Pairs member offset does not match TMap's");
417 }
