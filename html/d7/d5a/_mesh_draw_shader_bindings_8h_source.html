37 bool operator==(const FMeshDrawShaderBindingsLayout& Rhs) const
38 {
39 // Since 4.25, FShader (the owner of this memory) is no longer shared across the shadermaps and gets deleted at the same time as the owning shadermap.
40 // To prevent crashes when a singe mesh draw command is shared across multiple MICs with compatible shaders, consider shader bindings belonging to different FShaders different.
41 return &ParameterMapInfo == &Rhs.ParameterMapInfo;
42 }
43
44 inline uint32 GetLooseDataSizeBytes() const
45 {
46 uint32 DataSize = 0;
47 for (const FShaderLooseParameterBufferInfo& Info : ParameterMapInfo.LooseParameterBuffers)
48 {
49 DataSize += Info.Size;
50 }
51 return DataSize;
52 }
53
54 inline uint32 GetDataSizeBytes() const
55 {
56 uint32 DataSize = sizeof(void*) *
57 (ParameterMapInfo.UniformBuffers.Num()
58 + ParameterMapInfo.TextureSamplers.Num()
59 + ParameterMapInfo.SRVs.Num());
60
61 // Allocate a bit for each SRV tracking whether it is a FRHITexture* or FRHIShaderResourceView*
62 DataSize += FMath::DivideAndRoundUp(ParameterMapInfo.SRVs.Num(), 8);
63
64 DataSize += GetLooseDataSizeBytes();
65
66 // Align to pointer size so subsequent packed shader bindings will have their pointers aligned
67 return Align(DataSize, sizeof(void*));
68 }
69
70protected:
71
72 // Note: pointers first in layout, so they stay aligned
73 inline uint32 GetUniformBufferOffset() const
74 {
75 return 0;
76 }
77
78 inline uint32 GetSamplerOffset() const
79 {
80 return ParameterMapInfo.UniformBuffers.Num() * sizeof(FRHIUniformBuffer*);
81 }
82
83 inline uint32 GetSRVOffset() const
84 {
85 return GetSamplerOffset()
86 + ParameterMapInfo.TextureSamplers.Num() * sizeof(FRHISamplerState*);
87 }
88
89 inline uint32 GetSRVTypeOffset() const
90 {
91 return GetSRVOffset()
92 + ParameterMapInfo.SRVs.Num() * sizeof(FRHIShaderResourceView*);
93 }
94
95 inline uint32 GetLooseDataOffset() const
96 {
97 return GetSRVTypeOffset()
98 + FMath::DivideAndRoundUp(ParameterMapInfo.SRVs.Num(), 8);
99 }
100
101 friend class FMeshDrawShaderBindings;
102};

192 template<class ParameterType>
193 void Add(FShaderParameter Parameter, const ParameterType& Value)
194 {
195 static_assert(!TIsUECoreVariant<ParameterType, double>::Value, "FMeshDrawSingleShaderBindings cannot Add double core variants! Switch to float variant.");
196 static_assert(!TIsUECoreVariant<typename std::remove_pointer<typename TDecay<ParameterType>::Type>::type, double>::Value, "FMeshDrawSingleShaderBindings cannot Add double core variants! Switch to float variant.");
197 static_assert(!TIsPointer<ParameterType>::Value, "Passing by pointer is not valid.");
198 checkfSlow(Parameter.IsInitialized(), TEXT("Parameter was not serialized"));
199
200 if (Parameter.IsBound())
201 {
202 bool bFoundParameter = false;
203 uint8* LooseDataOffset = GetLooseDataStart();
204
205 TArrayView<const FShaderLooseParameterBufferInfo> LooseParameterBuffers(ParameterMapInfo.LooseParameterBuffers);
206 for (int32 LooseBufferIndex = 0; LooseBufferIndex < LooseParameterBuffers.Num(); LooseBufferIndex++)
207 {
208 const FShaderLooseParameterBufferInfo& LooseParameterBuffer = LooseParameterBuffers[LooseBufferIndex];
209
210 if (LooseParameterBuffer.BaseIndex == Parameter.GetBufferIndex())
211 {
212 TArrayView<const FShaderLooseParameterInfo> Parameters(LooseParameterBuffer.Parameters);
213 for (int32 LooseParameterIndex = 0; LooseParameterIndex < Parameters.Num(); LooseParameterIndex++)
214 {
215 const FShaderLooseParameterInfo LooseParameter = Parameters[LooseParameterIndex];
216
217 if (Parameter.GetBaseIndex() == LooseParameter.BaseIndex)
218 {
219 checkSlow(Parameter.GetNumBytes() == LooseParameter.Size);
220 ensureMsgf(sizeof(ParameterType) == Parameter.GetNumBytes(), TEXT("Size mismatch between the value being set and the size the shader requires. Setting %" SIZE_T_FMT " bytes on loose parameter at BaseIndex %u, which requires %u bytes. This can cause GPU hangs, depending on usage."), sizeof(ParameterType), Parameter.GetBaseIndex(), Parameter.GetNumBytes());
221 const int32 NumBytesToSet = FMath::Min<int32>(sizeof(ParameterType), Parameter.GetNumBytes());
222 FMemory::Memcpy(LooseDataOffset, &Value, NumBytesToSet);
223 const int32 NumBytesToClear = FMath::Min<int32>(0, Parameter.GetNumBytes() - NumBytesToSet);
224 FMemory::Memset(LooseDataOffset + NumBytesToSet, 0x00, NumBytesToClear);
225 bFoundParameter = true;
226 break;
227 }
228
229 LooseDataOffset += LooseParameter.Size;
230 }
231 break;
232 }
233
234 LooseDataOffset += LooseParameterBuffer.Size;
235 }
236
237 checkfSlow(bFoundParameter, TEXT("Attempted to set loose parameter at BaseIndex %u, Size %u which was never in the shader's parameter map."), Parameter.GetBaseIndex(), Parameter.GetNumBytes());
238 }

274 template<typename ElementType>
275 inline int FindSortedArrayBaseIndex(const TConstArrayView<ElementType>& Array, uint32 BaseIndex)
276 {
277 int Index = 0;
278 int Size = Array.Num();
279
280 //start with binary search for larger lists
281 while (Size > 8)
282 {
283 const int LeftoverSize = Size % 2;
284 Size = Size / 2;
285
286 const int CheckIndex = Index + Size;
287 const int IndexIfLess = CheckIndex + LeftoverSize;
288
289 Index = Array[CheckIndex].BaseIndex < BaseIndex ? IndexIfLess : Index;
290 }
291
292 //small size array optimization
293 const int ArrayEnd = FMath::Min(Index + Size + 1, Array.Num());
294 while (Index < ArrayEnd)
295 {
296 if (Array[Index].BaseIndex == BaseIndex)
297 {
298 return Index;
299 }
300 Index++;
301 }
302
303 return -1;
304 }

306 inline void WriteBindingUniformBuffer(const FRHIUniformBuffer* Value, uint32 BaseIndex)
307 {
308 int32 FoundIndex = FindSortedArrayBaseIndex(MakeArrayView(ParameterMapInfo.UniformBuffers), BaseIndex);
309 if (FoundIndex >= 0)

328 inline void WriteBindingSRV(FRHIShaderResourceView* Value, uint32 BaseIndex)
329 {

331 if (FoundIndex >= 0)
