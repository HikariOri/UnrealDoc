431 bool IsSameAs(const TDynamicMeshTriangleAttribute<AttribValueType, AttribDimension>& Other, bool bIgnoreDataLayout) const
432 {
433 if (!bIgnoreDataLayout)
434 {
435 if (AttribValues.Num() != Other.AttribValues.Num())
436 {
437 return false;
438 }
439
440 for (int Idx = 0, NumValues = AttribValues.Num(); Idx < NumValues; Idx++)
441 {
442 if (AttribValues[Idx] != Other.AttribValues[Idx])
443 {
444 return false;
445 }
446 }
447 }
448 else
449 {
450 FRefCountVector::IndexIterator ItTid = ParentMesh->GetTrianglesRefCounts().BeginIndices();
451 const FRefCountVector::IndexIterator ItTidEnd = ParentMesh->GetTrianglesRefCounts().EndIndices();
452 FRefCountVector::IndexIterator ItTidOther = Other.ParentMesh->GetTrianglesRefCounts().BeginIndices();
453 const FRefCountVector::IndexIterator ItTidEndOther = Other.ParentMesh->GetTrianglesRefCounts().EndIndices();
454
455 while (ItTid != ItTidEnd && ItTidOther != ItTidEndOther)
456 {
457 for (int32 i = 0; i < AttribDimension; ++i)
458 {
459 const AttribValueType AttribValue = AttribValues[*ItTid * AttribDimension + i];
460 const AttribValueType AttribValueOther = Other.AttribValues[*ItTidOther * AttribDimension + i];
461 if (AttribValue != AttribValueOther)
462 {
463 // Triangle attribute value is not the same.
464 return false;
465 }
466 }
467 ++ItTid;
468 ++ItTidOther;
469 }
470
471 if (ItTid != ItTidEnd || ItTidOther != ItTidEndOther)
472 {
473 // Number of triangle attribute values is not the same.
474 return false;
475 }
476 }
477
478 return true;
479 }

597template<typename AttribValueType, int AttribDimension>
598void FDynamicMeshTriangleAttributeChange<AttribValueType, AttribDimension>::StoreAllFinalTriangles(const FDynamicMeshAttributeBase* Attribute, const TArray<int>& TriangleIDs)
599{
600 const TDynamicMeshTriangleAttribute<AttribValueType, AttribDimension>* AttribCast = static_cast<const TDynamicMeshTriangleAttribute<AttribValueType, AttribDimension>*>(Attribute);
601 NewTriangleAttributes.Reserve(NewTriangleAttributes.Num() + TriangleIDs.Num());
602 for (int TriangleID : TriangleIDs)

610template<typename AttribValueType, int AttribDimension>
611bool FDynamicMeshTriangleAttributeChange<AttribValueType, AttribDimension>::Apply(FDynamicMeshAttributeBase* Attribute, bool bRevert) const
612{
613 TDynamicMeshTriangleAttribute<AttribValueType, AttribDimension>* AttribCast = static_cast<TDynamicMeshTriangleAttribute<AttribValueType, AttribDimension>*>(Attribute);
614 const TArray<FChangeTriangleAttribute> *Changes = bRevert ? &OldTriangleAttributes : &NewTriangleAttributes;
615 for (const FChangeTriangleAttribute& Change : *Changes)
616 {
617 bool bIsTriangle = AttribCast->GetParentMesh()->IsTriangle(Change.TriangleID);
618 checkSlow(bIsTriangle);
619 if (bIsTriangle)
620 {
621 AttribCast->SetValue(Change.TriangleID, Change.Data);
622 }
623 }
624 return true;
