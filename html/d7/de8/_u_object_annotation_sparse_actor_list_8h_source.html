39template<typename InAllocatorType>
40void FUObjectAnnotationSparseActorList<InAllocatorType>::AddOwner(const UObject* Object, const AActor* Owner)
41{
42 if (!Object || !Owner)
43 {
44 return;
45 }
46
47 UE::TRWScopeLock AnnotationMapLock(this->AnnotationMapCritical, SLT_Write);
48
49 if (this->AnnotationMap.IsEmpty() && ReverseAnnotationMap.IsEmpty() && !this->bRegistered)
50 {
51 GUObjectArray.AddUObjectDeleteListener(this);
52 this->bRegistered = true;
53 }
54
55 FAnnotationSparseActorList<InAllocatorType>& Entry = this->AnnotationMap.FindOrAdd(Object);
56 Entry.OwnerList.AddUnique(Owner);
57 ReverseAnnotationMap.FindOrAdd(Owner).AddUnique(Object);
58}

60template<typename InAllocatorType>
61void FUObjectAnnotationSparseActorList<InAllocatorType>::RemoveOwner(const UObject* Object, const AActor* Owner)
62{
63 if (!Object || !Owner)
64 {
65 return;
66 }
67
68 UE::TRWScopeLock AnnotationMapLock(this->AnnotationMapCritical, SLT_Write);
69 if (FAnnotationSparseActorList<InAllocatorType>* Entry = this->AnnotationMap.Find(Object))
70 {
71 Entry->OwnerList.Remove(Owner);
72
73 // Remove the annotation entirely if we removed the last owner
74 if (Entry->IsDefault())
75 {
76 this->AnnotationMap.Remove(Object);
77 }
78 }
79
80 if (TArray<const UObject*>* Objects = ReverseAnnotationMap.Find(Owner))
81 {
82 Objects->Remove(Object);
83 if (Objects->IsEmpty())
84 {
85 ReverseAnnotationMap.Remove(Owner);
86 }
87 }
88
89 if (this->AnnotationMap.IsEmpty() && ReverseAnnotationMap.IsEmpty() && this->bRegistered)
90 {
91 GUObjectArray.RemoveUObjectDeleteListener(this);
92 this->bRegistered = false;
93 }
94}

96template<typename InAllocatorType>
97void FUObjectAnnotationSparseActorList<InAllocatorType>::NotifyUObjectDeleted(const UObjectBase* Object, int32 Index)
98{
99 // UE::TRWScopeLock AnnotationMapLock
100 {
101 UE::TRWScopeLock AnnotationMapLock(this->AnnotationMapCritical, SLT_Write);
102
103 // Object being deleted is potentially a owner
104 if (const AActor* Owner = Cast<const AActor>(Object))
105 {
106 if (TArray<const UObject*>* Objects = ReverseAnnotationMap.Find(Owner))
107 {
108 for (const UObject* TempObject : *Objects)
109 {
110 if (FAnnotationSparseActorList<InAllocatorType>* Entry = this->AnnotationMap.Find(TempObject))
111 {
112 Entry->OwnerList.Remove(Owner);
113 if (Entry->IsDefault())
114 {
115 this->AnnotationMap.Remove(TempObject);
116 }
117 }
118 }
119 ReverseAnnotationMap.Remove(Owner);
120 }
121 }
122 // Object being deleted is a UObject, find the owners and remove them.
123 else
124 {
125 const UObject* TempObject = Cast<const UObject>(Object);
126 if (FAnnotationSparseActorList<InAllocatorType>* Owners = this->AnnotationMap.Find(TempObject))
127 {
128 for (TWeakObjectPtr<const AActor> ObjectOwner : Owners->OwnerList)
129 {
130 if (ObjectOwner.IsValid())
131 {
132 if (TArray<const UObject*>* Objects = ReverseAnnotationMap.Find(ObjectOwner.Get()))
133 {
134 Objects->Remove(TempObject);
135 if (Objects->IsEmpty())
136 {
137 ReverseAnnotationMap.Remove(ObjectOwner.Get());
138 }
139 }
140 }
141 }
142 this->AnnotationMap.Remove(TempObject);
143 }
144 }
145
146 if (this->AnnotationMap.IsEmpty() && ReverseAnnotationMap.IsEmpty() && this->bRegistered)
147 {
148 GUObjectArray.RemoveUObjectDeleteListener(this);
149 this->bRegistered = false;
150 }
151 }
152
153 // Needs to be at the end because in non shipping builds it will check if we cleaned up our objects.
154 // Make sure you have released the lock before this point or you will deadlock.
155 Base::NotifyUObjectDeleted(Object, Index);
156}
