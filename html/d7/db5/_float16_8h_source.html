158inline FFloat16 FFloat16::GetClampedNonNegativeAndFinite() const
159{
160 FFloat16 ReturnValue;

163 ReturnValue.Encoded = Encoded;

174inline FFloat16 FFloat16::GetClampedFinite() const
175{
176 FFloat16 ReturnValue;
177
178 if ( (Encoded&0x7c00) == 0x7c00 )
