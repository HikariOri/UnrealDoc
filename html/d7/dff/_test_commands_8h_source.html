182template <typename ResultType, typename ResultCommandType = void>
183class TAsyncExecute : public IAutomationLatentCommand
184{
185public:
186 template<typename... ResultCommandArgs>
187 TAsyncExecute(
188 FAutomationTestBase& TestRunner,
189 TFunction<TAsyncResult<ResultType>()> AsyncAction,
190 TOptional<FTimespan> AsyncActionTimeout = CQTest::DefaultTimeout,
191 const TCHAR* Description = nullptr,
192 ResultCommandArgs&&... Args
193 )
194 {
195 RunSequence.Append(MakeShared<FExecute>(
196 TestRunner,
197 [&Result = AsyncResult, Action = MoveTemp(AsyncAction)]() { Result = Action(); },
198 AsyncExecuteDetails::CreateInternalCommandDescription(Descriptions, Description, TEXT("Execute async action"))
199 ));
200 RunSequence.Append(MakeShared<FWaitUntil>(
201 TestRunner,
202 [&Result = AsyncResult]() { return Result.GetFuture().IsReady(); },
203 AsyncActionTimeout,
204 AsyncExecuteDetails::CreateInternalCommandDescription(Descriptions, Description, TEXT("Wait async result"))
205 ));
206 if constexpr (!std::is_void_v<ResultCommandType>)
207 {
208 static_assert(!std::is_void_v<ResultType>,
209 "The <ResultType> 'void' should not be used with <ResultCommandType>. "
210 "Instead, consider using a standalone command for any subsequent actions.");
211
212 RunSequence.Append(AsyncExecuteDetails::TResultCommandFactory<ResultType, ResultCommandType>::Create(
213 TestRunner,
214 AsyncResult,
215 Descriptions,
216 Description,
217 std::forward<ResultCommandArgs>(Args)...)
218 );
219 }
220 };
221
222 bool Update() override { return RunSequence.Update(); }
223
224private:
225 FRunSequence RunSequence;
226 TAsyncResult<ResultType> AsyncResult;
227 TArray<FString> Descriptions;
228};
