296
297 if (Extension.Equals(TEXT("uasset"), ESearchCase::IgnoreCase))
298 {
299 Icon = StyleSet->GetBrush("SlateFileDialogs.UAsset16");
300 }
301 else if (Extension.Equals(TEXT("uproject"), ESearchCase::IgnoreCase))

303 Icon = StyleSet->GetBrush("SlateFileDialogs.UProject16");

307 Icon = StyleSet->GetBrush("SlateFileDialogs.Model3D");
308 }
309 else if (Extension.Equals(TEXT("cpp"), ESearchCase::IgnoreCase) ||
310 Extension.Equals(TEXT("h"), ESearchCase::IgnoreCase) ||
311 Extension.Equals(TEXT("txt"), ESearchCase::IgnoreCase) ||
312 Extension.Equals(TEXT("log"), ESearchCase::IgnoreCase))
313 {
314 Icon = StyleSet->GetBrush("SlateFileDialogs.TextFile");
315 }
316 else if (Extension.Equals(TEXT("wav"), ESearchCase::IgnoreCase) ||
317 Extension.Equals(TEXT("mp3"), ESearchCase::IgnoreCase) ||
318 Extension.Equals(TEXT("ogg"), ESearchCase::IgnoreCase))
319 {
320 Icon = StyleSet->GetBrush("SlateFileDialogs.Audio");
321 }
322 else if (Extension.Equals(TEXT("mp4"), ESearchCase::IgnoreCase) ||
323 Extension.Equals(TEXT("webm"), ESearchCase::IgnoreCase))
324 {
325 Icon = StyleSet->GetBrush("SlateFileDialogs.Video");
326 }
327 else if (Extension.Equals(TEXT("png"), ESearchCase::IgnoreCase) ||
328 Extension.Equals(TEXT("jpg"), ESearchCase::IgnoreCase) ||
329 Extension.Equals(TEXT("tga"), ESearchCase::IgnoreCase) ||
330 Extension.Equals(TEXT("bmp"), ESearchCase::IgnoreCase))
331 {
332 Icon = StyleSet->GetBrush("SlateFileDialogs.Image");
333 }
334 else
335 {
336 Icon = StyleSet->GetBrush("SlateFileDialogs.PlaceHolder");
337 FolderIconVisibility = EVisibility::Hidden;
338 }
339 }
340
341 if (ColumnName == TEXT("Pathname"))
342 {
343 return SNew(SHorizontalBox)
344
345 + SHorizontalBox::Slot()
346 .HAlign(HAlign_Left)
347 .VAlign(VAlign_Top)
348 .AutoWidth()
349 .Padding(FMargin(5.0f, 2.0f))

353 .Visibility(FolderIconVisibility)
354 ]
355
356 + SHorizontalBox::Slot()
357 .HAlign(HAlign_Left)
358 .VAlign(VAlign_Top)
359 .AutoWidth()
360 .Padding(FMargin(5.0f, 0.0f, 0.0f, 0.0f))
361 [
362 SNew(STextBlock)

365 ];

367 else if (ColumnName == TEXT("ModDate"))
