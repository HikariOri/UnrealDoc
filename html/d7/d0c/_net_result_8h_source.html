373 FString ToString(ENetResultString ConversionType=ENetResultString::WithoutChain) const
374 {
375 using namespace UE::Net::Private;
376
377 FString ReturnVal;
378 const bool bIncludeChain = ConversionType == ENetResultString::WithChain;
379
380 for (FConstIterator It(*this); It; ++It)
381 {
382 const TCHAR* ResultLex = ToCStr(LexToString(static_cast<ResultEnum>(It->Result)));
383
384 if (ConversionType == ENetResultString::ResultEnumOnly)
385 {
386 ReturnVal = ResultLex;
387 }
388 else
389 {
390 NetResultToString(ReturnVal, ResultLex, It->ErrorContext, ConversionType);
391 }
392
393 if (!bIncludeChain)
394 {
395 break;
396 }
397 }
398
399 return ReturnVal;
400 }
