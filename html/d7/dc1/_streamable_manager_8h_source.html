967template <typename CallableT>
968void FStreamableHandle::ForEachLoadedAsset(CallableT Callable) const
969{
970 if (HasLoadCompleted())
971 {
972 for (const FSoftObjectPath& Ref : RequestedAssets)
973 {
974 // Try manager, should be faster and will handle redirects better
975 if (IsActive())
976 {
977 Invoke(Callable, OwningManager->GetStreamed(Ref));
978 }
979 else
980 {
981 Invoke(Callable, Ref.ResolveObject());
982 }
983 }
984
985 // Check child handles
986 for (const TSharedPtr<FStreamableHandle>& ChildHandle : ChildHandles)
987 {
988 for (const FSoftObjectPath& Ref : ChildHandle->RequestedAssets)
989 {
990 // Try manager, should be faster and will handle redirects better
991 if (IsActive())
992 {
993 Invoke(Callable, OwningManager->GetStreamed(Ref));
994 }
995 else
996 {
997 Invoke(Callable, Ref.ResolveObject());
998 }
999 }
1000 }
1001 }
