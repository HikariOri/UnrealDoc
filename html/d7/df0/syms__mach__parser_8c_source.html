9SYMS_API SYMS_MachBinAccel*
10syms_mach_bin_from_base_range(SYMS_Arena *arena, void *base, SYMS_U64Range range){
11 SYMS_MachBinAccel *result = (SYMS_MachBinAccel*)&syms_format_nil;
12
13 //- read properties from magic
14 SYMS_U32 magic = 0;
15 syms_based_range_read_struct(base, range, 0, &magic);
16
17 SYMS_B32 is_mach = syms_false;
18 SYMS_B32 is_swapped = syms_false;
19 SYMS_B32 is_32 = syms_false;
20 switch (magic){
21 case SYMS_MACH_MAGIC_32:
22 {
23 is_mach = syms_true;
24 is_32 = syms_true;
25 }break;
26 case SYMS_MACH_MAGIC_64:
27 {
28 is_mach = syms_true;
29 }break;
30
31 case SYMS_MACH_CIGAM_32:
32 {
33 is_mach = syms_true;
34 is_swapped = syms_true;
35 is_32 = syms_true;
36 }break;
37 case SYMS_MACH_CIGAM_64:
38 {
39 is_mach = syms_true;
40 is_swapped = syms_true;
41 }break;
42 }
43
44 if (is_mach){
45 //- read header
46 SYMS_U64 after_header_off = 0;
47 SYMS_MachHeader64 header = {0};
48 if (is_32){
49 SYMS_MachHeader32 header32 = {0};
50 after_header_off = syms_based_range_read_struct(base, range, 0, &header32);
51 if (is_swapped){
52 syms_bswap_in_place(SYMS_MachHeader32, &header32);
53 }
54 syms_mach_header64_from_header32(&header, &header32);
55 }
56 else{
57 after_header_off = syms_based_range_read_struct(base, range, 0, &header);
58 if (is_swapped){
59 syms_bswap_in_place(SYMS_MachHeader64, &header);
60 }
61 }
62
63 //- gather segment and section lists
64 SYMS_ArenaTemp scratch = syms_get_scratch(&arena, 1);
65
66 SYMS_MachSegmentNode *segment_first = 0;
67 SYMS_MachSegmentNode *segment_last = 0;
68 SYMS_U32 segment_count = 0;
69
70 SYMS_MachSectionNode *section_first = 0;
71 SYMS_MachSectionNode *section_last = 0;
72 SYMS_U32 section_count = 0;
73
74 SYMS_U64Range bind_ranges[SYMS_MachBindTable_COUNT]; syms_memzero_struct(&bind_ranges[0]);
75 SYMS_U64Range export_range = syms_make_u64_range(0,0);
76
77 SYMS_MachSymtabCommand symtab; syms_memzero_struct(&symtab);
78
79 SYMS_MachDylibList dylib_list; syms_memzero_struct(&dylib_list);
80
81 { // push null dylib
82 SYMS_MachDylib dylib; syms_memzero_struct(&dylib);
83 syms_mach_dylib_list_push(scratch.arena, &dylib_list, &dylib, syms_make_u64_range(0,0));
84 }
85
86 SYMS_U64 next_cmd_off = after_header_off;
87 for (SYMS_U32 i = 0; i < header.ncmds; i += 1){
88 //- align read offset
89 SYMS_U64 cmd_off = 0;
90 if (is_32){
91 cmd_off = SYMS_AlignPow2(next_cmd_off, 4);
92 }
93 else{
94 cmd_off = SYMS_AlignPow2(next_cmd_off, 8);
95 }
96
97 //- read command
98 SYMS_MachLoadCommand lc = {0};
99 syms_based_range_read_struct(base, range, cmd_off, &lc);
100 switch (lc.type){
101 case SYMS_MachLoadCommandType_SYMTAB:
102 {
103 syms_based_range_read_struct(base, range, cmd_off, &symtab);
104 }break;
105
106 case SYMS_MachLoadCommandType_SEGMENT:
107 {
108 SYMS_MachSegmentCommand32 segment_command32 = {0};
109 syms_based_range_read_struct(base, range, cmd_off, &segment_command32);
110 if (is_swapped){
111 syms_bswap_in_place(SYMS_MachSegmentCommand32, &segment_command32);
112 }
113 SYMS_U64 after_seg_off = cmd_off + sizeof(SYMS_MachSegmentCommand32);
114
115 // push segment node
116 SYMS_MachSegmentNode *segment_node = syms_push_array_zero(scratch.arena, SYMS_MachSegmentNode, 1);
117 syms_mach_segment64_from_segment32(&segment_node->data, &segment_command32);
118 SYMS_QueuePush(segment_first, segment_last, segment_node);
119 segment_count += 1;
120
121 // loop over sections
122 SYMS_U64 next_sec_off = after_seg_off;
123 SYMS_U64 sec_count = segment_command32.nsects;
124 for (SYMS_U32 k = 0; k < sec_count; k += 1){
125 // read section 64
126 SYMS_U64 sec_off = next_sec_off;
127 SYMS_MachSection32 section32 = {0};
128 syms_based_range_read_struct(base, range, sec_off, &section32);
129 if (is_swapped){
130 syms_bswap_in_place(SYMS_MachSection32, &section32);
131 }
132 next_sec_off = sec_off + sizeof(SYMS_MachSection32);
133
134 // push section node
135 SYMS_MachSectionNode *section_node = syms_push_array_zero(scratch.arena, SYMS_MachSectionNode, 1);
136 syms_mach_section64_from_section32(&section_node->data, &section32);
137 SYMS_QueuePush(section_first, section_last, section_node);
138 section_count += 1;
139 }
140 }break;
141
142 case SYMS_MachLoadCommandType_SEGMENT_64:
143 {
144 // read segment 64
145 SYMS_MachSegmentCommand64 segment_command64 = {0};
146 syms_based_range_read_struct(base, range, cmd_off, &segment_command64);
147 if (is_swapped){
148 syms_bswap_in_place(SYMS_MachSegmentCommand64, &segment_command64);
149 }
150 SYMS_U64 after_seg_off = cmd_off + sizeof(SYMS_MachSegmentCommand64);
151
152 // push segment node
153 SYMS_MachSegmentNode *segment_node = syms_push_array_zero(scratch.arena, SYMS_MachSegmentNode, 1);
154 segment_node->data = segment_command64;
155 SYMS_QueuePush(segment_first, segment_last, segment_node);
156 segment_count += 1;
157
158 // loop over sections
159 SYMS_U64 next_sec_off = after_seg_off;
160 SYMS_U64 sec_count = segment_command64.nsects;
161 for (SYMS_U32 k = 0; k < sec_count; k += 1){
162 // read section 64
163 SYMS_U64 sec_off = next_sec_off;
164 SYMS_MachSection64 section64 = {0};
165 syms_based_range_read_struct(base, range, sec_off, &section64);
166 if (is_swapped){
167 syms_bswap_in_place(SYMS_MachSection64, &section64);
168 }
169 next_sec_off = sec_off + sizeof(SYMS_MachSection64);
170
171 // push section node
172 SYMS_MachSectionNode *section_node = syms_push_array_zero(scratch.arena, SYMS_MachSectionNode, 1);
173 section_node->data = section64;
174 SYMS_QueuePush(section_first, section_last, section_node);
175 section_count += 1;
176 }
177 }break;
178
179 case SYMS_MachLoadCommandType_DYLD_INFO_ONLY:
180 {
181 SYMS_MachDyldInfoCommand dyld;
182 syms_based_range_read_struct(base, range, cmd_off, &dyld);
183 if (is_swapped){
184 syms_bswap_in_place(SYMS_MachDyldInfoCommand, &dyld);
185 }
186
187 bind_ranges[SYMS_MachBindTable_REGULAR] = syms_make_u64_inrange(range, dyld.bind_off, dyld.bind_size);
188 bind_ranges[SYMS_MachBindTable_LAZY] = syms_make_u64_inrange(range, dyld.lazy_bind_off, dyld.lazy_bind_size);
189 bind_ranges[SYMS_MachBindTable_WEAK] = syms_make_u64_inrange(range, dyld.weak_bind_off, dyld.weak_bind_size);
190 export_range = syms_make_u64_inrange(range, dyld.export_off, dyld.export_size);
191 }break;
192
193 case SYMS_MachLoadCommandType_LOAD_WEAK_DYLIB:
194 case SYMS_MachLoadCommandType_LOAD_DYLIB:
195 case SYMS_MachLoadCommandType_LOAD_UPWARD_DYLIB:
196 case SYMS_MachLoadCommandType_LAZY_LOAD_DYLIB:
197 {
198 SYMS_MachDylibCommand cmd; syms_memzero_struct(&cmd);
199 syms_based_range_read_struct(base, range, cmd_off, &cmd);
200 SYMS_MachDylib *dylib = &cmd.dylib;
201 if (is_swapped){
202 syms_bswap_in_place(SYMS_MachDylib, dylib);
203 }
204 SYMS_U64 name_offset = cmd_off + dylib->name.offset;
205 SYMS_U64Range name_range = syms_make_u64_inrange(range, name_offset, (cmd_off + lc.size) - name_offset);
206 syms_mach_dylib_list_push(scratch.arena, &dylib_list, dylib, name_range);
207 }break;
208 }
209
210 next_cmd_off = cmd_off + lc.size;
211 }
212
213 //- segment array from list
214 SYMS_MachSegmentCommand64 *segments = syms_push_array(arena, SYMS_MachSegmentCommand64, segment_count);
215 {
216 SYMS_MachSegmentNode *segment_node = segment_first;
217 SYMS_MachSegmentCommand64 *segment_ptr = segments;
218 SYMS_MachSegmentCommand64 *segment_opl = segments + segment_count;
219 for (; segment_ptr < segment_opl; segment_ptr += 1, segment_node = segment_node->next){
220 *segment_ptr = segment_node->data;
221 }
222 }
223
224 //- section array from list
225 SYMS_MachSection64 *sections = syms_push_array(arena, SYMS_MachSection64, section_count);
226 {
227 SYMS_MachSectionNode *section_node = section_first;
228 SYMS_MachSection64 *section_ptr = sections;
229 SYMS_MachSection64 *section_opl = sections + section_count;
230 for (; section_ptr < section_opl; section_ptr += 1, section_node = section_node->next){
231 *section_ptr = section_node->data;
232 }
233 }
234
235 //- fill result
236 result = syms_push_array(arena, SYMS_MachBinAccel, 1);
237 result->load_command_count = header.ncmds;
238 result->load_commands = syms_make_u64_inrange(range, after_header_off, header.sizeofcmds);
239 result->format = SYMS_FileFormat_MACH;
240 result->arch = syms_mach_arch_from_cputype(header.cputype);
241 result->is_swapped = is_swapped;
242 result->symtab = symtab;
243 result->segment_count = segment_count;
244 result->segments = segments;
245 result->section_count = section_count;
246 result->sections = sections;
247 syms_memmove(&result->bind_ranges[0], &bind_ranges[0], sizeof(bind_ranges));
248 result->export_range = export_range;
249 { // convert dylib list to array
250 result->dylib_count = 0;
251 result->dylibs = syms_push_array(arena, SYMS_MachParsedDylib, dylib_list.count);
252 for (SYMS_MachDylibNode *n = dylib_list.first; n != 0; n = n->next){
253 result->dylibs[result->dylib_count++] = n->data;
254 }
255 }
256
257 syms_release_scratch(scratch);
258 }
259
260 return(result);
261}

423syms_mach_sec_info_array_from_bin(SYMS_Arena *arena, SYMS_String8 data, SYMS_MachBinAccel *bin){
424 SYMS_SecInfoArray array;
425 array.count = bin->section_count;
426 array.sec_info = syms_push_array_zero(arena, SYMS_SecInfo, array.count);
427
428 SYMS_MachSection64 *mach_sec = bin->sections;
429 SYMS_MachSection64 *mach_sec_opl = bin->sections + array.count;

432 SYMS_U8 *name_ptr = mach_sec->sectname;
433 SYMS_U8 *ptr = name_ptr;
434 SYMS_U8 *opl = name_ptr + SYMS_ARRAY_SIZE(mach_sec->sectname);
435 for (;ptr < opl && *ptr != 0; ptr += 1);
436
437 SYMS_String8 name = syms_str8_range(name_ptr, ptr);
438
439 sec_info->name = syms_push_string_copy(arena, name);
440 // TODO(allen): figure out when these ranges actually apply and when the section isn't
441 // there one side or the other.
442 sec_info->vrange = syms_make_u64_range(mach_sec->addr, mach_sec->addr + mach_sec->size);
443 sec_info->frange = syms_make_u64_range(mach_sec->offset, mach_sec->offset + mach_sec->size);
444 }
445 return array;
446}

476SYMS_API SYMS_MachBindList
477syms_mach_binds_from_base_range(SYMS_Arena *arena, void *base, SYMS_U64Range range, SYMS_U32 address_size, SYMS_MachBindTable bind_type){
478 SYMS_U64 read_offset = 0;
479 SYMS_MachBindList list; syms_memzero_struct(&list);
480 SYMS_MachBind state; syms_memzero_struct(&state);
481 for (;;){
482 SYMS_U8 encoded_byte = 0;
483 read_offset += syms_based_range_read_struct(base, range, read_offset, &encoded_byte);
484
485 SYMS_U8 opcode = encoded_byte & SYMS_MachBindOpcode_MASK;
486 SYMS_U8 imm = encoded_byte & SYMS_MachBindOpcode_IMM_MASK;
487
488 switch (opcode){
489 case SYMS_MachBindOpcode_DONE: goto exit;
490
491 case SYMS_MachBindOpcode_SET_DYLIB_ORDINAL_IMM:
492 {
493 state.dylib = imm;
494 } break;
495 case SYMS_MachBindOpcode_SET_DYLIB_ORDINAL_ULEB:
496 {
497 read_offset += syms_based_range_read_uleb128(base, range, read_offset, &state.dylib);
498 } break;
499 case SYMS_MachBindOpcode_SET_DYLIB_SPECIAL_IMM:
500 {
501 SYMS_ASSERT_PARANOID("TODO: SET_DYLIB_SPECIAL_IMM");
502 } break;
503 case SYMS_MachBindOpcode_SET_SYMBOL_TRAILING_FLAGS_IMM:
504 {
505 state.flags = imm;
506 state.symbol_name = syms_based_range_read_string(base, range, read_offset);
507 read_offset += (state.symbol_name.size + 1);
508 } break;
509 case SYMS_MachBindOpcode_SET_TYPE_IMM:
510 {
511 state.type = imm;
512 } break;
513 case SYMS_MachBindOpcode_SET_ADDEND_SLEB:
514 {
515 read_offset += syms_based_range_read_sleb128(base, range, read_offset, &state.addend);
516 } break;
517 case SYMS_MachBindOpcode_SET_SEGMENT_AND_OFFSET_ULEB:
518 {
519 state.segment = imm;
520 read_offset += syms_based_range_read_uleb128(base, range, read_offset, &state.segment_offset);
521 } break;
522 case SYMS_MachBindOpcode_ADD_ADDR_ULEB:
523 {
524 SYMS_U64 addend = 0;
525 read_offset += syms_based_range_read_uleb128(base, range, read_offset, &addend);
526 state.segment_offset += addend;
527 } break;
528 case SYMS_MachBindOpcode_DO_BIND_ADD_ADDR_ULEB:
529 {
530 SYMS_U64 addend = 0;
531 read_offset += syms_based_range_read_uleb128(base, range, read_offset, &addend);
532 state.segment_offset += addend;
533 } goto do_bind;
534 case SYMS_MachBindOpcode_DO_BIND_ADD_ADDR_IMM_SCALED:
535 {
536 state.segment_offset += imm;
537 } goto do_bind;
538 case SYMS_MachBindOpcode_DO_BIND_ULEB_TIMES_SKIPPING_ULEB:
539 {
540 SYMS_U64 count = 0, skip = 0;
541 read_offset += syms_based_range_read_uleb128(base, range, read_offset, &count);
542 read_offset += syms_based_range_read_uleb128(base, range, read_offset, &skip);
543
544 state.segment_offset += skip * address_size;
545 } goto do_bind;
546 case SYMS_MachBindOpcode_DO_BIND:
547 {
548 } goto do_bind;
549 do_bind:
550 {
551 if (bind_type == SYMS_MachBindTable_LAZY){
552 for (;read_offset < syms_u64_range_size(range);){
553 SYMS_U8 value = 0;
554 SYMS_U64 read_size = syms_based_range_read_struct(base, range, read_offset, &value);
555 if (value != 0){
556 break;
557 }
558 read_offset += read_size;
559 }
560 }

617syms_mach_parse_export_node(SYMS_Arena *arena, void *base, SYMS_U64Range range, SYMS_String8 name, SYMS_U64 read_offset){
618 SYMS_MachExport *node = 0;
619 SYMS_U64 export_size = 0;
620 SYMS_U64 read_size = syms_based_range_read_uleb128(base, range, read_offset, &export_size);
621 if (read_size > 0){

623
624 // push new node
625 node = syms_push_array_zero(arena, SYMS_MachExport, 1);
626 node->name = name;
627 node->is_export_info = export_size != 0;
628
629 // read export discriminator
630 SYMS_U64Range export_range = syms_make_u64_inrange(range, read_offset, export_size);
631 read_offset += export_size;
632
633 // parse export info
634 if (node->is_export_info){
635 SYMS_U64 export_read_offset = 0;
636 export_read_offset += syms_based_range_read_uleb128(base, export_range, export_read_offset, &node->flags);
637 if (node->flags & SYMS_MachExportSymbolFlags_REEXPORT){
638 export_read_offset += syms_based_range_read_uleb128(base, export_range, export_read_offset, &node->dylib_ordinal);

640 export_read_offset += (node->import_name.size + 1);
641 } else{
642 export_read_offset += syms_based_range_read_uleb128(base, export_range, export_read_offset, &node->address);
643 if (node->flags & SYMS_MachExportSymbolFlags_STUB_AND_RESOLVED){
644 export_read_offset += syms_based_range_read_uleb128(base, export_range, export_read_offset, &node->resolver);
645 }
646 }
647 }
648
649 // read child info
650 read_offset += syms_based_range_read_struct(base, range, read_offset, &node->child_count);
651 if (node->child_count > 0){
652 node->children = syms_push_array(arena, SYMS_MachExport *, node->child_count);
653 for (SYMS_U8 child_idx = 0; child_idx < node->child_count; child_idx += 1){
654 SYMS_String8 name = syms_based_range_read_string(base, range, read_offset);
655 read_offset += (name.size + 1);
656 SYMS_U64 child_offset = 0;
657 read_offset += syms_based_range_read_uleb128(base, range, read_offset, &child_offset);
658 node->children[child_idx] = syms_mach_parse_export_node(arena, base, range, name, child_offset);
659 }
660 }
661 }

673syms_mach_exports_from_bin(SYMS_Arena *arena, SYMS_String8 data, SYMS_MachBinAccel *bin){
674 SYMS_ArenaTemp scratch = syms_get_scratch(&arena, 1);
675
676 // read and copy library names
677 SYMS_String8 *dylib_names = syms_push_array(scratch.arena, SYMS_String8, bin->dylib_count);
678 for (SYMS_U32 i = 0; i < bin->dylib_count; ++i){
679 SYMS_String8 name = syms_based_range_read_string((void*)data.str, bin->dylibs[i].name, 0);
680 dylib_names[i] = syms_push_string_copy(arena, name);
681 }
682
683 // build export trie
684 SYMS_MachExport *root_export = syms_build_mach_export_trie(scratch.arena, (void*)data.str, bin->export_range);
685
686 // traverse trie
687 SYMS_ExportNode *first_export = 0;
688 SYMS_ExportNode *last_export = 0;
689 SYMS_U32 export_count = 0;
690 if (root_export){
691 SYMS_MachExportFrame *stack = 0;
692 SYMS_MachExportFrame *free_frames = 0;
693
694 SYMS_MachExportFrame *frame = syms_push_array_zero(arena, SYMS_MachExportFrame, 1);
695 frame->node = root_export;
696 SYMS_StackPush(stack, frame);
697
698 for (; stack != 0; ){
699 SYMS_B32 pop_frame = frame->child_idx >= frame->node->child_count;
700 if (pop_frame){
701 if (frame->node->is_export_info){
702 // append strings from nodes on the stack
703 SYMS_ArenaTemp temp = syms_arena_temp_begin(scratch.arena);
704 SYMS_String8List name_parts; syms_memzero_struct(&name_parts);
705 for (SYMS_MachExportFrame *f = frame; f != 0; f = f->next){
706 syms_string_list_push_front(temp.arena, &name_parts, f->node->name);
707 }
708 SYMS_StringJoin join; syms_memzero_struct(&join);
709 SYMS_String8 name = syms_string_list_join(arena, &name_parts, &join);
710 syms_arena_temp_end(temp);
711
712 SYMS_ExportNode *export_node = syms_push_array(scratch.arena, SYMS_ExportNode, 1);
713
714 // add export to the list
715 SYMS_QueuePush(first_export, last_export, export_node);
716 export_count += 1;
717
718 // initialize export
719 SYMS_Export *exp = &export_node->data;
720 if (frame->node->flags & SYMS_MachExportSymbolFlags_REEXPORT){
721 // TODO(nick): need dylib with reexports to test this code path
722 if (frame->node->dylib_ordinal > 0 && frame->node->dylib_ordinal < bin->dylib_count){
723 exp->name = name;
724 exp->address = 0;
725 exp->ordinal = 0;
726 exp->forwarder_library_name = dylib_names[frame->node->dylib_ordinal-1];
727 exp->forwarder_import_name = syms_push_string_copy(arena, frame->node->import_name);
728 } else{
729 SYMS_ASSERT_PARANOID(!"invalid dylib ordinal");
730 }
731 } else{
732 exp->name = name;
733 exp->address = frame->node->address;
734 exp->ordinal = 0;
735 exp->forwarder_library_name = syms_str8(0,0);
736 exp->forwarder_import_name = syms_str8(0,0);
737 }
738 }
739
740 // pop current frame
741 SYMS_MachExportFrame *to_free = frame;
742 frame = SYMS_StackPop(stack);
743
744 // push frame to free list for later reuse
745 to_free->next = 0;
746 SYMS_StackPush(free_frames, to_free);
747 } else{
748 // push frame
749 SYMS_MachExportFrame *new_frame = free_frames;
750 if (new_frame){
751 SYMS_StackPop(free_frames);
752 } else{
753 new_frame = syms_push_array_zero(arena, SYMS_MachExportFrame, 1);
754 }
755 new_frame->next = 0;
756 new_frame->child_idx = 0;
757 new_frame->node = frame->node->children[frame->child_idx++];
758 frame = SYMS_StackPush(stack, new_frame);
759 }
760 }
761 }
762
763 SYMS_ExportArray export_array;
764 export_array.count = 0;
765 export_array.exports = syms_push_array(arena, SYMS_Export, export_count);
766 for (SYMS_ExportNode *n = first_export; n != 0; n = n->next){
767 export_array.exports[export_array.count++] = n->data;
768 }
769
770 syms_release_scratch(scratch);
771 return export_array;
772}
