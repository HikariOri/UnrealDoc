265 ECurveEditorTreeFilterState Get(FCurveEditorTreeItemID ItemID) const
266 {
267 if (!bIsActive)
268 {
269 // If not active, everything is treated as having matched the (non-existent) filters
270 return ECurveEditorTreeFilterState::Match;
271 }
272
273 const ECurveEditorTreeFilterState* State = FilterStates.Find(ItemID);
274 return State ? *State : ECurveEditorTreeFilterState::NoMatch;
275 }
276
277 template <typename CallableT>
278 void ForEachItemState(CallableT Callable) const
279 {
280 Algo::ForEach(FilterStates, Callable);
281 }
282
286 void SetFilterState(FCurveEditorTreeItemID ItemID, ECurveEditorTreeFilterState NewState)
287 {
288 const ECurveEditorTreeFilterState* Existing = FilterStates.Find(ItemID);
289 if (Existing)
290 {
291 if ((*Existing & ECurveEditorTreeFilterState::Match) != ECurveEditorTreeFilterState::NoMatch)
292 {
293 --NumMatched;
294 }
295 else if ((*Existing & ECurveEditorTreeFilterState::MatchBitMask) != ECurveEditorTreeFilterState::NoMatch)
296 {
297 --NumMatchedImplicitly;
298 }
299 }
300
301 if ((NewState & ECurveEditorTreeFilterState::MatchBitMask) == ECurveEditorTreeFilterState::NoMatch)
302 {
303 FilterStates.Remove(ItemID);
304 }
305 else
306 {
307 FilterStates.Add(ItemID, NewState);
308
309 if ((NewState & ECurveEditorTreeFilterState::Match) != ECurveEditorTreeFilterState::NoMatch)
310 {
311 ++NumMatched;
312 }
313 else
314 {
315 ++NumMatchedImplicitly;
316 }
317 }
318 }

326 }
