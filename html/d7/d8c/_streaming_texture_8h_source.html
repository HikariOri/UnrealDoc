42 FStreamingRenderAssetFlags() = default;
43 FStreamingRenderAssetFlags(const FStreamingRenderAssetFlags& Other) noexcept : Storage(Other.Storage.load(std::memory_order_relaxed)) {}
44 FStreamingRenderAssetFlags(FStreamingRenderAssetFlags&& Other) noexcept : Storage(Other.Storage.load(std::memory_order_relaxed)) {}
45
46 FStreamingRenderAssetFlags& operator=(const FStreamingRenderAssetFlags& Other) noexcept { Storage.store(Other.Storage.load(std::memory_order_relaxed), std::memory_order_relaxed); return *this; }
47 FStreamingRenderAssetFlags& operator=(FStreamingRenderAssetFlags&& Other) noexcept { Storage.store(Other.Storage.load(std::memory_order_relaxed), std::memory_order_relaxed); return *this; }
48
49 bool Get(EBit Bit) const
50 {
51 return (Storage.load(std::memory_order_relaxed) & (uint16(1) << Bit)) != 0;
52 }
53
54 void Set(EBit Bit, bool Value)
55 {
56 const uint16 Mask = uint16(1) << Bit;
57 if (Value)
58 {
59 Storage.fetch_or(Mask, std::memory_order_relaxed);
60 }
61 else
62 {
63 Storage.fetch_and(static_cast<uint16>(~Mask), std::memory_order_relaxed);
64 }
65 }
66
67private:
68 std::atomic<uint16> Storage{ 0 };

260 FORCEINLINE bool HasUpdatePending(bool bIsStreamingPaused, bool bHasViewPoint) const
261 {
262 const bool bBudgetedMipsIsValid = bHasViewPoint || GetFlag(FStreamingRenderAssetFlags::ForceFullyLoadHeuristic); // Force fully load don't need any viewpoint info.
263 // If paused, nothing will update anytime soon.
264 // If more mips will be streamed in eventually, wait.
265 // Otherwise, if the distance based computation had no viewpoint, wait.
266 return !bIsStreamingPaused && (BudgetedMips > ResidentMips || !bBudgetedMipsIsValid);
267 }
