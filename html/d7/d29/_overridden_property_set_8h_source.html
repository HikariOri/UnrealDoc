225/*
226 * Override operation type for each property node
227 */
228UENUM()
229enum class EOverriddenPropertyOperation : uint8
230{
231 None = 0, /* no overridden operation was recorded on this property */
232 Modified, /* some sub property has recorded overridden operation */
233 Replace, /* everything has been overridden from this property down to every sub property/sub object*/
234 Add, /* this element was added in the container */
235 Remove, /* this element was removed from the container */
236 SubObjectsShadowing, /* only used to serialize/collect the subobjects, this saved data can be skipped at loading unless loading a loose property or in a placeholder which will keep reference to subobjects */
237 Externalized, /* this indicates the elements are stored externally */
238};

244 {
245 return TOptional<EOverriddenPropertyOperation>();
