103 struct TTupleElementGetterByIndex
104 {
105 template <typename DeducedType, typename TupleType>
106 static inline decltype(auto) GetImpl(const volatile TTupleBaseElement<DeducedType, Index, TupleSize>& Element, TupleType&& Tuple)
107 {
108#if UE_TUPLE_REFERENCE_WORKAROUND
109 if constexpr (!std::is_reference_v<TupleType> && std::is_reference_v<DeducedType>)
110 {
111 return (DeducedType)GetImpl(Element, Tuple);
112 }
113 else
114#endif
115 {
116 // Brackets are important here - we want a reference type to be returned, not object type
117 decltype(auto) Result = (ForwardAsBase<TupleType, TTupleBaseElement<DeducedType, Index, TupleSize>>(Tuple).Value);
118
119 // Keep tuple rvalue references to rvalue reference elements as rvalues, because that's how std::get() works, not how C++ struct member access works.
120 return static_cast<std::conditional_t<!std::is_reference_v<TupleType> && std::is_rvalue_reference_v<DeducedType>, DeducedType, decltype(Result)>>(Result);
121 }
122 }
123
124 template <typename TupleType>
125 static UE_FORCEINLINE_HINT decltype(auto) Get(TupleType&& Tuple)
126 {
127 return GetImpl(Tuple, Forward<TupleType>(Tuple));
128 }
