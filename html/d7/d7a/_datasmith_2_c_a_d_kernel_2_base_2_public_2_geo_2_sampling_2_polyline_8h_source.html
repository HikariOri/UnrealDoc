209 void FindIntersectionsWithIso(const EIso Iso, double IsoParameter, TArray<double>& OutIntersections)
210 {
211 OutIntersections.Empty(8);
212
213 int32 UIndex = Iso == EIso::IsoU ? 0 : 1;
214 int32 VIndex = Iso == EIso::IsoU ? 1 : 0;
215
216 TFunction<void(const int32, const int32, const int32)> ComputeIntersection = [&](const int32 Index1, const int32& Index2, const int32 MinIndex)
217 {
218 if (IsoParameter > Points[Index1][UIndex] && IsoParameter <= Points[Index2][UIndex])
219 {
220 double Intersection = (IsoParameter - Points[Index1][UIndex]) / (Points[Index2][UIndex] - Points[Index1][UIndex]);
221 OutIntersections.Add(PolylineTools::LinearInterpolation(Coordinates, MinIndex, Intersection));
222 }
223 };
224
225 for (int32 Index = 1; Index < Points.Num(); ++Index)
226 {
227 if (!FMath::IsNearlyEqual(Points[Index - 1][UIndex], Points[Index][UIndex]))
228 {
229 if (Points[Index - 1][UIndex] < Points[Index][UIndex])
230 {
231 ComputeIntersection(Index - 1, Index, Index - 1);
232 }
233 else
234 {
235 ComputeIntersection(Index, Index - 1, Index - 1);
236 }
237 }
238 }
