289 virtual FCbObject ExportGlobalFormatConfig(const FTextureBuildSettings& BuildSettings) const override
290 {
291 FTextureBuildSettings BaseSettings = GetBaseTextureBuildSettings(BuildSettings);

293 FCbObject BaseObj = GetBaseFormatObject(BuildSettings.TextureFormatName)->ExportGlobalFormatConfig(BaseSettings);
294 FCbObject ChildObj = ExportGlobalChildFormatConfig(BuildSettings);

301 FCbWriter Writer;

303

306 Writer.AddObject("BaseTextureFormatConfig", BaseObj);
307 }
308
309 if (ChildObj)

311 Writer.AddObject("ChildTextureFormatConfig", ChildObj);
312 }
