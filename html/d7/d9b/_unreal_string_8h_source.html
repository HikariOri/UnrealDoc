31[[nodiscard]] inline FString BytesToString(const uint8* In, int32 Count)
32{
33 FString Result;
34 Result.Empty(Count);
35
36 while (Count)
37 {
38 // Put the byte into an int16 and add 1 to it, this keeps anything from being put into the string as a null terminator
39 int16 Value = *In;
40 Value += 1;
41
42 Result += FString::ElementType(Value);
43
44 ++In;
45 Count--;
46 }
47 return Result;
48}

133[[nodiscard]] inline const uint8 TCharToNibble(const TCHAR Hex)
134{
135 if (Hex >= '0' && Hex <= '9')
136 {
137 return uint8(Hex - '0');
138 }
139 if (Hex >= 'A' && Hex <= 'F')
140 {
141 return uint8(Hex - 'A' + 10);
142 }
143 if (Hex >= 'a' && Hex <= 'f')
144 {
145 return uint8(Hex - 'a' + 10);
146 }
147 checkf(false, TEXT("'%c' (0x%02X) is not a valid hexadecimal digit"), Hex, Hex);
148 return 0;
149}
