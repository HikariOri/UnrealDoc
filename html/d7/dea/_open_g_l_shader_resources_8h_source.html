61struct FOpenGLShaderBindings
62{
63 TArray<TArray<CrossCompiler::FPackedArrayInfo>> PackedUniformBuffers;
64 TArray<CrossCompiler::FPackedArrayInfo> PackedGlobalArrays;
65 TArray<FOpenGLShaderVarying> InputVaryings;
66 TArray<FOpenGLShaderVarying> OutputVaryings;
67 CrossCompiler::FShaderBindingInOutMask InOutMask;
68
69 uint8 NumSamplers = 0;
70 uint8 NumUniformBuffers = 0;
71 uint8 NumUAVs = 0;
72 bool bFlattenUB = false;
73
74 FSHAHash VaryingHash; // Not serialized, built during load to allow us to diff varying info but avoid the memory overhead.
75
76 inline FArchive& Serialize(FArchive& Ar, FShaderResourceTable& ShaderResourceTable);
77};
78
79inline FArchive& FOpenGLShaderBindings::Serialize(FArchive& Ar, FShaderResourceTable& ShaderResourceTable)
80{
81 Ar << PackedUniformBuffers;
82 Ar << PackedGlobalArrays;
83 Ar << InputVaryings;
84 Ar << OutputVaryings;
85 Ar << ShaderResourceTable;
86 Ar << InOutMask;
87 Ar << NumSamplers;
88 Ar << NumUniformBuffers;
89 Ar << NumUAVs;
90 Ar << bFlattenUB;
91
92 if (Ar.IsLoading())
93 {
94 // hash then strip out the Input/OutputVaryings at load time.
95 // The hash ensures varying diffs still affect operator== and GetTypeHash()
96 FSHA1 HashState;
97 auto HashVarying = [](FSHA1& InHashState, const TArray<FOpenGLShaderVarying>& InInputVaryings)
98 {
99 for (const FOpenGLShaderVarying& Varying : InInputVaryings)
100 {
101 InHashState.Update((const uint8*)&Varying.Location, sizeof(Varying.Location));
102 InHashState.Update((const uint8*)Varying.Varying.GetData(), Varying.Varying.Num() * sizeof(ANSICHAR));
103 }
104 };
105 HashVarying(HashState, InputVaryings);
106 HashVarying(HashState, OutputVaryings);
107 HashState.Final();
108 HashState.GetHash(&VaryingHash.Hash[0]);
109
110 InputVaryings.Empty();
111 OutputVaryings.Empty();
112 }
113
114 return Ar;
115}

131inline FArchive& FOpenGLCodeHeader::Serialize(FArchive& Ar, FShaderResourceTable& SRT)
132{
133 Ar << GlslMarker;
134 Ar << FrequencyMarker;
135 Bindings.Serialize(Ar, SRT);
136 Ar << ShaderName;
137 int32 NumInfos = UniformBuffersCopyInfo.Num();
138 Ar << NumInfos;
139 if (Ar.IsSaving())
140 {
141 for (int32 Index = 0; Index < NumInfos; ++Index)
142 {
143 Ar << UniformBuffersCopyInfo[Index];
144 }
145 }
146 else if (Ar.IsLoading())
147 {
148 UniformBuffersCopyInfo.Empty(NumInfos);
149 for (int32 Index = 0; Index < NumInfos; ++Index)
150 {
151 CrossCompiler::FUniformBufferCopyInfo Info;
152 Ar << Info;
153 UniformBuffersCopyInfo.Add(Info);
154 }
155 }
156 return Ar;
157}
