304 void InitDevicesFromConfig()
305 {
306 if (bChangingDeviceConfig)
307 {
308 return;
309 }
310 bChangingDeviceConfig = true;
311
312 int NumDevices = 0;
313 for (;; ++NumDevices)
314 {
315 FString DeviceName, DeviceUser, DevicePass;
316
317 FString DeviceBaseKey(FString::Printf(TEXT("LinuxTargetPlatfrom_%s_Device_%d"), *PlatformName(), NumDevices));
318 FString DeviceNameKey = DeviceBaseKey + TEXT("_Name");
319 if (!GConfig->GetString(TEXT("/Script/LinuxTargetPlatform.LinuxTargetSettings"), *DeviceNameKey, DeviceName, GEngineIni))
320 {
321 // no such device
322 break;
323 }
324
325 if (!AddDevice(DeviceName, false))
326 {
327 break;
328 }
329
330 // set credentials, if any
331 FString DeviceUserKey = DeviceBaseKey + TEXT("_User");
332 if (GConfig->GetString(TEXT("/Script/LinuxTargetPlatform.LinuxTargetSettings"), *DeviceUserKey, DeviceUser, GEngineIni))
333 {
334 FString DevicePassKey = DeviceBaseKey + TEXT("_Pass");
335 if (GConfig->GetString(TEXT("/Script/LinuxTargetPlatform.LinuxTargetSettings"), *DevicePassKey, DevicePass, GEngineIni))
336 {
337 for (const auto& DeviceIter : Devices)
338 {
339 ITargetDevicePtr Device = DeviceIter.Value;
340 if (Device.IsValid() && Device->GetId().GetDeviceName() == DeviceName)
341 {
342 Device->SetUserCredentials(DeviceUser, DevicePass);
343 }
344 }
345 }
346 }
347 }
348
349 bChangingDeviceConfig = false;
350 }
351
352 void SaveDevicesToConfig()
353 {
354 if (bChangingDeviceConfig)
355 {
356 return;
357 }
358 bChangingDeviceConfig = true;
359
360 int DeviceIndex = 0;
361 for (const auto& DeviceIter : Devices)
362 {
363 ITargetDevicePtr Device = DeviceIter.Value;
364
365 FString DeviceBaseKey(FString::Printf(TEXT("LinuxTargetPlatfrom_%s_Device_%d"), *TSuper::PlatformName(), DeviceIndex));
366 FString DeviceNameKey = DeviceBaseKey + TEXT("_Name");
367
368 if (Device.IsValid())
369 {
370 FString DeviceName = Device->GetId().GetDeviceName();
371 // do not save a local device on Linux or it will be duplicated
372 if (PLATFORM_LINUX && DeviceName == FPlatformProcess::ComputerName())
373 {
374 continue;
375 }
376
377 GConfig->SetString(TEXT("/Script/LinuxTargetPlatform.LinuxTargetSettings"), *DeviceNameKey, *DeviceName, GEngineIni);
378
379 FString DeviceUser, DevicePass;
380 if (Device->GetUserCredentials(DeviceUser, DevicePass))
381 {
382 FString DeviceUserKey = DeviceBaseKey + TEXT("_User");
383 FString DevicePassKey = DeviceBaseKey + TEXT("_Pass");
384
385 GConfig->SetString(TEXT("/Script/LinuxTargetPlatform.LinuxTargetSettings"), *DeviceUserKey, *DeviceUser, GEngineIni);
386 GConfig->SetString(TEXT("/Script/LinuxTargetPlatform.LinuxTargetSettings"), *DevicePassKey, *DevicePass, GEngineIni);
387 }
388
389 ++DeviceIndex; // needs to be incremented here since we cannot allow gaps
390 }
391 }
392
393 bChangingDeviceConfig = false;
394 }
395#endif // WITH_ENGINE

564 // Set up which formats to include based on bMultiTargetFormat_* config keys.
565 // Pass all single-format target platform controls; this will filter and deduplicate them.
566 void LoadFormats(TArray<ITargetPlatformControls*> SingleFormatTPs)
567 {
568 // sort formats by priority so higher priority formats are packaged (and thus used by the device) first
569 // use GetVariantPriority() which has hardcoded defaults when the config key is absent
570 SingleFormatTPs.Sort([](const ITargetPlatformControls& A, const ITargetPlatformControls& B)
571 {
572 return A.GetVariantPriority() > B.GetVariantPriority();
573 });
574
575 FormatTargetPlatforms.Empty();
576 FormatTargetString = TEXT("");
577
578 // Server multi-targets don't cook visual texture data; nothing to populate.
579 if (TProperties::IsServerOnly())
580 {
581 return;
582 }
583
584 TSet<FString> SeenFormats;
585
586 // Only pick up single-format platforms that match our own type (Game/Client), so that
587 // e.g. MultiClientTP uses Client texture settings rather than the higher-priority Game variants.
588 const EBuildTargetType MyType = TProperties::IsClientOnly() ? EBuildTargetType::Client : EBuildTargetType::Game;
589
590 for (ITargetPlatformControls* SingleFormatTP : SingleFormatTPs)
591 {
592 FString FlavorName = SingleFormatTP->GetTargetPlatformInfo().PlatformFlavor.ToString();
593 if (FlavorName.IsEmpty())
594 {
595 continue;
596 }
597
598 const EBuildTargetType TPType = SingleFormatTP->GetTargetPlatformInfo().PlatformType;
599 // only use each flavor once, match our platform type, and skip server platforms (servers don't cook visual data)
600 if (SeenFormats.Contains(FlavorName) || TPType == EBuildTargetType::Server || TPType != MyType)
601 {
602 continue;
603 }
604 SeenFormats.Add(FlavorName);
605
606 bool bEnabled = false;
607 FString SettingsName = FString(TEXT("bMultiTargetFormat_")) + FlavorName;
608 FConfigCacheIni::ForPlatform(TEXT("Linux"))->GetBool(LINUX_SECTION_TEXT, *SettingsName, bEnabled, GEngineIni);
609 if (bEnabled)
610 {
611 if (FormatTargetPlatforms.Num())
612 {
613 FormatTargetString += TEXT(",");
614 }
615 FormatTargetString += FlavorName;
616 FormatTargetPlatforms.Add(SingleFormatTP);
617 }
618 }
619
620 PlatformInfo::UpdatePlatformDisplayName(
621 FString(TProperties::IniPlatformName()) + TEXT("_Multi"), DisplayName());
622 }
