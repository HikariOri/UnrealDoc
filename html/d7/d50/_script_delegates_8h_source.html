245 void ProcessEvent(WeakPtrType ObjectPtr, UFunction* Function, void* Parameters) const override
246 {
247 // The incoming Parameters will point to a block of arguments which were passed to the dynamic delegate
248 // invocation. These will be combined with the payload in order to produce a new parameters struct to
249 // be passed to the ProcessEvent call.
250
251 // If the payload replaces every argument, just pass the tuple itself as it should be the same layout.
252 if constexpr (DynamicDelegatePayloadIsLayoutCompatible<void(ParamTypes...), VarsTupleType>())
253 {
254 ObjectPtr.Get()->ProcessEvent(Function, (void*)&Payload);
255 }
256 // ... otherwise we need to combine Params and Payload into a single struct
257 else
258 {
259#if UE_DYNAMIC_DELEGATE_UNSCOPED_ENUM_HACK
260 using FullTupleType = TTuple<std::conditional_t<UE::CUnscopedEnum<std::decay_t<ParamTypes>>, TEnumAsByte<std::decay_t<ParamTypes>>, std::decay_t<ParamTypes>>...>;
261#else
262 using FullTupleType = TTuple<std::decay_t<ParamTypes>...>;
263#endif
264 using ParamsTupleType = TTuplePopBack_T<FullTupleType, sizeof...(VarTypes)>;
265
266 ((ParamsTupleType*)Parameters)->ApplyBefore(
267 [&Payload = Payload, ObjectPtr, Function](auto&&... Params)
268 {
269 Payload.ApplyAfter(
270 [&Payload = Payload, ObjectPtr, Function](auto&&... ParamsAndVars)
271 {
272 FullTupleType NewParameters(Forward<decltype(ParamsAndVars)>(ParamsAndVars)...);
273 ObjectPtr.Get()->ProcessEvent(Function, &NewParameters);
274 },
275 Forward<decltype(Params)>(Params)...
276 );
277 }
278 );
279 }

803 D.FunctionName = *FunctionNameWithOptionalPayloadFlagStr;
804 }

817#if UE_USE_DYNAMIC_DELEGATE_PAYLOADS

831#endif

837 }
838#endif
839 };
840
841 if (bSerializePayloadData)
842 {
843 if (bPayloadDataUsesCompactFormat && !UnderlyingAr.IsTextFormat())

845 // Binary v263+: bHasPayload already determined from suffix
846 if (bHasPayload)

848 SerializePayload(Record.EnterField(TEXT("Payload")));

857 }

982 template <class UObjectTemplate>
983 void ProcessDelegate( void* Parameters ) const
984 {
985 UObjectTemplate* ObjectPtr;
986 UFunction* Function;
987
988 { // to avoid MT access check if the delegate is deleted from inside of its callback, we don't cover the callback execution
989 // by access protection scope
990 // the `const` on the method is a lie
991 FWriteAccessScope WriteScope = const_cast<TScriptDelegate*>(this)->GetWriteAccessScope();
992
993 checkf( Object.IsValid() != false, TEXT( "ProcessDelegate() called with no object bound to delegate!" ) );
994 checkf( FunctionName != NAME_None, TEXT( "ProcessDelegate() called with no function name set!" ) );
995
996 // Object was pending kill, so we cannot execute the delegate. Note that it's important to assert
997 // here and not simply continue execution, as memory may be left uninitialized if the delegate is
998 // not able to execute, resulting in much harder-to-detect code errors. Users should always make
999 // sure IsBound() returns true before calling ProcessDelegate()!
1000 ObjectPtr = static_cast<UObjectTemplate*>(Object.Get()); // Down-cast
1001 checkSlow( IsValid(ObjectPtr) );
1002
1003 // Object *must* implement the specified function
1004 Function = ObjectPtr->FindFunctionChecked(FunctionName);
1005 }
1006
1007 // Execute the delegate!
1008#if UE_USE_DYNAMIC_DELEGATE_PAYLOADS
1009 if (this->Payload)
1010 {
1011 Payload->ProcessEvent(ObjectPtr, Function, Parameters);
1012 }
1013 else
1014#endif
1015 {
1016 ObjectPtr->ProcessEvent(Function, Parameters);
1017 }
1018 }
