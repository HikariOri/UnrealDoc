718{

720 {
721 DiagnosticCheck();

725 Min.Y = FMath::Min(Min.Y, Other.Y);
726 Min.Z = FMath::Min(Min.Z, Other.Z);

728 Max.X = FMath::Max(Max.X, Other.X);
729 Max.Y = FMath::Max(Max.Y, Other.Y);
730 Max.Z = FMath::Max(Max.Z, Other.Z);
731 }
732 else
733 {
734 Min = Max = Other;
735 IsValid = 1;
736 }
737
738 return *this;

767template<typename T>
768inline TVector<T> TBox<T>::GetClosestPointTo( const TVector<T>& Point ) const
769{
770 DiagnosticCheck();
771
772 // start by considering the point inside the box
773 TVector<T> ClosestPoint = Point;
774
775 // now clamp to inside box if it's outside
776 if (Point.X < Min.X)
777 {
778 ClosestPoint.X = Min.X;
779 }
780 else if (Point.X > Max.X)
781 {
782 ClosestPoint.X = Max.X;
783 }
784
785 // now clamp to inside box if it's outside
786 if (Point.Y < Min.Y)
787 {
788 ClosestPoint.Y = Min.Y;
789 }
790 else if (Point.Y > Max.Y)
791 {
792 ClosestPoint.Y = Max.Y;
793 }
794
795 // Now clamp to inside box if it's outside.
796 if (Point.Z < Min.Z)
797 {
798 ClosestPoint.Z = Min.Z;
799 }
800 else if (Point.Z > Max.Z)
801 {
802 ClosestPoint.Z = Max.Z;
803 }
804
805 return ClosestPoint;

869template<typename T>
870TBox<T> TBox<T>::TransformBy(const TMatrix<T>& M) const
871{
872 // if we are not valid, return another invalid box.
873 if (!IsValid)
874 {
875 return TBox<T>(ForceInit);
876 }
877
878 DiagnosticCheck();
879
880 TBox<T> NewBox;
881
882 const TVectorRegisterType<T> VecMin = VectorLoadFloat3_W0(&Min);
883 const TVectorRegisterType<T> VecMax = VectorLoadFloat3_W0(&Max);
884
885 const TVectorRegisterType<T> m0 = VectorLoadAligned(M.M[0]);
886 const TVectorRegisterType<T> m1 = VectorLoadAligned(M.M[1]);
887 const TVectorRegisterType<T> m2 = VectorLoadAligned(M.M[2]);
888 const TVectorRegisterType<T> m3 = VectorLoadAligned(M.M[3]);
889
890 const TVectorRegisterType<T> Half = VectorSetFloat1((T)0.5f); // VectorSetFloat1() can be faster than SetFloat3(0.5, 0.5, 0.5, 0.0). Okay if 4th element is 0.5, it's multiplied by 0.0 below and we discard W anyway.
891 const TVectorRegisterType<T> Origin = VectorMultiply(VectorAdd(VecMax, VecMin), Half);
892 const TVectorRegisterType<T> Extent = VectorMultiply(VectorSubtract(VecMax, VecMin), Half);
893
894 TVectorRegisterType<T> NewOrigin = VectorMultiply(VectorReplicate(Origin, 0), m0);
895 NewOrigin = VectorMultiplyAdd(VectorReplicate(Origin, 1), m1, NewOrigin);
896 NewOrigin = VectorMultiplyAdd(VectorReplicate(Origin, 2), m2, NewOrigin);
897 NewOrigin = VectorAdd(NewOrigin, m3);
898
899 TVectorRegisterType<T> NewExtent = VectorAbs(VectorMultiply(VectorReplicate(Extent, 0), m0));
900 NewExtent = VectorAdd(NewExtent, VectorAbs(VectorMultiply(VectorReplicate(Extent, 1), m1)));
901 NewExtent = VectorAdd(NewExtent, VectorAbs(VectorMultiply(VectorReplicate(Extent, 2), m2)));
902
903 const TVectorRegisterType<T> NewVecMin = VectorSubtract(NewOrigin, NewExtent);
904 const TVectorRegisterType<T> NewVecMax = VectorAdd(NewOrigin, NewExtent);
905
906 VectorStoreFloat3(NewVecMin, &(NewBox.Min.X));
907 VectorStoreFloat3(NewVecMax, &(NewBox.Max.X));
908
909 NewBox.IsValid = 1;
910
911 return NewBox;
912}

935template<typename T>
936TBox<T> TBox<T>::InverseTransformBy(const TTransform<T>& M) const
937{
938 // if we are not valid, return another invalid box.
939 if (!IsValid)
940 {
941 return TBox<T>(ForceInit);
942 }
943
944 TVector<T> Vertices[8];
945 GetVertices(Vertices);
946
947 TBox<T> NewBox(ForceInit);
948
949 for (int32 VertexIndex = 0; VertexIndex < UE_ARRAY_COUNT(Vertices); VertexIndex++)
950 {
951 TVector<T> ProjectedVertex = M.InverseTransformPosition(Vertices[VertexIndex]);
952 NewBox += ProjectedVertex;
953 }
954
955 return NewBox;
956}

958template<typename T>
959TBox<T> TBox<T>::TransformProjectBy(const TMatrix<T>& ProjM) const
960{
961 // if we are not valid, return another invalid box.
962 if (!IsValid)
963 {
964 return TBox<T>(ForceInit);
965 }
966
967 TVector<T> Vertices[8];
968 GetVertices(Vertices);
969
970 TBox<T> NewBox(ForceInit);
971
972 for (int32 VertexIndex = 0; VertexIndex < UE_ARRAY_COUNT(Vertices); VertexIndex++)
973 {
974 TVector4<T> ProjectedVertex = ProjM.TransformPosition(Vertices[VertexIndex]);
975 NewBox += ((TVector<T>)ProjectedVertex) / ProjectedVertex.W;
976 }
977
978 return NewBox;

981template<typename T>
982TBox<T> TBox<T>::Overlap(const TBox<T>& Other) const
983{
984 if (Intersect(Other) == false)
985 {
986 static TBox<T> EmptyBox(ForceInit);
987 return EmptyBox;
988 }
989
990 // otherwise they overlap
991 // so find overlapping box
992 TVector<T> MinVector, MaxVector;
993
994 MinVector.X = FMath::Max(Min.X, Other.Min.X);
995 MaxVector.X = FMath::Min(Max.X, Other.Max.X);
996
997 MinVector.Y = FMath::Max(Min.Y, Other.Min.Y);
998 MaxVector.Y = FMath::Min(Max.Y, Other.Max.Y);
999
1000 MinVector.Z = FMath::Max(Min.Z, Other.Min.Z);
1001 MaxVector.Z = FMath::Min(Max.Z, Other.Max.Z);
1002
1003 return TBox<T>(MinVector, MaxVector);
1004}
