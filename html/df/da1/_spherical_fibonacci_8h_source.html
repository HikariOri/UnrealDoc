45 TVector<RealType> Point(int32 Index) const

49 checkSlow(Index >= 0 && Index < N);

51 RealType phi = TMathUtil<RealType>::TwoPi * (div - TMathUtil<RealType>::Floor(div));

75 int32 FindIndex(const TVector<RealType>& P)
76 {
77 constexpr RealType PHI = static_cast<RealType>(UE_DOUBLE_GOLDEN_RATIO);
78
79 RealType phi = TMathUtil<RealType>::Min(TMathUtil<RealType>::Atan2(P.Y, P.X), TMathUtil<RealType>::Pi);
80 RealType cosTheta = P.Z;
81 RealType k = TMathUtil<RealType>::Max(2.0, TMathUtil<RealType>::Floor(
82 TMathUtil<RealType>::Log(N * TMathUtil<RealType>::Pi * TMathUtil<RealType>::Sqrt(5.0) * (1.0 - cosTheta * cosTheta)) / TMathUtil<RealType>::Log(PHI * PHI)));
83 RealType Fk = TMathUtil<RealType>::Pow(PHI, k) / TMathUtil<RealType>::Sqrt(5.0);
84
85 RealType F0 = TMathUtil<RealType>::Round(Fk);
86 RealType F1 = TMathUtil<RealType>::Round(Fk * PHI);
87
88 TMatrix2<RealType> B(
89 2.0 * TMathUtil<RealType>::Pi * MultiplyAddFrac(F0 + 1.0, PHI - 1.0) - 2.0 * TMathUtil<RealType>::Pi * (PHI - 1),
90 2.0 * TMathUtil<RealType>::Pi * MultiplyAddFrac(F1 + 1.0, PHI - 1.0) - 2.0 * TMathUtil<RealType>::Pi * (PHI - 1),
91 -2.0 * F0 / N, -2.0 * F1 / N);
92 TMatrix2<RealType> invB = B.Inverse();
93
94 //Vector2d c = floor(mul(invB, RealType2(phi, cosTheta - (1 - 1.0/N))));
95 TVector2<RealType> c(phi, cosTheta - (1.0 - 1.0/N));
96 c = invB * c;
97 c.X = TMathUtil<RealType>::Floor(c.X); c.Y = TMathUtil<RealType>::Floor(c.Y);
98
99 RealType d = TMathUtil<RealType>::MaxReal, j = 0;
100 for (int32 s = 0; s < 4; ++s)
101 {
102 TVector2<RealType> cosTheta_second((RealType)(s % 2) + c.X, (RealType)(s / 2) + c.Y);
103 cosTheta = B.Row1.Dot(cosTheta_second) + (1.0 - 1.0 / N);
104 cosTheta = TMathUtil<RealType>::Clamp(cosTheta, -1.0, +1.0) * 2.0 - cosTheta;
105 RealType i = TMathUtil<RealType>::Floor(N * 0.5 - cosTheta * N * 0.5);
106 phi = 2.0 * TMathUtil<RealType>::Pi * MultiplyAddFrac(i, PHI - 1);
107 cosTheta = 1.0 - (2.0 * i + 1.0) * (1.0 / N); // rcp(n);
108 RealType sinTheta = TMathUtil<RealType>::Sqrt(1.0 - cosTheta * cosTheta);
109 TVector<RealType> q(
110 TMathUtil<RealType>::Cos(phi) * sinTheta,
111 TMathUtil<RealType>::Sin(phi) * sinTheta,
112 cosTheta );
113 RealType SquaredDistance = DistanceSquared(q, P);
114 if (SquaredDistance < d)
115 {
116 d = SquaredDistance;
117 j = i;
118 }
119 }
120
121 return (int32)j;
122 }

131 {
132 return a * b + -TMathUtil<RealType>::Floor(a * b);
133 }

145template<typename RealType>
146class TFibonacciLattice
147{
148public:
149 int32 N = 64;
150
151 enum class EType
152 {
153 Square, // Fibonacci lattice on unit square
154 Disc // Fibonacci lattice on unit disc (spiral)
155 };
156 EType Type = EType::Square;
157
158public:
159 TFibonacciLattice(int32 NumPoints = 64, EType InType = EType::Square)
160 : N(NumPoints), Type(InType)
161 {
162 }
163
164 int32 Num() const
165 {
166 return N;
167 }
168
173 TVector2<RealType> Point(int32 Index) const
174 {
175 constexpr RealType PHI = static_cast<RealType>(UE_DOUBLE_GOLDEN_RATIO);
176 checkSlow(Index >= 0 && Index < N);
177
178 // Use 1-based indices to exclude the zero/origin term.
179 // Internally remap the range: [1, N+1].
180 const int32 Idx = Index + 1;
181 const int32 NumPts = N + 1;
182
183 // Unit Square Lattice:
184 // (xs,ys) --> (modf(i/phi, unused), i/n)
185 RealType Div = (RealType)Idx / PHI;
186 RealType X = (Div - TMathUtil<RealType>::Floor(Div));
187 RealType Y = (RealType)Idx / (RealType)NumPts;
188
189 if (Type == EType::Disc)
190 {
191 // Unit Disc Lattice:
192 // (theta, r) --> (2*pi*xs, sqrt(ys))
193 // (xd, yd) --> (r*cos(theta), r*sin(theta))
194 RealType Theta = TMathUtil<RealType>::TwoPi * X;
195 RealType R = TMathUtil<RealType>::Sqrt(Y);
196 X = R * TMathUtil<RealType>::Cos(Theta);
197 Y = R * TMathUtil<RealType>::Sin(Theta);
198 }
199 return TVector2<RealType>(X,Y);
200 }
201
206 TVector2<RealType> operator[](int32 Index) const
207 {
208 return Point(Index);
209 }
210};

219template<typename RealType>
220class THemisphericalFibonacci
221{
222public:
223 int32 N = 64;
224
225 enum class EDistribution
226 {
227 Uniform,
228 Cosine
229 };
230 EDistribution Distribution = EDistribution::Uniform;
231
232public:
233 THemisphericalFibonacci(int32 NumPoints = 64, EDistribution Dist = EDistribution::Uniform)
234 : N(NumPoints), Distribution(Dist)
235 {
236 }
237
238 int32 Num() const
239 {
240 return N;
241 }
242
247 TVector<RealType> Point(int32 Index) const
248 {
249 checkSlow(Index >= 0 && Index < N);
250
251 TVector<RealType> Point = {};
252 switch (Distribution)
253 {
254 case EDistribution::Uniform:
255 {
256 TSphericalFibonacci<RealType> Points(2 * N);
257 Point = Points[Index];
258 break;
259 }
260 case EDistribution::Cosine:
261 {
262 TFibonacciLattice<RealType> Points(N, TFibonacciLattice<RealType>::EType::Disc);
263 const TVector2<RealType> Pt = Points[Index];
264
265 // Planar projection of Fibonacci spiral (unit disc [xd, yd]) to
266 // hemisphere to achieve cosine weighted ray distribution.
267 // (Malley's method).
268 //
269 // Hemisphere: (x,y,z) --> (xd, yd, 1-xd^2-yd^2)
270 //
271 // Reference: Physically Based Rendering: From Theory to Implementation
272 // http://www.pbr-book.org/3ed-2018/Monte_Carlo_Integration/2D_Sampling_with_Multidimensional_Transformations.html#Cosine-WeightedHemisphereSampling
273 RealType Z = TMathUtil<RealType>::Sqrt(TMathUtil<RealType>::Max(0.0, 1.0 - Pt.X * Pt.X - Pt.Y * Pt.Y));
274 Point = TVector<RealType>(Pt.X, Pt.Y, Z);
275 break;
276 }
277 }
278 return Point;
279 }
280
285 TVector<RealType> operator[](int32 Index) const
286 {
287 return Point(Index);
288 }
289};
