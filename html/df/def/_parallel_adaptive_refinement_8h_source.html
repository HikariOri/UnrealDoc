22 struct FTriSplitRecord
23 {
24 uint16 SplitMask { 0 }; //< bitmask: per-edge refinement tagging [0..3)
25 std::atomic<uint16> GreenAdjMask { 0 }; //< bitmask: encode which of the neighbors [0..3) are green refinement
26 std::atomic_flag bClaimed = ATOMIC_FLAG_INIT;
27
28 inline void Reset()
29 {
30 SplitMask = 0;
31 GreenAdjMask.store(0);
32 bClaimed.clear();
33 }
34
35 // split one edge (green refinement)
36 bool IsSplit1() const
37 {
38 return SplitMask == 1 || SplitMask == 2 || SplitMask == 4;
39 }
40
41 // split two edges
42 bool IsSplit2() const
43 {
44 return SplitMask == 3 || SplitMask == 5 || SplitMask == 6;
45 }
46
47 // split all three edges (red refinement)
48 bool IsSplit3() const
49 {
50 return SplitMask == 7;
51 }
52 };
53
54 struct FEdgeSplitRecord
55 {
56 void Reset()
57 {
58 SplitVertex = IndexConstants::InvalidID;
59 HalfEdges[0] = FIndex2i(IndexConstants::InvalidID, IndexConstants::InvalidID);
60 HalfEdges[1] = FIndex2i(IndexConstants::InvalidID, IndexConstants::InvalidID);
61 NewEdge = IndexConstants::InvalidID;
62
63 bTagged.clear();
64 EntryCountPreWrite.store(0);
65 EntryCountPostWrite.store(0);
66 }
67
68 int32 SplitVertex { IndexConstants::InvalidID };
69
70 // each entry pair corresponds to the half-edge indices of one side of the split
71 FIndex2i HalfEdges[2] { FIndex2i(IndexConstants::InvalidID, IndexConstants::InvalidID),
72 FIndex2i(IndexConstants::InvalidID, IndexConstants::InvalidID) };
73
74 int32 NewEdge { IndexConstants::InvalidID };
75
76 std::atomic_flag bTagged = ATOMIC_FLAG_INIT; // tagged for refinement
77 std::atomic<int32> EntryCountPreWrite { 0 }; // current number of entries, before write
78 std::atomic<int32> EntryCountPostWrite { 0 }; // number of half-edge pairs written
79 };

81 struct FConcurrencyState
82 {
83 TArray<FEdgeSplitRecord> EdgeSplitRecords;
84 TArray<FTriSplitRecord> TriSplitRecords;
85 std::atomic<int32> HalfEdgeCount; // current number of halfedges, for parallel append
86 std::atomic<int32> EdgeCount; // current number of edges, for parallel append

88 void Resize(const int32 NumTris, const int32 NumEdges)
89 {
90 TRACE_CPUPROFILER_EVENT_SCOPE_TEXT("TParallelAdaptiveRefinement.FConcurrencyState.Resize");
91
92 for (FTriSplitRecord& Record : TriSplitRecords)
93 {
94 Record.Reset();
95 }
96 if (NumTris > TriSplitRecords.Num())
97 {
98 TriSplitRecords.AddDefaulted(NumTris - TriSplitRecords.Num());
99 }
100 check(TriSplitRecords.Num() == NumTris);
101
102 for (FEdgeSplitRecord &Record : EdgeSplitRecords)
103 {
104 Record.Reset();
105 }
106
107 if (NumEdges > EdgeSplitRecords.Num())
108 {
109 EdgeSplitRecords.AddDefaulted(NumEdges - EdgeSplitRecords.Num());
110 }
111 }
112 };

118 struct FOptions
119 {
120 int32 ParallelForBatchSize { 1024 };
121
122 //< pre-reserve enough space for growing up to this many vertices

128 // Initialize and run adaptive refinement
129 TParallelAdaptiveRefinement(
130 MeshType& InMesh, //< mesh interface
131 RefinementPolicyType& InRefinementPolicy, //< displacement and error interface
132 const FOptions& InOptions)
133 : Mesh(InMesh)
134 , RefinementPolicy(InRefinementPolicy)
135 , Options(InOptions)
136 {
137 const int ParallelForBatchSize = Options.ParallelForBatchSize;
138 const EParallelForFlags ParallelForFlags = ParallelForBatchSize > 0 ? EParallelForFlags::None : EParallelForFlags::ForceSingleThread;
139
140 const int32 ReserveTriangles = 2 * Options.ReserveVertices;
141 const int32 ReserveEdges = 3 * Options.ReserveVertices;
142
143 FConcurrencyState ConcurrencyState;
144 TArray<int32> RefinementCand; //< list of triangles to inspect for refinement
145 TArray<int32> TriSplitQueue; //< list of triangle indices scheduled for refinement
146 TArray<int32> EdgeSplitRequests; //< list of edge indices scheduled for refinement
147
148 RefinementCand.Reserve(ReserveTriangles);
149 TriSplitQueue.Reserve(ReserveTriangles);
150 EdgeSplitRequests.Reserve(ReserveEdges);
151
152 ConcurrencyState.TriSplitRecords.Reserve(ReserveTriangles);
153 ConcurrencyState.EdgeSplitRecords.Reserve(ReserveEdges);
154
155 Mesh.ReserveVertices(Options.ReserveVertices);
156 Mesh.ReserveTriangles(ReserveTriangles);
157
158 RefinementCand.Init(0, Mesh.MaxTriID());
159 for (int32 TriangleIdx(0); TriangleIdx<Mesh.MaxTriID(); ++TriangleIdx)
160 {
161 RefinementCand[TriangleIdx] = TriangleIdx;
162 }
163
164 std::atomic<int32> NumRefinementCand(RefinementCand.Num());
165
166 for (int32 Level(0); Level < 32; ++Level)
167 {
168 if (NumRefinementCand.load() == 0) {
169 break;
170 }
171
172 {
173 TRACE_CPUPROFILER_EVENT_SCOPE_TEXT("TParallelAdaptiveRefinement.ResizeRequests");
174 // reserve sufficient space for tris and edges to be split
175 TriSplitQueue.SetNum(Mesh.MaxTriID());
176 EdgeSplitRequests.SetNum(Mesh.EdgeCount());
177 ConcurrencyState.Resize(Mesh.MaxTriID(), Mesh.EdgeCount());
178 }
179
180 check(ConcurrencyState.TriSplitRecords.Num() == Mesh.MaxTriID());
181
182 std::atomic<int32> TriSplitQueueSize(0);
183 std::atomic<int32> NumEdgesToSplit(0);
184
185 auto TagForRefine = [&](const int32 Idx)
186 {
187 const int32 TriIndex = RefinementCand[Idx];
188
189 if (!Mesh.IsValidTri(TriIndex))
190 {
191 return;
192 }
193
194 if (ShouldRefine(TriIndex, Level))
195 {
196 if (!ConcurrencyState.TriSplitRecords[TriIndex].bClaimed.test_and_set())
197 {
198 TriSplitQueue[TriSplitQueueSize.fetch_add(1, std::memory_order_relaxed)] = TriIndex;
199 ConcurrencyState.TriSplitRecords[TriIndex].SplitMask = 7;
200
201 for (int32 LocalEdgeIdx=0; LocalEdgeIdx<3; ++LocalEdgeIdx)
202 {
203 const int32 EdgeIdx = Mesh.GetEdgeIndex(TriIndex, LocalEdgeIdx);
204 FEdgeSplitRecord& Edge = ConcurrencyState.EdgeSplitRecords[EdgeIdx];
205
206 // check whether this is the first time visiting this edge
207 if (!Edge.bTagged.test_and_set(std::memory_order_relaxed))
208 {
209 EdgeSplitRequests[ NumEdgesToSplit.fetch_add(1, std::memory_order_relaxed) ] = EdgeIdx;
210 }
211 }
212 }
213
214 if (Options.bApplyNeighborhoodRefinement)
215 {
216 // for more gradual mesh regularity, tag neighbors for refinement.
217 for (int NbIdx=0; NbIdx<3; ++NbIdx)
218 {
219 const int32 AdjTriIndex = Mesh.GetAdjTriangle(TriIndex, NbIdx);
220 if (AdjTriIndex != IndexConstants::InvalidID && RefinementPolicy.AllowRefine(AdjTriIndex, Level) &&
221 !ConcurrencyState.TriSplitRecords[AdjTriIndex].bClaimed.test_and_set())
222 {
223 check(Mesh.IsValidTri(AdjTriIndex));
224
225 TriSplitQueue[TriSplitQueueSize.fetch_add(1, std::memory_order_relaxed)] = AdjTriIndex;
226 ConcurrencyState.TriSplitRecords[AdjTriIndex].SplitMask = 7;
227
228 for (int32 LocalEdgeIdx=0; LocalEdgeIdx<3; ++LocalEdgeIdx)
229 {
230 const int32 EdgeIdx = Mesh.GetEdgeIndex(AdjTriIndex, LocalEdgeIdx);
231 FEdgeSplitRecord& Edge = ConcurrencyState.EdgeSplitRecords[EdgeIdx];
232
233 // check whether this is the first time visiting this edge
234 if (!Edge.bTagged.test_and_set(std::memory_order_relaxed))
235 {
236 EdgeSplitRequests[ NumEdgesToSplit.fetch_add(1, std::memory_order_relaxed) ] = EdgeIdx;
237 }
238 }
239 }
240 }
241 }
242 }
243 };
244
245 // tag triangles for refinement;
246 ParallelForTemplate( TEXT("TParallelAdaptiveRefinement.TagForRefine.PF"), NumRefinementCand.load(), ParallelForBatchSize, TagForRefine, ParallelForFlags );
247
248 NumRefinementCand.store(0);
249 if (TriSplitQueueSize.load() == 0)
250 {
251 break;
252 }
253
254 std::atomic<int32> NumNewInteriorEdges(0);
255 std::atomic<int32> NumNewTriangles(0);
256
257 // identify neighboring triangles closure split masks (T-junctions)
258 ParallelFor( TEXT("TParallelAdaptiveRefinement.ResolveNeighbors.PF"), Mesh.MaxTriID(), ParallelForBatchSize, [&](const int32 TriIndex)
259 {
260 if (!Mesh.IsValidTri(TriIndex))
261 {
262 return;
263 }
264
265 int32 SplitMask = ConcurrencyState.TriSplitRecords[TriIndex].SplitMask;
266
267 if (SplitMask != 0)
268 {
269 check(SplitMask == 7);
270 }
271 else
272 {
273 for (int LocalEdgeIdx=0; LocalEdgeIdx<3; ++LocalEdgeIdx)
274 {
275 const int32 AdjTriIndex = Mesh.GetAdjTriangle(TriIndex, LocalEdgeIdx);
276 if (AdjTriIndex != IndexConstants::InvalidID && ConcurrencyState.TriSplitRecords[AdjTriIndex].IsSplit3())
277 {
278 SplitMask |= 1 << LocalEdgeIdx;
279 }
280 }
281 }
282
283 if (ConcurrencyState.TriSplitRecords[TriIndex].SplitMask == 0 && SplitMask != 0)
284 {
285 // record that this triangle needs to be split as a consequence as well
286 TriSplitQueue[TriSplitQueueSize.fetch_add(1, std::memory_order_relaxed)] = TriIndex;
287 }
288
289 ConcurrencyState.TriSplitRecords[TriIndex].SplitMask = SplitMask;
290 ConcurrencyState.TriSplitRecords[TriIndex].GreenAdjMask.store(uint16(0));
291
292 if (SplitMask == 7)
293 {
294 // full split
295 NumNewInteriorEdges.fetch_add(3, std::memory_order_relaxed);
296 NumNewTriangles.fetch_add(3, std::memory_order_relaxed);
297 }
298 else if (SplitMask == 1 || SplitMask == 2 || SplitMask == 4)
299 {
300 // green refinement
301 NumNewInteriorEdges.fetch_add(1, std::memory_order_relaxed);
302 NumNewTriangles.fetch_add(1, std::memory_order_relaxed);
303 }
304 else if (SplitMask != 0)
305 {
306 NumNewInteriorEdges.fetch_add(2, std::memory_order_relaxed);
307 NumNewTriangles.fetch_add(2, std::memory_order_relaxed);
308 }
309 }, ParallelForFlags );
310
311 // for each green refined triangles, store the edge mask for each neighboring triangle that is marked for refinement
312 ParallelFor( TEXT("TParallelAdaptiveRefinement.IdentifyGreenNeighbors.PF"), TriSplitQueueSize.load(), ParallelForBatchSize, [&](int32 Idx)
313 {
314 const int32 TriIndex = TriSplitQueue[Idx];
315 check(ConcurrencyState.TriSplitRecords[TriIndex].SplitMask != 0);
316
317 if (ConcurrencyState.TriSplitRecords[TriIndex].IsSplit1())
318 {
319 // mark neighbors
320 for (int32 NbIdx=0; NbIdx<3; ++NbIdx)
321 {
322 const TPair<int32,int32> TriEdge = Mesh.GetAdjEdge(TriIndex, NbIdx);
323 if (TriEdge.Get<0>() != IndexConstants::InvalidID && ConcurrencyState.TriSplitRecords[TriEdge.Get<0>()].SplitMask != 0)
324 {
325 ConcurrencyState.TriSplitRecords[TriEdge.Get<0>()].GreenAdjMask.fetch_or(static_cast<uint16>(1u << TriEdge.Get<1>()));
326 }
327 }
328 }
329 }, ParallelForFlags);
330
331 // grow the arrays for edges and vertices
332 int32 PrevEdgeCount, PrevVtxCount;
333 {
334 TRACE_CPUPROFILER_EVENT_SCOPE_TEXT("TParallelAdaptiveRefinement.GrowVerts");
335
336 const int32 NewVertexCount = NumEdgesToSplit.load();
337 const int32 NewEdgeCount = NewVertexCount + NumNewInteriorEdges.load();
338
339 ConcurrencyState.EdgeCount.store(Mesh.EdgeCount() + NewVertexCount);
340
341 PrevEdgeCount = Mesh.AddEdges(NewEdgeCount);
342 PrevVtxCount = Mesh.AddVertices(NewVertexCount);
343
344 ConcurrencyState.EdgeSplitRecords.AddDefaulted( NewEdgeCount );
345 check(ConcurrencyState.EdgeSplitRecords.Num() == Mesh.EdgeCount());
346
347 RefinementPolicy.AllocateVertices(NewVertexCount);
348 }
349
350 // assign new edge-midpoints, assign displacement
351 ParallelForTemplate(TEXT("TParallelAdaptiveRefinement.MakeEdgeMidpoints.PF"), NumEdgesToSplit.load(), ParallelForBatchSize, [&](const int32 RequestIdx)
352 {
353 const int32 EdgeIndex = EdgeSplitRequests[RequestIdx];
354 FEdgeSplitRecord& Edge = ConcurrencyState.EdgeSplitRecords[EdgeIndex];
355
356 Edge.NewEdge = PrevEdgeCount + RequestIdx;
357 Edge.SplitVertex = PrevVtxCount + RequestIdx;
358
359 FIndex2i VertexIndices, Tris;
360
361 Mesh.GetEdge(EdgeIndex, VertexIndices, Tris);
362 Mesh.InterpolateVertex(0.5f, Edge.SplitVertex, EdgeIndex);
363 RefinementPolicy.VertexAdded(Edge.SplitVertex, Tris);
364
365 }, ParallelForFlags );
366
367 ConcurrencyState.HalfEdgeCount.store(Mesh.HalfEdgeCount());
368 {
369 TRACE_CPUPROFILER_EVENT_SCOPE_TEXT("TParallelAdaptiveRefinement.GrowHalfEdges");
370
371 Mesh.AddTriangles(NumNewTriangles.load());
372 RefinementCand.AddDefaulted(NumNewTriangles.load());
373 }
374
375 // walk through refinement queue, apply refinement, enqueue triangles to check for refinement
376 ParallelFor( TEXT("TParallelAdaptiveRefinement.RefineTri.PF"), TriSplitQueueSize.load(), ParallelForBatchSize, [&](int32 Idx)
377 {
378 const int32 TriIndex = TriSplitQueue[Idx];
379 check(ConcurrencyState.TriSplitRecords[TriIndex].SplitMask != 0);
380
381 if (ConcurrencyState.TriSplitRecords[TriIndex].IsSplit3())
382 {
383 const int32 FirstNewTri = Split3(TriIndex, ConcurrencyState);
384
385 // enqueue all 4 children as refinement candidates
386 const int32 QueueIdx = NumRefinementCand.fetch_add(4, std::memory_order_relaxed);
387 RefinementCand[QueueIdx+0] = FirstNewTri+0;
388 RefinementCand[QueueIdx+1] = FirstNewTri+1;
389 RefinementCand[QueueIdx+2] = FirstNewTri+2;
390 RefinementCand[QueueIdx+3] = TriIndex;
391 }
392 else if (ConcurrencyState.TriSplitRecords[TriIndex].IsSplit2())
393 {
394 Split2(TriIndex, ConcurrencyState);
395 }
396 else if (ConcurrencyState.TriSplitRecords[TriIndex].IsSplit1())
397 {
398 Split1(TriIndex, ConcurrencyState);
399 }
400
401 // All triangle split records should be in clean state after this loop
402 ConcurrencyState.TriSplitRecords[TriIndex].Reset();
403 }, ParallelForFlags );
404
405 check(ConcurrencyState.HalfEdgeCount.load() == Mesh.HalfEdgeCount());
406
407 // Mesh.CheckConsistency();
408
409 TriSplitQueueSize.store(0);
410 }
411 }

413 // split 1 edge, keep two edges (1 new triangle)
414 inline void Split1(int32 TriIndex, FConcurrencyState& ConcurrencyState)
415 {
416 const int32 SplitMask = ConcurrencyState.TriSplitRecords[TriIndex].SplitMask;
417 check(ConcurrencyState.TriSplitRecords[TriIndex].IsSplit1());
418
419 // determine the edge that is split
420 int32 LocSplitEdge = 0;
421 if (SplitMask == 2) LocSplitEdge = 1;
422 else if (SplitMask == 4) LocSplitEdge = 2;
423
424 const int32 NextLocEdge = (LocSplitEdge+1)%3;
425 const int32 PrevLocEdge = (LocSplitEdge+2)%3;
426
427 const int32 EdgeIndices[3] = { Mesh.GetEdgeIndex(TriIndex, 0),
428 Mesh.GetEdgeIndex(TriIndex, 1),
429 Mesh.GetEdgeIndex(TriIndex, 2) };
430
431 const FIndex3i BaseTri = Mesh.GetTriangle(TriIndex);
432
433 const int32 SplitEdgeIdx = Mesh.GetEdgeIndex(TriIndex, LocSplitEdge);
434 const int32 SplitVertexIdx = ConcurrencyState.EdgeSplitRecords[SplitEdgeIdx].SplitVertex;
435 check(SplitVertexIdx != IndexConstants::InvalidID);
436
437 const int32 AdjHalfEdges[3] = { Mesh.GetAdjHalfEdge(TriIndex, 0),
438 Mesh.GetAdjHalfEdge(TriIndex, 1),
439 Mesh.GetAdjHalfEdge(TriIndex, 2) };
440
441 check(AdjHalfEdges[LocSplitEdge] != IndexConstants::InvalidID);
442
443 // the adjacency for the outer edge doesn't change
444 FIndex3i NewTri = BaseTri;
445 NewTri[NextLocEdge] = SplitVertexIdx;
446
447 Mesh.SetTriangle(TriIndex, NewTri, TriIndex);
448
449 // allocate 3 new half-edges and indices. this is the first index to write to
450 const int32 HalfEdge = ConcurrencyState.HalfEdgeCount.fetch_add(3, std::memory_order_relaxed);
451 check(HalfEdge + 2 < Mesh.MaxTriID() * 3);
452
453 // first new triangle
454
455 const int32 NewTriIndex = HalfEdge / 3;
456 Mesh.SetTriangle(NewTriIndex, FIndex3i(SplitVertexIdx, BaseTri[NextLocEdge], BaseTri[PrevLocEdge]), TriIndex);
457
458 // for the interior edge
459 const int32 NewEdgeIndex = ConcurrencyState.EdgeCount.fetch_add(1, std::memory_order_relaxed);
460 check(NewEdgeIndex < Mesh.EdgeCount());
461
462 // interior edge
463 Mesh.LinkEdge(HalfEdge + 2, TriIndex * 3 + NextLocEdge, NewEdgeIndex, IndexConstants::InvalidID);
464
465 struct FLinkTask
466 {
467 int32 HalfEdge; //< halfedge to relink
468 int32 LocalEdge; //< local edge index
469 int32 AdjHalfEdge; //< adjacent half-edge before the update
470 int32 Edge; //<
471 };
472
473 const FLinkTask LinkTasks[2] = { {TriIndex * 3 + PrevLocEdge, PrevLocEdge, AdjHalfEdges[PrevLocEdge], EdgeIndices[PrevLocEdge] },
474 {HalfEdge + 1 , NextLocEdge, AdjHalfEdges[NextLocEdge], EdgeIndices[NextLocEdge] } };
475
476
477 // update adjacency for the unsplit edges
478 for (const FLinkTask& LinkTask : LinkTasks )
479 {
480 // only if the neighbor is of type split-1, we need a safe concurrent update. split-2 is constructed such that the unsplit
481 // edge doesn't require an adjacency update
482 if (LinkTask.AdjHalfEdge == IndexConstants::InvalidID || !(ConcurrencyState.TriSplitRecords[TriIndex].GreenAdjMask.load() & static_cast<uint16>(1 << LinkTask.LocalEdge)))
483 {
484 Mesh.LinkEdge(LinkTask.HalfEdge, LinkTask.AdjHalfEdge, LinkTask.Edge, LinkTask.Edge);
485 }
486 else
487 {
488 FEdgeSplitRecord& Edge = ConcurrencyState.EdgeSplitRecords[LinkTask.Edge];
489 const int32 EntryCountPreWrite = Edge.EntryCountPreWrite.fetch_add(1, std::memory_order_relaxed);
490 check(EntryCountPreWrite < 2);
491
492 Edge.HalfEdges[EntryCountPreWrite] = FIndex2i( LinkTask.HalfEdge, IndexConstants::InvalidID );
493
494 if (1 == Edge.EntryCountPostWrite.fetch_add(1, std::memory_order_acq_rel))
495 {
496 check(Edge.HalfEdges[0][1] == IndexConstants::InvalidID);
497 check(Edge.HalfEdges[1][1] == IndexConstants::InvalidID);
498
499 // second entry is being written, so we can now link the edges
500 Mesh.LinkEdge(Edge.HalfEdges[0][0], Edge.HalfEdges[1][0], LinkTask.Edge, LinkTask.Edge);
501 }
502 }
503 }
504
505 // update adjacency for the split edge
506 FIndex2i BoundaryHalfEdges = { 3 * TriIndex + LocSplitEdge, HalfEdge + 0 };
507 {
508 const int32 EdgeIdx = EdgeIndices[LocSplitEdge];
509 FEdgeSplitRecord& Edge = ConcurrencyState.EdgeSplitRecords[EdgeIdx];
510
511 const int32 EntryCountPreWrite = Edge.EntryCountPreWrite.fetch_add(1, std::memory_order_relaxed);
512 check(EntryCountPreWrite < 2);
513
514 Edge.HalfEdges[EntryCountPreWrite] = BoundaryHalfEdges;
515
516 // second atomic necessary to make sure the memory written to in other thread in slot 0 is available
517 // todo: might be faster to use std::memory_order_relaxed and use a atomic_thread_fence
518 if (1 == Edge.EntryCountPostWrite.fetch_add(1, std::memory_order_acq_rel))
519 {
520 check(Edge.HalfEdges[0][0] != IndexConstants::InvalidID);
521 check(Edge.HalfEdges[0][1] != IndexConstants::InvalidID);
522 check(Edge.HalfEdges[1][0] != IndexConstants::InvalidID);
523 check(Edge.HalfEdges[1][1] != IndexConstants::InvalidID);
524
525 // second entry is being written, so we can now link the edges
526 Mesh.LinkEdge(Edge.HalfEdges[0][0], Edge.HalfEdges[1][1], EdgeIdx, EdgeIdx);
527 Mesh.LinkEdge(Edge.HalfEdges[0][1], Edge.HalfEdges[1][0], Edge.NewEdge, EdgeIdx);
528 }
529 }
530 }

532 // split 2 edges, keep one edge (2 new triangles)
533 inline void Split2(int32 TriIndex, FConcurrencyState& ConcurrencyState)
534 {
535 const FIndex3i BaseTri = Mesh.GetTriangle(TriIndex);
536
537 const int32 SplitMask = ConcurrencyState.TriSplitRecords[TriIndex].SplitMask;
538
539 int32 LocPreserveEdge = 0; // the one edge that is NOT split
540 if (SplitMask == 3) {
541 LocPreserveEdge = 2;
542 } else if (SplitMask == 5) {
543 LocPreserveEdge = 1;
544 } else {
545 check(SplitMask == 6);
546 LocPreserveEdge = 0;
547 }
548
549 // edges to reconnect
550 const int32 EdgeIndices[2] = { Mesh.GetEdgeIndex(TriIndex, (LocPreserveEdge+1)%3),
551 Mesh.GetEdgeIndex(TriIndex, (LocPreserveEdge+2)%3) };
552
553 const int32 SplitVertexIdx0 = ConcurrencyState.EdgeSplitRecords[EdgeIndices[0]].SplitVertex;
554 const int32 SplitVertexIdx1 = ConcurrencyState.EdgeSplitRecords[EdgeIndices[1]].SplitVertex;
555
556 const int32 h0 = TriIndex * 3 + (LocPreserveEdge+0);
557 const int32 h1 = TriIndex * 3 + (LocPreserveEdge+1)%3;
558 const int32 h2 = TriIndex * 3 + (LocPreserveEdge+2)%3;
559
560 // the tri connected to the unsplit edge
561 FIndex3i NewTri = BaseTri;
562 NewTri[(LocPreserveEdge + 2) % 3] = SplitVertexIdx1;
563 Mesh.SetTriangle(TriIndex, NewTri, TriIndex);
564
565 // allocate 6 new half-edges and indices. this is the first index to write to
566 const int32 HalfEdge = ConcurrencyState.HalfEdgeCount.fetch_add(6, std::memory_order_relaxed);
567 check(HalfEdge + 5 < Mesh.MaxTriID() * 3);
568
569 // two new interior edges
570 const int32 NewEdgeIndex = ConcurrencyState.EdgeCount.fetch_add(2, std::memory_order_relaxed);
571
572
573 const int32 NewTriIndex = HalfEdge / 3;
574
575 // T0
576 Mesh.SetTriangle(NewTriIndex + 0, FIndex3i(BaseTri[(LocPreserveEdge + 2) % 3], SplitVertexIdx1, SplitVertexIdx0), TriIndex);
577
578 // T1
579 Mesh.SetTriangle(NewTriIndex + 1, FIndex3i(SplitVertexIdx0, SplitVertexIdx1, BaseTri[(LocPreserveEdge + 1) % 3]), TriIndex);
580
581 // interior edges
582 Mesh.LinkEdge(HalfEdge + 1, HalfEdge + 3, NewEdgeIndex , IndexConstants::InvalidID);
583 Mesh.LinkEdge(HalfEdge + 4, h1 , NewEdgeIndex+1, IndexConstants::InvalidID);
584
585 const FIndex2i BoundaryHalfEdges[2] = { { HalfEdge + 5, HalfEdge + 2 },
586 { HalfEdge + 0, h2 } };
587
588 for (int32 BoundaryIdx=0; BoundaryIdx<2; ++BoundaryIdx)
589 {
590 const int32 EdgeIdx = EdgeIndices[BoundaryIdx];
591 FEdgeSplitRecord& Edge = ConcurrencyState.EdgeSplitRecords[EdgeIdx];
592
593 const int32 EntryCountPreWrite = Edge.EntryCountPreWrite.fetch_add(1, std::memory_order_relaxed);
594 Edge.HalfEdges[EntryCountPreWrite] = BoundaryHalfEdges[BoundaryIdx];
595
596 // second atomic necessary to make sure the memory written to in other thread in slot 0 is available
597 // todo: might be faster to use std::memory_order_relaxed and use a atomic_thread_fence (not x86-64)
598 if (1 == Edge.EntryCountPostWrite.fetch_add(1, std::memory_order_acq_rel))
599 {
600 check( Edge.HalfEdges[0][0] != IndexConstants::InvalidID );
601 check( Edge.HalfEdges[0][1] != IndexConstants::InvalidID );
602 check( Edge.HalfEdges[1][0] != IndexConstants::InvalidID );
603 check( Edge.HalfEdges[1][1] != IndexConstants::InvalidID );
604
605 // second entry is being written, so we can now link the edges
606 Mesh.LinkEdge(Edge.HalfEdges[0][0], Edge.HalfEdges[1][1], EdgeIdx, EdgeIdx);
607 Mesh.LinkEdge(Edge.HalfEdges[0][1], Edge.HalfEdges[1][0], Edge.NewEdge, EdgeIdx);
608 }
609 }
610 }

612 // split 3 edges, triangle into 4 triangles (3 new triangles)
613 // return first index of newly added triangles
614 inline int32 Split3(int32 TriIndex, FConcurrencyState& ConcurrencyState)
615 {
616 const FIndex3i BaseTri = Mesh.GetTriangle(TriIndex);
617
618 const int32 BaseEdgeIndices[3] = { Mesh.GetEdgeIndex(TriIndex, 0),
619 Mesh.GetEdgeIndex(TriIndex, 1),
620 Mesh.GetEdgeIndex(TriIndex, 2) };
621
622 const int32 BaseAdj[3] = { Mesh.GetAdjTriangle(TriIndex, 0),
623 Mesh.GetAdjTriangle(TriIndex, 1),
624 Mesh.GetAdjTriangle(TriIndex, 2) };
625
626 const int32 SplitVertexIdx0 = ConcurrencyState.EdgeSplitRecords[BaseEdgeIndices[0]].SplitVertex;
627 const int32 SplitVertexIdx1 = ConcurrencyState.EdgeSplitRecords[BaseEdgeIndices[1]].SplitVertex;
628 const int32 SplitVertexIdx2 = ConcurrencyState.EdgeSplitRecords[BaseEdgeIndices[2]].SplitVertex;
629
630 // replace the base triangle with the one in the middle
631 Mesh.SetTriangle(TriIndex, FIndex3i(SplitVertexIdx0, SplitVertexIdx1, SplitVertexIdx2), TriIndex);
632
633 // allocate 9 new half-edges and indices. this is the first index to write to
634 const int32 HalfEdge = ConcurrencyState.HalfEdgeCount.fetch_add(9, std::memory_order_relaxed);
635 const int32 NewTriIndex = HalfEdge / 3;
636
637 // T0
638 Mesh.SetTriangle(NewTriIndex + 0, FIndex3i(BaseTri[0], SplitVertexIdx0, SplitVertexIdx2), TriIndex);
639
640 // T1
641 Mesh.SetTriangle(NewTriIndex + 1, FIndex3i(BaseTri[1], SplitVertexIdx1, SplitVertexIdx0), TriIndex);
642
643 // T2
644 Mesh.SetTriangle(NewTriIndex + 2, FIndex3i(BaseTri[2], SplitVertexIdx2, SplitVertexIdx1), TriIndex);
645
646 const int32 NewEdgeIndex = ConcurrencyState.EdgeCount.fetch_add(3, std::memory_order_relaxed);
647 check(NewEdgeIndex < Mesh.EdgeCount());
648
649 // interior edges
650 Mesh.LinkEdge( HalfEdge + 1, TriIndex * 3 + 2, NewEdgeIndex + 0, IndexConstants::InvalidID );
651 Mesh.LinkEdge( HalfEdge + 4, TriIndex * 3 + 0, NewEdgeIndex + 1, IndexConstants::InvalidID );
652 Mesh.LinkEdge( HalfEdge + 7, TriIndex * 3 + 1, NewEdgeIndex + 2, IndexConstants::InvalidID );
653
654 // pairs of halfedges corresponding to the split edge as seen from this triangle
655 const FIndex2i BoundaryHalfEdges[3] = { { HalfEdge + 0, HalfEdge + 5 },
656 { HalfEdge + 3, HalfEdge + 8 },
657 { HalfEdge + 6, HalfEdge + 2 } };
658
659 for (int32 LocalEdgeIdx=0; LocalEdgeIdx<3; ++LocalEdgeIdx)
660 {
661 const int32 EdgeIdx = BaseEdgeIndices[LocalEdgeIdx];
662 FEdgeSplitRecord& Edge = ConcurrencyState.EdgeSplitRecords[EdgeIdx];
663
664 if (BaseAdj[LocalEdgeIdx] == IndexConstants::InvalidID)
665 {
666 // boundary case: we can just update locally without needing data from the other side
667 Mesh.LinkEdge(BoundaryHalfEdges[LocalEdgeIdx][0], IndexConstants::InvalidID, EdgeIdx, EdgeIdx);
668 Mesh.LinkEdge(BoundaryHalfEdges[LocalEdgeIdx][1], IndexConstants::InvalidID, Edge.NewEdge, EdgeIdx);
669 continue;
670 }
671
672 const int32 EntryCountPreWrite = Edge.EntryCountPreWrite.fetch_add(1, std::memory_order_relaxed);
673 Edge.HalfEdges[EntryCountPreWrite] = BoundaryHalfEdges[LocalEdgeIdx];
674
675 // second atomic necessary to make sure the memory written to in other thread in slot 0 is available
676 // todo: might be faster to use std::memory_order_relaxed and use a atomic_thread_fence
677 if (1 == Edge.EntryCountPostWrite.fetch_add(1, std::memory_order_acq_rel))
678 {
679 check(Edge.HalfEdges[0][0] != IndexConstants::InvalidID);
680 check(Edge.HalfEdges[0][1] != IndexConstants::InvalidID);
681 check(Edge.HalfEdges[1][0] != IndexConstants::InvalidID);
682 check(Edge.HalfEdges[1][1] != IndexConstants::InvalidID);
683
684 // second entry is being written, so we can now link the edges
685 Mesh.LinkEdge(Edge.HalfEdges[0][0], Edge.HalfEdges[1][1], EdgeIdx, EdgeIdx);
686 Mesh.LinkEdge(Edge.HalfEdges[0][1], Edge.HalfEdges[1][0], Edge.NewEdge, EdgeIdx);
687 }
688 }
689 check(HalfEdge % 3 == 0);
690 return HalfEdge / 3;
691 }

696 return RefinementPolicy.ShouldRefine(TriIndex, SplitVertexBary, Iter);
