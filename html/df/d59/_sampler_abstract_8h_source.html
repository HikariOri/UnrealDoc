93 virtual void SamplingInitializing()
94 {
95 IsOptimalSegments.Empty(100);
96 Sampling.Empty(100);
97
98 CandidatePoints.Empty(100);
99
100 // Initialization of the algorithm with the not derivable coordinates of the curve and at least 5 temporary points
101 int32 ComplementaryPointOffset = 0;
102 {
103 TArray<double> NotDerivableCoordinates;
104 GetNotDerivableCoordinates(NotDerivableCoordinates);
105
106 NotDerivableCoordinates.Insert(Boundary.Min, 0);
107 NotDerivableCoordinates.Add(Boundary.Max);
108
109 ComplementaryPointOffset = NotDerivableCoordinates.Num() < 10 ? 10 : 1;
110
111 NextCoordinates.Empty(NotDerivableCoordinates.Num() * (ComplementaryPointOffset + 1));
112 NextCoordinates.Add(Boundary.Min);
113
114 for (int32 Index = 0; Index < NotDerivableCoordinates.Num() - 1; ++Index)
115 {
116 if (AddIntermediateCoordinates(NotDerivableCoordinates[Index], NotDerivableCoordinates[Index + 1], ComplementaryPointOffset))
117 {
118 NextCoordinates.Add(NotDerivableCoordinates[Index + 1]);
119 }
120 }
121 }
122
123 // #cadkernel_check: Could we do a MoveTem?
124 CandidatePoints.SwapCoordinates(NextCoordinates);
125 EvaluatesNewCandidatePoints();
126
127 // Not Derivable points and boundaries initialize the sampling
128 ComplementaryPointOffset++;
129 for (int32 Index = 0, ISampling = 0; Index < CandidatePoints.Size(); Index += ComplementaryPointOffset, ++ISampling)
130 {
131 Sampling.EmplaceAt(ISampling, CandidatePoints, Index);
132 IsOptimalSegments.Add(false);
133 }
134 // Remove last entered entry to IsOptimalSegments as this is to track segments
135 IsOptimalSegments.Pop();
136
137 CandidatePoints.RemoveComplementaryPoints(ComplementaryPointOffset);
138 }

143 void RunSampling()
144 {
145 if (IsOptimalSegments.Num() == 0)
146 {
147 // #cadkernel_check: What does this mean that IsOptimalSegments is empty.
148 // Why only one CandidatePoints?
149 return;
150 }
151
152 // < 100000 : check to avoid that the process loops endlessly
153 int32 CandidatePointsCount = CandidatePoints.Size();
154 while (CandidatePointsCount && Sampling.Coordinates.Num() < 100000)
155 {
156 NextCoordinates.Empty();
157
158 StartSamplingSegmentIndex = 0;
159 EndStudySegment.Index = 0;
160
161 // indices of TmpPoints and TmpCoordinates
162 int32 FirstCandidateIndex = 0;
163 int32 LastCandidateIndex = 1;
164
165 int32 SegmentIndex = 0;
166
167 do
168 {
169 if (IsOptimalSegments[StartSamplingSegmentIndex])
170 {
171 StartSamplingSegmentIndex++;
172 continue;
173 }
174
175 EndSamplingSegmentIndex = StartSamplingSegmentIndex + 1;
176
177 do
178 {
179 if ((LastCandidateIndex < CandidatePointsCount && TmpPolylineCoordinates[LastCandidateIndex] < Sampling.Coordinates[EndSamplingSegmentIndex]))
180 {
181 EndStudySegment.Set(CandidatePoints, LastCandidateIndex);
182 }
183 else
184 {
185 EndStudySegment.Set(Sampling, EndSamplingSegmentIndex);
186 }
187
188 // if the next candidate point is biggest than the next sampling point, no chord check can be done.
189 // an intermediate point is added.
190 // the next segment is processed
191 if ((FirstCandidateIndex >= CandidatePointsCount) || (FirstCandidateIndex < CandidatePointsCount && TmpPolylineCoordinates[FirstCandidateIndex] > EndStudySegment.Polyline->Coordinates[EndStudySegment.Index]))
192 {
193 AddIntermediateCoordinates(Sampling.Coordinates[StartSamplingSegmentIndex], Sampling.Coordinates[EndSamplingSegmentIndex], 1);
194 StartSamplingSegmentIndex = EndSamplingSegmentIndex;
195 break;
196 }
197
198 int32 NeededPointNum = CheckSamplingError(FirstCandidateIndex, LastCandidateIndex);
199 if (NeededPointNum == 0 || (NeededPointNum == 1 && ((LastCandidateIndex + 1) == CandidatePointsCount || (((LastCandidateIndex + 1) < CandidatePointsCount) && TmpPolylineCoordinates[LastCandidateIndex + 1] > Sampling.Coordinates[EndSamplingSegmentIndex]))))
200 {
201 if (EndStudySegment.IsSamplingPoint())
202 {
203 IsOptimalSegments[StartSamplingSegmentIndex] = true;
204 }
205 else
206 {
207 LastCandidateIndex++;
208 continue;
209 }
210 }
211 else
212 {
213 CompletesPolyline(NeededPointNum, LastCandidateIndex - 1);
214 }
215
216 // An existing point is reached
217 if (EndStudySegment.IsSamplingPoint())
218 {
219 StartSamplingSegmentIndex = EndSamplingSegmentIndex;
220 FirstCandidateIndex = LastCandidateIndex;
221 LastCandidateIndex++;
222 }
223 else
224 {
225 StartSamplingSegmentIndex = EndSamplingSegmentIndex - 1;
226 FirstCandidateIndex = LastCandidateIndex + 1;
227 LastCandidateIndex += 2;
228 }
229
230 } while (EndStudySegment.IsCandidatePoint());
231
232 } while (StartSamplingSegmentIndex < IsOptimalSegments.Num() && EndStudySegment.Index < Sampling.Coordinates.Num() - 1);
233
234 CandidatePoints.SwapCoordinates(NextCoordinates);
235 EvaluatesNewCandidatePoints();
236
237 CandidatePointsCount = CandidatePoints.Size();
238
239#ifdef DEBUG_CURVE_SAMPLING
240 DisplaySampling(CurveIndex == CurveToDisplay, 0);
241#endif
242 }
243
244#ifdef CHECK_RESULT_MAKE_ISO_POLYLINE
245 CheckResult();
246#endif
247
248 }

251 bool AddIntermediateCoordinates(double UMin, double UMax, int32 PointNum)
252 {
253 if (FMath::IsNearlyEqual(UMin, UMax, UE_DOUBLE_SMALL_NUMBER))
254 {
255 return false;
256 }
257 double DeltaCoord = (UMax - UMin) / (PointNum + 1);
258 double UCoord = UMin;
259 for (int32 Index = 0; Index < PointNum; Index++)
260 {
261 UCoord += DeltaCoord;
262 NextCoordinates.Add(UCoord);
263 }
264
265 return true;

276 void CompletesPolyline(int32 NeededPointNumber, int32 IntermediateIndex)
277 {
278 if (NeededPointNumber > 50)
279 {
280 NeededPointNumber = 50;
281 }
282
283 const TArray<double>& CandidateCoordinates = CandidatePoints.GetCoordinates();
284
285 if (NeededPointNumber > 1)
286 {
287 Sampling.EmplaceAt(EndSamplingSegmentIndex, CandidatePoints, IntermediateIndex);
288 IsOptimalSegments.EmplaceAt(EndSamplingSegmentIndex - 1, NeededPointNumber == 2 ? true : false);
289 EndSamplingSegmentIndex++;
290 }
291
292 if (EndStudySegment.IsCandidatePoint())
293 {
294 Sampling.EmplaceAt(EndSamplingSegmentIndex, CandidatePoints, EndStudySegment.Index);
295 IsOptimalSegments.EmplaceAt(EndSamplingSegmentIndex - 1, NeededPointNumber <= 2 ? true : false);
296 EndSamplingSegmentIndex++;
297 }
298 else
299 {
300 EndStudySegment.Index = EndSamplingSegmentIndex;
301 if (NeededPointNumber <= 2)
302 {
303 IsOptimalSegments[EndSamplingSegmentIndex - 1] = true;
304 }
305 }
306
307 if (NeededPointNumber > 2)
308 {
309 NeededPointNumber -= 2;
310 AddIntermediateCoordinates(Sampling.Coordinates[StartSamplingSegmentIndex], CandidateCoordinates[IntermediateIndex], NeededPointNumber);
311 AddIntermediateCoordinates(CandidateCoordinates[IntermediateIndex], EndStudySegment.Polyline->Coordinates[EndStudySegment.Index], NeededPointNumber);
312 }

326 int32 CountOfNeededPointsToRespectChordError(const PointType& PointA, const PointType& PointB, double ChordError)
327 {
328 double ABLength = PointType::Distance(PointA, PointB) / 2;
329 // Max segment length to respect the chord error
330 double LengthToHave = sqrt(FMath::Square(ABLength) * DesiredChordError / ChordError);
331 return (int32)(2 + (ABLength / LengthToHave + 0.5));
332 };

334 int32 CheckTangentError(const PointType& APoint, double ACoordinate, const PointType& BPoint, double BCoordinate, int32 FirstIndex, int32 EndIndex, int32 InStartSamplingSegmentIndex)
335 {
336 PointType Middle = CandidatePoints.GetPoints()[FirstIndex];
337
338 int32 CountOfNeededPoints = 2;
339
340 if (FirstIndex > 0 && InStartSamplingSegmentIndex > 0)
341 {
342 // get the first previous point in Sampling or CandidatePoints
343 double PreviousSampingCoordinate;
344 int32 PreviousSampingIndex = GetFirstNeighbor(InStartSamplingSegmentIndex - 1, ACoordinate, Sampling, -1, PreviousSampingCoordinate);
345
346 double PreviousCandidateCoordinate;
347 int32 PreviousCandidateIndex = GetFirstNeighbor(FirstIndex - 1, ACoordinate, CandidatePoints, -1, PreviousCandidateCoordinate);
348
349 const PointType& PreviousPoint = PreviousSampingCoordinate > PreviousCandidateCoordinate ? Sampling.GetPoints()[PreviousSampingIndex] : CandidatePoints.GetPoints()[PreviousCandidateIndex];
350
351 double ChordA = DistanceOfPointToSegment<PointType>(APoint, PreviousPoint, Middle);
352 if (ChordA > 4 * DesiredChordError)
353 {
354 CountOfNeededPoints = CountOfNeededPointsToRespectChordError(APoint, PreviousPoint, ChordA);
355 }
356 }
357
358 Middle = CandidatePoints.GetPoints()[EndIndex - 1];
359 if (EndIndex + 1 < CandidatePoints.Coordinates.Num())
360 {
361 double NextSampingCoordinate;
362 int32 NextSampingIndex = GetFirstNeighbor(StartSamplingSegmentIndex + 1, BCoordinate, Sampling, +1, NextSampingCoordinate);
363
364 double NextCandidateCoordinate;
365 int32 NextCandidateIndex = GetFirstNeighbor(EndIndex, BCoordinate, CandidatePoints, +1, NextCandidateCoordinate);
366
367 const PointType& NextPoint = NextSampingCoordinate < NextCandidateCoordinate ? Sampling.GetPoints()[NextSampingIndex] : CandidatePoints.GetPoints()[NextCandidateIndex];
368
369 double ChordB = DistanceOfPointToSegment<PointType>(BPoint, NextPoint, Middle);
370 if (ChordB > 4 * DesiredChordError)
371 {
372 CountOfNeededPoints = FMath::Max(CountOfNeededPointsToRespectChordError(BPoint, NextPoint, ChordB), CountOfNeededPoints);
373 }
374 }
375
376 return CountOfNeededPoints;
377 };

459 void CheckResult()
460 {
461 // Check the result
462 NextCoordinates.Empty(Sampling.Coordinates.Num());
463
464 for (int32 Index = 0; Index < Sampling.Coordinates.Num() - 1; ++Index)
465 {
466 AddIntermediateCoordinates(Sampling.Coordinates[Index], Sampling.Coordinates[Index + 1], 1);
467 }
468 CandidatePoints.SwapCoordinates(NextCoordinates);
469 EvaluatesNewCandidatePoints();
470
471 double MinError[2] = { HUGE_VALUE, HUGE_VALUE };
472 double MaxError[2] = { 0., 0. };
473 double ErrorSum[2] = { 0., 0. };
474 double SquareErrorSum[2] = { 0., 0. };
475
476 int32 EvaluationCount = CandidatePoints.Coordinates.Num();
477
478 for (int32 Index = 0; Index < EvaluationCount; ++Index)
479 {
480 double CurvilinearAbscise;
481 PointType ProjectedPoint = ProjectPointOnSegment<PointType>(CandidatePoints.GetPoints()[Index], Sampling.GetPoints()[Index], Sampling.GetPoints()[Index + 1], CurvilinearAbscise);
482 double ChordError = ProjectedPoint.Distance(CandidatePoints.GetPoints()[Index]);
483
484 MinError[0] = FMath::Min(MinError[0], ChordError);
485 MaxError[0] = FMath::Max(MaxError[0], ChordError);
486 ErrorSum[0] += ChordError;
487 SquareErrorSum[0] += FMath::Square(ChordError);
488
489 double ABCoordinate = Sampling.Coordinates[Index + 1] - Sampling.Coordinates[Index];
490 double AMCoordinate = CandidatePoints.Coordinates[Index] - Sampling.Coordinates[Index];
491 PointType MApproximate = Sampling.GetPoints()[Index] + (AMCoordinate / ABCoordinate) * (Sampling.GetPoints()[Index + 1] - Sampling.GetPoints()[Index]);
492 double ParametrizationError = MApproximate.Distance(ProjectedPoint);
493
494 MinError[1] = FMath::Min(MinError[1], ParametrizationError);
495 MaxError[1] = FMath::Max(MaxError[1], ParametrizationError);
496 ErrorSum[1] += ParametrizationError;
497 SquareErrorSum[1] += FMath::Square(ParametrizationError);
498 }
499 double ChordMed = ErrorSum[0] / EvaluationCount;
500 double ChordStandartDeviation = sqrt(SquareErrorSum[0] / EvaluationCount - FMath::Square(ChordMed));
501
502 double ParamMed = ErrorSum[1] / EvaluationCount;
503 double ParamStandartDeviation = sqrt(SquareErrorSum[1] / EvaluationCount - FMath::Square(ParamMed));
504
505 UE_LOGF(LogCADKernelBase, Verbose, "Chord Desired Med Sd Min Max %f %f %f %f %f", MaxError, ChordMed, ChordStandartDeviation, MinError[0], MaxError[0]);
506 UE_LOGF(LogCADKernelBase, Verbose, "Param Desired Med Sd Min Max %f %f %f %f %f", MaxError, ParamMed, ParamStandartDeviation, MinError[1], MaxError[1]);
507 }
508#endif
