428 // Update the mask that indicates what shader registers are used in the descriptor table.
429 template<typename DescriptorRangeType>
430 inline void UpdateCBVRegisterMaskWithDescriptorRange(EShaderFrequency SF, const DescriptorRangeType& Range)
431 {
432 const uint32 StartRegister = Range.BaseShaderRegister;
433 const uint32 EndRegister = StartRegister + Range.NumDescriptors;
434 const uint32 StartStage = (SF == SF_NumFrequencies) ? SF_Vertex : SF;
435 const uint32 EndStage = (SF == SF_NumFrequencies) ? SF_Compute : SF;
436 for (uint32 CurrentStage = StartStage; CurrentStage <= EndStage; CurrentStage++)
437 {
438 for (uint32 Register = StartRegister; Register < EndRegister; Register++)
439 {
440 // The bit shouldn't already be set for the current register.
441 check((Stage[CurrentStage].CBVRegisterMask & (1 << Register)) == 0);
442 Stage[CurrentStage].CBVRegisterMask |= (1 << Register);
443 }
444 }
445 }

447 // Update the mask that indicates what shader registers are used in the root descriptor.
448 template<typename DescriptorType>
449 inline void UpdateCBVRegisterMaskWithDescriptor(EShaderFrequency SF, const DescriptorType& Descriptor)
450 {
451 const uint32 StartStage = (SF == SF_NumFrequencies) ? SF_Vertex : SF;
452 const uint32 EndStage = (SF == SF_NumFrequencies) ? SF_Compute : SF;
453 const uint32& Register = Descriptor.ShaderRegister;
454 for (uint32 CurrentStage = StartStage; CurrentStage <= EndStage; CurrentStage++)
455 {
456 // The bit shouldn't already be set for the current register.
457 check((Stage[CurrentStage].CBVRegisterMask & (1 << Register)) == 0);
458 Stage[CurrentStage].CBVRegisterMask |= (1 << Register);
459 }
