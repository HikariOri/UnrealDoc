121 struct FElementCounter
122 {
123 uint16& Get(const int32 TypeIndex)
124 {
125 if (LIKELY(DenseIndexToElementIndex.IsValidIndex(LastAccessedDenseIndex)
126 && DenseIndexToElementIndex[LastAccessedDenseIndex] == TypeIndex))
127 {
128 return DenseCounters[LastAccessedDenseIndex];
129 }
130 if (UNLIKELY(DenseIndexToElementIndex.IsEmpty()))
131 {
132 // add a new one at the end
133 LastAccessedDenseIndex = DenseIndexToElementIndex.Add(static_cast<uint16>(TypeIndex));
134 return DenseCounters.Add_GetRef(0);
135 }
136
137 const int32 DenseIndex = Algo::LowerBound(DenseIndexToElementIndex, TypeIndex);
138 LastAccessedDenseIndex = DenseIndex;
139
140 if (DenseIndex == DenseIndexToElementIndex.Num()
141 || DenseIndexToElementIndex[DenseIndex] != TypeIndex)
142 {
143 // in this case DenseIndex indicates the new element insertion index
144 DenseIndexToElementIndex.Insert(static_cast<uint16>(TypeIndex), DenseIndex);
145 return DenseCounters.Insert_GetRef(0, DenseIndex);
146 }
147
148 return DenseCounters[DenseIndex];
149 }
150
151 SIZE_T GetAllocatedSize() const
152 {
153 return DenseCounters.GetAllocatedSize() + DenseIndexToElementIndex.GetAllocatedSize();
154 }
155
156 private:
157 int32 LastAccessedDenseIndex = INDEX_NONE;
158 TArray<uint16, TInlineAllocator<16>> DenseCounters;
159 TArray<uint16, TInlineAllocator<16>> DenseIndexToElementIndex;
160 };
161 FElementCounter SparseElementTypesCount;

983template<typename TStruct>
984void FMassArchetypeData::SetFragmentData(FMassArchetypeEntityCollection::FConstEntityRangeArrayView EntityRangeContainer, const TStruct& FragmentSource)
985{
986 check(FragmentSource.IsValid());
987 const UScriptStruct* FragmentType = FragmentSource.GetScriptStruct();
988 check(FragmentType);
989
990 if (UE::Mass::IsSparse(FragmentType) == false)
991 {
992 const int32* FragmentIndex = FragmentIndexMap.Find(FragmentType);
993 if (LIKELY(FragmentIndex))
994 {
995 const int32 FragmentTypeSize = FragmentType->GetStructureSize();
996 const uint8* FragmentSourceMemory = FragmentSource.GetMemory();
997 check(FragmentSourceMemory);
998
999 for (FMassArchetypeChunkIterator ChunkIterator(EntityRangeContainer); ChunkIterator; ++ChunkIterator)
1000 {
1001 uint8* FragmentMemory = static_cast<uint8*>(FragmentConfigs[*FragmentIndex].GetFragmentData(Chunks[ChunkIterator->ChunkIndex].GetRawMemory(), ChunkIterator->SubchunkStart));
1002 // Note that we cannot handle the copying with a single FragmentType->CopyScriptStruct call
1003 // because we have a single FragmentSource instance we're using as a copy source (i.e. we don't have a source array).
1004 for (int i = ChunkIterator->Length; i; --i, FragmentMemory += FragmentTypeSize)
1005 {
1006 FragmentType->CopyScriptStruct(FragmentMemory, FragmentSourceMemory);
1007 }
1008 }
1009 }
1010 else
1011 {
1012 UE_LOGF(LogMass, Warning
1013 , "Attempting to set value of fragment of type %ls, while it's not part of the archetype's composition"
1014 , *FragmentType->GetName());
1015 }
1016 }
1017 else
1018 {
1019 bSparseElementsBitSetDirty = true;
1020 for (FMassArchetypeChunkIterator ChunkIterator(EntityRangeContainer); ChunkIterator; ++ChunkIterator)
1021 {
1022 FMassArchetypeChunk& Chunk = Chunks[ChunkIterator->ChunkIndex];
1023 const FMassArchetypeEntityCollection::FArchetypeEntityRange& EntityRange = *ChunkIterator;
1024
1025 UE::Mass::FChunkSparseElements& SparseElements = Chunk.GetSparseElements(NumEntitiesPerChunk);
1026 SparseElements.BatchAdd(FragmentType, EntityRange.SubchunkStart, EntityRange.Length);
1027
1028 const FMassEntityHandle* EntitiesArray = Chunk.GetEntityArray(EntityListOffsetWithinChunk);
1029 TArrayView<const FMassEntityHandle> EntityHandlesView(&EntitiesArray[EntityRange.SubchunkStart], EntityRange.Length);
1030 SparseElementsStorage->BatchAddElementInstancesToEntities(EntityHandlesView, FragmentSource);
1031 }
1032 }
