18 FIntVector2 SanitizeAndroidScreenSize(const FIntVector2&& MaxScreenDims, const FIntVector2&& RequestedScreenDims)
19 {
20 FIntVector2 SanitizedDims;
21 // ensure Width and Height is multiple of 8
22 SanitizedDims.X = (RequestedScreenDims.X / 8) * 8;
23 SanitizedDims.Y = (RequestedScreenDims.Y / 8) * 8;
24
25 // clamp to native resolution
26 SanitizedDims.X = FPlatformMath::Min(SanitizedDims.X, MaxScreenDims.X);
27 SanitizedDims.Y = FPlatformMath::Min(SanitizedDims.Y, MaxScreenDims.Y);
28 return SanitizedDims;
29 }

31 void ApplyContentScaleFactor(int32& InOutScreenWidth, int32& InOutScreenHeight)
32 {
33 // CSF is a multiplier to 1280x720
34 static IConsoleVariable* CVarScale = IConsoleManager::Get().FindConsoleVariable(TEXT("r.MobileContentScaleFactor"));
35 float RequestedContentScaleFactor = CVarScale->GetFloat();
36
37 static IConsoleVariable* CVarResX = IConsoleManager::Get().FindConsoleVariable(TEXT("r.Mobile.DesiredResX"));
38 static IConsoleVariable* CVarResY = IConsoleManager::Get().FindConsoleVariable(TEXT("r.Mobile.DesiredResY"));
39 int32 RequestedResX = CVarResX->GetInt();
40 int32 RequestedResY = CVarResY->GetInt();
41
42 FString CmdLineCSF;
43 if (FParse::Value(FCommandLine::Get(), TEXT("mcsf="), CmdLineCSF, false))
44 {
45 RequestedContentScaleFactor = FCString::Atof(*CmdLineCSF);
46 }
47
48 FString CmdLineMDRes;
49 if (FParse::Value(FCommandLine::Get(), TEXT("mobileresx="), CmdLineMDRes, false))
50 {
51 RequestedResX = FCString::Atoi(*CmdLineMDRes);
52 }
53 if (FParse::Value(FCommandLine::Get(), TEXT("mobileresy="), CmdLineMDRes, false))
54 {
55 RequestedResY = FCString::Atoi(*CmdLineMDRes);
56 }
57
58 // 0 means to use native size
59 if (RequestedContentScaleFactor == 0.0f && RequestedResX <= 0 && RequestedResY <= 0)
60 {
61 UE_LOGF(LogAndroidWindowUtils, Log, "Setting Width=%d and Height=%d (requested scale = 0 = auto)", InOutScreenWidth, InOutScreenHeight);
62 }
63 else
64 {
65 int32 Width = InOutScreenWidth;
66 int32 Height = InOutScreenHeight;
67
68 if (RequestedResX > 0)
69 {
70 // set long side for orientation to requested X (height > width == GAndroidIsPortrait)
71 if (InOutScreenHeight > InOutScreenWidth)
72 {
73 Height = RequestedResX;
74 Width = FMath::TruncToInt32((float)Height * ((float)InOutScreenWidth / (float)InOutScreenHeight) + 0.5f);
75 }
76 else
77 {
78 Width = RequestedResX;
79 Height = FMath::TruncToInt32((float)Width * ((float)InOutScreenHeight / (float)InOutScreenWidth) + 0.5f);
80 }
81 }
82 else if (RequestedResY > 0)
83 {
84 // set short side for orientation to requested Y (height > width == GAndroidIsPortrait)
85 if (InOutScreenHeight > InOutScreenWidth)
86 {
87 Width = RequestedResY;
88 Height = FMath::TruncToInt32((float)Width * ((float)InOutScreenHeight / (float)InOutScreenWidth) + 0.5f);
89 }
90 else
91 {
92 Height = RequestedResY;
93 Width = FMath::TruncToInt32((float)Height * ((float)InOutScreenWidth / (float)InOutScreenHeight) + 0.5f);
94 }
95 }
96 else
97 {
98 const float AspectRatio = (float)InOutScreenWidth / (float)InOutScreenHeight;
99
100 // set resolution relative to 1280x720 ((height > width == GAndroidIsPortrait)
101 if (InOutScreenHeight > InOutScreenWidth)
102 {
103 Height = FMath::TruncToInt32(1280.f * RequestedContentScaleFactor);
104 }
105 else
106 {
107 Height = FMath::TruncToInt32(720.f * RequestedContentScaleFactor);
108 }
109
110 // apply the aspect ration to get the width
111 Width = FMath::TruncToInt32((float)Height * AspectRatio + 0.5f);
112 }
113
114 const FIntVector2 Sanitized = SanitizeAndroidScreenSize(FIntVector2(InOutScreenWidth, InOutScreenHeight), FIntVector2(Width, Height));
115 InOutScreenWidth = Sanitized.X;
116 InOutScreenHeight = Sanitized.Y;
117
118 UE_LOGF(LogAndroidWindowUtils, Log, "Setting Width=%d and Height=%d (requested scale = %f)", InOutScreenWidth, InOutScreenHeight, RequestedContentScaleFactor);
119 }
120 }
