331template <>
332class TCsvPersistentCustomStat<float> : public FCsvPersistentCustomStatBase
333{
334 friend struct FCsvPersistentCustomStats;
335public:
336 TCsvPersistentCustomStat<float>(FName InName, uint32 InCategoryIndex, bool bInResetEachFrame)
337 : FCsvPersistentCustomStatBase(InName, InCategoryIndex, bInResetEachFrame, ECsvPersistentCustomStatType::Float)
338 , Value(0.0f)
339 {}
340
341 float Add(float Rhs)
342 {
343 float Previous = Value.load(std::memory_order_consume);
344 float Desired = Previous + Rhs;
345 while (!Value.compare_exchange_weak(Previous, Desired, std::memory_order_release, std::memory_order_consume))
346 {
347 Desired = Previous + Rhs;
348 }
349 return Desired;
350 }
351 float Sub(float Rhs)
352 {
353 return Add(-Rhs);
354 }
355 void Set(float NewVal)
356 {
357 return Value.store(NewVal, std::memory_order_relaxed);
358 }
359 float GetValue()
360 {
361 return Value.load();
362 }
363 static ECsvPersistentCustomStatType GetClassStatType() { return ECsvPersistentCustomStatType::Float; }
364protected:
365 std::atomic<float> Value;
366};
