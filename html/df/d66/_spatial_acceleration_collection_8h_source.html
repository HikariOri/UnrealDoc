93template <typename TObj>
94struct TSpatialCollectionBucket
95{
96 TSpatialCollectionBucket() = default;
97 TSpatialCollectionBucket(const TSpatialCollectionBucket<TObj>& Other) = delete;
98 TSpatialCollectionBucket(TSpatialCollectionBucket<TObj>&& Other) = default;
99 TSpatialCollectionBucket<TObj>& operator=(const TSpatialCollectionBucket<TObj>& Other) = delete;
100 TSpatialCollectionBucket<TObj>& operator=(TSpatialCollectionBucket<TObj>&& Other) = default;
101
102 TArray<TObj> Objects;
103 TArray<uint16> FreeIndices;
104
105 int32 Add(TObj&& Obj)
106 {
107 uint16 Idx;
108 if (FreeIndices.Num())
109 {
110 Idx = FreeIndices.Pop();
111 Objects[Idx] = MoveTemp(Obj);
112 }
113 else
114 {
115 Idx = static_cast<uint16>(Objects.Add(MoveTemp(Obj)));
116 }
117
118 return Idx;
119 }
120
121 void UpdateOrAddAt(uint16 Idx, TObj&& Obj)
122 {
123 uint16 ObjectNum = static_cast<uint16>(Objects.Num());
124 if (ObjectNum <= Idx)
125 {
126 Objects.SetNum(Idx + 1);
127 for (uint16 FreeIndex = ObjectNum; FreeIndex < Idx - 1; FreeIndex++)
128 {
129 FreeIndices.Add(FreeIndex);
130 }
131 }
132 else
133 {
134 FreeIndices.Remove(Idx);
135 }
136
137 Objects[Idx] = MoveTemp(Obj);
138 }
139
140 void Remove(uint16 Idx)
141 {
142 if (Objects.Num() == Idx + 1)
143 {
144 Objects.Pop(EAllowShrinking::No);
145 }
146 else
147 {
148 FreeObjHelper(Objects[Idx]);
149 FreeIndices.Add(Idx);
150 }
151 }
152
153 void CopyFrom(const TSpatialCollectionBucket& Src)
154 {
155 Objects.SetNum(Src.Objects.Num());
156 for (int32 ObjIdx = 0; ObjIdx < Objects.Num(); ++ObjIdx)
157 {
158 Objects[ObjIdx] = CopyFromHelper(Src.Objects[ObjIdx]);
159 }
160 FreeIndices = Src.FreeIndices;
161 }
162
163};

407 virtual FSpatialAccelerationIdx AddSubstructure(TUniquePtr<ISpatialAcceleration<TPayloadType, T, d>>&& Substructure, uint16 BucketIdx, uint16 BucketInnerIdx) override
408 {
409 check(BucketIdx < MaxBuckets);
410 FSpatialAccelerationIdx Result;
411 Result.Bucket = BucketIdx;
412
413 ISpatialAcceleration<TPayloadType, T, d>* AccelPtr = Substructure.Get();
414 BucketEntryType BucketEntry;
415 BucketEntry.Acceleration = MoveTemp(Substructure);
416
417 const int32 TypeIdx = GetTypeIdx(AccelPtr);
418 switch (TypeIdx)
419 {
420 case 0: BucketEntry.TypeInnerIdx = static_cast<uint16>(GetAccelerationsPerType<0>(Types).Add(&AccelPtr->template AsChecked<typename std::tuple_element<0, std::tuple<TSpatialAccelerationTypes...>>::type>())); break;
421 case 1: BucketEntry.TypeInnerIdx = static_cast<uint16>(GetAccelerationsPerType<ClampedIdx(1)>(Types).Add(&AccelPtr->template AsChecked<typename std::tuple_element<ClampedIdx(1), std::tuple<TSpatialAccelerationTypes...>>::type>())); break;
422 case 2: BucketEntry.TypeInnerIdx = static_cast<uint16>(GetAccelerationsPerType<ClampedIdx(2)>(Types).Add(&AccelPtr->template AsChecked<typename std::tuple_element<ClampedIdx(2), std::tuple<TSpatialAccelerationTypes...>>::type>())); break;
423 case 3: BucketEntry.TypeInnerIdx = static_cast<uint16>(GetAccelerationsPerType<ClampedIdx(3)>(Types).Add(&AccelPtr->template AsChecked<typename std::tuple_element<ClampedIdx(3), std::tuple<TSpatialAccelerationTypes...>>::type>())); break;
424 case 4: BucketEntry.TypeInnerIdx = static_cast<uint16>(GetAccelerationsPerType<ClampedIdx(4)>(Types).Add(&AccelPtr->template AsChecked<typename std::tuple_element<ClampedIdx(4), std::tuple<TSpatialAccelerationTypes...>>::type>())); break;
425 case 5: BucketEntry.TypeInnerIdx = static_cast<uint16>(GetAccelerationsPerType<ClampedIdx(5)>(Types).Add(&AccelPtr->template AsChecked<typename std::tuple_element<ClampedIdx(5), std::tuple<TSpatialAccelerationTypes...>>::type>())); break;
426 case 6: BucketEntry.TypeInnerIdx = static_cast<uint16>(GetAccelerationsPerType<ClampedIdx(6)>(Types).Add(&AccelPtr->template AsChecked<typename std::tuple_element<ClampedIdx(6), std::tuple<TSpatialAccelerationTypes...>>::type>())); break;
427 case 7: BucketEntry.TypeInnerIdx = static_cast<uint16>(GetAccelerationsPerType<ClampedIdx(7)>(Types).Add(&AccelPtr->template AsChecked<typename std::tuple_element<ClampedIdx(7), std::tuple<TSpatialAccelerationTypes...>>::type>())); break;
428 }
429
430 this->ActiveBucketsMask |= (1 << BucketIdx);
431
432 Result.InnerIdx = BucketInnerIdx;
433 Buckets[BucketIdx].UpdateOrAddAt(BucketInnerIdx, MoveTemp(BucketEntry));
434
435 return Result;
436 }
