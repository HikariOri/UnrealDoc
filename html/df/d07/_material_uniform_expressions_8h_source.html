461inline UTextureCollection* GetIndexedTextureCollection(const FMaterial& Material, int32 TextureCollectionIndex)
462{
463 UTextureCollection* IndexedTextureCollection = nullptr;
464 const TArrayView<const TObjectPtr<UTextureCollection>> ReferencedTextureCollections = Material.GetReferencedTextureCollections();
465 if (ReferencedTextureCollections.IsValidIndex(TextureCollectionIndex))
466 {
467 IndexedTextureCollection = ReferencedTextureCollections[TextureCollectionIndex];
468 }
469
470 if (IndexedTextureCollection == nullptr)
471 {
472 static bool bWarnedOnce = false;
473 if (!bWarnedOnce)
474 {
475 UE_LOGF(LogMaterial, Warning, "%ls: Requesting an invalid TextureIndex! (%u / %u)", *Material.GetFriendlyName(), TextureCollectionIndex, ReferencedTextureCollections.Num());
476 bWarnedOnce = true;
477 }
478 }
479
480 return IndexedTextureCollection;
481}
