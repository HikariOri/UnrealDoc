465 {
466 check(!InName.IsNone());
467
468 // Walk looking for non-abstract default child. (e.g. Surround->Surround5X->Surround 5.1)
469 const FChannelTypeFamily* Channel = FindChannel(InName);
470 while (Channel && Channel->IsAbstract())

472 Channel = Channel->GetDefaultChild();
