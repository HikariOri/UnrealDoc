211template<typename ParentSectionClass>
212FText TSubSectionMixin<ParentSectionClass>::GetSectionToolTip() const
213{
214 const UMovieScene* MovieScene = SubSectionObject.GetTypedOuter<UMovieScene>();
215 const UMovieSceneSequence* InnerSequence = SubSectionObject.GetSequence();
216 const UMovieScene* InnerMovieScene = InnerSequence ? InnerSequence->GetMovieScene() : nullptr;
217
218 if (!MovieScene || !InnerMovieScene || !SubSectionObject.HasStartFrame() || !SubSectionObject.HasEndFrame())
219 {
220 return FText::GetEmpty();
221 }
222
223 FFrameRate InnerTickResolution = InnerMovieScene->GetTickResolution();
224
225 // Calculate the length of this section and convert it to the timescale of the sequence's internal sequence
226 FFrameTime SectionLength = ConvertFrameTime(SubSectionObject.GetExclusiveEndFrame() - SubSectionObject.GetInclusiveStartFrame(), MovieScene->GetTickResolution(), InnerTickResolution);
227
228 // Calculate the inner start time of the sequence in both full tick resolution and frame number
229 FFrameTime StartOffset = SubSectionObject.GetOffsetTime().Get(0);
230 FFrameTime InnerStartTime = InnerMovieScene->GetPlaybackRange().GetLowerBoundValue() + StartOffset;
231 int32 InnerStartFrame = ConvertFrameTime(InnerStartTime, InnerTickResolution, InnerMovieScene->GetDisplayRate()).RoundToFrame().Value;
232
233 // Calculate the length, which is limited by both the outer section length and internal sequence length, in terms of internal frames
234 int32 InnerFrameLength = ConvertFrameTime(FMath::Min(SectionLength, InnerMovieScene->GetPlaybackRange().GetUpperBoundValue() - InnerStartTime), InnerTickResolution, InnerMovieScene->GetDisplayRate()).RoundToFrame().Value;
235
236 // Calculate the inner frame number of the end frame
237 int32 InnerEndFrame = InnerStartFrame + InnerFrameLength;
238
239 FText MetaDataText = FSubTrackEditorUtil::GetMetaDataText(InnerSequence);
240
241 if (MetaDataText.IsEmpty())
242 {
243 return FText::Format(LOCTEXT("ToolTipContentFormat", "{0} - {1} ({2} frames @ {3})"),
244 InnerStartFrame,
245 InnerEndFrame,
246 InnerFrameLength,
247 InnerMovieScene->GetDisplayRate().ToPrettyText()
248 );
249 }
250 else
251 {
252 return FText::Format(LOCTEXT("ToolTipContentFormatWithMetaData", "{0} - {1} ({2} frames @ {3})\n\n{4}"),
253 InnerStartFrame,
254 InnerEndFrame,
255 InnerFrameLength,
256 InnerMovieScene->GetDisplayRate().ToPrettyText(),
257 MetaDataText
258 );
259 }
260}

262template<typename ParentSectionClass>
263TOptional<FFrameTime> TSubSectionMixin<ParentSectionClass>::GetSectionTime(FSequencerSectionPainter& InPainter) const
264{
265 if (!InPainter.bIsSelected)
266 {
267 return TOptional<FFrameTime>();
268 }
269
270 TSharedPtr<ISequencer> Sequencer = GetSequencer();
271 if (!Sequencer)
272 {
273 return TOptional<FFrameTime>();
274 }
275
276 FFrameTime CurrentTime = Sequencer->GetLocalTime().Time;
277 if (!SubSectionObject.GetRange().Contains(CurrentTime.FrameNumber))
278 {
279 return TOptional<FFrameTime>();
280 }
281
282 if (!SubSectionObject.GetSequence() || !SubSectionObject.GetSequence()->GetMovieScene())
283 {
284 return TOptional<FFrameTime>();
285 }
286
287 const UMovieScene* SubSequenceMovieScene = SubSectionObject.GetSequence()->GetMovieScene();
288 const FFrameTime HintFrameTime = CurrentTime * SubSectionObject.OuterToInnerTransform();
289
290 return HintFrameTime;
291}
