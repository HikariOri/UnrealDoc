5/*
6 * This header contains types used to store and decode references between compiled-in objects in
7 * different modules.
8 * e.g. a UClass in a project module which extends Actor needs a pointer to the AActor UClass
9 * instance in its UStruct::SuperStruct field. Some compilers and linkers will not allow this
10 * field to be constinit-initialized to the address of a constinit variable in another module.
11 *
12 * The values encoded are produced by UHT such that they correspond to UObject initialization
13 * data handled by FCompiledInObjectRegistry.
14 *
15 * Incomplete features:
16 * - Stability of references across live coding/hot reload recompiles (if UHT runs)
17 * - Stability of references for installed-engine scenarios (i.e. references for engine types
18 * must be fixed when the engine build is created so that plugins/projects can link against
19 * the right objects at runtime)
20 *
21 * Non-features:
22 * - There is no ability to distinguish the encoding of these compiled in references from
23 * other encoded pointers used by TObjectPtr such as late-resolved object handles or remote
24 * object handles.
25 * - Therefore there is no ability to defer resolution of compiled in references until an
26 * arbitrary later point or resolve them on demand. They can only be resolved by the
27 * CoreUObject module as part of initializing a module which contains UObjects.
28 *
29 */
30

43// Decoded reference to a compiled in object
44struct FCompiledInObjectReference
45{
46 uint32 ModuleId = 0;
47 uint32 ObjectId = 0;
48
49 static inline constexpr int32 NumObjectIdBits = 32;
50 static inline constexpr uint32 MaskObjectId = 0xFFFF'FFFF;
51 static inline constexpr uint32 MaxModuleId = 0x7FFF'FFFF;
52
53 // To distinguish compiled-in references from pointers the low bit is always 1
54 // UObjects should have a natural alignment greater than 1, allowing this.
55 static inline constexpr uint32 CompiledInRefMask = 0x1;
56 static inline constexpr int32 CompiledInRefShift = 1;
57
58 static_assert(sizeof(UPTRINT) >= 8, "FCompiledInObjectReference requires pointers to be 8 bytes or more");
59
60 /* Check whether a pointer is an encoded reference or a raw pointer */
61 [[nodiscard]] static constexpr bool IsEncodedRef(UPTRINT InEncoded)
62 {
63 return (InEncoded & CompiledInRefMask) == CompiledInRefMask;
64 }
65
66 /* Decode an encoded pointer into this structure type */
67 [[nodiscard]] static constexpr FCompiledInObjectReference FromEncoded(UPTRINT InEncoded)
68 {
69 InEncoded >>= CompiledInRefShift;
70 uint32 ModuleId = InEncoded >> NumObjectIdBits;
71 uint32 ObjectId = InEncoded & MaskObjectId;
72 return { .ModuleId = ModuleId, .ObjectId = ObjectId };
73 }
74
75 /* Return this structure as an encoded pointer */
76 [[nodiscard]] constexpr UPTRINT GetEncoded() const
77 {
78 const UPTRINT CombinedId = ((UPTRINT)ModuleId << NumObjectIdBits) | ObjectId;
79 return (CombinedId << CompiledInRefShift) | CompiledInRefMask;
80 }
81};

83// Encoded pointer to another compiled-in object for runtime linking
84struct FCompiledInObjectPtr
85{
86 [[nodiscard]] constexpr FCompiledInObjectPtr() = default;
87 [[nodiscard]] explicit constexpr FCompiledInObjectPtr(FCompiledInObjectReference InRef)
88 : Encoded(InRef.GetEncoded())
89 {
90 }
91 constexpr FCompiledInObjectPtr(const FCompiledInObjectPtr&) = default;
92 constexpr FCompiledInObjectPtr& operator=(const FCompiledInObjectPtr&) = default;
93
94 [[nodiscard]] constexpr bool IsUnset() const
95 {
96 return !FCompiledInObjectReference::IsEncodedRef(Encoded);
97 }
98
99 [[nodiscard]] constexpr bool IsEncodedRef() const
100 {
101 return FCompiledInObjectReference::IsEncodedRef(Encoded);
102 }
103
104 [[nodiscard]] constexpr FCompiledInObjectReference GetDecoded() const
105 {
106 return FCompiledInObjectReference::FromEncoded(Encoded);
107 }
108
109private:
110 uint64 Encoded = 0;
111};

118// Wrapper for a compiled-in object pointer which is not a TObjectPtr to be used in other UHT-generated
119// structures
120// T is not required to derive from UObject, e.g. it may be FStructBaseChain.
121// In that case if the pointer is an encoded reference, it will resolve to a UObject which can be cast to the required type.
122// e.g. lists of implemented interfaces, struct base chains
123template<typename T>
124struct TCompiledInObjectPtr
125{
126 union
127 {
128 FCompiledInObjectPtr CompiledInPtr;
129 T* Pointer = nullptr;
130 };
131
132 constexpr TCompiledInObjectPtr() = default;
133 constexpr TCompiledInObjectPtr(const TCompiledInObjectPtr&) = default;
134 constexpr TCompiledInObjectPtr& operator=(const TCompiledInObjectPtr&) = default;
135
136 // Implicit compile-time construction from object pointer
137 template<UE::CNotCVRefTo<TCompiledInObjectPtr> U> requires (UE::CConvertibleTo<U, T*> && !UE::CSameAs<T, U>)
138 [[nodiscard]] constexpr TCompiledInObjectPtr(U&& InPtr)
139 : Pointer(InPtr)
140 {
141 }
142
143 // Implicit compile-time construction from UHT encoded reference
144 [[nodiscard]] constexpr TCompiledInObjectPtr(FCompiledInObjectReference InRef)
145 : CompiledInPtr(InRef)
146 {
147 }

149 // Operators to allow this or T* to be used depending on IS_MONOLITHIC
150 operator T*() const

152 return GetPointer();
