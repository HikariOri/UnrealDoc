312// DDC-Verify collects data about a subset of newly generated derived data, and whether it matches existing cached values
313// This object is used to track statistics about verification results and is collected per DDC bucket
314struct FDerivedDataCacheVerificationStat
315{
316 FString BucketName; // name of the bucket
317 uint32 RecordsTested = 0; // num records that were tested against the cache
318 uint32 RecordsMatchedCache = 0; // num records that match existing cache data
319 uint32 RecordsNotInCache = 0; // records that have no existing data in the cache
320 uint32 ValuesTested = 0; // multiple values per record
321 uint64 ValuesTestedBytes = 0; // total size of values tested
322 uint32 ValuesMatchedCache = 0; // num values that match existing cache data
323 uint64 ValuesMatchedCacheBytes = 0; // total size of matching data
324 uint32 ValuesNotInCache = 0; // num values that have no existing data in the cache
325 uint64 ValuesNotInCacheBytes = 0; // num bytes of new values written to the cache
326 uint32 ValuesChanged = 0; // num values existing in the cache that don't match the new data
327 uint64 ValuesChangedBytes = 0; // num bytes of changed values detected (note this tracks the entire value, not the individual changes)
328 uint32 ValuesNotInRequest = 0; // num values that were in the cache before, but not in the new data
329 uint64 ValuesNotInRequestBytes = 0; // num bytes of data we are no longer writing in values
330
331 const FDerivedDataCacheVerificationStat& operator+=(const FDerivedDataCacheVerificationStat& OtherStat)
332 {
333 RecordsTested += OtherStat.RecordsTested;
334 RecordsMatchedCache += OtherStat.RecordsMatchedCache;
335 RecordsNotInCache += OtherStat.RecordsNotInCache;
336 ValuesTested += OtherStat.ValuesTested;
337 ValuesTestedBytes += OtherStat.ValuesTestedBytes;
338 ValuesMatchedCache += OtherStat.ValuesMatchedCache;
339 ValuesMatchedCacheBytes += OtherStat.ValuesMatchedCacheBytes;
340 ValuesNotInCache += OtherStat.ValuesNotInCache;
341 ValuesNotInCacheBytes += OtherStat.ValuesNotInCacheBytes;
342 ValuesChanged += OtherStat.ValuesChanged;
343 ValuesChangedBytes += OtherStat.ValuesChangedBytes;
344 ValuesNotInRequest += OtherStat.ValuesNotInRequest;
345 ValuesNotInRequestBytes += OtherStat.ValuesNotInRequestBytes;
346 return *this;
347 }
