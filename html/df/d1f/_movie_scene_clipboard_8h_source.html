89 {

95 // Check for same type

104 if (ConversionFunction)
105 {
106 FMovieSceneClipboardKey Key = ConversionFunction->operator()(*this);
107 Value = static_cast<const TKey<T>&>(Key.Data.GetValue()).Value;
108 return true;
109 }
