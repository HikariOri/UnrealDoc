131 TSharedPtr<FString> InitRowWidgetContent()
132 {
133 TSharedPtr<FString> InitialValue = MakeShareable(new FString(LOCTEXT("DataTable_None", "None").ToString()));
134
135 FName RowContains;
136 const FPropertyAccess::Result RowResult = RowContentsPropertyHandle->GetValue(RowContains);
137 RowContents.Empty();
138
140 UDataTable* DataTable = NULL;
141 DataTablePropertyHandle->GetValue((UObject*&)DataTable);
142
143 FName ColumnName = NAME_None;
144 ColumnNamePropertyHandle->GetValue(ColumnName);
145
146 if (DataTable != NULL && ColumnName != NAME_None)
147 {
148 TArray<TArray<FString> > RowData = DataTable->GetTableData();
149 if (RowData.Num() > 0)
150 {
151 TArray<FString>& ColumnTitles = RowData[0];
152
153 // find the column we are interested in
154 int Column;
155 for (Column = 0; Column < ColumnTitles.Num(); ++Column)
156 {
157 if (ColumnName.ToString() == ColumnTitles[Column])
158 {
159 break;
160 }
161 }
162
163 TSet<FString> Strings;
164
165 // go through the rest of the rows and add the contents of the column we want to our list
166 for (int Row = 1; Row < RowData.Num(); ++Row)
167 {
168 TSharedRef<FString> RowContentsItem = MakeShareable(new FString(RowData[Row][Column]));
169 if (!Strings.Contains(*RowContentsItem))
170 {
171 RowContents.Add(RowContentsItem);
172 Strings.Add(*RowContentsItem);
173 }
175 if (RowData[Row][Column] == RowContains.ToString())
176 {
177 InitialValue = RowContentsItem;
178 }
179 }
180 }
181 }
182
184 if (RowResult != FPropertyAccess::MultipleValues)
185 {
186 FName NewValue = FName(**InitialValue);
187 RowContentsPropertyHandle->SetValue(NewValue);
188 }
189
190 return InitialValue;
191 }
192
194 TSharedPtr<FString> InitColumnWidgetContent()
195 {
196 TSharedPtr<FString> InitialValue = MakeShareable(new FString(LOCTEXT("DataTable_None", "None").ToString()));
197
198 FName ColumnName;
199 const FPropertyAccess::Result ColumnResult = ColumnNamePropertyHandle->GetValue(ColumnName);
200 ColumnNames.Empty();
201
203 UDataTable* DataTable = NULL;
204 DataTablePropertyHandle->GetValue((UObject*&)DataTable);
205
206 if (DataTable != NULL)
207 {
208 TArray<TArray<FString> > RowData = DataTable->GetTableData();
209 if (RowData.Num() > 0)
210 {
211 TArray<FString>& ColumnTitles = RowData[0];
212
213 for (int Column = 0; Column < ColumnTitles.Num(); ++Column)
214 {
216 TSharedRef<FString> ColumnNameItem = MakeShareable(new FString(ColumnTitles[Column]));
217 ColumnNames.Add(ColumnNameItem);
218
220 if (ColumnTitles[Column] == ColumnName.ToString())
221 {
222 InitialValue = ColumnNameItem;
223 }
224 }
225 }
226 }
227
229 if (ColumnResult != FPropertyAccess::MultipleValues)
230 {
231 FName NewValue = FName(**InitialValue);
232 ColumnNamePropertyHandle->SetValue(NewValue);
233 }
234

329 bool ShouldFilterAsset(const FAssetData& AssetData)
330 {
331 if (!RowTypeFilter.IsNone())
332 {
333 static const FName RowStructureTagName("RowStructure");
334 FString RowStructure;
335 if (AssetData.GetTagValue<FString>(RowStructureTagName, RowStructure))
336 {
337 if (RowStructure == RowTypeFilter.ToString())
338 {
339 return false;
340 }
341
342 // This is slow, but at the moment we don't have an alternative to the short struct name search
343 UScriptStruct* RowStruct = UClass::TryFindTypeSlow<UScriptStruct>(RowStructure);
344 if (RowStruct && RowFilterStruct && RowStruct->IsChildOf(RowFilterStruct))
345 {
346 return false;
347 }
348 }
349 return true;
350 }
351 return false;
352 }
