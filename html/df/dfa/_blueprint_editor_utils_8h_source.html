353 template <typename SignatureType>
354 static void CreateFunctionGraph(UBlueprint* Blueprint, class UEdGraph* Graph, bool bIsUserCreated, SignatureType* SignatureFromObject)
355 {
356 // Give the schema a chance to fill out any required nodes (like the entry node or results node)
357 const UEdGraphSchema* Schema = Graph->GetSchema();
358 const UEdGraphSchema_K2* K2Schema = Cast<const UEdGraphSchema_K2>(Graph->GetSchema());
359
360 Schema->CreateDefaultNodesForGraph(*Graph);
361
362 if ( K2Schema != NULL )
363 {
364 K2Schema->CreateFunctionGraphTerminators(*Graph, SignatureFromObject);
365
366 if ( bIsUserCreated )
367 {
368 // We need to flag the entry node to make sure that the compiled function is callable from Kismet2
369 int32 ExtraFunctionFlags = ( FUNC_BlueprintCallable | FUNC_BlueprintEvent | FUNC_Public );
370 if ( BPTYPE_FunctionLibrary == Blueprint->BlueprintType )
371 {
372 ExtraFunctionFlags |= FUNC_Static;
373 }
374 if ( BPTYPE_Const == Blueprint->BlueprintType )
375 {
376 ExtraFunctionFlags |= FUNC_Const;
377 }
378 // We need to mark the function entry as editable so that we can
379 // set metadata on it if it is an editor utility blueprint/widget:
380 K2Schema->MarkFunctionEntryAsEditable(Graph, true);
381 if( IsEditorUtilityBlueprint( Blueprint ))
382 {
383 if( FKismetUserDeclaredFunctionMetadata* MetaData = GetGraphFunctionMetaData( Graph ))
384 {
385 MetaData->bCallInEditor = true;
386 }
387 }
388 K2Schema->AddExtraFunctionFlags(Graph, ExtraFunctionFlags);
389 }
390 }
391 }

504 template< class T >
505 static inline void GetAllNodesOfClass( const UBlueprint* Blueprint, TArray<T*>& OutNodes )
506 {
507 TArray<UEdGraph*> AllGraphs;
508 Blueprint->GetAllGraphs(AllGraphs);
509 for(int32 i=0; i<AllGraphs.Num(); i++)
510 {
511 check(AllGraphs[i] != NULL);
512 TArray<T*> GraphNodes;
513 AllGraphs[i]->GetNodesOfClass<T>(GraphNodes);
514 OutNodes.Append(GraphNodes);
515 }
516 }
