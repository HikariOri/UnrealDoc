40// Primitive types of a single scalar.
41// Note: These are listed in precision order. Converting one to the other is then simply performed taking the max EScalarKind.
42enum class EScalarKind : uint8
43{
44 Boolean, // Boolean type (true/false).
45 Integer, // 32-bit signed integer type.
46 Float, // 32-bit floating-point type.
47 Double, // Large World Coordinates type.
48};

65// Describes a primitive type (scalar, vector, or matrix).
66struct FPrimitive
67{
68 // The underlying scalar data type (e.g. Bool, Int, Float).
69 EScalarKind ScalarKind;
70
71 // Number of rows. For a vector, this is the number of elements. For a scalar, this is 1.
72 int8 NumRows;
73
74 // Number of columns. For a vector or scalar, this is 1. For a matrix, this is > 1.
75 int8 NumColumns;
76
77 // If true, this represents an LWC inverse matrix. Only supported for EScalarKind::LWC and 4x4 dimensions.
78 bool bIsLWCInverseMatrix;
79
80 // Returns whether this primitive type is arithmetic (it supports arithmetic operations like addition).
81 bool IsArithmetic() const { return ScalarKindIsArithmetic(ScalarKind); }
82
83 // Returns whether this type is a primitive scalar (1x1).
84 bool IsScalar() const { return NumComponents() == 1; }
85
86 // Returns whether this primitive type is a row vector (rows = 1, columns > 1).
87 bool IsRowVector() const { return NumRows == 1 && NumColumns > 1; }
88
89 // Returns whether this primitive type is a column vector (rows > 1, columns = 1).
90 bool IsColumnVector() const { return NumRows > 1 && NumColumns == 1; }
91
92 // Returns whether this primitive type is a matrix (rows > 1, columns > 1).
93 bool IsMatrix() const { return NumRows > 1 && NumColumns > 1; }
94
95 // Returns whether this is a boolean primitive type (of any dimension).
96 bool IsBoolean() const { return ScalarKind == EScalarKind::Boolean; }
97
98 // Returns whether this is a single boolean scalar.
99 bool IsBoolScalar() const { return IsBoolean() && IsScalar(); }
100
101 // Returns whether this is an integer primitive type (of any dimension).
102 bool IsInteger() const { return ScalarKind == EScalarKind::Integer; }
103
104 // Returns whether this is a float primitive type (of any dimension).
105 bool IsFloat() const { return ScalarKind == EScalarKind::Float; }
106
107 // Returns whether this is an LWC primitive type (of any dimension).
108 bool IsDouble() const { return ScalarKind == EScalarKind::Double; }
109
110 // Returns whether this is any floating point primitive type (e.g. float, LWC).
111 bool IsAnyFloat() const { return ScalarKindIsAnyFloat(ScalarKind); }
112
113 // Returns the total number of scalar components in this primitive type.
114 int32 NumComponents() const { return NumRows * NumColumns; }
115
116 // Returns a new FType with the same dimensions but a different scalar kind.
117 FType ToScalarKind(EScalarKind InScalarKind) const;
118
119 // Returns the scalar FType corresponding to this primitive's scalar kind.
120 FType ToScalar() const;
121
122 // Returns a vector FType with the given number of rows and this primitive's scalar kind.
123 FType ToVector(int NumColumns) const;
124
125 bool operator==(const FPrimitive& Other) const = default;
126 bool operator!=(const FPrimitive& Other) const = default;
127};
