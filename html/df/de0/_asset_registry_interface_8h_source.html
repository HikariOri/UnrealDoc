220 struct FDependencyQuery
221 {
226 UE::AssetRegistry::EDependencyProperty Required;
231 UE::AssetRegistry::EDependencyProperty Excluded;
232
240 TArray<UE::AssetRegistry::EDependencyProperty, TInlineAllocator<1>> RequiredUnions;
241
242 FDependencyQuery()
243 {
244 Required = UE::AssetRegistry::EDependencyProperty::None;
245 Excluded = UE::AssetRegistry::EDependencyProperty::None;
246 }
247
248 inline FDependencyQuery(EDependencyQuery QueryFlags)
249 {
250 if (!EnumHasAnyFlags(QueryFlags, EDependencyQuery::Propagation))
251 {
252 Required = (!!(QueryFlags & EDependencyQuery::Game) ? UE::AssetRegistry::EDependencyProperty::Game : UE::AssetRegistry::EDependencyProperty::None)
253 | (!!(QueryFlags & EDependencyQuery::Build) ? UE::AssetRegistry::EDependencyProperty::Build : UE::AssetRegistry::EDependencyProperty::None);
254 }
255 else
256 {
257 Required = UE::AssetRegistry::EDependencyProperty::None;
258 EnumRemoveFlags(QueryFlags, EDependencyQuery::Game | EDependencyQuery::NotGame | EDependencyQuery::Build | EDependencyQuery::NotBuild);
259 RequiredUnions.Add(UE::AssetRegistry::EDependencyProperty::Game | UE::AssetRegistry::EDependencyProperty::Build);
260 }
261 Required |= (!!(QueryFlags & EDependencyQuery::Hard) ? UE::AssetRegistry::EDependencyProperty::Hard : UE::AssetRegistry::EDependencyProperty::None)
262 | (!!(QueryFlags & EDependencyQuery::Direct) ? UE::AssetRegistry::EDependencyProperty::Direct : UE::AssetRegistry::EDependencyProperty::None)
263 | (!!(QueryFlags & EDependencyQuery::CookRule) ? UE::AssetRegistry::EDependencyProperty::CookRule : UE::AssetRegistry::EDependencyProperty::None);
264
265 Excluded = (!!(QueryFlags & EDependencyQuery::NotHard) ? UE::AssetRegistry::EDependencyProperty::Hard : UE::AssetRegistry::EDependencyProperty::None)
266 | (!!(QueryFlags & EDependencyQuery::NotGame) ? UE::AssetRegistry::EDependencyProperty::Game : UE::AssetRegistry::EDependencyProperty::None)
267 | (!!(QueryFlags & EDependencyQuery::NotBuild) ? UE::AssetRegistry::EDependencyProperty::Build : UE::AssetRegistry::EDependencyProperty::None)
268 | (!!(QueryFlags & EDependencyQuery::NotDirect) ? UE::AssetRegistry::EDependencyProperty::Direct : UE::AssetRegistry::EDependencyProperty::None)
269 | (!!(QueryFlags & EDependencyQuery::NotCookRule) ? UE::AssetRegistry::EDependencyProperty::CookRule : UE::AssetRegistry::EDependencyProperty::None);
270 }
