318 static constexpr uint32 ChunkSizeInBytes = 16 * 1024;
319 static constexpr uint32 RemainingBytes = ChunkSizeInBytes - Align<uint32>(sizeof(FHeader), alignof(FHeader));
320 static constexpr uint32 MaxEventsPerChunk = RemainingBytes / Align<uint32>(sizeof(FEvent), alignof(FEvent));
321
322 TStaticArray<TTypeCompatibleBytes<FEvent>, MaxEventsPerChunk> Elements;
323
324 static RHI_API TLockFreePointerListUnordered<void, PLATFORM_CACHE_LINE_SIZE> MemoryPool;
325
326 void* operator new(size_t Size)
327 {
328 check(Size == sizeof(FChunk));
329
330 void* Memory = MemoryPool.Pop();
331 if (!Memory)
332 {
333 Memory = FMemory::Malloc(sizeof(FChunk), alignof(FChunk));
334 }
335
336 // UE-295331 Investigation : Fill memory with garbage on allocation to catch use-after-free etc.
337 FMemory::Memset(Memory, 0xf7, sizeof(FChunk));
338
339 return Memory;
340 }
341
342 void operator delete(void* Pointer)
343 {
344 // UE-295331 Investigation : Fill memory with garbage on deallocation to catch use-after-free etc.
345 FMemory::Memset(Pointer, 0xe5, sizeof(FChunk));
346
347 MemoryPool.Push(Pointer);
348 }

389 template <typename TEventType, typename... TArgs>
390 TEventType& Emplace(TArgs&&... Args)
391 {
392 static_assert(std::is_trivially_destructible_v<TEventType>, "Destructors are not called on GPU profiler events, so the types must be trivially destructible.");
393
394 if (!Current)
395 {
396 Current = new FChunk;
397 if (!First)
398 {
399 First = Current;
400 }
401 }
402
403 if (Current->Header.Num >= FChunk::MaxEventsPerChunk)
404 {
405 FChunk* NewChunk = new FChunk;
406 Current->Header.Next = NewChunk;
407 Current = NewChunk;
408 }
409
410 FEvent* Event = Current->GetElement(Current->Header.Num++);
411 new (Event) FEvent(TEventType(Forward<TArgs>(Args)...));
412
413 TEventType& Data = Event->Value.Get<TEventType>();
414
415 #if WITH_RHI_BREADCRUMBS
416 if constexpr (
417 std::is_same_v<UE::RHI::GPUProfiler::FEvent::FBeginBreadcrumb, TEventType> ||
418 std::is_same_v<UE::RHI::GPUProfiler::FEvent::FEndBreadcrumb , TEventType> ||
419 std::is_same_v<UE::RHI::GPUProfiler::FEvent::FFrameBoundary , TEventType>
420 )
421 {
422 if (Data.Breadcrumb)
423 {
424 // Attach the breadcrumb allocator for begin/end breadcrumb events.
425 // This keeps the breadcrumbs alive until the events have been consumed by the profilers.
426 Current->Header.BreadcrumbAllocators.AddUnique(Data.Breadcrumb->Allocator);
427 }
428 }
429 #endif
430
431 return Data;
432 }
433
434 bool IsEmpty() const
435 {
436 return First == nullptr;
437 }
438
439 void Append(FEventStream&& Other)
440 {
441 check(Queue == Other.Queue);
442
443 if (IsEmpty())
444 {
445 Current = Other.Current;
446 First = Other.First;
447 }
448 else if (!Other.IsEmpty())
449 {
450 Current->Header.Next = Other.First;
451 Current = Other.Current;
452 }
453
454 Other.Current = nullptr;
455 Other.First = nullptr;
456 }
