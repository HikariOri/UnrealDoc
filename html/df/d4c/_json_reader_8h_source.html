57 bool ReadNext( EJsonNotation& Notation )
58 {
59 if (!ErrorMessage.IsEmpty())
60 {
61 Notation = EJsonNotation::Error;
62 return false;
63 }
64
65 if (Stream == nullptr)
66 {
67 Notation = EJsonNotation::Error;
68 SetErrorMessage(TEXT("Null Stream"));
69 return true;
70 }
71
72 const bool AtEndOfStream = Stream->AtEnd();
73
74 if (AtEndOfStream && !FinishedReadingRootObject)
75 {
76 Notation = EJsonNotation::Error;
77 SetErrorMessage(TEXT("Improperly formatted."));
78 return true;
79 }
80
81 if (FinishedReadingRootObject && !AtEndOfStream)
82 {
83 Notation = EJsonNotation::Error;
84 SetErrorMessage(TEXT("Unexpected additional input found."));
85 return true;
86 }
87
88 if (AtEndOfStream)
89 {
90 return false;
91 }
92
93 bool ReadWasSuccess = false;
94 Identifier.Empty();
95
96 do
97 {
98 EJson CurrentState = EJson::None;
99
100 if (ParseState.Num() > 0)
101 {
102 CurrentState = ParseState.Top();
103 }
104
105 switch (CurrentState)
106 {
107 case EJson::Array:
108 ReadWasSuccess = ReadNextArrayValue( /*OUT*/ CurrentToken );
109 break;
110
111 case EJson::Object:
112 ReadWasSuccess = ReadNextObjectValue( /*OUT*/ CurrentToken );
113 break;
114
115 case EJson::None:
116 case EJson::Null:
117 case EJson::String:
118 case EJson::Number:
119 case EJson::Boolean:
120 default:
121 ReadWasSuccess = ReadStart( /*OUT*/ CurrentToken );
122 break;
123 }
124 }
125 while (ReadWasSuccess && (CurrentToken == EJsonToken::None));
126
127#if WITH_JSON_INLINED_NOTATIONMAP
128 JSON_NOTATIONMAP_DEF;
129#endif // WITH_JSON_INLINED_NOTATIONMAP
130
131 Notation = TokenToNotationTable[(int32)CurrentToken];
132 FinishedReadingRootObject = ParseState.Num() == 0;
133
134 if (!ReadWasSuccess || (Notation == EJsonNotation::Error))
135 {
136 Notation = EJsonNotation::Error;
137
138 if (ErrorMessage.IsEmpty())
139 {
140 SetErrorMessage(TEXT("Unknown Error Occurred"));
141 }
142
143 return true;
144 }
145
146 if (FinishedReadingRootObject && !Stream->AtEnd())
147 {
148 ReadWasSuccess = ParseWhiteSpace();
149 }
150
151 return ReadWasSuccess;
152 }

268 bool ReadUntilMatching( const EJsonNotation ExpectedNotation )
269 {
270 uint32 ScopeCount = 0;
271 EJsonNotation Notation;
272
273 while (ReadNext(Notation))
274 {
275 if ((ScopeCount == 0) && (Notation == ExpectedNotation))
276 {
277 return true;
278 }
279
280 switch (Notation)
281 {
282 case EJsonNotation::ObjectStart:
283 case EJsonNotation::ArrayStart:
284 ++ScopeCount;
285 break;
286
287 case EJsonNotation::ObjectEnd:
288 case EJsonNotation::ArrayEnd:
289 --ScopeCount;
290 break;
291
292 case EJsonNotation::Boolean:
293 case EJsonNotation::Null:
294 case EJsonNotation::Number:
295 case EJsonNotation::String:
296 break;
297
298 case EJsonNotation::Error:
299 return false;
300 break;
301 }
302 }
303
304 return !Stream->IsError();
305 }

307 bool ReadStart( EJsonToken& Token )
308 {
309 if (!ParseWhiteSpace())
310 {
311 return false;
312 }
313
314 Token = EJsonToken::None;
315
316 if (NextToken(Token) == false)
317 {
318 return false;
319 }
320
321 if ((Token != EJsonToken::CurlyOpen) && (Token != EJsonToken::SquareOpen))
322 {
323 SetErrorMessage(TEXT("Open Curly or Square Brace token expected, but not found."));
324 return false;
325 }
326
327 return true;

330 bool ReadNextObjectValue( EJsonToken& Token )
331 {
332 const bool bCommaPrepend = Token != EJsonToken::CurlyOpen;
333 Token = EJsonToken::None;
334
335 if (NextToken(Token) == false)
336 {
337 return false;
338 }
339
340 if (Token == EJsonToken::CurlyClose)
341 {
342 return true;
343 }
344 else
345 {
346 if (bCommaPrepend)
347 {
348 if (Token != EJsonToken::Comma)
349 {
350 SetErrorMessage( TEXT("Comma token expected, but not found.") );
351 return false;
352 }
353
354 Token = EJsonToken::None;
355
356 if (!NextToken(Token))
357 {
358 return false;
359 }
360 }
361
362 if (Token != EJsonToken::String)
363 {
364 SetErrorMessage( TEXT("String token expected, but not found.") );
365 return false;
366 }
367
368 // Move value if possible. A conversion will happen if StringValue is not an FString
369 Identifier = FString(MoveTemp(StringValue));
370 Token = EJsonToken::None;
371
372 if (!NextToken(Token))
373 {
374 return false;
375 }
376
377 if (Token != EJsonToken::Colon)
378 {
379 SetErrorMessage( TEXT("Colon token expected, but not found.") );
380 return false;
381 }
382
383 Token = EJsonToken::None;
384
385 if (!NextToken(Token))
386 {
387 return false;
388 }
389 }
390
391 return true;

431 bool NextToken( EJsonToken& OutToken )
432 {
433 while (!Stream->AtEnd())
434 {
435 CharType Char;
436 if (!Serialize(&Char, sizeof(CharType)))
437 {
438 return false;
439 }
440 ++CharacterNumber;
441
442 if (Char == CharType('\0'))
443 {
444 break;
445 }
446
447 if (IsLineBreak(Char))
448 {
449 ++LineNumber;
450 CharacterNumber = 0;
451 }
452
453 if (!IsWhitespace(Char))
454 {
455 if (IsJsonNumber(Char))
456 {
457 bool bParseNumberSucceed = ParseNumberToken(Char);
458 if (!bParseNumberSucceed && Char != '-') // Could be -NaN, will return false when fail to parse it as -NaN later on
459 {
460 return false;
461 }
462
463 if (bParseNumberSucceed)
464 {
465 OutToken = EJsonToken::Number;
466 return true;
467 }
468 }
469
470 switch (Char)
471 {
472 case CharType('{'):
473 OutToken = EJsonToken::CurlyOpen; ParseState.Push( EJson::Object );
474 return true;
475
476 case CharType('}'):
477 {
478 OutToken = EJsonToken::CurlyClose;
479 if (ParseState.Num())
480 {
481 ParseState.Pop();
482 return true;
483 }
484 else
485 {
486 SetErrorMessage(TEXT("Unknown state reached while parsing Json token."));
487 return false;
488 }
489 }
490
491 case CharType('['):
492 OutToken = EJsonToken::SquareOpen; ParseState.Push( EJson::Array );
493 return true;
494
495 case CharType(']'):
496 {
497 OutToken = EJsonToken::SquareClose;
498 if (ParseState.Num())
499 {
500 ParseState.Pop();
501 return true;
502 }
503 else
504 {
505 SetErrorMessage(TEXT("Unknown state reached while parsing Json token."));
506 return false;
507 }
508 }
509
510 case CharType(':'):
511 OutToken = EJsonToken::Colon;
512 return true;
513
514 case CharType(','):
515 OutToken = EJsonToken::Comma;
516 return true;
517
518 case CharType('\"'):
519 {
520 if (!ParseStringToken())
521 {
522 return false;
523 }
524
525 OutToken = EJsonToken::String;
526 }
527 return true;
528
529 case CharType('t'): case CharType('T'):
530 case CharType('f'): case CharType('F'):
531 case CharType('i'): case CharType('I'):
532 case CharType('n'): case CharType('N'):
533 case CharType('-'):
534 {
535 StoredStringType Test;
536 Test += Char;
537
538 while (!Stream->AtEnd())
539 {
540 if (!Serialize(&Char, sizeof(CharType)))
541 {
542 return false;
543 }
544
545 if (IsAlphaNumber(Char)
546 || Char == '(' || Char == ')') // Could be "-nan(ind)" depending on the platform and impl of standard library when write
547 {
548 ++CharacterNumber;
549 Test += Char;
550 }
551 else
552 {
553 // backtrack and break
554 Stream->Seek(Stream->Tell() - sizeof(CharType));
555 break;
556 }
557 }
558
559 if (Test == CHARTEXT(CharType, "False"))
560 {
561 BoolValue = false;
562 OutToken = EJsonToken::False;
563 return true;
564 }
565
566 if (Test == CHARTEXT(CharType, "True"))
567 {
568 BoolValue = true;
569 OutToken = EJsonToken::True;
570 return true;
571 }
572
573 if (Test == CHARTEXT(CharType, "Null"))
574 {
575 OutToken = EJsonToken::Null;
576 return true;
577 }
578
579 if (Test.Compare(CHARTEXT(CharType, "Inf"), ESearchCase::IgnoreCase) == 0)
580 {
581 StringValue = Test;
582 NumberValue = std::numeric_limits<double>::infinity();
583 OutToken = EJsonToken::Number;
584 return true;
585 }
586
587 if (Test.Compare(CHARTEXT(CharType, "-Inf"), ESearchCase::IgnoreCase) == 0)
588 {
589 StringValue = Test;
590 NumberValue = -std::numeric_limits<double>::infinity();
591 OutToken = EJsonToken::Number;
592 return true;
593 }
594
595 if (Test.Compare(CHARTEXT(CharType, "NaN"), ESearchCase::IgnoreCase) == 0)
596 {
597 StringValue = Test;
598 NumberValue = std::numeric_limits<double>::quiet_NaN();
599 OutToken = EJsonToken::Number;
600 return true;
601 }
602
603 if (Test.Compare(TEXT("-NaN"), ESearchCase::IgnoreCase) == 0 ||
604 Test.Compare(TEXT("-NaN(ind)"), ESearchCase::IgnoreCase) == 0)
605 {
606 StringValue = Test;
607 NumberValue = -std::numeric_limits<double>::quiet_NaN();
608 OutToken = EJsonToken::Number;
609 return true;
610 }
611
612 SetErrorMessage( TEXT("Invalid Json Token. Check that your member names have quotes around them!") );
613 return false;
614 }
615
616 default:
617 SetErrorMessage( TEXT("Invalid Json Token.") );
618 return false;
619 }
620 }
621 }
622
623 SetErrorMessage( TEXT("Invalid Json Token.") );
624 return false;

627 bool ParseStringToken()
628 {
629 TStringBuilderWithBuffer<StoredCharType, 512> StringBuffer;
630 TStringBuilderWithBuffer<WIDECHAR, 512> UTF16CodePoints;
631
632 // Add escaped surrogate pairs
633 auto ConditionallyAddCodePoints = [&StringBuffer, &UTF16CodePoints]() -> void
634 {
635 if (UTF16CodePoints.Len() > 0)
636 {
637 // Will convert to StoredCharType encoding if needed
638 StringBuffer.Append(UTF16CodePoints);
639 UTF16CodePoints.Reset();
640 }
641 };
642
643 while (true)
644 {
645 if (Stream->AtEnd())
646 {
647 SetErrorMessage( TEXT("String Token Abruptly Ended.") );
648 return false;
649 }
650
651 CharType Char;
652 if (!Serialize(&Char, sizeof(CharType)))
653 {
654 return false;
655 }
656 ++CharacterNumber;
657
658 if (Char == CharType('\"'))
659 {
660 ConditionallyAddCodePoints();
661 break;
662 }
663
664 if (Char == CharType('\\'))
665 {
666 if (!Serialize(&Char, sizeof(CharType)))
667 {
668 return false;
669 }
670 ++CharacterNumber;
671
672 if (Char != CharType('u'))
673 {
674 ConditionallyAddCodePoints();
675 }
676
677 switch (Char)
678 {
679 case CharType('\"'): case CharType('\\'): case CharType('/'): StringBuffer.AppendChar(Char); break;
680 case CharType('f'): StringBuffer.AppendChar(CharType('\f')); break;
681 case CharType('r'): StringBuffer.AppendChar(CharType('\r')); break;
682 case CharType('n'): StringBuffer.AppendChar(CharType('\n')); break;
683 case CharType('b'): StringBuffer.AppendChar(CharType('\b')); break;
684 case CharType('t'): StringBuffer.AppendChar(CharType('\t')); break;
685 case CharType('u'):
686 // 4 hex digits, like \uAB23, which is a 16 bit number that we would usually see as 0xAB23
687 {
688 int32 HexNum = 0;
689
690 for (int32 Radix = 3; Radix >= 0; --Radix)
691 {
692 if (Stream->AtEnd())
693 {
694 SetErrorMessage( TEXT("String Token Abruptly Ended.") );
695 return false;
696 }
697
698 if (!Serialize(&Char, sizeof(CharType)))
699 {
700 return false;
701 }
702 ++CharacterNumber;
703
704 int32 HexDigit = FParse::HexDigit(static_cast<TCHAR>(Char));
705
706 if ((HexDigit == 0) && (Char != CharType('0')))
707 {
708 SetErrorMessage( TEXT("Invalid Hexadecimal digit parsed.") );
709 return false;
710 }
711
712 //@TODO: FLOATPRECISION: this is gross
713 HexNum += HexDigit * (int32)FMath::Pow(16.f, (float)Radix);
714 }
715
716 UTF16CodePoints.AppendChar((UTF16CHAR)HexNum);
717 }
718 break;
719
720 default:
721 SetErrorMessage( TEXT("Bad Json escaped char.") );
722 return false;
723 }
724 }
725 else
726 {
727 ConditionallyAddCodePoints();
728 StringBuffer.AppendChar(Char);
729 }
730 }
731
732 StringValue = StoredStringType(StringBuffer);
733
734 return true;
735 }

737 bool ParseNumberToken( CharType FirstChar )
738 {
739 StoredStringType String;
740 int32 State = 0;
741 bool UseFirstChar = true;
742 bool StateError = false;
743
744 while (true)
745 {
746 if (Stream->AtEnd())
747 {
748 SetErrorMessage( TEXT("Number Token Abruptly Ended.") );
749 return false;
750 }
751
752 CharType Char;
753 if (UseFirstChar)
754 {
755 Char = FirstChar;
756 UseFirstChar = false;
757 }
758 else
759 {
760 if (!Serialize(&Char, sizeof(CharType)))
761 {
762 return false;
763 }
764 ++CharacterNumber;
765 }
766
767 // The following code doesn't actually derive the Json Number: that is handled
768 // by the function FPlatformString::Atod below. This code only ensures the Json Number is
769 // EXACTLY to specification
770 if (IsJsonNumber(Char))
771 {
772 // ensure number follows Json format before converting
773 // This switch statement is derived from a finite state automata
774 // derived from the Json spec. A table was not used for simplicity.
775 switch (State)
776 {
777 case 0:
778 if (Char == CharType('-')) { State = 1; }
779 else if (Char == CharType('0')) { State = 2; }
780 else if (IsNonZeroDigit(Char)) { State = 3; }
781 else { StateError = true; }
782 break;
783
784 case 1:
785 if (Char == CharType('0')) { State = 2; }
786 else if (IsNonZeroDigit(Char)) { State = 3; }
787 else { StateError = true; }
788 break;
789
790 case 2:
791 if (Char == CharType('.')) { State = 4; }
792 else if (Char == CharType('e') || Char == CharType('E')) { State = 5; }
793 else { StateError = true; }
794 break;
795
796 case 3:
797 if (IsDigit(Char)) { State = 3; }
798 else if (Char == CharType('.')) { State = 4; }
799 else if (Char == CharType('e') || Char == CharType('E')) { State = 5; }
800 else { StateError = true; }
801 break;
802
803 case 4:
804 if (IsDigit(Char)) { State = 6; }
805 else { StateError = true; }
806 break;
807
808 case 5:
809 if (Char == CharType('-') ||Char == CharType('+')) { State = 7; }
810 else if (IsDigit(Char)) { State = 8; }
811 else { StateError = true; }
812 break;
813
814 case 6:
815 if (IsDigit(Char)) { State = 6; }
816 else if (Char == CharType('e') || Char == CharType('E')) { State = 5; }
817 else { StateError = true; }
818 break;
819
820 case 7:
821 if (IsDigit(Char)) { State = 8; }
822 else { StateError = true; }
823 break;
824
825 case 8:
826 if (IsDigit(Char)) { State = 8; }
827 else { StateError = true; }
828 break;
829
830 default:
831 SetErrorMessage( TEXT("Unknown state reached in Json Number Token.") );
832 return false;
833 }
834
835 if (StateError)
836 {
837 break;
838 }
839
840 String += Char;
841 }
842 else
843 {
844 // backtrack once because we read a non-number character
845 Stream->Seek(Stream->Tell() - sizeof(CharType));
846 --CharacterNumber;
847 // and now the number is fully tokenized
848 break;
849 }
850 }
851
852 // ensure the number has followed valid Json format
853 if (!StateError && ((State == 2) || (State == 3) || (State == 6) || (State == 8)))
854 {
855 StringValue = String;
856 NumberValue = FPlatformString::Atod(*String);
857 return true;
858 }
859
860 if (FirstChar != '-') // Could be -NaN, will set the error message when fail to parse it as -NaN later on
861 {
862 SetErrorMessage( TEXT("Poorly formed Json Number Token.") );
863 }
864
865 return false;
866 }

868 bool ParseWhiteSpace()
869 {
870 while (!Stream->AtEnd())
871 {
872 CharType Char;
873 if (!Serialize(&Char, sizeof(CharType)))
874 {
875 return false;
876 }
877 ++CharacterNumber;
878
879 if (IsLineBreak(Char))
880 {
881 ++LineNumber;
882 CharacterNumber = 0;
883 }
884
885 if (!IsWhitespace(Char))
886 {
887 // backtrack and break
888 Stream->Seek(Stream->Tell() - sizeof(CharType));
889 --CharacterNumber;
890 break;
891 }
892 }
893 return true;

908 bool IsJsonNumber( const CharType& Char )
909 {
910 return ((Char >= CharType('0') && Char <= CharType('9')) ||
911 Char == CharType('-') || Char == CharType('.') || Char == CharType('+') || Char == CharType('e') || Char == CharType('E'));
912 }
913
915 bool IsDigit( const CharType& Char )
916 {
917 return (Char >= CharType('0') && Char <= CharType('9'));
918 }
919
920 bool IsNonZeroDigit( const CharType& Char )
921 {
922 return (Char >= CharType('1') && Char <= CharType('9'));
923 }
924
926 bool IsAlphaNumber( const CharType& Char )
927 {
928 return (Char >= CharType('a') && Char <= CharType('z')) || (Char >= CharType('A') && Char <= CharType('Z'));
929 }
