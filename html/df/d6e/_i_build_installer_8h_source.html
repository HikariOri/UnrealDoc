591inline void LexFromString(EBuildPatchInstallError& Error, const TCHAR* Buffer)
592{
593#define RETURN_IF_EQUAL(Value) if (FCString::Stricmp(Buffer, TEXT(#Value)) == 0) { Error = EBuildPatchInstallError::Value; return; }
594 const TCHAR* const Prefix = TEXT("EBuildPatchInstallError::");
595 const SIZE_T PrefixLen = FCString::Strlen(Prefix);
596 if (FCString::Strnicmp(Buffer, Prefix, PrefixLen) == 0)
597 {
598 Buffer += PrefixLen;
599 }
600 RETURN_IF_EQUAL(NoError);
601 RETURN_IF_EQUAL(DownloadError);
602 RETURN_IF_EQUAL(FileConstructionFail);
603 RETURN_IF_EQUAL(MoveFileToInstall);
604 RETURN_IF_EQUAL(BuildVerifyFail);
605 RETURN_IF_EQUAL(ApplicationClosing);
606 RETURN_IF_EQUAL(ApplicationError);
607 RETURN_IF_EQUAL(UserCanceled);
608 RETURN_IF_EQUAL(PrerequisiteError);
609 RETURN_IF_EQUAL(InitializationError);
610 RETURN_IF_EQUAL(PathLengthExceeded);
611 RETURN_IF_EQUAL(OutOfDiskSpace);
612 RETURN_IF_EQUAL(CustomUninstallActionError);
613 // Did not match
614 Error = EBuildPatchInstallError::InvalidOrMax;
615 return;
616#undef RETURN_IF_EQUAL
617}
