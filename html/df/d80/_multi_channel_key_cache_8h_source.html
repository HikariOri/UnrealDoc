186 void RecomputeForDrag(const TSharedRef<FToolableTimeline>& InTimeline
187 , const FFrameTime& InDeltaTickTime)
188 {
189 const TSharedRef<FToolableTimeSliderController> TimeSliderController = InTimeline->GetTimeSliderController();
190 const FFrameRate TickResolution = TimeSliderController->GetTickResolution();
191 const FFrameRate DisplayRate = TimeSliderController->GetDisplayRate();
192
193 const bool bSnapEnabled = TimeSliderController->IsSnapEnabled();
194 const int32 ChannelCount = ChannelCache.Num();
195
196 // Compute the new key times
197 // ParallelFor has overhead; don't pay it for tiny workloads
198 if (ChannelCount >= ParallelThreshold)
199 {
200 TArray<FChannelKeyRangeCache<TKeyCacheType>>& ChannelCacheRef = ChannelCache;
201
202 ParallelFor(ChannelCount, [&ChannelCacheRef, &TickResolution, &DisplayRate, InDeltaTickTime, bSnapEnabled]
203 (const int32 InChannelIndex)
204 {
205 if (ensure(ChannelCacheRef.IsValidIndex(InChannelIndex)))
206 {
207 RecomputeLastDraggedFrameTimesFromDrag(ChannelCacheRef[InChannelIndex]
208 , TickResolution, DisplayRate, InDeltaTickTime, bSnapEnabled);
209 }
210 });
211 }
212 else
213 {
214 for (int32 ChannelIndex = 0; ChannelIndex < ChannelCount; ++ChannelIndex)
215 {
216 RecomputeLastDraggedFrameTimesFromDrag(ChannelCache[ChannelIndex]
217 , TickResolution, DisplayRate, InDeltaTickTime, bSnapEnabled);
218 }
219 }

258 void ApplyKeyTimes(const TSharedRef<FToolableTimeline>& InTimeline, const bool bInNotifyMovieSceneDataChanged = true)
259 {
260 TArray<FPreparedChannelApply> PreparedChannels;
261 PrepareKeyTimeChanges(InTimeline, PreparedChannels);
262
263 bool bAnyChanged = false;
264
265 for (FPreparedChannelApply& Prepared : PreparedChannels)
266 {
267 if (!Prepared.bAnyChanged || !Prepared.KeyArea.IsValid())
268 {
269 continue;
270 }
271
272 UObject* const OwningObject = Prepared.OwningObject.Get();
273 UMovieSceneSection* const Section = Prepared.Section.Get();
274 if (!Section || Section->IsReadOnly())
275 {
276 continue;
277 }
278
279 MarkOwningObjectModifiedOnce(OwningObject);
280
281 if (Prepared.Extents.bInitialized)
282 {
283 const TRange<FFrameNumber> SectionRange = Section->GetRange();
284 const FFrameNumber MinFrame = Prepared.Extents.Min.FloorToFrame();
285 const FFrameNumber MaxFrame = Prepared.Extents.Max.CeilToFrame();
286
287 if (!SectionRange.Contains(MinFrame))
288 {
289 Section->ExpandToFrame(MinFrame);
290 }
291 if (!SectionRange.Contains(MaxFrame))
292 {
293 Section->ExpandToFrame(MaxFrame);
294 }
295 }
296
297 Prepared.AppliedFrames.SetNumUninitialized(Prepared.KeyHandles.Num());
298 Prepared.KeyArea->SetKeyTimes(Prepared.KeyHandles, Prepared.RequestedFrames);
299 Prepared.KeyArea->GetKeyTimes(Prepared.KeyHandles, Prepared.AppliedFrames);
300
301 for (int32 KeyIndex = 0; KeyIndex < Prepared.KeyChanges.Num(); ++KeyIndex)
302 {
303 FPreparedKeyTimeChange& KeyChange = Prepared.KeyChanges[KeyIndex];
304 if (ensure(KeyChange.KeyCache) && ensure(Prepared.AppliedFrames.IsValidIndex(KeyIndex)))
305 {
306 // Track the key's actual post-apply frame rather than assuming the requested
307 // frame always stuck. This keeps future drag ticks from being skipped forever
308 // when SetKeyTime is rejected, clamped, or otherwise lands somewhere else.
309 KeyChange.KeyCache->LastAppliedFrameTime = FFrameTime(Prepared.AppliedFrames[KeyIndex]);
310 }
311 }
312 bAnyChanged = true;
313 }
314
315 if (bAnyChanged)
316 {
317 for (const TWeakObjectPtr<> WeakObject : WeakModifiedObjectsThisDrag)
318 {
319 if (UMovieSceneSignedObject* const SignedObject = Cast<UMovieSceneSignedObject>(WeakObject.Get()))
320 {
321 SignedObject->MarkAsChanged();
322 }
323 }
324
325 if (const TSharedPtr<ISequencer> Sequencer = InTimeline->GetSequencer())
326 {
327 if (bInNotifyMovieSceneDataChanged)
328 {
329 Sequencer->NotifyMovieSceneDataChanged(EMovieSceneDataChangeType::TrackValueChanged);
330 }
331 else
332 {
333 InTimeline->GetTimeSliderController()->InvalidateKeyRendererCache();
334 Sequencer->RequestEvaluate();
335 }
336 }
337 }

373 static void RecomputeLastDraggedFrameTimesFromDrag(FChannelKeyRangeCache<TKeyCacheType>& InChannelCache
374 , const FFrameRate& InTickResolution
375 , const FFrameRate& InDisplayRate
376 , const FFrameTime& InDeltaTickTime
377 , const bool bInSnapToFrame)
378 {
379 InChannelCache.ForEachKey([&InTickResolution, &InDisplayRate, bInSnapToFrame, &InDeltaTickTime]
380 (FChannelKeyRangeCache<TKeyCacheType>& InKeyRangeCache
381 , const TViewModelPtr<FChannelModel>& InChannelModel
382 , const int32 InKeyIndex
383 , TKeyCacheType& InKeyCache)
384 {
385 FFrameTime NewKeyTime = InKeyCache.InitialFrameTime + InDeltaTickTime;
386
387 if (bInSnapToFrame)
388 {
389 const FFrameTime DisplayTime = FFrameRate::TransformTime(NewKeyTime, InTickResolution, InDisplayRate);
390 const FFrameNumber SnappedDisplayFrame = DisplayTime.RoundToFrame();
391 NewKeyTime = FFrameRate::TransformTime(FFrameTime(SnappedDisplayFrame), InDisplayRate, InTickResolution);
392 }
393
394 InKeyCache.LastDraggedFrameTime = NewKeyTime;
395
396 return true;
397 });
398 }

400 void PrepareKeyTimeChanges(const TSharedRef<FToolableTimeline>& InTimeline
401 , TArray<FPreparedChannelApply>& OutPreparedChannels)
402 {
403 const int32 ChannelCount = ChannelCache.Num();
404
405 OutPreparedChannels.Reset();
406 OutPreparedChannels.SetNum(ChannelCount);
407
408 auto PrepareSingleChannel = [this, &OutPreparedChannels](const int32 InChannelIndex)
409 {
410 if (!ensure(ChannelCache.IsValidIndex(InChannelIndex)))
411 {
412 return;
413 }
414
415 FChannelKeyRangeCache<TKeyCacheType>& InChannelCache = ChannelCache[InChannelIndex];
416
417 const TViewModelPtr<FChannelModel> InChannelModel = InChannelCache.WeakChannelModel.Pin();
418 if (!InChannelModel.IsValid())
419 {
420 return;
421 }
422
423 const TSharedPtr<IKeyArea> KeyArea = InChannelModel->GetKeyArea();
424 if (!KeyArea.IsValid())
425 {
426 return;
427 }
428
429 UMovieSceneSection* const Section = InChannelModel->GetSection();
430 if (!Section || Section->IsReadOnly())
431 {
432 return;
433 }
434
435 UObject* const OwningObject = InChannelModel->GetOwningObject();
436
437 FPreparedChannelApply Prepared;
438 Prepared.ChannelIndex = InChannelIndex;
439 Prepared.KeyArea = KeyArea;
440 Prepared.Section = Section;
441 Prepared.OwningObject = OwningObject;
442 Prepared.KeyChanges.Reserve(InChannelCache.KeyCache.Num());
443 Prepared.KeyHandles.Reserve(InChannelCache.KeyCache.Num());
444 Prepared.RequestedFrames.Reserve(InChannelCache.KeyCache.Num());
445
446 for (TKeyCacheType& InKeyCache : InChannelCache.KeyCache)
447 {
448 // IMPORTANT:
449 // RecomputeForDrag() already produced LastDraggedFrameTime.
450 // Do not resnap here.
451 const FFrameTime NewKeyTime = InKeyCache.LastDraggedFrameTime;
452
453 // Skip if this key would apply to the same rounded frame as last time.
454 if (AreFrameTimesEquivalentForApply(InKeyCache.LastAppliedFrameTime, NewKeyTime))
455 {
456 continue;
457 }
458
459 FPreparedKeyTimeChange& NewChange = Prepared.KeyChanges.AddDefaulted_GetRef();
460 NewChange.Handle = InKeyCache.Handle;
461 NewChange.NewTime = NewKeyTime;
462 NewChange.RequestedFrame = NewKeyTime.RoundToFrame();
463 NewChange.KeyCache = &InKeyCache;
464
465 Prepared.KeyHandles.Add(NewChange.Handle);
466 Prepared.RequestedFrames.Add(NewChange.RequestedFrame);
467
468 if (!Prepared.Extents.bInitialized)
469 {
470 Prepared.Extents.Min = NewKeyTime;
471 Prepared.Extents.Max = NewKeyTime;
472 Prepared.Extents.bInitialized = true;
473 }
474 else
475 {
476 Prepared.Extents.Min = FMath::Min(Prepared.Extents.Min, NewKeyTime);
477 Prepared.Extents.Max = FMath::Max(Prepared.Extents.Max, NewKeyTime);
478 }
479
480 Prepared.bAnyChanged = true;
481 }
482
483 OutPreparedChannels[InChannelIndex] = MoveTemp(Prepared);
484 };
485
486 if (ChannelCount >= ParallelThreshold)
487 {
488 ParallelFor(ChannelCount, [&PrepareSingleChannel](const int32 InChannelIndex)
489 {
490 PrepareSingleChannel(InChannelIndex);
491 });
492 }
493 else
494 {
495 for (int32 ChannelIndex = 0; ChannelIndex < ChannelCount; ++ChannelIndex)
496 {
497 PrepareSingleChannel(ChannelIndex);
498 }
499 }
500 }
