140 static FRSAKeyHandle ParseRSAKeyFromJson(TSharedPtr<FJsonObject> InObj)
141 {
142 TSharedPtr<FJsonObject> PublicKey = InObj->GetObjectField(TEXT("PublicKey"));
143 TSharedPtr<FJsonObject> PrivateKey = InObj->GetObjectField(TEXT("PrivateKey"));
144
145 FString PublicExponentBase64, PrivateExponentBase64, PublicModulusBase64, PrivateModulusBase64;
146
147 if (PublicKey->TryGetStringField(TEXT("Exponent"), PublicExponentBase64)
148 && PublicKey->TryGetStringField(TEXT("Modulus"), PublicModulusBase64)
149 && PrivateKey->TryGetStringField(TEXT("Exponent"), PrivateExponentBase64)
150 && PrivateKey->TryGetStringField(TEXT("Modulus"), PrivateModulusBase64))
151 {
152 check(PublicModulusBase64 == PrivateModulusBase64);
153
154 TArray<uint8> PublicExponent, PrivateExponent, Modulus;
155 FBase64::Decode(PublicExponentBase64, PublicExponent);
156 FBase64::Decode(PrivateExponentBase64, PrivateExponent);
157 FBase64::Decode(PublicModulusBase64, Modulus);
158
159 return FRSA::CreateKey(PublicExponent, PrivateExponent, Modulus);
160 }
161 else
162 {
163 return nullptr;
164 }
165 }

174 static FKeyChainLoadResult LoadKeyChainFromFileEx(const FString& InFilename, FKeyChain& OutCryptoSettings, EKeyChainLoadFlags Flags = EKeyChainLoadFlags::Default)
175 {
176 // Open file
177 FArchive* File = IFileManager::Get().CreateFileReader(*InFilename);
178 if (!File)
179 {
180 return FKeyChainLoadResult::Error(FString::Printf(TEXT("Crypto keys file '%s' does not exist"), *InFilename));
181 }
182
183 // Parse JSON
184 TSharedPtr<FJsonObject> RootObject;
185 TSharedRef<TJsonReader<UTF8CHAR>> Reader = TJsonReaderFactory<UTF8CHAR>::Create(File);
186 if (!FJsonSerializer::Deserialize(Reader, RootObject) || !RootObject.IsValid())
187 {
188 delete File;
189 return FKeyChainLoadResult::Error(FString::Printf(TEXT("Failed to parse JSON from '%s'"), *InFilename));
190 }
191 delete File;
192
193 const bool bLoadSigningKeys = EnumHasAnyFlags(Flags, EKeyChainLoadFlags::LoadSigningKeys);
194
195 // Load primary encryption key
196 const TSharedPtr<FJsonObject>* EncryptionKeyObject;
197 if (RootObject->TryGetObjectField(TEXT("EncryptionKey"), EncryptionKeyObject))
198 {
199 FString EncryptionKeyBase64;
200 if ((*EncryptionKeyObject)->TryGetStringField(TEXT("Key"), EncryptionKeyBase64))
201 {
202 if (EncryptionKeyBase64.Len() > 0)
203 {
204 TArray<uint8> Key;
205 FBase64::Decode(EncryptionKeyBase64, Key);
206 checkf(Key.Num() == sizeof(FAES::FAESKey::Key), TEXT("Primary encryption key has invalid size: %d bytes (expected %d) in file '%s'"), Key.Num(), (int32)sizeof(FAES::FAESKey::Key), *InFilename);
207 FNamedAESKey NewKey;
208 NewKey.Name = TEXT("Default");
209 NewKey.Guid = FGuid();
210 FMemory::Memcpy(NewKey.Key.Key, &Key[0], sizeof(FAES::FAESKey::Key));
211 OutCryptoSettings.GetEncryptionKeys().Add(NewKey.Guid, NewKey);
212 }
213 }
214 }
215
216 // Load RSA signing key (optional, requires engine crypto)
217 const TSharedPtr<FJsonObject>* SigningKey = nullptr;
218 if (RootObject->TryGetObjectField(TEXT("SigningKey"), SigningKey))
219 {
220 if (bLoadSigningKeys)
221 {
222 // Try to parse RSA key - this may fail if engine crypto isn't initialized
223 FRSAKeyHandle KeyHandle = ParseRSAKeyFromJson(*SigningKey);
224 if (KeyHandle == InvalidRSAKeyHandle)
225 {
226 return FKeyChainLoadResult::Error(TEXT("Failed to load RSA signing key (engine crypto may not be initialized)"));
227 }
228 OutCryptoSettings.SetSigningKey(KeyHandle);
229 }
230 // If signing keys not requested, just skip them silently
231 }
232
233 // Load secondary encryption keys
234 const TArray<TSharedPtr<FJsonValue>>* SecondaryEncryptionKeyArray = nullptr;
235 if (RootObject->TryGetArrayField(TEXT("SecondaryEncryptionKeys"), SecondaryEncryptionKeyArray))
236 {
237 for (TSharedPtr<FJsonValue> EncryptionKeyValue : *SecondaryEncryptionKeyArray)
238 {
239 FNamedAESKey NewKey;
240 TSharedPtr<FJsonObject> SecondaryEncryptionKeyObject = EncryptionKeyValue->AsObject();
241 FGuid::Parse(SecondaryEncryptionKeyObject->GetStringField(TEXT("Guid")), NewKey.Guid);
242 NewKey.Name = SecondaryEncryptionKeyObject->GetStringField(TEXT("Name"));
243 FString KeyBase64 = SecondaryEncryptionKeyObject->GetStringField(TEXT("Key"));
244
245 TArray<uint8> Key;
246 FBase64::Decode(KeyBase64, Key);
247 checkf(Key.Num() == sizeof(FAES::FAESKey::Key), TEXT("Secondary encryption key '%s' (%s) has invalid size: %d bytes (expected %d) in file '%s'"), *NewKey.Name, *NewKey.Guid.ToString(), Key.Num(), (int32)sizeof(FAES::FAESKey::Key), *InFilename);
248 FMemory::Memcpy(NewKey.Key.Key, &Key[0], sizeof(FAES::FAESKey::Key));
249
250 checkf(!OutCryptoSettings.GetEncryptionKeys().Contains(NewKey.Guid) || OutCryptoSettings.GetEncryptionKeys()[NewKey.Guid].Key == NewKey.Key, TEXT("Duplicate encryption key GUID %s with conflicting key values in file '%s'"), *NewKey.Guid.ToString(), *InFilename);
251 OutCryptoSettings.GetEncryptionKeys().Add(NewKey.Guid, NewKey);
252 }
253 }
254
255 FGuid EncryptionKeyOverrideGuid;
256 OutCryptoSettings.SetPrincipalEncryptionKey(OutCryptoSettings.GetEncryptionKeys().Find(EncryptionKeyOverrideGuid));
257
258 return FKeyChainLoadResult::Success();
259 }
