315private:
316 EPersistentStorageManagerFileSizeFlags TryUpdateQuota(const int64 OldFileSize, const int64 NewFileSize, EPersistentStorageManagerFileSizeFlags Flags)
317 {
318 check(OldFileSize >= 0);
319 check(NewFileSize >= 0);
320
321 if (NewFileSize <= OldFileSize)
322 {
323 UsedQuota -= (OldFileSize - NewFileSize);
324 return EPersistentStorageManagerFileSizeFlags::None;
325 }
326
327 if (EnumHasAnyFlags(Flags, EPersistentStorageManagerFileSizeFlags::OnlyUpdateIfLess))
328 {
329 return EPersistentStorageManagerFileSizeFlags::OnlyUpdateIfLess;
330 }
331
332 if (EnumHasAnyFlags(Flags, EPersistentStorageManagerFileSizeFlags::RespectQuota) && StorageQuota >= 0)
333 {
334 int64 OldUsedQuota = UsedQuota;
335 int64 NewUsedQuota;
336 do
337 {
338 NewUsedQuota = OldUsedQuota + (NewFileSize - OldFileSize);
339 if (NewUsedQuota > StorageQuota)
340 {
341 return EPersistentStorageManagerFileSizeFlags::RespectQuota;
342 }
343 } while (!UsedQuota.compare_exchange_weak(OldUsedQuota, NewUsedQuota));
344
345 return EPersistentStorageManagerFileSizeFlags::None;
346 }
347
348 UsedQuota += (NewFileSize - OldFileSize);
349 return EPersistentStorageManagerFileSizeFlags::None;
350 }
