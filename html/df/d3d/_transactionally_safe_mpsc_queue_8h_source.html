33 [[nodiscard]] TTransactionallySafeMpscQueue()
34 {
35 FNode* Node = ::new (AllocatorType::Malloc(sizeof(FNode), alignof(FNode))) FNode;
36 Head = First = Tail = Node;
37 }
38
39 ~TTransactionallySafeMpscQueue()
40 {
41 // Nobody should have a reference to this class anymore, but we still take the mutex to
42 // guarantee that the queue isn't modified by another thread while a transaction is underway.
43 UE::TScopeLock Lock(Mutex);
44 // Free all recycled/sentinel nodes (First up to and including Tail).
45 FNode* Node = First;
46 while (Node != Tail)
47 {
48 FNode* Next = Node->Next;
49 AllocatorType::Free(Node);
50 Node = Next;
51 }
52
53 // Free the sentinel (Tail) and advance to occupied nodes.
54 Node = Tail->Next;
55 AllocatorType::Free(Tail);
56
57 // Destroy and free all occupied nodes.
58 while (Node != nullptr)
59 {
60 FNode* Next = Node->Next;
61 DestructItem((ElementType*)&Node->Value);
62 AllocatorType::Free(Node);
63 Node = Next;
64 }
65 }

78 // Returns NullOpt if the queue is empty.
79 // Spin-waiting on Dequeue from within an AutoRTFM transaction is likely to deadlock,
80 // as the matching Enqueue will be waiting on Mutex.
81 TOptional<ElementType> Dequeue()
82 {
83 Mutex.Lock();
84 FNode* LocalTail = Tail;
85 FNode* LocalTailNext = LocalTail->Next;
86 Mutex.Unlock();
87
88 if (LocalTailNext == nullptr)
89 {
90 return NullOpt;
91 }
92
93 ElementType* TailNextValue = (ElementType*)&LocalTailNext->Value;
94 TOptional<ElementType> Value{ MoveTemp(*TailNextValue) };
95 DestructItem(TailNextValue);
96
97 Mutex.Lock();
98 Tail = LocalTailNext;
99 Mutex.Unlock();
100
101 return Value;
102 }

104 bool Dequeue(ElementType& OutElem)
105 {
106 TOptional<ElementType> LocalElement = Dequeue();
107 if (LocalElement.IsSet())
108 {
109 OutElem = MoveTempIfPossible(LocalElement.GetValue());
110 return true;
111 }
112
113 return false;
114 }

116 [[nodiscard]] bool IsEmpty() const
117 {
118 UE::TScopeLock Lock(Mutex);
119 FNode* LocalTail = Tail;
120 FNode* LocalTailNext = LocalTail->Next;
121
122 return LocalTailNext == nullptr;
123 }

125 // As there can be only one consumer, a consumer can safely "peek" the tail of the queue.
126 // Returns a pointer to the tail if the queue is not empty, nullptr otherwise.
127 // There's no overload with TOptional as it doesn't support references.
128 [[nodiscard]] ElementType* Peek() const
129 {
130 UE::TScopeLock Lock(Mutex);
131 FNode* LocalTail = Tail;
132 FNode* LocalTailNext = LocalTail->Next;
133
134 return LocalTailNext
135 ? (ElementType*)&LocalTailNext->Value
136 : nullptr;
137 }

139private:
140 struct FNode
141 {
142 FNode* Next = nullptr;
143 TTypeCompatibleBytes<ElementType> Value;
144 };

146 FNode* AllocNode()
147 {
148 // Attempt to recycle a consumed node from the free list (First..Tail range).
149 {
150 UE::TScopeLock Lock(Mutex);
151 if (First != Tail)
152 {
153 FNode* Node = First;
154 First = First->Next;
155 Node->Next = nullptr;
156 return Node;
157 }
158 }
159
160 // Free list is empty; allocate a new node.
161 return ::new (AllocatorType::Malloc(sizeof(FNode), alignof(FNode))) FNode();
162 }
163
164 // This mutex guards all accesses to this structure or to FNode::Next.
165 mutable UE::FTransactionallySafeMutex Mutex;
166
167 FNode* Tail; // tail of the queue (consumer end)
168 FNode* Head; // head of the queue (producer end)
169 FNode* First; // head of the recycled node free list
170};
