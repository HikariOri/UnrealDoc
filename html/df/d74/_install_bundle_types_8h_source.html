150 Result.CurrentSizeOnDisk = CurrentSizeOnDisk + Other.CurrentSizeOnDisk;
151 Result.CurrentSizeOnDiskOtherDirs = CurrentSizeOnDiskOtherDirs + Other.CurrentSizeOnDiskOtherDirs;
