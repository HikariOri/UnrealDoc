332inline ChangeMaskStorageType* FDeltaCompressionBaselineManager::GetChangeMaskPointerForBaseline(const FPerObjectInfo* ObjectInfo, uint32 ConnId, uint32 BaselineIndex) const
333{
334 // Layout of storage per object is:
335 // ChangeMaskForConnections[0 .. ConnectionCount]
336 // ChangeMaskForBaselines[Conn0[0+1] .. ConnConnectionCount[0+1]]
337 const uint32 ChangeMaskStride = ObjectInfo->ChangeMaskStride;
338 const SIZE_T BaselineChangeMaskOffset = (MaxConnectionCount + 2U*ConnId + BaselineIndex)*ChangeMaskStride;
339 ChangeMaskStorageType* ChangeMaskPointer = ObjectInfo->ChangeMasksForConnections + BaselineChangeMaskOffset;
340 return ChangeMaskPointer;
341}
342
343inline ChangeMaskStorageType* FDeltaCompressionBaselineManager::GetChangeMaskPointerForConnection(const FPerObjectInfo* ObjectInfo, uint32 ConnId) const
344{
345 const uint32 ChangeMaskStride = ObjectInfo->ChangeMaskStride;
346 const SIZE_T ConnectionChangeMaskOffset = ConnId*ChangeMaskStride;
347 ChangeMaskStorageType* ChangeMaskPointer = ObjectInfo->ChangeMasksForConnections + ConnectionChangeMaskOffset;
348 return ChangeMaskPointer;
349}
