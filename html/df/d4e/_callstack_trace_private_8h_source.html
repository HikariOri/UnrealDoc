60 public:
61 struct FBacktraceEntry
62 {
63 uint64 Hash = 0;
64 uint32 FrameCount = 0;
65 uint64* Frames;
66 };
67
68 FCallstackTracer(FMalloc* InMalloc)
69 : KnownSet(InMalloc)
70 , CallstackIdCounter(1) // 0 is reserved for "unknown callstack"
71 {
72 }
73
74 uint32 AddCallstack(const FBacktraceEntry& Entry)
75 {
76 bool bAlreadyAdded = false;
77
78 // Our set implementation doesn't allow for zero entries (zero represents an empty element
79 // in the hash table), so if we get one due to really bad luck in our 64-bit Id calculation,
80 // treat it as a "1" instead, for purposes of tracking if we've seen that callstack.
81 const uint64 Hash = FMath::Max(Entry.Hash, 1ull);
82 uint32 Id;
83 KnownSet.Find(Hash, &Id, &bAlreadyAdded);
84 if (!bAlreadyAdded)
85 {
86 Id = CallstackIdCounter.fetch_add(1, std::memory_order_relaxed);
87 // On the first callstack reserve memory up front
88 if (Id == 1)
89 {
90 KnownSet.Reserve(InitialReserveCount);
91 }
92#if !UE_CALLSTACK_TRACE_RESERVE_GROWABLE
93 // If configured as not growable, start returning unknown id's when full.
94 if (Id >= InitialReserveCount)
95 {
96 return 0;
97 }
98#endif //UE_CALLSTACK_TRACE_RESERVE_GROWABLE
99 KnownSet.Emplace(Hash, Id);
100#if UE_CALLSTACK_TRACE_UNCOMPRESSED
101 UE_TRACE_LOG(Memory, CallstackSpec, CallstackChannel)
102 << CallstackSpec.CallstackId(Id)
103 << CallstackSpec.Frames(Entry.Frames, Entry.FrameCount);
104#else
105 uint8 Compressed[UE_CALLSTACK_TRACE_MAX_FRAMES * 9];
106 const uint32 CompressedSize = FCallstackDeltaVarInt::Compress(
107 MakeArrayView(Entry.Frames, Entry.FrameCount),
108 Compressed
109 );
110
111 UE_TRACE_LOG(Memory, CallstackSpecDeltaVarInt, CallstackChannel)
112 << CallstackSpecDeltaVarInt.CallstackId(Id)
113 << CallstackSpecDeltaVarInt.CompressedFrames(Compressed, CompressedSize);
114#endif //UE_CALLSTACK_TRACE_UNCOMPRESSED
115 }
116
117 return Id;
118 }

121 {
122 std::atomic_uint64_t Key;
123 std::atomic_uint32_t Value;
124
125 inline uint64 GetKey() const { return Key.load(std::memory_order_relaxed); }
126 inline uint32 GetValue() const { return Value.load(std::memory_order_relaxed); }
127 inline bool IsEmpty() const { return Key.load(std::memory_order_relaxed) == 0; }
128 inline void SetKeyValue(uint64 InKey, uint32 InValue)
129 {
130 Value.store(InValue, std::memory_order_release);
131 Key.store(InKey, std::memory_order_relaxed);
132 }
133 static inline uint32 KeyHash(uint64 Key) { return static_cast<uint32>(Key); }
134 static inline void ClearEntries(FEncounteredCallstackSetEntry* Entries, int32 EntryCount)
135 {
136 memset(Entries, 0, EntryCount * sizeof(FEncounteredCallstackSetEntry));
137 }
138 };

146 };
