12// Helper struct used to handle graph (editor) node alpha options
13struct FAnimGraphNodeAlphaOptions
14{
15 // This assumes that a anim node contains the following members:
16 //
17 // UPROPERTY(EditAnywhere, BlueprintReadWrite, Category = Alpha, meta = (PinShownByDefault))
18 // float Alpha;
19 //
20 // UPROPERTY(EditAnywhere, BlueprintReadWrite, Category = Alpha)
21 // FInputScaleBias AlphaScaleBias;
22 //
23 // UPROPERTY(EditAnywhere, BlueprintReadWrite, Category = Alpha, meta = (DisplayName = "Blend Settings"))
24 // FInputAlphaBoolBlend AlphaBoolBlend;
25 //
26 // UPROPERTY(EditAnywhere, BlueprintReadWrite, Category = Alpha, meta = (PinShownByDefault))
27 // FName AlphaCurveName;
28 //
29 // UPROPERTY(EditAnywhere, BlueprintReadWrite, Category = Alpha)
30 // FInputScaleBiasClamp AlphaScaleBiasClamp;
31 //
32 // UPROPERTY(EditAnywhere, BlueprintReadWrite, Category = Alpha)
33 // EAnimAlphaInputType AlphaInputType;
34 //
35 // UPROPERTY(EditAnywhere, BlueprintReadWrite, Category = Alpha, meta = (PinShownByDefault, DisplayName = "bEnabled", DisplayAfter = "AlphaScaleBias"))
36 // bool bAlphaBoolEnabled;
37 //
38 // This is implemented in this way (rather than encapsulating the options in their own struct) because
39 // we dont support exposing pins from sub-structs on anim nodes (or anything else for that matter).

41 // Called from an UAnimGraphNode_Base-derived classes CustomizePinData to update alpha pins
42 template<typename AnimNodeType>
43 static void HandleCustomizePinData(const AnimNodeType& InAnimNode, UEdGraphPin* InPin)
44 {
45 if (InPin->PinName == GET_MEMBER_NAME_STRING_CHECKED(AnimNodeType, Alpha))
46 {
47 InPin->bHidden = (InAnimNode.AlphaInputType != EAnimAlphaInputType::Float);
48
49 if (!InPin->bHidden)
50 {
51 InPin->PinFriendlyName = InAnimNode.AlphaScaleBias.GetFriendlyName(InAnimNode.AlphaScaleBiasClamp.GetFriendlyName(InPin->PinFriendlyName));
52 }
53 }
54
55 if (InPin->PinName == GET_MEMBER_NAME_STRING_CHECKED(AnimNodeType, bAlphaBoolEnabled))
56 {
57 InPin->bHidden = (InAnimNode.AlphaInputType != EAnimAlphaInputType::Bool);
58 }
59
60 if (InPin->PinName == GET_MEMBER_NAME_STRING_CHECKED(AnimNodeType, AlphaCurveName))
61 {
62 InPin->bHidden = (InAnimNode.AlphaInputType != EAnimAlphaInputType::Curve);
63
64 if (!InPin->bHidden)
65 {
66 InPin->PinFriendlyName = InAnimNode.AlphaScaleBiasClamp.GetFriendlyName(InPin->PinFriendlyName);
67 }
68 }
69 }

71 // Called from an UAnimGraphNode_Base-derived classes PostEditChangeProperty to update alpha properties and pins
72 template<typename AnimNodeType>
73 static void HandlePostEditChangeProperty(const AnimNodeType& InAnimNode, UAnimGraphNode_Base* InNode, FPropertyChangedEvent& PropertyChangedEvent)
74 {
75 const FName PropertyName = (PropertyChangedEvent.Property ? PropertyChangedEvent.Property->GetFName() : NAME_None);
76
77 // Reconstruct node to show updates to PinFriendlyNames.
78 if ((PropertyName == GET_MEMBER_NAME_STRING_CHECKED(AnimNodeType, AlphaScaleBias))
79 || (PropertyName == GET_MEMBER_NAME_STRING_CHECKED(FInputScaleBiasClamp, bMapRange))
80 || (PropertyName == GET_MEMBER_NAME_STRING_CHECKED(FInputRange, Min))
81 || (PropertyName == GET_MEMBER_NAME_STRING_CHECKED(FInputRange, Max))
82 || (PropertyName == GET_MEMBER_NAME_STRING_CHECKED(FInputScaleBiasClamp, Scale))
83 || (PropertyName == GET_MEMBER_NAME_STRING_CHECKED(FInputScaleBiasClamp, Bias))
84 || (PropertyName == GET_MEMBER_NAME_STRING_CHECKED(FInputScaleBiasClamp, bClampResult))
85 || (PropertyName == GET_MEMBER_NAME_STRING_CHECKED(FInputScaleBiasClamp, ClampMin))
86 || (PropertyName == GET_MEMBER_NAME_STRING_CHECKED(FInputScaleBiasClamp, ClampMax))
87 || (PropertyName == GET_MEMBER_NAME_STRING_CHECKED(FInputScaleBiasClamp, bInterpResult))
88 || (PropertyName == GET_MEMBER_NAME_STRING_CHECKED(FInputScaleBiasClamp, InterpSpeedIncreasing))
89 || (PropertyName == GET_MEMBER_NAME_STRING_CHECKED(FInputScaleBiasClamp, InterpSpeedDecreasing)))
90 {
91 InNode->ReconstructNode();
92 }
93
94 if (PropertyName == GET_MEMBER_NAME_STRING_CHECKED(AnimNodeType, AlphaInputType))
95 {
96 FScopedTransaction Transaction(NSLOCTEXT("AnimGraphNodeAlphaOptions", "ChangeAlphaInputType", "Change Alpha Input Type"));
97 InNode->Modify();
98
99 // Break links to pins going away
100 for (int32 PinIndex = 0; PinIndex < InNode->Pins.Num(); ++PinIndex)
101 {
102 UEdGraphPin* Pin = InNode->Pins[PinIndex];
103 if (Pin->PinName == GET_MEMBER_NAME_STRING_CHECKED(AnimNodeType, Alpha))
104 {
105 if (InAnimNode.AlphaInputType != EAnimAlphaInputType::Float)
106 {
107 Pin->BreakAllPinLinks();
108 InNode->RemoveBindings(Pin->PinName);
109 }
110 }
111 else if (Pin->PinName == GET_MEMBER_NAME_STRING_CHECKED(AnimNodeType, bAlphaBoolEnabled))
112 {
113 if (InAnimNode.AlphaInputType != EAnimAlphaInputType::Bool)
114 {
115 Pin->BreakAllPinLinks();
116 InNode->RemoveBindings(Pin->PinName);
117 }
118 }
119 else if (Pin->PinName == GET_MEMBER_NAME_STRING_CHECKED(AnimNodeType, AlphaCurveName))
120 {
121 if (InAnimNode.AlphaInputType != EAnimAlphaInputType::Curve)
122 {
123 Pin->BreakAllPinLinks();
124 InNode->RemoveBindings(Pin->PinName);
125 }
126 }
127 }
128
129 InNode->ReconstructNode();
130
131 FBlueprintEditorUtils::MarkBlueprintAsStructurallyModified(InNode->GetBlueprint());
132 }
133 }

135 // Called from an UAnimGraphNode_Base-derived classes CustomizeDetails to update alpha properties and pins
136 template<typename AnimNodeType>
137 static void HandleCustomizeDetails(const AnimNodeType& InAnimNode, TSharedRef<IPropertyHandle> InNodeHandle, IDetailLayoutBuilder& InDetailBuilder)
138 {
139 if (InAnimNode.AlphaInputType != EAnimAlphaInputType::Bool)
140 {
141 InDetailBuilder.HideProperty(InNodeHandle->GetChildHandle(GET_MEMBER_NAME_CHECKED(AnimNodeType, bAlphaBoolEnabled)));
142 InDetailBuilder.HideProperty(InNodeHandle->GetChildHandle(GET_MEMBER_NAME_CHECKED(AnimNodeType, AlphaBoolBlend)));
143 }
144
145 if (InAnimNode.AlphaInputType != EAnimAlphaInputType::Float)
146 {
147 InDetailBuilder.HideProperty(InNodeHandle->GetChildHandle(GET_MEMBER_NAME_CHECKED(AnimNodeType, Alpha)));
148 InDetailBuilder.HideProperty(InNodeHandle->GetChildHandle(GET_MEMBER_NAME_CHECKED(AnimNodeType, AlphaScaleBias)));
149 }
150
151 if (InAnimNode.AlphaInputType != EAnimAlphaInputType::Curve)
152 {
153 InDetailBuilder.HideProperty(InNodeHandle->GetChildHandle(GET_MEMBER_NAME_CHECKED(AnimNodeType, AlphaCurveName)));
154 }
155
156 if ((InAnimNode.AlphaInputType != EAnimAlphaInputType::Float)
157 && (InAnimNode.AlphaInputType != EAnimAlphaInputType::Curve))
158 {
159 InDetailBuilder.HideProperty(InNodeHandle->GetChildHandle(GET_MEMBER_NAME_CHECKED(AnimNodeType, AlphaScaleBiasClamp)));
160 }
161 }
162};
