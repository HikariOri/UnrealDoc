29template<typename InEventType, typename SettingsType>
30class EnumerateAsyncAlgoritm
31{
32public:
33 struct FEnumerationStackEntry
34 {
35 double StartTime;
36 InEventType Event;
37 };
38
39 typedef TFunctionRef<void(int32 /*InitialStackDepth*/)> InitialStackCallback;
40
41 static void EnumerateEventsDownSampled(FAsyncEnumerateTaskData<InEventType, SettingsType>& Data, InitialStackCallback InitCallback, typename ITimeline<InEventType>::EventCallback Callback)
42 {
43 const FDetailLevel<InEventType, SettingsType>& DetailLevel = *Data.DetailLevel;
44
45 auto ScopeEntryIterator = DetailLevel.ScopeEntries.GetIteratorFromPage(Data.StartPageIndex);
46 const FEventScopeEntryPage* ScopePage = ScopeEntryIterator.GetCurrentPage();
47 const FEventScopeEntryPage* InitialPage = ScopePage;
48
49 auto EventsIterator = DetailLevel.Events.GetIteratorFromItem(ScopePage->BeginEventIndex);
50 FEnumerationStackEntry EventStack[SettingsType::MaxDepth];
51 int32 CurrentStackDepth = ScopePage->InitialStackCount;
52
53 // Init the enumeration EventStack using the start page initial stack.
54 for (int32 InitialStackIndex = 0; InitialStackIndex < CurrentStackDepth; ++InitialStackIndex)
55 {
56 FEnumerationStackEntry& EnumerationStackEntry = EventStack[InitialStackIndex];
57 const FEventStackEntry& EventStackEntry = ScopePage->InitialStack[InitialStackIndex];
58 EnumerationStackEntry.StartTime = DetailLevel.GetScopeEntryTime(EventStackEntry.EnterScopeIndex);
59 EnumerationStackEntry.Event = DetailLevel.GetEvent(EventStackEntry.EventIndex);
60 }
61
62 // Check if we must also send the initial stack events to the caller.
63 if ((Data.SortOrder == EEventSortOrder::ByEndTime) ||
64 (Data.TaskIndex == 0 && Data.SortOrder == EEventSortOrder::ByStartTime))
65 {
66 InitCallback(0);
67
68 for (int32 StackIndex = 0; StackIndex < CurrentStackDepth; ++StackIndex)
69 {
70 FEnumerationStackEntry& StackEntry = EventStack[StackIndex];
71 if (Callback(true, StackEntry.StartTime, StackEntry.Event) == EEventEnumerate::Stop)
72 {
73 return;
74 }
75 }
76 }
77 else
78 {
79 InitCallback(CurrentStackDepth);
80 }
81
82 const FEventScopeEntry* ScopeEntry = ScopeEntryIterator.GetCurrentItem();
83 const InEventType* Event = EventsIterator.GetCurrentItem();
84 uint64 CurrentPageIndex = Data.StartPageIndex;
85
86 // Iterate the pages owned by this task.
87 while (ScopeEntry && CurrentPageIndex < Data.EndPageIndex)
88 {
89 if (ScopeEntry->Time < 0.0) // "begin" event
90 {
91 check(CurrentStackDepth < SettingsType::MaxDepth);
92 FEnumerationStackEntry& StackEntry = EventStack[CurrentStackDepth++];
93 StackEntry.Event = *Event;
94 StackEntry.StartTime = -ScopeEntry->Time;
95
96 if (Callback(true, -ScopeEntry->Time, StackEntry.Event) == EEventEnumerate::Stop)
97 {
98 return;
99 }
100
101 Event = EventsIterator.NextItem();
102 }
103 else // "end" event
104 {
105 check(CurrentStackDepth > 0);
106 --CurrentStackDepth;
107
108 FEnumerationStackEntry& StackEntry = EventStack[CurrentStackDepth];
109 if (Callback(false, ScopeEntry->Time, StackEntry.Event) == EEventEnumerate::Stop)
110 {
111 return;
112 }
113
114 }
115 ScopeEntry = ScopeEntryIterator.NextItem();
116 CurrentPageIndex = ScopeEntryIterator.GetCurrentPageIndex();
117 }
118
119 // Check if we must find the end times for the events remaining on the enumeration stack after we have iterated
120 // all pages owned by this task.
121 if (Data.TaskIndex != Data.NumTasks - 1 && Data.SortOrder == EEventSortOrder::ByEndTime)
122 {
123 return;
124 }
125
126 if (ShouldStopSearchingForEndEvents(Data, EventStack, InitialPage, CurrentStackDepth, DetailLevel))
127 {
128 return;
129 }
130
131 if (ScopeEntry)
132 {
133 do
134 {
135 const FEventScopeEntryPage* CurrentScopePage = ScopeEntryIterator.GetCurrentPage();
136 check(CurrentStackDepth <= CurrentScopePage->InitialStackCount);
137
138 while (CurrentStackDepth > 0 && CurrentScopePage->InitialStack[CurrentStackDepth - 1].EndTime > 0)
139 {
140 --CurrentStackDepth;
141 InEventType CurrentEvent = DetailLevel.GetEvent(CurrentScopePage->InitialStack[CurrentStackDepth].EventIndex);
142 if (Callback(false, CurrentScopePage->InitialStack[CurrentStackDepth].EndTime, CurrentEvent) == EEventEnumerate::Stop)
143 {
144 return;
145 }
146
147 if (ShouldStopSearchingForEndEvents(Data, EventStack, InitialPage, CurrentStackDepth, DetailLevel))
148 {
149 return;
150 }
151 }
152 } while (ScopeEntryIterator.NextPage());
153 }
154
155 // If there are still events on the stack at this point, it means analysis has not yet reached their end,
156 // so we truncate them to the time of last received event.
157 while (CurrentStackDepth > 0)
158 {
159 FEnumerationStackEntry& StackEntry = EventStack[--CurrentStackDepth];
160 if (Callback(false, DetailLevel.InsertionState.LastTime, StackEntry.Event) == EEventEnumerate::Stop)
161 {
162 return;
163 }
164
165 if (ShouldStopSearchingForEndEvents(Data, EventStack, InitialPage, CurrentStackDepth, DetailLevel))
166 {
167 return;
168 }
169 }
170 }
171
172private:
173 static bool ShouldStopSearchingForEndEvents(FAsyncEnumerateTaskData<InEventType, SettingsType>& Data, FEnumerationStackEntry* EventStack, const FEventScopeEntryPage* InitialPage, int32 CurrentStackDepth, const FDetailLevel<InEventType, SettingsType>& DetailLevel)
174 {
175 if (Data.SortOrder == EEventSortOrder::ByStartTime &&
176 Data.TaskIndex > 0 &&
177 CurrentStackDepth > 0 &&
178 CurrentStackDepth <= InitialPage->InitialStackCount)
179 {
180 double InitialStartTime = DetailLevel.GetScopeEntryTime(InitialPage->InitialStack[CurrentStackDepth - 1].EnterScopeIndex);
181
182 if (EventStack[CurrentStackDepth - 1].StartTime == InitialStartTime)
183 {
184 return true;
185 }
186 }
187
188 return false;
189 }

206 void EnumerateOrderedByStartTime()
207 {
208 struct FStackEntry
209 {
210 uint64 OutputEventIndex;
211 };
212
213 FStackEntry EventStack[SettingsType::MaxDepth];
214 uint32 CurrentDepth = 0;
215 uint32 MinimumDepth = 0;
216
217 struct FOutputEvent
218 {
219 double StartTime;
220 double EndTime;
221 uint32 Depth;
222 EventType Event;
223 };
224
225 TArray<FOutputEvent> OutputEvents;
226
227 EnumerateAsyncAlgoritm<InEventType, SettingsType>::EnumerateEventsDownSampled(Data,
228 [&CurrentDepth, &MinimumDepth]
229 (int32 InInitialDepth)
230 {
231 MinimumDepth = InInitialDepth;
232 CurrentDepth = InInitialDepth;
233 },
234 [this, &EventStack, &OutputEvents, &CurrentDepth, &MinimumDepth]
235 (bool IsEnter, double Time, const EventType& Event)
236 {
237 if (IsEnter)
238 {
239 FStackEntry& StackEntry = EventStack[CurrentDepth];
240 StackEntry.OutputEventIndex = OutputEvents.Num();
241 FOutputEvent& OutputEvent = OutputEvents.AddDefaulted_GetRef();
242 OutputEvent.StartTime = Time;
243 OutputEvent.EndTime = Time;
244 OutputEvent.Depth = CurrentDepth;
245 OutputEvent.Event = Event;
246 ++CurrentDepth;
247 }
248 else
249 {
250 {
251 FStackEntry& StackEntry = EventStack[--CurrentDepth];
252 if (CurrentDepth >= MinimumDepth)
253 {
254 FOutputEvent* OutputEvent = OutputEvents.GetData() + StackEntry.OutputEventIndex;
255 OutputEvent->EndTime = Time;
256 }
257 }
258 if (CurrentDepth <= MinimumDepth)
259 {
260 MinimumDepth = CurrentDepth;
261 for (FOutputEvent& OutputEvent : OutputEvents)
262 {
263 if (OutputEvent.EndTime >= Data.StartTime && OutputEvent.StartTime <= Data.EndTime)
264 {
265 if (Data.EventRangeCallback(OutputEvent.StartTime, OutputEvent.EndTime, OutputEvent.Depth, OutputEvent.Event, Data.TaskIndex) == EEventEnumerate::Stop)
266 {
267 return EEventEnumerate::Stop;
268 }
269 }
270 }
271 OutputEvents.Empty(OutputEvents.Num());
272 }
273 }
274
275 return EEventEnumerate::Continue;
276 });
277
278 check(OutputEvents.Num() == 0);
279 }
280
281 void EnumerateOrderedByEndTime()
282 {
283 struct FStackEntry
284 {
285 double StartTime;
286 };
287
288 FStackEntry EventStack[SettingsType::MaxDepth];
289 uint32 CurrentDepth = 0;
290
291 EnumerateAsyncAlgoritm<InEventType, SettingsType>::EnumerateEventsDownSampled(Data,
292 [&CurrentDepth]
293 (int32 InInitialDepth)
294 {
295 CurrentDepth = InInitialDepth;
296 },
297 [this, &EventStack, &CurrentDepth]
298 (bool IsEnter, double Time, const EventType& Event)
299 {
300 if (IsEnter)
301 {
302 FStackEntry& StackEntry = EventStack[CurrentDepth];
303 StackEntry.StartTime = Time;
304 ++CurrentDepth;
305 }
306 else
307 {
308 FStackEntry& StackEntry = EventStack[--CurrentDepth];
309
310 if (Time >= Data.StartTime && StackEntry.StartTime <= Data.EndTime)
311 {
312 if (Data.EventRangeCallback(StackEntry.StartTime, Time, CurrentDepth, Event, Data.TaskIndex) == EEventEnumerate::Stop)
313 {
314 return EEventEnumerate::Stop;
315 }
316 }
317 }
318
319 return EEventEnumerate::Continue;
320 });
321 }

323 void DoWork()
324 {
325 if (Data.EventRangeCallback != nullptr)
326 {
327 if (Data.SortOrder == EEventSortOrder::ByEndTime)
328 {
329 EnumerateOrderedByEndTime();
330 }
331 else if (Data.SortOrder == EEventSortOrder::ByStartTime)
332 {
333 EnumerateOrderedByStartTime();
334 }
335 }
336 else
337 {
338 EnumerateAsyncAlgoritm<InEventType, SettingsType>::EnumerateEventsDownSampled(Data,
339 [](int32 InInitialDepth)
340 {
341 },
342 [this](bool IsEnter, double Time, const EventType& Event)
343 {
344 return this->Data.EventCallback(IsEnter, Time, Event, Data.TaskIndex);
345 });
346 }
