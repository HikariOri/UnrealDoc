75 friend bool operator==(const FOutlinerSizing& A, const FOutlinerSizing& B)
76 {
77 return
78 A.Height == B.Height &&
79 A.PaddingTop == B.PaddingTop &&
80 A.PaddingBottom == B.PaddingBottom &&
81 A.Flags == B.Flags;
82 }
83 friend bool operator!=(const FOutlinerSizing& A, const FOutlinerSizing& B)
84 {
85 return !(A == B);
86 }

88 float GetSeparatorHeight() const
89 {
90 return EnumHasAnyFlags(Flags, EOutlinerSizingFlags::IncludeSeparator)
91 ? 1.f
92 : 0.f;
93 }
94
95 float GetTotalHeight() const
96 {
97 return Height + PaddingTop + PaddingBottom + GetSeparatorHeight();
98 }
99
100 void Accumulate(const FOutlinerSizing& Other)
101 {
102 const float ThisSeparatorHeight = GetSeparatorHeight();
103 const float OtherSeparatorHeight = Other.GetSeparatorHeight();
104
105 Height = FMath::Max(Height + ThisSeparatorHeight, Other.Height + OtherSeparatorHeight);
106 PaddingTop = FMath::Max(PaddingTop, Other.PaddingTop);
107 PaddingBottom = FMath::Max(PaddingBottom, Other.PaddingBottom);
108 Flags |= Other.Flags;
109
110 Height -= GetSeparatorHeight();
111 }
112
113 float Height = 0.f;
114 float PaddingTop = 0.f;
115 float PaddingBottom = 0.f;
116 EOutlinerSizingFlags Flags = EOutlinerSizingFlags::IncludeSeparator;
