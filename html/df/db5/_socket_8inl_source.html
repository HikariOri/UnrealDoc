162 struct FPollFdBase
163 {
164 SocketType fd;
165 int16 events;
166 int16 revents;
167 static const int16 POLLIN = 1 << 0;
168 static const int16 POLLOUT = 1 << 1;
169 static const int16 POLLERR = 1 << 2;
170 static const int16 POLLHUP = POLLERR;
171 static const int16 POLLNVAL= POLLERR;
172 };
173#endif
174
176struct FPollFd
177 : public FPollFdBase
178{
179 static const int16 PollIn = int16(POLLIN);
180 static const int16 PollOut = int16(POLLOUT);
181 static const int16 PollErr = int16(POLLERR);
182 static const int16 PollHup = int16(POLLHUP);
183 static const int16 PollNVal= int16(POLLNVAL);
184 static const int16 PollAll = PollIn|PollOut|PollErr|PollHup|PollNVal;
185
186 FPollFd(SocketType Socket, int16 Events)
187 {
188 fd = FdType(Socket);
189 events = EventType(Events);
190 revents = EventType(0);
191 }
192
193 bool HasTrigger() const { return !!revents; }
194 bool HasSendTrigger() const { return !!(revents & EventType(PollOut)); }
195 bool HasRecvTrigger() const { return !!(revents & EventType(PollIn)); }
196
197private:
198 // The following is odd because POLLFD varies subtly from one platform to
199 // the next. With typedefs we can cleanly set members and not get narrowing
200 // warnings from the compiler.
201 using FdType = decltype(fd);
202 using EventType = decltype(events);
203};

223int32 FPoller::Wait(FPollFd* Polls, int32 Num, int32 TimeOutMs)
224{
225#if defined(IAS_HTTP_USE_POLL)
226 static_assert(sizeof(FPollFd) == sizeof(pollfd));
227 return Poll_Wait(Handle, Polls, Num, TimeOutMs);
228#else
229 timeval TimeVal = {};
230 timeval* TimeValPtr = (TimeOutMs >= 0 ) ? &TimeVal : nullptr;
231 if (TimeOutMs > 0)
232 {
233 int32 TimeoutUs = TimeOutMs << 10;
234 TimeVal = { TimeOutMs >> 10, TimeoutUs & ((1 << 20) - 1) };
235 }
236
237 fd_set FdSetRead; FD_ZERO(&FdSetRead);
238 fd_set FdSetWrite; FD_ZERO(&FdSetWrite);
239 fd_set FdSetExcept; FD_ZERO(&FdSetExcept);
240
241 SocketType MaxFd = 0;
242 for (FPollFd& Poll : MakeArrayView(Polls, Num))
243 {
244 fd_set* RwSet = (Poll.events & FPollFd::PollIn) ? &FdSetRead : &FdSetWrite;
245 FD_SET(Poll.fd, RwSet);
246 FD_SET(Poll.fd, &FdSetExcept);
247 MaxFd = FMath::Max(Poll.fd, MaxFd);
248 }
249
250 int32 Result = Socket_Select(int32(MaxFd + 1), &FdSetRead, &FdSetWrite, &FdSetExcept, TimeValPtr);
251 if (Result <= 0)
252 {
253 return Result;
254 }
255
256 for (FPollFd& Poll : MakeArrayView(Polls, Num))
257 {
258 if (FD_ISSET(Poll.fd, &FdSetExcept))
259 {
260 Poll.revents = FPollFd::PollErr;
261 continue;
262 }
263
264 fd_set* RwSet = (Poll.events & FPollFd::PollIn) ? &FdSetRead : &FdSetWrite;
265 if (FD_ISSET(Poll.fd, RwSet))
266 {
267 Poll.revents = Poll.events;
268 }
269 }
270
271 return Result;
272#endif // IAS_HTTP_USE_POLL
273}

321int32 FWaiter::Wait(TArrayView<FWaiter> Waiters, FPoller& Poller, int32 TimeoutMs)
322{
323 uint16 Events[] = {
324 FPollFd::PollErr,
325 FPollFd::PollOut,
326 FPollFd::PollIn,
327 FPollFd::PollIn | FPollFd::PollOut,
328 };
329 TArray<FPollFd, TFixedAllocator<64>> Polls;
330 for (FWaiter& Waiter : Waiters)
331 {
332 Polls.Emplace(Waiter.Socket, Events[uint32(Waiter.WaitOn)]);
333 }
334
335 // Poll the sockets
336 int32 Result = Poller.Wait(Polls.GetData(), Polls.Num(), TimeoutMs);
337 if (Result <= 0)
338 {
339 return Result;
340 }
341
342 // Transfer poll results to the input sockets. We don't transfer across error
343 // states. Subsequent sockets ops can take care of that instead.
344 for (uint32 i = 0, n = Waiters.Num(); i < n; ++i)
345 {
346 const FPollFd& Poll = Polls[i];
347 if (!Poll.HasTrigger())
348 {
349 continue;
350 }
351
352 uint8 Value = 0;
353 if (Poll.HasSendTrigger()) Value |= uint8(EWhat::Send);
354 if (Poll.HasRecvTrigger()) Value |= uint8(EWhat::Recv);
355 Waiters[i].Ready = Value ? Value : uint8(EWhat::Both);
356 }
357
358 return Result;
359}

442FOutcome FSocket::Connect(uint32 IpAddress, uint32 Port)
443{
444 check(IsValid());
445
446 Trace(Socket, ETrace::Connect);
447 auto Return = [this] (FOutcome Outcome) {
448 Trace(Socket, ETrace::Connect, &Outcome);
449 return Outcome;
450 };
451
452 IpAddress = Socket_HtoNl(IpAddress);
453
454 sockaddr_in AddrInet = { sizeof(sockaddr_in) };
455 AddrInet.sin_family = AF_INET;
456 AddrInet.sin_port = Socket_HtoNs(uint16(Port));
457 memcpy(&(AddrInet.sin_addr), &IpAddress, sizeof(IpAddress));
458
459 int32 Result = Socket_Connect(Socket, &(sockaddr&)AddrInet, uint32(sizeof(AddrInet)));
460
461 if (IsSocketResult(EWOULDBLOCK) | IsSocketResult(EINPROGRESS))
462 {
463 return Return(FOutcome::Waiting());
464 }
465
466 if (Result < 0)
467 {
468 return Return(FOutcome::Error("Socket connect failed", Result));

482FOutcome FSocket::Send(const char* Data, uint32 Size)
483{
484 Trace(Socket, ETrace::Send);
485 int32 Result = Socket_Send(Socket, Data, Size, MsgFlagType(MSG_NOSIGNAL));
486
487 auto Return = [this] (FOutcome Outcome) {
488 Trace(Socket, ETrace::Send, &Outcome);
489 return Outcome;
490 };
491
492 if (Result > 0) return Return(FOutcome::Ok(Result));
493 if (Result == 0) return Return(FOutcome::Error("Send ATH0"));
494 if (IsSocketResult(EWOULDBLOCK)) return Return(FOutcome::Waiting());
495
496 if (IsSocketResult(ENOTCONN))
497 {
498 int32 Error = 0;
499 socklen_t ErrorSize = sizeof(Error);
500 Result = Socket_GetSockOpt(Socket, SOL_SOCKET, SO_ERROR, (char*)&Error, &ErrorSize);
501 if (Result < 0 || Error != 0)
502 {
503 Result = (Result < 0) ? LastSocketResult() : Error;
504 return Return(FOutcome::Error("Error while connecting", Result));
505 }
506
507 return Return(FOutcome::Waiting());
508 }
509
510 Result = LastSocketResult();
511 return Return(FOutcome::Error("Send", Result));
512}

539bool FSocket::SetBlocking(bool bBlocking)
540{
541#if defined(IAS_HTTP_HAS_NONBLOCK_IMPL)
542 return SetNonBlockingSocket(Socket, !bBlocking);
543#elif PLATFORM_MICROSOFT
544 unsigned long NonBlockingMode = (bBlocking != true);
545 return (ioctlsocket(Socket, FIONBIO, &NonBlockingMode) != SOCKET_ERROR);
546#else
547 int32 Flags = fcntl(Socket, F_GETFL, 0);

549 {
550 return false;
551 }
552
553 int32 NewFlags = bBlocking
554 ? (Flags & ~int32(O_NONBLOCK))

583static void Trace(UPTRINT Socket, ETrace Action, const FOutcome* Outcome)
584{
585 static uint32 SockScopeId = LaneTrace_NewScope("Iax/Socket");
586 static uint32 ConnScopeId = LaneTrace_NewScope("Iax/Connect");
587 static uint32 SendScopeId = LaneTrace_NewScope("Iax/Send");
588 static uint32 RecvScopeId = LaneTrace_NewScope("Iax/Recv");
589 static uint32 WaitScopeId = LaneTrace_NewScope("Iax/Wait");
590 static uint32 IdleScopeId = LaneTrace_NewScope("Iax/Idle");
591
592 if (Action == ETrace::SocketCreate)
593 {
594 FLaneTrace* Lane = LaneEstate_Build(GSocketLaneEstate, Socket);
595 LaneTrace_Enter(Lane, SockScopeId);
596 LaneTrace_Enter(Lane, IdleScopeId);
597 return;
598 }
599
600 if (Action == ETrace::SocketDestroy)
601 {
602 LaneEstate_Demolish(GSocketLaneEstate, Socket);
603 return;
604 }
605
606 FLaneTrace* Lane = LaneEstate_Lookup(GSocketLaneEstate, Socket);
607
608 if (Action == ETrace::Send)
609 {
610 (Outcome == nullptr)
611 ? LaneTrace_Enter(Lane, SendScopeId)
612 : LaneTrace_Leave(Lane);
613 return;
614 }
615
616 if (Action == ETrace::Connect)
617 {
618 if (Outcome == nullptr)
619 {
620 LaneTrace_Change(Lane, ConnScopeId);
621 }
622 else if (!Outcome->IsWaiting())
623 {
624 LaneTrace_Change(Lane, IdleScopeId);
625 }
626 return;
627 }
628
629 if (Outcome == nullptr)
630 {
631 LaneTrace_Change(Lane, RecvScopeId);
632 return;
633 }
634
635 if (Outcome->IsWaiting())
636 {
637 LaneTrace_Change(Lane, WaitScopeId);
638 return;
639 }
640
641 LaneTrace_Change(Lane, IdleScopeId);
642}
