63enum class ERayTracingPipelineCacheFlags : uint8
64{
65 // Query the pipeline cache, create pipeline if necessary.
66 // Compilation may happen on a task, but RHIThread will block on it before translating the RHICmdList.
67 // Therefore the RHIThread may stall when creating large / complex pipelines.
68 Default = 0,
69
70 // Query the pipeline cache, create a background task to create the pipeline if necessary.
71 // GetAndOrCreateRayTracingPipelineState() may return NULL if pipeline is not ready.
72 // Caller must use an alternative fallback PSO to render current frame and may retry next frame.
73 // Pipeline creation task will not block RenderThread or RHIThread, allowing hitch-free rendering.
74 NonBlocking = 1 << 0,
75};

91// Unique request ID of PSOPrecache which can be used to boost the priority of a PSO precache requests if it's needed for rendering
92struct FPSOPrecacheRequestID
93{
94 enum class EType
95 {
96 Invalid = 0,
97 Graphics,
98 Compute
99 };
100
101 FPSOPrecacheRequestID() : Type((uint32)EType::Invalid), RequestID(0) { }
102
103 EType GetType() const { return (EType)Type; }
104 bool IsValid() const
105 {
106 return Type != (uint32)EType::Invalid;
107 }
108 bool operator==(const FPSOPrecacheRequestID& Other) const
109 {
110 return Type == Other.Type && RequestID == Other.RequestID;
111 }
112 bool operator!=(const FPSOPrecacheRequestID& rhs) const
113 {
114 return !(*this == rhs);
115 }
116
117 uint32 Type : 2; //< PSO request type
118 uint32 RequestID : 30; //< Unique request ID
119};

138// Statistics about PSOs created at runtime, including stats for PSOs which took over a pre-defined threshold to be created.
139// These only track PSOs requested by the renderer on the critical path (not from background precaching).
140// Note that while slow PSO creation times usually result in a hitch at runtime, it's not a 1-to-1 correlation.
141// There are cases where multiple PSOs are slow in a single frame and therefore cause a single hitch, and cases
142// where a PSO might not be slow enough to cause a visible hitch for a certain frame (depending on the threshold).
143struct FPSORuntimeCreationStats
144{
145 // The total number of PSOs created after a request by the renderer. Does not include PSOs created from background precaching.
146 uint32 TotalPSOCreations = 0;
147
148 // The number of compute and graphics PSOs that took too long to create.
149 uint32 ComputePSOHitches = 0;
150 uint32 GraphicsPSOHitches = 0;
151
152 // How many of the PSOs that took too long to create were previously precached in the background.
153 uint32 PreviouslyPrecachedPSOHitches = 0;
154
155 // How many of the PSOs that too long to create were previously precached in the background and are suspected to have hit a full compilation again.
156 // These use a higher threshold and usually indicate the driver cache is being missed or is not working.
157 uint32 SuspectedUnhealthyDriverCachePSOHitches = 0;
158
159 // Whether the driver cache is suspected to be unhealthy due to the number of very long creation times for precached PSO (controlled by a configurable threshold).
160 bool bDriverCacheSuspectedUnhealthy = false;
161};
