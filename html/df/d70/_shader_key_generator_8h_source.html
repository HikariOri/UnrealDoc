164inline void FShaderKeyGenerator::Append(FStringView Value)
165{
166 switch (OutputType)
167 {
168 case EOutputType::Text:
169 ResultString->Append(Value);
170 return;
171 case EOutputType::Binary:
172 ResultFunc(Value.GetData(), Value.Len() * sizeof(Value.GetData()[0]));
173 return;
174 default:
175 checkNoEntry();
176 }
177}
