88 auto InRange = [&](const int32 Idx, const int32 Value)
89 {
90 const FInt16Range& Range = SwitchSizeRanges[Idx];
91 if (Value <= Range.GetLowerBoundValue())
92 {
93 return -1;
94 }
95

97 {
98 return 1;
99 }

102 };

104 // Starting from the existing widget, check each size range for the matching parent
105 int32 NumIterations = 0;
106 while (NumIterations < SwitchSizeRanges.Num())
107 {
108 const int32 InRangeResult = InRange(TargetIdx, FMath::FloorToInt(CurrentSize - UE_KINDA_SMALL_NUMBER));
109 if (InRangeResult <= 0)

112 if (TargetIdx == 0 || InRangeResult == 0)
113 {
114 return TargetIdx;

118 TargetIdx -= 1;
119 }
120 else
121 {

123 if (TargetIdx == SwitchSizeRanges.Num() - 1)
124 {
125 return TargetIdx;
126 }
127
128 // Otherwise offset TargetIdx
129 TargetIdx += 1;
130 }

132 ++NumIterations;
133 }
