79 virtual TSharedPtr<IToolTip> GetToolTip() override
80 {
81 ILauncherProfilePtr LaunchProfile = LaunchProfileAttr.Get();
82 if (LaunchProfile.IsValid() && LaunchProfile->HasValidationError())
83 {
84 ErrorToolTipWidget->SetContentWidget(

88 return ErrorToolTipWidget;
