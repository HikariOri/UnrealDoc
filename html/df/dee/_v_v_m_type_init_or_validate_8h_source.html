101 template <typename ValueType>
102 static FString FormatValue(ValueType Value)
103 {
104 if constexpr (std::is_enum<ValueType>::value)
105 {
106 using UEnumType = std::underlying_type<ValueType>::type;
107 UEnumType UnderlyingValue = (UEnumType)Value;
108 return LexToString(UnderlyingValue);
109 }
110 else if constexpr (std::is_same_v<FName, ValueType>)
111 {
112 return Value.ToString();
113 }
114 else if constexpr (std::is_same_v<FProperty, std::remove_pointer_t<ValueType>>)
115 {
116 return Value ? Value->GetFullName() : TEXT("null");
117 }
118 else if constexpr (std::is_base_of_v<UObject, std::remove_pointer_t<ValueType>>)
119 {
120 return Value ? Value->GetFullName() : TEXT("null");
121 }
122 else if constexpr (std::is_same_v<FGuid, ValueType>)
123 {
124 return LexToString(Value);
125 }
126 else if constexpr (std::is_same_v<FUtf8String, ValueType>)
127 {
128 return FString(Value);
129 }
130 else
131 {
132 return FString::Format(TEXT("{0}"), {Value}); // yeah, this is horrible
133 }
134 }
135
136 template <typename ObjectType>
137 static FString FormatValue(TObjectPtr<ObjectType> Value)
138 {
139 return FormatValue(Value.Get());
140 }

142 template <typename DestinationType, typename ValueType>
143 void LogValueMismatch(const DestinationType& Destination, const ValueType& Value, const TCHAR* What, const TCHAR* SubWhat = nullptr) const
144 {
145 if (SubWhat != nullptr)
146 {
147 LogError(FString::Format(TEXT("'{0}:{1}:{2}' doesn't have the correct value. Expected: '{3}' Got: '{4}'"),
148 {Field->GetName(), What, SubWhat, FormatValue(Value), FormatValue(Destination)}));
149 }
150 else
151 {
152 LogError(FString::Format(TEXT("'{0}:{1}' doesn't have the correct value. Expected: '{2}' Got: '{3}'"),
153 {Field->GetName(), What, FormatValue(Value), FormatValue(Destination)}));
154 }
155 }
156
157 virtual void LogError(const FString& text) const = 0;
