116 [[nodiscard]] bool OrderIndependentCompareEqual(const TMAPBASE& Other) const
117 {
118 // first check counts (they should be the same obviously)
119 if (Num() != Other.Num())
120 {
121 return false;
122 }
123
124 // since we know the counts are the same, we can just iterate one map and check for existence in the other
125 for (typename ElementSetType::TConstIterator It(Pairs); It; ++It)
126 {
127 const ValueType* BVal = Other.Find(It->Key);
128 if (BVal == nullptr)
129 {
130 return false;
131 }
132 if (!(*BVal == It->Value))
133 {
134 return false;
135 }
136 }
137
138 // all fields in A match B and A and B's counts are the same (so there can be no fields in B not in A)
139 return true;
140 }

242 template<typename Allocator> int32 GetKeys(TArray<KeyType, Allocator>& OutKeys) const
243 {
244 OutKeys.Reset();
245
246 TSet<KeyType> VisitedKeys;
247 VisitedKeys.Reserve(Num());
248
249 // Presize the array if we know there are supposed to be no duplicate keys
250 if constexpr (!KeyFuncs::bAllowDuplicateKeys)
251 {
252 OutKeys.Reserve(Num());
253 }
254
255 for (typename ElementSetType::TConstIterator It(Pairs); It; ++It)
256 {
257 // Even if bAllowDuplicateKeys is false, we still want to filter for duplicate
258 // keys due to maps with keys that can be invalidated (UObjects, TWeakObj, etc.)
259 if (!VisitedKeys.Contains(It->Key))
260 {
261 OutKeys.Add(It->Key);
262 VisitedKeys.Add(It->Key);
263 }
264 }
265
266 return OutKeys.Num();
267 }
