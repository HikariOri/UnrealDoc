46 Deletion = 1ULL << 1, // The property is able to be deleted.

50 DropDownMenuButton = 1ULL << 5, // Drop-down menu (down arrow button) is enabled.

57 Menu_TypeSelector = 1ULL << 17, // Adds the type selector pill widget to the drop-down.
58 Menu_Rename = 1ULL << 18, // Renaming the property from the drop-down menu. Requires property renaming enabled.
59 Menu_Delete = 1ULL << 19, // Deleting the property from the drop-down menu.
60 Menu_Categories = 1ULL << 20, // Create new/remove from categories.
61 Menu_MetadataSpecifiers = 1ULL << 21, // Common metadata specifier options.
62
63 AllMenuOptions = Menu_TypeSelector | Menu_Rename | Menu_Delete | Menu_Categories | Menu_MetadataSpecifiers,
64
65 // Insert new menu features above this line.
66 Deprecated = 1ULL << 63, // To allow for deprecating older features.
