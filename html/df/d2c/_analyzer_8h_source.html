391template <typename ValueType>
392IAnalyzer::FEventFieldHandle IAnalyzer::FEventTypeInfo::GetFieldHandle(const ANSICHAR* FieldName) const
393{
394 int16 SizeAndType;
395 FEventFieldHandle Handle = GetFieldHandleImpl(FieldName, SizeAndType);
396 if (std::is_floating_point<ValueType>::value)
397 {
398 checkf((SizeAndType < 0), TEXT("Field is not a float-type field"));
399 }
400 else
401 {
402 checkf(SizeAndType >= sizeof(ValueType), TEXT("Field is to small to read as hinted type ValueType"));
403 }
404 return Handle;
405}

408template <typename ValueType>
409ValueType IAnalyzer::CoerceValue(const void* Addr, int8 SizeAndType)
410{
411 using integral8 = std::conditional_t<std::is_signed_v<ValueType>, int8, uint8>;
412 using integral16 = std::conditional_t<std::is_signed_v<ValueType>, int16, uint16>;
413 using integral32 = std::conditional_t<std::is_signed_v<ValueType>, int32, uint32>;
414 using integral64 = std::conditional_t<std::is_signed_v<ValueType>, int64, uint64>;
415 switch (SizeAndType)
416 {
417 case -4: return ValueType(*(const float*)(Addr));
418 case -8: return ValueType(*(const double*)(Addr));
419 case 1: return ValueType(*(const integral8*)(Addr));
420 case 2: return ValueType(*(const integral16*)(Addr));
421 case 4: return ValueType(*(const integral32*)(Addr));
422 case 8: return ValueType(*(const integral64*)(Addr));
423 default: return ValueType(0);
424 }
425}
