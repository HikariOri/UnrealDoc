73uint32 GetNowMs()
74{
75 // 4.2MM seconds will give us 50 days of uptime.
76 static uint64 Freq = 0;
77 static uint64 Base = 0;
78 if (Freq == 0)
79 {
80 Freq = uint64(1.0 / FPlatformTime::GetSecondsPerCycle64());
81 Base = FPlatformTime::Cycles64();
82 }
83 uint64 NowBig = ((FPlatformTime::Cycles64() - Base) * 1000) / Freq;
84 uint32 NowMs = uint32(NowBig);
85 check(NowMs == NowBig);
86 return NowMs + 1;
87 // +1 bias ----^ ...so 0 can be considered "not set"
88}

186template <uint32 Base=10>
187static int64 CrudeToInt(FAnsiStringView View)
188{
189 static_assert(Base == 10 || Base == 16);
190
191 // FCStringAnsi::* is not used to mitigate any locale hiccups. By
192 // initialising 'Value' with MSB set we can detect cases where View did not
193 // start with digits. This works as we won't be using this on huge numbers.
194 int64 Value = 0x8000'0000'0000'0000ll;
195 for (int32 c : View)
196 {
197 uint32 Digit = c - '0';
198 if (Digit > 9u)
199 {
200 if constexpr (Base != 16)
201 {
202 break;
203 }
204 else
205 {
206 Digit = (c | 0x20) - 'a';
207 if (Digit > uint32('f' - 'a'))
208 {
209 break;
210 }
211 Digit += 10;
212 }
213 }
214 Value *= Base;
215 Value += Digit;
216 }
217 return Value;
218};

242static int32 ParseUrl(FAnsiStringView Url, FUrlOffsets& Out)
243{
244 if (Url.Len() < 5)
245 {
246 return -1;
247 }
248
249 Out = {};
250
251 const char* Start = Url.GetData();
252 const char* Cursor = Start;
253
254 // Scheme
255 int32 i = 0;
256 for (; i < 5; ++i)
257 {
258 if (uint32(Cursor[i] - 'a') > uint32('z' - 'a'))
259 {
260 break;
261 }
262 }
263
264 Out.SchemeLength = uint8(i);
265 FAnsiStringView Scheme = Url.Left(i);
266 if (Scheme != "http" && Scheme != "https")
267 {
268 return -1;
269 }
270
271 // Separator and authority
272 if (Cursor[i] != ':' || Cursor[i + 1] != '/' || Cursor[i + 2] != '/')
273 {
274 return -1;
275 }
276 i += 3;
277
278 struct { int32 c; int32 i; } Seps[2];
279 int32 SepCount = 0;
280 for (; i < Url.Len(); ++i)
281 {
282 int32 c = Cursor[i];
283 if (c < '-') break;
284 if (c != ':' && c != '@' && c != '/') continue;
285 if (c == '/' || SepCount >= 2) break;
286
287 if (c == '@' && SepCount)
288 {
289 SepCount -= (Seps[SepCount - 1].c == ':');
290 }
291 Seps[SepCount++] = { c, i };
292 }
293
294 if (i > 0xff || i <= Scheme.Len() + 3)
295 {
296 return -1;
297 }
298
299 if (i < Url.Len())
300 {
301 Out.Path = uint8(i);
302 }
303
304 Out.HostName = { uint8(Scheme.Len() + 3), uint8(i) };
305
306 switch (SepCount)
307 {
308 case 0:
309 break;
310
311 case 1:
312 if (Seps[0].c == ':')
313 {
314 Out.Port = { Seps[0].i + 1, i };
315 Out.HostName.Right = uint8(Seps[0].i);
316 }
317 else
318 {
319 Out.UserInfo = { Out.HostName.Left, Seps[0].i };
320 Out.HostName.Left += uint8(Seps[0].i + 1);
321 Out.HostName.Right += uint8(Seps[0].i + 1);
322 }
323 break;
324
325 case 2:
326 if ((Seps[0].c != '@') | (Seps[1].c != ':'))
327 {
328 return -1;
329 }
330 Out.UserInfo = { Out.HostName.Left, Seps[0].i };
331 Out.Port.Left = uint8(Seps[1].i + 1);
332 Out.Port.Right = Out.HostName.Right;
333 Out.HostName.Left = Out.UserInfo.Right + 1;
334 Out.HostName.Right = Out.Port.Left - 1;
335 break;
336
337 default:
338 return -1;
339 }
340
341 bool Bad = false;
342 Bad |= (Out.HostName.Len() == 0);
343 Bad |= bool(int32(Out.UserInfo) & (Out.UserInfo.Len() == 0));
344
345 if (Out.Port.Left)
346 {
347 Bad |= (Out.Port.Len() == 0);
348 for (int32 j = 0, n = Out.Port.Len(); j < n; ++j)
349 {
350 Bad |= (uint32(Start[Out.Port.Left + j] - '0') > 9);
351 }
352 }
353
354 return Bad ? -1 : 1;
355}

453template <typename T>
454T* FBuffer::Alloc(uint32 Count)
455{
456 uint32 AlignBias = uint32(Data + Used) & (alignof(T) - 1);
457 if (AlignBias)
458 {
459 AlignBias = alignof(T) - AlignBias;
460 }
461
462 uint32 PotentialUsed = Used + AlignBias + (sizeof(T) * Count);
463 if (PotentialUsed > Max)
464 {
465 Extend(PotentialUsed, 256);
466 }
467
468 void* Ret = GetDataPtr() + Used + AlignBias;
469 Used = PotentialUsed;
470 return (T*)Ret;
471}

474FBuffer::FMutableSection FBuffer::GetMutableFree(uint32 MinSize, uint32 PageSize)

477
478 uint32 PotentialUsed = Used + MinSize;
479 if (PotentialUsed > Max)
480 {
481 Extend(PotentialUsed, PageSize);
482 }
483
484 return FMutableSection{ GetDataPtr() + Used, Max - Used };

495void FBuffer::Extend(uint32 AtLeast, uint32 PageSize)
496{
497 checkSlow(((PageSize - 1) & PageSize) == 0);
498
499 --PageSize;
500 Max = (AtLeast + PageSize) & ~PageSize;
501
502 if (!Inline)
503 {
504 Data = UPTRINT(FMemory::Realloc(GetDataPtr(), Max, alignof(FBuffer)));
505 return;
506 }
507
508 const char* PrevData = GetDataPtr();
509 Data = UPTRINT(FMemory::Malloc(Max, alignof(FBuffer)));
510 ::memcpy(GetDataPtr(), PrevData, Used);
511 Inline = 0;
512}

522class FThrottler
523{
524public:
525 FThrottler();
526 void SetLimit(uint32 KiBPerSec);
527 int32 GetAllowance();
528 void ReturnUnused(int32 Unused);
529
530private:
531 friend void ThrottleTest(FAnsiStringView);
532 int32 GetAllowance(int64 CycleDelta);
533 int64 CycleFreq;
534 int64 CycleLast = 0;
535 int64 CyclePeriod = 0;
536 uint32 Limit = 0;
537
538 enum : uint32 {
539 LIMITLESS = MAX_int32,
540 SLICES_POW2 = 5,
541 };
542};
543
545FThrottler::FThrottler()
546{
547 CycleFreq = int64(1.0 / FPlatformTime::GetSecondsPerCycle64());
548 check(CycleFreq >> SLICES_POW2);
549}
550
552void FThrottler::SetLimit(uint32 KiBPerSec)
553{
554 // 512MiB/s might as well be limitless.
555 KiBPerSec = (KiBPerSec < (512 << 10)) ? KiBPerSec : 0;
556 if (KiBPerSec)
557 {
558 KiBPerSec = FMath::Max(KiBPerSec, 1u << SLICES_POW2);
559 }
560 Limit = KiBPerSec << 10;
561}
562
564int32 FThrottler::GetAllowance()
565{
566 int64 Cycle = FPlatformTime::Cycles64();
567 int64 CycleDelta = Cycle - CycleLast;
568 CycleLast = Cycle;
569 return GetAllowance(CycleDelta);
570}
571
573int32 FThrottler::GetAllowance(int64 CycleDelta)
574{
575 if (Limit == 0)
576 {
577 return LIMITLESS;
578 }
579
580 int64 CycleSlice = CycleFreq >> SLICES_POW2;
581 CycleDelta = FMath::Min(CycleDelta, CycleSlice);
582 CyclePeriod -= CycleDelta;
583 if (CyclePeriod > 0)
584 {
585 return 0 - int32((CyclePeriod * 1000ll) / CycleFreq);
586 }
587 CyclePeriod += CycleSlice;
588
589 int32 Released = Limit >> SLICES_POW2;
590 return Released;
591}
592
594void FThrottler::ReturnUnused(int32 Unused)
595{
596 if (Limit == 0 || Unused == 0)
597 {
598 return;
599 }
600
601 int64 CycleReturn = (CycleFreq * Unused) / Limit;
602 CycleLast -= CycleReturn;
603}
