170 static FORCEINLINE void* MapBufferRange(GLenum Type, uint32 InOffset, uint32 InSize, EResourceLockMode LockMode)
171 {
172 GLenum Access;
173 switch (LockMode)
174 {
175 case EResourceLockMode::RLM_ReadOnly:
176 Access = GL_MAP_READ_BIT;
177 break;
178 case EResourceLockMode::RLM_ReadOnlyPersistent:
179 Access = (GL_MAP_READ_BIT | GL_MAP_PERSISTENT_BIT | GL_MAP_COHERENT_BIT);
180 break;
181 case EResourceLockMode::RLM_WriteOnly:
182 Access = OpenGLConsoleVariables::bUseUnsynchronizedBufferMapping ? (GL_MAP_INVALIDATE_BUFFER_BIT | GL_MAP_WRITE_BIT | GL_MAP_UNSYNCHRONIZED_BIT) : (GL_MAP_INVALIDATE_BUFFER_BIT | GL_MAP_WRITE_BIT);
183 break;
184 case EResourceLockMode::RLM_WriteOnlyUnsynchronized:
185 Access = (GL_MAP_WRITE_BIT | GL_MAP_UNSYNCHRONIZED_BIT);
186 break;
187 case EResourceLockMode::RLM_WriteOnlyPersistent:
188 Access = (GL_MAP_WRITE_BIT | GL_MAP_PERSISTENT_BIT | GL_MAP_COHERENT_BIT);
189 break;
190 case EResourceLockMode::RLM_ReadWrite:
191 default:
192 Access = (GL_MAP_READ_BIT | GL_MAP_WRITE_BIT);
193 }
194 return glMapBufferRange(Type, InOffset, InSize, Access);
195 }
