150 template<typename T, typename = typename TEnableIf<TIsDerivedFrom<T, USubsystem>::IsDerived>::Type>
151 static T* FetchSubsystemInstance(UWorld* World)
152 {
153 check(World);
154 if constexpr (TIsDerivedFrom<T, UWorldSubsystem>::IsDerived)
155 {
156 return UWorld::GetSubsystem<T>(World);
157 }
158 else if constexpr (TIsDerivedFrom<T, UGameInstanceSubsystem>::IsDerived)
159 {
160 return UGameInstance::GetSubsystem<T>(World->GetGameInstance());
161 }
162 else if constexpr (TIsDerivedFrom<T, ULocalPlayerSubsystem>::IsDerived)
163 {
164 // note that this default implementation will work only for the first player in a local-coop game
165 // to customize this behavior specialize the FetchSubsystemInstance template function for the type you need.
166 return ULocalPlayer::GetSubsystem<T>(World->GetFirstLocalPlayerFromController());
167 }
168 else
169 {
170 checkf(false, TEXT("FMassSubsystemAccess::FetchSubsystemInstance: Unhandled world-related USubsystem class %s"), *T::StaticClass()->GetName());
171 }
172 }

196 template<typename T>
197 T* GetSubsystemInternal(const int32 SystemIndex)
198 {
199 checkSlow(SystemIndex != INDEX_NONE);
200 if (UNLIKELY(Subsystems.IsValidIndex(SystemIndex) == false))
201 {
202 Subsystems.AddZeroed(SystemIndex - Subsystems.Num() + 1);
203 }
204
205 T* SystemInstance = (T*)Subsystems[SystemIndex];
206 if (SystemInstance == nullptr)
207 {
208 if constexpr (DoesRequireWorld<T>())
209 {
210 SystemInstance = FetchSubsystemInstance<std::remove_const_t<T>>(GetWorld());
211 }
212 else
213 {
214 SystemInstance = FetchSubsystemInstance<std::remove_const_t<T>>();
215 }
216 Subsystems[SystemIndex] = SystemInstance;
217 }
218 return SystemInstance;
219 }
220
221 template<typename T>
222 T* GetSubsystemInternal(const int32 SystemIndex, const TSubclassOf<USubsystem> SubsystemClass)
223 {
224 checkSlow(SystemIndex != INDEX_NONE);
225 if (UNLIKELY(Subsystems.IsValidIndex(SystemIndex) == false))
226 {
227 Subsystems.AddZeroed(SystemIndex - Subsystems.Num() + 1);
228 }
229
230 USubsystem* SystemInstance = (T*)Subsystems[SystemIndex];
231 if (SystemInstance == nullptr)
232 {
233 SystemInstance = FetchSubsystemInstance(GetWorld(), SubsystemClass);
234 Subsystems[SystemIndex] = SystemInstance;
235 }
236 return Cast<T>(SystemInstance);
237 }
