186 [[nodiscard]] constexpr std::conditional_t<std::is_const_v<StorageElementType>, const InElementType, InElementType>& operator*() const
187 {
188 if constexpr (bReverse)
189 {
190 return *(Ptr - 1);
191 }
192 else
193 {
194 return *Ptr;

196 }
