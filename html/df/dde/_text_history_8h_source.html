113template <typename T>
114const TCHAR* ReadScopedEnumFromBuffer(const TCHAR* Buffer, const FString& Scope, T& OutValue)
115{
116 if (PeekInsensitiveMarker(Buffer, *Scope, Scope.Len()))
117 {
118 // Parsing something of the form: EEnumName::...
119 Buffer += Scope.Len();
120
121 FString EnumValueString;
122 Buffer = ReadAlnumFromBuffer(Buffer, EnumValueString);
123
124 if (Buffer && LexTryParseString(OutValue, *EnumValueString))
125 {
126 return Buffer;
127 }
128 }
129
130 return nullptr;
131}

137 Buffer += LexToString(Value);
