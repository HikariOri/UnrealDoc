15inline void InterpolateNormalsAndUVs(
16 const FVector3f& Barycentrics,
17 const FLerpVert& Vert0,
18 const FLerpVert& Vert1,
19 const FLerpVert& Vert2,
20 FVector2f& OutUV,
21 FVector3f& OutNormal)
22{
23 OutUV = Vert0.UVs[0] * Barycentrics.X;
24 OutUV += Vert1.UVs[0] * Barycentrics.Y;
25 OutUV += Vert2.UVs[0] * Barycentrics.Z;
26
27 OutNormal = Vert0.TangentZ * Barycentrics.X;
28 OutNormal += Vert1.TangentZ * Barycentrics.Y;
29 OutNormal += Vert2.TangentZ * Barycentrics.Z;
30 OutNormal.Normalize();
31}

63template <typename T>
64inline int32 GetNumSamples(
65 const UE::Math::TVector<T> Barycentrics[3],
66 const FVector2f VertexUVs[3],
67 const FVector2f MapRes)
68{
69 FVector2f UVs[3];
70 for (int k = 0; k < 3; k++)
71 {
72 UVs[k] = VertexUVs[0] * Barycentrics[k].X;
73 UVs[k] += VertexUVs[1] * Barycentrics[k].Y;
74 UVs[k] += VertexUVs[2] * Barycentrics[k].Z;
75
76 UVs[k].X *= MapRes[0];
77 UVs[k].Y *= MapRes[1];
78 }
79
80 FVector2f Edge01 = UVs[1] - UVs[0];
81 FVector2f Edge12 = UVs[2] - UVs[1];
82 FVector2f Edge20 = UVs[0] - UVs[2];
83
84 T MaxEdgeLength = FMath::Sqrt(FMath::Max3(
85 Edge01.SizeSquared(),
86 Edge12.SizeSquared(),
87 Edge20.SizeSquared()));
88
89 T AreaInTexels = FMath::Abs(0.5 * (Edge01 ^ Edge12));
90
91 return static_cast<int32>(FMath::CeilToInt(FMath::Max(MaxEdgeLength, AreaInTexels)));
92}
