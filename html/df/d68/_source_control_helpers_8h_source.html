63 void SetFromStatus(const FSourceControlStatePtr& InStatus)
64 {
65 if (InStatus)
66 {
67 bIsValid = true;
68
69 // Copy over state info
70 // - make these assignments a method of FSourceControlState if anything else sets a state
71 bIsUnknown = InStatus->IsUnknown();
72 bIsSourceControlled = InStatus->IsSourceControlled();
73 bCanCheckIn = InStatus->CanCheckIn();
74 bCanCheckOut = InStatus->CanCheckout();
75 bIsCheckedOut = InStatus->IsCheckedOut();
76 bIsCurrent = InStatus->IsCurrent();

81 bCanDelete = InStatus->CanDelete();
82 bCanAdd = InStatus->CanAdd();
83 bIsConflicted = InStatus->IsConflicted();
84 bCanRevert = InStatus->CanRevert();
85 bIsModified = InStatus->IsModified();
86 bIsCheckedOutOther = InStatus->IsCheckedOutOther();
87 bIsCheckedOutInOtherBranch = InStatus->IsCheckedOutInOtherBranch();
88 bIsModifiedInOtherBranch = InStatus->IsModifiedInOtherBranch();
89 Filename = InStatus->GetFilename();
90
91 if (bIsCheckedOutOther)
92 {
93 InStatus->IsCheckedOutOther(&CheckedOutOther);
94 }
95
96 //Go through the history to find the most recent user
97 const int32 HistorySize = InStatus->GetHistorySize();
98 for ( int32 Index = 0; Index < HistorySize; Index++ )
99 {
100 if ( const ISourceControlRevision* Revision = InStatus->GetHistoryItem( Index ).Get() )
101 {
102 const FString User = Revision->GetUserName();
103 if ( !User.IsEmpty() )
104 {
105 PreviousUser = User;
106 break;
107 }
108 }
109 }
110 }
111 else
112 {
113 bIsValid = false;
114 }
115 }
