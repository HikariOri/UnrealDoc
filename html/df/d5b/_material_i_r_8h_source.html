97// Base entity of all IR graph nodes.
98//
99// An IR module is a graph of values, connected by their "uses" relations. The graph of IR
100// values is built by the `MaterialIRModuleBuilder` as the result of crawling through and
101// analyizing the MaterialExpression graph contained in the translated Material. During
102// this processing, IR values are emitted, and linked together. After the graph is
103// constructed, it is itself analyzed: the builder will call `FMaterialIRValueAnalyzer::Analyze()`
104// in each active* (i.e. truly used) value in the graph, making sure a value is analyzed only
105// after its dependencies have been analyzed.
106// A few notes:
107// - FValues are automatically zero-initialized.
108// - FValues are intended to be simple and inert data records. They cannot have non-trivia
109// ctor, dtor or copy operators.
110// - The ModuleBuilder relies on this property to efficiently hashing values so that it
111// will reuse the same value instance instead of creating multiple instances of the
112// same computation (for a more efficient output shader).
113// - All values have a MIR type.
114// - Pure FValue instances are values that do not have other dependencies (called "uses").
115// - If a value has some other value as dependency, it means that it is the result of a
116// calculation on those values. Values that have dependencies are Instructions (they
117// derive from FInstruction).
118struct FValue
119{
120 // The runtime type this value has.
121 FType Type;
122
123 // Used to discern the concrete C++ type of this value (e.g. Subscript)
124 TEnumAsByte<EValueKind> Kind;
125
126 // Set of fundamental flags true for this value.
127 EValueFlags Flags;
128
129 // The set of properties that are true for this value. Some flags might have been
130 // set but some upstream dependency leading to this value and not this value directly.
131 // See `EGraphProperties` for more information.
132 EGraphProperties GraphProperties;
133
134 // Offset where this value will be written in the Preshader buffer plus one, with zero
135 // indicating the value is not a Preshader. During value analyze, before PreshaderFixup has
136 // been called, stores a unique "register" index. PreshaderFixup translates these registers
137 // to offsets once locations in the output buffer have all been computed, taking into
138 // account lifetimes and aliasing of the registers.
139 int32 Analysis_PreshaderOffset;
140
141 // Returns whether this value has been analyzed for specified stage.
142 bool IsAnalyzed(EStage State) const;
143
144 // Returns whether specified flags are true for this value.
145 bool HasFlags(EValueFlags InFlags) const;

293// A constant value.
294//
295// A constant represents a translation-time known scalar primitive value. Operations on
296// constant values can be folded by the builder, that is they can be evaluated statically
297// while the builder constructs the IR graph of an input material.
298struct FConstant : TValue<VK_Constant>
299{
300 union
301 {
302 bool Boolean;
303 TInteger Integer;
304 TFloat Float;
305 TDouble Double;
306 };
307
308 // Returns the constant value of given type T. The type must be bool, integral or floating point.
309 template <typename T>
310 T Get() const
311 {
312 if constexpr (std::is_same_v<T, bool>)
313 {
314 return Boolean;
315 }
316 else if constexpr (std::is_integral_v<T>)
317 {
318 return Integer;
319 }
320 else if constexpr (std::is_floating_point_v<T>)
321 {
322 return Float;
323 }
324 else
325 {
326 check(false && "unexpected type T.");
327 }
328 }
329};

428// A block of instructions.
429//
430// A block is a sequence of instructions in order of execution. Blocks are organized in a
431// tree-like structure, used to model blocks nested inside other blocks.
432struct FBlock
433{
434 // This block's parent block, if any. If this is null, this is a *root block.
435 FBlock* Parent;
436
437 // The linked-list head of the instructions contained in this block.
438 // Links are contained in the FInstruction::Next field.
439 FInstruction* Instructions;
440
441 // Depth of this block in the tree structure (root blocks have level zero).
442 int32 Level;
443
444 // Finds and returns the common block between this and Other, if any. It returns null
445 // if no common block was found.
446 // Note: This has O(n) complexity, where n is the maximum depth of the tree structure.
447 FBlock* FindCommonParentWith(MIR::FBlock* Other);
448};

468// Base struct of an instruction value.
469//
470// An instruction is a value in a well defined order of evaluation.
471// Instructions have a parent block and are organized in a linked list: the Next field
472// indicates the instruction that will execute immediately after this one.
473// Since the material shader has multiple stages, the same instruction can belong to two
474// different graphs of execution, which explains why all fields in this structure have
475// a different possible value per stage.
476//
477// Note: All fields in this struct are not expected to be set by the user when emitting
478// an instruction, and are instead automatically populated by the builder.
479struct FInstruction : FValue
480{
481 // Array of linkage information (how this instruction is connected inside the IR graph)
482 // for each entry point registered in the module.
483 TArrayView<FInstructionLinkage> Linkage;
484
485 // Returns the block in which the dependency with specified index should execute.
486 FBlock* GetTargetBlockForUse(int32 EntryPointIndex, int32 UseIndex);
487
488 // Returns the next instruction in specified entry point.
489 FInstruction* GetNext(int32 EntryPointIndex) const { return Linkage[EntryPointIndex].Next; }
490
491 // Returns the number of user sin specified entry point.
492 uint32 GetNumUsers(int32 EntryPointIndex) const { return Linkage[EntryPointIndex].NumUsers; }
493
494 // Returns the block this instruction belongs to in specified entry point.
495 const FBlock* GetBlock(int32 EntryPointIndex) const { return Linkage[EntryPointIndex].Block; }
496
497 // Retrieves a user-specified name string if this instruction provides one.
498 FStringView GetName() const;
499
500 // Retrieves the priority if this instruction is used as a entry point output.
501 int32 GetOutputPriority() const;
502};
