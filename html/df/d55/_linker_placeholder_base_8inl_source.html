16template<class PlaceholderType>
17int32 TLinkerImportPlaceholder<PlaceholderType>::ResolveAllPlaceholderReferences(UObject* ReplacementObj)
18{
19 if (UObjectRedirector* ReplacementRedirector = Cast<UObjectRedirector>(ReplacementObj))
20 {
21 if (FLinkerLoad* ReplacementLinker = ReplacementRedirector->GetLinker())
22 {
23 if (!ReplacementRedirector->HasAnyFlags(RF_LoadCompleted))
24 {
25 // we're in the midst of serializing this redirector
26 // somewhere up the stack, in some scenario like this:
27 //
28 // - ClassA and ClassC both depend on ClassB
29 // - ClassB has a redirector to ClassB_2
30 // - ClassB_2 depends on ClassC
31 //
32 // if ClassA is loaded first, it then goes to load ClassB, which
33 // seeks to serialize in its UObjectRedirector; before that's
34 // set it loads ClassB_2 and subsequently ClassC; ClassC ends up
35 // here, needing to use the ClassB redirector, but we haven't
36 // returned up the stack for it to be set yet... here we force
37 // it to finish preloading (like we do in VerifyImport):
38 UE::TUniqueLock LinkerLock(ReplacementLinker->Mutex);
39 ReplacementRedirector->SetFlags(RF_NeedLoad);
40 ReplacementLinker->Preload(ReplacementRedirector);
41 }
42 }
43 ReplacementObj = ReplacementRedirector->DestinationObject;
44 }
45
46 PlaceholderType* TypeCheckedReplacement = CastChecked<PlaceholderType>(ReplacementObj, ECastCheckedType::NullAllowed);
47
48 int32 ReplacementCount = ResolvePropertyReferences(TypeCheckedReplacement);
49 ReplacementCount += ResolveScriptReferences(TypeCheckedReplacement);
50
51#if USE_DEFERRED_DEPENDENCY_CHECK_VERIFICATION_TESTS
52 ReplacementCount += ChildObjects.Num();
53 ChildObjects.Empty();
54#endif//USE_DEFERRED_DEPENDENCY_CHECK_VERIFICATION_TESTS
55
56 ReplacementCount += DerivedFunctions.Num();
57 for (auto Entry : DerivedFunctions)
58 {
59 Entry->SetSuperStruct(TypeCheckedReplacement);
60 }
61 DerivedFunctions.Empty();
62
63 ReplacementCount += FLinkerPlaceholderBase::ResolveAllPlaceholderReferences(ReplacementObj);
64
65 return ReplacementCount;
66}

83template<class PlaceholderType>
84void TLinkerImportPlaceholder<PlaceholderType>::AddReferencingProperty(FFieldVariant ReferencingProperty)
85{
86#if USE_DEFERRED_DEPENDENCY_CHECK_VERIFICATION_TESTS
87 UObject* ThisAsObject = GetPlaceholderAsUObject();
88 check(ThisAsObject != nullptr);
89
90 if (!ReferencingProperty.IsUObject())
91 {
92 if (FLinkerLoad* PropertyLinker = ReferencingProperty.Get<FProperty>()->GetLinker())
93 {
94 check(ThisAsObject->GetOutermost() == PropertyLinker->LinkerRoot);
95 check(PropertyLinker->LoadFlags & LOAD_DeferDependencyLoads);
96 }
97 }
98 // if this check hits, then we're adding dependencies after we've
99 // already resolved the placeholder (it won't be resolved again)
100 check(!IsMarkedResolved());
101#endif // USE_DEFERRED_DEPENDENCY_CHECK_VERIFICATION_TESTS
102
103 ReferencingProperties.Add(ReferencingProperty);
104}

157//------------------------------------------------------------------------------
158template<class PlaceholderType>
159int32 TLinkerImportPlaceholder<PlaceholderType>::ResolveScriptReferences(PlaceholderType* ReplacementObj)
160{
161 int32 ReplacementCount = 0;
162 PlaceholderType* PlaceholderObj = CastChecked<PlaceholderType>(GetPlaceholderAsUObject());
163
164 for (PlaceholderType** ScriptRefPtr : ReferencingScriptExpressions)
165 {
166 PlaceholderType*& ScriptRef = *ScriptRefPtr;
167 if (ScriptRef == PlaceholderObj)
168 {
169 ScriptRef = ReplacementObj;
170 ++ReplacementCount;
171 }
172 }
173
174 ReferencingScriptExpressions.Empty();
175 return ReplacementCount;
176}
