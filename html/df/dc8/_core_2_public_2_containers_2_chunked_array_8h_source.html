254 int32 Add( int32 Count=1 )
255 {
256 check(Count>=0);
257 checkSlow(NumElements>=0);
258
259 const int32 OldNum = NumElements;
260 const int32 NewNumElements = OldNum + Count;
261 const int32 NewNumChunks = (NewNumElements + NumElementsPerChunk - 1)/NumElementsPerChunk;
262 NumElements = NewNumElements;
263 for (int32 NumChunks = Chunks.Num(); NumChunks < NewNumChunks; ++NumChunks)
264 {
265 Chunks.Add(new FChunk);
266 }
267
268 return OldNum;

271 template<typename OtherAllocator>
272 void CopyToLinearArray(TArray<ElementType, OtherAllocator>& DestinationArray)
273 {
274 if (NumElements > 0)
275 {
276 int32 OriginalNumElements = DestinationArray.Num();
277 DestinationArray.AddUninitialized(NumElements);
278 InElementType* CopyDestPtr = &DestinationArray[OriginalNumElements];
279
280 for (int32 ChunkIndex = 0; ChunkIndex < Chunks.Num(); ChunkIndex++)
281 {
282 const int32 NumElementsInCurrentChunk = FMath::Min<int32>(NumElements - ChunkIndex * NumElementsPerChunk, NumElementsPerChunk);
283 check(NumElementsInCurrentChunk > 0);
284 ConstructItems<ElementType>(CopyDestPtr, &Chunks[ChunkIndex].Elements[0], NumElementsInCurrentChunk);
285 CopyDestPtr += NumElementsInCurrentChunk;
286 }
287 }
288 }

290 template<typename OtherAllocator>
291 void MoveToLinearArray(TArray<ElementType, OtherAllocator>& DestinationArray)
292 {
293 if (NumElements > 0)
294 {
295 int32 OriginalNumElements = DestinationArray.Num();
296 DestinationArray.AddUninitialized(NumElements);
297 InElementType* CopyDestPtr = &DestinationArray[OriginalNumElements];
298
299 for (int32 ChunkIndex = 0; ChunkIndex < Chunks.Num(); ChunkIndex++)
300 {
301 const int32 NumElementsInCurrentChunk = FMath::Min<int32>(NumElements - ChunkIndex * NumElementsPerChunk, NumElementsPerChunk);
302 check(NumElementsInCurrentChunk > 0);
303 MoveConstructItems<ElementType>(CopyDestPtr, &Chunks[ChunkIndex].Elements[0], NumElementsInCurrentChunk);
304 CopyDestPtr += NumElementsInCurrentChunk;
305 }
306
307 Empty();
308 }
