86struct FNotifyNodeInterface : public INotifyNodeObjectInterface
87{
88 FAnimNotifyEvent* NotifyEvent;
89
90 // Cached notify name (can be generated by blueprints so want to cache this instead of hitting VM)
91 FName CachedNotifyName;
92
93 // Stable Guid that allows us to refer to notify event
94 FGuid Guid;
95
96 FNotifyNodeInterface(FAnimNotifyEvent* InAnimNotifyEvent) : NotifyEvent(InAnimNotifyEvent), Guid(NotifyEvent->Guid) {}
97 virtual ENotifyNodeObjectTypes::Type GetType() const override { return ENotifyNodeObjectTypes::NOTIFY; }
98 virtual FAnimNotifyEvent* GetNotifyEvent() override { return NotifyEvent; }
99 virtual int GetTrackIndex() const override { return NotifyEvent->TrackIndex; }
100 virtual float GetTime(EAnimLinkMethod::Type ReferenceFrame = EAnimLinkMethod::Absolute) const override { return NotifyEvent->GetTime(ReferenceFrame); }
101 virtual float GetDuration() override { return NotifyEvent->GetDuration(); }
102 virtual FName GetName() override { return CachedNotifyName; }
103 virtual bool IsBranchingPoint() override { return NotifyEvent->IsBranchingPoint(); }
104 virtual TOptional<FLinearColor> GetEditorColor() override
105 {
106 TOptional<FLinearColor> ReturnColour;
107 if (NotifyEvent->Notify)
108 {
109 ReturnColour = NotifyEvent->Notify->GetEditorColor();
110 }
111 else if (NotifyEvent->NotifyStateClass)
112 {
113 ReturnColour = NotifyEvent->NotifyStateClass->GetEditorColor();
114 }
115 return ReturnColour;
116 }
117
118 virtual FText GetNodeTooltip(const UAnimSequenceBase* Sequence) override
119 {
120 FText ToolTipText = MakeTooltipFromTime(Sequence, NotifyEvent->GetTime(), NotifyEvent->GetDuration());
121
122 if (NotifyEvent->IsBranchingPoint())
123 {
124 ToolTipText = FText::Format(LOCTEXT("AnimNotify_ToolTipBranchingPoint", "{0} (BranchingPoint)"), ToolTipText);
125 }
126
127 UObject* NotifyToDisplayClassOf = NotifyEvent->Notify;
128 if (NotifyToDisplayClassOf == nullptr)
129 {
130 NotifyToDisplayClassOf = NotifyEvent->NotifyStateClass;
131 }
132
133 if (NotifyToDisplayClassOf != nullptr)
134 {
135 ToolTipText = FText::Format(LOCTEXT("AnimNotify_ToolTipNotifyClass", "{0}\nClass: {1}"), ToolTipText, NotifyToDisplayClassOf->GetClass()->GetDisplayNameText());
136 }
137
138 return ToolTipText;
139 }
140
141 virtual TOptional<UObject*> GetObjectBeingDisplayed() override
142 {
143 if (NotifyEvent->Notify)
144 {
145 return TOptional<UObject*>(NotifyEvent->Notify);
146 }
147
148 if (NotifyEvent->NotifyStateClass)
149 {
150 return TOptional<UObject*>(NotifyEvent->NotifyStateClass);
151 }
152 return TOptional<UObject*>();
153 }
154
155 virtual void SetTime(float Time, EAnimLinkMethod::Type ReferenceFrame = EAnimLinkMethod::Absolute) override { NotifyEvent->SetTime(Time, ReferenceFrame); }
156 virtual void SetDuration(float Duration) override { NotifyEvent->SetDuration(Duration); }
157
158 virtual void HandleDrop(class UAnimSequenceBase* Sequence, float Time, int32 TrackIndex) override
159 {
160 float EventDuration = NotifyEvent->GetDuration();
161
162 NotifyEvent->Link(Sequence, Time, NotifyEvent->GetSlotIndex());
163 NotifyEvent->RefreshTriggerOffset(Sequence->CalculateOffsetForNotify(NotifyEvent->GetTime()));
164
165 if (EventDuration > 0.0f)
166 {
167 NotifyEvent->EndLink.Link(Sequence, NotifyEvent->GetTime() + EventDuration, NotifyEvent->GetSlotIndex());
168 NotifyEvent->RefreshEndTriggerOffset(Sequence->CalculateOffsetForNotify(NotifyEvent->EndLink.GetTime()));
169 }
170 else
171 {
172 NotifyEvent->EndTriggerTimeOffset = 0.0f;
173 }
174
175 NotifyEvent->TrackIndex = TrackIndex;
176 }
177
178 virtual void CacheName() override
179 {
180 if (NotifyEvent->Notify)
181 {
182 CachedNotifyName = FName(*NotifyEvent->Notify->GetNotifyName());
183 }
184 else if (NotifyEvent->NotifyStateClass)
185 {
186 CachedNotifyName = FName(*NotifyEvent->NotifyStateClass->GetNotifyName());
187 }
188 else
189 {
190 CachedNotifyName = NotifyEvent->NotifyName;
191 }
192 }
193
194 virtual void Delete(UAnimSequenceBase* Seq) override
195 {
196 for (int32 I = 0; I < Seq->Notifies.Num(); ++I)
197 {
198 if (NotifyEvent == &(Seq->Notifies[I]))
199 {
200 Seq->Notifies.RemoveAt(I);
201 Seq->PostEditChange();
202 Seq->MarkPackageDirty();
203 break;
204 }
205 }
206 }
207
208
209 virtual void MarkForDelete(UAnimSequenceBase* Seq) override
210 {
211 for (int32 I = 0; I < Seq->Notifies.Num(); ++I)
212 {
213 if (NotifyEvent == &(Seq->Notifies[I]))
214 {
215 Seq->Notifies[I].Guid = FGuid();
216 break;
217 }
218 }
219 }
220
221 virtual void ExportForCopy(UAnimSequenceBase* Seq, FString& StrValue) const override
222 {
223 int32 Index = INDEX_NONE;
224 for (int32 NotifyIdx = 0; NotifyIdx < Seq->Notifies.Num(); ++NotifyIdx)
225 {
226 if (NotifyEvent == &Seq->Notifies[NotifyIdx])
227 {
228 Index = NotifyIdx;
229 break;
230 }
231 }
232
233 check(Index != INDEX_NONE);
234
235 FArrayProperty* ArrayProperty = NULL;
236 uint8* PropertyData = Seq->FindNotifyPropertyData(Index, ArrayProperty);
237 if (PropertyData && ArrayProperty)
238 {
239 ArrayProperty->Inner->ExportTextItem_Direct(StrValue, PropertyData, PropertyData, Seq, PPF_Copy);
240 }
241 }
242
243 virtual FGuid GetGuid() const override
244 {
245 return Guid;
246 }
247
248 static void RemoveInvalidNotifies(UAnimSequenceBase* SeqBase)
249 {
250 SeqBase->Notifies.RemoveAll([](const FAnimNotifyEvent& InNotifyEvent) { return !InNotifyEvent.Guid.IsValid(); });
251 SeqBase->PostEditChange();
252 SeqBase->MarkPackageDirty();
253 }
