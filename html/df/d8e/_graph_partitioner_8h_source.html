75template< typename FGetCenter >
76void FGraphPartitioner::BuildLocalityLinks( FDisjointSet& DisjointSet, const FBounds3f& Bounds, TConstArrayView< const int32 > GroupIndexes, FGetCenter& GetCenter )
77{
78 TArray< uint32 > SortKeys;
79 SortKeys.AddUninitialized( NumElements );
80 SortedTo.AddUninitialized( NumElements );
81
82 const bool bElementGroups = !GroupIndexes.IsEmpty(); // Only create locality links between elements with the same group index
83
84 ParallelFor( TEXT("BuildLocalityLinks.PF"), NumElements, 4096,
85 [&]( uint32 Index )
86 {
87 FVector3f Center = GetCenter( Index );
88 FVector3f CenterLocal = ( Center - Bounds.Min ) / FVector3f( Bounds.Max - Bounds.Min ).GetMax();
89
90 uint32 Morton;
91 Morton = FMath::MortonCode3( uint32( CenterLocal.X * 1023 ) );
92 Morton |= FMath::MortonCode3( uint32( CenterLocal.Y * 1023 ) ) << 1;
93 Morton |= FMath::MortonCode3( uint32( CenterLocal.Z * 1023 ) ) << 2;
94 SortKeys[ Index ] = Morton;
95 } );
96
97 RadixSort32( SortedTo.GetData(), Indexes.GetData(), NumElements,
98 [&]( uint32 Index )
99 {
100 return SortKeys[ Index ];
101 } );
102
103 SortKeys.Empty();
104
105 Swap( Indexes, SortedTo );
106 for( uint32 i = 0; i < NumElements; i++ )
107 {
108 SortedTo[ Indexes[i] ] = i;
109 }
110
111 TArray< FRange > IslandRuns;
112 IslandRuns.AddUninitialized( NumElements );
113
114 // Run length acceleration
115 // Range of identical IslandID denoting that elements are connected.
116 // Used for jumping past connected elements to the next nearby disjoint element.
117 {
118 uint32 RunIslandID = 0;
119 uint32 RunFirstElement = 0;
120
121 for( uint32 i = 0; i < NumElements; i++ )
122 {
123 uint32 IslandID = DisjointSet.Find( Indexes[i] );
124
125 if( RunIslandID != IslandID )
126 {
127 // We found the end so rewind to the beginning of the run and fill.
128 for( uint32 j = RunFirstElement; j < i; j++ )
129 {
130 IslandRuns[j].End = i;
131 }
132
133 // Start the next run
134 RunIslandID = IslandID;
135 RunFirstElement = i;
136 }
137
138 IslandRuns[i].Begin = RunFirstElement;
139 }
140 // Finish the last run
141 for( uint32 j = RunFirstElement; j < NumElements; j++ )
142 {
143 IslandRuns[j].End = NumElements;
144 }
145 }
146
147 for( uint32 i = 0; i < NumElements; i++ )
148 {
149 uint32 Index = Indexes[i];
150
151 uint32 RunLength = IslandRuns[i].Num();
152 if( RunLength < 128 )
153 {
154 uint32 IslandID = DisjointSet[ Index ];
155 int32 GroupID = bElementGroups ? GroupIndexes[ Index ] : 0;
156
157 FVector3f Center = GetCenter( Index );
158
159 const uint32 MaxLinksPerElement = 5;
160
161 uint32 ClosestIndex[MaxLinksPerElement];
162 float ClosestDist2[MaxLinksPerElement];
163 for (int32 k = 0; k < MaxLinksPerElement; k++)
164 {
165 ClosestIndex[k] = ~0u;
166 ClosestDist2[k] = MAX_flt;
167 }
168
169 for( int Direction = 0; Direction < 2; Direction++ )
170 {
171 uint32 Limit = Direction ? NumElements - 1 : 0;
172 uint32 Step = Direction ? 1 : -1;
173
174 uint32 Adj = i;
175 for( int32 Iterations = 0; Iterations < 16; Iterations++ )
176 {
177 if( Adj == Limit )
178 break;
179 Adj += Step;
180
181 uint32 AdjIndex = Indexes[ Adj ];
182 uint32 AdjIslandID = DisjointSet[ AdjIndex ];
183 int32 AdjGroupID = bElementGroups ? GroupIndexes[AdjIndex] : 0;
184 if( IslandID == AdjIslandID || ( GroupID != AdjGroupID ) )
185 {
186 // Skip past this run
187 if( Direction )
188 Adj = IslandRuns[ Adj ].End - 1;
189 else
190 Adj = IslandRuns[ Adj ].Begin;
191 }
192 else
193 {
194 // Add to sorted list
195 float AdjDist2 = ( Center - GetCenter( AdjIndex ) ).SizeSquared();
196 for( int k = 0; k < MaxLinksPerElement; k++ )
197 {
198 if( AdjDist2 < ClosestDist2[k] )
199 {
200 Swap( AdjIndex, ClosestIndex[k] );
201 Swap( AdjDist2, ClosestDist2[k] );
202 }
203 }
204 }
205 }
206 }
207
208 for( int k = 0; k < MaxLinksPerElement; k++ )
209 {
210 if( ClosestIndex[k] != ~0u )
211 {
212 // Add both directions
213 LocalityLinks.AddUnique( Index, ClosestIndex[k] );
214 LocalityLinks.AddUnique( ClosestIndex[k], Index );
215 }
216 }
217 }
218 }
