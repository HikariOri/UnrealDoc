79template<typename T>
80bool IsInterclusterEdge(const TPBDRigidParticleHandle<T, 3>& Particle, const TConnectivityEdge<T>& Edge)
81{
82 if (!Edge.Sibling)
83 {
84 return false;
85 }
86
87 const TPBDRigidClusteredParticleHandle<T, 3>* ClusterParticle = Particle.CastToClustered();
88 const TPBDRigidClusteredParticleHandle<T, 3>* SiblingParticle = Edge.Sibling->CastToClustered();
89
90 if (!ClusterParticle || !SiblingParticle)
91 {
92 return false;
93 }
94
95 return ClusterParticle->Parent() != SiblingParticle->Parent();
96}
