69template <typename NativeType, typename = void>
70struct TToVValue
71{
72 TOptional<NativeType> Value;
73
74 void Emplace() { Value.Emplace(); }
75
76 TToVValue& operator=(NativeType Other)
77 {
78 Value = MoveTemp(Other);
79 return *this;

337 static FOpResult FromVValue(FAllocationContext Context, const VValue Value, TFromVValue<FVerseRational>& OutNative)
338 {
339 V_REQUIRE_CONCRETE(Value);
340 TFromVValue<int64> Numerator;
341 TFromVValue<int64> Denominator;
342 if (Value.IsInt())
343 {
344 FOpResult NumeratorResult = FromVValue(Context, Value, Numerator);
345 if (!NumeratorResult.IsReturn())
346 {
347 return NumeratorResult;
348 }
349 Denominator.Value = 1;
350 }
351 else if (VRational* Rational = Value.DynamicCast<VRational>())
352 {
353 FOpResult NumeratorResult = FromVValue(Context, Rational->Numerator.Get(), Numerator);
354 if (!NumeratorResult.IsReturn())
355 {
356 return NumeratorResult;
357 }
358 FOpResult DenominatorResult = FromVValue(Context, Rational->Denominator.Get(), Denominator);
359 if (!DenominatorResult.IsReturn())
360 {
361 return DenominatorResult;
362 }
363 }
364 else
365 {
366 V_DIE("Unexpected rational value");
367 }
368 OutNative.Value.Numerator = Numerator.GetValue();
369 OutNative.Value.Denominator = Denominator.GetValue();
370 return {FOpResult::Return};
371 }

460 template <typename... ElementTypes, size_t... Indices>
461 static FOpResult FromVValue(FAllocationContext Context, const VValue Value, TFromVValue<TNativeTuple<ElementTypes...>>& OutNative, std::index_sequence<Indices...>)
462 {
463 V_REQUIRE_CONCRETE(Value);
464 const VArrayBase& Array = Value.StaticCast<VArrayBase>();
465 FOpResult Result{FOpResult::Return};
466 auto ElementFromVValue = [&](const VValue ElementValue, auto& OutNativeElement) {
467 // Only convert this element if the previous element successfully converted
468 if (Result.IsReturn())
469 {
470 TFromVValue<typename TRemoveReference<decltype(OutNativeElement)>::Type> NativeElement;
471 Result = FromVValue(Context, ElementValue, NativeElement);
472 if (Result.IsReturn())
473 {
474 OutNativeElement = NativeElement.GetValue();
475 }
476 }
477 };
478 (ElementFromVValue(Array.GetValue(Indices), OutNative.Value.template Get<Indices>()), ...);
479 return Result;
480 }

497 template <class ElementType>
498 static FOpResult FromVValue(FAllocationContext Context, const VValue Value, TFromVValue<TArray<ElementType>>& OutNative)
499 {
500 V_REQUIRE_CONCRETE(Value);
501 V_DIE_UNLESS(Value.IsCellOfType<VArrayBase>());
502 const VArrayBase& Array = Value.StaticCast<VArrayBase>();
503 ensure(OutNative.Value.IsEmpty());
504 OutNative.Value.Reserve(Array.Num());
505 for (const VValue Element : Array)
506 {
507 TFromVValue<ElementType> NativeElement;
508 FOpResult ElementOpResult = FromVValue(Context, Element, NativeElement);
509 if (!ElementOpResult.IsReturn())
510 {
511 return ElementOpResult;
512 }
513 OutNative.Value.Add(NativeElement.GetValue());
514 }
515 return {FOpResult::Return};
516 }

518 template <class KeyType, class ValueType>
519 static FOpResult FromVValue(FAllocationContext Context, const VValue Value, TFromVValue<TMap<KeyType, ValueType>>& OutNative)
520 {
521 V_REQUIRE_CONCRETE(Value);
522 V_DIE_UNLESS(Value.IsCellOfType<VMap>());
523 const VMap& Map = Value.StaticCast<VMap>();
524 ensure(OutNative.Value.IsEmpty());
525 OutNative.Value.Reserve(Map.Num());
526 for (const TPair<VValue, VValue> Pair : Map)
527 {
528 TFromVValue<KeyType> NativeKey;
529 FOpResult KeyOpResult = FromVValue(Context, Pair.Key, NativeKey);
530 if (!KeyOpResult.IsReturn())
531 {
532 return KeyOpResult;
533 }
534 TFromVValue<ValueType> NativeValue;
535 FOpResult ValueOpResult = FromVValue(Context, Pair.Value, NativeValue);
536 if (!ValueOpResult.IsReturn())
537 {
538 return ValueOpResult;
539 }
540 OutNative.Value.Add(NativeKey.GetValue(), NativeValue.GetValue());
541 }
542 return {FOpResult::Return};
