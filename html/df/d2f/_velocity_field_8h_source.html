228 static FSolverVec3 CalculateForceLowLevel(const FSolverVec3 &InV, const FSolverVec3 &InN, const FSolverReal InQuarterRho, const FSolverReal InDoubleArea,
229 FSolverReal InCdI, FSolverReal InCdO, const FSolverReal InClI, const FSolverReal InClO, const FSolverReal InCp, const FSolverReal InTr)
230 {
231 // Set the aerodynamic forces
232 const FSolverReal VDotN = FSolverVec3::DotProduct(InV, InN);
233 const FSolverReal VSquare = FSolverVec3::DotProduct(InV, InV);
234
235 const FSolverReal TurbulenceFactor = VSquare > UE_SMALL_NUMBER ? ((1.f - InTr) / FMath::Sqrt(VSquare) + InTr) : 1.f;
236
237 return InQuarterRho * InDoubleArea * TurbulenceFactor * (VDotN >= (FSolverReal)0. ? // The flow can hit either side of the triangle, so the normal might need to be reversed
238 (InCdI - InClI) * VDotN * InV + InClI * VSquare * InN :
239 (InClO - InCdO) * VDotN * InV - InClO * VSquare * InN) + InDoubleArea * (FSolverReal)0.5 * InCp * InN;
240 }

242 FSolverVec3 CalculateForce(const TConstArrayView<FSolverVec3>& Xs, const TConstArrayView<FSolverVec3>& Vs, int32 ElementIndex, const FSolverVec3& InVelocity, const FSolverReal CdI, const FSolverReal CdO, const FSolverReal ClI, const FSolverReal ClO, const FSolverReal Cp) const
243 {
244 const TVec3<int32>& Element = Elements[ElementIndex];
245
246 // Calculate the normal and the area of the surface exposed to the flow
247 FSolverVec3 N = FSolverVec3::CrossProduct(
248 Xs[Element[2]] - Xs[Element[0]],
249 Xs[Element[1]] - Xs[Element[0]]);
250 const FSolverReal DoubleArea = N.SafeNormalize();
251
252 // Calculate the direction and the relative velocity of the triangle to the flow
253 const FSolverVec3& SurfaceVelocity = (FSolverReal)(1. / 3.) * (
254 Vs[Element[0]] +
255 Vs[Element[1]] +
256 Vs[Element[2]]);
257 const FSolverVec3 V = InVelocity - SurfaceVelocity;
258
259 return CalculateForceLowLevel(V, N, QuarterRho, DoubleArea, CdI, CdO, ClI, ClO, Cp, TurbulenceRatio);
260 }

267 FSolverVec3 CalculateForce(const TConstArrayView<FSolverVec3>& Xs, const TConstArrayView<FSolverVec3>& Vs, int32 ElementIndex, const FSolverVec3& InVelocity, const FSolverReal CdI, const FSolverReal CdO, const FSolverReal ClI, const FSolverReal ClO, const FSolverReal Cp, const FSolverReal MaxVelocitySquared) const
268 {
269 checkSlow(MaxVelocitySquared > (FSolverReal)0);
270
271 const TVec3<int32>& Element = Elements[ElementIndex];
272
273 // Calculate the normal and the area of the surface exposed to the flow
274 FSolverVec3 N = FSolverVec3::CrossProduct(
275 Xs[Element[2]] - Xs[Element[0]],
276 Xs[Element[1]] - Xs[Element[0]]);
277 const FSolverReal DoubleArea = N.SafeNormalize();
278
279 // Calculate the direction and the relative velocity of the triangle to the flow
280 const FSolverVec3& SurfaceVelocity = (FSolverReal)(1. / 3.) * (
281 Vs[Element[0]] +
282 Vs[Element[1]] +
283 Vs[Element[2]]);
284 FSolverVec3 V = InVelocity - SurfaceVelocity;
285
286 // Clamp the velocity
287 const FSolverReal RelVelocitySquared = V.SquaredLength();
288 if (RelVelocitySquared > MaxVelocitySquared)
289 {
290 V *= FMath::Sqrt(MaxVelocitySquared / RelVelocitySquared);
291 }
292
293
294 return CalculateForceLowLevel(V, N, QuarterRho, DoubleArea, CdI, CdO, ClI, ClO, Cp, TurbulenceRatio);
295 }
