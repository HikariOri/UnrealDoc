87// Needs to stay inline for shader formats which can't depend on the RHI module
88inline const TCHAR* GetShaderFrequencyString(EShaderFrequency Frequency, bool bIncludePrefix = true)
89{
90 const TCHAR* String = TEXT("SF_NumFrequencies");
91 switch (Frequency)
92 {
93 case SF_Vertex: String = TEXT("SF_Vertex"); break;
94 case SF_Mesh: String = TEXT("SF_Mesh"); break;
95 case SF_Amplification: String = TEXT("SF_Amplification"); break;
96 case SF_Geometry: String = TEXT("SF_Geometry"); break;
97 case SF_Pixel: String = TEXT("SF_Pixel"); break;
98 case SF_Compute: String = TEXT("SF_Compute"); break;
99 case SF_RayGen: String = TEXT("SF_RayGen"); break;
100 case SF_RayMiss: String = TEXT("SF_RayMiss"); break;
101 case SF_RayHitGroup: String = TEXT("SF_RayHitGroup"); break;
102 case SF_RayCallable: String = TEXT("SF_RayCallable"); break;
103 case SF_WorkGraphRoot: String = TEXT("SF_WorkGraphRoot"); break;
104 case SF_WorkGraphComputeNode: String = TEXT("SF_WorkGraphComputeNode"); break;
105
106 default:
107 break;
108 }
109
110 // Skip SF_
111 int32 Index = bIncludePrefix ? 0 : 3;
112 String += Index;
113 return String;
114}
