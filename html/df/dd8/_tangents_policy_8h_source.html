64 template<typename T>
65 static std::enable_if_t<TIsContiguousFloat<T>::value, void>
66 ComputeClampableTangent(
67 float PrevTime, const T& PrevPoint,
68 float CurTime, const T& CurPoint,
69 float NextTime, const T& NextPoint,
70 float Tension,
71 bool bWantClamping,
72 T& OutTangent)
73 {
74 if (bWantClamping)
75 {
76 static_assert(std::is_trivially_copyable_v<T>, "Type must be trivially copyable.");
77 static_assert(sizeof(T) % sizeof(float) == 0, "Type size must be a multiple of float.");
78
79 const float* Prev = reinterpret_cast<const float*>(&PrevPoint);
80 const float* Cur = reinterpret_cast<const float*>(&CurPoint);
81 const float* Next = reinterpret_cast<const float*>(&NextPoint);
82 float* Out = reinterpret_cast<float*>(&OutTangent);
83
84 constexpr int32 N = int32(sizeof(T) / sizeof(float));
85 for (int32 i = 0; i < N; ++i)
86 {
87 const float Clamped = ClampFloatTangent(
88 Prev[i], PrevTime,
89 Cur[i], CurTime,
90 Next[i], NextTime);
91
92 Out[i] = (1.0f - Tension) * Clamped;
93 }
94 }
95 else
96 {
97 AutoCalcTangent(PrevPoint, CurPoint, NextPoint, Tension, OutTangent);
98 const float PrevToNext = FMath::Max(UE_KINDA_SMALL_NUMBER, NextTime - PrevTime);
99 OutTangent /= PrevToNext;
100 }
101 }
