16// Represents a full driver version with multiple version numbers,
17// e.g. 30.0.13023.4001.
18class FDriverVersion
19{
20public:
21 FDriverVersion() = default;
22
23 FDriverVersion(const FString& DriverVersionString)
24 {
25 Parse(DriverVersionString);
26 }
27
28 void Parse(const FString& DriverVersionString)
29 {
30 Values.Empty();
31 FString RemainingString = DriverVersionString;
32 FString CurrentVersionNumber;
33
34 uint32_t CurrentVersionNumberIdx = 0;
35 while (RemainingString.Split(TEXT("."), &CurrentVersionNumber, &RemainingString))
36 {
37 Values.Add(FCString::Atoi(*CurrentVersionNumber));
38 }
39 Values.Add(FCString::Atoi(*RemainingString));
40 }
41
42 bool operator>(const FDriverVersion& Other) const
43 {
44 for (int32 Idx = 0; Idx < Values.Num(); ++Idx)
45 {
46 if (Values[Idx] > Other.Values[Idx])
47 {
48 return true;
49 }
50 if (Values[Idx] < Other.Values[Idx])
51 {
52 return false;
53 }
54 }
55
56 return false;
57 }
58
59 bool operator==(const FDriverVersion& Other) const
60 {
61 return !(*this > Other) && !(Other > *this);
62 }
63 bool operator<(const FDriverVersion& Other) const
64 {
65 return Other > *this;
66 }
67 bool operator!=(const FDriverVersion& Other) const
68 {
69 return !(*this == Other);
70 }
71 bool operator>=(const FDriverVersion& Other) const
72 {
73 return (*this == Other) || (*this > Other);
74 }
75 bool operator<=(const FDriverVersion& Other) const
76 {
77 return (*this == Other) || (*this < Other);
78 }
79
80 // Index from the left (major version first).
81 uint32 GetVersionValue(uint32 Index) const
82 {
83 return Values[Index];
84 }
85
86 const TArray<uint32>& GetVersionValues() const
87 {
88 return Values;
89 }

100// Find the comparison operator in the input string, and advances the In pointer to point to the first character
101// after the comparison operator.
102// Defaults to Equal if no comparison operator is present.
103inline EComparisonOp ParseComparisonOp(const TCHAR*& In)
104{
105 if (In[0] == '=' && In[1] == '=')
106 {
107 In += 2;
108 return ECO_Equal;
109 }
110 if (In[0] == '!' && In[1] == '=')
111 {
112 In += 2;
113 return ECO_NotEqual;
114 }
115 if (*In == '>')
116 {
117 ++In;
118 if (*In == '=')
119 {
120 ++In;
121 return ECO_GreaterOrEqual;
122 }
123 return ECO_Greater;
124 }
125 if (*In == '<')
126 {
127 ++In;
128 if (*In == '=')
129 {
130 ++In;
131 return ECO_LessOrEqual;
132 }
133 return ECO_Less;
134 }
135 return ECO_Equal;
136}

138// Compare two objects given the comparison operator.
139template <class T>
140bool CompareWithOp(const T& A, EComparisonOp Op, const T& B)
141{
142 switch (Op)
143 {
144 case ECO_Equal: return A == B;
145 case ECO_NotEqual: return A != B;
146 case ECO_Greater: return A > B;
147 case ECO_GreaterOrEqual: return A >= B;
148 case ECO_Less: return A < B;
149 case ECO_LessOrEqual: return A <= B;
150 }
151 check(0);
152 return false;
153}

224 static FString GetNVIDIAUnifiedVersion(const FString& InternalVersion)
225 {
226 // Ignore the Windows/DirectX version by taking the last digits of the internal version
227 // and moving the version dot. Coincidentally, that's the user-facing string. For example:
228 // 9.18.13.4788 -> 3.4788 -> 347.88
229 if (InternalVersion.Len() < 6)
230 {
231 return InternalVersion;
232 }
233
234 FString RightPart = InternalVersion.Right(6);
235 RightPart = RightPart.Replace(TEXT("."), TEXT(""));
236 RightPart.InsertAt(3, TEXT("."));
237 return RightPart;
238 }
239
240 static FString GetIntelUnifiedVersion(const FString& InternalVersion)
241 {
242 // https://www.intel.com/content/www/us/en/support/articles/000005654/graphics.html
243 // Drop off the OS and DirectX version. For example:
244 // 27.20.100.8935 -> 100.8935
245 int32 DotIndex = InternalVersion.Find(TEXT("."), ESearchCase::CaseSensitive, ESearchDir::FromStart);
246 if (DotIndex != INDEX_NONE)
247 {
248 DotIndex = InternalVersion.Find(TEXT("."), ESearchCase::CaseSensitive, ESearchDir::FromStart, DotIndex + 1);
249 if (DotIndex != INDEX_NONE)
250 {
251 return InternalVersion.RightChop(DotIndex + 1);
252 }
253 }
254 return InternalVersion;
255 }
256
257 FString GetUnifiedDriverVersion() const
258 {
259 // We use the internal version as the base to avoid instability or changes to the user-facing driver versioning scheme,
260 // and we remove the parts of the version that identify unimportant factors like Windows or DirectX version.
261 const FString& FullVersion = InternalDriverVersion;
262
263 if (IsNVIDIA() && (InternalDriverVersion != UserDriverVersion))
264 {
265 return GetNVIDIAUnifiedVersion(FullVersion);
266 }
267 else if (IsIntel())
268 {
269 return GetIntelUnifiedVersion(FullVersion);
270 }
271 return FullVersion;
272 }

283// Represents an optional set of constraints that a driver configuration entry can include:
284// * The RHI being used (e.g. "D3D12" will result in the entry to be considered only when running in D3D12).
285// * The adapter name, using a regular expression (e.g. ".*RTX.*" will result in the entry to be considered only for cards that contain the string "RTX").
286// * The adapter generation. This is a vendor-specific integer and may not be implemented for all vendors.
287struct FDriverConfigEntryConstraints {
288 struct FIntegerConstraint {
289 EComparisonOp ComparisonOp;
290 uint32 Value;
291
292 static FIntegerConstraint FromString(const FString& IntegerConstraintString)
293 {
294 FIntegerConstraint Constraint;
295 const TCHAR* Chars = *IntegerConstraintString;
296 Constraint.ComparisonOp = ParseComparisonOp(Chars);
297 Constraint.Value = FCString::Atoi(Chars);
298 return Constraint;
299 }
300 };
301
302 TOptional<FString> RHINameConstraint;
303 TOptional<FRegexPattern> AdapterNameRegexConstraint;
304 TOptional<FIntegerConstraint> AdapterGenerationConstraint;
305
306 FDriverConfigEntryConstraints() = default;
307
308 FDriverConfigEntryConstraints(const TCHAR* Entry)
309 {
310 FString RHINameString;
311 FParse::Value(Entry, TEXT("RHI="), RHINameString);
312 if (!RHINameString.IsEmpty())
313 {
314 RHINameConstraint = RHINameString;
315 }
316
317 FString AdapterNameRegexString;
318 FParse::Value(Entry, TEXT("AdapterNameRegex="), AdapterNameRegexString);
319 if (!AdapterNameRegexString.IsEmpty())
320 {
321 AdapterNameRegexConstraint.Emplace(AdapterNameRegexString);
322 }
323
324 FString AdapterGenerationString;
325 FParse::Value(Entry, TEXT("AdapterGeneration="), AdapterGenerationString);
326 if (!AdapterGenerationString.IsEmpty())
327 {
328 AdapterGenerationConstraint = FIntegerConstraint::FromString(AdapterGenerationString);
329 }
330 }
331
332 bool AreConstraintsSatisfied(const FGPUDriverInfo& Info, uint32& OutNumSatisfiedConstraints) const
333 {
334 OutNumSatisfiedConstraints = 0;
335
336 if (RHINameConstraint)
337 {
338 if (*RHINameConstraint != Info.RHIName)
339 {
340 return false;
341 }
342 OutNumSatisfiedConstraints++;
343 }
344
345 if (AdapterNameRegexConstraint)
346 {
347 FRegexMatcher Matcher(*AdapterNameRegexConstraint, Info.DeviceDescription);
348 if (!Matcher.FindNext())
349 {
350 return false;
351 }
352 OutNumSatisfiedConstraints++;
353 }
354
355 if (AdapterGenerationConstraint)
356 {
357 if (!Info.IsAdapterGenerationKnown() || !CompareWithOp(Info.AdapterGeneration, AdapterGenerationConstraint->ComparisonOp, AdapterGenerationConstraint->Value))
358 {
359 return false;
360 }
361 OutNumSatisfiedConstraints++;
362 }
363
364 return true;
365 }
366
367 bool HasConstraints()
368 {
369 return RHINameConstraint || AdapterNameRegexConstraint || AdapterGenerationConstraint;
370 }

373// This corresponds to one DriverDenyList entry in the Hardware.ini configuration file. One entry includes:
374// * The drivers being denylisted, specified through one of the following:
375// * A driver version, along with any comparison operator (e.g. "<473.47" would denylist any driver which version is less than 473.47). This uses the "unified" driver version (see FGPUDriverInfo::GetUnifiedDriverVersion).
376// * A driver date, along with any comparison operator (e.g. "<=10-31-2023" would denylist any driver which release date is on or before Oct. 31, 2023).
377// * A set of optional constraints that the denylist entry will consider when the driver is being tested (see FDriverConfigEntryConstraints).
378// * The reason for the driver(s) being denylisted, e.g.visual glitches, stability, or performance issues.

434 // Checks whether the given driver is denylisted by this entry, and also returns the number
435 // of constraints that have been satisfied in OutNumSatisfiedConstraints.
436 bool AppliesToDriver(const FGPUDriverInfo& Info, uint32& OutNumSatisfiedConstraints) const
437 {
438 OutNumSatisfiedConstraints = 0;
439
440 if (!IsValid())
441 {
442 return false;
443 }
444
445 // Check constraints first. Only apply the driver version/date check
446 // if all constraints are satisfied.
447 if (!AreConstraintsSatisfied(Info, OutNumSatisfiedConstraints))
448 {
449 return false;
450 }
451
452 if (DriverVersion && Info.IsSameDriverVersionGeneration(DriverVersion->Version))
453 {
454 FDriverVersion CurrentDriverVersion(Info.GetUnifiedDriverVersion());
455 return CompareWithOp(CurrentDriverVersion, DriverVersion->ComparisonOp, DriverVersion->Version);
456 }
457
458 if (DriverDate)
459 {
460 FDateTime CurrentDriverDate = ParseMonthDayYearDate(Info.DriverDate);
461 return CompareWithOp(CurrentDriverDate, DriverDate->ComparisonOp, DriverDate->Date);
462 }
463
464 return false;
465 }
466
467 // Returns true whether this denylist entry will always apply to the latest drivers available,
468 // i.e. the comparison function is > or >=.
469 bool AppliesToLatestDrivers()
470 {
471 if (DriverVersion)
472 {
473 return DriverVersion->ComparisonOp == ECO_Greater || DriverVersion->ComparisonOp == ECO_GreaterOrEqual;
474 }
475 if (DriverDate)
476 {
477 return DriverDate->ComparisonOp == ECO_Greater || DriverDate->ComparisonOp == ECO_GreaterOrEqual;
478 }
479 return false;
480 }
481
482 bool IsValid() const
483 {
484 return DriverVersion.IsSet() != DriverDate.IsSet();
485 }

487 TOptional<FDriverVersionDenyListEntry> DriverVersion;
488 TOptional<FDriverDateDenyListEntry> DriverDate;
489
490 FString DenylistReason;

590 template <typename TEntryType>
591 TOptional<TEntryType> GetBestConfigEntry(const TCHAR* ConfigEntryName) const
592 {
593 const FString Section = GetVendorSectionName();
594
595 if (Section.IsEmpty())
596 {
597 return {};
598 }
599
600 // Multiple entries can match, but some entries may match on more
601 // constraints (e.g. both RHI and adapter name). Choose the entry
602 // that satisfies the most constraints.
603 TPair<TEntryType, uint32> MostRelevantEntry;
604
605 TArray<FString> EntryStrings;
606 GConfig->GetArray(*Section, ConfigEntryName, EntryStrings, GHardwareIni);
607 for (const FString& EntryString : EntryStrings)
608 {
609 ensure(!EntryString.IsEmpty());
610 TEntryType Entry(*EntryString);
611
612 uint32 NumSatisfiedConstraints = 0;
613 if (Entry.AppliesToDriver(DriverInfo, NumSatisfiedConstraints))
614 {
615 if (NumSatisfiedConstraints >= MostRelevantEntry.Value)
616 {
617 MostRelevantEntry = { Entry, NumSatisfiedConstraints };
618 }
619 }
620 }
621
622 if (MostRelevantEntry.Key.IsValid())
623 {
624 return MostRelevantEntry.Key;
625 }
626 return {};
627 }
