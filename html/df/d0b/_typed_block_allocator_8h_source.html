208template <typename ElementType>
209ElementType* TTypedBlockAllocatorBase<ElementType>::BaseAlloc()
210{
211 ElementType* Result = nullptr;
212 FAllocationBlock* Block = nullptr;
213 for (; NextBlock < Blocks.Num(); ++NextBlock)
214 {
215 if (Blocks[NextBlock].NextIndex < Blocks[NextBlock].Capacity)
216 {
217 Block = &Blocks[NextBlock];
218 break;
219 }
220 }
221
222 if (!Block)
223 {
224 check(NextBlock == Blocks.Num());
225 // Double our NumAllocations with each new block, up to a maximum block size
226 int32 BlockCapacity = FMath::Max(NumAllocations, MinBlockSize);
227 if (MaxBlockSize > MinBlockSize)
228 {
229 BlockCapacity = FMath::Min(BlockCapacity, MaxBlockSize);
230 }
231 Block = &Blocks.Emplace_GetRef(BlockCapacity);
232 }
233
234 Result = Block->Elements[Block->NextIndex].GetTypedPtr();
235 ++Block->NextIndex;
236 ++NumAllocations;
237
238 return Result;
239}

303template <typename ElementType>
304void TTypedBlockAllocatorBase<ElementType>::BaseReserveDelta(int32 AllocationCount,
305 int32 InMaxBlockSize, int32 NumAvailableOnFreeList)
306{
307 int32 DeltaAllocationCount = AllocationCount - NumAvailableOnFreeList;
308 for (const FAllocationBlock& Block : TConstArrayView<FAllocationBlock>(Blocks).RightChop(NextBlock))
309 {
310 DeltaAllocationCount -= Block.Capacity - Block.NextIndex;
311 }
312 if (InMaxBlockSize == 0)
313 {
314 InMaxBlockSize = MaxBlockSize;
315 }
316 if (DeltaAllocationCount <= 0)
317 {
318 return;
319 }
320
321 // Allocate blocks until we have enough capacity to cover the Reservation. As with blocks allocated from
322 // Alloc, set the unclamped capacity high enough to double our number of allocations. But also set it high
323 // enough to cover the remaining count, and use the different MaxBlockSize if passed in.
324 int32 BlockCapacity = FMath::Max(DeltaAllocationCount, NumAllocations);
325 BlockCapacity = FMath::Max(BlockCapacity, MinBlockSize);
326 if (InMaxBlockSize > MinBlockSize)
327 {
328 BlockCapacity = FMath::Min(BlockCapacity, InMaxBlockSize);
329 }
330 Blocks.Emplace(BlockCapacity);
331 DeltaAllocationCount -= BlockCapacity;
332
333 if (DeltaAllocationCount > 0)
334 {
335 // For further blocks, use the MaxBlockSize
336 check(InMaxBlockSize > MinBlockSize); // Otherwise we would have allocated all of DeltaAllocationCount in a single block
337 BlockCapacity = InMaxBlockSize;
338 while (DeltaAllocationCount > 0)
339 {
340 Blocks.Emplace(BlockCapacity);
341 DeltaAllocationCount -= BlockCapacity;
342 }
343 }
344}

437{
438 ElementType* Result = nullptr;
