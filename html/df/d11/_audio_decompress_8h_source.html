571 void DoWork()
572 {
573 LLM_SCOPE(ELLMTag::AudioMisc);
574
575 switch(TaskType)
576 {
577 case ERealtimeAudioTaskType::CompressedInfo:
578 AudioBuffer->ReadCompressedInfo(WaveData);
579 break;
580
581 case ERealtimeAudioTaskType::Decompress:
582 if (bSkipFirstBuffer)
583 {
584#if PLATFORM_ANDROID
585 // Only skip one buffer on Android
586 AudioBuffer->ReadCompressedData((uint8*)AudioData, NumPrecacheFrames, bLoopingMode );
587#else
588 // If we're using cached data we need to skip the first two reads from the data
589 AudioBuffer->ReadCompressedData((uint8*)AudioData, NumPrecacheFrames, bLoopingMode);
590 AudioBuffer->ReadCompressedData((uint8*)AudioData, NumPrecacheFrames, bLoopingMode);
591#endif
592 }
593 bLooped = AudioBuffer->ReadCompressedData((uint8*)AudioData, MONO_PCM_BUFFER_SAMPLES, bLoopingMode);
594 break;
595
596 case ERealtimeAudioTaskType::Procedural:
597 BytesWritten = WaveData->GeneratePCMData((uint8*)AudioData, MaxSamples);
598 break;
599
600 default:
601 check(false);
602 }
603 }
