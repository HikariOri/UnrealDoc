22enum class ESymbolQueryResult : uint8
23{
24 Pending, // Symbol is pending resolution
25 OK, // Symbol has been correctly resolved
26 NotLoaded, // Module debug data could not be loaded or found
27 Mismatch, // Module debug data could not be loaded because debug data did not match traced binary
28 NotFound, // Symbol was not found in module debug data
29 StatusNum
30};

96enum class EModuleStatus : uint8
97{

99 Pending, // Module is pending load
100 Loaded, // Module has been successfully loaded
101 VersionMismatch, // Debug data was found, but did not match traced version

128struct FModule

142 // If status is loaded, contains the path to the debug info. If failure state contains an error message.
143 const TCHAR* StatusMessage;
144 // Statistics about the module
145 struct SymbolStats
146 {
147 std::atomic<uint32> Discovered = 0; // number of symbols discovered in the trace (used symbols)
148 std::atomic<uint32> Cached = 0; // number of symbols loaded from cache
149 std::atomic<uint32> Resolved = 0; // number of symbols successfully resolved from the loaded module
150 std::atomic<uint32> Failed = 0; // number of symbols failed to resolve
151 std::atomic<uint32> Available = 0; // number of symbols available in the loaded module's debug information
152 } Stats;
