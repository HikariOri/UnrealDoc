123FORCEINLINE bool appLineCheckTriangle(const FVector4& Start, const FVector4& End, const FVector4& Dir, const FVector4& V0, const FVector4& V1, const FVector4& V2, const FVector4& Normal, float& IntersectionTime)
124{
125 const FPlane::FReal StartDist = FPlane(Normal).PlaneDot(Start);
126 const FPlane::FReal EndDist = FPlane(Normal).PlaneDot(End);
127
128 // Check if the line is completely on one side of the triangle, or if it's co-planar.
129#if 1
130 if ((StartDist == EndDist) || (StartDist < -0.001f && EndDist < -0.001f) || (StartDist > 0.001f && EndDist > 0.001f))
131#else
132 if ( (StartDist * EndDist) > -0.0001f )
133#endif
134 {
135 return false;
136 }
137
138 // Figure out when it will hit the triangle
139 float Time = float(-StartDist / (EndDist - StartDist)); // LWC_TODO: Precision loss
140
141 // If this triangle is not closer than the previous hit, reject it
142 if (Time < 0.f || Time >= IntersectionTime)
143 {
144 return false;
145 }
146
147 // Calculate the line's point of intersection with the node's plane
148 const FVector4& Intersection = Start + Dir * Time;
149 const FVector4* Verts[3] =
150 {
151 &V0, &V1, &V2
152 };
153
154 // Check if the point of intersection is inside the triangle's edges.
155 for( int32 SideIndex = 0; SideIndex < 3; SideIndex++ )
156 {
157 const FVector4& SideDirection = Normal ^ (*Verts[(SideIndex + 1) % 3] - *Verts[SideIndex]);
158 const FVector4::FReal SideW = Dot3(SideDirection, *Verts[SideIndex]);
159 const FVector4::FReal DotW = Dot3(SideDirection, Intersection);
160 if ((DotW - SideW) >= 0.001f)
161 {
162 return false;
163 }
164 }
165 IntersectionTime = Time;
166 return true;
167}

191static int32 appLineCheckTriangleSOA(const FVector3SOA& Start, const FVector3SOA& End, const FVector3SOA& Dir, const FTriangleSOA& Triangle4, float& InOutIntersectionTime)
192{
193 VectorRegister TriangleMask;
194
195 VectorRegister StartDist;
196 StartDist = VectorMultiplyAdd( Triangle4.Normals.X, Start.X, Triangle4.Normals.W );
197 StartDist = VectorMultiplyAdd( Triangle4.Normals.Y, Start.Y, StartDist );
198 StartDist = VectorMultiplyAdd( Triangle4.Normals.Z, Start.Z, StartDist );
199
200 VectorRegister EndDist;
201 EndDist = VectorMultiplyAdd( Triangle4.Normals.X, End.X, Triangle4.Normals.W );
202 EndDist = VectorMultiplyAdd( Triangle4.Normals.Y, End.Y, EndDist );
203 EndDist = VectorMultiplyAdd( Triangle4.Normals.Z, End.Z, EndDist );
204
205 // Are both end-points of the line on the same side of the triangle (or parallel to the triangle plane)?
206 TriangleMask = VectorCompareLE(VectorMultiply(StartDist, EndDist), GSmallNegativeNumber);
207 if ( VectorMaskBits(TriangleMask) == 0 )
208 {
209 return -1;
210 }
211
212 // Figure out when it will hit the triangle
213 VectorRegister Time = VectorDivide( StartDist, VectorSubtract(StartDist, EndDist) );
214
215 // If this triangle is not closer than the previous hit, reject it
216 VectorRegister IntersectionTime = VectorLoadFloat1( &InOutIntersectionTime );
217 TriangleMask = VectorBitwiseAnd( TriangleMask, VectorCompareGE( Time, VectorZero() ) );
218 TriangleMask = VectorBitwiseAnd( TriangleMask, VectorCompareLT( Time, IntersectionTime ) );
219 if ( VectorMaskBits(TriangleMask) == 0 )
220 {
221 return -1;
222 }
223
224 // Calculate the line's point of intersection with the node's plane
225 const VectorRegister IntersectionX = VectorMultiplyAdd( Dir.X, Time, Start.X );
226 const VectorRegister IntersectionY = VectorMultiplyAdd( Dir.Y, Time, Start.Y );
227 const VectorRegister IntersectionZ = VectorMultiplyAdd( Dir.Z, Time, Start.Z );
228
229 // Check if the point of intersection is inside the triangle's edges.
230 for( int32 SideIndex = 0; SideIndex < 3; SideIndex++ )
231 {
232 const VectorRegister EdgeX = VectorSubtract( Triangle4.Positions[(SideIndex + 1) % 3].X, Triangle4.Positions[SideIndex].X );
233 const VectorRegister EdgeY = VectorSubtract( Triangle4.Positions[(SideIndex + 1) % 3].Y, Triangle4.Positions[SideIndex].Y );
234 const VectorRegister EdgeZ = VectorSubtract( Triangle4.Positions[(SideIndex + 1) % 3].Z, Triangle4.Positions[SideIndex].Z );
235 const VectorRegister SideDirectionX = VectorNegateMultiplyAdd( Triangle4.Normals.Z, EdgeY, VectorMultiply(Triangle4.Normals.Y, EdgeZ) );
236 const VectorRegister SideDirectionY = VectorNegateMultiplyAdd( Triangle4.Normals.X, EdgeZ, VectorMultiply(Triangle4.Normals.Z, EdgeX) );
237 const VectorRegister SideDirectionZ = VectorNegateMultiplyAdd( Triangle4.Normals.Y, EdgeX, VectorMultiply(Triangle4.Normals.X, EdgeY) );
238 VectorRegister SideW;
239 SideW = VectorMultiply( SideDirectionX, Triangle4.Positions[SideIndex].X );
240 SideW = VectorMultiplyAdd( SideDirectionY, Triangle4.Positions[SideIndex].Y, SideW );
241 SideW = VectorMultiplyAdd( SideDirectionZ, Triangle4.Positions[SideIndex].Z, SideW );
242 VectorRegister DotW;
243 DotW = VectorMultiply( SideDirectionX, IntersectionX );
244 DotW = VectorMultiplyAdd( SideDirectionY, IntersectionY, DotW );
245 DotW = VectorMultiplyAdd( SideDirectionZ, IntersectionZ, DotW );
246 TriangleMask = VectorBitwiseAnd( TriangleMask, VectorCompareLT( VectorSubtract(DotW, SideW), GSmallNumber ) );
247 if ( VectorMaskBits(TriangleMask) == 0 )
248 {
249 return -1;
250 }
251 }
252
253 // Set all non-hitting times to 1.0
254 Time = VectorSelect( TriangleMask, Time, VectorOne() );
255
256 // Get the best intersection time out of the 4 possibilities.
257 VectorRegister BestTimes = VectorMin( Time, VectorSwizzle(Time, 2, 3, 0, 0) );
258 BestTimes = VectorMin( BestTimes, VectorSwizzle(BestTimes, 1, 0, 0, 0) );
259 IntersectionTime = VectorReplicate( BestTimes, 0 );
260
261 // Get the triangle index that corresponds to the best time.
262 // NOTE: This will pick the first triangle, in case there are multiple hits at the same spot.
263 int32 SubIndex = VectorMaskBits( VectorCompareEQ( Time, IntersectionTime ) );
264 SubIndex = appCountTrailingZeros( SubIndex );
265
266 // Return results.
267 VectorStoreFloat1( IntersectionTime, &InOutIntersectionTime );
268 return SubIndex;
269}

432 FBox SplitTriangleList(int32 Start,int32 NumTris,
433 TArray<FkDOPBuildCollisionTriangle<KDOP_IDX_TYPE> >& BuildTriangles,
434 kDOPArray<FTriangleSOA>& SOATriangles,
435 kDOPArray<NodeType>& Nodes)
436 {
437 // Figure out if we are a leaf node or not
438 if (NumTris > 4)
439 {
440 // Still too many triangles, so continue subdividing the triangle list
441 bIsLeaf = 0;
442 Occupancy = 0;
443 int32 BestPlane = -1;
444 double BestMean = 0.f;
445 double BestVariance = 0.f;
446
447 // Determine how to split using the splatter algorithm
448 {
449 double Mean[NUM_PLANES] = { 0 };
450 double Variance[NUM_PLANES] = { 0 };
451
452 // Compute the mean for the triangle list
453 for (int32 nTriangle = Start; nTriangle < Start + NumTris; nTriangle++)
454 {
455 // Project the centroid of the triangle against the plane
456 // normals and accumulate to find the total projected
457 // weighting
458 FVector4 Centroid = BuildTriangles[nTriangle].GetCentroid();
459
460 for (int32 nPlane = 0; nPlane < NUM_PLANES; nPlane++)
461 {
462 Mean[nPlane] += Centroid[nPlane];
463 }
464 }
465
466 // Divide by the number of triangles to get the average
467 for (int32 nPlane = 0; nPlane < NUM_PLANES; nPlane++)
468 {
469 Mean[nPlane] /= NumTris;
470 }
471
472 // Compute variance of the triangle list
473 for (int32 nTriangle = Start; nTriangle < Start + NumTris; nTriangle++)
474 {
475 // Project the centroid again
476 FVector4 Centroid = BuildTriangles[nTriangle].GetCentroid();
477
478 // Now calculate the variance and accumulate it
479 for (int32 nPlane = 0; nPlane < NUM_PLANES; nPlane++)
480 {
481 Variance[nPlane] += (Centroid[nPlane] - Mean[nPlane]) * (Centroid[nPlane] - Mean[nPlane]);
482 }
483 }
484
485 // Determine which plane is the best to split on
486 for (int32 nPlane = 0; nPlane < NUM_PLANES; nPlane++)
487 {
488 // Get the average variance
489 Variance[nPlane] /= NumTris;
490 if (Variance[nPlane] >= BestVariance)
491 {
492 BestPlane = nPlane;
493 BestVariance = Variance[nPlane];
494 BestMean = Mean[nPlane];
495 }
496 }
497 }
498
499 // Now that we have the plane to split on, work through the triangle
500 // list placing them on the left or right of the splitting plane
501 int32 Left = Start - 1;
502 int32 Right = Start + NumTris;
503 // Keep working through until the left index passes the right; we test that condition after each interior loop
504 for (;;)
505 {
506 // Loop invariants: (1) Left < Right,
507 // (2) All triangles <= Left belong on the left, all triangles >= Right belong on the right
508 // (3) Left+1 is an untested triangle, Right-1 is an untested triangle
509 FVector4::FReal Dot;
510 // Increment Left until it points to triangle that belongs on the right, or Right==Left
511 for (++Left; Left < Right; ++Left)
512 {
513 Dot = BuildTriangles[Left].GetCentroid()[BestPlane];
514 if (Dot < BestMean)
515 {
516 break;
517 }
518 }
519 if (Left == Right)
520 {
521 break;
522 }
523 // Decrement Right until it points to triangle that belongs on the left, or Right==Left
524 for (--Right; Left < Right; --Right)
525 {
526 Dot = BuildTriangles[Right].GetCentroid()[BestPlane];
527 if (Dot >= BestMean)
528 {
529 break;
530 }
531 }
532 if (Left == Right)
533 {
534 break;
535 }
536 // Left points to a triangle that belongs on the Right; Right points to a triangle that belongs on the Left. Swap them.
537 Swap(BuildTriangles[Left], BuildTriangles[Right]);
538 }
539 // After loop array is partitioned and Left is the first index that belongs on the right side of the plane.
540 // Check for wacky degenerate case where more than GKDOPMaxTrisPerLeaf
541 // fall all in the same kDOP
542 if (Left == Start + NumTris || Right == Start)
543 {
544 Left = Start + (NumTris / 2);
545 }
546 // Add the two child nodes
547 n.LeftNode = Nodes.AddZeroed(2);
548 n.RightNode = n.LeftNode + 1;
549 // Have the left node recursively subdivide it's list and set bounding volume.
550 FBox LeftBoundingVolume = Nodes[n.LeftNode].SplitTriangleList(Start,Left - Start,BuildTriangles,SOATriangles,Nodes);
551 BoundingVolumes.SetBox(0,LeftBoundingVolume);
552 // Set unused index 2,3 to child nodes of left node.
553 BoundingVolumes.SetBox(2,Nodes[n.LeftNode].BoundingVolumes.GetBox(0));
554 BoundingVolumes.SetBox(3,Nodes[n.LeftNode].BoundingVolumes.GetBox(1));
555
556 // And now have the right node recursively subdivide it's list and set bounding volume.
557 FBox RightBoundingVolume = Nodes[n.RightNode].SplitTriangleList(Left,Start + NumTris - Left,BuildTriangles,SOATriangles,Nodes);
558 BoundingVolumes.SetBox(1,RightBoundingVolume);
559
560 // Non-leaf node bounds are the "sum" of the left and right nodes' volumes.
561 return LeftBoundingVolume + RightBoundingVolume;
562 }
563 else
564 {
565 // Build SOA triangles
566
567 // "NULL triangle", used when a leaf can't fill all 4 triangles in a FTriangleSOA.
568 // No line should ever hit these triangles, set the values so that it can never happen.
569 FkDOPBuildCollisionTriangle<KDOP_IDX_TYPE> EmptyTriangle(0,FVector4(0,0,0,0),FVector4(0,0,0,0),FVector4(0,0,0,0));
570
571 t.StartIndex = SOATriangles.Num();
572 t.NumTriangles = Align<int32>(NumTris, 4) / 4;
573 SOATriangles.AddZeroed( t.NumTriangles );
574
575 int32 BuildTriIndex = Start;
576 for ( uint32 SOAIndex=0; SOAIndex < t.NumTriangles; ++SOAIndex )
577 {
578 FkDOPBuildCollisionTriangle<KDOP_IDX_TYPE>* Tris[4] = { &EmptyTriangle, &EmptyTriangle, &EmptyTriangle, &EmptyTriangle };
579 FTriangleSOA& SOA = SOATriangles[t.StartIndex + SOAIndex];
580 int32 SubIndex = 0;
581 for ( ; SubIndex < 4 && BuildTriIndex < (Start+NumTris); ++SubIndex, ++BuildTriIndex )
582 {
583 Tris[SubIndex] = &BuildTriangles[BuildTriIndex];
584 SOA.Payload[SubIndex] = Tris[SubIndex]->MaterialIndex;
585 }
586 for ( ; SubIndex < 4; ++SubIndex )
587 {
588 SOA.Payload[SubIndex] = 0xffffffff;
589 }
590
591 SOA.Positions[0].X = VectorSet( Tris[0]->V0.X, Tris[1]->V0.X, Tris[2]->V0.X, Tris[3]->V0.X );
592 SOA.Positions[0].Y = VectorSet( Tris[0]->V0.Y, Tris[1]->V0.Y, Tris[2]->V0.Y, Tris[3]->V0.Y );
593 SOA.Positions[0].Z = VectorSet( Tris[0]->V0.Z, Tris[1]->V0.Z, Tris[2]->V0.Z, Tris[3]->V0.Z );
594 SOA.Positions[1].X = VectorSet( Tris[0]->V1.X, Tris[1]->V1.X, Tris[2]->V1.X, Tris[3]->V1.X );
595 SOA.Positions[1].Y = VectorSet( Tris[0]->V1.Y, Tris[1]->V1.Y, Tris[2]->V1.Y, Tris[3]->V1.Y );
596 SOA.Positions[1].Z = VectorSet( Tris[0]->V1.Z, Tris[1]->V1.Z, Tris[2]->V1.Z, Tris[3]->V1.Z );
597 SOA.Positions[2].X = VectorSet( Tris[0]->V2.X, Tris[1]->V2.X, Tris[2]->V2.X, Tris[3]->V2.X );
598 SOA.Positions[2].Y = VectorSet( Tris[0]->V2.Y, Tris[1]->V2.Y, Tris[2]->V2.Y, Tris[3]->V2.Y );
599 SOA.Positions[2].Z = VectorSet( Tris[0]->V2.Z, Tris[1]->V2.Z, Tris[2]->V2.Z, Tris[3]->V2.Z );
600
601 const FVector4& Tris0LocalNormal = Tris[0]->GetLocalNormal();
602 const FVector4& Tris1LocalNormal = Tris[1]->GetLocalNormal();
603 const FVector4& Tris2LocalNormal = Tris[2]->GetLocalNormal();
604 const FVector4& Tris3LocalNormal = Tris[3]->GetLocalNormal();
605
606 SOA.Normals.X = VectorSet( Tris0LocalNormal.X, Tris1LocalNormal.X, Tris2LocalNormal.X, Tris3LocalNormal.X );
607 SOA.Normals.Y = VectorSet( Tris0LocalNormal.Y, Tris1LocalNormal.Y, Tris2LocalNormal.Y, Tris3LocalNormal.Y );
608 SOA.Normals.Z = VectorSet( Tris0LocalNormal.Z, Tris1LocalNormal.Z, Tris2LocalNormal.Z, Tris3LocalNormal.Z );
609 SOA.Normals.W = VectorSet( -Tris0LocalNormal.W, -Tris1LocalNormal.W, -Tris2LocalNormal.W, -Tris3LocalNormal.W );
610 }
611
612 // No need to subdivide further so make this a leaf node
613 bIsLeaf = 1;
614
615 check(NumTris <= std::numeric_limits<uint8>::max());
616 Occupancy = static_cast<uint8>(NumTris);
617
618 // Generate bounding volume for leaf which is passed up the call chain.
619 FBox BoundingVolume(ForceInit);
620 for (int32 TriangleIndex=Start; TriangleIndex<Start + NumTris; TriangleIndex++)
621 {
622 BoundingVolume += FVector(BuildTriangles[TriangleIndex].V0);
623 BoundingVolume += FVector(BuildTriangles[TriangleIndex].V1);
624 BoundingVolume += FVector(BuildTriangles[TriangleIndex].V2);
625 }
626 BoundingVolumes.SetBox(0,BoundingVolume);
627 BoundingVolumes.SetBox(1,BoundingVolume);
628 BoundingVolumes.SetBox(2,BoundingVolume);
629 BoundingVolumes.SetBox(3,BoundingVolume);
630
631 return BoundingVolume;
632 }
