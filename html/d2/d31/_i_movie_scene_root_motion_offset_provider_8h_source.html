10// Modular feature for systems that apply root motion to scene components.
11// Allows the transform track editor to subtract root motion when auto-keying,
12// preventing double-application of root motion offsets.
13class IMovieSceneRootMotionOffsetProvider : public IModularFeature
14{
15public:
16 virtual ~IMovieSceneRootMotionOffsetProvider() = default;
17
18 // Returns the root motion offset currently applied to the given object,
19 // or Identity if this provider doesn't handle the object.
20 virtual FTransform GetRootMotionOffset(const UMovieSceneEntitySystemLinker* Linker, UObject* AnimatedObject) const = 0;
21
22 // Returns the root motion offset for a specific editing context (e.g. a UControlRig
23 // on a particular mixer layer). When EditingContext is non-null, the provider may
24 // return a per-layer offset rather than the total. Default delegates to the 2-arg version.
25 virtual FTransform GetRootMotionOffset(const UMovieSceneEntitySystemLinker* Linker, UObject* AnimatedObject, const UObject* EditingContext) const
26 {
27 return GetRootMotionOffset(Linker, AnimatedObject);
28 }
29
30 // Recalculates the initial actor transform baseline after the user manually
31 // changes the transform.
32 virtual void UpdateRootMotionOffset(UMovieSceneEntitySystemLinker* Linker, UObject* AnimatedObject) {}
33
34 static MOVIESCENETOOLS_API FName GetModularFeatureName();
35};
