243 // Spawns an intermediate node associated with the source node (for error purposes)
244 template <typename NodeType>
245 NodeType* SpawnIntermediateNode(UEdGraphNode* SourceNode, UEdGraph* ParentGraph = nullptr)
246 {
247 if (ParentGraph == nullptr)
248 {
249 ParentGraph = SourceNode->GetGraph();
250 }
251
252 NodeType* Result = ParentGraph->CreateIntermediateNode<NodeType>();
253 MessageLog.NotifyIntermediateObjectCreation(Result, SourceNode); // this might be useful to track back function entry nodes to events.
254 Result->CreateDeterministicGuid();
255
256 AutoAssignNodePosition(Result);
257
258 return Result;
259 }
