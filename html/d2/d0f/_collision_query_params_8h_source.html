404 if (ensure ((int32)NewChannel < MaxCollisionChannelCount))

423 }

431 }
432 }

434 void SetAllObjectsQueryFlag(int64 NewQueryFlag)
435 {
436 // if all object query has changed, make sure to apply to static object query
437 AllObjectQueryFlag = NewQueryFlag;
438 AllStaticObjectQueryFlag = AllObjectQueryFlag & AllStaticObjectQueryFlag;
439 }
