77static uint32 Ssl_ContextProvision(SSL_CTX* Context)
78{
79 uint32 NumAdded = 0;
80
81#if PLATFORM_WINDOWS
82 HANDLE StoreHandle = CertOpenSystemStoreW(0, L"ROOT");
83 for (PCCERT_CONTEXT CertContext = nullptr;;)
84 {
85 CertContext = CertEnumCertificatesInStore(StoreHandle, CertContext);
86 if (CertContext == nullptr)
87 {
88 break;
89 }
90
91 if ((CertContext->dwCertEncodingType & X509_ASN_ENCODING) == 0)
92 {
93 continue;
94 }
95
96 const uint8* CertData = (uint8*)(CertContext->pbCertEncoded);
97 uint32 CertSize = CertContext->cbCertEncoded;
98 X509* FiveOhNine = d2i_X509(nullptr, &CertData, CertSize);
99 if (FiveOhNine == nullptr)
100 {
101 continue;
102 }
103
104 X509_STORE* Store = SSL_CTX_get_cert_store(Context);
105 int32 Result = X509_STORE_add_cert(Store, FiveOhNine);
106 NumAdded += (Result == 1);
107
108 X509_free(FiveOhNine);
109 }
110
111 CertCloseStore(StoreHandle, 0);
112#endif // PLATFORM_WINDOWS
113
114 return NumAdded;
115}

118static uint32 Ssl_ContextProvision(SSL_CTX* Context, FMemoryView PemData)
119{
120 const void* Data = PemData.GetData();
121 uint32 Size = uint32(PemData.GetSize());
122 BIO* Bio = BIO_new_mem_buf(Data, Size);
123
124 uint32 NumAdded = 0;
125 while (true)
126 {
127 X509* FiveOhNine = PEM_read_bio_X509(Bio, nullptr, 0, nullptr);
128 if (FiveOhNine == nullptr)
129 {
130 break;
131 }
132
133 X509_STORE* Store = SSL_CTX_get_cert_store(Context);
134 int32 Result = X509_STORE_add_cert(Store, FiveOhNine);
135 NumAdded += (Result == 1);
136
137 X509_free(FiveOhNine);
138 }
139
140 BIO_free(Bio);
141 return NumAdded;
142}

145static UPTRINT Ssl_ContextCreate(FMemoryView PemData)
146{
147 if (static bool InitOnce = false; !InitOnce)
148 {
149 // While OpenSSL will lazily initialise itself, the defaults used will fail
150 // initialisation on some platforms. So we have a go here. We do not register
151 // anything for clean-up as we do not know if anyone else has done so.
152 uint64 InitOpts = OPENSSL_INIT_NO_ATEXIT;
153 OPENSSL_init_ssl(InitOpts, nullptr);
154 InitOnce = true;
155 }
156
157 auto* Method = TLS_client_method();
158 SSL_CTX* Context = SSL_CTX_new(Method);
159 checkf(Context != nullptr, TEXT("ERR_get_error() == %ld"), ERR_get_error());
160
161 SSL_CTX_set_options(Context, SSL_OP_NO_SSLv2|SSL_OP_NO_SSLv3);
162
163 uint32 NumAdded;
164 if (!PemData.IsEmpty())
165 {
166 NumAdded = Ssl_ContextProvision(Context, PemData);
167 }
168 else
169 {
170 NumAdded = Ssl_ContextProvision(Context);
171 }
172
173 if (NumAdded == 0)
174 {
175 SSL_CTX_free(Context);
176 return 0;
177 }
178
179#if defined(IAS_HTTP_EXPLICIT_VERIFY_TIME)
180 if (X509_VERIFY_PARAM* VerifyParam = SSL_CTX_get0_param(Context); VerifyParam != nullptr)
181 {
182 int32 AliasTown;
183 std::tm Utc = {};
184 FPlatformTime::UtcTime(
185 Utc.tm_year, Utc.tm_mon,
186 AliasTown,
187 Utc.tm_mday, Utc.tm_hour, Utc.tm_min,
188 AliasTown, AliasTown
189 );
190
191 Utc.tm_year -= 1900;
192 Utc.tm_mon -= 1;
193
194 time_t Now = std::mktime(&Utc);
195
196 X509_VERIFY_PARAM_set_time(VerifyParam, Now);
197 }
198#endif
199
200 return UPTRINT(Context);
201}

213static int32 Ssl_BioWrite(BIO* Bio, const char* Data, size_t Size, size_t* BytesWritten)
214{
215 *BytesWritten = 0;
216 BIO_clear_retry_flags(Bio);
217
218 auto* Peer = (FPeer*)BIO_get_data(Bio);
219 FOutcome Outcome = Peer->Send(Data, int32(Size));
220 if (Outcome.IsWaiting())

265static SSL* Ssl_Create(FCertRootsRef Certs, const char* HostName=nullptr)
266{
267 static_assert(OPENSSL_VERSION_NUMBER >= 0x10100000L, "version supporting autoinit required");
268
269 if (Certs == ECertRootsRefType::Default)
270 {
271 Certs = FCertRoots::Explicit(GDefaultCertRoots);
272 check(Certs != 0);
273 }
274 auto* Context = (SSL_CTX*)Certs;
275
276 static BIO_METHOD* BioMethod = nullptr;
277 if (BioMethod == nullptr)
278 {
279 int32 BioId = BIO_get_new_index() | BIO_TYPE_SOURCE_SINK;
280 BioMethod = BIO_meth_new(BioId, "IasBIO");
281 BIO_meth_set_write_ex(BioMethod, Ssl_BioWrite);
282 BIO_meth_set_read_ex(BioMethod, Ssl_BioRead);
283 BIO_meth_set_ctrl(BioMethod, Ssl_BioControl);
284 }
285
286 BIO* Bio = BIO_new(BioMethod);
287
288 // SSL_MODE_ENABLE_PARTIAL_WRITE ??!!!
289
290 SSL* Ssl = SSL_new(Context);
291 SSL_set_connect_state(Ssl);
292 SSL_set0_rbio(Ssl, Bio);
293 SSL_set0_wbio(Ssl, Bio);
294 BIO_up_ref(Bio);
295
296 if (HostName != nullptr)
297 {
298 SSL_set_tlsext_host_name(Ssl, HostName);
299 }
300
301 return Ssl;
302}

319static void Ssl_SetupAlpn(SSL* Ssl, int32 HttpVersion)
320{
321 FAnsiStringView AlpnProtos;
322 switch (HttpVersion)
323 {
324 case 1: AlpnProtos = "\x08" "http/1.1"; break;
325 case 2: AlpnProtos = "\x02" "h2"; break;
326 default: break;
327 }
328
329 check(!AlpnProtos.IsEmpty());
330 SSL_set_alpn_protos(Ssl, (uint8*)(AlpnProtos.GetData()), int32(AlpnProtos.Len()));
331}

334static int32 Ssl_GetProtocolVersion(SSL* Ssl)
335{
336 int32 Proto = 1;
337
338 const char* AlpnProto = nullptr;
339 uint32 AlpnProtoLen;
340 SSL_get0_alpn_selected(Ssl, &(const uint8*&)AlpnProto, &AlpnProtoLen);
341 if (AlpnProto == nullptr)
342 {
343 return Proto;
344 }
345
346 FAnsiStringView Needle(AlpnProto, AlpnProtoLen);
347 FAnsiStringView Candidates[] = {
348 "http/1.1",
349 "h2",
350 };
351 for (uint32 i = 0; i < UE_ARRAY_COUNT(Candidates); ++i)
352 {
353 const FAnsiStringView& Candidate = Candidates[i];
354 if (AlpnProtoLen != uint32(Candidate.Len()))
355 {
356 continue;
357 }
358
359 if (Candidate != Needle)
360 {
361 continue;
362 }
363
364 Proto = i + 1;
365 break;
366 }
367
368 return Proto;
369}

372static FOutcome Ssl_GetOutcome(SSL* Ssl, int32 SslResult, const char* Message="tls error")
373{
374 int32 Error = SSL_get_error(Ssl, SslResult);
375 if (Error != SSL_ERROR_WANT_READ && Error != SSL_ERROR_WANT_WRITE)
376 {
377 return FOutcome::Error(Message, Error);
378 }
379 return FOutcome::Waiting();
380}

383static FOutcome Ssl_Handshake(SSL* Ssl)
384{
385 int32 Result = SSL_do_handshake(Ssl);
386 if (Result == 0) return FOutcome::Error("unsuccessful tls handshake");
387 if (Result != 1) return Ssl_GetOutcome(Ssl, Result, "tls handshake error");
388
389 Result = SSL_get_verify_result(Ssl);
390 if (Result == X509_V_OK)
391 {
392 return FOutcome::Ok();
393 }
394
395 const char* Msg;
396 switch (Result)
397 {
398 case X509_V_ERR_CERT_HAS_EXPIRED: Msg = "certificate has expired (check local date)"; break;
399 case X509_V_ERR_CERT_NOT_YET_VALID: Msg = "certificate not yet valid (check local date)"; break;
400 case X509_V_ERR_DEPTH_ZERO_SELF_SIGNED_CERT:Msg = "self-signed certificate"; break;
401 case X509_V_ERR_SELF_SIGNED_CERT_IN_CHAIN: Msg = "self-signed certificate in chain"; break;
402 default: Msg = "x509 verification error"; break;
403 }
404 return FOutcome::Error(Msg, Result);
405}

670FOutcome FHttpPeer::Handshake()
671{
672 if (Ssl != nullptr)
673 {
674 if (FOutcome Outcome = FTlsPeer::Handshake(); !Outcome.IsOk())
675 {

680 if (Negotiated != 1 && Negotiated != 2)

684 Proto = uint8(Negotiated);

686 /* intentional fallthrough */
687 }

689 FOutcome Outcome = FOutcome::None();
690 switch (Proto)

692 case 1: Outcome = HandshakeOne(*this, PeerData); break;
693 case 2: Outcome = HandshakeTwo(*this, PeerData); break;
694 default: return FOutcome::Error("Unsupported");

697 if (!Outcome.IsOk())

699 return Outcome;
