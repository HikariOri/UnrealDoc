281 TArray<FTextureStage> Textures;
282 TArray<FOpenGLSamplerState*> SamplerStates;

292
293 // NumCombinedTextures must be greater than or equal to FOpenGL::GetMaxCombinedTextureImageUnits()
294 // NumCombinedUAVUnits must be greater than or equal to FOpenGL::GetMaxCombinedUAVUnits()
295 virtual void InitializeResources(int32 NumCombinedTextures, int32 NumCombinedUAVUnits)
296 {
297 check(NumCombinedTextures >= FOpenGL::GetMaxCombinedTextureImageUnits());
298 check(NumCombinedUAVUnits >= FOpenGL::GetMaxCombinedUAVUnits());
299 check(Textures.IsEmpty() && SamplerStates.IsEmpty() && UAVs.Num() == 0);
300 Textures.SetNum(NumCombinedTextures);
301 SamplerStates.SetNumZeroed(NumCombinedTextures);
302
303 UAVs.Reserve(NumCombinedUAVUnits);
304 UAVs.AddDefaulted(NumCombinedUAVUnits);
305 }
306
307 virtual void CleanupResources()
308 {
309 SamplerStates.Empty();
310 Textures.Empty();
311 UAVs.Empty();
312 }
