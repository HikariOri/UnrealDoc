12// Simple locking circular buffer
13// Uses 32bit wrap around logic explained here:
14// https://www.snellman.net/blog/archive/2016-12-13-ring-buffers/
15class FLockingCircularSampleBuffer
16{
17public:
18 FLockingCircularSampleBuffer(int32 InInitialCapacity)
19 {
20 SetCapacity(InInitialCapacity);
21 }
22
23 // Allow the caller to lock the CS ahead of calling (if necessary)
24 FCriticalSection& GetCriticialSection() { return CS; }
25 const FCriticalSection& GetCriticialSection() const { return CS; }
26
27 // This doesn't preserve contents.
28 void SetCapacity(int32 InCapacity)
29 {
30 check(InCapacity > 1);
31
32 if (!FMath::IsPowerOfTwo(InCapacity))
33 {
34 InCapacity = FMath::RoundUpToPowerOfTwo(InCapacity);
35 }
36
37 FScopeLock Lock(&CS);
38 Buffer.SetNumZeroed(InCapacity, EAllowShrinking::Yes);
39 Mask = InCapacity - 1;
40 Read = 0; // Empty.
41 Write = 0;
42
43 check(Mask != 0);
44 }
45 int32 GetCapacity() const
46 {
47 FScopeLock Lock(&CS);
48 return Buffer.Num();
49 }
50 int32 Num() const
51 {
52 FScopeLock Lock(&CS);
53 return Write - Read;
54 }
55
56 // Get number of samples that can be pushed onto the buffer before it is full.
57 int32 Remainder() const
58 {
59 FScopeLock Lock(&CS);
60 return Buffer.Num() - Num();
61 }
62
63 int32 Push(const float* InBuffer, int32 InSize)
64 {
65 int32 CanPush = FMath::Min(Remainder(), InSize);
66 FScopeLock Lock(&CS);
67 for (int32 i = 0; i < CanPush; ++i)
68 {
69 Enqueue(InBuffer[i]);
70 }
71 return CanPush;
72 }
73 int32 Pop(float* OutBuffer, int32 InNumSamples)
74 {
75 int32 CanPop = FMath::Min(Num(), InNumSamples);
76 FScopeLock Lock(&CS);
77 for (int32 i = 0; i < CanPop; ++i)
78 {
79 OutBuffer[i] = Dequeue();
80 }
81 return CanPop;
82 }
83
84 int32 PushZeros(int32 InNumSamplesOfSilence)
85 {
86 int32 CanPush = FMath::Min(Remainder(), InNumSamplesOfSilence);
87 FScopeLock Lock(&CS);
88 for (int32 i = 0; i < CanPush; ++i)
89 {
90 Enqueue(0.f);
91 }
92 return CanPush;
93 }
94
95 // Reset read/write pointers, discarding all buffered data without reallocating.
96 void Flush()
97 {
98 FScopeLock Lock(&CS);
99 Read = 0;
100 Write = 0;
101 }
102
103 // Grow the buffer to at least InMinCapacity if current capacity is insufficient. Never shrinks.
104 // Flushes read/write pointers (does not preserve contents).
105 void EnsureCapacity(int32 InMinCapacity)
106 {
107 check(InMinCapacity > 1);
108
109 if (!FMath::IsPowerOfTwo(InMinCapacity))
110 {
111 InMinCapacity = FMath::RoundUpToPowerOfTwo(InMinCapacity);
112 }
113
114 FScopeLock Lock(&CS);
115 if (InMinCapacity > Buffer.Num())
116 {
117 Buffer.SetNumZeroed(InMinCapacity, EAllowShrinking::No);
118 Mask = InMinCapacity - 1;
119 }
120 Read = 0;
121 Write = 0;
122 }
123
124private:
125 uint32 Read = 0; // These grow indefinitely until wrap at 2^32,
126 uint32 Write = 0; // this allows us to use full capacity as write >= read.
127 uint32 Mask = 0;
128 TArray<float> Buffer;
129 mutable FCriticalSection CS;
130
131 // NOTE: Not forceinline as compiler does better job without.
132 void Enqueue(const float InFloat)
133 {
134 Buffer[Write++ & Mask] = InFloat;
135 }
136 float Dequeue()
137 {
138 return Buffer[Read++ & Mask];
139 }
140};
