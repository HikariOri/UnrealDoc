1209 void RemoveAtImpl(int32_t Index, int32_t Count, bool bAllowShrinking)
1210 {
1211 if (Count)
1212 {
1213 CheckInvariants();
1214 ULANG_ASSERTF((Count >= 0) & (Index >= 0) & (Index + Count <= _ArrayNum), "Cannot remove elements from array due to invalid parameters.");
1215
1216 DestructElements(GetData() + Index, Count);
1217
1218 // Skip memmove in the common case that there is nothing to move.
1219 int32_t NumToMove = _ArrayNum - Index - Count;
1220 if (NumToMove)
1221 {
1222 memmove
1223 (
1224 (uint8_t*)_ElementStorage.GetAllocation() + (Index)* sizeof(ElementType),
1225 (uint8_t*)_ElementStorage.GetAllocation() + (Index + Count) * sizeof(ElementType),
1226 NumToMove * sizeof(ElementType)
1227 );
1228 }
1229 _ArrayNum -= Count;
1230
1231 if (bAllowShrinking)
1232 {
1233 ResizeShrink();
1234 }
1235 }
1236 }
