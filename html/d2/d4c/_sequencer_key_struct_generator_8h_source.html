200template<typename ChannelType>
201void PostConstructKeyInstance(const TMovieSceneChannelHandle<ChannelType>& ChannelHandle, FKeyHandle InHandle, FStructOnScope* Struct)
202{
203 ChannelType* Channel = ChannelHandle.Get();
204 if (Channel)
205 {
206 const UMovieSceneKeyStructType* GeneratedStructType = CastChecked<const UMovieSceneKeyStructType>(Struct->GetStruct());
207 void* StructPtr = Struct->GetStructMemory();
208
209 const int32 InitialKeyIndex = Channel->GetData().GetIndex(InHandle);
210
211 // Copy the initial value into the struct
212 if (InitialKeyIndex != INDEX_NONE)
213 {
214 const uint8* SrcValueData = GeneratedStructType->SourceValuesProperty->ContainerPtrToValuePtr<uint8>(Channel);
215 uint8* DestValueData = GeneratedStructType->DestValueProperty->ContainerPtrToValuePtr<uint8>(StructPtr);
216
217 FScriptArrayHelper SourceValuesArray(GeneratedStructType->SourceValuesProperty.Get(), SrcValueData);
218 GeneratedStructType->SourceValuesProperty->Inner->CopyCompleteValue(DestValueData, SourceValuesArray.GetRawPtr(InitialKeyIndex));
219 }

223template<typename ChannelType>
224TSharedPtr<FStructOnScope> FSequencerKeyStructGenerator::CreateKeyStructInstance(const TMovieSceneChannelHandle<ChannelType>& ChannelHandle, FKeyHandle InHandle)
225{
226 ChannelType* Channel = ChannelHandle.Get();
227 if (Channel)
228 {
229 const int32 KeyIndex = Channel->GetData().GetIndex(InHandle);
230 if (KeyIndex != INDEX_NONE)
231 {
232 UMovieSceneKeyStructType* GeneratedStructType = InstanceGeneratedStruct(Channel, this);
233 if (!GeneratedStructType)
234 {
235 GeneratedStructType = DefaultInstanceGeneratedStruct(ChannelType::StaticStruct());
236 }
237
238 if (GeneratedStructType && GeneratedStructType->IsComplete())
239 {
240 TSharedPtr<FStructOnScope> StructInstance = CreateInitialStructInstance(Channel, GeneratedStructType, KeyIndex);
241
242 auto CopyInstanceToKeyLambda = [ChannelHandle, InHandle, StructPtr = StructInstance.Get()](const FPropertyChangedEvent&)
243 {
244 FSequencerKeyStructGenerator::CopyInstanceToKey(ChannelHandle, InHandle, StructPtr);
245 };
246
247 FGeneratedMovieSceneKeyStruct* KeyStruct = reinterpret_cast<FGeneratedMovieSceneKeyStruct*>(StructInstance->GetStructMemory());
248 KeyStruct->OnPropertyChangedEvent = CopyInstanceToKeyLambda;
249
250 PostConstructKeyInstance(ChannelHandle, InHandle, StructInstance.Get());
251
252 return StructInstance;
253 }
254 }
255 }

257 return nullptr;

260template<typename ChannelType>
261void FSequencerKeyStructGenerator::CopyInstanceToKey(const TMovieSceneChannelHandle<ChannelType>& DestChannelHandle, FKeyHandle DestHandle, FStructOnScope* SourceInstance)
262{
263 ChannelType* DestinationChannel = DestChannelHandle.Get();
264 if (DestinationChannel)
265 {
266 const int32 KeyIndex = DestinationChannel->GetData().GetIndex(DestHandle);
267 if (KeyIndex != INDEX_NONE)
268 {
269 const UMovieSceneKeyStructType* GeneratedStructType = CastChecked<UMovieSceneKeyStructType>(SourceInstance->GetStruct());
270
271 // Copy the new value back into the channel
272 {
273 uint8* DestValueData = GeneratedStructType->SourceValuesProperty->ContainerPtrToValuePtr<uint8>(DestinationChannel);
274 const uint8* SrcValueData = GeneratedStructType->DestValueProperty->ContainerPtrToValuePtr<uint8>(SourceInstance->GetStructMemory());
275
276 FScriptArrayHelper SourceValuesArray(GeneratedStructType->SourceValuesProperty.Get(), DestValueData);
277 GeneratedStructType->DestValueProperty->CopyCompleteValue(SourceValuesArray.GetRawPtr(KeyIndex), SrcValueData);
278 }
279
280 // Set the new key time
281 {
282 FFrameNumber SrcTimeData = *GeneratedStructType->DestTimeProperty->ContainerPtrToValuePtr<FFrameNumber>(SourceInstance->GetStructMemory());
283 DestinationChannel->SetKeyTime(DestHandle, SrcTimeData);
284 }
285 }
