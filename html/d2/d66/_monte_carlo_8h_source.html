18FORCEINLINE FVector3f UniformSampleSphere( FVector2f E )
19{
20 float CosPhi = 1.0f - 2.0f * E.X;
21 float SinPhi = FMath::Sqrt( 1.0f - CosPhi * CosPhi );
22
23 float Theta = 2.0f * PI * E.Y;
24 float SinTheta, CosTheta;
25 FMath::SinCos( &SinTheta, &CosTheta, Theta );
26
27 return FVector3f(
28 SinPhi * CosTheta,
29 SinPhi * SinTheta,

34FORCEINLINE FVector2f GaussianSampleDisk( FVector2f E, float Sigma, float Window )
35{
36 // Scale distribution to set non-unit variance
37 // Variance = Sigma^2
38
39 // Window to [-Window, Window] output
40 // Without windowing we could generate samples far away on the infinite tails.
41 float InWindow = FMath::Exp( -0.5f * FMath::Square( Window / Sigma ) );
42
43 // Box-Muller transform
44 float Radius = Sigma * FMath::Sqrt( -2.0f * FMath::Loge( (1.0f - E.X) * InWindow + E.X ) );
45
46 float Theta = 2.0f * PI * E.Y;
47 float SinTheta, CosTheta;
48 FMath::SinCos( &SinTheta, &CosTheta, Theta );
49 return Radius * FVector2f( CosTheta, SinTheta );
50}

53FORCEINLINE uint32 EvolveSobolSeed( uint32& Seed )
54{
55 // constant from: https://www.pcg-random.org/posts/does-it-beat-the-minimal-standard.html
56 const uint32 MCG_C = 2739110765;
57 Seed += MCG_C;
58
59 // Generated using https://github.com/skeeto/hash-prospector
60 // Estimated Bias ~583
61 uint32 Hash = Seed;
62 Hash *= 0x92955555u;
63 Hash ^= Hash >> 15;
64 return Hash;
65}

67FORCEINLINE FVector4f LatticeSampler( uint32 SampleIndex, uint32& Seed )
68{
69 // Same as FastOwenScrambling, but without the final reversebits
70 uint32 LatticeIndex = SampleIndex + EvolveSobolSeed( Seed );
71 LatticeIndex ^= LatticeIndex * 0x9c117646u;
72 LatticeIndex ^= LatticeIndex * 0xe0705d72u;
73
74 // Lattice parameters taken from:
75 // Weighted compound integration rules with higher order convergence for all N
76 // Fred J. Hickernell, Peter Kritzer, Frances Y. Kuo, Dirk Nuyens
77 // Numerical Algorithms - February 2012
78 FUintVector4 Result = LatticeIndex * FUintVector4( 1, 364981, 245389, 97823 );
79
80 return (Result >> 8) * 5.96046447754e-08f; // * 2^-24
81}

83FORCEINLINE uint32 FastOwenScrambling( uint32 Index, uint32 Seed )
84{
85 // Laine and Karras / Stratified Sampling for Stochastic Transparency / EGSR 2011
86 Index += Seed; // randomize the index by our seed (pushes bits toward the left)
87 Index ^= Index * 0x9c117646u;
88 Index ^= Index * 0xe0705d72u;
89 return ReverseBits( Index );
90}

92FORCEINLINE FVector2f SobolSampler( uint32 SampleIndex, uint32& Seed )
93{
94 // first scramble the index to decorelate from other 4-tuples
95 uint32 SobolIndex = FastOwenScrambling( SampleIndex, EvolveSobolSeed( Seed ) );
96 // now get Sobol' point from this index
97 FUintVector2 Result( SobolIndex );
98 // y component can be computed without iteration
99 // "An Implementation Algorithm of 2D Sobol Sequence Fast, Elegant, and Compact"
100 // Abdalla Ahmed, EGSR 2024
101 // See listing (19) in the paper
102 // The code is different here because we want the output to be bit-reversed, but
103 // the methodology is the same
104 Result.Y ^= Result.Y >> 16;
105 Result.Y ^= (Result.Y & 0xFF00FF00) >> 8;
106 Result.Y ^= (Result.Y & 0xF0F0F0F0) >> 4;
107 Result.Y ^= (Result.Y & 0xCCCCCCCC) >> 2;
108 Result.Y ^= (Result.Y & 0xAAAAAAAA) >> 1;
109
110 // finally scramble the points to avoid structured artifacts
111 Result.X = FastOwenScrambling( Result.X, EvolveSobolSeed( Seed ) );
112 Result.Y = FastOwenScrambling( Result.Y, EvolveSobolSeed( Seed ) );
113
114 // output as float in [0,1) taking care not to skew the distribution
115 // due to the non-uniform spacing of floats in this range
116 return (Result >> 8) * 5.96046447754e-08f; // * 2^-24
117}
