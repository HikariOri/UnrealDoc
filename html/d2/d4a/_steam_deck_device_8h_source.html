51 static TArray<ITargetDevicePtr> DiscoverDevices(const ITargetPlatformControls& TargetPlatformControls, const TCHAR* RuntimeOSName)
52 {
53 TArray<FString> EngineIniSteamDeckDevices;
54
55 // Expected ini format: +SteamDeckDevice=(IpAddr=10.1.33.19,Name=MySteamDeck,UserName=deck,Password=password)
56 GConfig->GetArray(TEXT("SteamDeck"), TEXT("SteamDeckDevice"), EngineIniSteamDeckDevices, GEngineIni);
57
58 TArray<ITargetDevicePtr> SteamDevices;
59 for (const FString& Device : EngineIniSteamDeckDevices)
60 {
61 FString IpAddr;
62 FString Name;
63 FString ConfigUserName;
64 FString ConfigPassword;
65
66 // As of SteamOS version 3.1, "deck" is the required UserName to be used when making a remote connection
67 // to the device. Eventually it will be configurable, so we will allow users to set it via the config.
68 static const FString DefaultUserName = TEXT("deck");
69
70 if (!FParse::Value(*Device, TEXT("IpAddr="), IpAddr))
71 {
72 UE_LOGF(LogHAL, Error, "You must specify the 'IpAddr' field to connect to a SteamDeck!");
73 continue;
74 }
75
76 // Name is not required, if not set use the IpAddr. This is what is displayed in the Editor
77 if (!FParse::Value(*Device, TEXT("Name="), Name))
78 {
79 static const FString DefaultNamePrefix = "[SteamDeck] ";
80 Name = DefaultNamePrefix + IpAddr;
81 }
82
83 // If the user has not specified a UserName in the config, fallback to the default user name for the SteamDeck
84 if (!FParse::Value(*Device, TEXT("UserName="), ConfigUserName))
85 {
86 ConfigUserName = DefaultUserName;
87 }
88
89 // If the user has not specified a Password in the config, use an rsa key that is part of the SteamOS Devkit Client
90 if (!FParse::Value(*Device, TEXT("Password="), ConfigPassword))
91 {
92 ConfigPassword = FString();
93 }
94
95 SteamDevices.Add(MakeShareable(new TSteamDeckDevice<ParentDeviceClass>(IpAddr, Name, ConfigUserName, ConfigPassword, TargetPlatformControls, RuntimeOSName)));
96 }
97
98 return SteamDevices;
99 }
100
101private:
102 FString IpAddr;
103 FString DeviceName;
104 FString UserName;
105 FString Password;
106 FString RuntimeOSName;
107};
