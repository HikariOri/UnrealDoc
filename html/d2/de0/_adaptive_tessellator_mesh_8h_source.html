49 // create the topology. directed edges will map to their opposite directed edge in opposite direction
50 if (!bTriangleSoup)
51 {
52 struct FSortEdge
53 {
54 size_t EdgeIndex;
55 int32 VtxA, VtxB;
56
57 FORCEINLINE bool operator<(const FSortEdge& other) const
58 {
59 return VtxA < other.VtxA || (VtxA == other.VtxA && VtxB < other.VtxB);
60 }
61 };
62
63 TArray<FSortEdge> SortedHalfEdges;
64 SortedHalfEdges.Reserve(AdjEdges.Num());
65
66 const size_t NumEdges = HalfEdgeCount();
67 for( size_t i = 0; i < NumEdges; ++i )
68 {
69 const int32 V0 = Indexes[i];
70 const int32 V1 = Indexes[Cycle3(i)];
71 SortedHalfEdges.Emplace( i, V0 < V1 ? V0 : V1, V0 < V1 ? V1 : V0 );
72 }
73 SortedHalfEdges.Sort();
74
75 for ( size_t i = 0; i < NumEdges-1; ++i)
76 {
77 if (SortedHalfEdges[i].VtxA == SortedHalfEdges[i+1].VtxA && SortedHalfEdges[i].VtxB == SortedHalfEdges[i+1].VtxB)
78 {
79 AdjEdges[SortedHalfEdges[i ].EdgeIndex] = SortedHalfEdges[i+1].EdgeIndex;
80 AdjEdges[SortedHalfEdges[i+1].EdgeIndex] = SortedHalfEdges[i ].EdgeIndex;
81 ++i;
82 }
83 }
84 }
85 else
86 {
87 FEdgeHash EdgeHash( HalfEdgeCount() );
88 for( int32 EdgeIndex = 0; EdgeIndex < HalfEdgeCount(); EdgeIndex++ )
89 {
90 EdgeHash.ForAllMatching( EdgeIndex, true,
91 [ this ]( int32 CornerIndex )
92 {
93 return Verts[ Indexes[ CornerIndex ] ].Position;
94 },
95 [&]( int32 EdgeIndex0, int32 EdgeIndex1 )
96 {
97 if( AdjEdges[ EdgeIndex0 ] < 0 &&
98 AdjEdges[ EdgeIndex1 ] < 0 )
99 {
100 AdjEdges[ EdgeIndex0 ] = EdgeIndex1;
101 AdjEdges[ EdgeIndex1 ] = EdgeIndex0;
102 }
103 } );
104 }
105 }

265 // Added interpolation of corner vertices of triangle according to barycentric coordinates and return new vertex index.
266 inline int32 AddInterpolatedVertex(const FVector3f Barycentrics, const int32 TriIndex)
267 {
268 ensure( FMath::Abs( Barycentrics.X + Barycentrics.Y + Barycentrics.Z - 1.0f ) < 1e-4f );
269 ensure( 0.0f <= Barycentrics.X && Barycentrics.X < 1.0f );
270 ensure( 0.0f <= Barycentrics.Y && Barycentrics.Y < 1.0f );
271 ensure( 0.0f <= Barycentrics.Z && Barycentrics.Z < 1.0f );
272
273 const FIndex3i Tri = GetTriangle(TriIndex);
274
275 const FLerpVert& Vert0 = Verts[ Tri.A ];
276 const FLerpVert& Vert1 = Verts[ Tri.B ];
277 const FLerpVert& Vert2 = Verts[ Tri.C ];
278
279 FLerpVert NewVert;
280 NewVert = Vert0 * Barycentrics.X;
281 NewVert += Vert1 * Barycentrics.Y;
282 NewVert += Vert2 * Barycentrics.Z;
283
284 return Verts.Add( NewVert );
285 }

293
294 const FLerpVert& Vert0 = Verts[VertexIndex0];
295 const FLerpVert& Vert1 = Verts[VertexIndex1];
296
297 FLerpVert NewVert;
298 NewVert = Vert0 * Barycentrics.X;
299 NewVert += Vert1 * Barycentrics.Y;
300
301 return Verts.Add(NewVert);

354inline UE::Geometry::FEdgeSplitInfo FMinimalMesh::SplitEdge(const int32 TriIndex, const int32 TriEdgeIndex, const float SplitWeight)
355{
356 UE::Geometry::FEdgeSplitInfo SplitInfo;
357
358 const FIndex3i Triangle = GetTriangle(TriIndex);
359 int32 NewIndex = AddInterpolatedVertex(FVector2f(SplitWeight, 1.f - SplitWeight),
360 GetVertexIndex(TriIndex, TriEdgeIndex),
361 GetVertexIndex(TriIndex, Cycle3(TriEdgeIndex)));
362 SplitInfo.NewVertIndex[0] = NewIndex;
363
364 // Split edge
365 int32 Edge[2];
366 Edge[0] = TriIndex * 3 + TriEdgeIndex;
367 Edge[1] = AdjEdges[ Edge[0] ];
368
369 const int32 NumNewTris = Edge[1] < 0 ? 1 : 2;
370
371 SplitInfo.OldTriIndex[0] = TriIndex;
372 SplitInfo.OldTriIndex[1] = Edge[1] / 3;
373
374 SplitInfo.NewTriIndex[0] = MaxTriID();
375 SplitInfo.NewTriIndex[1] = SplitInfo.NewTriIndex[0] + 1;
376
377 bool bIsSeam = false;
378 if( Edge[1] < 0 )
379 {
380 SplitInfo.OldTriIndex[1] = -1;
381 SplitInfo.NewTriIndex[1] = -1;
382 SplitInfo.NewVertIndex[1] = -1;
383 }
384 else
385 {
386 check( Verts[ Indexes[ Edge[0] ] ].Position == Verts[ Indexes[ Cycle3( Edge[1] ) ] ].Position );
387 check( Verts[ Indexes[ Edge[1] ] ].Position == Verts[ Indexes[ Cycle3( Edge[0] ) ] ].Position );
388 if (EdgeSplitMode == EEdgeSplitMode::PreserveSeams)
389 {
390 // material seam
391 if (MaterialIndexes[SplitInfo.OldTriIndex[0]] != MaterialIndexes[SplitInfo.OldTriIndex[1]])
392 {
393 bIsSeam = true;
394 }
395 else if (!HasIdenticalUVs(Indexes[Edge[0]], Indexes[Cycle3(Edge[1])]) ||
396 !HasIdenticalUVs(Indexes[Edge[1]], Indexes[Cycle3(Edge[0])]))
397 {
398 bIsSeam = true;
399 }
400 }
401 }
402
403 /*
404 v2
405 /|\
406 e1/ | \e2
407 / | \
408 v1/ 0|1 \v0
409 o====+====o
410 v0\ 1|0 /v1
411 \ | /
412 e2\ | /e1
413 \|/
414 v2
415 */
416 for( int32 j = 0; j < NumNewTris; j++ )
417 {
418 const uint32 e0 = Edge[j];
419 const uint32 e1 = Cycle3( Edge[j] );
420 const uint32 e2 = Cycle3( Edge[j], 2 );
421
422 uint32 OldIndex0 = Indexes[ e0 ];
423 uint32 OldIndex1 = Indexes[ e1 ];
424 uint32 OldIndex2 = Indexes[ e2 ];
425
426 int32 OldAdjEdge1 = AdjEdges[ e1 ];
427 int32 OldAdjEdge2 = AdjEdges[ e2 ];
428
429 if( j == 1 )
430 {
431 // only introduce seem when material indices are different.
432 if (EdgeSplitMode == EEdgeSplitMode::CrackFree ||
433 (EdgeSplitMode == EEdgeSplitMode::PreserveSeams && !bIsSeam))
434 {
435 SplitInfo.NewVertIndex[j] = SplitInfo.NewVertIndex[0];
436 }
437 else
438 {
439 NewIndex = AddInterpolatedVertex(FVector2f(SplitWeight, 1.f - SplitWeight), OldIndex1, OldIndex0);
440 SplitInfo.NewVertIndex[j] = NewIndex;
441 }
442 }
443
444 Indexes.AddUninitialized(3);
445 AdjEdges.AddUninitialized(3);
446 MaterialIndexes.Add( CopyTemp( MaterialIndexes[ SplitInfo.OldTriIndex[j] ] ) );
447
448 // replace v0
449 uint32 i = SplitInfo.OldTriIndex[j] * 3;
450 Indexes[ i + 0 ] = NewIndex;
451 Indexes[ i + 1 ] = OldIndex1;
452 Indexes[ i + 2 ] = OldIndex2;
453
454 AdjEdges[ i + 0 ] = SplitInfo.NewTriIndex[j^1] * 3;
455 LinkEdge( i + 1, OldAdjEdge1 );
456 AdjEdges[ i + 2 ] = SplitInfo.NewTriIndex[j] * 3 + 1;
457
458 // replace v1
459 i = SplitInfo.NewTriIndex[j] * 3;
460 Indexes[ i + 0 ] = OldIndex0;
461 Indexes[ i + 1 ] = NewIndex;
462 Indexes[ i + 2 ] = OldIndex2;
463
464 AdjEdges[ i + 0 ] = SplitInfo.OldTriIndex[j^1] * 3;
465 AdjEdges[ i + 1 ] = SplitInfo.OldTriIndex[j] * 3 + 2;
466 LinkEdge( i + 2, OldAdjEdge2 );
467 }
468
469 return SplitInfo;
470}

472inline UE::Geometry::FPokeInfo FMinimalMesh::PokeTriangle(const int32 TriIndex, const FVector3f Barycentrics)
473{
474 UE::Geometry::FPokeInfo PokeInfo;
475
476 const FIndex3i Tri = GetTriangle(TriIndex);
477 const int32 NewIndex = AddInterpolatedVertex(Barycentrics, TriIndex);
478
479 ensure( FMath::Abs( Barycentrics.X + Barycentrics.Y + Barycentrics.Z - 1.0f ) < 1e-4f );
480 ensure( 0.0f <= Barycentrics.X && Barycentrics.X < 1.0f );
481 ensure( 0.0f <= Barycentrics.Y && Barycentrics.Y < 1.0f );
482 ensure( 0.0f <= Barycentrics.Z && Barycentrics.Z < 1.0f );
483
484 const FLerpVert& Vert0 = Verts[ Tri.A ];
485 const FLerpVert& Vert1 = Verts[ Tri.B ];
486 const FLerpVert& Vert2 = Verts[ Tri.C ];
487
488 FLerpVert NewVert;
489 NewVert = Vert0 * Barycentrics.X;
490 NewVert += Vert1 * Barycentrics.Y;
491 NewVert += Vert2 * Barycentrics.Z;
492
493 int32 OldAdjEdges[3];
494 OldAdjEdges[0] = AdjEdges[ TriIndex * 3 + 0 ];
495 OldAdjEdges[1] = AdjEdges[ TriIndex * 3 + 1 ];
496 OldAdjEdges[2] = AdjEdges[ TriIndex * 3 + 2 ];
497
498 PokeInfo.NewTriIndex[0] = TriIndex;
499 PokeInfo.NewTriIndex[1] = MaxTriID();
500 PokeInfo.NewTriIndex[2] = PokeInfo.NewTriIndex[1] + 1;
501
502 // add 2 triangles
503 Indexes.AddUninitialized(6);
504 AdjEdges.AddUninitialized(6);
505 MaterialIndexes.AddUninitialized(2);
506
507 for( uint32 EdgeIndex = 0; EdgeIndex < 3; EdgeIndex++ )
508 {
509 const uint32 e0 = EdgeIndex;
510 const uint32 e1 = (1 << e0) & 3;
511 const uint32 e2 = (1 << e1) & 3;
512
513 uint32 i = PokeInfo.NewTriIndex[ EdgeIndex ] * 3;
514 Indexes[ i + 0 ] = Tri[ e0 ];
515 Indexes[ i + 1 ] = Tri[ e1 ];
516 Indexes[ i + 2 ] = NewIndex;
517
518 LinkEdge( i + 0, OldAdjEdges[ e0 ] );
519 AdjEdges[ i + 1 ] = PokeInfo.NewTriIndex[ e1 ] * 3 + 2;
520 AdjEdges[ i + 2 ] = PokeInfo.NewTriIndex[ e2 ] * 3 + 1;
521
522 MaterialIndexes[ PokeInfo.NewTriIndex[ EdgeIndex ] ] = MaterialIndexes[ TriIndex ];
523 }
524
525 PokeInfo.EdgeIndex[0] = PokeInfo.EdgeIndex[1] = PokeInfo.EdgeIndex[2] = 0;
526 PokeInfo.NewVertIndex = NewIndex;
527
528 return PokeInfo;
529}

531inline UE::Geometry::FFlipEdgeInfo FMinimalMesh::FlipEdge(const int32 TriIndex, const int32 TriEdgeIndex)
532{
533 const int32 EdgeIndex = TriIndex * 3 + TriEdgeIndex;
534 const int32 AdjEdge = AdjEdges[ EdgeIndex ];
535 check(AdjEdge >= 0);
536
537 const int32 AdjTriIndex = AdjEdge / 3;
538
539 /*
540 v2 v0,v1 vA
541 /\ /|\ /\
542 e1/ \e2 e2/ | \e1 eD/ \eA
543 / \ / | \ / \
544 v1/ e0 \v0 / | \ / \
545 o========o => v2 e0|e0 v2 vD vB
546 v0\ e0 /v1 \ | / \ /
547 \ / \ | / \ /
548 e2\ /e1 e1\ | /e2 eC\ /eB
549 \/ \|/ \/
550 v2 v1'v0 vC
551 */
552 const int32 eA = TriIndex * 3 + ( EdgeIndex + 2 ) % 3;
553 const int32 eB = AdjTriIndex * 3 + ( AdjEdge + 1 ) % 3;
554 const int32 eC = AdjTriIndex * 3 + ( AdjEdge + 2 ) % 3;
555 const int32 eD = TriIndex * 3 + ( EdgeIndex + 1 ) % 3;
556
557 int32 IndexA = Indexes[ eA ];
558 int32 IndexB = Indexes[ eB ];
559 int32 IndexC = Indexes[ eC ];
560 int32 IndexD = Indexes[ eD ];
561
562 int32 AdjEdgeA = AdjEdges[ eA ];
563 int32 AdjEdgeB = AdjEdges[ eB ];
564 int32 AdjEdgeC = AdjEdges[ eC ];
565 int32 AdjEdgeD = AdjEdges[ eD ];
566
567 Indexes[ TriIndex * 3 + 0 ] = IndexC;
568 Indexes[ TriIndex * 3 + 1 ] = IndexA;
569 Indexes[ TriIndex * 3 + 2 ] = IndexB;
570
571 Indexes[ AdjTriIndex * 3 + 0 ] = IndexA;
572 Indexes[ AdjTriIndex * 3 + 1 ] = IndexC;
573 Indexes[ AdjTriIndex * 3 + 2 ] = IndexD;
574
575 LinkEdge( TriIndex * 3 , AdjTriIndex * 3 );
576 LinkEdge( TriIndex * 3 + 1, AdjEdgeA );
577 LinkEdge( TriIndex * 3 + 2, AdjEdgeB );
578
579 LinkEdge( AdjTriIndex * 3 + 1, AdjEdgeC );
580 LinkEdge( AdjTriIndex * 3 + 2, AdjEdgeD );
581
582 return { { TriIndex, AdjTriIndex }, { 0, 0 } };
583}
