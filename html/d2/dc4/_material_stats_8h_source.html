735FORCEINLINE void FShaderPlatformSettings::SetPresentInGrid(const bool bValue)
736{
737 if (bAllowPresenceInGrid) // if we're allowed to be present in the stats grid
738 {
739 if (!bAlwaysOn) // and bAlwaysOn is false - then any bValue is fine
740 {
741 bPresentInGrid = bValue;
742 }
743 else if (bValue == true) // otherwise only enabling is fine, as we shouldn't be able to disable this platform
744 {
745 bPresentInGrid = bValue;
746 }
747 }
748 else
749 {
750 if (bValue == false) // if we're not allowed to be present, only disabling this platform is fine
751 {
752 bPresentInGrid = bValue;
753 }
754 }
