335template< typename T >
336FORCEINLINE bool TSimpTri<T>::ReplaceVertexIsValid( const TSimpVert<T>* oldV, const FVector3f& pos ) const
337{
338 checkSlow( oldV );
339 checkSlow( oldV == verts[0] || oldV == verts[1] || oldV == verts[2] );
340
341 uint32 k;
342 if( oldV == verts[0] )
343 k = 0;
344 else if( oldV == verts[1] )
345 k = 1;
346 else
347 k = 2;
348
349 const FVector3f& v0 = verts[k]->GetPos();
350 const FVector3f& v1 = verts[ k = (1 << k) & 3 ]->GetPos();
351 const FVector3f& v2 = verts[ k = (1 << k) & 3 ]->GetPos();
352
353 const FVector3f d21 = v2 - v1;
354 const FVector3f d01 = v0 - v1;
355 const FVector3f dp1 = pos - v1;
356
357 FVector3f n0 = d01 ^ d21;
358 FVector3f n1 = dp1 ^ d21;
359
360 return (n0 | n1) > 0.0f;
361}

363template< typename T >
364FORCEINLINE void TSimpTri<T>::ReplaceVertex( TSimpVert<T>* oldV, TSimpVert<T>* newV )
365{
366 checkSlow( oldV && newV );
367 checkSlow( oldV == verts[0] || oldV == verts[1] || oldV == verts[2] );
368 checkSlow( newV != verts[0] && newV != verts[1] && newV != verts[2] );
369
370 if( oldV == verts[0] )
371 verts[0] = newV;
372 else if( oldV == verts[1] )
373 verts[1] = newV;
374 else
375 verts[2] = newV;
376
377 checkSlow( !HasVertex( oldV ) );
378}
