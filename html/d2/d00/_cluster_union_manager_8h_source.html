299 template<typename TParticle>
300 void TransferClusterUnionShapeData(const TUniquePtr<Chaos::FPerShapeData>& ShapeData, TParticle* TemplateParticle, const TUniquePtr<Chaos::FPerShapeData>& TemplateShape, int32 ActorId, int32 ComponentId)
301 {
302 if (ShapeData && TemplateShape)
303 {
304 {
305 FCollisionData Data = TemplateShape->GetCollisionData();
306 Data.UserData = nullptr;
307 ShapeData->SetCollisionData(Data);
308 }
309
310 {
311 const Chaos::Filter::FInstanceData InstanceData(ActorId, ComponentId);
312 ShapeData->SetFilterInstanceData(InstanceData);
313 ShapeData->SetShapeFilterData(TemplateShape->GetShapeFilterData());
314 }
315
316 {
317 const int32 NumTemplateMaterials = TemplateShape->NumMaterials();
318
319 if(NumTemplateMaterials == 1)
320 {
321 // We have a special case for 1 material, calling GetMaterials
322 // would allocate a material storage object to apply to the shape
323 // whereas for 1 material we store it inline to the shape data
324 ShapeData->SetMaterial(TemplateShape->GetMaterial(0));
325 }
326 else if(NumTemplateMaterials > 1)
327 {
328 ShapeData->SetMaterials(TemplateShape->GetMaterials());
329 }
330 }
331 }

340 template<typename TClusterParticle, typename TParticle, typename TLambda>
341 void ModifyAdditionOfChildrenToClusterUnionGeometry(TClusterParticle* ClusterParticle, const TArray<TParticle*>& Particles, int32 ActorId, int32 ComponentId, TLambda&& Func)
342 {
343 if (!ClusterParticle || Particles.IsEmpty())
344 {
345 return;
346 }
347
348 FImplicitObjectUnion* ImplicitUnion = ClusterParticle->GetGeometry() ? ClusterParticle->GetGeometry()->template AsA<FImplicitObjectUnion>() : nullptr;
349 const int32 OldNumChildShapes = ImplicitUnion ? ImplicitUnion->GetNumRootObjects() : 0;
350
351 Func();
352
353 // We must have a union at this point since the contract is that Func is adding a shape
354 // which means we must be in a union already.
355 check(ClusterParticle->GetGeometry() != nullptr);
356 ImplicitUnion = ClusterParticle->GetGeometry()->template AsA<FImplicitObjectUnion>();
357 check(ImplicitUnion != nullptr);
358
359 const int32 NewNumChildShapes = ImplicitUnion->GetNumRootObjects();
360 const FShapesArray& ShapeArray = ClusterParticle->ShapesArray();
361 check(ShapeArray.Num() == NewNumChildShapes);
362 check(Particles.Num() == NewNumChildShapes - OldNumChildShapes);
363
364 for (int32 Index = 0; Index < Particles.Num(); ++Index)
365 {
366 TParticle* Particle = Particles[Index];
367 if (!Particle)
368 {
369 continue;
370 }
371
372 const TUniquePtr<Chaos::FPerShapeData>& TemplateShape = Particle->ShapesArray()[0];
373 if (!TemplateShape)
374 {
375 continue;
376 }
377
378 const int32 ShapeIndex = OldNumChildShapes + Index;
379 if (const TUniquePtr<Chaos::FPerShapeData>& ShapeData = ShapeArray[ShapeIndex])
380 {
381 TransferClusterUnionShapeData(ShapeData, Particle, TemplateShape, ActorId, ComponentId);
382 }
383 }

389 template<typename TClusterParticle, typename TParticle>
390 void RemoveParticlesFromClusterUnionGeometry(TClusterParticle* ClusterParticle, const TArray<TParticle*>& ShapeParticles, TArray<TParticle*>& AllChildParticles)
391 {
392 if (!ensure(ClusterParticle))
393 {
394 return;
395 }
396
397 // Don't try to remove anything if the shape array is already empty
398 if (!ClusterParticle->ShapesArray().IsEmpty() && !AllChildParticles.IsEmpty())
399 {
400#if WITH_EDITOR
401 ensureMsgf(
402 AllChildParticles.Num() == ClusterParticle->ShapesArray().Num(),
403 TEXT("RemoveParticlesFromClusterUnionGeometry : More than one shape per child particle : ShapeArray=[%d] AllChildParticles=[%d] ShapeParticles=[%d]"),
404 ClusterParticle->ShapesArray().Num(),
405 AllChildParticles.Num(),
406 ShapeParticles.Num()
407 );
408#else
409 checkf(
410 AllChildParticles.Num() == ClusterParticle->ShapesArray().Num(),
411 TEXT("RemoveParticlesFromClusterUnionGeometry : More than one shape per child particle : ShapeArray=[%d] AllChildParticles=[%d] ShapeParticles=[%d]"),
412 ClusterParticle->ShapesArray().Num(),
413 AllChildParticles.Num(),
414 ShapeParticles.Num()
415 );
416#endif
417
418
419 TArray<int32> ShapeIndicesToRemove;
420 ShapeIndicesToRemove.Reserve(ShapeParticles.Num());
421 for (TParticle* ShapeParticle : ShapeParticles)
422 {
423 check(ShapeParticle != nullptr);
424 const int32 Index = AllChildParticles.Find(ShapeParticle);
425 if (Index != INDEX_NONE)
426 {
427 ShapeIndicesToRemove.Add(Index);
428 }
429 }
430
431 ShapeIndicesToRemove.Sort();
432
433 ClusterParticle->RemoveShapesAtSortedIndices(ShapeIndicesToRemove);
434
435 RemoveArrayItemsAtSortedIndices(AllChildParticles, ShapeIndicesToRemove);
436
437#if WITH_EDITOR
438 ensureMsgf(
439 AllChildParticles.Num() == ClusterParticle->ShapesArray().Num(),
440 TEXT("RemoveParticlesFromClusterUnionGeometry : More than one shape per child particle : ShapeArray=[%d] AllChildParticles=[%d] ShapeParticles=[%d]"),
441 ClusterParticle->ShapesArray().Num(),
442 AllChildParticles.Num(),
443 ShapeParticles.Num()
444 );
445#else
446 checkf(
447 AllChildParticles.Num() == ClusterParticle->ShapesArray().Num(),
448 TEXT("RemoveParticlesFromClusterUnionGeometry : More than one shape per child particle : ShapeArray=[%d] AllChildParticles=[%d] ShapeParticles=[%d]"),
449 ClusterParticle->ShapesArray().Num(),
450 AllChildParticles.Num(),
451 ShapeParticles.Num()
452 );
453#endif
454 }
455 // If we remove particles from the cluster union geometry then we need to switch the geometry back to a FImplicitObjectUnionClustered to avoid errors with empty unions.
456 if (ClusterParticle->ShapesArray().IsEmpty())
457 {
458 ClusterParticle->SetGeometry(MakeImplicitObjectPtr<FImplicitObjectUnionClustered>());
459 }
460 }
