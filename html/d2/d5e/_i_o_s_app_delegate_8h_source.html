34enum class EAudioFeature : uint8
35{
36 Playback, // Audio Not affected by the ringer switch
37 Record, // Recording only, unless Playback or VoiceChat is also set
38
39 DoNotMixWithOthers, // Do not mix audio with other applications
40
41 VoiceChat, // set AVAudioSessionModeVoiceChat when both Playback and Record are enabled
42 UseReceiver, // use receiver instead of speaker when both Playback and Record are enabled. Headsets will still be preferred if they are present
43 DisableBluetoothSpeaker, // disable the use of Bluetooth A2DP speakers when both Playback and Record are enabled
44
45 BluetoothMicrophone, // enable the use of Bluetooth HFP headsets when Record is enabled
46
47 BackgroundAudio, // continue to play audio in the background. Requires an appropriate background mode to be set in Info.plist
48
49 AllowHapticsAndSystemSoundsDuringRecording, // allow haptic feedback and system sounds when Record or PlayAndRecord category is active
50
51 NumFeatures,
52};
