497 FMovieSceneSequenceEditor* FindSequenceEditor(UClass* SequenceClass) const
498 {
499 check(SequenceClass);
500
501 UClass* MostRelevantClass = nullptr;
502 FMovieSceneSequenceEditor* SequenceEditor = nullptr;
503
504 for (const FSequenceEditorEntry& Entry : SequenceEditors)
505 {
506 if (SequenceClass->IsChildOf(Entry.ApplicableClass))
507 {
508 if (!MostRelevantClass || Entry.ApplicableClass->IsChildOf(MostRelevantClass))
509 {
510 MostRelevantClass = Entry.ApplicableClass;
511 SequenceEditor = Entry.Editor.Get();
512 }
513 }
514 }
515
516 return SequenceEditor;
517 }
