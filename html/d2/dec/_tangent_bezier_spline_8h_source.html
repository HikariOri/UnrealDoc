230 virtual bool Serialize(FArchive& Ar) override
231 {
232 // Call immediate parent's Serialize (TSplineWrapper)
233 if (!TSplineWrapper<TPolyBezierSpline<ValueType>>::Serialize(Ar))
234 {
235 return false;
236 }
237
238 Ar << Tension;
239 // Serialize as uint8 for compatibility
240 int32 NumTangentModes = TangentModes.Num();
241 Ar << NumTangentModes;
242
243 if (Ar.IsLoading())
244 {
245 TangentModes.SetNum(NumTangentModes);
246
247 InstallFindWindowOverride();
248 }
249
250 for (int32 i = 0; i < NumTangentModes; ++i)
251 {
252 uint8 TangentModeValue = static_cast<uint8>(TangentModes[i]);
253 Ar << TangentModeValue;
254
255 if (Ar.IsLoading())
256 {
257 TangentModes[i] = static_cast<ETangentMode>(TangentModeValue);
258 }
259 }
260
261 Ar << bStationaryEndpoint;
262
263 if (Ar.IsLoading())
264 {
265 const int32 OldNumTangentModes = TangentModes.Num();
266 const int32 NumPoints = GetNumPoints();
267
268 if (OldNumTangentModes != NumPoints)
269 {
270 UE_LOGF(LogSpline, Display, "Incorrect number of tangent modes loaded (loaded %d, expected %d), fixing up.", OldNumTangentModes, NumPoints);
271
272 TangentModes.SetNum(NumPoints);
273 const int32 NewNumTangentModes = TangentModes.Num();
274
275 for (int32 TangentModeIdx = OldNumTangentModes; TangentModeIdx < NewNumTangentModes; ++TangentModeIdx)
276 {
277 TangentModes[TangentModeIdx] = ETangentMode::Auto;
278 }
279 }
280 }
281
282 return true;
283 }

461 bool AppendPoints(const TArray<FTangentBezierControlPoint>& Points, EParameterizationPolicy Policy)
462 {
463 const int32 NumPoints = Points.Num();
464
465 // Need at least 2 points to add a valid segment
466 if (NumPoints < 2)
467 {
468 UE_LOGF(LogSpline, Warning, "AppendPoints requires at least 2 points to add a valid segment. Got %d points.", NumPoints);
469 return false;
470 }
471
472 // If the spline is currently empty (though it shouldn't be based on our design),
473 // this is equivalent to SetPoints
474 if (GetNumPoints() == 0)
475 {
476 return SetPoints(Points, Policy);
477 }
478
479 // For each point, use AddPoint
480 for (int32 i = 0; i < NumPoints; ++i)
481 {
482 // Skip the first point if it's the same as our last existing point
483 // (to avoid duplicates at connection point)
484 if (i == 0)
485 {
486 const int32 ExistingNumPoints = GetNumPoints();
487 const ValueType LastExistingPoint = GetValue(ExistingNumPoints - 1);
488
489 // Check if connection points are reasonably close
490 constexpr double ConnectionTolerance = 1e-4;
491 bool bSameEndpoint;
492
493 if constexpr (Math::TIsQuaternionType<ValueType>::value)
494 {
495 // angular distance
496 const double dot = Math::Dot(LastExistingPoint, Points[0].Position);
497 const double ang = FMath::Acos(FMath::Clamp(dot, -1.0, 1.0)); // radians
498 bSameEndpoint = (ang <= ConnectionTolerance);
499 }
500 else
501 {
502 const double DistanceSquared = Math::SizeSquared(LastExistingPoint - Points[0].Position);
503 bSameEndpoint = (DistanceSquared <= ConnectionTolerance);
504 }
505
506 if (bSameEndpoint)
507 {
508 // Points are close - the first existing point is the same as our first new point
509 // So update the last point's tangent mode if needed
510 if (Points[0].TangentMode != ETangentMode::User && Points[0].TangentMode != ETangentMode::Broken)
511 {
512 SetPointTangentMode(ExistingNumPoints - 1, Points[0].TangentMode);
513 }
514 continue; // Skip adding this point
515 }
516 }
517
518 // Add each point normally
519 AppendPoint(Points[i], Policy);
520 }
521
522 return true;
523 }

692 {
693 if (Index < 0 || Index >= GetNumPoints())
694 {
695 return;
696 }

836 void SetPointTangentMode(int32 Index, ETangentMode Mode)
837 {
838 const int32 NumPoints = GetNumPoints();
839
840 if (Index < 0 || Index >= NumPoints)
841 {
842 return;
843 }
844
845 if (!TangentModes.IsValidIndex(Index))
846 {
847 // If we get here, TangentModes was in a bad state. We will correct the number of TangentModes, defaulting new points to Auto, before proceeding with the operation.
848

852 TangentModes.SetNum(NumPoints);
853 const int32 NewTangentModesNum = TangentModes.Num();

856 {
857 TangentModes[TangentModesIdx] = ETangentMode::Auto;
858 }
859 }
860

896 {

916 void UpdateTangent(int32 Index)
917 {
918 const int32 NumPoints = GetNumPoints();
919 const int32 NumSegments = GetNumberOfSegments();
920
921 if (Index < 0 || Index >= NumPoints || !ensureAlwaysMsgf(TangentModes.IsValidIndex(Index), TEXT("No tangent mode for point!")))
922 {
923 return;
924 }
925
926 const ETangentMode Mode = TangentModes[Index];
927
928 if (Mode == ETangentMode::User || Mode == ETangentMode::Broken)
929 {
930 return;
931 }
932
933 if constexpr (Math::TIsQuaternionType<ValueType>::value)
934 {
935 if (Mode == ETangentMode::Linear || Mode == ETangentMode::LegacyLinear)
936 {
937 const ValueType Q = GetValue(Index).GetNormalized();
938 SetTangentIn(Index, Q);
939 SetTangentOut(Index, Q);
940 return;
941 }
942 }
943
944 // Get the position of the current point
945 const ValueType& Current = GetValue(Index);
946
947 // Calculate indices handling closed loops
948 const int32 PrevIndex = (Index > 0) ? (Index - 1) : (this->IsClosedLoop() ? NumPoints - 1 : Index);
949 const int32 NextIndex = (Index < NumPoints - 1) ? (Index + 1) : (this->IsClosedLoop() ? 0 : Index);
950 const ValueType& Prev = GetValue(PrevIndex);
951 const ValueType& Next = GetValue(NextIndex);
952
953 // Get the actual parameter values for each point
954 float PrevParam = GetParameter(PrevIndex);
955 float CurrentParam = GetParameter(Index);
956 float NextParam = GetParameter(NextIndex);
957
958 // If we are closed and updating the first or last point, we need to do some special work
959 // to get virtual PrevParam or NextParam. The 'virtual' parameter is not the actual parameter
960 // of the previous or next point, but the current parameter +- the knot delta (closing segment length).
961 if (this->IsClosedLoop())
962 {
963 if (Index == 0 && PrevIndex == NumPoints - 1)
964 {
965 // First point, prev wraps from end - subtract the closing segment length
966 // The closing segment is the last segment in a closed loop
967 PrevParam = CurrentParam - GetSegmentParameterRange(NumSegments - 1).Length();
968 }
969 else if (Index == NumPoints - 1 && NextIndex == 0)
970 {
971 // Last point, next wraps to start - add the closing segment length
972 NextParam = CurrentParam + GetSegmentParameterRange(NumSegments - 1).Length();
973 }
974 }
975
976 const float dTL = FMath::Max(UE_KINDA_SMALL_NUMBER, CurrentParam - PrevParam); // span to the *left*
977 const float dTR = FMath::Max(UE_KINDA_SMALL_NUMBER, NextParam - CurrentParam ); // span to the *right*
978
979 // Calculate tangents based on mode
980 ValueType InTangent = GetTangentIn(Index); // Keep existing if not changing
981 ValueType OutTangent = GetTangentOut(Index);
982
983 switch (Mode)
984 {
985 case ETangentMode::Auto:
986 case ETangentMode::AutoClamped:
987 {
988 const bool bIsStationaryEndpoint = bStationaryEndpoint &&
989 (Index == 0 || Index == NumPoints - 1) &&
990 !this->IsClosedLoop();
991
992 if (bIsStationaryEndpoint)
993 {
994 if constexpr (Math::TIsQuaternionType<ValueType>::value)
995 {
996 InTangent = Current; // control = key rotation
997 OutTangent = Current;
998 }
999 else
1000 {
1001 // Stationary endpoints get zero tangent
1002 InTangent = ValueType(0);
1003 OutTangent = ValueType(0);
1004 }
1005 }
1006 else
1007 {
1008 const bool bWantClamping = (Mode == ETangentMode::AutoClamped);
1009 TangentOps::ComputeAutoTangent(
1010 PrevParam, Prev, CurrentParam, Current, NextParam, Next,
1011 Tension, bWantClamping, InTangent);
1012 OutTangent = InTangent;
1013 }
1014 }
1015 break;
1016
1017 case ETangentMode::Linear:
1018 {
1019 InTangent = Current - Prev;
1020 OutTangent = Next - Current;
1021 }
1022 break;
1023
1024 case ETangentMode::LegacyLinear:
1025 {
1026 // FInterpCurve-compatible behavior
1027 OutTangent = Next - Current;
1028 InTangent = IsCurveKey(TangentModes[PrevIndex]) ? InTangent = OutTangent : (Current - Prev);
1029 }
1030 break;
1031 case ETangentMode::LegacyAuto:
1032 case ETangentMode::LegacyAutoClamped:
1033 {
1034 if (bStationaryEndpoint && (Index == 0 || Index == NumPoints - 1) && !this->IsClosedLoop())
1035 {
1036 // Stationary endpoints get zero tangent
1037 InTangent = ValueType(0);
1038 OutTangent = ValueType(0);
1039 }
1040 else if (IsCurveKey(TangentModes[PrevIndex]))
1041 {
1042 // Use TTangentOps which handles clamping correctly
1043 const bool bWantClamping = (Mode == ETangentMode::LegacyAutoClamped);
1044 TangentOps::ComputeAutoTangent(
1045 PrevParam, Prev,
1046 CurrentParam, Current,
1047 NextParam, Next,
1048 Tension,
1049 bWantClamping,
1050 InTangent);
1051 OutTangent = InTangent;
1052 }
1053 else
1054 {
1055 // Previous is Linear or Constant - copy its tangents (legacy behavior)
1056 InTangent = GetTangentIn(PrevIndex);
1057 OutTangent = GetTangentOut(PrevIndex);
1058 }
1059 }
1060 break;
1061
1062 case ETangentMode::Constant:
1063 {
1064 // Zero tangents for constant interpolation
1065 InTangent = ValueType();
1066 OutTangent = ValueType();
1067 }
1068 break;
1069
1070 case ETangentMode::User:
1071 break;
1072
1073 case ETangentMode::Unknown:
1074 default:
1075 // Default to Auto mode if unknown
1076 {
1077 ValueType Tangent;
1078 TangentOps::ComputeAutoTangent(
1079 PrevParam, Prev,
1080 CurrentParam, Current,
1081 NextParam, Next,
1082 Tension,
1083 false,
1084 Tangent);
1085 InTangent = Tangent;
1086 OutTangent = Tangent;
1087 }
1088 break;
1089 }
1090
1091 // Apply the calculated tangents
1092 SetTangentIn(Index, InTangent);
1093 SetTangentOut(Index, OutTangent);
1094 }
