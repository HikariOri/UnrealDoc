324 template <typename T>
325 void WriteNetTokenWithKnownType(FNetSerializationContext& Context, FNetToken NetToken)
326 {
327#if UE_NET_VALIDATE_NETTOKENTYPE
328 // We only verify this in dev builds as it should trap most programmer errors.
329 if (NetToken.IsValid())
330 {
331 const T* DataStore = GetDataStore<T>();
332 if (!DataStore || (DataStore->GetTypeId() != NetToken.GetTypeId()))
333 {
334 UE_LOGF(LogNetToken, Error, "Tried to write NetToken %ls using invalid NetTokenStore %ls", *NetToken.ToString(), *(T::GetTokenStoreName().ToString()));
335 // Just to get some attention to the log.
336 ensure(DataStore && (DataStore->GetTypeId() == NetToken.GetTypeId()));
337 }
338 }
339#endif
340 const bool bWriteTypeId = false;
341 FNetTokenStore::InternalWriteNetToken(Context, NetToken, bWriteTypeId);
342 }
343
347 template <typename T>
348 FNetToken ReadNetTokenWithKnownType(FNetSerializationContext& Context)
349 {
350 FNetToken NetToken;
351 if (T* DataStore = GetDataStore<T>())
352 {
353 NetToken = DataStore->ReadNetToken(Context);
354 }
355 else
356 {
357 UE_LOGF(LogNetToken, Error, "ReadNetTokenWithKnownType Tried to read NetToken using invalid NetTokenStore %ls", *(T::GetTokenStoreName().ToString()));
358 // Just to get some attention to the log.
359 ensure(DataStore);
360 }
361
362 return NetToken;
363 }

434inline FNetTokenDataStore::FNetTokenStoreKey FNetTokenDataStore::GetNextNetTokenStoreKey()
435{
436 UE_IRIS_PARALLEL_EXPR(FScopedNetTokenStoreAccess ScopeAccess(&TokenStore));
437 uint32 NextTokenStoreKeyIndex = (uint32)StoredTokens.Num();
438 if (ensure(NextTokenStoreKeyIndex < FNetToken::MaxNetTokenCount))
439 {
440 FNetTokenStoreKey TokenKey(NextTokenStoreKeyIndex);
441
442 // We fill in the actual key later
443 StoredTokens.Add(FNetToken());
444
445 return TokenKey;
446 }
447
448 return FNetTokenStoreKey();
