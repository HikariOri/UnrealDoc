494 template <class CharType>
495 static bool Deserialize(TJsonReader<CharType>& Reader, typename Policy::FStackState& OutStackState, EFlags InOptions)
496 {
497 typename Policy::FGlobalState GlobalState;
498 TArray<TSharedRef<typename Policy::FStackState>> ScopeStack;
499 TSharedPtr<typename Policy::FStackState> CurrentState;
500
501 typename Policy::FValue NewValue;
502 EJsonNotation Notation;
503
504 while (Reader.ReadNext(Notation))
505 {
506 FString Identifier = Reader.GetIdentifier();
507
508 Policy::ResetValue(NewValue);
509
510 bool bWasValueRead = false;
511
512 switch( Notation )
513 {
514 case EJsonNotation::ObjectStart:
515 {
516 if (CurrentState.IsValid())
517 {
518 ScopeStack.Push(CurrentState.ToSharedRef());
519 }
520
521 CurrentState = MakeShared<typename Policy::FStackState>(GlobalState);
522 CurrentState->Identifier = Identifier;
523 Policy::ReadObjectStart(*CurrentState);
524 }
525 break;
526
527 case EJsonNotation::ObjectEnd:
528 {
529 if (ScopeStack.Num() > 0)
530 {
531 Identifier = CurrentState->Identifier;
532 Policy::ReadObjectEnd(*CurrentState, NewValue);
533 bWasValueRead = true;
534 CurrentState = ScopeStack.Pop();
535 }
536 }
537 break;
538
539 case EJsonNotation::ArrayStart:
540 {
541 if (CurrentState.IsValid())
542 {
543 ScopeStack.Push(CurrentState.ToSharedRef());
544 }
545
546 CurrentState = MakeShared<typename Policy::FStackState>(GlobalState);
547 Policy::ReadArrayStart(*CurrentState);
548 CurrentState->Identifier = Identifier;
549 }
550 break;
551
552 case EJsonNotation::ArrayEnd:
553 {
554 if (ScopeStack.Num() > 0)
555 {
556 Identifier = CurrentState->Identifier;
557 Policy::ReadArrayEnd(*CurrentState, NewValue);
558 bWasValueRead = true;
559 CurrentState = ScopeStack.Pop();
560 }
561 }
562 break;
563
564 case EJsonNotation::Boolean:
565 Policy::ReadBoolean(Reader, NewValue); bWasValueRead = true;
566 break;
567
568 case EJsonNotation::String:
569 Policy::ReadString(Reader, NewValue); bWasValueRead = true;
570 break;
571
572 case EJsonNotation::Number:
573 {
574 if (EnumHasAnyFlags(InOptions, EFlags::StoreNumbersAsStrings))
575 {
576 Policy::ReadNumberAsString(Reader, NewValue);
577 }
578 else
579 {
580 Policy::ReadNumber(Reader, NewValue);
581 }
582
583 bWasValueRead = true;
584 }
585 break;
586
587 case EJsonNotation::Null:
588 Policy::ReadNull(NewValue); bWasValueRead = true;
589 break;
590
591 case EJsonNotation::Error:
592 return false;
593 break;
594 }
595
596 if (bWasValueRead && CurrentState.IsValid())
597 {
598 if (CurrentState->Type == EJson::Object)
599 {
600 Policy::AddValueToObject(*CurrentState, Identifier, NewValue);
601 }
602 else
603 {
604 Policy::AddValueToArray(*CurrentState, NewValue);
605 }
606 }
607 }
608
609 if (!CurrentState.IsValid() || !Reader.GetErrorMessage().IsEmpty())
610 {
611 return false;
612 }
613
614 OutStackState = *CurrentState.Get();
615
616 return true;
617 }

619 template <class CharType, class PrintPolicy>
620 static bool Serialize(const TSharedRef<typename Policy::FElement>& StartingElement, TJsonWriter<CharType, PrintPolicy>& Writer, bool bCloseWriter)
621 {
622 TArray<TSharedRef<typename Policy::FElement>> ElementStack;
623 ElementStack.Push(StartingElement);
624
625 while (ElementStack.Num() > 0)
626 {
627 TSharedRef<typename Policy::FElement> Element = ElementStack.Pop();
628
629 // Empty keys are valid identifiers only when writing inside an object.
630 const bool bWriteValueOnly = !Element->bIsKeyValuePair || (Element->Identifier.IsEmpty() && Writer.GetCurrentElementType() != EJson::Object);
631
632 if (Policy::SerializeIfBool(Element, Writer, bWriteValueOnly))
633 {
634 continue;
635 }
636
637 if (Policy::SerializeIfNumber(Element, Writer, bWriteValueOnly))
638 {
639 continue;
640 }
641
642 if (Policy::SerializeIfString(Element, Writer, bWriteValueOnly))
643 {
644 continue;
645 }
646
647 if (Policy::SerializeIfNull(Element, Writer, bWriteValueOnly))
648 {
649 continue;
650 }
651
652 if (Policy::SerializeIfArray(ElementStack, Element, Writer, bWriteValueOnly))
653 {
654 continue;
655 }
656
657 if (Policy::SerializeIfObject(ElementStack, Element, Writer, bWriteValueOnly))
658 {
659 continue;
660 }
661
662 UE_LOGF(LogJson, Fatal, "Could not print Json Value, unrecognized type.");
663 }
664
665 if (bCloseWriter)
666 {
667 return Writer.Close();
668 }
669 else
670 {
671 return true;
672 }
673 }
