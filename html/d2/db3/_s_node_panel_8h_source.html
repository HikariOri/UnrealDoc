469 virtual FVector2D ComputeDesiredSize(float) const override
470 {
471 for( int32 ChildIndex = 0; ChildIndex < Children.Num(); ++ChildIndex )
472 {
473 if( Children[ ChildIndex ].Zone == ENodeZone::Center )
474 {
475 const FNodeSlot& CenterZone = Children[ ChildIndex ];
476 const EVisibility ChildVisibility = CenterZone.GetWidget()->GetVisibility();
477 if( ChildVisibility != EVisibility::Collapsed )
478 {
479 return ( FVector2D(CenterZone.GetWidget()->GetDesiredSize() + CenterZone.GetPadding().GetDesiredSize()) ) * FVector2D(GetDesiredSizeScale());
480 }
481 }
482 }
483 return FVector2D::ZeroVector;
484 }
485
486 virtual float GetRelativeLayoutScale(int32 ChildIndex, float LayoutScaleMultiplier) const override
487 {
488 const FNodeSlot& ThisSlot = Children[ChildIndex];
489 if (!ThisSlot.AllowScale.Get())
490 {
491 // Child slots that do not allow zooming should scale themselves to negate the node panel's zoom.
492 TSharedPtr<SNodePanel> ParentPanel = GetParentPanel();
493 if (ParentPanel.IsValid())
494 {
495 return 1.0f / ParentPanel->GetZoomAmount();
496 }
497 }
498
499 return 1.0f;
500 }

502 virtual void OnArrangeChildren( const FGeometry& AllottedGeometry, FArrangedChildren& ArrangedChildren ) const override
503 {
504 for( int32 ChildIndex = 0; ChildIndex < Children.Num(); ++ChildIndex )
505 {
506 const FNodeSlot& CurChild = Children[ChildIndex];
507 const EVisibility ChildVisibility = CurChild.GetWidget()->GetVisibility();
508 if ( ArrangedChildren.Accepts(ChildVisibility) )
509 {
510 const FMargin SlotPadding(CurChild.GetPadding());
511 // If this child is not allowed to scale, its scale relative to its parent should undo the parent widget's scaling.
512 FVector2f Size;
513
514 if( CurChild.bSizeAttrIsSet )
515 {
516 Size = CurChild.Size.Get();
517 }
518 else
519 {
520 AlignmentArrangeResult XResult = AlignChild<Orient_Horizontal>(AllottedGeometry.GetLocalSize().X, CurChild, SlotPadding);
521 AlignmentArrangeResult YResult = AlignChild<Orient_Vertical>(AllottedGeometry.GetLocalSize().Y, CurChild, SlotPadding);
522 Size = FVector2f( XResult.Size, YResult.Size );
523 }
524 const FArrangedWidget ChildGeom =
525 AllottedGeometry.MakeChild(
526 CurChild.GetWidget(),
527 CurChild.Offset.Get(),
528 Size,
529 GetRelativeLayoutScale(ChildIndex, AllottedGeometry.Scale)
530 );
531 ArrangedChildren.AddWidget( ChildVisibility, ChildGeom );
532 }
533 }
