202template<typename Settings>
203typename TStateStream<Settings>::FHandle TStateStream<Settings>::Game_CreateInstance(const FStaticState& Ss, const FDynamicState& Ds, uint32 LaneId)
204{
205 if (LaneId == DefaultLaneId)
206 {
207 check(ActiveDefaultLaneId != DefaultLaneId);
208 LaneId = ActiveDefaultLaneId;
209 }
210
211 check(CurrentTick);
212
213 // We usually have several dynamic states stored for each instance
214 // and consecutive ticks could refer the same dynamic state if nothing changed
215 const uint32 InstanceIndex = Instances.Emplace(Ss, 1, CurrentTick->Index);
216 const uint32 DynamicStateIndex = DynamicStates.Add(Ds);
217
218 CurrentTickLock.WriteLock();
219
220 // We make sure the current tick dynamic states / instances have at least the stream instances size
221 if (static_cast<uint32>(CurrentTick->DynamicStates.Num()) <= InstanceIndex)
222 {
223 CurrentTick->DynamicStates.SetNum(InstanceIndex + 1);
224 CurrentTick->ModifiedInstances.SetNum(InstanceIndex + 1, false);
225 }
226 CurrentTick->ModifiedInstances[InstanceIndex] = true;
227 CurrentTick->DynamicStates[InstanceIndex] = DynamicStateIndex;
228
229 CurrentTickLock.WriteUnlock();
230
231 return { *this, InstanceIndex + 1 };
232}

245template<typename Settings>
246void TStateStream<Settings>::Game_Release(uint32 HandleId)
247{
248 check(CurrentTick);
249 check(HandleId != 0);
250 uint32 InstanceIndex = HandleId - 1;
251 FInstance& Instance = Instances[InstanceIndex];
252 check(Instance.RefCount);
253 if (--Instance.RefCount)
254 {
255 return;
256 }
257
258 // If no more refs we are going to delete the tick store its index onto the delete tick
259 // the static and dynamic states should be made internal
260 Instance.DeleteTick = CurrentTick->Index;
261 MakeInternal(Instance.StaticState);
262
263 CurrentTickLock.ReadLock();
264
265 CurrentTick->ModifiedInstances[InstanceIndex] = true;
266 uint32 DsIndex = CurrentTick->DynamicStates[InstanceIndex];
267
268 CurrentTickLock.ReadUnlock();
269
270 MakeInternal(DynamicStates[DsIndex]);
271}

288template<typename Settings>
289typename TStateStream<Settings>::FDynamicState& TStateStream<Settings>::Edit(uint32 HandleId, double TimeFactor)
290{
291 check(HandleId != 0);
292 uint32 InstanceIndex = HandleId - 1;
293
294 check(CurrentTick);
295 check(InstanceIndex < uint32(CurrentTick->DynamicStates.Num()));
296
297 CurrentTickLock.ReadLock();
298
299 FDynamicState* DsState;
300
301 if (CurrentTick->ModifiedInstances[InstanceIndex])
302 {
303 // If the instance had already been modified just return the
304 // current tick dynamic states for that instance
305 uint32 DsIndex = CurrentTick->DynamicStates[InstanceIndex];
306 CurrentTickLock.ReadUnlock();
307 DsState = &DynamicStates[DsIndex];
308 }
309 else
310 {
311 // If not modified, create a new dynamic state and reference
312 // it in the current tick ones.
313 uint32& IndexRef = CurrentTick->DynamicStates[InstanceIndex];
314 uint32 OldIndex = IndexRef;
315 CurrentTick->ModifiedInstances[InstanceIndex] = true;
316 uint32 DsIndex;
317 void* DsPtr = DynamicStates.AddUninitialized(DsIndex);
318 IndexRef = DsIndex;
319 CurrentTickLock.ReadUnlock();
320
321 FDynamicState& OldDs = DynamicStates[OldIndex];
322 DsState = new (DsPtr) FDynamicState(OldDs);
323
324 // Why do wee need to make the old state internal since
325 // this state could still be referenced by previous ticks?
326 MakeInternal(OldDs);
327 }
328 return *DsState;
329}

436template<typename Settings>
437void TStateStream<Settings>::Render_Update(StateStreamTime AbsoluteTime)
438{
439 bool Requested = RendEnabledRequested;
440 if (Requested != RendEnabled)
441 {
442 RendEnabled = Requested;
443
444 // If the render update is enable and if it has changed we are in creation mode
445 // We need to find the tick with previous time lower than the requested absolute time
446 // and use it to fill the render dynamic state
447 if (RendEnabled)
448 {
449 // Figure out which tick to use and recreate instances
450 check(NewestAvailableTick);
451 FTick* Tick = NewestAvailableTick;
452 while (true)
453 {
454 if (AbsoluteTime >= Tick->Time || !Tick->PrevTick)
455 {
456 break;
457 }
458 Tick = Tick->PrevTick;
459 }
460
461 RendTick = Tick;
462 RendTime = AbsoluteTime;
463
464 for (uint32 InstanceIndex = 0, E = RendTick->DynamicStates.Num(); InstanceIndex != E; ++InstanceIndex)
465 {
466 FInstance& Instance = Instances[InstanceIndex];
467 if (Instance.CreateTick > Tick->Index || Instance.DeleteTick <= Tick->Index)
468 {
469 continue;
470 }
471 check(!Instance.RendDynamicState.IsSet());
472 uint32 DynamicStateIndex = Tick->DynamicStates[InstanceIndex];
473 FStateStreamCopyContext Context;
474
475 // Why do we use the render dynamic state from the prev tick and not interpolate from prev->next?
476 Instance.RendDynamicState.Emplace(Context, DynamicStates[DynamicStateIndex]);
477 Render_OnCreateInline(Instance.StaticState, *Instance.RendDynamicState, Instance.UserData, false);
478 }
479 }
480 // If the render update is disable and if it has changed we are in deletion mode
481 // We need to to reset all th instances render dynamic state
482 else if (RendTick)
483 {
484 // Cleanup all instances
485 for (uint32 InstanceIndex = 0, E = RendTick->DynamicStates.Num(); InstanceIndex != E; ++InstanceIndex)
486 {
487 FInstance& Instance = Instances[InstanceIndex];
488 if (!Instance.RendDynamicState)
489 {
490 continue;
491 }
492 Render_OnDestroyInline(Instance.StaticState, *Instance.RendDynamicState, Instance.UserData);
493 Instance.RendDynamicState.Reset();
494 checkf(!Instance.UserData, TEXT("Make sure to clear UserData in Render_OnDestroy"));
495 }
496 RendTick = nullptr;
497 }
498 return;
499 }
500 else if (!RendEnabled)
501 {
502 return;
503 }
504
505 if (!NewestAvailableTick || AbsoluteTime == RendTime)
506 {
507 return;
508 }
509
510 check(AbsoluteTime > RendTime); // Only play forward for now
511
512 uint32 PrevTick = 0;
513 bool IsFirstTick = false;
514 if (RendTick)
515 {
516 PrevTick = RendTick->Index;
517 }
518 else
519 {
520 RendTick = OldestAvailableTick;
521 IsFirstTick = true;
522 }
523
524 StateStreamTime PrevTime = RendTime;
525 RendTime = AbsoluteTime;
526
527 // We are still inside the same tick
528 if (RendTime <= RendTick->Time)
529 {
530 // Just interpolate or apply RendTick
531 ApplyChanges(*RendTick, RendTime, PrevTick, PrevTime, RendTick->ModifiedInstances);
532 return;
533 }
534
535 // We were at exact end of last handled tick.. move into next
536 if (PrevTime == RendTick->Time)
537 {
538 FTick* NewRendTick = RendTick->NextTick;
539
540 if (!NewRendTick) // We've caught up with game.. set RendTime back to PrevTime and return
541 {
542 RendTime = PrevTime;
543 return;
544 }
545
546 if (!IsFirstTick)
547 RendTick = NewRendTick;
548
549 // We don't need to include RendTick Modifications.
550 if (RendTime <= RendTick->Time)
551 {
552 // Just interpolate or apply RendTick
553 ApplyChanges(*RendTick, RendTime, PrevTick, PrevTime, RendTick->ModifiedInstances);
554 return;
555 }
556 }
557
558 // We are overlapping between two or more ticks
559 TBitArray<> ModifiedInstances(RendTick->ModifiedInstances);
560 while (RendTick->Time < RendTime)
561 {
562 if (!RendTick->NextTick)
563 {
564 RendTime = RendTick->Time;
565 break;
566 }
567 RendTick = RendTick->NextTick;
568 ModifiedInstances.CombineWithBitwiseOR(RendTick->ModifiedInstances, EBitwiseOperatorFlags::MaxSize);
569 }
570
571 ApplyChanges(*RendTick, RendTime, PrevTick, PrevTime, ModifiedInstances);
572}

597template<typename Settings>
598void TStateStream<Settings>::Render_GarbageCollect()
599{
600 if (!OldestAvailableTick)
601 {
602 return;
603 }
604
605 FTick* RendTickUsed = RendTick;
606 if (RendTick && RendTick->Time != RendTime && RendTick->PrevTick)
607 RendTickUsed = RendTick->PrevTick;
608
609 if (!RendTickUsed)
610 {
611 RendTickUsed = OldestAvailableTick->NextTick;
612 if (!RendTickUsed)
613 {
614 return;
615 }
616 }
617
618 FTick* Tick = OldestAvailableTick;
619 while (Tick != RendTickUsed)
620 {
621 FTick* Next = Tick->NextTick;
622 check (Next);
623
624 // We can remove all DynamicStates that are different in Next tick since we know this is the last tick using the state
625 for (TConstSetBitIterator<> It(Next->ModifiedInstances); It; ++It)
626 {
627 uint32 InstanceIndex = It.GetIndex();
628 if (InstanceIndex >= uint32(Tick->DynamicStates.Num()))
629 {
630 continue;
631 }
632
633 // If instance was created in next tick we ignore this. If instance is deleted in tick we handle it further down
634 FInstance& Instance = Instances[InstanceIndex];
635 if (Instance.DeleteTick == Tick->Index || Instance.CreateTick == Next->Index)
636 {
637 continue;
638 }
639
640 uint32 DynamicStateIndex = Tick->DynamicStates[InstanceIndex];
641 if (DynamicStateIndex == Next->DynamicStates[InstanceIndex])
642 {
643 continue;
644 }
645
646 DynamicStates.Remove(DynamicStateIndex);
647 }
648
649 // Remove all instances who was deleted in Tick
650 for (TConstSetBitIterator<> It(Tick->ModifiedInstances); It; ++It)
651 {
652 uint32 InstanceIndex = It.GetIndex();
653 FInstance& Instance = Instances[InstanceIndex];
654
655 if (Instance.DeleteTick != Tick->Index)
656 {
657 continue;
658 }
659
660 DynamicStates.Remove(Tick->DynamicStates[InstanceIndex]);
661 Instance.RendDynamicState.Reset();
662 Instances.Remove(InstanceIndex);
663 }
664
665 delete Tick;
666 Tick = Next;
667 }
668 OldestAvailableTick = Tick;
669
670}

715template<typename Settings>
716void TStateStream<Settings>::ApplyChanges(const FTick& Tick, StateStreamTime Time, uint32 PrevTickIndex, StateStreamTime PrevTime, const TBitArray<>& ModifiedInstances)
717{
718 // TODO: This will do lots of allocations
719 struct FCreateInfo { FInstance* Instance; bool IsDestroyInSameFrame; };
720 TArray<FCreateInfo> Creates;
721 TArray<FInstance*> Updates;
722 TArray<FInstance*> Destroys;
723
724 for (TConstSetBitIterator<> It(ModifiedInstances); It; ++It)
725 {
726 uint32 InstanceIndex = It.GetIndex();
727 FInstance& Instance = Instances[InstanceIndex];
728
729 if (Instance.DeleteTick <= PrevTickIndex) // Already deleted
730 {
731 continue;
732 }
733
734 FStaticState& Ss = Instance.StaticState;
735
736 bool IsCreate = Instance.CreateTick > PrevTickIndex && Instance.CreateTick <= Tick.Index;
737 bool IsDestroy = Instance.DeleteTick > PrevTickIndex && Instance.DeleteTick <= Tick.Index;
738
739 if (IsCreate)
740 {
741 if (Settings::SkipCreatingDeletes && IsDestroy)
742 {
743 continue;
744 }
745
746 check(!Instance.RendDynamicState.IsSet());
747 uint32 DynamicStateIndex = Tick.DynamicStates[InstanceIndex];
748 FStateStreamCopyContext Context;
749 Instance.RendDynamicState.Emplace(Context, DynamicStates[DynamicStateIndex]);
750 }
751 else if (Instance.CreateTick == Tick.Index) // Still in the tick it was created, no interpolation possible
752 {
753 continue;
754 }
755
756 FDynamicState& Ds = *Instance.RendDynamicState;
757
758 if (Tick.Time == Time || !Tick.PrevTick)
759 {
760 if (!IsCreate)
761 {
762 FStateStreamCopyContext Context;
763 Ds.Apply(Context, DynamicStates[Tick.DynamicStates[InstanceIndex]]);
764 }
765 }
766 else
767 {
768 const FTick& PrevTick = *Tick.PrevTick;
769 uint32 FromIndex = PrevTick.DynamicStates[InstanceIndex];
770 uint32 ToIndex = Tick.DynamicStates[InstanceIndex];
771
772 StateStreamTime DeltaTime = Tick.Time - PrevTick.Time;
773 StateStreamTime TimeInTo = Time - PrevTick.Time;
774 double Factor = (DeltaTime == 0.0) ? 0.0f : double(TimeInTo) / DeltaTime;
775
776 FDynamicState& From = DynamicStates[FromIndex];
777 FDynamicState& To = DynamicStates[ToIndex];
778
779 FStateStreamInterpolateContext Context;
780 Context.Factor = Factor;
781 Ds.Interpolate(Context, From, To);
782 }
783
784 if (IsCreate)
785 {
786 Creates.Add({&Instance, IsDestroy});
787 }
788
789 if (IsDestroy)
790 {
791 if (IsCreate)
792 {
793 DeferredDestroys.Add(&Instance);
794 }
795 else
796 {
797 Destroys.Add(&Instance);
798 }
799 continue;
800 }
801
802 if (!IsCreate)
803 {
804 Updates.Add(&Instance);
805 }
806 }
807
808 for (FCreateInfo& Info : Creates)
809 {
810 FInstance& Instance = *Info.Instance;
811 Render_OnCreateInline(Instance.StaticState, *Instance.RendDynamicState, Instance.UserData, Info.IsDestroyInSameFrame);
812 }
813
814 for (FInstance* Instance : Updates)
815 {
816 Render_OnUpdateInline(Instance->StaticState, *Instance->RendDynamicState, Instance->UserData);
817 }
818
819 for (FInstance* Instance : Destroys)
820 {
821 Render_OnDestroyInline(Instance->StaticState, *Instance->RendDynamicState, Instance->UserData);
822 Instance->RendDynamicState.Reset();
823 checkf(!Instance->UserData, TEXT("Make sure to clear UserData in Render_OnDestroy"));
824 }
825}
