92 // StartCycles can never be greater than current time as there is no way to construct a timeout starting in the future

98 {
99 if (WillNeverExpire())

102 }
103
104 // We convert to double separately to avoid underflow on the cycles
105 // This could also be done with some branches or treating cycles as signed int64
106 return GetTimeoutSeconds() - GetElapsedSeconds();
107 }

113 }

116 void SetTimeoutSeconds(double TimeoutSeconds)
117 {
118 if (TimeoutSeconds <= 0.0)
119 {
120 SetToExpired();
121 }
122 else
123 {
124 TimeoutCycles = FPlatformTime::SecondsToCycles64(TimeoutSeconds);
125 }
126 }
127
129 void ModifyTimeoutSeconds(double DeltaTimeoutSeconds)
130 {
131 if (IsAlwaysExpired() || WillNeverExpire())
132 {
133 return;
134 }
135
136 if (DeltaTimeoutSeconds >= 0.0)
137 {
138 TimeoutCycles += FPlatformTime::SecondsToCycles64(DeltaTimeoutSeconds);
139 }
140 else
141 {
142 uint64 RemovedCycles = FPlatformTime::SecondsToCycles64(-DeltaTimeoutSeconds);
143 if (RemovedCycles >= TimeoutCycles)
144 {
145 SetToExpired();
146 }
147 else
148 {
149 TimeoutCycles -= RemovedCycles;
150 }
151 }
152 }

216 friend bool operator==(FTimeout Left, FTimeout Right)
217 {
218 // Timeout cycles need to match which handles differentiating between always and never
219 // For normal timeouts, also check the start cycles
220 return Left.TimeoutCycles == Right.TimeoutCycles
221 && (Left.WillNeverExpire() || Left.IsAlwaysExpired() || Left.StartCycles == Right.StartCycles);
222 }
223
