321inline void SerializeForLog(FCbWriter& Writer, const TOptional<ValueType>& Optional)
322{
323 Writer.BeginArray();

484template <typename LogType, typename... FieldArgTypes>
485UE_COLD UE_DEBUG_SECTION void LogWithFields(const FLogCategoryBase& Category, const LogType& Log, FieldArgTypes&&... FieldArgs)
486{
487 if constexpr (sizeof...(FieldArgTypes) == 0)
488 {
489 LogWithNoFields(Category, Log);
490 }
491 else
492 {
493 constexpr int32 FieldCount = FLogFieldCreator::template GetCount<FieldArgTypes...>();
494 static_assert(FieldCount > 0);
495 FLogField Fields[FieldCount];
496 FLogFieldCreator::Create(Fields, (FieldArgTypes&&)FieldArgs...);
497 LogWithFieldArray(Category, Log, Fields, FieldCount);
498 }
499}
