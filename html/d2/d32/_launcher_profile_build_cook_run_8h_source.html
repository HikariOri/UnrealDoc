467 virtual bool ShouldBuild() override
468 {
469 bool bBuild = true;
470 if (BuildMode == ELauncherProfileBuildModes::DoNotBuild)
471 {
472 bBuild = false;
473 }
474 else if (BuildMode == ELauncherProfileBuildModes::Auto)
475 {
476 if (FApp::GetEngineIsPromotedBuild())
477 {
478 bBuild = false;
479
480 TArray<FString> TargetPlatformNames = FindPlatforms();
481 for (const FString& TargetPlatformName : TargetPlatformNames)
482 {
483 // Get the target we're building for
484 const ITargetPlatform* TargetPlatform = GetTargetPlatformManager()->FindTargetPlatform(TargetPlatformName);
485 const PlatformInfo::FTargetPlatformInfo& PlatformInfo = TargetPlatform->GetTargetPlatformInfo();
486
487 // Figure out which target we're building
488 FString ReceiptDir;
489 FString TargetName;
490 if (TryGetDefaultTargetName(FPaths::GetProjectFilePath(), PlatformInfo.PlatformType, TargetName))
491 {
492 ReceiptDir = FPaths::GetPath(FPaths::GetProjectFilePath());
493 }
494 else if (TryGetDefaultTargetName(FString(), PlatformInfo.PlatformType, TargetName))
495 {
496 FText Reason;
497 if (TargetPlatform->RequiresTempTarget(false, BuildConfiguration, false, Reason))
498 {
499 UE_LOGF(LogLauncherProfile, Log, "Project requires temp target (%ls)", *Reason.ToString());
500 ReceiptDir = FPaths::GetPath(FPaths::GetProjectFilePath());
501 }
502 else
503 {
504 UE_LOGF(LogLauncherProfile, Log, "Project does not require temp target");
505 ReceiptDir = FPaths::EngineDir();
506 }
507 }
508 else
509 {
510 UE_LOGF(LogLauncherProfile, Log, "Unable to find any targets for platform %ls - forcing build", *TargetPlatformName);
511 bBuild = true;
512 break;
513 }
514
515 // Check if the existing target is valid
516 FString BuildPlatform = PlatformInfo.DataDrivenPlatformInfo->UBTPlatformString;
517 if (!HasPromotedTarget(*ReceiptDir, *TargetName, *BuildPlatform, BuildConfiguration, nullptr))
518 {
519 bBuild = true;
520 break;
521 }
522 }
523 }
524 }
525 return bBuild;
526 }

820 TArray<FString> FindPlatforms()
821 {
822 TArray<FString> Platforms;
823 if (GetCookMode() == ELauncherProfileCookModes::ByTheBook)
824 {
825 Platforms = GetCookedPlatforms();
826 }
827
828 // determine deployment platforms
829 ILauncherDeviceGroupPtr DeviceGroup = GetDeployedDeviceGroup();
830 FName Variant = NAME_None;
831
832 // Loading the Device Proxy Manager to get the needed Device Manager.
833 ITargetDeviceServicesModule& DeviceServiceModule = FModuleManager::LoadModuleChecked<ITargetDeviceServicesModule>(TEXT("TargetDeviceServices"));
834 TSharedRef<ITargetDeviceProxyManager> DeviceProxyManager = DeviceServiceModule.GetDeviceProxyManager();
835
836 if (DeviceGroup.IsValid() && Platforms.Num() < 1)
837 {
838 const TArray<FString>& Devices = DeviceGroup->GetDeviceIDs();
839 // for each deployed device...
840 for (int32 DeviceIndex = 0; DeviceIndex < Devices.Num(); ++DeviceIndex)
841 {
842 const FString& DeviceId = Devices[DeviceIndex];
843
844 TSharedPtr<ITargetDeviceProxy> DeviceProxy = DeviceProxyManager->FindProxyDeviceForTargetDevice(DeviceId);
845
846 if (DeviceProxy.IsValid())
847 {
848 // add the platform
849 Variant = DeviceProxy->GetTargetDeviceVariant(DeviceId);
850 Platforms.AddUnique(DeviceProxy->GetTargetPlatformName(Variant));
851 }
852 }
853 }
854
855 return Platforms;
856 }
