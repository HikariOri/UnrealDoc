110struct FZenWorkspaces
111{

117 };

119 struct Workspace
120 {
121 FString Id;
122 FString BaseDir;
123 bool bDynamicShare;
124
125 TArray<Share> WorkspaceShares;
126 };
127
128 TArray<Workspace> ZenWorkspaces;
129 bool bIsValid = false;
130
131 bool operator==(const FZenWorkspaces& Other) const
132 {
133 if (ZenWorkspaces.Num() != Other.ZenWorkspaces.Num())
134 {
135 return false;
136 }
137
138 for (int32 Idx = 0; Idx < ZenWorkspaces.Num(); ++Idx)
139 {
140 const Workspace& CurrentWorkspace = ZenWorkspaces[Idx];
141 const Workspace& OtherWorkspace = Other.ZenWorkspaces[Idx];
142
143 if (CurrentWorkspace.WorkspaceShares.Num() != OtherWorkspace.WorkspaceShares.Num())
144 {
145 return false;
146 }
147
148 if (CurrentWorkspace.Id != OtherWorkspace.Id ||
149 CurrentWorkspace.BaseDir != OtherWorkspace.BaseDir ||
150 CurrentWorkspace.bDynamicShare != OtherWorkspace.bDynamicShare)
151 {
152 return false;
153 }
154
155 for (int32 ShareIdx = 0; ShareIdx < CurrentWorkspace.WorkspaceShares.Num(); ++ShareIdx)
156 {
157 const Share& CurrentShare = CurrentWorkspace.WorkspaceShares[ShareIdx];
158 const Share& OtherShare = OtherWorkspace.WorkspaceShares[ShareIdx];
159
160 if (CurrentShare.Id != OtherShare.Id ||
161 CurrentShare.Dir != OtherShare.Dir ||
162 CurrentShare.Alias != OtherShare.Alias)
163 {
164 return false;
165 }
166 }
167 }
168
169 return true;
170 }
