11struct FTextureMemoryStats

13 // Hardware state (never change after device creation):
14
15 // -1 if unknown, in bytes
16 int64 DedicatedVideoMemory = -1;
17
18 // -1 if unknown, in bytes
19 int64 DedicatedSystemMemory = -1;
20
21 // -1 if unknown, in bytes
22 int64 SharedSystemMemory = -1;
23
24 // Total amount of "graphics memory" that we think we can use for all our graphics resources, in bytes. -1 if unknown.
25 int64 TotalGraphicsMemory = -1;
26
27 // Size of memory allocated to streaming textures, in bytes
28 uint64 StreamingMemorySize = 0;
29
30 // Size of memory allocated to non-streaming textures, in bytes
31 uint64 NonStreamingMemorySize = 0;
32
33 // Size of the largest memory fragment, in bytes
34 int64 LargestContiguousAllocation = 0;
35
36 // 0 if streaming pool size limitation is disabled, in bytes
37 int64 TexturePoolSize = 0;
38
39 int64 GetTotalDeviceWorkingMemory() const
40 {
41 if (GRHIDeviceIsIntegrated)
42 {
43 // Max in case the device failed to report the available working memory
44 return FMath::Max(TotalGraphicsMemory, DedicatedVideoMemory);
45 }
46
47 return DedicatedVideoMemory;
48 }
49
50 bool AreHardwareStatsValid() const
51 {
52 // pardon the redundancy, have a broken compiler (__EMSCRIPTEN__) that needs these types spelled out...
53 return ((int64)DedicatedVideoMemory >= 0 && (int64)DedicatedSystemMemory >= 0 && (int64)SharedSystemMemory >= 0);
54 }
55
56 bool IsUsingLimitedPoolSize() const
57 {
58 return TexturePoolSize > 0;
59 }
60
61 int64 ComputeAvailableMemorySize() const
62 {
63 return FMath::Max<int64>(TexturePoolSize - StreamingMemorySize, 0);
64 }
65};

182struct FRHIMemoryStats
183{
184 // Budget assigned by the OS/driver. Total memory the app should use.
185 uint64 BudgetLocal = 0; // GPU VRAM budget
186 uint64 BudgetSystem = 0; // System/host memory budget
187
188 // Currently used memory.
189 uint64 UsedLocal = 0; // GPU VRAM used
190 uint64 UsedSystem = 0; // System/host memory used
191
192 // Over-budget memory
193 uint64 DemotedLocal = 0;
194 uint64 DemotedSystem = 0;
195
196 // Available memory within budget
197 uint64 AvailableLocal = 0;
198 uint64 AvailableSystem = 0;
199
200 bool IsOverBudget() const
201 {
202 return DemotedLocal > 0 || DemotedSystem > 0;
203 }
204
205 bool IsValid() const
206 {
207 return BudgetLocal > 0 || BudgetSystem > 0;
208 }
209};
