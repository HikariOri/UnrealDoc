26template<typename T>
27class TLocklessGrowingStorage
28{
29public:
30
31 void Push(T&& In)
32 {
33 Queue.ProduceItem(MoveTemp(In));
34 ItemCount.fetch_add(1, std::memory_order_relaxed);
35 }
36
37 void ExtractAll(TArray<T>& Out)
38 {
39 const int32 PreviousNum = Out.Num();
40 Queue.ConsumeAllFifo([&Out](T Item)
41 {
42 Out.Add(MoveTemp(Item));
43 });
44 // Subtract only what we drained so concurrent producers' increments stand. ItemCount
45 // is informational (memory stats) and still races with live pushes by construction.
46 const int32 DrainedCount = Out.Num() - PreviousNum;
47 ItemCount.fetch_sub(DrainedCount, std::memory_order_relaxed);
48 }
49
50 uint32 GetAllocatedSize() const
51 {
52 // TConsumeAllMpmcQueue allocates a private FNode (next-pointer + storage for T)
53 // per outstanding item. Approximation: treat each node as a next-ptr + the item.
54 // ItemCount is a relaxed counter racing with live producers/consumers and can briefly
55 // observe negative skew (ExtractAll's fetch_sub completing before a concurrent Push's
56 // fetch_add). Clamp at zero so the unsigned multiplication does not produce an
57 // astronomical reported size in memory stats.
58 constexpr uint32 PerItemBytes = sizeof(T) + sizeof(void*);
59 const int32 ClampedItemCount = FMath::Max<int32>(0, ItemCount.load(std::memory_order_relaxed));
60 return static_cast<uint32>(ClampedItemCount) * PerItemBytes;
61 }
62
63private:
64 UE::TConsumeAllMpmcQueue<T> Queue;
65 std::atomic<int32> ItemCount{0};
66};

285 template <typename T>
286 struct TSimpleSparseArray
287 {
288 int32 Add(T&& InElement)
289 {
290 if (UsedElementsCount == UsedElements.Num())
291 {
292 UsedElements.Add(false, GSimpleStreamableAssetManagerSparseArrayGrowSize);
293 Elements.AddDefaulted(GSimpleStreamableAssetManagerSparseArrayGrowSize);
294 }
295 const int32 Index = UsedElements.FindAndSetFirstZeroBit(FreeElementIndexHint);
296 check(Index != INDEX_NONE);
297 FreeElementIndexHint = Index + 1;
298 ++UsedElementsCount;
299 Elements[Index] = MoveTemp(InElement);
300 return Index;
301 }
302
303 void Reset(int32 Index)
304 {
305 if (UsedElements.Num() > Index && UsedElements[Index])
306 {
307 UsedElements[Index] = false;
308 FreeElementIndexHint = FMath::Min(FreeElementIndexHint, Index);
309 Elements[Index] = T{};
310 --UsedElementsCount;
311 }
312 }
313
314 void Empty()
315 {
316 FreeElementIndexHint = 0;
317 UsedElementsCount = 0;
318 UsedElements.Empty();
319 Elements.Empty();
320 }
321
322 int32 Num() const { return UsedElementsCount; }
323
324 SIZE_T GetAllocatedSize(void) const
325 {
326 return sizeof(TSimpleSparseArray)
327 + UsedElements.GetAllocatedSize()
328 + Elements.GetAllocatedSize();
329 }
330
331 // Returns a contiguous view over [0..LastUsed+1) covering every set bit in UsedElements
332 // up to the highest, or an empty view when no bit is set (FindLast returns INDEX_NONE,
333 // so the view length is 0). The view INCLUDES holes at unused indices in between --
334 // consumers must filter via the per-element validity check (typically
335 // ObjectRegistrationIndex == INDEX_NONE) before using an entry. When the array is
336 // sparse the per-iteration cost still scales with the prefix length rather than the
337 // used-element count; a future tightening could iterate UsedElements directly via
338 // TConstSetBitIterator on the screen-size hot path. Cheap when UsedElementsCount is
339 // close to LastUsed+1.
340 TConstArrayView<T> GetSparseView() const
341 {
342 const int32 Max = UsedElements.FindLast(true);
343 return TConstArrayView<T>{ Elements.GetData(), Max + 1};
344 }
345 private:
346 int32 FreeElementIndexHint = 0;
347 int32 UsedElementsCount = 0;
348 TBitArray<> UsedElements;
349 TArray<T> Elements;
350 };
