143struct FGraphTraversalCounter
144{
145private:
146 // Number of the last external frame where we updated the counter (total global frame count)
147 uint64 LastSyncronizedFrame;
148
149 // Number of counter increments since last reset
150 int16 Counter;
151
152 // Number of external frames that URO is currently scheduled to skip
153 int16 SkipFrames;
154
155public:
156
157 FGraphTraversalCounter()
158 : LastSyncronizedFrame(INDEX_NONE)
159 , Counter(INDEX_NONE)
160 , SkipFrames(0)
161
162 {}
163
164 int16 Get() const
165 {
166 return Counter;
167 }
168
169 bool HasEverBeenUpdated() const
170 {
171 return (Counter != INDEX_NONE) && (LastSyncronizedFrame != INDEX_NONE);
172 }
173
175 void Increment()
176 {
177 Counter++;
178 LastSyncronizedFrame = GFrameCounter;
179
180 // Avoid wrapping over back to INDEX_NONE, as this means 'never been traversed'
181 if (Counter == INDEX_NONE)
182 {
183 Counter++;
184 }
185 }
186
188 void SetMaxSkippedFrames(int16 InMaxSkippedFrames)
189 {
190 SkipFrames = InMaxSkippedFrames;
191 }
192
194 int16 GetMaxSkippedFrames() const
195 {
196 return SkipFrames;
197 }
198
200 void Reset()
201 {
202 Counter = INDEX_NONE;
203 LastSyncronizedFrame = INDEX_NONE;
204 SkipFrames = 0;
205 }
206
208 void SynchronizeWith(const FGraphTraversalCounter& InMasterCounter)
209 {
210 Counter = InMasterCounter.Counter;
211 LastSyncronizedFrame = InMasterCounter.LastSyncronizedFrame;
212 SkipFrames = InMasterCounter.SkipFrames;
213 }
214
216 bool IsSynchronized_Counter(const FGraphTraversalCounter& InOtherCounter) const
217 {
218 return HasEverBeenUpdated() && Counter == InOtherCounter.Counter;
219 }
220
222 bool IsSynchronized_Frame(const FGraphTraversalCounter& InOtherCounter) const
223 {
224 return HasEverBeenUpdated() && LastSyncronizedFrame == InOtherCounter.LastSyncronizedFrame;
225 }
226
228 bool IsSynchronized_All(const FGraphTraversalCounter& InOtherCounter) const
229 {
230 return HasEverBeenUpdated() && Counter == InOtherCounter.Counter && LastSyncronizedFrame == InOtherCounter.LastSyncronizedFrame;
231 }
232
234 bool WasSynchronizedCounter(const FGraphTraversalCounter& InOtherCounter) const
235 {
236 // Test if we're currently in sync with our master counter
237 if(IsSynchronized_Counter(InOtherCounter))
238 {
239 return true;
240 }
241
242 // If not, test if the Master Counter is a frame ahead of us
243 FGraphTraversalCounter TestCounter(*this);
244 TestCounter.Increment();
245
246 return TestCounter.IsSynchronized_Counter(InOtherCounter);
247 }
248
258 bool WasSynchronizedLastFrame(const FGraphTraversalCounter& InOtherCounter) const
259 {
260 // Test if we're currently in sync with our master counter
261 if(IsSynchronized_Frame(InOtherCounter))
262 {
263 return true;
264 }
265
266 // If we aren't synced then check if we were synced last frame
267 return LastSyncronizedFrame >= (GFrameCounter - SkipFrames) - 1;
268 }
269};

458// Used by UAnimSequenceBase::SortNotifies() to sort its Notifies array
459inline bool FAnimNotifyEvent::operator<(const FAnimNotifyEvent& Other) const
460{
461 float ATime = GetTriggerTime();
462 float BTime = Other.GetTriggerTime();
463
464 // When we've got the same time sort on the track index. Explicitly
465 // using SMALL_NUMBER here incase the underlying default changes as
466 // notifies can have an offset of KINDA_SMALL_NUMBER to be consider
467 // distinct
468 if (FMath::IsNearlyEqual(ATime, BTime, UE_SMALL_NUMBER))
469 {
470 return TrackIndex < Other.TrackIndex;
471 }
472 else
473 {
474 return ATime < BTime;
475 }
476}
