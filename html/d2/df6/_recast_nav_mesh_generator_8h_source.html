217struct FRcTileBox
218{
219 int32 XMin, XMax, YMin, YMax;
220
221 FRcTileBox(const FBox& UnrealBounds, const FVector& RcNavMeshOrigin, const FVector::FReal TileSizeInWorldUnits)
222 {
223 check(TileSizeInWorldUnits > 0);
224 checkf(!RcNavMeshOrigin.ContainsNaN(), TEXT("%hs: RcNavMeshOrigin ContainsNaN"), __FUNCTION__);
225 checkf(!UnrealBounds.ContainsNaN(), TEXT("%hs: UnrealBounds ContainsNaN"), __FUNCTION__);
226
227 if (!ensureMsgf(UnrealBounds.IsValid, TEXT("%hs: UnrealBounds !IsValid"), __FUNCTION__))
228 {
229 // This is a bug but we'll handle it as well as we can.
230
231 // Invalid bounds, set to empty range.
232 // Max is set to -1, because the range is inclusive, used like: for (int32 y = TileBox.YMin; y <= TileBox.YMax; ++y)
233 XMin = 0;
234 XMax = -1;
235 YMin = 0;
236 YMax = -1;
237 return;
238 }
239
240 auto CalcMaxCoordExclusive = [](const FVector::FReal MaxAsFloat, const int32 MinCoord) -> int32
241 {
242 FVector::FReal UnusedIntPart;
243 // If MaxCoord falls exactly on the boundary of a tile
244 if (FMath::Modf(MaxAsFloat, &UnusedIntPart) == 0)
245 {
246 // Return the lower tile
247 return FMath::Max(ClampToInt32(FMath::FloorToInt(MaxAsFloat) - 1), MinCoord);
248 }
249 // Otherwise use default behaviour
250 return ClampToInt32(FMath::FloorToInt(MaxAsFloat));
251 };
252
253 const FBox RcAreaBounds = Unreal2RecastBox(UnrealBounds);
254 XMin = ClampToInt32(FMath::FloorToInt((RcAreaBounds.Min.X - RcNavMeshOrigin.X) / TileSizeInWorldUnits));
255 XMax = CalcMaxCoordExclusive((RcAreaBounds.Max.X - RcNavMeshOrigin.X) / TileSizeInWorldUnits, XMin);
256 YMin = ClampToInt32(FMath::FloorToInt((RcAreaBounds.Min.Z - RcNavMeshOrigin.Z) / TileSizeInWorldUnits));
257 YMax = CalcMaxCoordExclusive((RcAreaBounds.Max.Z - RcNavMeshOrigin.Z) / TileSizeInWorldUnits, YMin);
258 }
259
260 inline bool Contains(const FIntPoint& Point) const
261 {
262 return Point.X >= XMin && Point.X <= XMax
263 && Point.Y >= YMin && Point.Y <= YMax;
264 }
265
266 static inline int32 ClampToInt32(const int64 Value)
267 {
268#if !NO_LOGGING
269 UE_CLOG(!IntFitsIn<int32>(Value), LogNavigation, Warning,
270 TEXT("FRcTileBox clamped a NavMesh transform value to fit in int32. Old value: %" UINT64_FMT), Value);
271#endif // !NO_LOGGING
272
273 return static_cast<int32>(FMath::Clamp(Value, MIN_int32, MAX_int32));
274 }
275};
