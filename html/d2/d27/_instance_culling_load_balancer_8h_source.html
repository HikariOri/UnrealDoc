19class FInstanceCullingLoadBalancerBase

21public:
22 static constexpr uint32 ThreadGroupSize = 64U;
23
24 // Number of bits needed for prefix sum storage
25 static constexpr uint32 PrefixBits = 6U;//ILog2Const(uint32(ThreadGroupSize));
26 static_assert((1U << PrefixBits) == ThreadGroupSize, "ThreadGroupSize and PrefixBits must be kept in sync");
27 static constexpr uint32 PrefixBitMask = (1U << PrefixBits) - 1U;
28
29 static constexpr uint32 NumInstancesItemBits = PrefixBits + 1U;
30 static constexpr uint32 NumInstancesItemMask = (1U << NumInstancesItemBits) - 1U;
31

33 struct FPackedBatch
34 {
35 uint32 FirstItem_NumItems;
36 };
37
38 FPackedBatch PackBatch(uint32 FirstItem, uint32 NumItems)
39 {
40 checkSlow(NumItems < (1U << NumInstancesItemBits));
41 checkSlow(FirstItem < (1U << (32U - NumInstancesItemBits)));
42
43 return FPackedBatch{ (FirstItem << NumInstancesItemBits) | (NumItems & NumInstancesItemMask) };
44 }
45
46 struct FPackedItem
47 {
48 // packed 32-NumInstancesItemBits:NumInstancesItemBits - need one more bit for the case where one item has ThreadGroupSize work to do
49 uint32 InstanceDataOffset_NumInstances;
50 // packed 32-PrefixBits:PrefixBits
51 uint32 Payload_BatchPrefixOffset;
52 };
53 FPackedItem PackItem(uint32 InstanceDataOffset, uint32 NumInstances, uint32 Payload, uint32 BatchPrefixSum)
54 {
55 checkSlow(NumInstances < (1U << NumInstancesItemBits));
56 checkSlow(InstanceDataOffset < (1U << (32U - NumInstancesItemBits)));
57 checkSlow(BatchPrefixSum < (1U << PrefixBits));
58 checkSlow(Payload < (1U << (32U - PrefixBits)));
59
60 return FPackedItem
61 {
62 (InstanceDataOffset << NumInstancesItemBits) | (NumInstances & NumInstancesItemMask),
63 (Payload << PrefixBits) | (BatchPrefixSum & PrefixBitMask)
64 };
65 }

119template <typename InAllocatorType = FDefaultAllocator>
120class TInstanceCullingLoadBalancer : public FInstanceCullingLoadBalancerBase
121{
122public:
123 using AllocatorType = InAllocatorType;
124
125 void ReserveStorage(int32 NumBatches, int32 NumItems)
126 {
127 Data->Batches.Empty(NumBatches);
128 Data->Items.Empty(NumItems);
129 }
130
134 void Add(uint32 InstanceDataOffset, uint32 NumInstanceDataEntries, uint32 Payload)
135 {
136 uint32 InstancesAdded = 0;
137 while (InstancesAdded < NumInstanceDataEntries)
138 {
139 uint32 MaxInstancesThisBatch = ThreadGroupSize - CurrentBatchPrefixSum;
140
141 if (MaxInstancesThisBatch > 0)
142 {
143 const uint32 NumInstancesThisItem = FMath::Min(MaxInstancesThisBatch, NumInstanceDataEntries - InstancesAdded);
144
145 Data->Items.Add(PackItem(InstanceDataOffset + InstancesAdded, NumInstancesThisItem, Payload, CurrentBatchPrefixSum));
146 if (CurrentBatchNumItems * PrefixBits < sizeof(CurrentBatchPackedPrefixSum) * 8U)
147 {
148 CurrentBatchPackedPrefixSum |= CurrentBatchPackedPrefixSum << (PrefixBits * CurrentBatchNumItems);
149 }
150 CurrentBatchNumItems += 1U;
151 InstancesAdded += NumInstancesThisItem;
152 CurrentBatchPrefixSum += NumInstancesThisItem;
153
154 }
155
156 // Flush batch if it is not possible to add any more items (for one of the reasons)
157 if (MaxInstancesThisBatch <= 0U || CurrentBatchPrefixSum >= ThreadGroupSize)
158 {
159 Data->Batches.Add(PackBatch(CurrentBatchFirstItem, CurrentBatchNumItems));
160 CurrentBatchFirstItem = uint32(Data->Items.Num());
161 CurrentBatchPrefixSum = 0u;
162 CurrentBatchNumItems = 0U;
163 CurrentBatchPackedPrefixSum = 0U;
164 }
165 }
166 TotalInstances += InstancesAdded;
167 }
168
169 bool IsEmpty() const
170 {
171 return Data->Items.IsEmpty();
172 }
173
174 FGPUData Upload(FRDGBuilder& GraphBuilder, int32 NumGroupsPerBatch = 1)
175 {
176 FinalizeBatches();
177
178 const auto& LocalBatches = Data->Batches;
179 const auto& LocalItems = Data->Items;
180
181 GraphBuilder.AllocObject<TRefCountPtr<FData>>(Data);
182
183 return FInstanceCullingLoadBalancerBase::Upload(GraphBuilder, LocalBatches, LocalItems, ERDGInitialDataFlags::NoCopy, NumGroupsPerBatch);
184 }
185
186 /* Const variant that assumes the batches have already been finalized */
187 FGPUData UploadFinalized(FRDGBuilder& GraphBuilder, int32 NumGroupsPerBatch = 1) const
188 {
189 check(CurrentBatchNumItems == 0);
190
191 const auto& LocalBatches = Data->Batches;
192 const auto& LocalItems = Data->Items;
193
194 GraphBuilder.AllocObject<TRefCountPtr<FData>>(Data);
195
196 return FInstanceCullingLoadBalancerBase::Upload(GraphBuilder, LocalBatches, LocalItems, ERDGInitialDataFlags::NoCopy, NumGroupsPerBatch);
197 }
198
202 void FinalizeBatches()
203 {
204 if (CurrentBatchNumItems != 0)
205 {
206 Data->Batches.Add(PackBatch(CurrentBatchFirstItem, CurrentBatchNumItems));
207 CurrentBatchNumItems = 0;
208 }
209 }
210
216 FIntVector GetWrappedCsGroupCount(int32 NumGroupsPerBatch = 1) const
217 {
218 return FInstanceCullingLoadBalancerBase::GetWrappedCsGroupCount(Data->Batches, NumGroupsPerBatch);
219 }
220
221 const TArray<FPackedBatch, AllocatorType> &GetBatches() const
222 {
223 check(CurrentBatchNumItems == 0);
224 return Data->Batches;
225 };
226
227 const TArray<FPackedItem, AllocatorType> &GetItems() const
228 {
229 check(CurrentBatchNumItems == 0);
230 return Data->Items;
231 }
232
233 uint32 GetTotalNumInstances() const { return TotalInstances; }
234
235 template <typename AllocatorType>
236 void AppendData(const TInstanceCullingLoadBalancer<AllocatorType> &Other)
237 {
238 Data->Batches.Append(Other.GetBatches().GetData(), Other.GetBatches().Num());
239 Data->Items.Append(Other.GetItems().GetData(), Other.GetItems().Num());
240 TotalInstances += Other.GetTotalNumInstances();
241 }
242
243 bool HasSingleInstanceItemsOnly() const
244 {
245 return TotalInstances == Data->Items.Num();
246 }
