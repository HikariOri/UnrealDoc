34template<typename Type>
35struct FThreadSafeQueue
36{
37private:
38 mutable FCriticalSection SynchronizationObject; // made this mutable so this class can have const functions and still be thread safe
39 TRingBuffer<Type> Items;
40public:
41 void Enqueue(const Type& Item)
42 {
43 FScopeLock ScopeLock(&SynchronizationObject);
44 Items.Add(Item);
45 }
46
47 void Enqueue(Type&& Item)
48 {
49 FScopeLock ScopeLock(&SynchronizationObject);
50 Items.Add(MoveTempIfPossible(Item));
51 }
52
53 void EnqueueUnique(const Type& Item)
54 {
55 FScopeLock ScopeLock(&SynchronizationObject);
56 for (const Type& Existing : Items)
57 {
58 if (Existing == Item)
59 {
60 return;
61 }
62 }
63 Items.PushBack(Item);
64 }
65
66 bool Dequeue(Type* Result)
67 {
68 FScopeLock ScopeLock(&SynchronizationObject);
69 if (Items.Num())
70 {
71 *Result = Items.PopFrontValue();
72 return true;
73 }
74 return false;
75 }
76
77 void DequeueAll(TArray<Type>& Results)
78 {
79 FScopeLock ScopeLock(&SynchronizationObject);
80 Results.Reserve(Results.Num() + Items.Num());
81 while (!Items.IsEmpty())
82 {
83 Results.Add(Items.PopFrontValue());
84 }
85 }
86
87 bool HasItems() const
88 {
89 FScopeLock ScopeLock(&SynchronizationObject);
90 return Items.Num() > 0;
91 }
92
93 void Remove(const Type& Item)
94 {
95 FScopeLock ScopeLock(&SynchronizationObject);
96 Items.Remove(Item);
97 }
98
99 void CopyItems(const TArray<Type>& InItems) const
100 {
101 FScopeLock ScopeLock(&SynchronizationObject);
102 Items.Empty(InItems.Num());
103 for (const Type& Item : InItems)
104 {
105 Items.PushBack(Item);
106 }
107 }
108
109 int Num() const
110 {
111 FScopeLock ScopeLock(&SynchronizationObject);
112 return Items.Num();
113 }
114
115 void Empty()
116 {
117 FScopeLock ScopeLock(&SynchronizationObject);
118 Items.Empty();
119 }
120};

123template <typename T>
124class FThreadSafeSet
125{
126 TSet<T> InnerSet;
127 FCriticalSection SetCritical;
128public:
129 void Add(T InValue)
130 {
131 FScopeLock SetLock(&SetCritical);
132 InnerSet.Add(InValue);
133 }
134 bool AddUnique(T InValue)
135 {
136 FScopeLock SetLock(&SetCritical);
137 if (!InnerSet.Contains(InValue))
138 {
139 InnerSet.Add(InValue);
140 return true;
141 }
142 return false;
143 }
144 bool Contains(T InValue)
145 {
146 FScopeLock SetLock(&SetCritical);
147 return InnerSet.Contains(InValue);
148 }
149 void Remove(T InValue)
150 {
151 FScopeLock SetLock(&SetCritical);
152 InnerSet.Remove(InValue);
153 }
154 void Empty()
155 {
156 FScopeLock SetLock(&SetCritical);
157 InnerSet.Empty();
158 }
159
160 void GetValues(TSet<T>& OutSet)
161 {
162 FScopeLock SetLock(&SetCritical);
163 OutSet.Append(InnerSet);
164 }
