58class FPackageObjectIndex
59{
60 static constexpr uint64 IndexBits = 62ull;
61 static constexpr uint64 IndexMask = (1ull << IndexBits) - 1ull;
62 static constexpr uint64 TypeMask = ~IndexMask;
63 static constexpr uint64 TypeShift = IndexBits;
64 static constexpr uint64 Invalid = ~0ull;
65
66 uint64 TypeAndId = Invalid;
67
68 enum EType
69 {
70 Export,
71 ScriptImport,
72 PackageImport,
73 Null,
74 TypeCount = Null,
75 };
76 static_assert((TypeCount - 1) <= (TypeMask >> TypeShift), "FPackageObjectIndex: Too many types for TypeMask");
77
78 inline explicit FPackageObjectIndex(EType InType, uint64 InId) : TypeAndId((uint64(InType) << TypeShift) | InId) {}
79
80 COREUOBJECT_API static uint64 GenerateImportHashFromObjectPath(const FStringView& ObjectPath);
81 COREUOBJECT_API static uint64 GenerateImportHashFromVersePath(FUtf8StringView VersePath);
82
83public:
84 FPackageObjectIndex() = default;
85
86 inline static FPackageObjectIndex FromExportIndex(const int32 Index)
87 {
88 return FPackageObjectIndex(Export, Index);
89 }
90
91 inline static FPackageObjectIndex FromScriptPath(const FStringView& ScriptObjectPath)
92 {
93 return FPackageObjectIndex(ScriptImport, GenerateImportHashFromObjectPath(ScriptObjectPath));
94 }
95
96#if WITH_VERSE_VM || defined(__INTELLISENSE__)
97 inline static FPackageObjectIndex FromVersePath(Verse::VUniqueString* VersePath)
98 {
99 return FromVersePath(VersePath->AsStringView());
100 }
101#endif
102 inline static FPackageObjectIndex FromVersePath(FUtf8StringView VersePath)
103 {
104 return FPackageObjectIndex(ScriptImport, GenerateImportHashFromVersePath(VersePath));
105 }
106
107 inline static FPackageObjectIndex FromPackageImportRef(const FPackageImportReference& PackageImportRef)
108 {
109 uint64 Id = static_cast<uint64>(PackageImportRef.GetImportedPackageIndex()) << 32 | PackageImportRef.GetImportedPublicExportHashIndex();
110 check(!(Id & TypeMask));
111 return FPackageObjectIndex(PackageImport, Id);
112 }
113
114 inline bool IsNull() const
115 {
116 return TypeAndId == Invalid;
117 }
118
119 inline bool IsExport() const
120 {
121 return (TypeAndId >> TypeShift) == Export;
122 }
123
124 inline bool IsImport() const
125 {
126 return IsScriptImport() || IsPackageImport();
127 }
128
129 inline bool IsScriptImport() const
130 {
131 return (TypeAndId >> TypeShift) == ScriptImport;
132 }
133
134 inline bool IsPackageImport() const
135 {
136 return (TypeAndId >> TypeShift) == PackageImport;
137 }
138
139 inline uint32 ToExport() const
140 {
141 check(IsExport());
142 return uint32(TypeAndId);
143 }
144
145 inline FPackageImportReference ToPackageImportRef() const
146 {
147 uint32 ImportedPackageIndex = static_cast<uint32>((TypeAndId & IndexMask) >> 32);
148 uint32 ExportHash = static_cast<uint32>(TypeAndId);
149 return FPackageImportReference(ImportedPackageIndex, ExportHash);
150 }
151
152 inline uint64 Value() const
153 {
154 return TypeAndId & IndexMask;
155 }
156
157 inline bool operator==(FPackageObjectIndex Other) const
158 {
159 return TypeAndId == Other.TypeAndId;
160 }
161
162 inline bool operator!=(FPackageObjectIndex Other) const
163 {
164 return TypeAndId != Other.TypeAndId;
165 }
166
167 friend FArchive& operator<<(FArchive& Ar, FPackageObjectIndex& Value)
168 {
169 Ar << Value.TypeAndId;
170 return Ar;
171 }
172
173 inline friend uint32 GetTypeHash(const FPackageObjectIndex& Value)
174 {
175 return uint32(Value.TypeAndId);
176 }
177};

179class FPublicExportKey
180{
181public:
182 FPublicExportKey()
183 {
184 }
185
186 bool IsNull() const
187 {
188 return GetExportHash() == 0;
189 }
190
191 FPackageId GetPackageId() const
192 {
193 return FPackageId::FromValue(uint64(PackageIdHigh) << 32 | PackageIdLow);
194 }
195
196 uint64 GetExportHash() const
197 {
198 return uint64(ExportHashHigh) << 32 | ExportHashLow;
199 }
200
201 inline bool operator==(const FPublicExportKey& Other) const
202 {
203 return GetExportHash() == Other.GetExportHash() &&
204 GetPackageId() == Other.GetPackageId();
205 }
206
207 inline bool operator!=(const FPublicExportKey& Other) const
208 {
209 return GetExportHash() != Other.GetExportHash() ||
210 GetPackageId() != Other.GetPackageId();
211 }
212
213 inline friend uint32 GetTypeHash(const FPublicExportKey& In)
214 {
215 return HashCombine(GetTypeHash(In.GetPackageId()), GetTypeHash(In.GetExportHash()));
216 }
217
218 static FPublicExportKey MakeKey(FPackageId PackageId, uint64 ExportHash)
219 {
220 check(PackageId.IsValid());
221 check(ExportHash);
222 uint64 PackageIdValue = PackageId.Value();
223 return FPublicExportKey(uint32(PackageIdValue >> 32), uint32(PackageIdValue), uint32(ExportHash >> 32), uint32(ExportHash));
224 }
225
226 static FPublicExportKey FromPackageImport(FPackageObjectIndex ObjectIndex, const TArrayView<const FPackageId>& ImportedPackageIds, const TArrayView<const uint64>& ImportedPublicExportHashes)
227 {
228 check(ObjectIndex.IsPackageImport());
229 FPackageImportReference PackageImportRef = ObjectIndex.ToPackageImportRef();
230 FPackageId PackageId = ImportedPackageIds[PackageImportRef.GetImportedPackageIndex()];
231 uint64 ExportHash = ImportedPublicExportHashes[PackageImportRef.GetImportedPublicExportHashIndex()];
232 return MakeKey(PackageId, ExportHash);
233 }
234
235private:
236 FPublicExportKey(uint32 InPackageIdHigh, uint32 InPackageIdLow, uint32 InExportHashHigh, uint32 InExportHashLow)
237 : PackageIdHigh(InPackageIdHigh)
238 , PackageIdLow(InPackageIdLow)
239 , ExportHashHigh(InExportHashHigh)
240 , ExportHashLow(InExportHashLow)
241 {
242 }
243
244 uint32 PackageIdHigh = 0;
245 uint32 PackageIdLow = 0;
246 uint32 ExportHashHigh = 0;
247 uint32 ExportHashLow = 0;
248};
