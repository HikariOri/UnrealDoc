219 static inline void AddRasterizeToRectsPass(
220 FRDGBuilder& GraphBuilder,
221 const FGlobalShaderMap* GlobalShaderMap,
222 FRDGEventName&& PassName,
223 const TShaderRef<TPixelShaderClass>& PixelShader,
224 TPassParameters* Parameters,
225 FIntPoint ViewportSize,
226 FRDGBufferSRVRef RectCoordBufferSRV,
227 uint32 NumRects,
228 FRHIBlendState* BlendState = nullptr,
229 FRHIRasterizerState* RasterizerState = nullptr,
230 FRHIDepthStencilState* DepthStencilState = nullptr,
231 uint32 StencilRef = 0,
232 FIntPoint TextureSize = FIntPoint(1, 1),
233 FRDGBufferSRVRef RectUVBufferSRV = nullptr,
234 uint32 DownsampleFactor = 1,
235 ERDGPassFlags AdditionalPassFlags = ERDGPassFlags::None,
236 float DepthIncrement = 0.0f)
237 {
238 FRasterizeToRectsVS::FPermutationDomain PermutationVector;
239 PermutationVector.Set<FRasterizeToRectsVS::FRectUV>(RectUVBufferSRV != nullptr);
240 auto VertexShader = GlobalShaderMap->GetShader<FRasterizeToRectsVS>(PermutationVector);
241
242 Parameters->VS.InvViewSize = FVector2f(1.0f / ViewportSize.X, 1.0f / ViewportSize.Y);
243 Parameters->VS.InvTextureSize = FVector2f(1.0f / TextureSize.X, 1.0f / TextureSize.Y);
244 Parameters->VS.DownsampleFactor = 1.0f / DownsampleFactor;
245 Parameters->VS.RectCoordBuffer = RectCoordBufferSRV;
246 Parameters->VS.RectUVBuffer = RectUVBufferSRV;
247 Parameters->VS.DepthIncrement = DepthIncrement;
248
249 ClearUnusedGraphResources(PixelShader, &Parameters->PS);
250
251 ERDGPassFlags PassFlags = ERDGPassFlags::Raster;
252 PassFlags |= AdditionalPassFlags;
253 const bool bSkipRenderPass = EnumHasAnyFlags(PassFlags, ERDGPassFlags::SkipRenderPass);
254
255 GraphBuilder.AddPass(
256 Forward<FRDGEventName>(PassName),
257 Parameters,
258 PassFlags,
259 [Parameters, GlobalShaderMap, VertexShader, PixelShader, ViewportSize, BlendState, RasterizerState, DepthStencilState, StencilRef, bSkipRenderPass, NumRects](FRDGAsyncTask, FRHICommandList& RHICmdList)
260 {
261 if (bSkipRenderPass)
262 {
263 FRHIRenderPassInfo RPInfo;
264 RPInfo.ResolveRect.X1 = 0;
265 RPInfo.ResolveRect.Y1 = 0;
266 RPInfo.ResolveRect.X2 = ViewportSize.X;
267 RPInfo.ResolveRect.Y2 = ViewportSize.Y;
268 RHICmdList.BeginRenderPass(RPInfo, TEXT("RasterizeToRects"));
269 }
270
271 FGraphicsPipelineStateInitializer GraphicsPSOInit;
272 RHICmdList.ApplyCachedRenderTargets(GraphicsPSOInit);
273
274 RHICmdList.SetViewport(0.0f, 0.0f, 0.0f, (float)ViewportSize.X, (float)ViewportSize.Y, 1.0f);
275
276 GraphicsPSOInit.BlendState = BlendState ? BlendState : TStaticBlendState<>::GetRHI();
277 GraphicsPSOInit.RasterizerState = RasterizerState ? RasterizerState : TStaticRasterizerState<>::GetRHI();
278 GraphicsPSOInit.DepthStencilState = DepthStencilState ? DepthStencilState : TStaticDepthStencilState<false, CF_Always>::GetRHI();
279
280 GraphicsPSOInit.PrimitiveType = GRHISupportsRectTopology ? PT_RectList : PT_TriangleList;
281
282 GraphicsPSOInit.BoundShaderState.VertexDeclarationRHI = GTileVertexDeclaration.VertexDeclarationRHI;
283 GraphicsPSOInit.BoundShaderState.VertexShaderRHI = VertexShader.GetVertexShader();
284 GraphicsPSOInit.BoundShaderState.PixelShaderRHI = PixelShader.GetPixelShader();
285
286 SetGraphicsPipelineState(RHICmdList, GraphicsPSOInit, StencilRef);
287
288 SetShaderParameters(RHICmdList, VertexShader, VertexShader.GetVertexShader(), Parameters->VS);
289 SetShaderParameters(RHICmdList, PixelShader, PixelShader.GetPixelShader(), Parameters->PS);
290
291 const uint32 NumPrimitives = GRHISupportsRectTopology ? 1 : 2;
292 const uint32 NumInstances = NumRects;
293 RHICmdList.DrawPrimitive(0, NumPrimitives, NumInstances);
294
295 if (bSkipRenderPass)
296 {
297 RHICmdList.EndRenderPass();
298 }
299 });
300 }
