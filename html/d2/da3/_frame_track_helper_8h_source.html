33struct FFrameTrackSample
34{

38 double EndTime; // max end time of all frames in this sample

40 double LargestFrameStartTime; // start time of the largest frame
41 double LargestFrameDuration; // duration of the largest frame
42
43 FFrameTrackSample()
44 : NumFrames(0)
45 , TotalDuration(0.0)
46 , StartTime(DBL_MAX)
47 , EndTime(-DBL_MAX)
48 , LargestFrameIndex(0)
49 , LargestFrameStartTime(0.0)
50 , LargestFrameDuration(0.0)
51 {}
