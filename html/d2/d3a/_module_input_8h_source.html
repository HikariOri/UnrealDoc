58 template<int32 MaxValue, uint32 NumBits>
59 struct TCompressedFloatDetails
60 {
61 // NumBits = 8:
62 static constexpr int32 MaxBitValue = (1 << (NumBits - 1)) - 1; // 0111 1111 - Max abs value we will serialize
63 static constexpr int32 Bias = (1 << (NumBits - 1)); // 1000 0000 - Bias to pivot around (in order to support signed values)
64 static constexpr int32 SerIntMax = (1 << (NumBits - 0)); // 1 0000 0000 - What we pass into SerializeInt
65 static constexpr int32 MaxDelta = (1 << (NumBits - 0)) - 1; // 1111 1111 - Max delta is
66 };
67
68 template<int32 MaxValue, uint32 NumBits, typename T UE_REQUIRES(std::is_floating_point_v<T>&& NumBits < 32)>
69 bool ToCompressedFloat(const T InValue, uint32& OutCompressedFloat)
70 {
71 using Details = ModularQuantize::TCompressedFloatDetails<MaxValue, NumBits>;
72
73 bool clamp = false;
74 int64 ScaledValue;
75 if (MaxValue > Details::MaxBitValue)
76 {
77 // We have to scale this down
78 const T Scale = T(Details::MaxBitValue) / MaxValue;
79 ScaledValue = FMath::TruncToInt(Scale * InValue);
80 }
81 else
82 {
83 // We will scale up to get extra precision. But keep is a whole number preserve whole values
84 constexpr int32 Scale = Details::MaxBitValue / MaxValue;
85 ScaledValue = FMath::RoundToInt(Scale * InValue);
86 }
87
88 uint32 Delta = static_cast<uint32>(ScaledValue + Details::Bias);
89
90 if (Delta > Details::MaxDelta)
91 {
92 clamp = true;
93 Delta = static_cast<int32>(Delta) > 0 ? Details::MaxDelta : 0;
94 }
95
96 OutCompressedFloat = Delta;
97
98 return !clamp;
99 }
100
101 template<int32 MaxValue, uint32 NumBits, typename T UE_REQUIRES(std::is_floating_point_v<T>&& NumBits < 32)>
102 bool FromCompressedFloat(const uint32 InCompressed, T& OutValue )
103 {
104 using Details = ModularQuantize::TCompressedFloatDetails<MaxValue, NumBits>;
105
106 uint32 Delta = InCompressed;
107 T UnscaledValue = static_cast<T>(static_cast<int32>(Delta) - Details::Bias);
108
109 if constexpr (MaxValue > Details::MaxBitValue)
110 {
111 // We have to scale down, scale needs to be a float:
112 constexpr T InvScale = MaxValue / (T)Details::MaxBitValue;
113 OutValue = UnscaledValue * InvScale;
114 }
115 else
116 {
117 constexpr int32 Scale = Details::MaxBitValue / MaxValue;
118 constexpr T InvScale = T(1) / (T)Scale;
119
120 OutValue = UnscaledValue * InvScale;
121 }
122
123 return true;
124 }

126 template<int32 MaxValue, uint32 NumBits, typename T UE_REQUIRES(std::is_floating_point_v<T>&& NumBits < 32)>
127 bool WriteCompressedFloat(const T Value, FArchive& Ar)
128 {
129 using Details = ModularQuantize::TCompressedFloatDetails<MaxValue, NumBits>;
130
131 uint32 CompressedValue;
132 bool clamp = ModularQuantize::ToCompressedFloat<MaxValue, NumBits>(Value, CompressedValue);
133
134 Ar.SerializeInt(CompressedValue, Details::SerIntMax);
135
136 return !clamp;
137 }

139 template<int32 MaxValue, uint32 NumBits, typename T UE_REQUIRES(std::is_floating_point_v<T>&& NumBits < 32)>
140 bool ReadCompressedFloat(T& Value, FArchive& Ar)
141 {
142 using Details = ModularQuantize::TCompressedFloatDetails<MaxValue, NumBits>;
143
144 uint32 CompressedValue;
145 Ar.SerializeInt(CompressedValue, Details::SerIntMax);
146
147 ModularQuantize::FromCompressedFloat<MaxValue, NumBits>(CompressedValue, Value);
148
149 return true;
150 }

152 // Required because we seialize quantized vector in seperate parts depending on input type
153 template<int32 MaxValue, uint32 NumBits>
154 bool SerializeFixedFloat(double& InOutValue, FArchive& Ar)
155 {
156 if (Ar.IsSaving())
157 {
158 bool success = true;
159 success &= ModularQuantize::WriteCompressedFloat<MaxValue, NumBits>(InOutValue, Ar);
160 return success;
161 }
162
163 ModularQuantize::ReadCompressedFloat<MaxValue, NumBits>(InOutValue, Ar);
164 return true;
165 }

167 template<int32 MaxValue, uint32 NumBits, typename T UE_REQUIRES(std::is_floating_point_v<T>&& NumBits < 32)>
168 void QuantizeValue(T& Value)
169 {
170 uint32 CompressedValue = 0;
171 ModularQuantize::ToCompressedFloat<MaxValue, NumBits>(Value, CompressedValue);
172 ModularQuantize::FromCompressedFloat<MaxValue, NumBits>(CompressedValue, Value);
173 }
174
175 template<int32 MaxValue, uint32 NumBits, typename T UE_REQUIRES(std::is_floating_point_v<T>&& NumBits < 32)>
176 bool QuantizedIsNearlyEqual(const T& Left,const T& Right)
177 {
178 uint32 CompressedValue = 0;
179 T QuantLeft, QuantRight;
180 ModularQuantize::ToCompressedFloat<MaxValue, NumBits>(Left, CompressedValue);
181 ModularQuantize::FromCompressedFloat<MaxValue, NumBits>(CompressedValue, QuantLeft);
182 ModularQuantize::ToCompressedFloat<MaxValue, NumBits>(Right, CompressedValue);
183 ModularQuantize::FromCompressedFloat<MaxValue, NumBits>(CompressedValue, QuantRight);
184 return FMath::IsNearlyEqual(QuantLeft, QuantRight);
185 }
