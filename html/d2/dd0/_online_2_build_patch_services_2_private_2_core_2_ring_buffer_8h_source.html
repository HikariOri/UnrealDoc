192template< typename DataType >
193void TRingBuffer< DataType >::Enqueue(const DataType& Val)
194{
195 Data[DataIndex++] = Val;
196
197 if (DataIndex == BufferDataSize)
198 {
199 DataIndex = 0;
200 }
201
202 ++TotalNumDataPushed;
203 NumDataAvailable = FMath::Clamp< uint32 >(NumDataAvailable + 1, 0, BufferDataSize);
204}

206template< typename DataType >
207void TRingBuffer< DataType >::Enqueue(const DataType* ValBuf, const uint32& BufLen)
208{
209 check(BufLen <= BufferDataSize);
210 const uint32 FirstPartLen = BufferDataSize - DataIndex;
211 FMemory::Memcpy(Data + DataIndex, ValBuf, sizeof(DataType)* FMath::Min(FirstPartLen, BufLen));
212 if (FirstPartLen < BufLen)
213 {
214 FMemory::Memcpy(Data, ValBuf + FirstPartLen, sizeof(DataType)* (BufLen - FirstPartLen));
215 }
216 DataIndex += BufLen;
217 if (DataIndex >= BufferDataSize)
218 {
219 DataIndex -= BufferDataSize;
220 }
221 TotalNumDataPushed += BufLen;
222 NumDataAvailable = FMath::Clamp<uint32>(NumDataAvailable + BufLen, 0, BufferDataSize);
223}

261template< typename DataType >
262uint32 TRingBuffer< DataType >::Peek(DataType* ValBuf, const uint32& BufLen) const
263{
264 const uint32 DataProvided = FMath::Min(BufLen, NumDataAvailable);
265 const uint32 BottomIdx = BottomIndex();
266 const uint32 BottomPartLen = BufferDataSize - BottomIdx;
267 FMemory::Memcpy(ValBuf, Data + BottomIdx, sizeof(DataType)* FMath::Min(BottomPartLen, DataProvided));
268 if (BottomPartLen < DataProvided)
269 {
270 FMemory::Memcpy(ValBuf + BottomPartLen, Data, sizeof(DataType)* (DataProvided - BottomPartLen));
271 }
272 return DataProvided;
273}

275template< typename DataType >
276int32 TRingBuffer< DataType >::SerialCompare(const DataType* SerialBuffer, uint32 CompareLen) const
277{
278 check(CompareLen <= BufferDataSize);
279 const uint32 FirstPartLen = BufferDataSize - DataIndex;
280 int32 FirstCmp = FMemory::Memcmp(Data + BottomIndex(), SerialBuffer, sizeof(DataType)* FMath::Min(FirstPartLen, CompareLen));
281 if (FirstCmp != 0 || FirstPartLen >= CompareLen)
282 {
283 return FirstCmp;
284 }
285 return FMemory::Memcmp(Data, &SerialBuffer[FirstPartLen], sizeof(DataType)* (CompareLen - FirstPartLen));
286}

288template< typename DataType >
289void TRingBuffer< DataType >::GetShaHash(FSHAHash& OutHash) const
290{
291 const uint32 FirstPartLen = FMath::Min(BufferDataSize - DataIndex, NumDataAvailable);
292 FSHA1 Sha;
293 Sha.Update(Data + BottomIndex(), sizeof(DataType) * FirstPartLen);
294 if (FirstPartLen < NumDataAvailable)
295 {
296 Sha.Update(Data, sizeof(DataType) * (NumDataAvailable - FirstPartLen));
297 }
298 Sha.Final();
299 Sha.GetHash(OutHash.Hash);
300}
