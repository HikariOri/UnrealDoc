31// Bit counts for Word3 of filter data.
32// (ExtraFilter (top NumExtraFilterBits) + MyChannel (next NumCollisionChannelBits) as ECollisionChannel + Flags (remaining NumFilterDataFlagBits)
33// [NumExtraFilterBits] [NumCollisionChannelBits] [NumFilterDataFlagBits] = 32 bits
34enum { NumCollisionChannelBits = 5 };
35enum { NumFilterDataFlagBits = 32 - NumExtraFilterBits - NumCollisionChannelBits };

212inline ECollisionChannel GetCollisionChannel(uint32 Word3)
213{
214 uint32 ChannelMask = (Word3 << NumExtraFilterBits) >> (32 - NumCollisionChannelBits);
215 return (ECollisionChannel)ChannelMask;
216}

219inline ECollisionChannel GetCollisionChannelAndExtraFilter(uint32 Word3, FMaskFilter& OutMaskFilter)

222 OutMaskFilter = Word3 >> (32 - NumExtraFilterBits);

226UE_DEPRECATED(5.8, "Use FPhysicsFilterBuilder directly instead")
227inline uint32 CreateChannelAndFilter(ECollisionChannel CollisionChannel, FMaskFilter MaskFilter)
228{
229 uint32 ResultMask = (uint32(MaskFilter) << NumCollisionChannelBits) | (uint32)CollisionChannel;
230 return ResultMask << NumFilterDataFlagBits;
231}

233UE_DEPRECATED(5.8, "Use FPhysicsFilterBuilder directly instead")
234inline void UpdateMaskFilter(uint32& Word3, FMaskFilter NewMaskFilter)
235{
236 static_assert(NumExtraFilterBits <= 8, "Only up to 8 extra filter bits are supported.");
237 Word3 &= (0xFFFFFFFFu >> NumExtraFilterBits); //we drop the top NumExtraFilterBits bits because that's where the new mask filter is going
238 Word3 |= uint32(NewMaskFilter) << (32 - NumExtraFilterBits);
239}

241UE_DEPRECATED(5.8, "Use ExtractCollisionResponseContainer instead")
242inline FCollisionResponseContainer ExtractSimCollisionResponseContainer(const FCollisionFilterData& InSimFilterData)
243{
244 FCollisionResponseContainer CollisionResponseContainer;
245
246 for (int32 ChannelIndex = 0; ChannelIndex < UE_ARRAY_COUNT(CollisionResponseContainer.EnumArray); ++ChannelIndex)
247 {
248 const uint64 ChannelBit = 1LLU << ChannelIndex;
249 if (uint64(InSimFilterData.Word1) & ChannelBit)
250 {
251 CollisionResponseContainer.EnumArray[ChannelIndex] = ECR_Block;
252 }
253 else
254 {
255 CollisionResponseContainer.EnumArray[ChannelIndex] = ECR_Ignore;
256 }
257 }
258
259 return CollisionResponseContainer;
260}

263inline FCollisionResponseContainer ExtractQueryCollisionResponseContainer(const FCollisionFilterData& InQueryFilterData)
264{
265 FCollisionResponseContainer CollisionResponseContainer;
266
267 for (int32 ChannelIndex = 0; ChannelIndex < UE_ARRAY_COUNT(CollisionResponseContainer.EnumArray); ++ChannelIndex)

273 }
274 else if (uint64(InQueryFilterData.Word2) & ChannelBit)
