76 inline FMeshEdge* FindOppositeEdge(int32 Index1, int32 Index2)
77 {
78 FMeshEdge* Edge = nullptr;
79 TArray<FMeshEdge*> EdgeList;
80
81 // Search the hash for a corresponding vertex
82 VertexToEdgeList.MultiFind((FVector)Vertices.Position[Index2], EdgeList);
83
84 // Now search through the array for a match or not
85 for (int32 EdgeIndex = 0; EdgeIndex < EdgeList.Num() && Edge == nullptr; ++EdgeIndex)
86 {
87 FMeshEdge* OtherEdge = EdgeList[EdgeIndex];
88 // See if this edge matches the passed in edge
89 if (OtherEdge != nullptr && DoesEdgeMatch(Index1, Index2, OtherEdge))
90 {
91 // We have a match
92 Edge = OtherEdge;
93 }
94 }
95 return Edge;
96 }

105 inline void AddEdge(int32 Index1, int32 Index2, int32 Triangle)
106 {
107 // If this edge matches another then just fill the other triangle
108 // otherwise add it
109 FMeshEdge* OtherEdge = FindOppositeEdge(Index1, Index2);
110 if (OtherEdge == nullptr)
111 {
112 // Add a new edge to the array
113 int32 EdgeIndex = Edges.AddZeroed();
114 Edges[EdgeIndex].Vertices[0] = Index1;
115 Edges[EdgeIndex].Vertices[1] = Index2;
116 Edges[EdgeIndex].Faces[0] = Triangle;
117 Edges[EdgeIndex].Faces[1] = INDEX_NONE;
118 // Also add this edge to the hash for faster searches
119 // NOTE: This relies on the array never being realloced!
120 VertexToEdgeList.Add((FVector)Vertices.Position[Index1], &Edges[EdgeIndex]);
121 }
122 else
123 {
124 OtherEdge->Faces[1] = Triangle;
125 }
126 }

152 void FindEdges(void)
153 {
154 // @todo Handle something other than trilists when building edges
155 int32 TriangleCount = Indices.Num() / 3;
156 int32 EdgeCount = 0;
157 // Work through all triangles building the edges
158 for (int32 Triangle = 0; Triangle < TriangleCount; Triangle++)
159 {
160 // Determine the starting index
161 int32 TriangleIndex = Triangle * 3;
162 // Get the indices for the triangle
163 int32 Index1 = Indices[TriangleIndex];
164 int32 Index2 = Indices[TriangleIndex + 1];
165 int32 Index3 = Indices[TriangleIndex + 2];
166 // Add the first to second edge
167 AddEdge(Index1, Index2, Triangle);
168 // Now add the second to third
169 AddEdge(Index2, Index3, Triangle);
170 // Add the third to first edge
171 AddEdge(Index3, Index1, Triangle);
172 }

191 static void BuildDepthOnlyIndexBuffer(
192 TArray<uint32>& OutDepthIndices,
193 const FConstMeshBuildVertexView& InVertices,
194 const TArray<uint32>& InIndices,
195 const TArrayView<FStaticMeshSection>& InSections
196 )
197 {
198 int32 NumVertices = InVertices.Position.Num();
199 if (InIndices.Num() <= 0 || NumVertices <= 0)
200 {
201 OutDepthIndices.Empty();
202 return;
203 }
204
205 // Create a mapping of index -> first overlapping index to accelerate the construction of the shadow index buffer.
206 TArray<FIndexAndZ> VertIndexAndZ;
207 VertIndexAndZ.Empty(NumVertices);
208 for (int32 VertIndex = 0; VertIndex < NumVertices; VertIndex++)
209 {
210 VertIndexAndZ.Emplace(VertIndex, FVector(InVertices.Position[VertIndex]));
211 }
212 VertIndexAndZ.Sort(FCompareIndexAndZ());
213
214 // Setup the index map. 0xFFFFFFFF == not set.
215 TArray<uint32> IndexMap;
216 IndexMap.AddUninitialized(NumVertices);
217 FMemory::Memset(IndexMap.GetData(), 0xFF, NumVertices * sizeof(uint32));
218
219 // Search for duplicates, quickly!
220 for (int32 i = 0; i < VertIndexAndZ.Num(); i++)
221 {
222 uint32 SrcIndex = VertIndexAndZ[i].Index;
223 float Z = VertIndexAndZ[i].Z;
224 IndexMap[SrcIndex] = FMath::Min(IndexMap[SrcIndex], SrcIndex);
225
226 // Search forward since we add pairs both ways.
227 for (int32 j = i + 1; j < VertIndexAndZ.Num(); j++)
228 {
229 if (FMath::Abs(VertIndexAndZ[j].Z - Z) > THRESH_POINTS_ARE_SAME * 4.01f)
230 break; // can't be any more dups
231
232 uint32 OtherIndex = VertIndexAndZ[j].Index;
233 if (InVertices.Position[SrcIndex].Equals(InVertices.Position[OtherIndex], 0.0f))
234 {
235 IndexMap[SrcIndex] = FMath::Min(IndexMap[SrcIndex], OtherIndex);
236 IndexMap[OtherIndex] = FMath::Min(IndexMap[OtherIndex], SrcIndex);
237 }
238 }
239 }
240
241 // Build the depth-only index buffer by remapping all indices to the first overlapping
242 // vertex in the vertex buffer.
243 OutDepthIndices.Empty();
244 for (int32 SectionIndex = 0; SectionIndex < InSections.Num(); ++SectionIndex)
245 {
246 const FStaticMeshSection& Section = InSections[SectionIndex];
247 int32 FirstIndex = Section.FirstIndex;
248 int32 LastIndex = FirstIndex + Section.NumTriangles * 3;
249 for (int32 SrcIndex = FirstIndex; SrcIndex < LastIndex; ++SrcIndex)
250 {
251 uint32 VertIndex = InIndices[SrcIndex];
252 OutDepthIndices.Add(IndexMap[VertIndex]);
253 }
254 }
255 }

257 static void CacheOptimizeVertexAndIndexBuffer(
258 FMeshBuildVertexData& Vertices,
259 TArray<TArray<uint32> >& PerSectionIndices,
260 TArray<int32>& WedgeMap
261 )
262 {
263 if (Vertices.Position.Num() <= 0)
264 {
265 return;
266 }
267
268 // Copy the vertices since we will be reordering them
269 FMeshBuildVertexData OriginalVertices;
270 OriginalVertices.Position = Vertices.Position;
271 OriginalVertices.TangentX = Vertices.TangentX;
272 OriginalVertices.TangentY = Vertices.TangentY;
273 OriginalVertices.TangentZ = Vertices.TangentZ;
274 OriginalVertices.CurveRadius = Vertices.CurveRadius;
275 OriginalVertices.Color = Vertices.Color;
276 OriginalVertices.UVs = Vertices.UVs;
277
278 // Initialize a cache that stores which indices have been assigned
279 TArray<int32> IndexCache;
280 IndexCache.AddUninitialized(Vertices.Position.Num());
281 FMemory::Memset(IndexCache.GetData(), INDEX_NONE, IndexCache.Num() * IndexCache.GetTypeSize());
282 int32 NextAvailableIndex = 0;
283
284 // Iterate through the section index buffers,
285 // Optimizing index order for the post transform cache (minimizes the number of vertices transformed),
286 // And vertex order for the pre transform cache (minimizes the amount of vertex data fetched by the GPU).
287 for (int32 SectionIndex = 0; SectionIndex < PerSectionIndices.Num(); SectionIndex++)
288 {
289 TArray<uint32>& Indices = PerSectionIndices[SectionIndex];
290
291 if (Indices.Num())
292 {
293 // Optimize the index buffer for the post transform cache with.
294 BuildOptimizationThirdParty::CacheOptimizeIndexBuffer(Indices);
295
296 // Copy the index buffer since we will be reordering it
297 TArray<uint32> OriginalIndices = Indices;
298
299 // Go through the indices and assign them new values that are coherent where possible
300 for (int32 Index = 0; Index < Indices.Num(); Index++)
301 {
302 const int32 CachedIndex = IndexCache[OriginalIndices[Index]];
303
304 if (CachedIndex == INDEX_NONE)
305 {
306 // No new index has been allocated for this existing index, assign a new one
307 Indices[Index] = NextAvailableIndex;
308 // Mark what this index has been assigned to
309 IndexCache[OriginalIndices[Index]] = NextAvailableIndex;
310 NextAvailableIndex++;
311 }
312 else
313 {
314 // Reuse an existing index assignment
315 Indices[Index] = CachedIndex;
316 }
317
318 // Reorder the vertices based on the new index assignment
319 Vertices.Position[Indices[Index]] = OriginalVertices.Position[OriginalIndices[Index]];
320 Vertices.TangentX[Indices[Index]] = OriginalVertices.TangentX[OriginalIndices[Index]];
321 Vertices.TangentY[Indices[Index]] = OriginalVertices.TangentY[OriginalIndices[Index]];
322 Vertices.TangentZ[Indices[Index]] = OriginalVertices.TangentZ[OriginalIndices[Index]];
323 if (Vertices.CurveRadius.Num() > 0)
324 {
325 Vertices.CurveRadius[Indices[Index]] = OriginalVertices.CurveRadius[OriginalIndices[Index]];
326 }
327 if (Vertices.Color.Num() > 0)
328 {
329 Vertices.Color[Indices[Index]] = OriginalVertices.Color[OriginalIndices[Index]];
330 }
331 for (int32 TexCoord = 0; TexCoord < Vertices.UVs.Num(); ++TexCoord)
332 {
333 Vertices.UVs[TexCoord][Indices[Index]] = OriginalVertices.UVs[TexCoord][OriginalIndices[Index]];
334 }
335 }
336 }
337 }
338
339 for (int32 WedgeIndex = 0; WedgeIndex < WedgeMap.Num(); ++WedgeIndex)
340 {
341 const int32 MappedIndex = WedgeMap[WedgeIndex];
342 if (MappedIndex != INDEX_NONE)
343 {
344 WedgeMap[WedgeIndex] = IndexCache[MappedIndex];
345 }
346 }
347 }
