45 void UpdateTotalCPUTime(FCollisionAnalyzer* Analyzer)
46 {
47 if(bIsGroup)
48 {
49 TotalCPUTime = 0.f;
50 // Iterate over each query in group
51 for(int32 i=0; i<QueriesInGroup.Num(); i++)
52 {
53 TSharedPtr<FQueryTreeItem> ChildItem = QueriesInGroup[i];
54 check(ChildItem.IsValid());
55 check(!ChildItem->bIsGroup);
56 check(ChildItem->QueryIndex < Analyzer->Queries.Num());
57 const FCAQuery& Query = Analyzer->Queries[ChildItem->QueryIndex];
58 TotalCPUTime += Query.CPUTime;
59 }
60
61 }
62 }
