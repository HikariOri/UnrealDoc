33// Carries the SDL_Window* plus the SDL window id captured at construction time (while the window is
34// provably alive). The id rides with the context through swapchain creation so the surface can be
35// released by that immutable, never-recycled id -- never by re-deriving it from a pointer that may
36// be stale or reused by destroy time. Ctor/IsValid are defined out-of-line so SDL stays out of this
37// header and IsValid can reach FVulkanLinuxPlatform::IsWindowValid.
38class FVulkanLinuxPlatformWindowContext : public FVulkanGenericPlatformWindowContext
39{
40public:
41 explicit FVulkanLinuxPlatformWindowContext(void* InWindowHandle);
42 uint32 GetWindowId() const { return WindowId; }
43 bool IsValid() const;
44private:
45 uint32 WindowId = 0;
46};
