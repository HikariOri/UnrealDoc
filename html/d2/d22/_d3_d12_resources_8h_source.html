367 bool IsResident() const
368 {
369#if ENABLE_RESIDENCY_MANAGEMENT
370
371 if (NeedsDeferredResidencyUpdate())
372 {
373 // We don't know the state because the set of residency handles is only known on the
374 // RHI Submission Thread and may change throughout the frame.
375 return true;
376 }
377
378 TConstArrayView<FD3D12ResidencyHandle*> ResidencyHandles = GetResidencyHandles();
379 if (ResidencyHandles.IsEmpty())
380 {
381 // No residency tracking for this resource
382 return true;
383 }
384
385 // Treat resource as resident if at least one backing heap is resident.
386 // Technically we should return a partial residency status.
387 for (FD3D12ResidencyHandle* Handle : ResidencyHandles)
388 {
389 if (Handle->ResidencyStatus == FD3D12ResidencyHandle::RESIDENCY_STATUS::RESIDENT)
390 {
391 return true;
392 }
393 }
394 return false;
395#else // ENABLE_RESIDENCY_MANAGEMENT
396 return true;
397#endif // ENABLE_RESIDENCY_MANAGEMENT
398 }

450 ED3D12Access GetOptimalInitialD3D12Access(ED3D12Access InD3D12Access, bool bAccurateWriteableStates) const
451 {
452 // Ignore the requested resource state for non tracked resource because RHI will assume it's always in default resource
453 // state then when a transition is required (will transition via scoped push/pop to requested state)
454 if (!bSRVOnly && InD3D12Access != ED3D12Access::Unknown)
455 {
456 return InD3D12Access;
457 }
458
459 if (bSRVOnly)
460 {
461 return ED3D12Access::SRVGraphics;
462 }
463 else if (bBuffer && !bUAV)
464 {
465 return (bReadBackResource) ? ED3D12Access::CopyDest : ED3D12Access::GenericRead;
466 }
467 else if (bWritable && bAccurateWriteableStates)
468 {
469 if (bDSV)
470 {
471 return ED3D12Access::DSVWrite;
472 }
473 else if (bRTV)
474 {
475 return ED3D12Access::RTV;
476 }
477 else if (bUAV)
478 {
479 check(InD3D12Access != ED3D12Access::Unknown);
480 return ED3D12Access::UAVMask;
481 }
482 }
483
484 return ED3D12Access::Common;
485 }

490 const uint32 bUAV : 1;
491 const uint32 bWritable : 1;

493 const uint32 bBuffer : 1;

502 void InitializeResourceState(
503 const D3D12_HEAP_PROPERTIES* InHeapProps,
504 ED3D12Access InInitialD3D12Access,
505 ED3D12ResourceStateMode InResourceStateMode,
506 ED3D12Access InDefaultD3D12Access)
507 {
508 check(IsValidAccess(InDefaultD3D12Access));
509 check(IsValidAccess(InInitialD3D12Access));
510
511 SubresourceCount = GetMipLevels() * GetArraySize() * GetPlaneCount();
512
513 if (InResourceStateMode == ED3D12ResourceStateMode::SingleState)
514 {
515 // make sure a valid default state is set
516 check(InInitialD3D12Access != ED3D12Access::Unknown);
517 check(InDefaultD3D12Access != ED3D12Access::Unknown);
518
519#if UE_BUILD_DEBUG
520 FPlatformAtomics::InterlockedIncrement(&NoStateTrackingResourceCount);
521#endif
522 InitialD3D12Access = InInitialD3D12Access;
523 DefaultD3D12Access = InDefaultD3D12Access;
524 bRequiresResourceStateTracking = false;
525 }
526 else
527 {
528 DetermineResourceStates(InHeapProps, InInitialD3D12Access, InResourceStateMode, InDefaultD3D12Access);
529 }
530
531 if (bRequiresResourceStateTracking)
532 {
533#if D3D12_RHI_RAYTRACING
534 // No state tracking for acceleration structures because they can't have another state
535 check(!EnumHasAnyFlags(InInitialD3D12Access, ED3D12Access::BVHRead | ED3D12Access::BVHWrite) ||
536 !EnumHasAnyFlags(InDefaultD3D12Access, ED3D12Access::BVHRead | ED3D12Access::BVHWrite));
537#endif // D3D12_RHI_RAYTRACING
538
539 check(InInitialD3D12Access != ED3D12Access::Unknown);
540 InitialD3D12Access = InInitialD3D12Access;
541 }

544 void DetermineResourceStates(
545 const D3D12_HEAP_PROPERTIES* InHeapProps,
546 ED3D12Access InInitialD3D12Access,
547 ED3D12ResourceStateMode InResourceStateMode,
548 ED3D12Access InDefaultD3D12Access)
549 {
550 const FD3D12ResourceTypeHelper Type(Desc, HeapType);
551
552 bDepthStencil = Type.bDSV;
553
554#ifdef PLATFORM_SUPPORTS_RESOURCE_COMPRESSION
555 SetCompressedState(D3D12_RESOURCE_STATE_COMMON);
556#endif
557
558 if (!Type.bWritable && InResourceStateMode != ED3D12ResourceStateMode::MultiState)
559 {
560 bRequiresResourceStateTracking = false;
561
562#if UE_BUILD_DEBUG
563 FPlatformAtomics::InterlockedIncrement(&NoStateTrackingResourceCount);
564#endif
565 if (InDefaultD3D12Access != ED3D12Access::Unknown)
566 {
567 DefaultD3D12Access = InDefaultD3D12Access;
568 }
569 else if (Type.bBuffer)
570 {
571 DefaultD3D12Access = DetermineInitialBufferD3D12Access(HeapType, InHeapProps);
572 }
573 else
574 {
575 check(Type.bSRVOnly);
576 DefaultD3D12Access = ED3D12Access::SRVGraphics;
577 }
578
579 check(InInitialD3D12Access != ED3D12Access::Unknown);
580 InitialD3D12Access = InInitialD3D12Access;
581 }
