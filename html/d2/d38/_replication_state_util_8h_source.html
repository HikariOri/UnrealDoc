76inline void MarkDirty(UE::Net::FReplicationStateHeader& InternalState, FNetBitArrayView& MemberChangeMask, const FReplicationStateMemberChangeMaskDescriptor& ChangeMaskInfo)
77{
78 // If this state is bound to a replicated object, notify the replication system that we have data to copy
79 // Note that we only check the first bit of the changemask, as this is used to indicate whether the property is dirty or not
80 // some properties might use additional bits but will only be treated as dirty if the parent bit is set.
81 if (InternalState.IsBound() && !MemberChangeMask.GetBit(ChangeMaskInfo.BitOffset))
82 {
83 FReplicationStateHeaderAccessor::MarkStateDirty(InternalState);
84 MarkNetObjectStateHeaderDirty(InternalState);
85 }
86
87 MemberChangeMask.SetBits(ChangeMaskInfo.BitOffset, ChangeMaskInfo.BitCount);
88}
