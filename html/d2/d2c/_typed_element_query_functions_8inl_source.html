83 template<typename RequestedReturnType, bool bIsConst>
84 static constexpr void StaticValidation()
85 {
86 static_assert(bIsValidQueryFunction, "The provided function is not compatible with a query callback. "
87 "One or more arguments are possibly incompatible.");
88
89 static_assert(ArgumentInfo::CountContexts() <= 1, "Only zero or one context can be added as an argument.");
90 static_assert(ArgumentInfo::CountFlowControls() <= 1, "Only zero or one flow control arguments can be added.");
91
92 if constexpr (bIsConst)
93 {
94 static_assert(ArgumentInfo::IsContextConst(), "Const functions require a const query context (TConstQueryContext).");
95 static_assert(ArgumentInfo::CountMutableColumns() == 0, "Const functions can not bind to mutable columns.");
96 }
97
98 if constexpr (std::is_same_v<RequestedReturnType, void>)
99 {
100 static_assert(ArgumentInfo::ResultIndex == INDEX_NONE,
101 "The query function being build has no return value and can't therefore accept a TResult.");
102 static_assert(std::is_same_v<ReturnType, void>,
103 "The return type of the provided query is expected to be void.");
104 }
105 else
106 {
107 if constexpr (bIsSingleRowProcessor)
108 {
109 if constexpr (ArgumentInfo::ResultIndex >= 0)
110 {
111 static_assert(ArgumentInfo::CountResults() == 1, "Only one TResult can be present.");
112 static_assert(std::is_same_v<typename ArgumentInfo::ResultType, RequestedReturnType>,
113 "The type used for TResult is not compatible with the result type expected for the query function.");
114 static_assert(std::is_same_v<ReturnType, void>,
115 "The return type of the provided query is expected to be void when a TResult is provided.");
116 }
117 else
118 {
119 static_assert(std::is_convertible_v<ReturnType, RequestedReturnType>,
120 "The type returned is not compatible with the result type expected for the query function.");
121 }
122 }
123 else
124 {
125 static_assert(ArgumentInfo::ResultIndex >= 0,
126 "Batch processing a function that expects a result it's required that a TResult is used.");
127 static_assert(ArgumentInfo::CountResults() == 1, "Only one TResult can be present.");
128 static_assert(std::is_same_v<typename ArgumentInfo::ResultType, RequestedReturnType>,
129 "The type used for TResult is not compatible with the result type expected for the query function.");
130 static_assert(std::is_same_v<ReturnType, void>,
131 "The return type of the provided query is expected to be void for batch processing functions.");
132 }
133 }

168 template<bool bIsConst, typename FunctionInfo, typename ArgumentInfo, typename ReturnType, typename FunctionType>
169 EFlowControl CallBody(
170 TResult<ReturnType>& QueryResult,
171 typename TQueryFunctionBase<ReturnType, bIsConst>::ContractType Contract,
172 typename TQueryFunctionBase<ReturnType, bIsConst>::FunctionResponseType Response,
173 typename TQueryFunctionBase<ReturnType, bIsConst>::FunctionSpecializationCallback Specialization,
174 FunctionType&& Callback,
175 EFlowControl IncompleteRowFlowControl)
176 {
177 using ArgumentList = typename ArgumentInfo::ArgumentList;
178
179 TConstArrayView<const UScriptStruct*> ConstColumnTypes = ArgumentInfo::ListConstColumns();
180 const void** ConstColumnsData = static_cast<const void**>(FMemory_Alloca(sizeof(void*) * ConstColumnTypes.Num()));
181 TArrayView<const void*> ConstColumns(ConstColumnsData, ConstColumnTypes.Num());
182
183 TConstArrayView<const UScriptStruct*> MutableColumnTypes;
184 TArrayView<void*> MutableColumns;
185 if constexpr (!bIsConst)
186 {
187 MutableColumnTypes = ArgumentInfo::ListMutableColumns();
188 void** MutableColumnsData = static_cast<void**>(FMemory_Alloca(sizeof(void*) * MutableColumnTypes.Num()));
189 MutableColumns = TArrayView<void*>(MutableColumnsData, MutableColumnTypes.Num());
190 }
191
192 EFlowControl Flow = EFlowControl::Continue;
193
194 ArgumentList Arguments;
195 ArgumentInfo::SetResult(Arguments, QueryResult);
196 ArgumentInfo::SetContext(Arguments, Contract);
197 ArgumentInfo::SetFlowControl(Arguments, Flow);
198
199 FRowHandleArrayView Rows = Response.GetBatchRows();
200 if (!Rows.IsEmpty())
201 {
202 bool bSpecializationResult;
203 if constexpr (bIsConst)
204 {
205 Response.GetConstColumns(ConstColumns, ConstColumnTypes);
206 bSpecializationResult = Specialization(Response, ConstColumns);
207 }
208 else
209 {
210 Response.GetConstColumns(ConstColumns, ConstColumnTypes);
211 Response.GetMutableColumns(MutableColumns, MutableColumnTypes);
212 bSpecializationResult = Specialization(Response, ConstColumns, MutableColumns);
213 }
214
215 if (bSpecializationResult)
216 {
217 ArgumentInfo::SetConstColumns(Arguments, ConstColumns);
218 if constexpr (!bIsConst)
219 {
220 ArgumentInfo::SetMutableColumns(Arguments, MutableColumns);
221 }
222
223 if constexpr (FunctionInfo::bIsSingleRowProcessor)
224 {
225 const RowHandle* It = Rows.GetData();
226 const RowHandle* End = Rows.GetData() + Rows.Num();
227 for (; It != End && Flow == EFlowControl::Continue; ++It)
228 {
229 Response.SetCurrentRow(*It);
230 if constexpr (ArgumentInfo::ResultIndex >= 0 || std::is_same_v<ReturnType, void>)
231 {
232 Arguments.ApplyBefore(Callback);
233 }
234 else
235 {
236 QueryResult.Add(*It, Arguments.ApplyBefore(Callback));
237 }
238 ArgumentInfo::IncrementColumns(Arguments);
239 }
240 }
241 else
242 {
243 Arguments.ApplyBefore(Callback);
244 }
245 return Flow;
246 }
247
248 return IncompleteRowFlowControl;
249 }
250 return EFlowControl::Break;
251 }

261 template<EFunctionCallConfig Config>
262 EFlowControl TQueryFunctionBase<ReturnType, bIsConst>::CallInternal(TResult<ReturnType>& Result, ContractType Contract, FunctionResponseType Response) const
263 {
264 if constexpr (EnumHasAnyFlags(Config, EFunctionCallConfig::VerifyCapabilityCompatibility))
265 {
266 if (!Contract.CheckCompatibility(CapabilityMask))
267 {
268 if constexpr (EnumHasAnyFlags(Config, EFunctionCallConfig::ReportCapabilityCompatibility))
269 {
270 Contract.ReportCompatibility(CapabilityMask);
271 }
272 return EFlowControl::Break;
273 }
274 }
275
276 constexpr EFlowControl IncompleteRowFlowControl = EnumHasAnyFlags(Config, EFunctionCallConfig::ContinueOnIncompleteRow) ? EFlowControl::Continue : EFlowControl::Break;
277
278 if constexpr (bIsConst)
279 {
280 auto Specialization = [](FunctionResponseType Response, TArrayView<const void*> ConstColumns)
281 {
282 if constexpr (EnumHasAnyFlags(Config, EFunctionCallConfig::VerifyColumns))
283 {
284 bool bMissingColumn = false;
285 for (const void* Column : ConstColumns)
286 {
287 bMissingColumn = bMissingColumn || Column == nullptr;
288 }
289 return !bMissingColumn;
290 }
291 else
292 {
293 return true;
294 }
295 };
296 return Function(Result, Contract, Response, Specialization, IncompleteRowFlowControl);
297 }
298 else
299 {
300 auto Specialization = [](FunctionResponseType Response, TArrayView<const void*> ConstColumns, TArrayView<void*> MutableColumns)
301 {
302 if constexpr (EnumHasAnyFlags(Config, EFunctionCallConfig::VerifyColumns))
303 {
304 bool bMissingColumn = false;
305 for (const void* Column : ConstColumns)
306 {
307 bMissingColumn = bMissingColumn || Column == nullptr;
308 }
309 for (void* Column : MutableColumns)
310 {
311 bMissingColumn = bMissingColumn || Column == nullptr;
312 }
313 return !bMissingColumn;
314 }
315 else
316 {
317 return true;
318 }
319 };
320 return Function(Result, Contract, Response, Specialization, IncompleteRowFlowControl);
