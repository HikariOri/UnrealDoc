81struct FAutomatedTestTagFilter
82{
83 GENERATED_BODY()
84
85public:
86
92 FAutomatedTestTagFilter(const FString& InContains)
93 {
94 FString FilterString = InContains.TrimStartAndEnd();
95 if (FilterString.IsEmpty())
96 {
97 TagFilter = nullptr;
98 }
99 else
100 {
101 TagFilter = MakeShared<FTextFilterExpressionEvaluator>(ETextFilterExpressionEvaluatorMode::BasicString);
102 TagFilter->SetFilterText(FText::FromString(FilterString));
103 }
104 }
105
106 virtual ~FAutomatedTestTagFilter() = default;
107
108 FAutomatedTestTagFilter() : FAutomatedTestTagFilter(TEXT("")) {}
109
110 virtual bool PassesFilter(const TSharedPtr< IAutomationReport >& InReport) const
111 {
112 if (TagFilter)
113 {
114 return TagFilter->TestTextFilter(FBasicStringFilterExpressionContext(InReport->GetTags()));
115 }
116 else // disabled filter, reject nothing
117 {
118 return true;
119 }
120 }
121
122private:
123 TSharedPtr<FTextFilterExpressionEvaluator> TagFilter;
124};

128* Describes a filter for a test group with exclude and tag options.
129*/
130USTRUCT()
131struct FAutomatedTestFilter : public FAutomatedTestFilterBase
132{
133 GENERATED_BODY()
134
135public:
136
137 FAutomatedTestFilter(FString InContains, bool InMatchFromStart = false, bool InMatchFromEnd = false)
138 : FAutomatedTestFilterBase(InContains, InMatchFromStart, InMatchFromEnd)
139 {
140 }
141
142 FAutomatedTestFilter() : FAutomatedTestFilter(TEXT("")) {}
143
145 UPROPERTY(Config)
146 TArray<FAutomatedTestFilterBase> Exclude;
147
149 UPROPERTY(Config)
150 TArray<FAutomatedTestTagFilter> Tags;
151
152 virtual bool PassesFilter(const TSharedPtr< IAutomationReport >& InReport) const override
153 {
154 bool bMeetsMatch = Super::PassesFilter(InReport);
155
156 if (bMeetsMatch && !Exclude.IsEmpty())
157 {
158 // Apply exclusion rules
159 for (const FAutomatedTestFilterBase& Filter : Exclude)
160 {
161 if (Filter.PassesFilter(InReport))
162 {
163 return false;
164 }
165 }
166 }
167
168 // Intersect with tag filters
169 if (bMeetsMatch && !Tags.IsEmpty())
170 {
171 for (const FAutomatedTestTagFilter& Filter : Tags)
172 {
173 if (Filter.PassesFilter(InReport))
174 {
175 return true;
176 }
177 }
178 bMeetsMatch = false; // failed to match any tag filter
179 }
180
181 return bMeetsMatch;
182 }
183};
