33 struct FContentionMatrix
34 {
35 public:
36 using WordType = uint64;
37 static constexpr int32 BitsPerWord = 64;
38 static constexpr int32 BitsPerWordLog2 = 6;
39
40 FContentionMatrix() = default;
41
42 explicit FContentionMatrix(int32 InNumElements)
43 {
44 Init(InNumElements);
45 }
46
51 void Init(int32 InNumElements)
52 {
53 check(InNumElements >= 0);
54 NumElements = InNumElements;
55
56 // Calculate how many 64-bit words we need per row
57 WordsPerRow = (NumElements + (BitsPerWord - 1)) / BitsPerWord;
58
59 // Total size = Rows * Width
60 const int32 TotalWords = NumElements * WordsPerRow;
61
62 MatrixBits.Reset();
63 MatrixBits.SetNumZeroed(TotalWords);
64 }
65
66 FORCEINLINE int32 GetSize() const
67 {
68 return NumElements;
69 }
70
71 FORCEINLINE int32 GetWordsPerRow() const
72 {
73 return WordsPerRow;
74 }
75
77 FORCEINLINE void Clear()
78 {
79 FMemory::Memzero(MatrixBits.GetData(), MatrixBits.Num() * sizeof(WordType));
80 }
81
83 void Set(int32 RowIndex, int32 ColumnIndex)
84 {
85 check(IsValidIndex(RowIndex) && IsValidIndex(ColumnIndex));
86 check(RowIndex != ColumnIndex);
87 const int32 WordIndex = (ColumnIndex >> BitsPerWordLog2);
88 const int32 BitOffset = (ColumnIndex & (BitsPerWord - 1));
89
90 // Calculate flattened index: Row * Stride + ColWord
91 const int32 FlatIndex = (RowIndex * WordsPerRow) + WordIndex;
92
93 MatrixBits[FlatIndex] |= (WordType(1) << BitOffset);
94 }
95
97 void SetMutual(int32 RowIndex, int32 ColumnIndex)
98 {
99 Set(RowIndex, ColumnIndex);
100 Set(ColumnIndex, RowIndex);
101 }
102
104 FORCEINLINE bool Test(int32 RowIndex, int32 ColumnIndex) const
105 {
106 check(IsValidIndex(RowIndex) && IsValidIndex(ColumnIndex));
107
108 const int32 WordIndex = (ColumnIndex >> BitsPerWordLog2);
109 const int32 BitOffset = (ColumnIndex & (BitsPerWord-1));
110 const int32 FlatIndex = (RowIndex * WordsPerRow) + WordIndex;
111
112 return ((MatrixBits[FlatIndex] >> BitOffset) & 1u) != 0;
113 }
114
116 FORCEINLINE const WordType* GetRowWordPointer(int32 RowIndex) const
117 {
118 check(IsValidIndex(RowIndex));
119 return &MatrixBits[RowIndex * WordsPerRow];
120 }
121
122 private:
123 int32 NumElements = 0;
124 int32 WordsPerRow = 0;
125
126 TArray<WordType> MatrixBits;
127
128 FORCEINLINE bool IsValidIndex(int32 Index) const
129 {
130 return Index >= 0 && Index < NumElements;
131 }
132 };

176 class FActiveChunkTracker
177 {
178 public:
179 using WordType = uint64;
180 static constexpr int32 MaxActiveChunks = UE_MASS_MAX_ACTIVE_CHUNKS;
181 static constexpr uint64 InvalidChunkID = 0;
182 FActiveChunkTracker() = default;
183
184 void Init(const FContentionMatrix& InMatrix)
185 {
186 MatrixRef = &InMatrix;
187 WordsPerRow = InMatrix.GetWordsPerRow();
188
189 // Initialize IDs to 0
190 for (int32 i = 0; i < MaxActiveChunks; ++i)
191 {
192 ChunkIDs[i].store(InvalidChunkID, std::memory_order_relaxed);
193 SlotRefCounts[i] = 0;
194 SlotLocks[i].clear();
195 }
196
197 SlotAssignmentLock.clear();
198
199 // Resize bitmask buffer: MaxActiveChunks slots * WordsPerRow
200 BitmaskBuffer.Reset();
201 BitmaskBuffer.SetNumZeroed(MaxActiveChunks * WordsPerRow);
202 }
203
207 bool TryAcquireAccess(FQueuedChunk& Chunk, int32 DirectChunkContentionIndex)
208 {
209 const uint64 ChunkID = Chunk.GetChunkID();
210 checkSlow(ChunkID != InvalidChunkID);
211 checkSlow(DirectChunkContentionIndex >= 0);
212 checkSlow(Chunk.ActiveChunkSlotIndex == -1);
213
214 // Fast scan (relaxed)
215 int32 FoundIndex = -1;
216 for (int32 i = 0; i < MaxActiveChunks; ++i)
217 {
218 if (ChunkIDs[i].load(std::memory_order_relaxed) == ChunkID)
219 {
220 FoundIndex = i;
221 break;
222 }
223 }
224
225 if (FoundIndex != -1)
226 {
227 // Existing slot
228 LockSlot(FoundIndex);
229
230 // Validate slot ownership
231 if (ChunkIDs[FoundIndex].load(std::memory_order_acquire) != ChunkID)
232 {
233 UnlockSlot(FoundIndex);
234 // Retry from scratch (extremely unlikely)
235 return TryAcquireAccess(Chunk, DirectChunkContentionIndex);
236 }
237
238 WordType* SlotBits = GetSlotBits(FoundIndex);
239 const WordType* JobRow = MatrixRef->GetRowWordPointer(DirectChunkContentionIndex);
240
241 // Contention check
242 for (int32 w = 0; w < WordsPerRow; ++w)
243 {
244 if ((SlotBits[w] & JobRow[w]) != 0)
245 {
246 UnlockSlot(FoundIndex);
247 return false;
248 }
249 }
250
251 // Mark active
252 AddJobToBitmask(SlotBits, DirectChunkContentionIndex);
253 ++SlotRefCounts[FoundIndex];
254
255 UnlockSlot(FoundIndex);
256
257 Chunk.ActiveChunkSlotIndex = FoundIndex;
258 return true;
259 }
260
261 // assign slot for new ChunkID
262 while (SlotAssignmentLock.test_and_set(std::memory_order_acquire))
263 {
264 FPlatformProcess::YieldCycles(10);
265 }
266
267 // Double-check under AllocationLock (someone may have added it)
268 for (int32 i = 0; i < MaxActiveChunks; ++i)
269 {
270 if (ChunkIDs[i].load(std::memory_order_relaxed) == ChunkID)
271 {
272 SlotAssignmentLock.clear(std::memory_order_release);
273 // Re-run the found path (will lock+validate)
274 return TryAcquireAccess(Chunk, DirectChunkContentionIndex);
275 }
276 }
277
278 int32 AssignedSlotIndex = -1;
279 // find an empty slot
280 for (int32 i = 0; i < MaxActiveChunks; ++i)
281 {
282 if (ChunkIDs[i].load(std::memory_order_acquire) == InvalidChunkID)
283 {
284 AssignedSlotIndex = i;
285 break;
286 }
287 }
288
289 if (AssignedSlotIndex == -1)
290 {
291 SlotAssignmentLock.clear(std::memory_order_release);
292 // Soft overflow: every slot is in use. Caller (FProcessingQueue::TryActivateChunk)
293 // already handles `false` as "couldn't acquire right now"; the chunk is left in
294 // the queue and retried later. If hit regularly, raise MaxActiveChunks via
295 // Target.MassMaxActiveChunks in the .Target.cs.
296 OnSlotExhausted();
297 return false;
298 }
299
300 // Initialize under slot lock BEFORE setting ChunkID
301 LockSlot(AssignedSlotIndex);
302
303#if DO_CHECK
304 check(ChunkIDs[AssignedSlotIndex].load(std::memory_order_relaxed) == InvalidChunkID);
305#endif
306
307 WordType* SlotBits = GetSlotBits(AssignedSlotIndex);
308 FMemory::Memzero(SlotBits, WordsPerRow * sizeof(WordType));
309
310#if DO_CHECK
311 const int32 WordIdx = (DirectChunkContentionIndex >> FContentionMatrix::BitsPerWordLog2);
312 check(WordIdx >= 0 && WordIdx < WordsPerRow);
313#endif
314
315 AddJobToBitmask(SlotBits, DirectChunkContentionIndex);
316 SlotRefCounts[AssignedSlotIndex] = 1;
317
318 // Publish LAST
319 ChunkIDs[AssignedSlotIndex].store(ChunkID, std::memory_order_release);
320
321 UnlockSlot(AssignedSlotIndex);
322 SlotAssignmentLock.clear(std::memory_order_release);
323
324 Chunk.ActiveChunkSlotIndex = AssignedSlotIndex;
325 return true;
326 }
327
328 void ReleaseAccess(FQueuedChunk& Chunk, int32 DirectChunkContentionIndex)
329 {
330 const int32 SlotIndex = static_cast<int32>(Chunk.ActiveChunkSlotIndex);
331 checkSlow(SlotIndex >= 0 && SlotIndex < MaxActiveChunks);
332
333 const uint64 ChunkID = Chunk.GetChunkID();
334 checkSlow(ChunkID != InvalidChunkID);
335 checkSlow(DirectChunkContentionIndex >= 0);
336
337 LockSlot(SlotIndex);
338
339 // Validate slot still belongs to this chunk
340 check(ChunkIDs[SlotIndex].load(std::memory_order_acquire) == ChunkID);
341
342 WordType* SlotBits = GetSlotBits(SlotIndex);
343
344 const int32 WordIdx = (DirectChunkContentionIndex >> FContentionMatrix::BitsPerWordLog2);
345
346#if DO_CHECK
347 check(WordIdx >= 0 && WordIdx < WordsPerRow);
348#endif
349 const int32 BitOffset = (DirectChunkContentionIndex & (FContentionMatrix::BitsPerWord - 1));
350 SlotBits[WordIdx] &= ~(WordType(1) << BitOffset);
351
352 --SlotRefCounts[SlotIndex];
353#if DO_CHECK
354 check(SlotRefCounts[SlotIndex] >= 0);
355#endif
356
357 if (SlotRefCounts[SlotIndex] == 0)
358 {
359 // release so prior writes are visible before anyone observes Invalid and reuses slot
360 ChunkIDs[SlotIndex].store(InvalidChunkID, std::memory_order_release);
361 }
362
363 UnlockSlot(SlotIndex);
364
365 Chunk.ActiveChunkSlotIndex = INDEX_NONE;
366 }
367
368
369 private:
371 UE_API void OnSlotExhausted();
372
373 const FContentionMatrix* MatrixRef = nullptr;
374 int32 WordsPerRow = 0;
375
376 alignas(64) std::atomic<uint64> ChunkIDs[MaxActiveChunks];
377 std::atomic_flag SlotLocks[MaxActiveChunks];
378 int32 SlotRefCounts[MaxActiveChunks];
379
380 std::atomic_flag SlotAssignmentLock;
381 TArray<WordType> BitmaskBuffer;
382
383 FORCEINLINE WordType* GetSlotBits(int32 SlotIndex)
384 {
385 return &BitmaskBuffer[SlotIndex * WordsPerRow];
386 }
387
388 FORCEINLINE void LockSlot(int32 SlotIndex)
389 {
390 while (SlotLocks[SlotIndex].test_and_set(std::memory_order_acquire))
391 {
392 FPlatformProcess::YieldCycles(10);
393 }
394 }
395
396 FORCEINLINE void UnlockSlot(int32 SlotIndex)
397 {
398 SlotLocks[SlotIndex].clear(std::memory_order_release);
399 }
400
401 FORCEINLINE void AddJobToBitmask(WordType* SlotBits, int32 JobIndex)
402 {
403 const int32 WordIdx = (JobIndex >> FContentionMatrix::BitsPerWordLog2);
404 const int32 BitOffset = (JobIndex & (FContentionMatrix::BitsPerWord - 1));
405 SlotBits[WordIdx] |= (WordType(1) << BitOffset);
406 }

412 struct FActiveContentionTracker
413 {
414 public:
415 using WordType = uint64;
416 static constexpr int32 BitsPerWord = 64;
417 static constexpr int32 BitsPerWordLog2 = 6;
418
419 FActiveContentionTracker(const FContentionMatrix& InMatrix)
420 : Matrix(InMatrix)
421 {
422 Init();
423 }
424
425 void Init()
426 {
427 const int32 Size = Matrix.GetSize();
428 ActiveContenderCount.SetNumZeroed(Size);
429 }
430
431 bool TryActivate(int32 ItemIndex)
432 {
433 check(IsValidIndex(ItemIndex));
434
435 if (HasAnyActiveContention(ItemIndex))
436 {
437 return false;
438 }
439
440 // assume success and update contention counts first
441 UpdateContentionCounts(ItemIndex, 1);
442
443 // see if a contender activated while this thread was in UpdateContentionCounts
444 if (HasAnyActiveContention(ItemIndex))
445 {
446 FScopeLock Lock(&ActiveContenderCriticalSection);
447 // check one more time inside the critical section to eliminate all possible races
448 if (HasAnyActiveContention(ItemIndex))
449 {
450 // undo the change to the contention counts
451 UpdateContentionCounts(ItemIndex, -1);
452 return false;
453 }
454 }
455
456 return true;
457 }
458
459 FORCEINLINE bool Deactivate(int32 ItemIndex)
460 {
461 check(IsValidIndex(ItemIndex));
462 UpdateContentionCounts(ItemIndex, -1);
463 return true;
464 }
465
466 FORCEINLINE bool HasAnyActiveContention(int32 ItemIndex)
467 {
468 check(IsValidIndex(ItemIndex));
469
470 // not all platforms support atomic std::atomic_ref, but it's typically faster so do it when we can
471#if defined(__cpp_lib_atomic_ref)
472 std::atomic_ref<int32> AtomicCount(ActiveContenderCount[ItemIndex]);
473 return AtomicCount.load(std::memory_order_acquire) > 0;
474#else
475 return FPlatformAtomics::AtomicRead(reinterpret_cast<volatile int32*>(&ActiveContenderCount[ItemIndex])) > 0;
476#endif
477 }
478
479 FORCEINLINE int32 GetActiveContentionCount(int32 ItemIndex)
480 {
481 check(IsValidIndex(ItemIndex));
482 // not all platforms support atomic std::atomic_ref, but it's typically faster so do it when we can
483#if defined(__cpp_lib_atomic_ref)
484 std::atomic_ref<int32> AtomicCount(ActiveContenderCount[ItemIndex]);
485 return AtomicCount.load(std::memory_order_acquire);
486#else
487 return FPlatformAtomics::AtomicRead(reinterpret_cast<volatile int32*>(&ActiveContenderCount[ItemIndex]));
488#endif
489 }
490
491 private:
492 const FContentionMatrix& Matrix;
493 TArray<int32> ActiveContenderCount;
494
495 FCriticalSection ActiveContenderCriticalSection;
496
497 FORCEINLINE bool IsValidIndex(int32 Index) const
498 {
499 return Index >= 0 && Index < Matrix.GetSize();
500 }
501
502 void UpdateContentionCounts(int32 ItemIndex, int32 Delta)
503 {
504 const WordType* RESTRICT RowWords = Matrix.GetRowWordPointer(ItemIndex);
505 const int32 MatrixSize = Matrix.GetSize();
506 const int32 WordsPerRow = Matrix.GetWordsPerRow();
507 const int32 DiagWordIndex = (ItemIndex >> BitsPerWordLog2);
508 const int32 DiagBitOffset = (ItemIndex & (BitsPerWord-1));
509
510 int32* RawCounts = ActiveContenderCount.GetData();
511
512 for (int32 RowWordIndex = 0, BaseBitIndex = 0; RowWordIndex < WordsPerRow; ++RowWordIndex, BaseBitIndex += BitsPerWord)
513 {
514 WordType ContenderWord = (RowWordIndex != DiagWordIndex)
515 ? RowWords[RowWordIndex]
516 : (RowWords[RowWordIndex] & ~(WordType(1) << DiagBitOffset));
517
518 while (ContenderWord != 0)
519 {
520 const int32 TrailingZeroCount = static_cast<int32>(FMath::CountTrailingZeros64(ContenderWord));
521 const int32 NeighborIndex = BaseBitIndex + TrailingZeroCount;
522
523 if (NeighborIndex < MatrixSize)
524 {
525 // not all platforms support atomic std::atomic_ref, but it's typically faster so do it when we can
526#if defined(__cpp_lib_atomic_ref)
527 std::atomic_ref<int32> NeighborCount(RawCounts[NeighborIndex]);
528 NeighborCount.fetch_add(Delta, std::memory_order_release);
529#else
530 FPlatformAtomics::InterlockedAdd(reinterpret_cast<volatile int32*>(&RawCounts[NeighborIndex]), Delta);
531#endif
532 }
533
534 ContenderWord &= (ContenderWord - 1);
535 }
536 }
537 }
538 };

541 struct FActiveChunkSet
542 {
543 private:
544 struct FChunkSlot
545 {
546 uint64 Index;
547
548 };
549 public:
550 static constexpr int32 Capacity = 32;
551 static constexpr uint64 EmptyValue = 0;
552 const int32 MaxSlots;
553
554 FActiveChunkSet(int32 InMaxSlots)
555 : MaxSlots(InMaxSlots)
556 {
557 checkf(MaxSlots <= Capacity, TEXT("FActiveChunkSet::MaxSlots (%d) exceeded Capacity (%d)."), MaxSlots, Capacity);
558
559 for (std::atomic<uint64>& Slot : Slots)
560 {
561 Slot.store(EmptyValue, std::memory_order_relaxed);
562 }
563 for (std::atomic<int32>& RefCount : RefCounts)
564 {
565 RefCount.store(0, std::memory_order_relaxed);
566 }
567 }
568
573 bool TryAddExclusive(uint64 ID)
574 {
575 check(ID != EmptyValue);
576
577 int32 CandidateIndex = INDEX_NONE;
578
579 for (;;)
580 {
581 for (int32 Index = 0; Index < MaxSlots; ++Index)
582 {
583 const uint64 Val = Slots[Index].load(std::memory_order_seq_cst);
584
585 if (Val == ID)
586 {
587 return false; // Already exists
588 }
589
590 if (CandidateIndex == INDEX_NONE && Val == EmptyValue)
591 {
592 CandidateIndex = Index;
593 break;
594 }
595 }
596
597 if (CandidateIndex != INDEX_NONE)
598 {
599 uint64 Expected = EmptyValue;
600 if (Slots[CandidateIndex].compare_exchange_strong(Expected, ID, std::memory_order_seq_cst))
601 {
602 // this is an exclusive add so we can just force refcount to 1 for the previously empty slot
603 RefCounts[CandidateIndex].store(1, std::memory_order_seq_cst);
604 return true;
605 }
606
607 // someone else took this slot or added an item, loop and try again
608 }
609 else
610 {
611 checkf(false, TEXT("FActiveChunkSet overflow! Active chunks exceeded Capacity (%d)."), Capacity);
612 return false;
613 }
614 }
615 }
616
622 int32 AddGetRefCount(uint64 ID)
623 {
624 check(ID != EmptyValue);
625
626 int32 CandidateIndex = INDEX_NONE;
627
628 for (;;)
629 {
630 for (int32 Index = 0; Index < MaxSlots; ++Index)
631 {
632 const uint64 Val = Slots[Index].load(std::memory_order_seq_cst);
633
634 if (Val == ID)
635 {
636 int32 NewCount = RefCounts[Index].fetch_add(1, std::memory_order_seq_cst) + 1;
637 return NewCount;
638 }
639
640 if (CandidateIndex == INDEX_NONE && Val == EmptyValue)
641 {
642 CandidateIndex = Index;
643 break;
644 }
645 }
646
647 if (CandidateIndex != INDEX_NONE)
648 {
649 uint64 Expected = EmptyValue;
650 if (Slots[CandidateIndex].compare_exchange_strong(Expected, ID, std::memory_order_seq_cst))
651 {
652 // this is an exclusive add so we can just force refcount to 1 for the previously empty slot
653 RefCounts[CandidateIndex].store(1, std::memory_order_seq_cst);
654 return 1;
655 }
656
657 // someone else took this slot or added an item, loop and try again
658 }
659 else
660 {
661 checkf(false, TEXT("FActiveChunkSet overflow! Active chunks exceeded MaxSlots (%d)."), MaxSlots);
662 return 0;
663 }
664 }
665 }
666
670 void Remove(uint64 ID)
671 {
672 checkSlow(ID != EmptyValue);
673
674 for (int32 Index = 0; Index < MaxSlots; ++Index)
675 {
676 std::atomic<uint64>& Slot = Slots[Index];
677
678 if (Slot.load(std::memory_order_seq_cst) == ID)
679 {
680 int32 NewCount = RefCounts[Index].fetch_add(-1, std::memory_order_seq_cst) - 1;
681
682 if (NewCount == 0)
683 {
684 Slot.store(EmptyValue, std::memory_order_seq_cst);
685 // double check that the refcount is still 0:
686 NewCount = RefCounts[Index].load(std::memory_order_seq_cst);
687 if (NewCount > 0)
688 {
689 // something else added a refcount between our operations
690 // restore the ID and move on
691 Slot.store(ID, std::memory_order_seq_cst);
692 }
693 return;
694 }
695 }
696 }
697 }
698
702 bool Contains(uint64 ID) const
703 {
704 checkSlow(ID != EmptyValue);
705 for (int32 i = 0; i < MaxSlots; ++i)
706 {
707 if (Slots[i].load(std::memory_order_acquire) == ID)
708 {
709 return true;
710 }
711 }
712 return false;
713 }
714
715 private:
716 // alignas(64) to avoid false sharing
717 alignas(64) std::atomic<uint64> Slots[Capacity];
718 alignas(64) std::atomic<int32> RefCounts[Capacity];
719 };

750 template <typename T>
751 struct FChangeListener
752 {
753 explicit FChangeListener(std::atomic<T>& InValueToTrack)
754 : ValueToTrack(InValueToTrack)
755 // Set the initial value at construction time to capture changes that happen between declaration and subsequent Wait calls.
756 , InitialValue(ValueToTrack.load(std::memory_order_acquire))
757 {
758 }
759
760 void WaitForChange()
761 {
762 // This spin can run inside a UE::Tasks worker (the queue runner, a processor worker, or
763 // ProcessChunks). Mark the worker as stalled so the scheduler can wake or spawn a standby worker
764 // to run the other queued Mass tasks that will advance the queue and release us. No-op when not
765 // on a task worker (e.g. the game thread).
766 LowLevelTasks::FOversubscriptionScope OversubscriptionScope;
767 for (;;)
768 {
769 const T CurrentValue = ValueToTrack.load(std::memory_order_acquire);
770 if (CurrentValue != InitialValue)
771 {
772 // caller might wait for changes again
773 InitialValue = CurrentValue;
774 return;
775 }
776 FPlatformProcess::YieldThread();
777 }
778 }
779 private:
780 std::atomic<T>& ValueToTrack;
781 T InitialValue;
782 };
