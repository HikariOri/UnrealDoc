65template <typename IntType, ERefCountIsQueryable Queryable = ERefCountIsQueryable::Yes>
66class TAtomicRefCount
67{
68public:
69 TAtomicRefCount() = default;
70
71 explicit TAtomicRefCount(IntType InitialRefCount)
72 : RefCount(InitialRefCount)
73 {
74 }
75
76 template <auto DeleteFn>
77 IntType AddRef() const
78 {
79 if (!AutoRTFM::IsClosed())
80 {
81 return ++RefCount;
82 }
83
84 IntType Refs;
85
86 UE_AUTORTFM_OPEN
87 {
88 Refs = RefCount.fetch_add(1, std::memory_order_relaxed);
89 };
90
91 // As we are inside a transaction, the rollback must undo our refcount bump.
92 // In general, this is best handled by Release(). However, there is one case
93 // that needs to be handled with special care. A brand-new object has a
94 // refcount of zero, and a rollback must return it to this zero-refcount state.
95 // However, calling AddRef() followed by Release() would not accomplish this;
96 // instead, it would free the object entirely! We need to guard against this,
97 // since it could lead to a double-free, so we detect the zero-reference case
98 // and special-case it.
99 if (Refs == 0)
100 {
101 UE_AUTORTFM_ONABORT(this)
102 {
103 RefCount.fetch_sub(1, std::memory_order_release);
104 // The refcount is likely zero now, but leaving the object alive isn't a leak.
105 // We are restoring the object back to its initial "just-created" state.
106 };
107 }
108 else
109 {
110 UE_AUTORTFM_ONABORT(this)
111 {
112 Release<DeleteFn>();
113 };
114 }
115
116 return Refs + 1 - TransactionallyDeferredReleaseCount;
117 }
118
119 template <auto DeleteFn>
120 IntType Release() const
121 {
122 #if DO_GUARD_SLOW
123 if (RefCount.load(std::memory_order_relaxed) == 0)
124 {
125 CheckRefCountIsNonZero();
126 }
127 #endif
128
129 if (AutoRTFM::IsClosed())
130 {
131 IntType Refs;
132
133 UE_AUTORTFM_OPEN
134 {
135 if constexpr (Queryable == ERefCountIsQueryable::Yes)
136 {
137 TransactionallyDeferredReleaseCount++;
138 }
139 Refs = RefCount.load(std::memory_order_relaxed) - TransactionallyDeferredReleaseCount;
140 };
141
142 if constexpr (Queryable == ERefCountIsQueryable::Yes)
143 {
144 UE_AUTORTFM_ONABORT(this)
145 {
146 #if DO_GUARD_SLOW
147 if (TransactionallyDeferredReleaseCount == 0)
148 {
149 CheckRefCountIsNonZero();
150 }
151 #endif
152 TransactionallyDeferredReleaseCount--;
153 };
154 }
155
156 // Refcount changes and frees are deferred until the transaction is concluded.
157 UE_AUTORTFM_ONCOMMIT(this)
158 {
159 if constexpr (Queryable == ERefCountIsQueryable::Yes)
160 {
161 #if DO_GUARD_SLOW
162 if (TransactionallyDeferredReleaseCount == 0)
163 {
164 CheckRefCountIsNonZero();
165 }
166 #endif
167 TransactionallyDeferredReleaseCount--;
168 }
169 ImmediatelyRelease<DeleteFn>();
170 };
171
172 return Refs;
173 }
174 else
175 {
176 return ImmediatelyRelease<DeleteFn>() - 1;
177 }
178 }
179
180 // This is equivalent to https://en.cppreference.com/w/cpp/memory/shared_ptr/use_count
181 // This loads with a relaxed memory ordering because a 'live' reference count is
182 // unstable by nature and so there's no benefit to try and enforce memory ordering
183 // around the reading of it.
184 IntType GetRefCount() const
185 requires (Queryable == ERefCountIsQueryable::Yes)
186 {
187 if (AutoRTFM::IsClosed())
188 {
189 IntType Refs;
190 UE_AUTORTFM_OPEN
191 {
192 Refs = RefCount.load(std::memory_order_relaxed) - TransactionallyDeferredReleaseCount;
193 };
194 return Refs;
195 }
196 else
197 {
198 return RefCount.load(std::memory_order_relaxed);
199 }
200 }
201
202private:
203 template <auto DeleteFn>
204 IntType ImmediatelyRelease() const
205 {
206 // fetch_sub returns the refcount _before_ it was decremented. std::memory_order_acq_rel is
207 // used so that, if we do end up executing the destructor, it's not possible for side effects
208 // from executing the destructor to end up being visible before we've determined that the
209 // reference count is actually zero.
210 IntType RefsBeforeRelease = RefCount.fetch_sub(1, std::memory_order_acq_rel);
211
212 #if DO_GUARD_SLOW
213 // A check-failure is issued if an object is over-released.
214 if (RefsBeforeRelease == 0)
215 {
216 CheckRefCountIsNonZero();
217 }
218 #endif
219
220 // We immediately free the object if its refcount has become zero.
221 if (RefsBeforeRelease == 1)
222 {
223 DeleteFn(const_cast<TAtomicRefCount*>(this));
224 }
225
226 return RefsBeforeRelease;
227 }
228
229 mutable std::atomic<IntType> RefCount = 0;
230 UE_NO_UNIQUE_ADDRESS mutable TQueryableTransactionalType<IntType, Queryable> TransactionallyDeferredReleaseCount = 0;
231};
