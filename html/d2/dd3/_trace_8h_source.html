55// Reference to a definition event.
56template<typename IdType>
57struct TEventRef
58{
59 using ReferenceType = IdType;
60
61 TEventRef(IdType InId, uint32 InTypeId)
62 : Id(InId)
63 , RefTypeId(InTypeId)
64 {
65 }
66
67 IdType Id;
68 uint32 RefTypeId;
69
70 uint64 GetHash() const
71 {
72 if constexpr (std::is_same_v<IdType, uint64>)
73 {
74 return (uint64(RefTypeId) << 32) ^ Id;
75 }
76 else
77 {
78 return (uint64(RefTypeId) << 32) | Id;
79 }
80 }
81
82private:
83 TEventRef() = delete;
84};
