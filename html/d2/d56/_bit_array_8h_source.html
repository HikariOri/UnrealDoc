73 template <typename AllocatorA, typename AllocatorB, typename OutAllocator, typename ProjectionType>
74 void BitwiseBinaryOperatorImpl(const TBitArray<AllocatorA>& InA, const TBitArray<AllocatorB>& InB, TBitArray<OutAllocator>& OutResult, EBitwiseOperatorFlags InFlags, ProjectionType&& InProjection)
75 {
76 check((const void*)&InA != (const void*)&InB && (const void*)&InA != (const void*)&OutResult && (const void*)&InB != (const void*)&OutResult);
77
78 if (EnumHasAnyFlags(InFlags, EBitwiseOperatorFlags::MinSize))
79 {
80
81 const int32 MinNumBits = FMath::Min(InA.Num(), InB.Num());
82 if (MinNumBits > 0)
83 {
84 OutResult.Reserve(MinNumBits);
85 OutResult.NumBits = MinNumBits;
86
87 typename TBitArray<AllocatorA>::FConstWordIterator IteratorA(InA);
88 typename TBitArray<AllocatorB>::FConstWordIterator IteratorB(InB);
89
90 typename TBitArray<OutAllocator>::FWordIterator IteratorResult(OutResult);
91
92 for ( ; IteratorResult; ++IteratorResult, ++IteratorA, ++IteratorB)
93 {
94 const uint32 NewValue = Invoke(InProjection, IteratorA.GetWord(), IteratorB.GetWord());
95 IteratorResult.SetWord(NewValue);
96 }
97 }
98
99 }
100 else if (EnumHasAnyFlags(InFlags, EBitwiseOperatorFlags::MaxSize))
101 {
102
103 const int32 MaxNumBits = FMath::Max(InA.Num(), InB.Num());
104 const uint32 MissingBitsFill = EnumHasAnyFlags(InFlags, EBitwiseOperatorFlags::OneFillMissingBits) ? ~0u : 0;
105
106 if (MaxNumBits)
107 {
108 OutResult.Reserve(MaxNumBits);
109 OutResult.NumBits = MaxNumBits;
110
111 typename TBitArray<AllocatorA>::FConstWordIterator IteratorA(InA);
112 typename TBitArray<AllocatorB>::FConstWordIterator IteratorB(InB);
113
114 IteratorA.FillMissingBits(MissingBitsFill);
115 IteratorB.FillMissingBits(MissingBitsFill);
116
117 typename TBitArray<OutAllocator>::FWordIterator IteratorResult(OutResult);
118
119 for ( ; IteratorResult; ++IteratorResult, ++IteratorA, ++IteratorB)
120 {
121 uint32 A = IteratorA ? IteratorA.GetWord() : MissingBitsFill;
122 uint32 B = IteratorB ? IteratorB.GetWord() : MissingBitsFill;
123
124 IteratorResult.SetWord(Invoke(InProjection, A, B));
125 }
126 }
127
128 }
129 else
130 {
131 checkf(false, TEXT("Invalid size flag specified for binary bitwise AND"));
132 }
133
134 OutResult.CheckInvariants();
135 }

138 template<typename OtherAllocator, typename OutAllocator, typename ProjectionType>
139 void BitwiseOperatorImpl(const TBitArray<OtherAllocator>& InOther, TBitArray<OutAllocator>& OutResult, EBitwiseOperatorFlags InFlags, ProjectionType&& InProjection)
140 {
141 if constexpr (std::is_same_v<OtherAllocator, OutAllocator>)
142 {
143 check(&InOther != &OutResult);
144 }
145
146 int32 NewNumBits = OutResult.NumBits;
147 if (EnumHasAnyFlags(InFlags, EBitwiseOperatorFlags::MinSize))
148 {
149 NewNumBits = FMath::Min(InOther.Num(), OutResult.Num());
150 }
151 else if (EnumHasAnyFlags(InFlags, EBitwiseOperatorFlags::MaxSize))
152 {
153 NewNumBits = FMath::Max(InOther.Num(), OutResult.Num());
154 }
155
156 const int32 SizeDifference = NewNumBits - OutResult.NumBits;
157 if (SizeDifference < 0)
158 {
159 OutResult.NumBits = NewNumBits;
160 OutResult.ClearPartialSlackBits();
161 }
162 else if (SizeDifference > 0)
163 {
164 const bool bPadValue = EnumHasAnyFlags(InFlags, EBitwiseOperatorFlags::OneFillMissingBits);
165 OutResult.Add(bPadValue, SizeDifference);
166 }
167
168 const uint32 MissingBitsFill = EnumHasAnyFlags(InFlags, EBitwiseOperatorFlags::OneFillMissingBits) ? ~0u : 0;
169 if (OutResult.NumBits != 0)
170 {
171 typename TBitArray<OtherAllocator>::FConstWordIterator IteratorOther(InOther);
172 IteratorOther.FillMissingBits(MissingBitsFill);
173
174 typename TBitArray<OutAllocator>::FWordIterator IteratorResult(OutResult);
175
176 for ( ; IteratorResult; ++IteratorResult, ++IteratorOther)
177 {
178 const uint32 OtherValue = IteratorOther ? IteratorOther.GetWord() : MissingBitsFill;
179 IteratorResult.SetWord(Invoke(InProjection, IteratorResult.GetWord(), OtherValue));
180 }
181 }
182
183 OutResult.CheckInvariants();
184 }

898 FORCENOINLINE void SetRange(int32 Index, int32 NumBitsToSet, bool Value)
899 {
900 check(Index >= 0 && NumBitsToSet >= 0 && Index + NumBitsToSet <= NumBits);
901
902 if (NumBitsToSet == 0)
903 {
904 return;
905 }
906
907 // Work out which uint32 index to set from, and how many
908 uint32 StartIndex = Index / NumBitsPerDWORD;
909 uint32 Count = (Index + NumBitsToSet + (NumBitsPerDWORD - 1)) / NumBitsPerDWORD - StartIndex;
910
911 // Work out masks for the start/end of the sequence
912 uint32 StartMask = FullWordMask << (Index % NumBitsPerDWORD);
913 uint32 EndMask = FullWordMask >> (NumBitsPerDWORD - (Index + NumBitsToSet) % NumBitsPerDWORD) % NumBitsPerDWORD;
914
915 uint32* Data = GetData() + StartIndex;
916 if (Value)
917 {
918 if (Count == 1)
919 {
920 *Data |= StartMask & EndMask;
921 }
922 else
923 {
924 *Data++ |= StartMask;
925 Count -= 2;
926 while (Count != 0)
927 {
928 *Data++ = ~0u;
929 --Count;
930 }
931 *Data |= EndMask;
932 }
933 }
934 else
935 {
936 if (Count == 1)
937 {
938 *Data &= ~(StartMask & EndMask);
939 }
940 else
941 {
942 *Data++ &= ~StartMask;
943 Count -= 2;
944 while (Count != 0)
945 {
946 *Data++ = 0;
947 --Count;
948 }
949 *Data &= ~EndMask;
950 }
951 }
952
953 CheckInvariants();
954 }

1033 * This is mainly implemented so that other code using TArray::RemoveSwap will have
1034 * matching indices.
1035 * @param BaseIndex - The index of the first bit to remove.

1102 int32 FindFromImpl(bool bValue, int32 StartIndex, int32 EndIndexExclusive) const
1103 {
1104 // Produce a mask for the first iteration
1105 uint32 Mask = ~0u << (StartIndex % FBitSet::BitsPerWord);
1106
1107 // Iterate over the array until we see a word with a matching bit
1108 const uint32 Test = bValue ? 0u : ~0u;
1109
1110 const uint32* RESTRICT DwordArray = GetData();
1111 const int32 DwordCount = FBitSet::CalculateNumWords(EndIndexExclusive);
1112 int32 DwordIndex = FMath::DivideAndRoundDown(StartIndex, NumBitsPerDWORD);
1113 while (DwordIndex < DwordCount && (DwordArray[DwordIndex] & Mask) == (Test & Mask))
1114 {
1115 ++DwordIndex;
1116 Mask = ~0u;
1117 }
1118
1119 if (DwordIndex < DwordCount)
1120 {
1121 // If we're looking for a false, then we flip the bits - then we only need to find the first one bit
1122 const uint32 Bits = (bValue ? DwordArray[DwordIndex] : ~DwordArray[DwordIndex]) & Mask;
1123 UE_ASSUME(Bits != 0);
1124 const int32 LowestBitIndex = FMath::CountTrailingZeros(Bits) + (DwordIndex << NumBitsPerDWORDLogTwo);
1125 if (LowestBitIndex < EndIndexExclusive)
1126 {
1127 return LowestBitIndex;
1128 }
1129 }
1130
1131 return INDEX_NONE;
1132 }

1134 int32 FindLastFromImpl(bool bValue, int32 EndIndexExclusive) const
1135 {
1136 // We need to early out on empty bit arrays in order to avoid iterating on the first (inexistent) DWord :
1137 if (NumBits == 0)
1138 {
1139 return INDEX_NONE;
1140 }
1141
1142 // Produce a mask for the first iteration
1143 uint32 Mask = GetLastWordMask(EndIndexExclusive);
1144
1145 // Iterate over the array until we see a word with a matching bit
1146 const uint32* RESTRICT DwordArray = GetData();
1147 uint32 DwordIndex = FBitSet::CalculateNumWords(EndIndexExclusive);
1148 const uint32 Test = bValue ? 0u : ~0u;
1149 for (;;)
1150 {
1151 if (DwordIndex == 0)
1152 {
1153 return INDEX_NONE;
1154 }
1155 --DwordIndex;
1156 if ((DwordArray[DwordIndex] & Mask) != (Test & Mask))
1157 {
1158 break;
1159 }
1160 Mask = ~0u;
1161 }
1162
1163 // If we're looking for a false, then we flip the bits - then we only need to find the last one bit
1164 const uint32 Bits = (bValue ? DwordArray[DwordIndex] : ~DwordArray[DwordIndex]) & Mask;
1165 UE_ASSUME(Bits != 0);
1166
1167 uint32 BitIndex = (NumBitsPerDWORD - 1) - FMath::CountLeadingZeros(Bits);
1168
1169 int32 Result = BitIndex + (DwordIndex << NumBitsPerDWORDLogTwo);
1170 return Result;
1171 }
