35 // This exists to try and catch incomplete types in the same way as UE::CCompleteType, but when it
36 // comes to conversions, a complete type checker is too blunt - there are many common conversion
37 // tests that can be accurately done without needing complete types, and we need to try to handle those.
38 template <typename From, typename To>
39 constexpr bool IsConvertibleTo()
40 {
41 // Conversion checks between pointer types are a common case which don't require full types
42 if constexpr (std::is_pointer_v<From> && std::is_pointer_v<To>)
43 {
44 using FromType = std::remove_pointer_t<From>;
45 using ToType = std::remove_pointer_t<To>;
46
47 // Losing cv qualifiers is always a convertibility failure
48 if constexpr ((std::is_const_v<FromType> && !std::is_const_v<ToType>) || (std::is_volatile_v<FromType> && !std::is_volatile_v<ToType>))
49 {
50 return false;
51 }
52 // Adding qualifiers to the same type is always possible
53 else if constexpr (std::is_same_v<const volatile FromType, const volatile ToType>)
54 {
55 return true;
56 }
57 // Conversion to void is always possible unless it's a function pointer
58 else if (std::is_void_v<ToType>)
59 {
60 return !std::is_function_v<FromType>;
61 }
62 // Fall back to the regular convertibility test, checking for complete types
63 else
64 {
65 return IsConvertibleToComplete<From, To>();
66 }
67 }
68 // Conversion checks between reference types are also a common case
69 else if constexpr (std::is_lvalue_reference_v<From> && std::is_lvalue_reference_v<To>)
70 {
71 using FromType = std::remove_reference_t<From>;
72 using ToType = std::remove_reference_t<To>;
73
74 // Losing cv qualifiers is always a convertibility failure
75 if constexpr ((std::is_const_v<FromType> && !std::is_const_v<ToType>) || (std::is_volatile_v<FromType> && !std::is_volatile_v<ToType>))
76 {
77 return false;
78 }
79 // Adding qualifiers to the same type is always possible
80 else if constexpr (std::is_same_v<const volatile FromType, const volatile ToType>)
81 {
82 return true;
83 }
84 // Fall back to the regular convertibility test, checking for complete types
85 else
86 {
87 return IsConvertibleToComplete<From, To>();
88 }
89 }
90 // Fall back to the regular convertibility test, checking for complete types
91 else
92 {
93 return IsConvertibleToComplete<From, To>();
94 }
95 }
96}
