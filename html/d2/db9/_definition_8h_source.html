36struct SQualifier
37{
38 enum class EType : uint8_t
39 {
40 Local, // i.e. `(local:)`, which is limited to a function's enclosing scopes.
41 NominalType, // This encompasses any nominal type or module.
42 LogicalScope, // This encompasses any parameterized type.
43 Unknown // Sentinel value for invalid/unresolved qualifiers.
44 };
45
47 union {
48 const CNominalType* _NominalType;
49 const CLogicalScope* _LogicalScope;
50 } U;
51
52 EType _Type;
53
54 SQualifier GetConstrainedQualifier() const;
55
56 const CNominalType* GetNominalType() const
57 {
58 if (_Type == EType::NominalType)
59 {
60 return U._NominalType;
61 }
62 return nullptr;
63 }
64 const CLogicalScope* GetLogicalScope() const {
65 if (_Type == EType::LogicalScope)
66 {
67 return U._LogicalScope;
68 }
69 return nullptr;
70 }
71 static SQualifier NominalType(const CNominalType* NominalType)
72 {
73 return { .U = {._NominalType = NominalType},._Type = EType::NominalType };
74 };
75 static SQualifier LogicalScope(const CLogicalScope* LogicalScope)
76 {
77 return { .U = {._LogicalScope = LogicalScope}, ._Type = EType::LogicalScope };
78 };
79 static SQualifier Local()
80 {
81 return { .U = {._NominalType = nullptr}, ._Type = EType::Local };
82 };
83 static SQualifier Unknown()
84 {
85 return { .U = {._NominalType = nullptr}, ._Type = EType::Unknown};
86 };
87
88 bool IsUnspecified() const
89 {
90 return _Type == EType::Unknown;
91 }
92
93 bool IsLocal() const
94 {
95 return _Type == EType::Local;
96 }
97
98 bool operator==(const SQualifier& Other) const
99 {
100 // Doesn't matter if we compare _NominalType or _LogicalType, it's only a pointer anyway.
101 return _Type == Other._Type && U._NominalType == Other.U._NominalType;
102 }
103
104 bool operator!=(const SQualifier& Other) const
105 {
106 return !(*this == Other);
107 }
108};

157 // If this definition is in the given visit set, return false.
158 // Otherwise, add this definition *and all definitions it overrides* to the visit set and
159 // return true.
160 bool TryMarkOverriddenAndConstrainedDefinitionsVisited(CVisitSet& VisitSet) const
161 {
162 if (!VisitSet.TryVisit(_PrototypeDefinition->_LastOverridingVisitKey))
163 {
164 return false;
165 }
166
167 // Mark this definition and the definitions overridden by it as visited.
168 for (const CDefinition* OverriddenDefinition = _PrototypeDefinition;
169 OverriddenDefinition;
170 OverriddenDefinition = OverriddenDefinition->GetOverriddenDefinition())
171 {
172 VisitSet.TryVisit(OverriddenDefinition->GetPrototypeDefinition()->_LastOverridingVisitKey);
173 }
174
175 // TODO? Should this one also use _PrototypeDefinition-> instead of this-> as above?
176 if (_ConstrainedDefinition)
177 {
178 VisitSet.TryVisit(_ConstrainedDefinition->GetPrototypeDefinition()->_LastOverridingVisitKey);
179 }
180
181 return true;
182 }

298 // These classes of functions are helpers to get to the scope of where
299 // the definition or definition's var access scope is defined. In Verse,
300 // access scopes are inherited when not explicitly defined. Note, this isn't
301 // the same as saying the same specifier is passed along the inheritance chain.
302 //
303 // For example:
304 // M1<public> := module {
305 // Base<public> := class {
306 // Field<internal>:int = 42
307 // }
308 // M2<public> := module:
309 // Child<public> := class(Base) {
310 // Field<override>:int = 50
311 // }
312 // }
313 // M2.Child's Field property is still accessible as internal to M1, not internal to M2.
