699inline const TCHAR* LexToString(BuildPatchServices::EFileSortOrder FileSortOrder)
700{
701#define CASE_ENUM_TO_STR(Value) case BuildPatchServices::EFileSortOrder::Value: return TEXT(#Value)
702 switch (FileSortOrder)
703 {
704 CASE_ENUM_TO_STR(AlphabeticalFilename);
705 CASE_ENUM_TO_STR(AlphabeticalTagThenFilename);
706 default: return TEXT("InvalidOrMax");
707 }
708#undef CASE_ENUM_TO_STR
709}

711inline void LexFromString(BuildPatchServices::EFileSortOrder& FileSortOrder, const TCHAR* Buffer)
712{
713#define RETURN_IF_EQUAL(Value) if (FCString::Stricmp(Buffer, TEXT(#Value)) == 0) { FileSortOrder = BuildPatchServices::EFileSortOrder::Value; return; }
714 const TCHAR* const Prefix = TEXT("EFileSortOrder::");
715 const SIZE_T PrefixLen = FCString::Strlen(Prefix);
716 if (FCString::Strnicmp(Buffer, Prefix, PrefixLen) == 0)
717 {
718 Buffer += PrefixLen;
719 }
720 RETURN_IF_EQUAL(AlphabeticalFilename);
721 RETURN_IF_EQUAL(AlphabeticalTagThenFilename);
722 // Did not match
723 FileSortOrder = BuildPatchServices::EFileSortOrder::InvalidOrMax;
724#undef RETURN_IF_EQUAL
725}
726inline FString LexToString(const TOptional<BuildPatchServices::FDiffAbortThreshold>& DiffAbortThreshold)
727{
728 constexpr auto* DiffAbortThresholdString = TEXT("Never Abort");
729 if (!DiffAbortThreshold.IsSet())
730 {
731 return DiffAbortThresholdString;
732 }
733
734 switch (DiffAbortThreshold.GetValue().Unit)
735 {
736 case BuildPatchServices::EDiffAbortThresholdUnits::Absolute:
737 return FString::FormatAsNumber(int32(DiffAbortThreshold.GetValue().Value / (1024 * 1024))) + TEXT(" MiB");
738 case BuildPatchServices::EDiffAbortThresholdUnits::Percentage:
739 return FString::FormatAsNumber(int32(DiffAbortThreshold.GetValue().Value)) + TEXT("%");
740 default:
741 checkf(false, TEXT("Unexpected EDiffAbortThresholdUnits value: %hhu"), EnumToUnderlyingType(DiffAbortThreshold.GetValue().Unit));
742 break;
743 }
744
745 return DiffAbortThresholdString;
746}
