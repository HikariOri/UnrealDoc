350 // Perceptual-based codec which supports all features across all platforms.

363 PlatformSpecific,
364
365 // The project defines the codec used for this asset.
366 ProjectDefined,

375 inline FName ToName(ESoundAssetCompressionType InDecoderType)
376 {
377 switch (InDecoderType)
378 {
379 case ESoundAssetCompressionType::RADAudio: return NAME_RADA;
380 case ESoundAssetCompressionType::BinkAudio: return NAME_BINKA;
381 case ESoundAssetCompressionType::ADPCM: return NAME_ADPCM;
382 case ESoundAssetCompressionType::PCM: return NAME_PCM;
383 case ESoundAssetCompressionType::Opus: return NAME_OPUS;
384 case ESoundAssetCompressionType::PlatformSpecific: return NAME_PLATFORM_SPECIFIC;
385 case ESoundAssetCompressionType::ProjectDefined: return NAME_PROJECT_DEFINED;
386 default:
387 ensure(false);
388 return TEXT("UNKNOWN");
389 }
390 }

397 case EDefaultAudioCompressionType::BinkAudio: return ESoundAssetCompressionType::BinkAudio;
398 case EDefaultAudioCompressionType::ADPCM: return ESoundAssetCompressionType::ADPCM;
399 case EDefaultAudioCompressionType::PCM: return ESoundAssetCompressionType::PCM;
400 case EDefaultAudioCompressionType::Opus: return ESoundAssetCompressionType::Opus;
401 case EDefaultAudioCompressionType::PlatformSpecific: return ESoundAssetCompressionType::PlatformSpecific;
