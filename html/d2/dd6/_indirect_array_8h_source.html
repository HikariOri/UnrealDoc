13 Indirect array.

477FArchive& operator<<(FArchive& Ar, TIndirectArray<T, Allocator>& A)
478{
479 A.CountBytes(Ar);
480 if (Ar.IsLoading())
481 {
482 // Load array.
483 int32 NewNum;
484 Ar << NewNum;
485 A.Empty(NewNum);
486 for (int32 Index = 0; Index < NewNum; Index++)
487 {
488 T* NewElement = new T;
489 Ar << *NewElement;
490 A.Add(NewElement);
491 }
492 }
493 else
494 {
495 // Save array.
496 int32 Num = A.Num();
497 Ar << Num;
498 for (int32 Index = 0; Index < Num; Index++)
499 {
500 Ar << A[Index];
501 }
502 }
