105struct FCallstackKey
106{
107 UObject* SerializedObject = nullptr;
108 FProperty* SerializedProperty = nullptr;
109 uint32 CppCallstackHash = MAX_uint32;
110 bool operator==(const FCallstackKey& Other) const
111 {
112 return Other.SerializedObject == SerializedObject &&
113 Other.SerializedProperty == SerializedProperty &&
114 Other.CppCallstackHash == CppCallstackHash;
115 }
116 friend uint32 GetTypeHash(const FCallstackKey& Key)
117 {
118 constexpr uint32 Prime = 101;
119 const uint32* WordPtr = reinterpret_cast<const uint32*>(&Key);
120 const uint32* WordPtrEnd = WordPtr + sizeof(Key) / sizeof(uint32);
121 uint32 Hash = 0;
122 while (WordPtr < WordPtrEnd)
123 {
124 Hash = Hash * Prime + *WordPtr;
125 ++WordPtr;
126 }
127 return Hash;
128 }
