173 void AddBitSet(const SMassBitSetDiff::FArguments& InArgs, TSharedRef<SWrapBox>& ButtonBox, const TBitSet& BaseBitSet, const TBitSet& BitSet
174 , const FSlateBrush* Brush, const EMassBitSetDiffPrune Prune, TSharedRef<FMassDebuggerModel> DebuggerModel)
175 {
176#if WITH_MASSENTITY_DEBUG
177 TBitSet CombinedBitSet = BaseBitSet + BitSet;
178 int32 SameCount = 0;
179
180 for (auto It = CombinedBitSet.GetIndexIterator(); It; ++It)
181 {
182 const bool bInBase = BaseBitSet.IsBitSet(*It);
183 const bool bInCurr = BitSet.IsBitSet(*It);
184
185 if (Prune == EMassBitSetDiffPrune::Same && bInBase && bInCurr)
186 {
187 SameCount++;
188 continue;
189 }
190
191 FString Name = CombinedBitSet.DebugGetStructTypeName(*It).ToString();
192 FLinearColor BorderColor = FLinearColor::White;
193 FLinearColor TextColor = InArgs._TextColor.Get();
194
195 FText Tooltip;
196
197 const FSlateBrush* Icon = nullptr;
198
199 if (bInBase && bInCurr)
200 {
201 // Same
202 BorderColor.A *= 0.6f;
203 TextColor.A *= 0.75f;
204 Tooltip = LOCTEXT("DiffExistsInBase", "Exists in base.");
205 }
206 else if (!bInBase && bInCurr)
207 {
208 // Added
209 TextColor = FMath::Lerp(TextColor, FLinearColor::White, 0.9f);
210 Icon = FAppStyle::Get().GetBrush("Icons.Plus");
211 Tooltip = LOCTEXT("DiffAddedToBase", "Added to base.");
212 }
213 else
214 {
215 // Removed
216 BorderColor.A *= 0.25f;
217 TextColor.A *= 0.5f;
218 Icon = FAppStyle::Get().GetBrush("Icons.Minus");
219 Tooltip = LOCTEXT("DiffRemovedFromBase", "Removed from base.");
220 }
221
222 ButtonBox->AddSlot()
223 .Padding(InArgs._SlotPadding)
224 [
225 SNew(SBorder)
226 .BorderImage(Brush)
227 .BorderBackgroundColor(BorderColor)
228 .ColorAndOpacity(TextColor)
229 .ToolTipText(Tooltip)
230 [
231 SNew(SHorizontalBox)
232 + SHorizontalBox::Slot()
233 .AutoWidth()
234 .Padding(3,0)
235 .VAlign(VAlign_Center)
236 [
237 SNew(SImage)
238 .Image(Icon)
239 .Visibility_Lambda([Icon] { return Icon != nullptr ? EVisibility::Visible : EVisibility::Collapsed; })
240 .DesiredSizeOverride(FVector2D(10.0f, 10.0f))
241 ]
242 + SHorizontalBox::Slot()
243 .AutoWidth()
244 .VAlign(VAlign_Center)
245 [
246 SNew(SButton)
247 .OnClicked_Lambda([Name, DebuggerModel]()
248 {
249 DebuggerModel->SelectFragment(FName(*Name));
250 return FReply::Handled();
251 })
252 [
253 SNew(STextBlock)
254 .Text(FText::FromString(Name))
255 ]
256 ]
257 ]
258 ];
259 }
260
261 if (SameCount > 0)
262 {
263 ButtonBox->AddSlot()
264 .Padding(InArgs._SlotPadding)
265 [
266 SNew(SHorizontalBox)
267 + SHorizontalBox::Slot()
268 .AutoWidth()
269 .VAlign(VAlign_Center)
270 [
271 SNew(STextBlock)
272 .Text(FText::Format(LOCTEXT("SameFragmentsFormat", "{0} same fragments"), FText::AsNumber(SameCount)))
273 ]
274 ];
275 }
276
277#endif // WITH_MASSENTITY_DEBUG
278 }
