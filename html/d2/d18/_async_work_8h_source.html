230 **/
231 void DestroyEvent()

233 FPlatformProcess::ReturnSynchEventToPool(DoneEvent);
234 DoneEvent = nullptr;

237 EQueuedWorkFlags GetQueuedWorkFlags() const final
238 {

242 /* Generic start function, not called directly
243 * @param bForceSynchronous if true, this job will be started synchronously, now, on this thread
244 **/
245 void Start(bool bForceSynchronous, FQueuedThreadPool* InQueuedPool, EQueuedWorkPriority InQueuedWorkPriority, EQueuedWorkFlags InQueuedWorkFlags, int64 InRequiredMemory, const TCHAR * InDebugName)
246 {
247 CaptureInheritedContext();
248
249 FScopeCycleCounter Scope(StatId, true);
250 DECLARE_SCOPE_CYCLE_COUNTER( TEXT( "FAsyncTask::Start" ), STAT_FAsyncTask_Start, STATGROUP_ThreadPoolAsyncTasks );
251 // default arg has InRequiredMemory == -1
252 RequiredMemory = InRequiredMemory;
253 DebugName = InDebugName;
254
255 FPlatformMisc::MemoryBarrier();
256 CheckIdle(); // can't start a job twice without it being completed first
257 WorkNotFinishedCounter.Increment();
258 QueuedPool = InQueuedPool;
259 Priority = InQueuedWorkPriority;
260 Flags = InQueuedWorkFlags;
261 if (bForceSynchronous)
262 {
263 QueuedPool = 0;
264 }
265 if (QueuedPool)
266 {
267 if (!DoneEvent)
268 {
269 DoneEvent = FPlatformProcess::GetSynchEventFromPool(true);
270 }
271 DoneEvent->Reset();
272 QueuedPool->AddQueuedWork(this, InQueuedWorkPriority);
273 }
274 else
275 {
276 // we aren't doing async stuff
277 DestroyEvent();
278 DoWork();
279 }
280 }
281
285 void DoWork()
286 {
287 {
288 UE::FInheritedContextScope InheritedContextScope = RestoreInheritedContext();
289 FScopeCycleCounter Scope(StatId, true);
290
291 DoTaskWork();
292 }
293 check(WorkNotFinishedCounter.GetValue() == 1);
294 WorkNotFinishedCounter.Decrement();
295 }
296
300 void FinishThreadedWork()
301 {
302 check(QueuedPool);
303 if (DoneEvent)
304 {
305 FScopeCycleCounter Scope(StatId, true);
306 DECLARE_SCOPE_CYCLE_COUNTER( TEXT( "FAsyncTask::FinishThreadedWork" ), STAT_FAsyncTask_FinishThreadedWork, STATGROUP_ThreadPoolAsyncTasks );
307 DoneEvent->Trigger();
308 }
309 }
310
314 void DoThreadedWork() final
315 {
316 DoWork();
317 FinishThreadedWork();
318 }
319
324 void Abandon() final
325 {
326 if (TryAbandonTask())
327 {
328 check(WorkNotFinishedCounter.GetValue() == 1);
329 WorkNotFinishedCounter.Decrement();
330 }
331 else
332 {
333 DoWork();
334 }
335 FinishThreadedWork();
336 }

342 void SyncCompletion(bool bIsLatencySensitive)
343 {
344 FPlatformMisc::MemoryBarrier();
345 if (QueuedPool)
346 {
347 TRACE_CPUPROFILER_EVENT_SCOPE(FAsyncTask::SyncCompletion);

349 DECLARE_SCOPE_CYCLE_COUNTER(TEXT("FAsyncTask::SyncCompletion"), STAT_FAsyncTask_SyncCompletion, STATGROUP_ThreadPoolAsyncTasks);
350
351 check(DoneEvent); // if it is not done yet, we must have an event
352 DoneEvent->Wait();
353 QueuedPool = 0;
354 }
355 CheckIdle();
356 }

433 void EnsureCompletion(bool bDoWorkOnThisThreadIfNotStarted = true, bool bIsLatencySensitive = false)
434 {
435 bool DoSyncCompletion = true;
436 if (bDoWorkOnThisThreadIfNotStarted)
437 {
438 if (QueuedPool)
439 {
440 if (QueuedPool->RetractQueuedWork(this))
441 {
442 // we got the job back, so do the work now and no need to synchronize
443 DoSyncCompletion = false;
444 DoWork();
445 FinishThreadedWork();
446 QueuedPool = 0;
447 }
448 }
449 else if (WorkNotFinishedCounter.GetValue()) // in the synchronous case, if we haven't done it yet, do it now
450 {
451 DoWork();
452 }
453 }
454 if (DoSyncCompletion)
455 {
456 SyncCompletion(bIsLatencySensitive);
457 }
458 CheckIdle(); // Must have had bDoWorkOnThisThreadIfNotStarted == false and needed it to be true for a synchronous job
459 }
460
465 bool Reschedule(FQueuedThreadPool* InQueuedPool = GThreadPool, EQueuedWorkPriority InQueuedWorkPriority = EQueuedWorkPriority::Normal)
466 {
467 if (QueuedPool)
468 {
469 if (QueuedPool->RetractQueuedWork(this))
470 {
471 QueuedPool = InQueuedPool;
472 Priority = InQueuedWorkPriority;
473 QueuedPool->AddQueuedWork(this, InQueuedWorkPriority);
474
475 return true;
476 }
477 }
478 return false;
479 }
480
486 bool Cancel()
487 {
488 if (QueuedPool)
489 {
490 if (QueuedPool->RetractQueuedWork(this))
491 {
492 check(WorkNotFinishedCounter.GetValue() == 1);
493 WorkNotFinishedCounter.Decrement();
494 FinishThreadedWork();
495 QueuedPool = 0;
496 return true;
497 }
498
499 if (!IsWorkDone())
500 {
501 MarkAsCanceled();
502 }
503 }
504 return false;
505 }
506
512 bool WaitCompletionWithTimeout(float TimeLimitSeconds)
513 {
514 if (TimeLimitSeconds <= 0.0f)
515 {
516 return IsDone();
517 }
518
519 FPlatformMisc::MemoryBarrier();
520 if (QueuedPool)
521 {
522 FScopeCycleCounter Scope(StatId);
523 DECLARE_SCOPE_CYCLE_COUNTER(TEXT("FAsyncTask::SyncCompletion"), STAT_FAsyncTask_SyncCompletion, STATGROUP_ThreadPoolAsyncTasks);
524
525 uint32 Ms = uint32(TimeLimitSeconds * 1000.0f) + 1;
526 check(Ms);
527
528 check(DoneEvent); // if it is not done yet, we must have an event
529 if (DoneEvent->Wait(Ms))
530 {
531 QueuedPool = 0;
532 CheckIdle();
533 return true;
534 }
535 return false;
536 }
537 CheckIdle();
538 return true;
539 }
540
544 bool IsDone()
545 {
546 if (!IsWorkDone())
547 {
548 return false;
549 }
550 SyncCompletion(/*bIsLatencySensitive = */false);
551 return true;
552 }
553
558 bool IsWorkDone() const
559 {
560 if (WorkNotFinishedCounter.GetValue())
561 {
562 return false;
563 }
564 return true;
565 }
566
570 bool IsIdle() const
571 {
572 return WorkNotFinishedCounter.GetValue() == 0 && QueuedPool == 0;
573 }
574
575 bool SetPriority(EQueuedWorkPriority QueuedWorkPriority)
576 {
577 return Reschedule(QueuedPool, QueuedWorkPriority);
578 }
579
580 EQueuedWorkPriority GetPriority() const
581 {
582 return Priority;
583 }
