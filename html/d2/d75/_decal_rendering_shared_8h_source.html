101 // Custom Render Passes have their own array of FViewInfo, outside the scene renderer's Views array. When accessing decal visibility
102 // for a specific view array, an offset must be added to account for where the relevant data is located in the visibility task data.
103 // To enforce this, the view offset is passed separately to the accessor functions below. INDEX_NONE is returned if decals are disabled
104 // for the given array of Views, where the task data constructor won't have added them to the list. It's up to the caller to check for
105 // INDEX_NONE and skip decal rendering (the accessors will assert if INDEX_NONE is passed in for ViewOffset).
106 int32 GetViewOffset(TConstArrayView<FViewInfo> Views) const
107 {
108 for (int32 ViewIndex = 0; ViewIndex < ViewPackets.Num(); ViewIndex++)
109 {
110 if (&ViewPackets[ViewIndex].View == Views.GetData())
111 {
112 return ViewIndex;
113 }
114 }
115 return INDEX_NONE;
116 }
117
118 TConstArrayView<const FVisibleDecal*> FinishRelevantDecals(int32 ViewOffset, int32 ViewIndex, EDecalRenderStage Stage)
119 {
120 check(ViewOffset != INDEX_NONE);
121 return ViewPackets[ViewOffset + ViewIndex].FinishRelevantDecals(Stage);
122 }
123
124 bool HasStage(int32 ViewOffset, int32 ViewIndex, EDecalRenderStage Stage) const
125 {
126 check(ViewOffset != INDEX_NONE);
127 return ViewPackets[ViewOffset + ViewIndex].HasStage(Stage);
128 }
129
130 void Finish(int32 ViewOffset, int32 ViewIndex)
131 {
132 check(ViewOffset != INDEX_NONE);
133 ViewPackets[ViewOffset + ViewIndex].Finish();
134 }
135
136private:
137 FDecalVisibilityTaskData(const FScene& Scene, TConstArrayView<FViewInfo*> Views);
138
139 TArray<FDecalVisibilityViewPacket, FRDGArrayAllocator> ViewPackets;

141 friend class FDecalVisibilityViewPacket;
142 RDG_FRIEND_ALLOCATOR_FRIEND(FDecalVisibilityTaskData);
