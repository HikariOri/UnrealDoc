180// IoChunk per shader group
181struct FShaderGroupIoChunk
182{
183 FIoChunkId ChunkId;
184 FIoBuffer CodeIoBuffer;
185 uint32 LoadOrderFactor = MAX_uint32; // the smaller the order, the more likely this will be loaded
186
187 static bool Sort(const FShaderGroupIoChunk& A, const FShaderGroupIoChunk& B)
188 {
189 if (A.LoadOrderFactor == B.LoadOrderFactor)
190 {
191 // Shader chunk IDs are the hash of the shader so this is consistent across builds.
192 return FMemory::Memcmp(A.ChunkId.GetData(), B.ChunkId.GetData(), A.ChunkId.GetSize()) < 0;
193 }
194 return A.LoadOrderFactor < B.LoadOrderFactor;
195 }
196};
