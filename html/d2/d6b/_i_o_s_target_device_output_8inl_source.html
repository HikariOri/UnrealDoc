51inline uint32 FIOSDeviceOutputReaderRunnable::Run(void)
52{
53 FString SyslogOutput;
54
55 while (StopTaskCounter.GetValue() == 0 && SyslogProcHandle.IsValid())
56 {
57 if (!FPlatformProcess::IsProcRunning(SyslogProcHandle))
58 {
59 // When user plugs out USB cable idevicesyslog process stops
60 // Keep trying to restore idevicesyslog connection until code that uses this object will not kill us
61 Output->Serialize(TEXT("Trying to restore connection to device..."), ELogVerbosity::Log, NAME_None);
62 FPlatformProcess::CloseProc(SyslogProcHandle);
63 if (StartSyslogProcess())
64 {
65 FPlatformProcess::Sleep(1.0f);
66 }
67 else
68 {
69 Output->Serialize(TEXT("Failed to start idevicesyslog proccess"), ELogVerbosity::Log, NAME_None);
70 Stop();
71 }
72 }
73 else
74 {
75 SyslogOutput.Append(FPlatformProcess::ReadPipe(SyslogReadPipe));
76
77 if (SyslogOutput.Len() > 0)
78 {
79 TArray<FString> OutputLines;
80 SyslogOutput.ParseIntoArray(OutputLines, TEXT("\n"), false);
81
82 if (!SyslogOutput.EndsWith(TEXT("\n")))
83 {
84 // partial line at the end, do not serialize it until we receive remainder
85 SyslogOutput = OutputLines.Last();
86 OutputLines.RemoveAt(OutputLines.Num() - 1);
87 }
88 else
89 {
90 SyslogOutput.Reset();
91 }
92
93 for (int32 i = 0; i < OutputLines.Num(); ++i)
94 {
95 Output->Serialize(*OutputLines[i], ELogVerbosity::Log, NAME_None);
96 }
97 }
98
99 FPlatformProcess::Sleep(0.1f);
100 }
101 }
102
103 return 0;
104}
