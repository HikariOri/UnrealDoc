41 static TArray<FBasicPropertyInfo> GetChangedPropertiesInfoUsingReflectionInterface(const UE::UndoHistory::IReflectionDataProvider& ReflectionData, const FSoftClassPath& InObjectClass, const TArray<FName>& InChangedProperties)
42 {
43 using namespace UE::UndoHistory;
44 TArray<FBasicPropertyInfo> Properties;
45
46 for (const FName& PropertyName : InChangedProperties)
47 {
48 if (TOptional<FPropertyReflectionData> PropertyData = ReflectionData.GetPropertyReflectionData(InObjectClass, PropertyName))
49 {
50 FString ClassName;
51 if (PropertyData->PropertyType == UE::UndoHistory::EPropertyType::ObjectProperty
52 || PropertyData->PropertyType == UE::UndoHistory::EPropertyType::StructProperty
53 || PropertyData->PropertyType == UE::UndoHistory::EPropertyType::EnumProperty)
54 {
55 ClassName = PropertyData->CppMacroType;
56 }
57 else if (PropertyData->PropertyType == UE::UndoHistory::EPropertyType::ArrayProperty)
58 {
59 ClassName = PropertyData->CppMacroType;
60 ClassName = FString::Printf(TEXT("TArray<%s>"), *ClassName);
61 }
62 else
63 {
64 ClassName = PropertyData->TypeName;
65 ClassName.RemoveFromEnd("Property");
66 }
67
68 Properties.Emplace(PropertyName.ToString(), MoveTemp(ClassName), PropertyData->PropertyFlags);
69 }
70 else
71 {
72 Properties.Emplace(PropertyName.ToString());
73 }
74 }
75
76 return Properties;
77 }

81 TArray<FBasicPropertyInfo> Properties;
82 for (const FName& PropertyName : InChangedProperties)

84 Properties.Emplace(PropertyName.ToString());
85 }
