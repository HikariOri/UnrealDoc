53 template<typename MeshType>
54 int32 ComputeBoundaryLoops(const MeshType& Mesh, TArray<int32>& OutPackedBoundaryVertLoops, TArray<int32>* OptionalOutPackedBoundaryTriLoops,
55 TFunctionRef<int32(int32)> VertexWeldFn = [](int32 VID) {return VID;}, FBoundaryFilterOptions FilterOptions = FBoundaryFilterOptions())
56 {
57 OutPackedBoundaryVertLoops.Reset();
58 if (OptionalOutPackedBoundaryTriLoops)
59 {
60 OptionalOutPackedBoundaryTriLoops->Reset();
61 }
62
63 // 1. Find the edges w/ no paired reverse edge (i.e. the potential hole boundary edges)
64
65 TMultiMap<int32, FIndex2i> OpenEdges; // VID -> (VID, TID)
66 // remove an edge from OpenEdges, assuming the vertices are already passed through VertexWeldFn
67 auto RemoveEdge = [&OpenEdges](int32 A, int32 B)
68 {
69 int32 NumRemovedPairs = 0;
70 for (TMultiMap<int32, FIndex2i>::TKeyIterator It = OpenEdges.CreateKeyIterator(A); It; ++It)
71 {
72 if (It.Value().A == B)
73 {
74 It.RemoveCurrent();
75 return true;
76 }
77 }
78 return false;
79 };
80 auto AddEdge = [&OpenEdges, &VertexWeldFn, &RemoveEdge](int32 A, int32 B, int32 TID)
81 {
82 A = VertexWeldFn(A);
83 B = VertexWeldFn(B);
84 if (A == B) // weld collapsed edge, skip it
85 {
86 return;
87 }
88 if (!RemoveEdge(B, A))
89 {
90 OpenEdges.Add(A, FIndex2i(B, TID));
91 }
92 };
93 // Consider all tri edges as possible boundary edges
94 // Note: This is necessary for meshes where we only have a vertex and triangle buffer --
95 // for dynamic mesh can filter to boundary edge and add a potentially much smaller subset of candidates to OpenEdges
96 for (int32 TID = 0; TID < Mesh.MaxTriangleID(); ++TID)
97 {
98 if (!Mesh.IsTriangle(TID))
99 {
100 continue;
101 }
102 FIndex3i Tri = Mesh.GetTriangle(TID);
103 for (int32 SubIdx = 0, Prev = 2; SubIdx < 3; Prev = SubIdx++)
104 {
105 AddEdge(Tri[Prev], Tri[SubIdx], TID);
106 }
107 }
108
109 // 2. Walk the open boundary edges to find the loops that define holes
110
111 const int32 MinEdgesPerLoop = FMath::Max(3, FilterOptions.MinEdges);
112 int32 NumHoles = 0;
113 while (true)
114 {
115 const auto& EdgeItr = OpenEdges.CreateConstIterator();
116 if (!EdgeItr)
117 {
118 break;
119 }
120
121 // Set up to add a new loop starting at this index
122 int32 BdryStartIdx = OutPackedBoundaryVertLoops.Add(INDEX_NONE);
123 // Helper to clear the added data, e.g. if we don't find a valid loop
124 auto ClearLoop = [BdryStartIdx, &OutPackedBoundaryVertLoops, &OptionalOutPackedBoundaryTriLoops]()
125 {
126 OutPackedBoundaryVertLoops.SetNum(BdryStartIdx);
127 if (OptionalOutPackedBoundaryTriLoops)
128 {
129 OptionalOutPackedBoundaryTriLoops->SetNum(BdryStartIdx);
130 }
131 };
132
133 int32 Start = EdgeItr.Key();
134 FIndex2i WalkPair = EdgeItr.Value();
135 int32 Walk = WalkPair.A;
136
137 OutPackedBoundaryVertLoops.Add(Start);
138 if (OptionalOutPackedBoundaryTriLoops)
139 {
140 OptionalOutPackedBoundaryTriLoops->Add(INDEX_NONE);
141 OptionalOutPackedBoundaryTriLoops->Add(WalkPair.B);
142 checkSlow(OptionalOutPackedBoundaryTriLoops->Num() == OutPackedBoundaryVertLoops.Num());
143 }
144 RemoveEdge(Start, Walk);
145 while (Walk != Start) // this will either loop back to start or dead end (note every non-dead-end iter removes an edge from OpenEdges)
146 {
147 OutPackedBoundaryVertLoops.Add(Walk);
148 FIndex2i* Next = OpenEdges.Find(Walk);
149 if (!Next)
150 {
151 // dead-ended chain of edges, discard the vertices
152 ClearLoop();
153 break;
154 }
155 int32 NextVal = Next->A;
156 if (OptionalOutPackedBoundaryTriLoops)
157 {
158 OptionalOutPackedBoundaryTriLoops->Add(Next->B);
159 }
160
161 RemoveEdge(Walk, NextVal);
162 Walk = NextVal;
163 }
164 int32 Count = OutPackedBoundaryVertLoops.Num() - BdryStartIdx - 1;
165 if (Count < MinEdgesPerLoop) // failed to add a boundary, try again next loop
166 {
167 ClearLoop();
168 continue;
169 }
170 if (FilterOptions.MinPerimeter > 0)
171 {
172 double Perim = 0;
173 FVector3d PrevPos = Mesh.GetVertex(OutPackedBoundaryVertLoops.Last());
174 for (int32 Idx = BdryStartIdx + 1; Perim < FilterOptions.MinPerimeter && Idx < OutPackedBoundaryVertLoops.Num(); ++Idx)
175 {
176 FVector3d CurPos = Mesh.GetVertex(OutPackedBoundaryVertLoops[Idx]);
177 Perim += FVector3d::Dist(PrevPos, CurPos);
178 PrevPos = CurPos;
179 }
180 if (Perim < FilterOptions.MinPerimeter)
181 {
182 ClearLoop();
183 continue;
184 }
185 }
186
187 OutPackedBoundaryVertLoops[BdryStartIdx] = Count; // record length of added chain
188 if (OptionalOutPackedBoundaryTriLoops)
189 {
190 (*OptionalOutPackedBoundaryTriLoops)[BdryStartIdx] = Count;
191 }
192 NumHoles++;
193 }
194 if (OptionalOutPackedBoundaryTriLoops)
195 {
196 check(OutPackedBoundaryVertLoops.Num() == OptionalOutPackedBoundaryTriLoops->Num());
197 }
198
199 return NumHoles;
200 }

236 template<typename MeshType>
237 void ExtrudeBoundaryLoopEdges(MeshType& Mesh, TConstArrayView<int32> PackedBoundaryVertLoops, TConstArrayView<int32> PackedBoundaryTriLoops,
238 TFunctionRef<FBoundaryOffsetVert(const FBoundaryVertexContext&)> GetOrCreateOffsetVertFn,
239 TFunctionRef<void(const FIndex3i& Tri, int32 SourceTID)> AddTriFn)
240 {
241 if (!ensure(PackedBoundaryVertLoops.Num() == PackedBoundaryTriLoops.Num()))
242 {
243 return;
244 }
245 for (int32 LoopCountIdx = 0; LoopCountIdx < PackedBoundaryVertLoops.Num(); LoopCountIdx += PackedBoundaryVertLoops[LoopCountIdx] + 1)
246 {
247 const int32 LoopCount = PackedBoundaryVertLoops[LoopCountIdx];
248 if (!ensure(PackedBoundaryTriLoops[LoopCountIdx] == LoopCount))
249 {
250 return;
251 }
252 const int32 LoopStartIdx = LoopCountIdx + 1;
253 const int32 LoopLastIdx = LoopCountIdx + LoopCount;
254
255 // For each boundary edge (Prev -> Cur), emit the two skirt-quad triangles. PackedBoundaryTriLoops[PrevIdx]
256 // is the tri containing the edge from PrevIdx's vertex to CurIdx's vertex, per ComputeBoundaryLoops.
257 for (int32 CurIdx = LoopStartIdx; CurIdx <= LoopLastIdx; ++CurIdx)
258 {
259 const int32 PrevIdx = (CurIdx == LoopStartIdx) ? LoopLastIdx : (CurIdx - 1);
260 const int32 PrevPrevIdx = (PrevIdx == LoopStartIdx) ? LoopLastIdx : (PrevIdx - 1);
261 const int32 NextIdx = (CurIdx == LoopLastIdx) ? LoopStartIdx : (CurIdx + 1);
262
263 const int32 WeldedCur = PackedBoundaryVertLoops[CurIdx];
264 const int32 WeldedPrev = PackedBoundaryVertLoops[PrevIdx];
265 const int32 WeldedPrevPrev = PackedBoundaryVertLoops[PrevPrevIdx];
266 const int32 WeldedNext = PackedBoundaryVertLoops[NextIdx];
267
268 const int32 EdgeTID = PackedBoundaryTriLoops[PrevIdx];
269
270 const FBoundaryOffsetVert CurOff = GetOrCreateOffsetVertFn({ WeldedCur, WeldedPrev, WeldedNext, EdgeTID });
271 const FBoundaryOffsetVert PrevOff = GetOrCreateOffsetVertFn({ WeldedPrev, WeldedPrevPrev, WeldedCur, EdgeTID });
272
273 AddTriFn(FIndex3i(CurOff.BaseVID, PrevOff.BaseVID, CurOff.SkirtVID), EdgeTID);
274 AddTriFn(FIndex3i(CurOff.SkirtVID, PrevOff.BaseVID, PrevOff.SkirtVID), EdgeTID);
275 }
276 }
