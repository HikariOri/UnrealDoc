32 /*
33 * A Content Source is a construct that can customize the look and behavior of the content browser through TEDS (Editor Data Storage) intended
34 * to replace the legacy asset specific content browser.
35 * Currently supports displaying an STedsTableViewer driven by a query specified by the content source in place of the legacy content browser
36 * layout and widgets
37 * Content Sources can be registered with IContentBrowserSingleton which will cause them to be displayed in a vertical toolbar next to the
38 * Content Browser allowing a user to select and activate them.
39 *
40 * NOTE: This API is experimental and subject to change
41 */
42 class IContentSource
43 {
44 public:
45 virtual ~IContentSource() = default;
46
47 // Get the internal name of the content source
48 virtual FName GetName() = 0;
49
50 // Get the user facing name of the content source
51 virtual FText GetDisplayName() = 0;
52
53 // Get an icon representing the content source
54 virtual FSlateIcon GetIcon() = 0;
55
56 // Get the init params used by the TEDS Table Viewer when this content source is active
57 virtual void GetAssetViewInitParams(FTableViewerInitParams& OutInitParams) = 0;
58
59 // Called when this content source is switched in
60 virtual void OnContentSourceEnabled() {}
61
62 // Called when this content source is swapped out
63 virtual void OnContentSourceDisabled() {}
64 };
