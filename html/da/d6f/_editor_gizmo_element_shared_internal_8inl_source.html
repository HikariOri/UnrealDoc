59 void GetSignedAxisBasis(EAxisList::Type InAxis, Math::TVector<RealType>& OutForward, Math::TVector<RealType>& OutUp, Math::TVector<RealType>& OutSide, const EAxisList::Type InAxisCoordinateSystem)
60 {
61 // Only Y is affected by the coordinate system
62 const FVector ModifiedY = FVector::YAxisVector * (InAxisCoordinateSystem == EAxisList::LeftUpForward ? -1 : 1);
63
64 switch (InAxis)
65 {
66 case EAxisList::X:
67 case EAxisList::Forward:
68 OutForward = Math::TVector<RealType>::XAxisVector;
69 OutUp = Math::TVector<RealType>::ZAxisVector;
70 OutSide = ModifiedY;
71 break;
72
73 case EAxisList::Y:
74 case EAxisList::Left:
75 OutForward = ModifiedY;
76 OutUp = Math::TVector<RealType>::ZAxisVector;
77 OutSide = Math::TVector<RealType>::XAxisVector;
78 break;
79
80 case EAxisList::Z:
81 case EAxisList::Up:
82 OutForward = Math::TVector<RealType>::ZAxisVector;
83 OutUp = Math::TVector<RealType>::XAxisVector;
84 OutSide = ModifiedY;
85 break;
86
87 // Screen, Uniform etc all use the same forward/up/side vectors
88 case EAxisList::Screen:
89 case EAxisList::All:
90 case EAxisList::LeftUpForward:
91 default:
92 OutForward = Math::TVector<RealType>::XAxisVector;
93 OutUp = Math::TVector<RealType>::ZAxisVector;
94 OutSide = ModifiedY;
95 break;
96 }
97 }

107 template <typename RealType UE_REQUIRES(std::is_floating_point_v<RealType>)>
108 void GetAxisBasis(EAxisList::Type InAxis, Math::TVector<RealType>& OutForward, Math::TVector<RealType>& OutUp, Math::TVector<RealType>& OutSide, const EAxisList::Type InAxisCoordinateSystem)
109 {
110 switch (InAxis)
111 {
112 case EAxisList::X:
113 case EAxisList::Forward:
114 OutForward = Math::TVector<RealType>::XAxisVector;
115 OutUp = Math::TVector<RealType>::ZAxisVector;
116 OutSide = Math::TVector<RealType>::YAxisVector;
117 break;
118
119 case EAxisList::Y:
120 case EAxisList::Left:
121 OutForward = Math::TVector<RealType>::YAxisVector;
122 OutUp = Math::TVector<RealType>::ZAxisVector;
123 OutSide = -Math::TVector<RealType>::XAxisVector;
124 if (InAxisCoordinateSystem == EAxisList::LeftUpForward)
125 {
126 // Both must be inverted to create a valid FFrame
127 OutForward *= -1;
128 OutSide *= -1;
129 }
130 break;
131
132 case EAxisList::Z:
133 case EAxisList::Up:
134 OutForward = Math::TVector<RealType>::ZAxisVector;
135 OutUp = -Math::TVector<RealType>::XAxisVector;
136 OutSide = Math::TVector<RealType>::YAxisVector;
137 break;
138
139 // Screen, Uniform etc all use the same forward/up/side vectors
140 case EAxisList::Screen:
141 case EAxisList::All:
142 case EAxisList::LeftUpForward:
143 default:
144 OutForward = Math::TVector<RealType>::XAxisVector;
145 OutUp = Math::TVector<RealType>::ZAxisVector;
146 OutSide = Math::TVector<RealType>::YAxisVector;
147 break;
148 }
149 }
