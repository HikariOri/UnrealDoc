54inline FStringView GetMimeType(EContentType Type)
55{
56 switch (Type)
57 {
58 case EContentType::Binary:
59 return TEXTVIEW("application/octet-stream");
60 case EContentType::Text:
61 return TEXTVIEW("text/plain");
62 case EContentType::JSON:
63 return TEXTVIEW("application/json");
64 case EContentType::CbObject:
65 return TEXTVIEW("application/x-ue-cb");
66 case EContentType::CbPackage:
67 return TEXTVIEW("application/x-ue-cbpkg");
68 case EContentType::YAML:
69 return TEXTVIEW("text/yaml");
70 case EContentType::CbPackageOffer:
71 return TEXTVIEW("application/x-ue-offer");
72 case EContentType::CompressedBinary:
73 return TEXTVIEW("application/x-ue-comp");
74 default:
75 return TEXTVIEW("unknown");
76 }
