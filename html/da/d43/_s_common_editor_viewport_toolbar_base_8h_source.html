60 UE_DEPRECATED(5.5, "Use the version of the show flags builder in FShowFlagMenuCommands::BuildShowFlagsMenu which takes a UToolMenu instead")
61 static inline void FillShowMenu(class FMenuBuilder& MenuBuilder, TArray<FShowMenuCommand> MenuCommands, int32 EntryOffset)
62 {
63 // Generate entries for the standard show flags
64 // Assumption: the first 'n' entries types like 'Show All' and 'Hide All' buttons, so insert a separator after them
65 for (int32 EntryIndex = 0; EntryIndex < MenuCommands.Num(); ++EntryIndex)
66 {
67 MenuBuilder.AddMenuEntry(MenuCommands[EntryIndex].ShowMenuItem, NAME_None, MenuCommands[EntryIndex].LabelOverride);
68 if (EntryIndex == EntryOffset - 1)
69 {
70 MenuBuilder.AddMenuSeparator();
71 }
72 }
73 }
