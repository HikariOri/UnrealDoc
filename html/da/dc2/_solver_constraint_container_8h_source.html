163 void UpdateSolverPartitionManager()
164 {
165 if (CVars::Chaos_SolverPartitionManager_Partitioning == 1 && SupportsPartitioning())
166 {
167 // Create a solver partition manager to apply greedy graph coloring
168 if (!SolverPartitionManager.IsValid() || SolverPartitionManager->GetType(Private::FSolverPartitionManager::FPasskey()) != Private::ESolverPartitionManagerType::GreedyConstraintColoring)
169 {
170 SolverPartitionManager = TUniquePtr<Private::FGreedyConstraintColoring>(new Private::FGreedyConstraintColoring(*this, Private::FSolverPartitionManager::FPasskey()));
171 }
172 }
173 else
174 {
175 // Clear the solver partition manager (no partitioning)
176 SolverPartitionManager.Reset();
177 }
178 }
