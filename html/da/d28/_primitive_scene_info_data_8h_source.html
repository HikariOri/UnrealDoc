14// An attachment counter is a thread-safe counter that is only ever
15// incremented on the game thread, and decremented on the rendering
16// thread.
17//
18// It also supports transactions, and will undo any increments on the
19// game thread if a transaction aborts.
20struct FAttachmentCounter final
21{
22 UE_NONCOPYABLE(FAttachmentCounter);
23
24 AUTORTFM_OPEN FAttachmentCounter() = default;
25
26#if UE_AUTORTFM
27 void Increment()
28 {
29 // TODO: should we `check(IsInGameThread());`?
30
31 UE_AUTORTFM_OPEN
32 {
33 Payload->Payload++;
34 };
35
36 UE_AUTORTFM_ONABORT(this)
37 {
38 Payload->Payload--;
39 };
40 }
41
42 void Decrement()
43 {
44 // TODO: should we `check(IsInRenderingThread());`?
45 AutoRTFM::UnreachableIfClosed("Should only be decremented on the render thread in the open");
46 Payload->Payload--;
47 }
48
49 int32 GetValue() const
50 {
51 int32 Result = 0;
52
53 UE_AUTORTFM_OPEN
54 {
55 Result = Payload->Payload;
56 };
57
58 return Result;
59 }
60private:
61
62 // Under AutoRTFM because we are creating and *sometimes* destroying
63 // `FAttachmentCounter`'s in the same transaction, we need it to be
64 // heap allocated, so that the memory doesn't just instantly go away
65 // and our on-abort handlers would then use-after-free.
66 struct FPayload final : public UE::Private::FQueryableRefCountedObject
67 {
68 std::atomic<int32> Payload = 0;
69 };
70
71 TRefCountPtr<FPayload> Payload = new FPayload();
72
73#else
74
75 void Increment()
76 {
77 // TODO: should we `check(IsInGameThread());`?
78 Payload++;
79 }
80
81 void Decrement()
82 {
83 // TODO: should we `check(IsInRenderingThread());`?
84 Payload--;
85 }
86
87 int32 GetValue() const
88 {
89 return Payload;
90 }
91private:
92 std::atomic<int32> Payload = 0;
93#endif
94};
