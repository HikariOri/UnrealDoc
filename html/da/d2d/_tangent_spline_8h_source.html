278template <typename AttrType>
279int32 FTangentSpline::AddAttributeValue(FName Name, const AttrType& Value, float Parameter)
280{
281 using AttrSplineType = UE::Geometry::Spline::TTangentBezierSpline<AttrType>;
282 using ControlPointType = UE::Geometry::Spline::TTangentBezierControlPoint<AttrType>;
283 using UE::Geometry::FInterval1f;
284
285 if (AttrSplineType* Channel = GetTypedAttributeChannel<AttrSplineType>(Name))
286 {
287 ON_SCOPE_EXIT
288 {
289 Channel->UpdateTangents();
290 };
291 ControlPointType ControlPoint = ControlPointType(Value);
292 int32 NumPoints = Channel->GetNumPoints();
293
294 float ParentSpaceParameter = Parameter;
295 FInterval1f MappedChildSpace = GetMappedChildSpace(Name);
296
297 // Special case for empty spline
298 if (NumPoints == 0)
299 {
300 Channel->AppendPoint(ControlPoint, 0.f);
301 SetAttributeChannelRange(Name, FInterval1f(ParentSpaceParameter, ParentSpaceParameter), EMappingRangeSpace::Parent);
302 return 0;
303 }
304
305 if (NumPoints == 1)
306 {
307 if (ParentSpaceParameter > MappedChildSpace.Min)
308 {
309 Channel->AppendPoint(ControlPoint, UE::Geometry::Spline::EParameterizationPolicy::Uniform);
310 SetAttributeChannelRange(Name, FInterval1f(MappedChildSpace.Min, ParentSpaceParameter), EMappingRangeSpace::Parent);
311 return 1;
312 }
313 else
314 {
315 Channel->PrependPoint(ControlPoint, UE::Geometry::Spline::EParameterizationPolicy::Uniform);
316 SetAttributeChannelRange(Name, FInterval1f(ParentSpaceParameter, MappedChildSpace.Min), EMappingRangeSpace::Parent);
317 return 0;
318 }
319 }
320
321 // append case
322 if (ParentSpaceParameter > MappedChildSpace.Max)
323 {
324 // It is important to compute ChildSpaceParameter before the AppendPoint call, otherwise it will not be correct.
325 float ChildSpaceParameter = MapParameterToChildSpace(Name, ParentSpaceParameter);
326 Channel->AppendPoint(ControlPoint, ChildSpaceParameter);
327
328 // By growing the child space and the mapped parent range proportionally, we keep the internal points stable in parent space.
329 SetAttributeChannelRange(Name, FInterval1f(MappedChildSpace.Min, ParentSpaceParameter), EMappingRangeSpace::Parent);
330
331 return Channel->GetNumPoints() - 1;
332 }
333
334 // prepend case
335 if (ParentSpaceParameter < MappedChildSpace.Min)
336 {
337 // It is important to compute ChildSpaceParameter before the AppendPoint call, otherwise it will not be correct.
338 float ChildSpaceParameter = MapParameterToChildSpace(Name, ParentSpaceParameter);
339 Channel->PrependPoint(ControlPoint, ChildSpaceParameter);
340
341 // By growing the child space and the mapped parent range proportionally, we keep the internal points stable in parent space.
342 SetAttributeChannelRange(Name, FInterval1f(ParentSpaceParameter, MappedChildSpace.Max), EMappingRangeSpace::Parent);
343
344 return 0;
345 }
346
347 float ChildSpaceParameter = MapParameterToChildSpace(Name, ParentSpaceParameter);
348 int NewPointIndex = Channel->InsertPoint(ControlPoint, ChildSpaceParameter);
349 return NewPointIndex;
350 }
351
352 return INDEX_NONE;
353}

421template <typename AttrType>
422int32 FTangentSpline::SetAttributeParameter(const FName& Name, int32 Index, float Parameter)
423{
424 using AttrSplineType = UE::Geometry::Spline::TTangentBezierSpline<AttrType>;
425 using UE::Geometry::FInterval1f;
426
427 if (AttrSplineType* Channel = GetTypedAttributeChannel<AttrSplineType>(Name))
428 {
429 ON_SCOPE_EXIT
430 {
431 Channel->UpdateTangents();
432 };
433 float ParentSpaceParameter = Parameter;
434
435 // Prevent collapse of the channel space and keep it starting at 0.
436 auto SanitizeChannelSpace = [Channel]()
437 {
438 const FInterval1f ChannelSpace = Channel->GetParameterSpace();
439
440 if (!ensureAlwaysMsgf(ChannelSpace.Min < ChannelSpace.Max, TEXT("Degenerate attribute channel parameter space detected, cannot sanitize.")))
441 {
442 return;
443 }
444
445 const float KnotOffset = -1.f * ChannelSpace.Min;
446 const float KnotScaleFactor = 1.f / (ChannelSpace.Max - ChannelSpace.Min);
447 TArray<UE::Geometry::Spline::FKnot> Knots = Channel->GetKnotVector();
448 for (UE::Geometry::Spline::FKnot& Knot : Knots)
449 {
450 Knot.Value += KnotOffset;
451 Knot.Value *= KnotScaleFactor;
452 }
453 Channel->SetKnotVector(Knots);
454 };
455
456 // Helper to prevent code duplication.
457 // Uses current value of Index and ParentSpaceParameter at the time of calling to update the attribute parameter.
458 auto SetParentSpaceParameter = [this, &ParentSpaceParameter, Name, Channel, &Index]() -> int32
459 {
460 const float CurrentChildSpaceParameter = MapParameterToChildSpace(Name, GetAttributeParameter<AttrType>(Name, Index));
461 const float DesiredChildSpaceParameter = MapParameterToChildSpace(Name, ParentSpaceParameter);
462
463 constexpr float MinStep = 2.f * UE_KINDA_SMALL_NUMBER;
464 float NewChildSpaceParameter = DesiredChildSpaceParameter > CurrentChildSpaceParameter
465 ? FMath::Max(DesiredChildSpaceParameter, CurrentChildSpaceParameter + MinStep)
466 : FMath::Min(DesiredChildSpaceParameter, CurrentChildSpaceParameter - MinStep);
467
468 return Channel->SetParameter(Index, NewChildSpaceParameter);
469 };
470
479
480 if (Channel->GetNumPoints() == 1) // Case 1: Moving the only attribute
481 {
482 SetAttributeChannelRange(Name, FInterval1f(ParentSpaceParameter, ParentSpaceParameter), EMappingRangeSpace::Parent);
483 return Index;
484 }
485 else if (Channel->GetNumPoints() == 2) // Case 2: Moving an end point while no internal points exist.
486 {
487 FInterval1f MappedRange = GetMappedChildSpace(Name);
488 float& RangeBound = Index == 0 ? MappedRange.Min : MappedRange.Max;
489 RangeBound = ParentSpaceParameter;
490
491 if (MappedRange.Min > MappedRange.Max)
492 {
493 // The mapping range will flip, swap end points and un-flip.
494 Channel->SetParameter(1, Channel->GetParameter(0) - 1.f); // It doesn't actually matter what we do here, as long as we get the ordering & mapping range correct.
495 SetAttributeChannelRange(Name, FInterval1f(MappedRange.Max, MappedRange.Min), EMappingRangeSpace::Parent);
496 Index = Index == 0 ? 1 : 0;
497 }
498 else
499 {
500 SetAttributeChannelRange(Name, MappedRange, EMappingRangeSpace::Parent);
501 }
502
503 return Index;
504 }
505 else if (Index == 0) // Case 3: Moving the first point (which has exactly 1 neighbor)
506 {
507 // Save internal attribute point parameters for later re-parameterization after changing the child space mapping.
508 TArray<float> InternalParameters = CacheChannelKnots<float>(Name);
509
510 float ParentSpaceUpperBound = GetMappedChildSpace(Name).Max;
511 const float NeighborParentSpaceParameter = GetAttributeParameter<AttrType>(Name, Index + 1);
512 const bool bAttributesWillReorder = ParentSpaceParameter > NeighborParentSpaceParameter;
513
514 // If the end point is passing its neighbor, we need to actually reshuffle the points.
515 if (bAttributesWillReorder)
516 {
517 Index = SetParentSpaceParameter();
518
519 // Shift values left up to the last invalidated index.
520 const int32 InvalidatedInternalParameterIdx = FMath::Clamp(Index - 1, 0, InternalParameters.Num() - 1);
521 for (int InternalParameterIdx = 1; InternalParameterIdx <= InvalidatedInternalParameterIdx; ++InternalParameterIdx)
522 {
523 InternalParameters[InternalParameterIdx - 1] = InternalParameters[InternalParameterIdx];
524 }
525
526 // Do one of 2 things to the actual invalidated index:
527 if (Index == Channel->GetNumPoints() - 1)
528 {
529 // Old upper end point is now an internal point and needs to be re-parameterized.
530 InternalParameters[InvalidatedInternalParameterIdx] = ParentSpaceUpperBound;
531 }
532 else
533 {
534 // Old lower end point is now an internal point and needs to be re-parameterized.
535 InternalParameters[InvalidatedInternalParameterIdx] = ParentSpaceParameter;
536 }
537 }
538
539 // Updates the lower bound of the mapping range.
540 ParentSpaceUpperBound = Index == Channel->GetNumPoints() - 1 ? ParentSpaceParameter : ParentSpaceUpperBound;
541 const float ParentSpaceLowerBound = bAttributesWillReorder ? NeighborParentSpaceParameter : ParentSpaceParameter;
542 SetAttributeChannelRange(Name, FInterval1f(ParentSpaceLowerBound, ParentSpaceUpperBound), EMappingRangeSpace::Parent);
543
544 // Re-parameterize the internal points so that they do not move in parent space after the mapping range changed
545 RestoreChannelKnots(Name, InternalParameters);
546
547 if (bAttributesWillReorder)
548 {
549 SanitizeChannelSpace();
550 }
551
552 return Index;
553 }
554 else if (Index == Channel->GetNumPoints() - 1) // Case 4: Moving the last point (which has exactly 1 neighbor)
555 {
556 // Save internal attribute point parameters for later re-parameterization after changing the child space mapping.
557 TArray<float> InternalParameters = CacheChannelKnots<float>(Name);
558
559 float ParentSpaceLowerBound = GetMappedChildSpace(Name).Min;
560 const float NeighborParentSpaceParameter = GetAttributeParameter<AttrType>(Name, Index - 1);
561 const bool bAttributesWillReorder = ParentSpaceParameter < NeighborParentSpaceParameter;
562
563 // If the end point is passing its neighbor, we need to actually reshuffle the points.
564 if (bAttributesWillReorder)
565 {
566 Index = SetParentSpaceParameter();
567
568 // Shift values right up to the last invalidated index.
569 const int32 InvalidatedInternalParameterIdx = FMath::Clamp(Index - 1, 0, InternalParameters.Num() - 1);
570 for (int InternalParameterIdx = InternalParameters.Num() - 2; InternalParameterIdx >= InvalidatedInternalParameterIdx; --InternalParameterIdx)
571 {
572 InternalParameters[InternalParameterIdx + 1] = InternalParameters[InternalParameterIdx];
573 }
574
575 // Do one of 2 things to the actual invalidated index:
576 if (Index == 0)
577 {
578 // Old upper end point is now an internal point and needs to be re-parameterized.
579 InternalParameters[InvalidatedInternalParameterIdx] = ParentSpaceLowerBound;
580 }
581 else
582 {
583 // Old lower end point is now an internal point and needs to be re-parameterized.
584 InternalParameters[InvalidatedInternalParameterIdx] = ParentSpaceParameter;
585 }
586 }
587
588 // Update the child space mapping
589 ParentSpaceLowerBound = Index == 0 ? ParentSpaceParameter : ParentSpaceLowerBound;
590 float ParentSpaceUpperBound = bAttributesWillReorder ? NeighborParentSpaceParameter : ParentSpaceParameter; // Prevent over-shrinking if we reshuffle.
591 SetAttributeChannelRange(Name, FInterval1f(ParentSpaceLowerBound, ParentSpaceUpperBound), EMappingRangeSpace::Parent);
592
593 // Re-parameterize the internal points so that they do not move in parent space after the mapping range changed
594 RestoreChannelKnots(Name, InternalParameters);
595
596 if (bAttributesWillReorder)
597 {
598 SanitizeChannelSpace();
599 }
600
601 return Index;
602 }
603 else // Case 5: Moving an internal point (which has exactly 2 neighbors)
604 {
605 FInterval1f MappedRange = GetMappedChildSpace(Name);
606 Index = SetParentSpaceParameter();
607
608 if (Index == 0)
609 {
610 // The internal point is now the lower end point
611 MappedRange.Min = ParentSpaceParameter;
612 }
613 else if (Index == Channel->GetNumPoints() - 1)
614 {
615 // The internal point is now the upper end point
616 MappedRange.Max = ParentSpaceParameter;
617 }
618
619 SetAttributeChannelRange(Name, MappedRange, EMappingRangeSpace::Parent);
620
621 return Index;
622 }
623 }
624
625 return INDEX_NONE;
626}

658template <typename AttrType>
659void FTangentSpline::RestoreChannelKnots(const FName& InName, const TArray<AttrType>& InKnots)
660{
661 using AttrSplineType = UE::Geometry::Spline::TTangentBezierSpline<AttrType>;
662
663 if (InKnots.Num() == 0) return;
664
665 if (AttrSplineType* Channel = GetTypedAttributeChannel<AttrSplineType>(InName))
666 {
667 // Provided knots are for internal points, bail out if we don't have the correct number of knots.
668 if (ensureAlwaysMsgf(InKnots.Num() == Channel->GetNumPoints() - 2, TEXT("Invalid number of knots!")))
669 {
670 for (int InternalIdx = 1; InternalIdx < Channel->GetNumPoints() - 1; ++InternalIdx)
671 {
672 Channel->SetParameter(InternalIdx, MapParameterToChildSpace(InName, InKnots[InternalIdx - 1]));
673 }
674 }
