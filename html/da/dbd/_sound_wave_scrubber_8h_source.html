100 void DecodeToDataChunk(FDecodedDataChunk& InOutDataChunk, float InDecoderSeekTimeSeconds);

103 // Returns INDEX_NONE if no decoded data was available for the given read index

107 // Returns the index for the decoded chunk

111 // Data for granulation

113 Audio::Grain::FEnvelope GrainEnvelope;

115 // The grain duration in frames
116 FVector2D TargetGrainDurationRange;
117 FVector2D GrainDurationRange;
118 float GrainDurationSeconds = 0.0f;
119
120 // Grain duration as computed by scrub velocity based on the grain duration ranges
121 int32 CurrentGrainDurationFrames = 0;
122 int32 CurrentHalfGrainDurationFrames = 0;
123
124 // Utility data useful for debugging the scrubber.
125 // The grain count is a running tally of the numebr of grains rendered
126 int32 GrainCount = 0;
127 // This should never go above 2
128 int32 NumActiveGrains = 0;
129
130 // Active grain data for a single grain
131 struct FGrain
132 {
133 // The number of frames this grain has rendered
134 // If this is larger than GrainDurationFrames it's an inactive grain
135 // this is used to look up the grain envelope to find the amplitude of the grain per frame
136 int32 CurrentRenderedFramesCount = INDEX_NONE;
137
138 // Index into the decoded data array
139 int32 DecodedDataChunkIndex = 0;
140
141 // The current read frame of the grain
142 int32 CurrentReadFrame = INDEX_NONE;
143
144 // The duration of this grain. Set when the grain spawns.
145 int32 GrainDurationFrames = 0;
146 };
147 // The grain data used to render the granular audio
148 TArray<FGrain> ActiveGrains;
149
150 // This is a frame count until we need to spawn another grain
151 int32 NumFramesTillNextGrainSpawn = 0;
152
153 // Spawns a new grain at the value of the current playhead
154 FGrain SpawnGrain();
155
156 // Makes sure the grain is referencing valid decoded data
157 void UpdateGrainDecodeData(FGrain& InGrain);
158
159 // Renders the grain, mixing it into the output buffer. Returns the frame index that is half
160 void RenderActiveGrains(TArrayView<float>& OutAudio, int32 InStartFrame, int32 InNumFramesToRender);
161
162 // Used for setting user parameters
163 FCriticalSection CritSect;
164
165 // Whether or not we're actively scrubbing audio playback.
166 bool bIsScrubbing = false;
167 bool bIsScrubbingDueToBeingStationary = true;
168
169 // Whether or not we scrub while the playhead doesn't move
170 bool bIsScrubbingWhileStationary = true;
171
172 // Whether or not we have an error with the decoder
173 bool bHasErrorWithDecoder = false;
174};
