164class FStreamingSourceShapeHelper
165{
166public:
167 inline static bool IsSourceAffectingGrid(const TSet<FName>& InSourceTargetGrids, EStreamingSourceTargetBehavior InSourceTargetBehavior, FName InGridName)
168 {
169 switch (InSourceTargetBehavior)
170 {
171 case EStreamingSourceTargetBehavior::Include:
172 return InSourceTargetGrids.IsEmpty() || InSourceTargetGrids.Contains(InGridName);
173 case EStreamingSourceTargetBehavior::Exclude:
174 return !InSourceTargetGrids.Contains(InGridName);
175 default:
176 checkNoEntry();
177 }
178
179 return false;
180 }
181
182 inline static void ForEachShape(float InGridLoadingRange, float InDefaultRadius, bool bInProjectIn2D, const FVector& InLocation, const FRotator& InRotation, const TArray<FStreamingSourceShape>& InShapes, TFunctionRef<void(const FSphericalSector&)> InOperation, float InExtraRadius = 0.f, float InExtraAngle = 0.f)
183 {
184 const FTransform Transform(bInProjectIn2D ? FRotator(0, InRotation.Yaw, 0) : InRotation, InLocation);
185 if (InShapes.IsEmpty())
186 {
187 if ((InDefaultRadius + InExtraRadius) > 0)
188 {
189 const FSphericalSector LocalShape(FVector::ZeroVector, InDefaultRadius + InExtraRadius);
190 check(LocalShape.IsValid()); // Radius, axis and angle should be all valid here
191
192 InOperation(LocalShape.TransformBy(Transform));
193 }
194 }
195 else
196 {
197 for (const FStreamingSourceShape& Shape : InShapes)
198 {
199 const FVector::FReal ShapeRadius = (Shape.bUseGridLoadingRange ? (InGridLoadingRange * Shape.LoadingRangeScale) : Shape.Radius) + InExtraRadius;
200 const FVector::FReal ShapeAngle = Shape.bIsSector ? FMath::Min(Shape.SectorAngle + InExtraAngle, 360.0f) : 360.0f;
201 const FRotator ShapeRotation = bInProjectIn2D ? FRotator(0, Shape.Rotation.Yaw, 0) : Shape.Rotation;
202 const FVector ShapeAxis = ShapeRotation.IsNearlyZero() ? FVector::ForwardVector : ShapeRotation.Vector();
203 const FSphericalSector LocalShape(bInProjectIn2D ? FVector(Shape.Location.X, Shape.Location.Y, 0) : Shape.Location, ShapeRadius, ShapeAxis, ShapeAngle);
204
205 if (LocalShape.IsValid())
206 {
207 InOperation(LocalShape.TransformBy(Transform));
208 }
209 }
210 }
211 }
