71 static inline float LambertWLower(const float X)
72 {
73 // Clamp to valid domain
74 const float XValid = FMath::Clamp(X, -1.0f / UE_EULERS_NUMBER + UE_SMALL_NUMBER, -UE_SMALL_NUMBER);
75
76 const float Redx = 0.625529587f * 0.588108778f + XValid;
77
78 // Polynomial approximation for when close to -(exp(-1))
79 if (Redx <= 0.09765625f)
80 {
81 const float R = FMath::Sqrt(Redx);
82 float W = -3.30250000e+2f;
83 W = W * R + 3.53563141e+2f;
84 W = W * R - 1.91617889e+2f;
85 W = W * R + 4.94172478e+1f;
86 W = W * R - 1.23464909e+1f;
87 W = W * R - 1.38704872e+0f;
88 W = W * R - 1.99431837e+0f;
89 W = W * R - 1.81044364e+0f;
90 W = W * R - 2.33166337e+0f;
91 W = W * R - 1.00000000e+0f;
92 return W;
93 }
94 // From "New approximations to the principal real - valued branch of the Lambert W function"
95 else
96 {
97 const float C0 = +1.68090820e-1f;
98 const float C1 = -2.96497345e-3f;
99 const float C2 = -2.87322998e-2f;
100 const float C3 = +7.07275391e-1f;
101
102 const float S = FMath::Log2(-XValid) * -UE_LN2 - 1.0f;
103 float T = FMath::Sqrt(S);
104 float W = -1.0f - S - (1.0f / (FMath::Exp2(C2 * T) * (C1 * T) + C3 / T + C0));
105
106 if (XValid > -0x1.0p-116f)
107 {
108 const float E = FMath::Exp2(W / UE_LN2 + 32.0f);
109 W = -(W * E - 0x1.0p32f * XValid) / (W * E + E) + W;
110 }
111 else
112 {
113 T = W / (1.0f + W);
114 W = T * FMath::Loge(XValid / W) + T;
115 }
116
117 return W;
118 }
119 }

172 template <typename T>
173 static void CriticalSpringDamper(
174 T& InOutX,
175 T& InOutV,
176 T TargetX,
177 float SmoothingTime,
178 float DeltaTime)
179 {
180 if (SmoothingTime < UE_SMALL_NUMBER)
181 {
182 InOutX = TargetX;
183 InOutV = T(0);
184 return;
185 }
186
187 float Y = SmoothingTimeToDamping(SmoothingTime) / 2.0f;
188 T J0 = InOutX - TargetX;
189 T J1 = InOutV + J0 * Y;
190 float EyDt = FMath::InvExpApprox(Y * DeltaTime);
191
192 InOutX = EyDt * (J0 + J1 * DeltaTime) + TargetX;
193 InOutV = EyDt * (InOutV - J1 * Y * DeltaTime);
194 }

606 template <typename TVector>
607 static void SpringCharacterUpdate(
608 TVector& InOutPosition,
609 TVector& InOutVelocity,
610 TVector& InOutAcceleration,
611 const TVector& TargetVelocity,
612 float SmoothingTime,
613 float DeltaTime,
614 float VDeadzone = 1e-2f,
615 float ADeadzone = 1e-4f)
616 {
617 float Y = SmoothingTimeToDamping(SmoothingTime) / 2.0f;
618 TVector J0 = InOutVelocity - TargetVelocity;
619 TVector J1 = InOutAcceleration + J0 * Y;
620 float EyDt = FMath::InvExpApprox(Y * DeltaTime);
621
622 InOutPosition = EyDt * (((-J1) / (Y * Y)) + ((-J0 - J1 * DeltaTime) / Y)) +
623 (J1 / (Y * Y)) + J0 / Y + TargetVelocity * DeltaTime + InOutPosition;
624 InOutVelocity = EyDt * (J0 + J1 * DeltaTime) + TargetVelocity;
625 InOutAcceleration = EyDt * (InOutAcceleration - J1 * Y * DeltaTime);
626
627 if ((TargetVelocity - InOutVelocity).SquaredLength() < FMath::Square(VDeadzone))
628 {
629 // We reached our target
630 InOutVelocity = TargetVelocity;
631
632 if (InOutAcceleration.SquaredLength() < FMath::Square(ADeadzone))
633 {
634 InOutAcceleration = TVector::ZeroVector;
635 }
636 }
637 }

706 static inline float SpringCharacterStoppingDistance(const float InitialVelocity, const float InitialAcceleration, const float SmoothingTime)
707 {
708 const float V = InitialVelocity;
709 const float A = InitialAcceleration;
710 const float Y = SmoothingTimeToDamping(SmoothingTime) / 2.0f;
711
712 if (FMath::Abs(A) < UE_SMALL_NUMBER)
713 {
714 return 2.0f * FMath::Abs(V) / FMath::Max(Y, UE_SMALL_NUMBER);
715 }
716
717 const float I0 = (V * Y * 2 + A) / FMath::Max(Y * Y, UE_SMALL_NUMBER);
718
719 const TOptional<float> T = SpringCharacterStoppingInflectionTime(V, A, SmoothingTime);
720
721 if (!T.IsSet())
722 {
723 return FMath::Abs(I0);
724 }
725 else
726 {
727 const float IT = (FMath::Exp(-Y * T.GetValue()) / FMath::Max(Y * Y, UE_SMALL_NUMBER)) * (T.GetValue() * A * Y + V * Y * (T.GetValue() * Y + 2.0f) + A);
728 return FMath::Abs(I0 - IT) + FMath::Abs(IT);
729 }
730 }

822 else

889 template <typename TVector>
890 static void VelocitySpringCharacterUpdate(
891 TVector& InOutPosition,
892 TVector& InOutVelocity,
893 TVector& InOutVelocityIntermediate,
894 TVector& InOutAcceleration,
895 TVector TargetVelocity,
896 float SmoothingTime,
897 float MaxAcceleration,
898 float DeltaTime,
899 float VDeadzone = 1e-2f,
900 float ADeadzone = 1e-4f)
901 {
902 TVector VDiff = TargetVelocity - InOutVelocityIntermediate;
903 float VDiffLength = VDiff.Length();
904 TVector VDiffDir = VDiffLength > 0.0001f
905 ? VDiff / VDiffLength
906 : TVector::ZeroVector;
907
908 float TGoalFuture = SmoothingTime;
909 TVector MaxVFuture = VDiffLength > TGoalFuture * MaxAcceleration
910 ? InOutVelocityIntermediate + (VDiffDir * MaxAcceleration) * TGoalFuture
911 : TargetVelocity;
912
913 float Y = SmoothingTimeToDamping(SmoothingTime) / 2.0f;
914 TVector J0 = InOutVelocity - MaxVFuture;
915 TVector J1 = InOutAcceleration + J0 * Y;
916 float EyDt = FMath::InvExpApprox(Y * DeltaTime);
917
918 InOutPosition = EyDt * (((-J1) / (Y * Y)) + ((-J0 - J1 * DeltaTime) / Y)) +
919 (J1 / (Y * Y)) + J0 / Y + MaxVFuture * DeltaTime + InOutPosition;
920 InOutVelocity = EyDt * (J0 + J1 * DeltaTime) + MaxVFuture;
921 InOutAcceleration = EyDt * (InOutAcceleration - J1 * Y * DeltaTime);
922 InOutVelocityIntermediate = VDiffLength > DeltaTime * MaxAcceleration
923 ? InOutVelocityIntermediate + VDiffDir * MaxAcceleration * DeltaTime
924 : TargetVelocity;
925
926 if ((TargetVelocity - InOutVelocity).SquaredLength() < FMath::Square(VDeadzone))
927 {
928 // We reached our target
929 InOutVelocity = TargetVelocity;
930
931 if (InOutAcceleration.SquaredLength() < FMath::Square(ADeadzone))
932 {
933 InOutAcceleration = TVector::ZeroVector;
934 }
935 }
936 }
