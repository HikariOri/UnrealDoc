71inline EItemFilterState CombineConjunctive(EItemFilterState InLeft, EItemFilterState InRight)
72{
73 switch (InLeft)
74 {
75 case EItemFilterState::Include: return InRight == EItemFilterState::Exclude ? EItemFilterState::Exclude : InLeft;
76 case EItemFilterState::Exclude: return EItemFilterState::Exclude;
77 case EItemFilterState::DoNotCare: return InRight;
78 default: checkNoEntry(); return EItemFilterState::DoNotCare;
79 }
80}

82inline EItemFilterState CombineDisjunctive(EItemFilterState InLeft, EItemFilterState InRight)

88 case EItemFilterState::DoNotCare: return InRight;
89 default: checkNoEntry(); return EItemFilterState::DoNotCare;
