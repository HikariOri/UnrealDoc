222 struct FFunctionRegistrar
223 {
224 FFunctionRegistrar(FString Name, TestMethod Func, int32 LineNumber, FString TestTags)
225 {
226 // Only register the test method if not already registered
227 if (!DerivedType::Methods.Contains(Name))
228 {
229 DerivedType::TestRunner->TestNames.Add(Name);
230 DerivedType::Methods.Add(Name, Func);
231 DerivedType::TestRunner->TestLineNumbers.Add(Name, LineNumber);
232
233 FString CommonTestTags = DerivedType::TestRunner->GetTestTags();
234 FString CompleteTags = FString::Printf(TEXT("%s%s"), *CommonTestTags, *TestTags);
235 if (!CompleteTags.IsEmpty())
236 {
237 FString FullName = FString::Printf(TEXT("%s.%s"), *DerivedType::TestRunner->GetBeautifiedTestName(), *Name);
238 bool Registered = FAutomationTestFramework::Get().RegisterAutomationTestTags(FullName, CompleteTags);
239 if (Registered)
240 {
241 MethodTags = CompleteTags;
242 }
243 }
244 }
245 }
246
247 const FString GetTestMethodTags() const { return MethodTags; }
248 private:
249 FString MethodTags;
250 };
