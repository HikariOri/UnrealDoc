126 bool AreAcksSatisfied() const
127 {
128 if (ReceivedAcks.Num() < NeededAcks.Num())
129 {
130 return false;
131 }
132 for (const FRemoteServerId& Needed : NeededAcks)
133 {
134 if (!ReceivedAcks.Contains(Needed))
135 {
136 return false;
137 }
138 }
139 return true;
140 }

241// ERemoteHeapCoverageFlow is declared at global scope (outside namespace UE) so that
242// UHT can generate reflection data for it via UENUM(). The enum is always present;
243// only the mutable FRemoteHeapCoverage tracking struct is compiled out in non-test builds.
244UENUM()
245enum class ERemoteHeapCoverageFlow : uint32
246{
247 // TickSubsystem flows
248 TickSubsystem_LineNotFound_AllocatingServer UMETA(DisplayName="TickSubsystem: line not found, we are allocator -> notify deallocated"),
249 TickSubsystem_LineRemote_Forward UMETA(DisplayName="TickSubsystem: line is Remote -> forward to LastMutatorServerId"),
250 TickSubsystem_ExclusiveRequest_LineLocked UMETA(DisplayName="TickSubsystem: lower-priority remote exclusive -> deferred"),
251 TickSubsystem_ExclusiveRequest_FromExclusive UMETA(DisplayName="TickSubsystem: evict Exclusive line, empty ForwardedServers"),
252 TickSubsystem_ExclusiveRequest_FromShared_NoPendingWrite UMETA(DisplayName="TickSubsystem: evict Shared line, build ForwardedServers"),
253 TickSubsystem_ExclusiveRequest_FromShared_PendingWrite UMETA(DisplayName="TickSubsystem: yield pending write to higher-priority requester"),
254 TickSubsystem_SharedRequest_LineLocked UMETA(DisplayName="TickSubsystem: lower-priority remote read -> deferred"),
255 TickSubsystem_SharedRequest_Served UMETA(DisplayName="TickSubsystem: send copy, downgrade Exclusive->Shared if needed"),
256 TickSubsystem_SharedRequest_HasPendingExclusiveUpgrade UMETA(DisplayName="TickSubsystem: add new child to NeededAcks, request eviction"),
257
258 // TickRequest flows
259 TickRequest_AlreadySatisfied UMETA(DisplayName="TickRequest: line present at required access level, skip"),
260 TickRequest_NewExclusiveRequest UMETA(DisplayName="TickRequest: no pending write request -> send new exclusive RPC"),
261 TickRequest_BoostExclusiveRequest UMETA(DisplayName="TickRequest: pending write exists at lower priority -> re-send at higher"),
262 TickRequest_NewReadRequest UMETA(DisplayName="TickRequest: no pending read request -> send new read RPC"),
263 TickRequest_BoostReadRequest UMETA(DisplayName="TickRequest: pending read exists at lower priority -> re-send at higher"),
264 TickRequest_ReadSkipped_HigherPriorityWrite UMETA(DisplayName="TickRequest: higher-priority write pending -> skip read request"),
265
266 // OnLineReceived flows
267 OnLineReceived_SharedRequestSatisfied UMETA(DisplayName="OnLineReceived: pending Shared request satisfied, removed"),
268 OnLineReceived_ExclusiveRequest_WaitingForAcks UMETA(DisplayName="OnLineReceived: pending Exclusive request, data arrived, still waiting on acks"),
269
270 // OnExclusiveAck flows
271 OnExclusiveAck_Abandoned UMETA(DisplayName="OnExclusiveAck: no matching write request -> forward ack onward"),
272 OnExclusiveAck_NewerVersion UMETA(DisplayName="OnExclusiveAck: ack version newer than tracked -> reset ack tracking"),
273 OnExclusiveAck_SameVersion_Promoted UMETA(DisplayName="OnExclusiveAck: all acks collected and data present -> promote to Exclusive"),
274
275 // OnLineDeallocated flow
276 OnLineDeallocated UMETA(DisplayName="OnLineDeallocated: remove line and pending requests from local cache"),
277
278 Count UMETA(Hidden)
279};
