26struct FBlueprintEditorApplicationModes
27{
28 // Mode identifiers
29 static UE_API const FName StandardBlueprintEditorMode;
30 static UE_API const FName BlueprintDefaultsMode;
31 static UE_API const FName BlueprintComponentsMode;
32 static UE_API const FName BlueprintInterfaceMode;
33 static UE_API const FName BlueprintMacroMode;
34 static FText GetLocalizedMode( const FName InMode )
35 {
36 static TMap< FName, FText > LocModes;
37
38 if (LocModes.Num() == 0)
39 {
40 LocModes.Add( StandardBlueprintEditorMode, NSLOCTEXT("BlueprintEditor", "StandardBlueprintEditorMode", "Graph") );
41 LocModes.Add( BlueprintDefaultsMode, NSLOCTEXT("BlueprintEditor", "BlueprintDefaultsMode", "Defaults") );
42 LocModes.Add( BlueprintComponentsMode, NSLOCTEXT("BlueprintEditor", "BlueprintComponentsMode", "Components") );
43 LocModes.Add( BlueprintInterfaceMode, NSLOCTEXT("BlueprintEditor", "BlueprintInterfaceMode", "Interface") );
44 LocModes.Add( BlueprintMacroMode, NSLOCTEXT("BlueprintEditor", "BlueprintMacroMode", "Macro") );
45 }
46
47 check( InMode != NAME_None );
48 const FText* OutDesc = LocModes.Find( InMode );
49 check( OutDesc );
50 return *OutDesc;
51 }
52private:
53 FBlueprintEditorApplicationModes() {}
54};
