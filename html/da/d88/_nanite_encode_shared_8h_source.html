200class FBitWriter
201{
202public:
203 FBitWriter(TArray<uint8>& Buffer) :
204 Buffer(Buffer),
205 PendingBits(0ull),
206 NumPendingBits(0)
207 {
208 }
209
210 void PutBits(uint32 Bits, uint32 NumBits)
211 {
212 check((uint64)Bits < (1ull << NumBits));
213 PendingBits |= (uint64)Bits << NumPendingBits;
214 NumPendingBits += NumBits;
215
216 while (NumPendingBits >= 8)
217 {
218 Buffer.Add((uint8)PendingBits);
219 PendingBits >>= 8;
220 NumPendingBits -= 8;
221 }
222 }
223
224 void Flush(uint32 Alignment=1)
225 {
226 if (NumPendingBits > 0)
227 Buffer.Add((uint8)PendingBits);
228 while (Buffer.Num() % Alignment != 0)
229 Buffer.Add(0);
230 PendingBits = 0;
231 NumPendingBits = 0;
232 }
233
234private:
235 TArray<uint8>& Buffer;
236 uint64 PendingBits;
237 int32 NumPendingBits;
238};

241template<typename TLambda>
242void ProcessPageClusters(const FPage& Page, const TArray<FClusterGroupPart>& Parts, TLambda&& Lambda)
243{
244 uint32 LocalClusterIndex = 0;
245 for (uint32 PartIndex = 0; PartIndex < Page.PartsNum; PartIndex++)
246 {
247 const FClusterGroupPart& Part = Parts[Page.PartsStartIndex + PartIndex];
248 for (uint32 i = 0; i < (uint32)Part.Clusters.Num(); i++)
249 {
250 Lambda(LocalClusterIndex, Part.Clusters[i]);
251 LocalClusterIndex++;
252 }
253 }
254 check(LocalClusterIndex == Page.NumClusters);
255}
