102
103 if (RequiredVertexCount > 0 )
104 {
105#if !SLATE_USE_32BIT_INDICES

109 int32 RequiredBufferSize = RequiredVertexCount*sizeof(VertexType);
110
111 // resize if needed

114 ResizeBuffer(RHICmdList, RequiredBufferSize);
