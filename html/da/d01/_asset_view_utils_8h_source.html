22 struct FLoadAssetsSettings
23 {
24 // Prompt the user with the number of unloaded assets before continuing to load
25 bool bAlwaysPromptBeforeLoading = false;
26 // Show a notification to the user that some assets failed to load
27 bool bShowFailureNotification = true;
28 // Follow redirectors and return the target assets.
29 // We usually don't want to follow redirects when loading objects for the Content Browser. It would
30 // allow a user to interact with a ghost/unverified asset as if it were still alive.
31 bool bFollowRedirectors = false;
32 // Allow the user to cancel the load operation
33 bool bAllowCancel = false;
34 // Advanced setting: allow loading of world partition map packages.
35 bool bLoadWorldPartitionMaps = false;
36 // Advanced setting: load all external objects/actors with e.g. world partition map packages.
37 bool bLoadAllExternalObjects = false;
38 };
