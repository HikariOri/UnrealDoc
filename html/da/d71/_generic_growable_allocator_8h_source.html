167 bool CanFitEntry(uint32 Size, uint32 Alignment)
168 {
169 // Compute MaxFreeEntrySize if necessary (should only happen if this chunk was just allocated from)
170 if (MaxFreeEntrySize == MaxFreeEntrySizeDirty)
171 {
172 MaxFreeEntrySize = 0;
173 for (FFreeEntry* Entry = FreeList; Entry; Entry = Entry->Next)
174 {
175 MaxFreeEntrySize = FMath::Max(Entry->BlockSize, MaxFreeEntrySize);
176 }
177 }
178
179 // Return false if we trivially don't fit
180 if (Size > MaxFreeEntrySize)
181 {
182 return false;
183 }
184
185 // Return true if we trivially do fit
186 if (Size + Alignment - 1 <= MaxFreeEntrySize)
187 {
188 return true;
189 }
190
191 // Slow method - search the free entries for a free chunk
192 bool bResult = false;
193 for (FFreeEntry *Entry = FreeList; Entry; Entry = Entry->Next)
194 {
195 if (Entry->CanFit(Size, Alignment))
196 {
197 bResult = true;
198 break;
199 }
200 }
201 return bResult;
202 }

211 FGrowableAllocationBase* Malloc(uint32 Size, uint32 Alignment, uint32 MinAllocationSize, int32 OwnerType)
212 {
213 checkSlow(Alignment != 0);
214
215 // multi-thread protection
216 FScopeLock ScopeLock(CriticalSection);
217
218 // Alignment here is assumed to be for location and size
219 const uint32 AlignedSize = Align<uint32>(Size, Alignment);
220
221 // Update stats.
222 const uint32 WastedSize = AlignedSize - Size;
223 TotalWaste += WastedSize;
224 CurrentAllocs++;
225 TotalAllocs++;
226
227 // look for a good free chunk
228 FFreeEntry* Prev = NULL;
229 FFreeEntry* FindEntry = NULL;
230
231 for (FFreeEntry* Entry = FreeList; Entry; Entry = Entry->Next)
232 {
233 if (Entry->CanFit(AlignedSize, Alignment))
234 {
235 FindEntry = Entry;
236 break;
237 }
238 Prev = Entry;
239 }
240
241 if (FindEntry != NULL)
242 {
243 if (FindEntry->BlockSize == MaxFreeEntrySize)
244 {
245 // We're probably about to just split our largest entry, so mark the max size as dirty to indicate it needs recomputing
246 MaxFreeEntrySize = MaxFreeEntrySizeDirty;
247 }
248 // Use it, leaving over any unused space
249 UsedMemorySize += AlignedSize;
250 bool bDelete;
251 uint32 Padding;
252 uint32 Offset = FindEntry->Split(AlignedSize, Alignment, bDelete, Padding, MinAllocationSize);
253 if (bDelete)
254 {
255 FFreeEntry*& PrevRef = Prev ? Prev->Next : FreeList;
256 PrevRef = FindEntry->Next;
257 delete FindEntry;
258 }
259
260 FGrowableAllocationBase* Allocation = CreateAllocationStruct();
261 Allocation->Size = AlignedSize;
262 Allocation->Padding = Padding;
263 Allocation->Offset = Offset;
264#if !UE_BUILD_SHIPPING
265 Allocation->OwnerType = OwnerType;
266#endif
267 LLM_IF_ENABLED(FLowLevelMemTracker::Get().OnLowLevelAlloc(ELLMTracker::Default, GetAddressForTracking(Offset), Size));
268 MemoryTrace_Alloc(uint64(GetAddressForTracking(Offset)), Size, Alignment, EMemoryTraceRootHeap::SystemMemory);
269
270 // let the implementation fill in any more
271 InitializeAllocationStruct(Allocation);
272 return Allocation;
273 }
274
275 // if no suitable blocks were found, we must fail
276 FPlatformMisc::LowLevelOutputDebugStringf(TEXT("Failed to allocate GPU memory (Size: %d)"), AlignedSize);
277 return nullptr;
278 }

280 bool Free(FGrowableAllocationBase* Memory)
281 {
282 // multi-thread protection
283 FScopeLock ScopeLock(CriticalSection);
284
285 uint64 Padding = Memory->Padding;
286 uint64 Size = Memory->Size;
287 uint64 AllocationSize = Padding + Size;
288 uint32 Offset = Memory->Offset;
289
290 LLM_IF_ENABLED(FLowLevelMemTracker::Get().OnLowLevelFree(ELLMTracker::Default, GetAddressForTracking(Offset)));
291 MemoryTrace_Free(uint64(GetAddressForTracking(Offset)), EMemoryTraceRootHeap::SystemMemory);
292
293 // we are now done with the Allocation object
294 DestroyAllocationStruct(Memory);
295
296 UsedMemorySize -= Size;
297 CurrentAllocs--;
298
299 // Search for a place to insert a new free entry
300 FFreeEntry* Prev = NULL;
301 FFreeEntry* Entry = FreeList;
302 while (Entry && Offset > Entry->Location)
303 {
304 Prev = Entry;
305 Entry = Entry->Next;
306 }
307
308 // Are we right before this free entry?
309 if (Entry && (Offset + Size) == Entry->Location)
310 {
311 // Join with chunk
312 Entry->Location -= uint32(AllocationSize);
313 Entry->BlockSize += uint32(AllocationSize);
314
315 // Can we join the two entries?
316 if (Prev && (Prev->Location + Prev->BlockSize) == Entry->Location)
317 {
318 Prev->BlockSize += Entry->BlockSize;
319 Prev->Next = Entry->Next;
320 MaxFreeEntrySize = FMath::Max(MaxFreeEntrySize, Prev->BlockSize);
321 delete Entry;
322 }
323 else
324 {
325 MaxFreeEntrySize = FMath::Max(MaxFreeEntrySize, Entry->BlockSize);
326 }
327 return true;
328 }
329
330 // Are we right after the previous free entry?
331 if (Prev && (Prev->Location + Prev->BlockSize + Padding) == Offset)
332 {
333 // Join with chunk
334 Prev->BlockSize += uint32(AllocationSize);
335
336 // Can we join the two entries?
337 if (Entry && (Prev->Location + Prev->BlockSize) == Entry->Location)
338 {
339 Prev->BlockSize += Entry->BlockSize;
340 Prev->Next = Entry->Next;
341 delete Entry;
342 }
343 MaxFreeEntrySize = FMath::Max(MaxFreeEntrySize, Prev->BlockSize);
344 return true;
345 }
346
347 // Insert a new entry.
348 FFreeEntry* NewFree = new FFreeEntry(Entry, uint32(Offset - Padding), AllocationSize);
349 FFreeEntry*& PrevRef = Prev ? Prev->Next : FreeList;
350 PrevRef = NewFree;
351 MaxFreeEntrySize = FMath::Max(MaxFreeEntrySize, NewFree->BlockSize);
352 return true;
353 }

404private:
405 class FFreeEntry
406 {
407 public:
409 FFreeEntry(FFreeEntry *NextEntry, uint32 InLocation, uint64 InSize)
410 : Location(InLocation)
411 , BlockSize(uint32(InSize))
412 , Next(NextEntry)
413 {

415
422 bool CanFit(uint64 AlignedSize, uint32 Alignment)
423 {
424 // location of the aligned allocation
425 uint32 AlignedLocation = Align(Location, Alignment);
426
427 // if we fit even after moving up the location for alignment, then we are good
428 return (AlignedSize + (AlignedLocation - Location)) <= BlockSize;
429 }
430
439 uint32 Split(uint64 UsedSize, uint32 Alignment, bool &bDelete, uint32& Padding, uint32 MinSize)
440 {
441 // make sure we are already aligned
442 check((UsedSize & (Alignment - 1)) == 0);
443
444 // this is the pointer to the free data
445 uint32 FreeLoc = Align(Location, Alignment);
446
447 // Adjust the allocated size for any alignment padding
448 Padding = FreeLoc - Location;
449 uint64 AllocationSize = UsedSize + uint64(Padding);
450
451 // see if there's enough space left over for a new free chunk (of at least a certain size)
452 if (BlockSize - AllocationSize >= MinSize)
453 {
454 // update this free entry to just point to what's left after using the UsedSize
455 Location += uint32(AllocationSize);
456 BlockSize -= uint32(AllocationSize);
457 bDelete = false;
458 }
459 // if no more room, then just remove this entry from the list of free items
460 else
461 {
462 bDelete = true;
463 }
464
465 // return a usable pointer!
466 return FreeLoc;
467 }
468
470 uint32 Location;
471
473 uint32 BlockSize;
474
476 FFreeEntry* Next;
477 };

560 GrowableAllocationBaseType* Malloc(uint32 Size, uint32 Alignment, int32 OwnerType)
561 {
562 // make sure we have some minimal alignment
563 Alignment = FPlatformMath::Max<uint32>(Alignment, SubAllocationAlignment);
564
565 // multi-thread protection
566 FScopeLock ScopeLock(&CriticalSection);
567
568 ChunkAllocatorType *AvailableChunk = NULL;
569
570 // align the size to match what Malloc does below
571 const uint32 AlignedSize = Align<uint32>(Size, Alignment);
572
573 // Update stats.
574 const uint32 WastedSize = AlignedSize - Size;
575 TotalAllocated += Size;
576 TotalWaste += WastedSize;
577 CurrentAllocs++;
578 TotalAllocs++;
579
580 // search for an existing alloc chunk with enough space
581 for (int32 ChunkIndex = 0; ChunkIndex < AllocChunks.Num(); ChunkIndex++)
582 {
583 ChunkAllocatorType* Chunk = AllocChunks[ChunkIndex];
584 if (Chunk && Chunk->CanFitEntry(AlignedSize, Alignment))
585 {
586 AvailableChunk = Chunk;
587 break;
588 }
589 }
590
591 // create a new chunk with enough space + alignment to Switch_GrowableHeapAlignment_MB and allocate out of it
592 if (AvailableChunk == nullptr)
593 {
594 AvailableChunk = CreateAllocChunk(AlignedSize);
595 }
596
597 // allocate from the space in the chunk
598 GrowableAllocationBaseType* Result = nullptr;
599 if (AvailableChunk)
600 {
601 Result = (GrowableAllocationBaseType*)AvailableChunk->Malloc(AlignedSize, Alignment, SubAllocationAlignment, OwnerType);
602 }
603
604 if (AvailableChunk == nullptr || Result == nullptr)
605 {
606 OutOfMemory(AlignedSize);
607 return nullptr;
608 }
609
610#if ALLOCATION_HISTOGRAM
611 TotalAllocationsHistogram.FindOrAdd(AlignedSize)++;
612 PeakAllocationsHistogram.FindOrAdd(AlignedSize)++;
613
614 // track per type allocation info
615 AllocationInfo& TypeAllocInfo = PerTypeAllocationInfo.FindOrAdd(Result->OwnerType);
616
617 TypeAllocInfo.Counts.Allocations++;
618 TypeAllocInfo.TotalAllocated += Result->Size;
619#if ALLOCATION_HISTOGRAM_DETAILED
620 TypeAllocInfo.AllocationHistogram.FindOrAdd(Result->Size).Allocations++;
621#endif
622#endif // ALLOCATION_HISTOGRAM
623
624#if STATS
625 if (OwnerTypeToStatIdMap)
626 {
627 INC_MEMORY_STAT_BY_FName(OwnerTypeToStatIdMap[Result->OwnerType], Result->Size);
628 }
629#endif
630
631 return Result;
632 }

634 bool Free(GrowableAllocationBaseType* Memory)
635 {
636 if (Memory == nullptr)
637 {
638 return true;
639 }
640
641 // multi-thread protection
642 FScopeLock ScopeLock(&CriticalSection);
643
644 // starting address space idx used by the chunk containing the allocation
645 for (int32 ChunkIndex = 0; ChunkIndex < AllocChunks.Num(); ChunkIndex++)
646 {
647 ChunkAllocatorType* Chunk = AllocChunks[ChunkIndex];
648 if (Chunk && Chunk->DoesChunkContainAllocation(Memory))
649 {
650#if ALLOCATION_HISTOGRAM
651 PeakAllocationsHistogram[Memory->Size]--;
652
653 // untrack per type allocation info
654 AllocationInfo& TypeAllocInfo = PerTypeAllocationInfo.FindOrAdd(Memory->OwnerType);
655
656 TypeAllocInfo.TotalAllocated -= Memory->Size;
657 TypeAllocInfo.Counts.Frees++;
658#if ALLOCATION_HISTOGRAM_DETAILED
659 TypeAllocInfo.AllocationHistogram[Memory->Size].Frees++;
660#endif
661#endif // ALLOCATION_HISTOGRAM
662
663#if STATS
664 if (OwnerTypeToStatIdMap)
665 {
666 DEC_MEMORY_STAT_BY_FName(OwnerTypeToStatIdMap[Memory->OwnerType], Memory->Size);
667 }
668#endif
669 // free space in the chunk
670 Chunk->Free(Memory);
671 CurrentAllocs--;
672
673 // never toss the only chunk
674 if (Chunk->IsEmpty())// && AllocChunks.Num() > 1)
675 {
676 // if empty then unmap and decommit physical memory
677 RemoveAllocChunk(Chunk);
678 }
679
680 // return success
681 return true;
682 }
683 }
684
685 // if we got here, we failed to free the pointer
686 UE_LOGF(LogCore, Fatal, "Tried to free invalid pointer");
687 return false;
688 }

801 ChunkAllocatorType* CreateAllocChunk(uint64 Size)
802 {
803 ChunkAllocatorType* NewChunk = new ChunkAllocatorType(Size, MemoryType, &CriticalSection, UserData);
804 NewChunk->Initialize();
805 INC_MEMORY_STAT_BY_FName(StatRegionName, NewChunk->HeapSize);
806
807 // add a new entry to list (reusing any empty slots)
808 int32 EmptyEntryIdx = AllocChunks.Find(NULL);
809 if (EmptyEntryIdx == INDEX_NONE)
810 {
811 EmptyEntryIdx = AllocChunks.Add(NewChunk);
812 }
813 else
814 {
815 AllocChunks[EmptyEntryIdx] = NewChunk;
816 }
817
818#if STATS
819 UpdateMemoryStatMaxSizes();
820#endif
821
822 return NewChunk;
823 }
