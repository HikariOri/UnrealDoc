344template<class TDataClass>
345bool UBlackboardComponent::SetValue(FBlackboard::FKey KeyID, typename TDataClass::FDataType Value)
346{
347 const FBlackboardEntry* EntryInfo = BlackboardAsset ? BlackboardAsset->GetKey(KeyID) : nullptr;
348 if ((EntryInfo == nullptr) || (EntryInfo->KeyType == nullptr) || (EntryInfo->KeyType->GetClass() != TDataClass::StaticClass()))
349 {
350 return false;
351 }
352
353 const uint16 DataOffset = EntryInfo->KeyType->HasInstance() ? sizeof(FBlackboardInstancedKeyMemory) : 0;
354 uint8* RawData = GetKeyRawData(KeyID) + DataOffset;
355 if (RawData)
356 {
357 UBlackboardKeyType* KeyOb = EntryInfo->KeyType->HasInstance() ? KeyInstances[KeyID] : EntryInfo->KeyType;
358 const bool bChanged = TDataClass::SetValue((TDataClass*)KeyOb, RawData, Value);
359 if (bChanged)
360 {
361 NotifyObservers(KeyID);
362 if (BlackboardAsset->HasSynchronizedKeys() && IsKeyInstanceSynced(KeyID))
363 {
364 UAISystem* AISystem = UAISystem::GetCurrentSafe(GetWorld());
365 for (auto Iter = AISystem->CreateBlackboardDataToComponentsIterator(*BlackboardAsset); Iter; ++Iter)
366 {
367 UBlackboardComponent* OtherBlackboard = Iter.Value();
368 if (OtherBlackboard != nullptr && ShouldSyncWithBlackboard(*OtherBlackboard))
369 {
370 UBlackboardData* const OtherBlackboardAsset = OtherBlackboard->GetBlackboardAsset();
371 const FBlackboard::FKey OtherKeyID = OtherBlackboardAsset ? OtherBlackboardAsset->GetKeyID(EntryInfo->EntryName) : FBlackboard::InvalidKey;
372 if (OtherKeyID != FBlackboard::InvalidKey)
373 {
374 UBlackboardKeyType* OtherKeyOb = EntryInfo->KeyType->HasInstance() ? OtherBlackboard->KeyInstances[OtherKeyID] : EntryInfo->KeyType;
375 uint8* OtherRawData = OtherBlackboard->GetKeyRawData(OtherKeyID) + DataOffset;
376
377 TDataClass::SetValue((TDataClass*)OtherKeyOb, OtherRawData, Value);
378 OtherBlackboard->NotifyObservers(OtherKeyID);
379 }
380 }
381 }
382 }
383 }
384
385 return true;
386 }
387
388 return false;
389}
