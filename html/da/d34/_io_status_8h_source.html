177template<typename T>
178void TIoStatusOr<T>::Reset()
179{
180 EIoErrorCode ErrorCode = StatusValue.GetErrorCode();
181 StatusValue = EIoErrorCode::Unknown;
182
183 if (ErrorCode == EIoErrorCode::Ok)
184 {
185 ((T*)&Value)->~T();
186 }
187}
