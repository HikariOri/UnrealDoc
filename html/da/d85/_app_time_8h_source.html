26 template <typename TValue>
27 struct TRelaxedAtomic
28 {
29 std::atomic<TValue> Value;
30
31 TRelaxedAtomic() = default;
32
33 TRelaxedAtomic(TRelaxedAtomic const& Other)
34 {
35 (*this) = Other.Value;
36 }
37
38 TRelaxedAtomic(TValue InValue)
39 {
40 (*this) = InValue;
41 }
42
43 TRelaxedAtomic& operator = (TValue InValue)
44 {
45 Value.store(InValue, std::memory_order_relaxed);
46 return *this;
47 }
48
49 TRelaxedAtomic& operator = (TRelaxedAtomic const& RHS)
50 {
51 TValue LoadedValue = RHS;
52 (*this) = LoadedValue;
53 return *this;
54 }
55
56 operator TValue() const
57 {
58 return Value.load(std::memory_order_relaxed);
59 }

99 // Returns the FAppTime value for the current thread's time context. May be called by the game thread, or decendent TaskGraph tasks of the game thread.
100 // Calling this function from any other context will result in an ensure failure. This is because the game thread is the authority on the FAppTime value,
101 // and this value is pipelined through the TaskGraph so that decendent tasks receive the correct time value for the frame they belong to.
102 // Reading the FAppTime value outside of these contexts is a data race with the game thread.

194 // Equivalent to GetCurrentTime(), but can be called from any thread, rather than only from threads with valid time contexts.
195 // The returned value is loaded via a relaxed atomic read, and therefore can race updates from the game thread.
196 //
197 // Use only for systems that require an approximate incrementing timer, where getting a value from a frame that is off-by-one is not important.
198 // Do not use in systems where the current time directly affects the game simulation or rendered pixels. Those systems should use GetCurrentTime()
