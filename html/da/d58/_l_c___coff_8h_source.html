54#if LC_64_BIT
55 /*
56 From the COFF spec: "5.2.1. Type Indicators"
57 The following relocation types are defined for x64 and compatible processors.
58 Constant Description
59 IMAGE_REL_AMD64_ABSOLUTE The relocation is ignored.
60 IMAGE_REL_AMD64_ADDR64 The 64-bit VA of the relocation target.
61 IMAGE_REL_AMD64_ADDR32 The 32-bit VA of the relocation target.
62 IMAGE_REL_AMD64_ADDR32NB The 32-bit address without an image base (RVA).
63 IMAGE_REL_AMD64_REL32 The 32-bit relative address from the byte following the relocation.
64 IMAGE_REL_AMD64_REL32_1 The 32-bit address relative to byte distance 1 from the relocation.
65 IMAGE_REL_AMD64_REL32_2 The 32-bit address relative to byte distance 2 from the relocation.
66 IMAGE_REL_AMD64_REL32_3 The 32-bit address relative to byte distance 3 from the relocation.
67 IMAGE_REL_AMD64_REL32_4 The 32-bit address relative to byte distance 4 from the relocation.
68 IMAGE_REL_AMD64_REL32_5 The 32-bit address relative to byte distance 5 from the relocation.
69 IMAGE_REL_AMD64_SECTION The 16-bit section index of the section that contains the target.
70 This is used to support debugging information.
71 IMAGE_REL_AMD64_SECREL The 32-bit offset of the target from the beginning of its section.
72 This is used to support debugging information and static thread local storage.
73 IMAGE_REL_AMD64_SECREL7 A 7-bit unsigned offset from the base of the section that contains the target.
74 IMAGE_REL_AMD64_TOKEN CLR tokens.
75 IMAGE_REL_AMD64_SREL32 A 32-bit signed span-dependent value emitted into the object.
76 IMAGE_REL_AMD64_PAIR A pair that must immediately follow every span-dependent value.
77 IMAGE_REL_AMD64_SSPAN32 A 32-bit signed span-dependent value that is applied at link time.
78
79 This means that only a handful of relocation types need to be supported.
80 */
81
82 // also used in 32-bit

95#else
96 /*
97 From the COFF spec: "5.2.1. Type Indicators"
98 The following relocation type indicators are defined for Intel 386 and compatible
99 processors.
100 Constant Description
101 IMAGE_REL_I386_ABSOLUTE The relocation is ignored.
102 IMAGE_REL_I386_DIR16 Not supported.
103 IMAGE_REL_I386_REL16 Not supported.
104 IMAGE_REL_I386_DIR32 The target's 32-bit VA.
105 IMAGE_REL_I386_DIR32NB The target's 32-bit RVA.

107 IMAGE_REL_I386_SECTION The 16-bit section index of the section that contains the target.
108 This is used to support debugging information.
109 IMAGE_REL_I386_SECREL The 32-bit offset of the target from the beginning of its section.
110 This is used to support debugging information and static thread local storage.
111 IMAGE_REL_I386_TOKEN The CLR token.
112 IMAGE_REL_I386_SECREL7 A 7-bit offset from the base of the section that contains the target.

127 static const char* ToString(Enum value)
128 {
129 switch (value)
130 {
131 case Type::RELATIVE: return "RELATIVE";
132 case Type::SECTION_RELATIVE: return "SECTION_RELATIVE";
133 case Type::VA_32: return "VA_32";
134 case Type::RVA_32: return "RVA_32";
135
136#if LC_64_BIT
137 case Type::RELATIVE_OFFSET_1: return "RELATIVE_OFFSET_1";
138 case Type::RELATIVE_OFFSET_2: return "RELATIVE_OFFSET_2";
139 case Type::RELATIVE_OFFSET_3: return "RELATIVE_OFFSET_3";
140 case Type::RELATIVE_OFFSET_4: return "RELATIVE_OFFSET_4";
141 case Type::RELATIVE_OFFSET_5: return "RELATIVE_OFFSET_5";
142 case Type::VA_64: return "VA_64";
143#endif
144
145 case Type::UNKNOWN: return "UNKNOWN";
146 }
147
148 return "Invalid type";
149 }

151 // returns the byte distance to the position of where the relocation should be applied
152 static uint32_t GetByteDistance(Enum type)
153 {
154 switch (type)
155 {
156 case Type::RELATIVE:
157 return 0u;
158
159#if LC_64_BIT
160 case Type::RELATIVE_OFFSET_1:
161 return 1u;
162
163 case Type::RELATIVE_OFFSET_2:
164 return 2u;
165
166 case Type::RELATIVE_OFFSET_3:
167 return 3u;
168
169 case Type::RELATIVE_OFFSET_4:
170 return 4u;
171
172 case Type::RELATIVE_OFFSET_5:
173 return 5u;
174
175 case Type::VA_64:
176#endif
177 case Type::SECTION_RELATIVE:
178 case Type::VA_32:
179 case Type::RVA_32:
180 case Type::UNKNOWN:
181 break;
182 }
183
184 LC_ERROR_DEV("Unexpected relocation type %s (%d)", ToString(type), type);
185 return 0u;
186 }
