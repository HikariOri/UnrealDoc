279struct FOnDemandInstallProgress
280{
282 uint64 TotalContentSize = 0;
284 uint64 TotalInstallSize = 0;
286 uint64 CurrentInstallSize = 0;
287
288 FOnDemandInstallProgress& Combine(const FOnDemandInstallProgress& Other)
289 {
290 TotalContentSize += Other.TotalContentSize;
291 TotalInstallSize += Other.TotalInstallSize;
292 CurrentInstallSize += Other.CurrentInstallSize;
293 return *this;
294 }
295
296 uint64 GetTotalDownloadSize() const
297 {
298 return TotalInstallSize;
299 }
300
301 uint64 GetAlreadyDownloadedSize() const
302 {
303 return CurrentInstallSize;
304 }
305
306 float GetRelativeProgress() const
307 {
308 const double Progress = (TotalInstallSize > 0) ?
309 FMath::Clamp(double(CurrentInstallSize) / double(TotalInstallSize), 0.0, 1.0) :
310 0;
311
312 return float(Progress);
313 }
314
315 float GetAbsoluteProgress() const
316 {
317 const double Progress = (TotalContentSize > 0) ?
318 FMath::Clamp(double(GetCachedSize()) / double(TotalContentSize), 0.0, 1.0) :
319 0;
320
321 return float(Progress);
322 }
323
324 uint64 GetCachedSize() const
325 {
326 return TotalContentSize - TotalInstallSize + CurrentInstallSize;
327 }
328
329 uint64 GetTotalSize() const
330 {
331 return TotalContentSize;
332 }
333};
