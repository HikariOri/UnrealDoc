87 void Sort(TArray<FSceneOutlinerTreeItemPtr>& Array)
88 {
89 TArray<FSortPayload> SortData;
90 SortData.Reserve(Array.Num());
91 const auto End = Array.Num();
92 for (int32 Index = 0; Index < End; ++Index)
93 {
94 const auto& Element = Array[Index];
95
96 PrimaryKeyType PrimaryKey = PrimaryFunction(*Element);
97
98 SecondaryKeyType SecondaryKey;
99 if (SecondarySortMode != EColumnSortMode::None)
100 {
101 SecondaryKey = SecondaryFunction(*Element);
102 }
103
104 const int32 DefaultSortID = Element.IsValid() ? GetTypeHash(Element->GetID()) : 0;
105 SortData.Emplace(Index, MoveTemp(PrimaryKey), MoveTemp(SecondaryKey), DefaultSortID);
106 }
107
108 SortData.Sort([&](const FSortPayload& One, const FSortPayload& Two){
109 if (PrimarySortMode == EColumnSortMode::Ascending && One.PrimaryKey != Two.PrimaryKey)
110 return One.PrimaryKey < Two.PrimaryKey;
111 else if (PrimarySortMode == EColumnSortMode::Descending && One.PrimaryKey != Two.PrimaryKey)
112 return One.PrimaryKey > Two.PrimaryKey;
113
114 if (SecondarySortMode == EColumnSortMode::Ascending && One.SecondaryKey != Two.SecondaryKey)
115 return One.SecondaryKey < Two.SecondaryKey;
116 else if (SecondarySortMode == EColumnSortMode::Descending && One.SecondaryKey != Two.SecondaryKey)
117 return One.SecondaryKey > Two.SecondaryKey;
118
119 return One.DefaultSortID < Two.DefaultSortID;
120 });
121
122 TArray<FSceneOutlinerTreeItemPtr> NewArray;
123 NewArray.Reserve(Array.Num());
124
125 for (const auto& Element : SortData)
126 {
127 NewArray.Add(Array[Element.OriginalIndex]);
128 }
129 Array = MoveTemp(NewArray);
130 }
