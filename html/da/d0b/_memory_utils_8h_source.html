33// Trivially movable types without destructors only; memcpy is used and destructors are not called
34template<typename T, typename SizeType=int32>
35class FLLMArray
36{
37public:
38 FLLMArray()
39 : Array(StaticArray)
40 , Count(0)
41 , Capacity(StaticArrayCapacity)
42 , Allocator(nullptr)
43 {
44 }
45
46 ~FLLMArray()
47 {
48 Clear(true);
49 }
50
51 void SetAllocator(UE::LLMPrivate::FLLMAllocator* InAllocator)
52 {
53 Allocator = InAllocator;
54 }
55
56 SizeType Num() const
57 {
58 return Count;
59 }
60
61 void Clear(bool ReleaseMemory = false)
62 {
63 if (ReleaseMemory)
64 {
65 if (Array != StaticArray)
66 {
67 Allocator->Free(Array, Capacity * sizeof(T));
68 Array = StaticArray;
69 }
70 Capacity = StaticArrayCapacity;
71 }
72 Count = 0;
73 }
74
75 void Add(const T& Item)
76 {
77 if (Count == Capacity)
78 {
79 SizeType NewCapacity = DefaultCapacity;
80 if (Capacity)
81 {
82 NewCapacity = Capacity + (Capacity / 2);
83 ensureMsgf(NewCapacity > Capacity, TEXT("Unsigned integer overflow."));
84 }
85 Reserve(NewCapacity);
86 }
87
88 Array[Count] = Item;
89 ++Count;
90 }
91
92 T RemoveLast()
93 {
94 LLMCheck(Count > 0);
95 --Count;
96 T Last = Array[Count];
97
98 return Last;
99 }
100
101 T& operator[](SizeType Index)
102 {
103 LLMCheck(Index >= 0 && Index < Count);
104 return Array[Index];
105 }
106
107 const T& operator[](SizeType Index) const
108 {
109 LLMCheck(Index >= 0 && Index < Count);
110 return Array[Index];
111 }
112
113 const T* GetData() const {
114 return Array;
115 }
116
117 T& GetLast()
118 {
119 LLMCheck(Count > 0);
120 return Array[Count - 1];
121 }
122
123 void Reserve(SizeType NewCapacity)
124 {
125 if (NewCapacity == Capacity)
126 {
127 return;
128 }
129
130 if (NewCapacity <= StaticArrayCapacity)
131 {
132 if (Array != StaticArray)
133 {
134 if (Count)
135 {
136 memcpy(StaticArray, Array, Count * sizeof(T));
137 }
138 Allocator->Free(Array, Capacity * sizeof(T));
139
140 Array = StaticArray;
141 Capacity = StaticArrayCapacity;
142 }
143 }
144 else
145 {
146 NewCapacity = AlignArbitrary(NewCapacity, ItemsPerPage);
147
148 T* NewArray = (T*)Allocator->Alloc(NewCapacity * sizeof(T));
149
150 if (Count)
151 {
152 memcpy(NewArray, Array, Count * sizeof(T));
153 }
154 if (Array != StaticArray)
155 {
156 Allocator->Free(Array, Capacity * sizeof(T));
157 }
158
159 Array = NewArray;
160 Capacity = NewCapacity;
161 }
162 }
163
164 void operator=(const FLLMArray<T>& Other)
165 {
166 Clear();
167 Reserve(Other.Count);
168 memcpy(Array, Other.Array, Other.Count * sizeof(T));
169 Count = Other.Count;
170 }
171
172 void Trim()
173 {
174 // Trim if usage has dropped below 3/4 of the total capacity
175 if (Array != StaticArray && Count < (Capacity - (Capacity / 4)))
176 {
177 Reserve(Count);
178 }
179 }
180
181private:
182 T* Array;
183 SizeType Count;
184 SizeType Capacity;
185
186 UE::LLMPrivate::FLLMAllocator* Allocator;
187
188 static const int StaticArrayCapacity = 64; // because the default size is so large this actually saves memory
189 T StaticArray[StaticArrayCapacity];
190
191 static const int ItemsPerPage = LLM_PAGE_SIZE / sizeof(T);
192 static const int DefaultCapacity = ItemsPerPage;
193};

256 // Add a value to this set.
257 // If this set already contains the value does nothing.
258 void Add(const TKey& Key, const TValue1& Value1, const TValue2& Value2)
259 {
260 SizeType KeyHash = Key.GetHashCode();
261 StripeData& Stripe = MapStripes[GetStripeIndex(KeyHash)];
262
263 UE::TUniqueLock AllocationScopeLock(Stripe.Mutex);
264
265 LLMCheck(Stripe.Map);
266
267 SizeType MapIndex = Stripe.GetMapIndex(Key, KeyHash);
268 SizeType KeyIndex = Stripe.Map[MapIndex];
269
270 if (KeyIndex != InvalidIndex)
271 {
272 static bool ShownWarning = false;
273 if (!ShownWarning)
274 {
275 FPlatformMisc::LowLevelOutputDebugString(TEXT("LLM WARNING: Replacing allocation in tracking map. Alloc/Free Mismatch.\n"));
276 ShownWarning = true;
277 }
278
279 Stripe.Values1[KeyIndex] = Value1;
280 Stripe.Values2[KeyIndex] = Value2;
281 }
282 else
283 {
284 // Be careful with the multiplication to avoid overflow, while still using only integer arithmetic for speed
285 // Margin/256U is the actual float we want to multiply by. The exact number doesn't matter, so use one order of operations
286 // when Capacity is low, and another when it is high
287 SizeType MaxCount = (Stripe.Capacity >= 256U * 256U ? (Stripe.Capacity / 256U) * Margin : (Margin * Stripe.Capacity) / 256U);
288 if (Stripe.Count >= MaxCount)
289 {
290 if (Stripe.Capacity * 2 < Stripe.Capacity)
291 {
292 // Trying to issue a check statement here will cause reentry into this function, use PLATFORM_BREAK directly instead
293 FPlatformMisc::LowLevelOutputDebugString(TEXT("LLM Error: Integer overflow in LLMap::Add, Capacity has reached its maximum size.\n"));
294 PLATFORM_BREAK();
295 }
296 if (Stripe.Count > MaxCount)
297 {
298 // Trying to issue a check statement here will cause reentry into this function, use PLATFORM_BREAK directly instead
299 // This shouldn't happen, because Count is only incremented here, and Capacity is only changed here, and Margin does not change, so Count should equal MaxCount before it goes over it
300 FPlatformMisc::LowLevelOutputDebugString(TEXT("LLM Assertion failure: Count > MaxCount.\n"));
301 PLATFORM_BREAK();
302 }
303 Stripe.Grow();
304 MapIndex = Stripe.GetMapIndex(Key, KeyHash);
305 }
306
307 if (Stripe.FreeKeyIndices.Num())
308 {
309 SizeType FreeIndex = Stripe.FreeKeyIndices.RemoveLast();
310 Stripe.Map[MapIndex] = FreeIndex;
311 Stripe.Keys[FreeIndex] = Key;
312 Stripe.KeyHashes[FreeIndex] = KeyHash;
313 Stripe.Values1[FreeIndex] = Value1;
314 Stripe.Values2[FreeIndex] = Value2;
315 }
316 else
317 {
318 Stripe.Map[MapIndex] = Stripe.Keys.Num();
319 Stripe.Keys.Add(Key);
320 Stripe.KeyHashes.Add(KeyHash);
321 Stripe.Values1.Add(Value1);
322 Stripe.Values2.Add(Value2);
323 }
324
325 ++Stripe.Count;
326 }
327 }

337 TKey Find(const TKey& Key, TValue1& OutValue1, TValue2& OutValue2) const
338 {
339 SizeType KeyHash = Key.GetHashCode();
340 const StripeData& Stripe = MapStripes[GetStripeIndex(KeyHash)];
341
342 UE::TUniqueLock AllocationScopeLock(Stripe.Mutex);
343
344 SizeType MapIndex = Stripe.GetMapIndex(Key, KeyHash);
345
346 SizeType KeyIndex = Stripe.Map[MapIndex];
347 if (KeyIndex == InvalidIndex)
348 {
349 return TKey();
350 }
351
352 OutValue1 = Stripe.Values1[KeyIndex];
353 OutValue2 = Stripe.Values2[KeyIndex];
354
355 return Stripe.Keys[KeyIndex];
356 }

358 bool Remove(const TKey& Key, Values& OutValues)
359 {
360 SizeType KeyHash = Key.GetHashCode();
361 StripeData& Stripe = MapStripes[GetStripeIndex(KeyHash)];
362
363 UE::TUniqueLock AllocationScopeLock(Stripe.Mutex);
364
365 LLMCheck(Stripe.Map);
366
367 SizeType MapIndex = Stripe.GetMapIndex(Key, KeyHash);
368 if (!Stripe.IsItemInUse(MapIndex))
369 {
370 return false;
371 }
372
373 SizeType KeyIndex = Stripe.Map[MapIndex];
374
375 OutValues.Key = Stripe.Keys[KeyIndex];
376 OutValues.Value1 = Stripe.Values1[KeyIndex];
377 OutValues.Value2 = Stripe.Values2[KeyIndex];
378
379 if (KeyIndex == Stripe.Keys.Num() - 1)
380 {
381 Stripe.Keys.RemoveLast();
382 Stripe.KeyHashes.RemoveLast();
383 Stripe.Values1.RemoveLast();
384 Stripe.Values2.RemoveLast();
385 }
386 else
387 {
388 Stripe.FreeKeyIndices.Add(KeyIndex);
389 }
390
391 // find first index in this array
392 SizeType IndexIter = MapIndex;
393 SizeType FirstIndex = MapIndex;
394 if (!IndexIter)
395 {
396 IndexIter = Stripe.Capacity;
397 }
398 --IndexIter;
399 while (Stripe.IsItemInUse(IndexIter))
400 {
401 FirstIndex = IndexIter;
402 if (!IndexIter)
403 {
404 IndexIter = Stripe.Capacity;
405 }
406 --IndexIter;
407 }
408
409 bool Found = false;
410 for (;;)
411 {
412 // find the last item in the array that can replace the item being removed
413 SizeType IndexIter2 = (MapIndex + 1) & (Stripe.Capacity - 1);
414
415 SizeType SwapIndex = InvalidIndex;
416 while (Stripe.IsItemInUse(IndexIter2))
417 {
418 SizeType SearchKeyIndex = Stripe.Map[IndexIter2];
419 const SizeType SearchHashCode = Stripe.KeyHashes[SearchKeyIndex];
420 const SizeType SearchInsertIndex = SearchHashCode & (Stripe.Capacity - 1);
421
422 if (Stripe.InRange(SearchInsertIndex, FirstIndex, MapIndex))
423 {
424 SwapIndex = IndexIter2;
425 Found = true;
426 }
427
428 IndexIter2 = (IndexIter2 + 1) & (Stripe.Capacity - 1);
429 }
430
431 // swap the item
432 if (Found)
433 {
434 Stripe.Map[MapIndex] = Stripe.Map[SwapIndex];
435 MapIndex = SwapIndex;
436 Found = false;
437 }
438 else
439 {
440 break;
441 }
442 }
443
444 // remove the last item
445 Stripe.Map[MapIndex] = InvalidIndex;
446
447 --Stripe.Count;
448
449 return true;
450 }

692 void Reserve(SizeType NewCapacity)
693 {
694 NewCapacity = FMath::Max(NewCapacity, (SizeType)1);
695 NewCapacity = static_cast<SizeType>(FPlatformMath::RoundUpToPowerOfTwo64(static_cast<uint64>(NewCapacity)));
696
697 // keep a copy of the old map
698 SizeType* OldMap = Map;
699 SizeType OldCapacity = Capacity;
700
701 LLMCheck(NewCapacity > OldCapacity);
702
703 // allocate the new table
704 Capacity = NewCapacity;
705 Map = (SizeType*)Allocator->Alloc(NewCapacity * sizeof(SizeType));
706
707 for (SizeType Index = 0; Index < NewCapacity; ++Index)
708 Map[Index] = InvalidIndex;
709
710 // copy the values from the old to the new table
711 SizeType* OldItem = OldMap;
712 for (SizeType Index = 0; Index < OldCapacity; ++Index, ++OldItem)
713 {
714 SizeType KeyIndex = *OldItem;
715
716 if (KeyIndex != InvalidIndex)
717 {
718 SizeType MapIndex = GetMapIndex(Keys[KeyIndex], KeyHashes[KeyIndex]);
719 Map[MapIndex] = KeyIndex;
720 }
721 }
722
723 Allocator->Free(OldMap, OldCapacity * sizeof(SizeType));
724 }
725
726 bool IsItemInUse(SizeType MapIndex) const
727 {
728 return Map[MapIndex] != InvalidIndex;
729 }
730
731 SizeType GetMapIndex(const TKey& Key, SizeType Hash) const
732 {
733 SizeType Mask = Capacity - 1;
734 SizeType MapIndex = Hash & Mask;
735 SizeType KeyIndex = Map[MapIndex];
736
737 while (KeyIndex != InvalidIndex && !(Keys[KeyIndex] == Key))
738 {
739 MapIndex = (MapIndex + 1) & Mask;
740 KeyIndex = Map[MapIndex];
741#ifdef PROFILE_LLMMAP
742 ++IterAcc;
743#endif
744 }
745
746#ifdef PROFILE_LLMMAP
747 ++IterCount;
748 double Average = IterAcc / (double)IterCount;
749 if (Average > 2.0)
750 {
751 static double LastWriteTime = 0.0;
752 double Now = FPlatformTime::Seconds();
753 if (Now - LastWriteTime > 5)
754 {
755 LastWriteTime = Now;
756 UE_LOGF(LogStats, Log, "WARNING: LLMMap average: %f\n", (float)Average);
757 }
758 }
759#endif
760 return MapIndex;
761 }

802 // We divide map items into multiple stripes by the high bits of the hash key, to reduce lock contention.
803 // Each stripe has its own critical section lock, and given even distribution of hash codes (due to the
804 // mix function applied to the key), multiple threads will usually end up with their own lock, and not
805 // need to wait on each other.

811// Pointer key for hash map
812struct PointerKey
813{
814 explicit PointerKey() = default;
815 explicit PointerKey(const void* InPointer, uint16 ExtraData=0)
816 : Pointer((UPTRINT(InPointer) << 16) | UPTRINT(ExtraData))
817 {
818 static_assert(sizeof(InPointer) == 8, "LLM only supports 64 bit platforms");
819 LLMEnsure(UPTRINT(InPointer) && UPTRINT(InPointer) <= 0x0000'ffff'ffff'ffff);
820 }
821 LLMNumAllocsType GetHashCode() const
822 {
823 return GetHashCodeImpl<sizeof(LLMNumAllocsType), sizeof(void*)>();
824 }
825 bool operator==(const PointerKey& other) const { return GetPointer() == other.GetPointer(); }
826 explicit operator bool () const { return !!Pointer; }
827 void* GetPointer() const { return (void*)(Pointer >> 16); };
828 uint64 GetExtraData() const { return uint64(Pointer & 0xffff); }
829
830private:
831 UPTRINT Pointer = 0;
832
833 template <int HashSize, int PointerSize>
834 LLMNumAllocsType GetHashCodeImpl() const
835 {
836 static_assert(HashSize == 0 && PointerSize == 0, "Converting void* to a LLMNumAllocsType - sized hashkey is not implemented for the current sizes.");
837 return (LLMNumAllocsType)(UPTRINT)GetPointer();

1136 template <typename T, typename GetEdgesType, typename SizeType>
1137 void TopologicalSort(T* VertexData, SizeType NumVertices, GetEdgesType GetEdges, ETopologicalSortOrder SortOrder)
1138 {
1139 if (NumVertices == 0)
1140 {
1141 return;
1142 }
1143 LLMCheckf(NumVertices > 0, TEXT("Invalid number of vertices"));
1144
1145 // In our traversal, we write vertices LeafToRoot. To make a stable sort, we need to iterate the input list from LeafToRoot as well.
1146 SizeType TraversalStart, TraversalEnd, TraversalDir;
1147 if (SortOrder == ETopologicalSortOrder::RootToLeaf)
1148 {
1149 TraversalStart = NumVertices - 1;
1150 TraversalEnd = -1;
1151 TraversalDir = -1;
1152 }
1153 else
1154 {
1155 TraversalStart = 0;
1156 TraversalEnd = NumVertices;
1157 TraversalDir = 1;
1158 }
1159
1160 TArray<SizeType, FDefaultLLMAllocator> LeafToRootOrder;
1161 LeafToRootOrder.Reserve(NumVertices);
1162
1163 {
1164 constexpr SizeType ExpectedMaxNumEdges = 16;
1165 struct FVisitData
1166 {
1167 SizeType Vertex;
1168 SizeType NextEdge;
1169 SizeType EdgeStart;
1170 };
1171
1172 TArray<FVisitData, FDefaultLLMAllocator> Stack;
1173 Stack.Reserve(NumVertices);
1174
1175 TArray<SizeType, FDefaultLLMAllocator> EdgeBuffer;
1176 EdgeBuffer.AddUninitialized(NumVertices);
1177 SizeType* EdgeBufferData = EdgeBuffer.GetData();
1178
1179 TArray<SizeType, FDefaultLLMAllocator> EdgesOnStack;
1180 EdgesOnStack.Reserve(NumVertices * ExpectedMaxNumEdges);
1181
1182 TArray<bool, FDefaultLLMAllocator> Visited;
1183 Visited.AddDefaulted(NumVertices);
1184
1185 auto PushVertex = [&EdgesOnStack, &Stack, &GetEdges, &Visited, EdgeBufferData, NumVertices](SizeType Vertex)
1186 {
1187 Visited[Vertex] = true;
1188 Stack.Add(FVisitData{ Vertex, EdgesOnStack.Num(), EdgesOnStack.Num() });
1189 SizeType NumEdges = static_cast<SizeType>(-1);
1190 GetEdges(Vertex, EdgeBufferData, NumEdges);
1191 LLMCheckf(0 <= NumEdges && NumEdges <= NumVertices, TEXT("GetEdges function passed into TopologicalSort did not write OutNumEdges, or wrote an invalid value"));
1192 EdgesOnStack.Append(EdgeBufferData, NumEdges);
1193 };
1194
1195 for (SizeType RootVertex = TraversalStart; RootVertex != TraversalEnd; RootVertex += TraversalDir)
1196 {
1197 if (Visited[RootVertex])
1198 {
1199 continue;
1200 }
1201
1202 PushVertex(RootVertex);
1203 while (Stack.Num() > 0)
1204 {
1205 FVisitData& VisitData = Stack.Last();
1206 bool bPushed = false;
1207 SizeType NumEdgesOnStack = EdgesOnStack.Num();
1208 for (; VisitData.NextEdge < NumEdgesOnStack; ++VisitData.NextEdge)
1209 {
1210 SizeType TargetVertex = EdgesOnStack[VisitData.NextEdge];
1211 if (!Visited[TargetVertex])
1212 {
1213 ++VisitData.NextEdge;
1214 PushVertex(TargetVertex);
1215 bPushed = true;
1216 break;
1217 }
1218 }
1219 if (!bPushed)
1220 {
1221 LeafToRootOrder.Add(VisitData.Vertex);
1222 EdgesOnStack.SetNum(VisitData.EdgeStart, EAllowShrinking::No);
1223 Stack.Pop(EAllowShrinking::No);
1224 }
1225 }
1226 }
1227 }
1228 LLMCheck(LeafToRootOrder.Num() == NumVertices); // This could only fail due to an internal logic error; all vertices should have been visited and added
1229
1230 TArray<T, FDefaultLLMAllocator> Original;
1231 Original.Reserve(NumVertices);
1232 for (SizeType n = 0; n < NumVertices; ++n)
1233 {
1234 Original.Add(MoveTemp(VertexData[n]));
1235 }
1236
1237 for (SizeType LeafToRootIndex = 0; LeafToRootIndex < NumVertices; ++LeafToRootIndex)
1238 {
1239 SizeType WriteIndex = TraversalStart + TraversalDir * LeafToRootIndex;
1240 SizeType ReadIndex = LeafToRootOrder[LeafToRootIndex];
1241 VertexData[WriteIndex] = MoveTemp(Original[ReadIndex]);
1242 }
1243 }
1244}
