132class FOcclusionRandomStream
133{
134 enum {NumSamples = 3571};
135public:
136
138 FOcclusionRandomStream()
139 : CurrentSample(0)
140 {
141 FRandomStream RandomStream(0x83246);
142 for (int32 Index = 0; Index < NumSamples; Index++)
143 {
144 Samples[Index] = RandomStream.GetFraction();
145 }
146 Samples[0] = 0.0f; // we want to make sure we have at least a few zeros
147 Samples[NumSamples/3] = 0.0f; // we want to make sure we have at least a few zeros
148 Samples[(NumSamples*2)/3] = 0.0f; // we want to make sure we have at least a few zeros
149 }
150
152 inline float GetFraction()
153 {
154 uint32 Current = CurrentSample.fetch_add(1);
155
156 if (Current >= NumSamples)
157 {
158 Current++;
159 CurrentSample.compare_exchange_strong(Current, 0);
160 Current = 0;
161 // It is intended here to not check if exchange worked or failed.
162 // It might be overkill to call recursively GetFraction if exchange failed.
163 // Another thread might already have reset CurrentSample and it is acceptable
164 // to have two threads returning the same Fraction at index 0
165 }
166 float Fraction = Samples[Current];
167 return Fraction;
168 }
169private:
170
172 std::atomic<uint32> CurrentSample;
174 float Samples[NumSamples];
175};
