68 // Visit each element of the subtree rooted at `this`.
69 // This visits nodes in pre-order: parents are visited before children.
70 template <typename FunctionType>
71 void ForEachInPreOrder(FunctionType&& Function)
72 {
73 if (LIKELY(!LastChild.Get()))
74 {
75 Function(*This());
76 return;
77 }
78
79 TArray<DerivedType*> ToVisit;
80 ToVisit.Push(This());
81 while (ToVisit.Num())
82 {
83 DerivedType* Derived = ToVisit.Pop();
84 Function(*Derived);
85 for (DerivedType* Child = Derived->LastChild.Get(); Child; Child = Child->Prev.Get())
86 {
87 ToVisit.Push(Child);
88 }
89 }
90 }
