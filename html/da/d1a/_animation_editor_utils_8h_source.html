154 // template version of simple creating animation asset
155 template< class T >
156 T* CreateAnimationAsset(UObject* SkeletonOrSkeletalMesh, const FString& AssetPath, const FString& InPrefix)
157 {
158 USkeletalMesh* SkeletalMesh = nullptr;
159 USkeleton* Skeleton = Cast<USkeleton>(SkeletonOrSkeletalMesh);
160 if (Skeleton == nullptr)
161 {
162 SkeletalMesh = CastChecked<USkeletalMesh>(SkeletonOrSkeletalMesh);
163 Skeleton = SkeletalMesh->GetSkeleton();
164 }
165
166 if (Skeleton)
167 {
168 FString Name;
169 FString PackageName;
170 // Determine an appropriate name
171 CreateUniqueAssetName(AssetPath, InPrefix, PackageName, Name);
172
173 // Create the asset, and assign its skeleton
174 FAssetToolsModule& AssetToolsModule = FModuleManager::GetModuleChecked<FAssetToolsModule>("AssetTools");
175 T* NewAsset = Cast<T>(AssetToolsModule.Get().CreateAsset(Name, FPackageName::GetLongPackagePath(PackageName), T::StaticClass(), NULL));
176
177 if (NewAsset)
178 {
179 NewAsset->SetSkeleton(Skeleton);
180 if (SkeletalMesh)
181 {
182 NewAsset->SetPreviewMesh(SkeletalMesh);
183 }
184 NewAsset->MarkPackageDirty();
185 }
186
187 return NewAsset;
188 }
189
190 return nullptr;
191 }

218 template <typename TFactory, typename T>
219 void ExecuteNewAnimAsset(TArray<TSoftObjectPtr<UObject>> SkeletonsOrSkeletalMeshes, const FString InSuffix, FAnimAssetCreated AssetCreated, bool bInContentBrowser, bool bAllowReplaceExisting)
220 {
221 if(bInContentBrowser && SkeletonsOrSkeletalMeshes.Num() == 1)
222 {
223 USkeletalMesh* SkeletalMesh = nullptr;
224 USkeleton* Skeleton = Cast<USkeleton>(SkeletonsOrSkeletalMeshes[0].LoadSynchronous());
225 if (Skeleton == nullptr)
226 {
227 SkeletalMesh = CastChecked<USkeletalMesh>(SkeletonsOrSkeletalMeshes[0].LoadSynchronous());
228 Skeleton = SkeletalMesh->GetSkeleton();
229 }
230
231 if(Skeleton)
232 {
233 // Determine an appropriate name for inline-rename
234 FString Name;
235 FString PackageName;
236 CreateUniqueAssetName(Skeleton->GetOutermost()->GetName(), InSuffix, PackageName, Name);
237
238 TFactory* Factory = NewObject<TFactory>();
239 Factory->TargetSkeleton = Skeleton;
240 Factory->PreviewSkeletalMesh = SkeletalMesh;
241
242 FContentBrowserModule& ContentBrowserModule = FModuleManager::LoadModuleChecked<FContentBrowserModule>("ContentBrowser");
243 ContentBrowserModule.Get().CreateNewAsset(Name, FPackageName::GetLongPackagePath(PackageName), T::StaticClass(), Factory);
244
245 if(AssetCreated.IsBound())
246 {
247 // @TODO: this doesn't work
248 //FString LongPackagePath = FPackageName::GetLongPackagePath(PackageName);
249 UObject* Parent = FindPackage(NULL, *PackageName);
250 UObject* NewAsset = FindObject<UObject>(Parent, *Name, EFindObjectFlags::None);
251 if(NewAsset)
252 {
253 TArray<UObject*> NewAssets;
254 NewAssets.Add(NewAsset);
255 if (!AssetCreated.Execute(NewAssets))
256 {
257 //Destroy the assets we just create
258 for (UObject* ObjectToDelete : NewAssets)
259 {
260 ObjectToDelete->ClearFlags(RF_Standalone | RF_Public);
261 ObjectToDelete->RemoveFromRoot();
262 ObjectToDelete->MarkAsGarbage();
263 }
264 CollectGarbage(GARBAGE_COLLECTION_KEEPFLAGS);
265 }
266 }
267 }
268 }
269 }
270 else
271 {
272 UObject* NameBaseObject = nullptr;
273 const bool bDoNotShowNameDialog = false;
274
275 CreateAnimationAssets(SkeletonsOrSkeletalMeshes, T::StaticClass(), InSuffix, AssetCreated, NameBaseObject, bDoNotShowNameDialog, bAllowReplaceExisting);
276 }
