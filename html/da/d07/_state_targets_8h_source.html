187class UGizmoTransformChangeStateTarget : public UObject, public IGizmoStateTarget
188{
189 GENERATED_BODY()
190public:
191 virtual void BeginUpdate()
192 {
193 if (TargetComponent.IsValid())
194 {
195 if (TransactionManager)
196 {
197 TransactionManager->BeginUndoTransaction(ChangeDescription);
198 }
199
200 InitialTransform = TargetComponent->GetComponentTransform();
201
202 for (TUniquePtr<IToolCommandChangeSource>& Source : DependentChangeSources)
203 {
204 Source->BeginChange();
205 }
206
207 for (IToolCommandChangeSource* Source : ExternalDependentChangeSources)
208 {
209 Source->BeginChange();
210 }
211 }
212 }
213
214 virtual void EndUpdate()
215 {
216 if (TargetComponent.IsValid())
217 {
218 FinalTransform = TargetComponent->GetComponentTransform();
219
220 if (TransactionManager)
221 {
222 TUniquePtr<FComponentWorldTransformChange> TransformChange
223 = MakeUnique<FComponentWorldTransformChange>(InitialTransform, FinalTransform);
224 TransactionManager->EmitObjectChange(TargetComponent.Get(), MoveTemp(TransformChange), ChangeDescription);
225
226 for (TUniquePtr<IToolCommandChangeSource>& Source : DependentChangeSources)
227 {
228 TUniquePtr<FToolCommandChange> Change = Source->EndChange();
229 if (Change)
230 {
231 UObject* Target = Source->GetChangeTarget();
232 TransactionManager->EmitObjectChange(Target, MoveTemp(Change), Source->GetChangeDescription());
233 }
234 }
235
236 for (IToolCommandChangeSource* Source : ExternalDependentChangeSources)
237 {
238 TUniquePtr<FToolCommandChange> Change = Source->EndChange();
239 if (Change)
240 {
241 UObject* Target = Source->GetChangeTarget();
242 TransactionManager->EmitObjectChange(Target, MoveTemp(Change), Source->GetChangeDescription());
243 }
244 }
245
246 TransactionManager->EndUndoTransaction();
247 }
248 }
249 }
250
251 virtual void CancelUpdate() override
252 {
253 if (TargetComponent.IsValid() && TransactionManager)
254 {
255 FinalTransform = InitialTransform;
256
257 for (TUniquePtr<IToolCommandChangeSource>& Source : DependentChangeSources)
258 {
259 Source->CancelChange();
260 }
261
262 for (IToolCommandChangeSource* Source : ExternalDependentChangeSources)
263 {
264 Source->CancelChange();
265 }
266
267 TransactionManager->CancelUndoTransaction();
268 }
269 }
