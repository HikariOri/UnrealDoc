32 // See FEncodedTextureDescription::GetNumSlices_WithDepth.
33 // Cubemaps = 6, Arrays = Count, Cubemaparrays = Count*6, Volume = Depth

36
37 // This is the SizeZ that gets passed to the RHI texture mip map stuff. See FStreamableTextureResource::SizeZ.
38 // It's weird because it's actually NOT used for cubemaps or cubemap arrays, however it does get saved in derived data
39 // in Texture2DMipMap, so we need to continue to pass through the previous values for cubemaps which are:

41 // *array* cubemaps = array_count * 6
42 // volume = depth
43 // arrays = array_count

45 {
46 return (bVolume || bTextureArray) ? NumSlicesWithDepth : 1;
47 }

386 // If there is a choice to be had between two formats: one with alpha and one without, this returns which
387 // one the texture expects.
388 bool GetTextureExpectsAlphaInPixelFormat(bool bTopMipAlphaDetected) const
389 {
390 // note the order of operations! ( ForceNo takes precedence )
391 if (bForceNoAlphaChannel)
392 {
393 return false;
394 }
395 if (bForceAlphaChannel)
396 {
397 return true;
398 }
399 return bTopMipAlphaDetected;
400 }
401
402 // bUnknownSourceAlphaFallback is whether we should treat the source as having alpha, in cases where the source alpha is unknown
403 void GetOutputAlphaFromKnownAlphaOrFallback(bool* bOutOutputHasAlpha, bool bUnknownSourceAlphaFallback = true) const
404 {
405 if ( bKnowAlphaTransparency )
406 {
407 *bOutOutputHasAlpha = bHasTransparentAlpha;
408 }
409 else
410 {
411 // fallback without known alpha
412 *bOutOutputHasAlpha = GetTextureExpectsAlphaInPixelFormat(bUnknownSourceAlphaFallback);
413 }
414 }
415
416 // returns false if we can't determine instead of passing through a fallback
417 bool GetOutputAlphaFromKnownAlphaOrFail(bool* bOutOutputHasAlpha) const
418 {
419 if (bKnowAlphaTransparency)
420 {
421 *bOutOutputHasAlpha = bHasTransparentAlpha;
422 return true;
423 }
424
425 if (bForceNoAlphaChannel)
426 {
427 *bOutOutputHasAlpha = false;
428 return true;
429 }
430 if (bForceAlphaChannel)
431 {
432 *bOutOutputHasAlpha = true;
433 return true;
434 }
435 return false;
436 }
