65{
66 GENERATED_BODY()
67
68 FPropertyBagContainerTypes() = default;
69
70 explicit FPropertyBagContainerTypes(EPropertyBagContainerType ContainerType)
71 {
72 if (ContainerType != EPropertyBagContainerType::None)
73 {
74 Add(ContainerType);
75 }
76 }
77
78 FPropertyBagContainerTypes(const std::initializer_list<EPropertyBagContainerType>& InTypes)
79 {
80 for (const EPropertyBagContainerType ContainerType : InTypes)
81 {
82 if (ContainerType != EPropertyBagContainerType::None)
83 {
84 Add(ContainerType);
85 }
86 }
87 }
88
89 bool Add(const EPropertyBagContainerType PropertyBagContainerType)
90 {
91 if (ensure(NumContainers < MaxNestedTypes))
92 {
93 if (PropertyBagContainerType != EPropertyBagContainerType::None)
94 {
95 Types[NumContainers] = PropertyBagContainerType;
96 NumContainers++;
97
98 return true;
99 }
100 }
101
102 return false;
103 }
104
105 void Reset()
106 {
107 for (EPropertyBagContainerType& Type : Types)
108 {
109 Type = EPropertyBagContainerType::None;
110 }
111 NumContainers = 0;
112 }
113
114 bool IsEmpty() const
115 {
116 return NumContainers == 0;
117 }
118
119 uint32 Num() const
120 {
121 return NumContainers;
122 }
123
124 bool CanAdd() const
125 {
126 return NumContainers < MaxNestedTypes;
127 }
128
129 EPropertyBagContainerType GetFirstContainerType() const
130 {
131 return NumContainers > 0 ? Types[0] : EPropertyBagContainerType::None;
132 }
133
134 EPropertyBagContainerType operator[] (int32 Index) const
135 {
136 return ensure(Index < NumContainers) ? Types[Index] : EPropertyBagContainerType::None;
137 }
138
139 UE_API EPropertyBagContainerType PopHead();
140
141 UE_API void Serialize(FArchive& Ar);
142
143 friend FArchive& operator<<(FArchive& Ar, FPropertyBagContainerTypes& ContainerTypesData)
144 {
145 ContainerTypesData.Serialize(Ar);
146 return Ar;
147 }
148
149 UE_API bool operator == (const FPropertyBagContainerTypes& Other) const;
150
151 UE_FORCEINLINE_HINT bool operator !=(const FPropertyBagContainerTypes& Other) const
152 {
153 return !(Other == *this);
154 }
155
156 friend UE_FORCEINLINE_HINT uint32 GetTypeHash(const FPropertyBagContainerTypes& PropertyBagContainerTypes)
157 {
158 return GetArrayHash(PropertyBagContainerTypes.Types.GetData(), PropertyBagContainerTypes.NumContainers);
159 }
160
161 EPropertyBagContainerType* begin() { return Types.GetData(); }
162 const EPropertyBagContainerType* begin() const { return Types.GetData(); }
163 EPropertyBagContainerType* end() { return Types.GetData() + NumContainers; }
164 const EPropertyBagContainerType* end() const { return Types.GetData() + NumContainers; }
