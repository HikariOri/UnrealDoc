123 void RecalculateCachedResults()
124 {
125 CachedResults = FPreflightResultData();
126
127 TMap<EPreflightState, int32> StateCounts;
128 TMap<EPreflightOutcome, int32> OutcomeCounts;
129 for (const FPFBatch& Batch : Batches)
130 {
131 for (const FPFStep& Step : Batch.Steps)
132 {
133 if (!Step.RetryByUser.IsEmpty())
134 {
135 // This step has been retried. Ignore it.
136 continue;
137 }
138
139 StateCounts.FindOrAdd(ParsePreflightEnum<EPreflightState>(Step.State), 0)++;
140 OutcomeCounts.FindOrAdd(ParsePreflightEnum<EPreflightOutcome>(Step.Outcome), 0)++;
141
142 if (!Step.Error.IsEmpty() && Step.Error != "None")
143 {
144 CachedResults.Errors.Add(Step.Error);
145 }
146 }
147 }
148
149 // Build the results.
150 const bool bHaveAllStepsFinished = (StateCounts.Contains(EPreflightState::Completed) || StateCounts.Contains(EPreflightState::Skipped))
151 && !StateCounts.Contains(EPreflightState::Running)
152 && !StateCounts.Contains(EPreflightState::Ready);
153 CachedResults.State = bHaveAllStepsFinished ? EPreflightState::Completed : EPreflightState::Running;
154 CachedResults.Outcome = EPreflightOutcome::Unspecified;
155
156 if (CachedResults.State == EPreflightState::Completed)
157 {
158 auto testOutcome = [this, &OutcomeCounts](EPreflightOutcome OutcomeToTest)
159 {
160 if (OutcomeCounts.Contains(OutcomeToTest) && OutcomeCounts[OutcomeToTest] > 0)
161 {
162 CachedResults.Outcome = OutcomeToTest;
163 }
164 };
165
166 // Override the outcome from the steps in this order.
167 testOutcome(EPreflightOutcome::Success);
168 testOutcome(EPreflightOutcome::Warnings);
169 testOutcome(EPreflightOutcome::Failure);
170 testOutcome(EPreflightOutcome::Unspecified);
171 }
172 }
173
174
175 bool operator==(const FPreflightData& other) const = default;
176 bool operator!=(const FPreflightData& other) const = default;
177};
