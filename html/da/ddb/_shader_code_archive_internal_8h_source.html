41 inline FIoChunkId GetShaderCodeArchiveChunkId(const FString& LibraryName, FName FormatName)
42 {
43 FString Name = FString::Printf(TEXT("%s-%s"), *LibraryName, *FormatName.ToString());
44 Name.ToLowerInline();
45 uint64 Hash = FShaderHash::HashBuffer(reinterpret_cast<const char*>(*Name), Name.Len() * sizeof(TCHAR)).Hash;
46 return CreateIoChunkId(Hash, 0, EIoChunkType::ShaderCodeLibrary);
47 }

49 template <int HashSize>
50 FIoChunkId GetShaderCodeChunkId(const uint8(&HashBytes)[HashSize])
51 {
52 static_assert(HashSize >= 11); // ensure the hash is big enough
53 uint8 Data[12];
54 FMemory::Memcpy(Data, HashBytes, 11); // 11 bytes from given hash
55 *reinterpret_cast<uint8*>(&Data[11]) = static_cast<uint8>(EIoChunkType::ShaderCode); // last byte is chunk type
56 FIoChunkId ChunkId;
57 ChunkId.Set(Data, 12);
58 return ChunkId;
59 }

61 inline FIoChunkId GetLibraryScopedShaderGroupChunkId(const FIoChunkId& ContentChunkId, const FIoChunkId& LibraryChunkId)
62 {
63 uint8 Input[24];
64 FMemory::Memcpy(Input, ContentChunkId.GetData(), 12);
65 FMemory::Memcpy(Input + 12, LibraryChunkId.GetData(), 12);
66 const uint64 Hash = FShaderHash::HashBuffer(Input, sizeof(Input)).Hash;
67 return CreateIoChunkId(Hash, 0, EIoChunkType::ShaderCode);
68 }
