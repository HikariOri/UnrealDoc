21// Describes a single field in a delegate payload schema. This is a CoreUObject-neutral
22// representation that can be constructed in Core from compile-time types and later interpreted
23// by CoreUObject to build FProperty instances and UScriptStructs.
24struct FDelegatePayloadFieldDesc
25{
26 // The name of the parameter this field corresponds to in the delegate's UFunction signature.
27 FName FieldName;
28
29 // The FProperty class name, e.g. NAME_IntProperty, NAME_StructProperty.
30 // Uses the pre-registered EName constants from UnrealNames.inl.
31 FName PropertyTypeName;
32
33 // Supplementary type identifier for types that need it:
34 // - StructProperty: struct path name (e.g. "Vector")
35 // - EnumProperty / ByteProperty with enum: enum name
36 // - Object/Class/SoftObject/etc: class name
37 // NAME_None for simple numeric/string types.
38 FName ExtendedTypeName;
39
40 // Child descriptors for container inner types:
41 // - ArrayProperty / SetProperty / OptionalProperty: 1 inner field (element type)
42 // - MapProperty: 2 inner fields (key type, value type)
43 // Empty for non-container types.
44 TArray<FDelegatePayloadFieldDesc> InnerFields;
45};
46
47// Describes the complete field layout of a delegate payload.
48struct FDelegatePayloadSchema
49{
50 TArray<FDelegatePayloadFieldDesc> Fields;
51};

54// Recursively builds a FDelegatePayloadFieldDesc for a native type, including container inner types.
55template <typename T>
56FDelegatePayloadFieldDesc MakePayloadFieldDesc(FName FieldName)
57{
58 using UnqualifiedT = std::remove_cv_t<T>;
59 using PropType = TNativeTypeToPropertyType_T<UnqualifiedT>;
60
61 FDelegatePayloadFieldDesc Desc;
62 Desc.FieldName = FieldName;
63 Desc.PropertyTypeName = TPropertyTypeToFName<PropType>::Get();
64
65 // Set ExtendedTypeName for types that need supplementary identification
66 if constexpr (UE::CUScriptStruct<UnqualifiedT>)
67 {
68 Desc.ExtendedTypeName = TBaseStructure<UnqualifiedT>::Get()->GetFName();
69 }
70 else if constexpr (TIsTEnumAsByte_V<UnqualifiedT>)
71 {
72 Desc.ExtendedTypeName = StaticEnum<typename UnqualifiedT::EnumType>()->GetFName();
73 }
74 else if constexpr (UE::CUEnum<UnqualifiedT>)
75 {
76 Desc.ExtendedTypeName = StaticEnum<UnqualifiedT>()->GetFName();
77 }
78
79 // Recurse into container inner types
80 if constexpr (std::is_same_v<PropType, FArrayProperty>)
81 {
82 Desc.InnerFields = { MakePayloadFieldDesc<typename T::ElementType>(NAME_None) };
83 }
84 else if constexpr (std::is_same_v<PropType, FSetProperty>)
85 {
86 Desc.InnerFields = { MakePayloadFieldDesc<typename T::ElementType>(NAME_None) };
87 }
88 else if constexpr (std::is_same_v<PropType, FOptionalProperty>)
89 {
90 Desc.InnerFields = { MakePayloadFieldDesc<typename T::ElementType>(NAME_None) };
91 }
92 else if constexpr (std::is_same_v<PropType, FMapProperty>)
93 {
94 Desc.InnerFields = { MakePayloadFieldDesc<typename T::KeyType>(NAME_None), MakePayloadFieldDesc<typename T::ValueType>(NAME_None) };
95 }
96
97 return Desc;
98}
