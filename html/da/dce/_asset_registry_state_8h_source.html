900inline void FAssetRegistryState::EnumerateMutableAssetsByPackageName(const FName PackageName,
901 TFunctionRef<bool(FAssetData* AssetData)> Callback) const
902{
903 using namespace UE::AssetRegistry::Private;
904
905#if !UE_ASSETREGISTRY_INDIRECT_ASSETDATA_POINTERS
906 if (const TArray<FAssetData*, TInlineAllocator<1>>* FoundAssetArray
907 = CachedAssetsByPackageName.Find(PackageName))
908 {
909 for (FAssetData* AssetData : *FoundAssetArray)
910 {
911 if (!Callback(AssetData))
912 {
913 break;
914 }
915 }
916 }
917#else
918 if (TOptional<TConstArrayView<FAssetDataPtrIndex>> AssetArray = CachedAssetsByPackageName.Find(PackageName))
919 {
920 for (FAssetDataPtrIndex Index : *AssetArray)
921 {
922 if (!Callback(CachedAssets[Index]))
923 {
924 break;
925 }
926 }
927 }
928#endif
929}
