144 void EnterUnwindRegion(FLabel OnUnwind)
145 {
146 int32 Offset = OpBytes.Num();
147
148 // Leave the region on the top of the stack. It will be re-entered when we leave the new region.
149 FUnwindRegion& Outer = UnwindStack.Top();
150 if (Outer.bHandlesUnwind)
151 {
152 UnwindEdges.Add(FUnwindEdge{Outer.Begin, Offset, CoerceOperand(Outer.OnUnwind)});
153 }
154
155 FUnwindRegion& Inner = UnwindStack.Emplace_GetRef(OnUnwind);
156 Inner.Begin = Offset;
157 Inner.bHandlesUnwind = false;
158 Inner.bLabelInvoked = false;
159 }
160
161 void NoteUnwind()
162 {
163 // The outer sentinel region never handles an unwind.
164 if (UnwindStack.Num() > 1)
165 {
166 FUnwindRegion& Region = UnwindStack.Top();
167 Region.bHandlesUnwind = true;
168 Region.bLabelInvoked = true;
169 }
170 }
171
172 // Returns true if any unwinds were recorded at this level of the stack.
173 bool LeaveUnwindRegion()
174 {
175 int32 Offset = OpBytes.Num();
176
177 FUnwindRegion Inner = UnwindStack.Pop();
178 if (Inner.bHandlesUnwind)
179 {
180 UnwindEdges.Add(FUnwindEdge{Inner.Begin, Offset, CoerceOperand(Inner.OnUnwind)});
181 }
182
183 // Re-enter the next region on the stack, but preserve its bLabelInvoked.
184 FUnwindRegion& Outer = UnwindStack.Top();
185 Outer.Begin = Offset;
186 Outer.bHandlesUnwind = false;
187
188 return Inner.bLabelInvoked;
189 }

207 template <typename Opcode, typename... ArgumentTypes>
208 int32 EmitOp(const FLocation& Location, ArgumentTypes&&... Arguments)
209 {
210 const int32 Offset = OpBytes.AddUninitialized(sizeof(Opcode));
211 Opcode* NewOp = new (std::launder(OpBytes.GetData() + Offset)) Opcode(CoerceOperand(Forward<ArgumentTypes>(Arguments))...);
212 checkSlow((BitCast<uint64>(NewOp) % alignof(Opcode)) == 0);
213 if (OpLocations.IsEmpty() || OpLocations.Top().Location != Location)
214 {
215 OpLocations.Emplace(Offset, Location);
216 }
217 if constexpr (std::is_same_v<Opcode, FOpReset>)
218 {
219 ResetOffsets.Add(Offset);
220 }
221 RegisterJumpRelocation(*NewOp);
222 MaybeNoteUnwind(*NewOp);
223 return Offset;
224 }
