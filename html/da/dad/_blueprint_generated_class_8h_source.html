124template<typename ElementType>
125class TSimpleRingBuffer
126{
127public:
128 TSimpleRingBuffer(int32 MaxItems)
129 : WriteIndex(0)
130 {
131 Storage.Empty(MaxItems);
132 }
133
134 int32 Num() const
135 {
136 return Storage.Num();
137 }
138
139 ElementType& operator()(int32 i)
140 {
141 // First element is at WriteIndex-1, second at 2, etc...
142 const int32 RingIndex = (WriteIndex - 1 - i);
143 const int32 WrappedRingIndex = (RingIndex < 0) ? (RingIndex + ArrayMax()) : RingIndex;
144
145 return Storage[WrappedRingIndex];
146 }
147
148 const ElementType& operator()(int32 i) const
149 {
150 // First element is at WriteIndex-1, second at 2, etc...
151 const int32 RingIndex = (WriteIndex - 1 - i);
152 const int32 WrappedRingIndex = (RingIndex < 0) ? (RingIndex + ArrayMax()) : RingIndex;
153
154 return Storage[WrappedRingIndex];
155 }
156
157 int32 ArrayMax() const
158 {
159 return Storage.GetSlack() + Storage.Num();
160 }
161
162 // Note: Doesn't call the constructor on objects stored in it; make sure to properly initialize the memory returned by WriteNewElementUninitialized
163 ElementType& WriteNewElementUninitialized()
164 {
165 const int32 OldWriteIndex = WriteIndex;
166 WriteIndex = (WriteIndex + 1) % ArrayMax();
167
168 if (Storage.GetSlack())
169 {
170 Storage.AddUninitialized(1);
171 }
172 return Storage[OldWriteIndex];
173 }
174
175 ElementType& WriteNewElementInitialized()
176 {
177 const int32 OldWriteIndex = WriteIndex;
178 WriteIndex = (WriteIndex + 1) % ArrayMax();
179
180 if (Storage.GetSlack())
181 {
182 Storage.Add(ElementType());
183 }
184 else
185 {
186 Storage[OldWriteIndex] = ElementType();
187 }
188 return Storage[OldWriteIndex];
189 }
190
191private:
192 TArray<ElementType> Storage;
193 int32 WriteIndex;
194};
