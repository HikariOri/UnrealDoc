26inline void CalculateUVBounds(

61template <typename T>
62inline UE::Math::TVector2<T> CalculateErrorBounds(
63 const UE::Math::TVector<T> Barycentrics[3],
64 const UE::Math::TVector<T> VertexNormals[3],
65 const FVector2f UVBounds[2],
66 const UE::Math::TVector<T> Displacements[3],
67 const FDisplacementMap& DisplacementMap,
68 const float Center,
69 const FVector2f MagnitudeBounds, // bounds
70 const int32 DisplacementBoundsRefinements)
71{
72 T MinBarycentric0 = FMath::Min3(Barycentrics[0].X, Barycentrics[1].X, Barycentrics[2].X);
73 T MaxBarycentric0 = FMath::Max3(Barycentrics[0].X, Barycentrics[1].X, Barycentrics[2].X);
74
75 T MinBarycentric1 = FMath::Min3(Barycentrics[0].Y, Barycentrics[1].Y, Barycentrics[2].Y);
76 T MaxBarycentric1 = FMath::Max3(Barycentrics[0].Y, Barycentrics[1].Y, Barycentrics[2].Y);
77
78 TAffine<T, 2> Barycentric0(MinBarycentric0, MaxBarycentric0, 0);
79 TAffine<T, 2> Barycentric1(MinBarycentric1, MaxBarycentric1, 1);
80 TAffine<T, 2> Barycentric2 = TAffine<T, 2 >(1.0) - Barycentric0 - Barycentric1;
81
82 TAffine<UE::Math::TVector<T>, 2> LerpedDisplacement;
83 LerpedDisplacement = TAffine<UE::Math::TVector<T>, 2>(Displacements[0]) * Barycentric0;
84 LerpedDisplacement += TAffine<UE::Math::TVector<T>, 2>(Displacements[1]) * Barycentric1;
85 LerpedDisplacement += TAffine<UE::Math::TVector<T>, 2>(Displacements[2]) * Barycentric2;
86
87 TAffine<UE::Math::TVector<T>, 2> Normal;
88 Normal = TAffine<UE::Math::TVector<T>, 2>(VertexNormals[0]) * Barycentric0;
89 Normal += TAffine<UE::Math::TVector<T>, 2>(VertexNormals[1]) * Barycentric1;
90 Normal += TAffine<UE::Math::TVector<T>, 2>(VertexNormals[2]) * Barycentric2;
91 Normal = Normalize(Normal);
92
93 const FVector2f DisplacementBounds = DisplacementMap.SampleHierarchical(UVBounds[0], UVBounds[1], DisplacementBoundsRefinements) - FVector2f(Center);
94
95 // Displacement can be negative, but Magnitude is always positive.
96 const FVector2f ScaledDisplacementBounds( FMath::Min(MagnitudeBounds[0]*DisplacementBounds[0], MagnitudeBounds[1]*DisplacementBounds[0]),
97 FMath::Max(MagnitudeBounds[0]*DisplacementBounds[1], MagnitudeBounds[1]*DisplacementBounds[1]));
98
99 TAffine<T, 2> Displacement( ScaledDisplacementBounds[0], ScaledDisplacementBounds[1] );
100 TAffine<T, 2> Error = (Normal * Displacement - LerpedDisplacement).SizeSquared();
101
102 return UE::Math::TVector2<T>(Error.GetMin(), Error.GetMax());
103}

116template <typename T>
117inline int32 GetNumSamples(
118 const UE::Math::TVector<T> Barycentrics[3],
119 const FVector2f VertexUVs[3],
120 const FVector2f MapRes)
121{
122 FVector2f UVs[3];
123 for (int k = 0; k < 3; k++)
124 {
125 UVs[k] = VertexUVs[0] * static_cast<float>(Barycentrics[k].X);
126 UVs[k] += VertexUVs[1] * static_cast<float>(Barycentrics[k].Y);
127 UVs[k] += VertexUVs[2] * static_cast<float>(Barycentrics[k].Z);
128
129 UVs[k].X *= MapRes[0];
130 UVs[k].Y *= MapRes[1];
131 }
132
133 FVector2f Edge01 = UVs[1] - UVs[0];
134 FVector2f Edge12 = UVs[2] - UVs[1];
135 FVector2f Edge20 = UVs[0] - UVs[2];
136
137 T MaxEdgeLength = FMath::Sqrt(FMath::Max3(
138 Edge01.SizeSquared(),
139 Edge12.SizeSquared(),
140 Edge20.SizeSquared()));
141
142 T AreaInTexels = FMath::Abs(0.5 * (Edge01 ^ Edge12));
143
144 return static_cast<int32>(FMath::CeilToInt(FMath::Max(MaxEdgeLength, AreaInTexels)));
145}
