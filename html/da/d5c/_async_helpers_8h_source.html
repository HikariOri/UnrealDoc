96 template<typename IntegerType>
97 void LockFreePeak(volatile IntegerType* PeakValue, IntegerType NewSample)
98 {
99 IntegerType CurrentPeak;
100 do
101 {
102 // Read the current value.
103 CurrentPeak = *PeakValue;
104 }
105 // If the value was lower than the sample, try to update it to NewSample if the current value has not been set by another thread.
106 // If the current value change since we read it, try again to see if our sample is still higher.
107 while (CurrentPeak < NewSample && FPlatformAtomics::InterlockedCompareExchange(PeakValue, NewSample, CurrentPeak) != CurrentPeak);
108 }

161 void ScheduleTask(TFunction<TResultType()>&& Task, FOnCompleteCb&& OnComplete = nullptr)
162 {
163 if (TaskFutures.Num() < MaxScheduledTasks)
164 {
165 // Space for new slot.
166 TaskFutures.Add(Async(ExecutionType, MoveTemp(Task)));
167 TaskCompleteFuncs.Add(MoveTemp(OnComplete));
168 }
169 else
170 {
171 // Find a finished slot, or complete the next from last search.
172 int32 NewTaskIndex = SlotSearchCount;
173 for (int32 i = 0; i < TaskFutures.Num(); i++)
174 {
175 const int32 TaskIdx = (SlotSearchCount + i) % TaskFutures.Num();
176 if (TaskFutures[TaskIdx].IsReady())
177 {
178 NewTaskIndex = TaskIdx;
179 break;
180 }
181 }
182 SlotSearchCount = (SlotSearchCount + 1) % TaskFutures.Num();
183 // Complete the slot we will be assigning to.
184 CompleteTask(*this, NewTaskIndex);
185 // Overwrite slot.
186 TaskFutures[NewTaskIndex] = Async(ExecutionType, MoveTemp(Task));
187 TaskCompleteFuncs[NewTaskIndex] = MoveTemp(OnComplete);
188 }
189 }
190
194 void WaitAll()
195 {
196 for (int32 i = 0; i < TaskFutures.Num(); i++)
197 {
198 CompleteTask(*this, i);
199 }
200 TaskFutures.Empty();
201 TaskCompleteFuncs.Empty();
202 }

204 private:

206 template<typename TTaskResultType, int32 Tasks>
207 void CompleteTask(FTaskScheduler<TTaskResultType, Tasks>& Scheduler, int32 Idx)
208 {
209 TTaskResultType Result = Scheduler.TaskFutures[Idx].Get();
210 if (Scheduler.TaskCompleteFuncs[Idx])
211 {
212 Scheduler.TaskCompleteFuncs[Idx](MoveTemp(Result));
213 }
214 }

216 template<int32 Tasks>
217 void CompleteTask(FTaskScheduler<void, Tasks>& Scheduler, int32 Idx)
218 {
219 Scheduler.TaskFutures[Idx].Wait();
220 if (Scheduler.TaskCompleteFuncs[Idx])
221 {
222 Scheduler.TaskCompleteFuncs[Idx]();
223 }
224 }
