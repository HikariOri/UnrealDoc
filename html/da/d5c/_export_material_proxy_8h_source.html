112 virtual int32 ReflectionAboutCustomWorldNormal(int32 CustomWorldNormal, int32 bNormalizeCustomWorldNormal) override
113 {
114 if (CustomWorldNormal == INDEX_NONE)
115 {
116 return INDEX_NONE;
117 }
118
119 int32 N = CustomWorldNormal;
120 int32 C = CameraVector();
121
122 if (bNormalizeCustomWorldNormal)
123 {
124 // N = N / sqrt(dot(N, N))
125 N = Compiler->Div(N, Compiler->SquareRoot(Compiler->Dot(N, N)));
126 }
127
128 // return 2 * dot(N, C) * N - C
129 return Compiler->Sub(Compiler->Mul(Compiler->Constant(2.0f), Compiler->Mul(Compiler->Dot(N, C), N)), C);
130 }

162 virtual int32 AccessCollectionParameter(UMaterialParameterCollection* ParameterCollection, int32 ParameterIndex, int32 ComponentIndex) override
163 {
164 if (!ParameterCollection || ParameterIndex == -1)
165 {
166 return INDEX_NONE;
167 }
168
169 // Collect names of all parameters
170 TArray<FName> ParameterNames;
171 ParameterCollection->GetParameterNames(ParameterNames, /*bVectorParameters=*/ false);
172 int32 NumScalarParameters = ParameterNames.Num();
173 ParameterCollection->GetParameterNames(ParameterNames, /*bVectorParameters=*/ true);
174
175 // Find a parameter corresponding to ParameterIndex/ComponentIndex pair
176 int32 Index;
177 for (Index = 0; Index < ParameterNames.Num(); Index++)
178 {
179 FGuid ParameterId = ParameterCollection->GetParameterId(ParameterNames[Index]);
180 int32 CheckParameterIndex, CheckComponentIndex;
181 ParameterCollection->GetParameterIndex(ParameterId, CheckParameterIndex, CheckComponentIndex);
182 if (CheckParameterIndex == ParameterIndex && CheckComponentIndex == ComponentIndex)
183 {
184 // Found
185 break;
186 }
187 }
188 if (Index >= ParameterNames.Num())
189 {
190 // Not found, should not happen
191 return INDEX_NONE;
192 }
193
194 // Create code for parameter
195 if (Index < NumScalarParameters)
196 {
197 const FCollectionScalarParameter* ScalarParameter = ParameterCollection->GetScalarParameterByName(ParameterNames[Index]);
198 check(ScalarParameter);
199 return Constant(ScalarParameter->DefaultValue);
200 }
201 else
202 {
203 const FCollectionVectorParameter* VectorParameter = ParameterCollection->GetVectorParameterByName(ParameterNames[Index]);
204 check(VectorParameter);
205 const FLinearColor& Color = VectorParameter->DefaultValue;
206 return Constant4(Color.R, Color.G, Color.B, Color.A);
207 }
208 }

398 {
399 case MP_EmissiveColor:

644 int32 CompileInputForCustomOutput(FMaterialCompiler* Compiler, const FMaterialCustomOutputAttributeDefintion* CustomAttribute, int32 InputIndex, uint32 ForceCastFlags) const
645 {
646 check(CustomAttribute);
647
648 UMaterialExpressionCustomOutput* Expression = GetCustomOutputExpression(CustomAttribute->FunctionName);
649 FExpressionInput* ExpressionInput = Expression ? Expression->GetInput(InputIndex) : nullptr;
650 int32 Result;
651
652 if (ExpressionInput)
653 {
654 Result = ExpressionInput->Compile(Compiler);
655 }
656 else
657 {
658 Result = CustomAttribute->CompileDefaultValue(Compiler);
659 }
660
661 if (CustomOutputToCompile == TEXT("ClearCoatBottomNormal"))
662 {
663 Result = CompileNormalEncoding(Compiler, CompileNormalTransform(Compiler, Result));
664 }
665
666 if (ForceCastFlags & MFCF_ForceCast)
667 {
668 Result = Compiler->ForceCast(Result, CustomAttribute->ValueType, ForceCastFlags);
669 }
670
671 return Result;
672 }

694 static int32 CompileNormalEncoding(FMaterialCompiler* Compiler, int32 NormalInput)
695 {
696 return Compiler->Add(
697 Compiler->Mul(NormalInput, Compiler->Constant(0.5f)), // [-1,1] * 0.5
698 Compiler->Constant(0.5f)); // [-0.5,0.5] + 0.5
699 }
700
701 static int32 CompileRefractionEncoding(FMaterialCompiler* Compiler, int32 RefractionInput)
702 {
703 // [1,Infinity] -> [1,0]
704 return Compiler->Div(
705 Compiler->Constant(1.0f),
706 Compiler->Max(Compiler->Constant(1.0f), RefractionInput));
707 }

709 static int32 CompileShadingModelEncoding(FMaterialCompiler* Compiler, int32 ShadingModelInput)
710 {
