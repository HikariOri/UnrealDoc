150 // RAII object to determine how Animation Modifier warning/errors are handled (ignore, user dialog, etc.)
151 struct FApplyModifiersScope
152 {
153 friend UAnimationModifier;
154
155 enum ESuppressionMode : uint8
156 {
157 // Do not change the error handling mode on this scope
158 // Use the mode set by parent scope or default (ShowDialog)
159 NoChange,
160 // Suppress error dialogs
161 // Suppress warnings dialogs, always apply modifiers
162 // No user interaction required
163 SuppressWarningAndError,
164 // Show error dialogs
165 // Suppress warnings dialogs, always apply modifiers
166 SuppressWarning,
167 // Show warning and error dialogs for first encounter
168 // Error dialog for each modifier class will only be showed once
169 ShowDialog,
170 // Always show the error or warning dialog
171 // Default behavior when no scope was open
172 ForceDialog,
173 // Suppress error dialogs
174 // Suppress warnings dialogs, always revert modifiers
175 // No user interaction required
176 RevertAtWarning,
177 };
178
179 FApplyModifiersScope(const FApplyModifiersScope&) = delete;
180 explicit FApplyModifiersScope(ESuppressionMode Mode = NoChange)
181 {
182 Open(Mode);
183 }
184
185 ~FApplyModifiersScope()
186 {
187 Close();
188 }
189
190 static bool IsScopePending() { return !ScopeModeStack.IsEmpty(); }
191protected:
193 static UE_API void HandleError(const UAnimationModifier* Modifier, const FText& Message, const FText& Title);
194
196 static UE_API bool HandleWarning(const UAnimationModifier* Modifier, const FText& Message, const FText& Title);
197
198private:
199 // Open a scope to control error handling when batch applying animation modifiers
200 static UE_API ESuppressionMode Open(ESuppressionMode Mode = NoChange);
