260 static int64 GetApproxRawDataArraySize(const TArray<FRawAnimSequenceTrack>& AnimData)

267 sizeof(FVector) * int64(RawTrack.PosKeys.Num()) +

531template<uint32 Alignment = DEFAULT_ALIGNMENT>
532class TMaybeMappedAllocator
533{
534public:
535 using SizeType = int32;
536
537 enum { NeedsElementType = false };
538 enum { RequireRangeCheck = true };
539
540 class ForAnyElementType
541 {
542 public:
543
545 ForAnyElementType()
546 : Data(nullptr)
547 , MappedHandle(nullptr)
548 , MappedRegion(nullptr)
549 {}
550
556 void MoveToEmpty(ForAnyElementType& Other)
557 {
558 checkSlow(this != &Other);
559
560 Reset();
561
562 Data = Other.Data;
563 Other.Data = nullptr;
564
565 MappedRegion = Other.MappedRegion;
566 Other.MappedRegion = nullptr;
567
568 MappedHandle = Other.MappedHandle;
569 Other.MappedHandle = nullptr;
570 }
571
573 ~ForAnyElementType()
574 {
575 Reset();
576 }
577
578 // FContainerAllocatorInterface
579 FScriptContainerElement* GetAllocation() const
580 {
581 return Data;
582 }
583 void ResizeAllocation(
584 SizeType CurrentNum,
585 SizeType NewMax,
586 SIZE_T NumBytesPerElement
587 )
588 {
589 if (MappedRegion || MappedHandle)
590 {
591 check(NewMax == 0); // Currently we can only support resizing of memory mapped regions to 0 size (ie delete)
592
593 delete MappedRegion;
594 delete MappedHandle;
595 MappedRegion = nullptr;
596 MappedHandle = nullptr;
597 Data = nullptr; // make sure we don't try to free this pointer
598 }
599 else if (Data || NewMax)
600 {
601 static_assert(sizeof(int32) <= sizeof(SIZE_T), "SIZE_T is expected to be larger than int32");
602
603 // Check for under/overflow
604 if (UNLIKELY(NewMax < 0 || NumBytesPerElement < 1 || NumBytesPerElement > (SIZE_T)MAX_int32))
605 {
606 UE::Animation::Private::OnInvalidMaybeMappedAllocatorNum(NewMax, NumBytesPerElement);
607 }
608
609 // Avoid calling FMemory::Realloc( nullptr, 0 ) as ANSI C mandates returning a valid pointer which is not what we want.
610 //checkSlow(((uint64)NewMax*(uint64)ElementTypeInfo.GetSize() < (uint64)INT_MAX));
611 Data = (FScriptContainerElement*)FMemory::Realloc(Data, NewMax*NumBytesPerElement, Alignment);
612 }
613 }
614 SizeType CalculateSlackReserve(SizeType NewMax, SIZE_T NumBytesPerElement) const
615 {
616 check(!MappedHandle && !MappedRegion); // this could be supported, but it probably is never what you want, so we will just assert.
617 return DefaultCalculateSlackReserve(NewMax, NumBytesPerElement, true, Alignment);
618 }
619 SizeType CalculateSlackShrink(SizeType NewMax, SizeType CurrentMax, SIZE_T NumBytesPerElement) const
620 {
621 check(!MappedHandle && !MappedRegion); // this could be supported, but it probably is never what you want, so we will just assert.
622 return DefaultCalculateSlackShrink(NewMax, CurrentMax, NumBytesPerElement, true, Alignment);
623 }
624 SizeType CalculateSlackGrow(SizeType NewMax, SizeType CurrentMax, SIZE_T NumBytesPerElement) const
625 {
626 check(!MappedHandle && !MappedRegion); // this could be supported, but it probably is never what you want, so we will just assert.
627 return DefaultCalculateSlackGrow(NewMax, CurrentMax, NumBytesPerElement, true, Alignment);
628 }
629
630 SIZE_T GetAllocatedSize(SizeType CurrentMax, SIZE_T NumBytesPerElement) const
631 {
632 return CurrentMax * NumBytesPerElement;
633 }
634
635 bool HasAllocation() const
636 {
637 return !!Data;
638 }
639
640 SizeType GetInitialCapacity() const
641 {
642 return 0;
643 }
644
645 void AcceptFileMapping(IMappedFileHandle* InMappedHandle, IMappedFileRegion* InMappedRegion, void *MallocPtr)
646 {
647 check(!MappedHandle && !Data); // we could support stuff like this, but that usually isn't what we want for streamlined loading
648 Reset(); // just in case
649 if (InMappedHandle || InMappedRegion)
650 {
651 MappedHandle = InMappedHandle;
652 MappedRegion = InMappedRegion;
653 Data = (FScriptContainerElement*)MappedRegion->GetMappedPtr(); //@todo mapped files should probably be const-correct
654 }
655 else
656 {
657 Data = (FScriptContainerElement*)MallocPtr;
658 }
659 check(IsAligned(Data, alignof(FScriptContainerElement)));
660 }
661
662 bool IsMapped() const
663 {
664 return MappedRegion || MappedHandle;
665 }
666 private:
667
668 FScriptContainerElement* Data;
669 IMappedFileHandle* MappedHandle;
670 IMappedFileRegion* MappedRegion;
671
672 void Reset()
673 {
674 if (MappedRegion || MappedHandle)
675 {
676 delete MappedRegion;
677 delete MappedHandle;
678 MappedRegion = nullptr;
679 MappedHandle = nullptr;
680 Data = nullptr; // make sure we don't try to free this pointer
681 }
682 if (Data)
683 {
684 FMemory::Free(Data);
685 Data = nullptr;
686 }
687 }
688
689
690 ForAnyElementType(const ForAnyElementType&);
691 ForAnyElementType& operator=(const ForAnyElementType&);
692 };
