53 [[nodiscard]] inline bool IsValidCodepoint(const uint32 Codepoint)
54 {
55 if ((Codepoint > 0x10FFFF) || // No Unicode codepoints above 10FFFFh, (for now!)
56 (Codepoint == 0xFFFE) || (Codepoint == 0xFFFF)) // illegal values.
57 {
58 return false;
59 }
60 return true;
61 }
62
64 [[nodiscard]] UE_FORCEINLINE_HINT bool IsHighSurrogate(const uint32 Codepoint)
65 {
66 return Codepoint >= HIGH_SURROGATE_START_CODEPOINT && Codepoint <= HIGH_SURROGATE_END_CODEPOINT;
67 }
68
70 [[nodiscard]] UE_FORCEINLINE_HINT bool IsLowSurrogate(const uint32 Codepoint)
71 {
72 return Codepoint >= LOW_SURROGATE_START_CODEPOINT && Codepoint <= LOW_SURROGATE_END_CODEPOINT;
73 }
74
75 [[nodiscard]] UE_FORCEINLINE_HINT uint32 EncodeSurrogate(const uint16 HighSurrogate, const uint16 LowSurrogate)
76 {
77 return ((HighSurrogate - HIGH_SURROGATE_START_CODEPOINT) << 10) + (LowSurrogate - LOW_SURROGATE_START_CODEPOINT) + 0x10000;
78 }
79
80 inline void DecodeSurrogate(const uint32 Codepoint, uint16& OutHighSurrogate, uint16& OutLowSurrogate)
81 {
82 const uint32 TmpCodepoint = Codepoint - 0x10000;
83 OutHighSurrogate = (uint16)((TmpCodepoint >> 10) + HIGH_SURROGATE_START_CODEPOINT);
84 OutLowSurrogate = (TmpCodepoint & 0x3FF) + LOW_SURROGATE_START_CODEPOINT;
85 }
86
88 [[nodiscard]] UE_FORCEINLINE_HINT bool IsEncodedSurrogate(const uint32 Codepoint)
89 {
90 return Codepoint >= ENCODED_SURROGATE_START_CODEPOINT && Codepoint <= ENCODED_SURROGATE_END_CODEPOINT;
91 }

94 template<typename CharType>
95 inline int32 InlineCombineSurrogates_Buffer(CharType* StrBuffer, int32 StrLen)
96 {
97 static_assert(sizeof(CharType) == 4, "CharType must be 4-bytes!");
98
99 for (int32 Index = 0; Index < StrLen; ++Index)
100 {
101 CharType Codepoint = StrBuffer[Index];
102
103 // Check if this character is a high-surrogate
104 if (StringConv::IsHighSurrogate(Codepoint))
105 {
106 // Ensure our string has enough characters to read from
107 if ((Index + 1) >= StrLen)
108 {
109 // Unpaired surrogate - replace with bogus char
110 StrBuffer[Index] = UNICODE_BOGUS_CHAR_CODEPOINT;
111 break;
112 }
113
114 const uint32 HighCodepoint = Codepoint;
115 Codepoint = StrBuffer[Index + 1];
116
117 // If our High Surrogate is set, check if this character is the matching low-surrogate
118 if (StringConv::IsLowSurrogate(Codepoint))
119 {
120 const uint32 LowCodepoint = Codepoint;
121
122 // Combine our high and low surrogates together to a single Unicode codepoint
123 Codepoint = StringConv::EncodeSurrogate(static_cast<uint16>(HighCodepoint), static_cast<uint16>(LowCodepoint));
124
125 StrBuffer[Index] = Codepoint;
126 {
127 const int32 ArrayNum = StrLen + 1;
128 const int32 IndexToRemove = Index + 1;
129 const int32 NumToMove = ArrayNum - IndexToRemove - 1;
130 if (NumToMove > 0)
131 {
132 FMemory::Memmove(StrBuffer + IndexToRemove, StrBuffer + IndexToRemove + 1, NumToMove * sizeof(CharType));
133 }
134 }
135 --StrLen;
136 continue;
137 }
138
139 // Unpaired surrogate - replace with bogus char
140 StrBuffer[Index] = UNICODE_BOGUS_CHAR_CODEPOINT;
141 }
142 else if (StringConv::IsLowSurrogate(Codepoint))
143 {
144 // Unpaired surrogate - replace with bogus char
145 StrBuffer[Index] = UNICODE_BOGUS_CHAR_CODEPOINT;
146 }
147 }
148
149 return StrLen;
150 }

223 class FTCHARToUTF8_Convert
224 {
225 // Leave this until we convert all the call sites and can change FromType to this.
226 using IntendedToType = UTF8CHAR;

240 template <typename BufferType>
241 [[nodiscard]] static int32 Utf8FromCodepoint(uint32 Codepoint, BufferType OutputIterator, uint32 OutputIteratorByteSizeRemaining)
242 {
243 // Ensure we have at least one character in size to write
244 if (OutputIteratorByteSizeRemaining < sizeof(ToType))
245 {
246 return 0;
247 }
248
249 const BufferType OutputIteratorStartPosition = OutputIterator;
250
251 if (!StringConv::IsValidCodepoint(Codepoint))
252 {
253 Codepoint = UNICODE_BOGUS_CHAR_CODEPOINT;
254 }
255 else if (StringConv::IsHighSurrogate(Codepoint) || StringConv::IsLowSurrogate(Codepoint)) // UTF-8 Characters are not allowed to encode codepoints in the surrogate pair range
256 {
257 Codepoint = UNICODE_BOGUS_CHAR_CODEPOINT;
258 }
259
260 // Do the encoding...
261 if (Codepoint < 0x80)
262 {
263 *(OutputIterator++) = (ToType)Codepoint;
264 }
265 else if (Codepoint < 0x800)
266 {
267 if (OutputIteratorByteSizeRemaining >= 2)
268 {
269 *(OutputIterator++) = (ToType)((Codepoint >> 6) | 128 | 64);
270 *(OutputIterator++) = (ToType)((Codepoint & 0x3F) | 128);
271 }
272 }
273 else if (Codepoint < 0x10000)
274 {
275 if (OutputIteratorByteSizeRemaining >= 3)
276 {
277 *(OutputIterator++) = (ToType) ((Codepoint >> 12) | 128 | 64 | 32);
278 *(OutputIterator++) = (ToType)(((Codepoint >> 6) & 0x3F) | 128);
279 *(OutputIterator++) = (ToType) ((Codepoint & 0x3F) | 128);
280 }
281 }
282 else
283 {
284 if (OutputIteratorByteSizeRemaining >= 4)
285 {
286 *(OutputIterator++) = (ToType) ((Codepoint >> 18) | 128 | 64 | 32 | 16);
287 *(OutputIterator++) = (ToType)(((Codepoint >> 12) & 0x3F) | 128);
288 *(OutputIterator++) = (ToType)(((Codepoint >> 6 ) & 0x3F) | 128);
289 *(OutputIterator++) = (ToType) ((Codepoint & 0x3F) | 128);
290 }
291 }
292
293 return UE_PTRDIFF_TO_INT32(OutputIterator - OutputIteratorStartPosition);
294 }

410template<typename InFromType, typename InToType>
411class TUTF32ToUTF16_Convert
412{
413 static_assert(sizeof(InFromType) == 4, "FromType must be 4 bytes!");
414 static_assert(sizeof(InToType) == 2, "ToType must be 2 bytes!");
415
416public:
417 typedef InFromType FromType;
418 typedef InToType ToType;
419
428 template <typename BufferType>
429 [[nodiscard]] static int32 Utf16FromCodepoint(uint32 Codepoint, BufferType OutputIterator, uint32 OutputIteratorNumRemaining)
430 {
431 // Ensure we have at least one character in size to write
432 if (OutputIteratorNumRemaining < 1)
433 {
434 return 0;
435 }
436
437 const BufferType OutputIteratorStartPosition = OutputIterator;
438
439 if (!StringConv::IsValidCodepoint(Codepoint))
440 {
441 Codepoint = UNICODE_BOGUS_CHAR_CODEPOINT;
442 }
443 else if (StringConv::IsHighSurrogate(Codepoint) || StringConv::IsLowSurrogate(Codepoint)) // Surrogate pairs range should not be present in UTF-32
444 {
445 Codepoint = UNICODE_BOGUS_CHAR_CODEPOINT;
446 }
447
448 // We want to write out two chars
449 if (StringConv::IsEncodedSurrogate(Codepoint))
450 {
451 // We need two characters to write the surrogate pair
452 if (OutputIteratorNumRemaining >= 2)
453 {
454 uint16 HighSurrogate = 0;
455 uint16 LowSurrogate = 0;
456 StringConv::DecodeSurrogate(Codepoint, HighSurrogate, LowSurrogate);
457
458 *(OutputIterator++) = (ToType)HighSurrogate;
459 *(OutputIterator++) = (ToType)LowSurrogate;
460 }
461 }
462 else if (Codepoint > ENCODED_SURROGATE_END_CODEPOINT)
463 {
464 // Ignore values higher than the supplementary plane range
465 *(OutputIterator++) = UNICODE_BOGUS_CHAR_CODEPOINT;
466 }
467 else
468 {
469 // Normal codepoint
470 *(OutputIterator++) = (ToType)Codepoint;
471 }
472
473 return UE_PTRDIFF_TO_INT32(OutputIterator - OutputIteratorStartPosition);
474 }

501 >
502 [[nodiscard]] static inline int32 ConvertedLength(const SrcBufferType* Source, int32 SourceLen)
503 {
504 UE::Core::Private::FCountingOutputIterator Dest;
505 const int32 DestLen = SourceLen * 2;
506 Convert_Impl(Dest, DestLen, (const FromType*)Source, SourceLen);

508 return Dest.GetCount();

511private:
512 template <typename DestBufferType>
513 static void Convert_Impl(DestBufferType& Dest, int32 DestLen, const FromType* Source, const int32 SourceLen)
514 {
515 for (int32 i = 0; i < SourceLen; ++i)
516 {
517 uint32 Codepoint = static_cast<uint32>(Source[i]);
518
519 if (!WriteCodepointToBuffer(Codepoint, Dest, DestLen))
520 {
521 // Could not write data, bail out
522 return;
523 }
524 }
525 }
526
527 template <typename DestBufferType>
528 static bool WriteCodepointToBuffer(const uint32 Codepoint, DestBufferType& Dest, int32& DestLen)
529 {
530 int32 WrittenChars = Utf16FromCodepoint(Codepoint, Dest, DestLen);
531 if (WrittenChars < 1)
532 {
533 return false;
534 }
535
536 Dest += WrittenChars;
537 DestLen -= WrittenChars;
538 return true;
539 }

542template<typename InFromType, typename InToType>
543class TUTF16ToUTF32_Convert
544{
545 static_assert(sizeof(InFromType) == 2, "FromType must be 2 bytes!");
546 static_assert(sizeof(InToType) == 4, "ToType must be 4 bytes!");
547
548public:
549 typedef InFromType FromType;
550 typedef InToType ToType;
551
560 template <
561 typename SrcBufferType

563 >
564 static UE_FORCEINLINE_HINT void Convert(ToType* Dest, const int32 DestLen, const SrcBufferType* Source, const int32 SourceLen)
565 {
566 Convert_Impl(Dest, DestLen, (const FromType*)Source, SourceLen);
567 }
568
576 template <
577 typename SrcBufferType
578 UE_REQUIRES(TIsCharEncodingCompatibleWith_V<SrcBufferType, FromType>)
579 >
580 [[nodiscard]] static int32 ConvertedLength(const SrcBufferType* Source, const int32 SourceLen)
581 {
582 UE::Core::Private::FCountingOutputIterator Dest;
583 Convert_Impl(Dest, MAX_int32, (const FromType*)Source, SourceLen);
584
585 return Dest.GetCount();
586 }
587
588private:
589 [[nodiscard]] static uint32 CodepointFromUtf16(const FromType*& SourceString, const uint32 SourceLengthRemaining)
590 {
591 checkSlow(SourceLengthRemaining > 0);
592
593 const FromType* CodeUnitPtr = SourceString;
594
595 uint32 Codepoint = *CodeUnitPtr;
596
597 // Check if this character is a high-surrogate
598 if (StringConv::IsHighSurrogate(Codepoint))
599 {
600 // Ensure our string has enough characters to read from
601 if (SourceLengthRemaining < 2)
602 {
603 // Skip to end and write out a single char (we always have room for at least 1 char)
604 SourceString += SourceLengthRemaining;
605 return UNICODE_BOGUS_CHAR_CODEPOINT;
606 }
607
608 const uint16 HighSurrogate = (uint16)Codepoint;
609 Codepoint = *(++CodeUnitPtr);
610
611 // If our High Surrogate is set, check if this character is the matching low-surrogate
612 if (StringConv::IsLowSurrogate(Codepoint))
613 {
614 const uint16 LowSurrogate = (uint16)Codepoint;
615
616 // Combine our high and low surrogates together to a single Unicode codepoint
617 Codepoint = StringConv::EncodeSurrogate(HighSurrogate, LowSurrogate);
618
619 SourceString += 2; // skip to next possible start of codepoint.
620 return Codepoint;
621 }
622 else
623 {
624 ++SourceString; // Did not find matching low-surrogate, write out a bogus character and continue
625 return UNICODE_BOGUS_CHAR_CODEPOINT;
626 }
627 }
628 else if (StringConv::IsLowSurrogate(Codepoint))
629 {
630 // Unpaired low surrogate
631 ++SourceString;
632 return UNICODE_BOGUS_CHAR_CODEPOINT;
633 }
634 else
635 {
636 // Single codepoint
637 ++SourceString;
638 return Codepoint;
639 }
640 }
641
645 template <typename DestBufferType>
646 static void Convert_Impl(DestBufferType& ConvertedBuffer, int32 DestLen, const FromType* Source, const int32 SourceLen)
647 {
648 const FromType* SourceEnd = Source + SourceLen;
649 while (Source < SourceEnd && DestLen > 0)
650 {
651 // Read our codepoint, advancing the source pointer
652 uint32 Codepoint = CodepointFromUtf16(Source, SourceEnd - Source);
653
654 *(ConvertedBuffer++) = Codepoint;
655 --DestLen;
656 }
657 }

982 * NOTE:
983 * The following converting types and macros should be considered
984 * deprecated and StringCast should be used instead.
985 *
986 * The reasons for deprecation include:
987 *
988 * - char is the wrong type for representing a UTF-8 encoding. UTF8CHAR should be used now, which will become char8_t from C++20 onwards.
989 * - The macros hide a reinterpret_cast of the argument, which is both misleading and dangerous.
990 * - The const return value is silently cast away, which is also misleading and dangerous.
991 * - The macros are frequently used to initialize a local variable, which ends up pointing at deleted memory.
992 * - The names are hardcoded with the types they cast from and to, making them unusable in generic contexts.
993 *
994 * Examples of migration:
995 * ANSICHAR* AnsiCharPtr = ...;
996 * UTF8CHAR* Utf8CharPtr = ...;
997 * TCHAR* TCharPtr = ...;
998 *
999 * Before:
1000 * FUTF8ToTCHAR Conv1(AnsiCharPtr); // Use Conv1.Get() and Conv1.Length()
1001 * FUTF8ToTCHAR Conv2(Utf8CharPtr); // Use Conv2.Get() and Conv2.Length()
1002 * FunctionTakingTCHARPtr(ANSI_TO_TCHAR(AnsiCharPtr));
1003 * Printf(TEXT("Narrow string: %s"), ANSI_TO_TCHAR(AnsiCharPtr));
1004 * Printf(TEXT("UTF-8 string: %s"), UTF8_TO_TCHAR(Utf8CharPtr));
1005 *
1006 * After:
1007 * auto Conv1 = StringCast<TCHAR>((const UTF8CHAR*)AnsiCharPtr); // Use Conv1.Get() and Conv1.Length()

1010 * Printf(TEXT("Narrow string: %hs"), AnsiCharPtr);
1011 * Printf(TEXT("Narrow string: %hs"), Utf8CharPtr);
