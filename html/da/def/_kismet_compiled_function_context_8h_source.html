257 void CopyAndPrependStatements(UEdGraphNode* Destination, UEdGraphNode* Source)
258 {
259 TArray<FBlueprintCompiledStatement*>* SourceStatementList = StatementsPerNode.Find(Source);
260 if (SourceStatementList)
261 {
262 // Mapping of jump targets for when kismet compile statements want to jump to other statements
263 TMap<FBlueprintCompiledStatement*, int32> JumpTargetIndexTable;
264
265 TArray<FBlueprintCompiledStatement*>& TargetStatementList = StatementsPerNode.FindOrAdd(Destination);
266
267 TargetStatementList.InsertUninitialized(0, SourceStatementList->Num());
268 for (int32 i = 0; i < SourceStatementList->Num(); ++i)
269 {
270 FBlueprintCompiledStatement* CopiedStatement = new FBlueprintCompiledStatement();
271 AllGeneratedStatements.Add(CopiedStatement);
272
273 *CopiedStatement = *((*SourceStatementList)[i]);
274
275 TargetStatementList[i] = CopiedStatement;
276
277 // If the statement is a jump target, keep a mapping of it so we can fix it up after this loop
278 if (CopiedStatement->bIsJumpTarget)
279 {
280 JumpTargetIndexTable.Add((*SourceStatementList)[i], i);
281 }
282 }
283
284 // Loop through all statements and remap the target labels to the mapped ones
285 for (int32 i = 0; i < TargetStatementList.Num(); i++)
286 {
287 FBlueprintCompiledStatement* Statement = TargetStatementList[i];
288 int32* JumpTargetIdx = JumpTargetIndexTable.Find(Statement->TargetLabel);
289 if (JumpTargetIdx)
290 {
291 Statement->TargetLabel = TargetStatementList[*JumpTargetIdx];
292 }
293 }
294 }
295 else
296 {
297 // A node that generated no code of it's own (Source) tried to inject code into (Destination).
298 // It is ok, for example: UK2Node_GetClassDefaults works like this.
299 // Moreover when KismetCompilerDebugOptions::EmitNodeComments is enabled there is always a Comment state generated anyway.
300 }
301 }

335 void InsertWireTrace(FBlueprintCompiledStatement* GotoStatement, UEdGraphPin* AssociatedExecPin)
336 {
337 // only need wire traces if we're debugging the blueprint is available (not for cooked builds)
338 if (IsDebuggingOrInstrumentationRequired() && (AssociatedExecPin != NULL))
339 {
340 UEdGraphNode* PreJumpNode = AssociatedExecPin->GetOwningNode();
341
342 TArray<FBlueprintCompiledStatement*>* NodeStatementList = StatementsPerNode.Find(PreJumpNode);
343 if (NodeStatementList != NULL)
344 {
345 // @TODO: this Find() is potentially costly (if the node initially generated a lot of statements)
346 int32 GotoIndex = NodeStatementList->Find(GotoStatement);
347 if (GotoIndex != INDEX_NONE)
348 {
349 FBlueprintCompiledStatement* PrevStatement = NULL;
350 if (GotoIndex > 0)
351 {
352 PrevStatement = (*NodeStatementList)[GotoIndex - 1];
353 }
354
355 // if a wire-trace has already been inserted for us
356 if ((PrevStatement != NULL) && PrevStatement->Type == GetWireTraceType())
357 {
358 PrevStatement->ExecContext = AssociatedExecPin;
359 }
360 else if (PrevStatement != NULL)
361 {
362 FBlueprintCompiledStatement* TraceStatement = new FBlueprintCompiledStatement();
363 TraceStatement->Type = GetWireTraceType();
364 TraceStatement->Comment = PreJumpNode->NodeComment.IsEmpty() ? PreJumpNode->GetName() : PreJumpNode->NodeComment;
365 TraceStatement->ExecContext = AssociatedExecPin;
366
367 NodeStatementList->Insert(TraceStatement, GotoIndex);
368 // AllGeneratedStatements is an unordered list, so it doesn't matter that we throw this at the end
369 AllGeneratedStatements.Add(TraceStatement);
370 }
371 }
372 }
373 }

377 UEdGraphPin* FindRequiredPinByName(const UEdGraphNode* Node, const FName PinName, EEdGraphPinDirection RequiredDirection = EGPD_MAX)
378 {
379 for (UEdGraphPin* Pin : Node->Pins)
380 {
381 if (Pin->PinName == PinName)
382 {
383 if ((Pin->Direction == RequiredDirection) || (RequiredDirection == EGPD_MAX))
384 {
385 return Pin;
386 }

388 {
389 MessageLog.Error(*FString::Printf(TEXT("Expected @@ to be an %s"), (RequiredDirection == EGPD_Output) ? TEXT("output") : TEXT("input")), Pin);
390 return nullptr;
391 }
392 }
393 }
394
395 MessageLog.Error(*FString::Printf(TEXT("Expected to find a pin named %s on @@"), *PinName.ToString()), Node);
396 return nullptr;
