467 {
468 if (ObjectsWithDependentObjectsInternalIndices.GetBit(ObjectIndex))

470 for (const FDependentObjectInfo& DependentObjectInfo : GetDependentObjectInfos(ObjectIndex))

472 Functor(DependentObjectInfo.NetRefIndex);
