36inline UMaterialCacheVirtualTextureDescriptor PackMaterialCacheTextureDescriptor(FMaterialCacheVirtualTextureAllocation* Allocation, uint32_t UVCoordinateIndex)
37{
38 checkf(UVCoordinateIndex <= 3 && UVCoordinateIndex < NANITE_MAX_UVS, TEXT("Out of bounds coordinate index, consider expanding bit-width of UVCoordinateIndex"));
39
40 UMaterialCacheVirtualTextureDescriptor Descriptor{};
41 if (!Allocation)
42 {
43 return Descriptor;
44 }
45
46 // The backing virtual texture may be null if the allocation failed
47 IAllocatedVirtualTexture* VTAllocation = Allocation->VirtualTexture;
48 if (!VTAllocation)
49 {
50 return Descriptor;
51 }
52
53 check(VTAllocation->GetWidthInBlocks() == VTAllocation->GetHeightInBlocks());
54
55 Descriptor.PageX = VTAllocation->GetVirtualPageX();
56 Descriptor.PageY = VTAllocation->GetVirtualPageY();
57 Descriptor.WidthHeightInPages = VTAllocation->GetWidthInTiles();
58 Descriptor.PageTableMipBias = FMath::FloorLog2(VTAllocation->GetVirtualTileSize());
59 Descriptor.SpaceID = VTAllocation->GetSpaceID();
60 Descriptor.MaxLevel = VTAllocation->GetMaxLevel();
61 Descriptor.PageFormat = VTAllocation->GetPageTableFormat() == EVTPageTableFormat::UInt32 ? 1 : 0;
62 Descriptor.AdaptiveBias = VTAllocation->GetAllocatedAdaptiveBias();
63 return Descriptor;
64}
