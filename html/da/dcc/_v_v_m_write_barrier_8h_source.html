23// Enumerator used to control whether a TWriteBarrier write is transactional
24// (reverts to its original value on transaction abort) or not.
25enum class EWriteMode
26{
27 // Transactional write if called from the closed, otherwise non-transactional.
28 Default,
29 // Non-transactional write regardless of AutoRTFM mode.
30 NonTransactional,
31 // Transactional write regardless of AutoRTFM mode.
32 Transactional,
33};

35// Returns true if WriteMode is EWriteMode::Transactional, or WriteMode is EWriteMode::Default
36// and is called from the closed.
37template <EWriteMode WriteMode>
38inline bool IsTransactional()
39{
40 return WriteMode == EWriteMode::Transactional || (WriteMode == EWriteMode::Default && AutoRTFM::IsClosed());
41}

43// This class is a smart pointer that runs a GC write barrier whenever the stored pointer is changed
44// Fundamental law of concurrent GC: when thoust mutated the heap thou shalt run the barrier template
45// When a VValue of heap pointer type (VCell*) is written, a barrier is run (int32s and floats don't run a barrier)
46// The barrier will inform the GC about a new edge in the heap, and GC will immediately mark the cell if a collection is ongoing
47// This is necessary because GC might have already visited the previous content of Value, and might miss the updated value in that case
48// No barrier is needed when _deleting_ heap references, therefore we don't care about the previous Value during mutation
49// It does not matter if the barrier is run before or after mutation since we unconditionally mark only the new value
50// We also run the barrier during TWriteBarrier construction since a white (= otherwise unreachable) cell might have been assigned
51// New cells are allocated black (marked, reachable) so we are not worried about those here
52template <typename T>
53struct TWriteBarrier

240 template <typename ContextType>
241 AUTORTFM_OPEN FORCENOINLINE static void RunBarrierSlow(ContextType Context, TValue Value)
242 {
243 if constexpr (bIsAux)
244 {
245 if (Value.GetPtr() && !FHeap::IsMarked(Value.GetPtr()))
246 {
247 FAccessContext(Context).RunAuxWriteBarrierNonNullDuringMarking(Value.GetPtr());
248 }
249 }
250 else if constexpr (bIsVValue)
251 {
252 if (VCell* Cell = Value.ExtractCell())
253 {
254 if (!FHeap::IsMarked(Cell))
255 {
256 // Delay construction of the context (which does the expensive TLS lookup), until we actually need the mark stack to do marking.
257 FAccessContext(Context).RunWriteBarrierNonNullDuringMarking(Cell);
258 }
259 }
260 else if (UObject* Object = Value.ExtractUObject())
261 {
262 if (UE::GC::GIsIncrementalReachabilityPending)
263 {
264 Object->VerseMarkAsReachable();
265 }
266 }
267 }
268 else
269 {
270 VCell* Cell = reinterpret_cast<VCell*>(Value);
271 if (Cell && !FHeap::IsMarked(Cell))
272 {
273 FAccessContext(Context).RunWriteBarrierNonNullDuringMarking(Cell);
274 }
275 }
276 }

281 if (FHeap::IsMarking())
282 {
283 RunBarrierSlow(Context, Value);

322// The AutoRTFM::EWriteFlags to use for a call to AutoRTFM::RecordOpenWrite()
323// when recording a change to a TWriteBarrier. The flags will have the
324// AutoRTFM::EWriteFlags::CustomRollback bit set, which will trigger a call to
325// RollbackAutoRTFMWrite() if the transaction is aborted.
326// The flags may also contain one or more AutoRTFM::EWriteFlags::CustomFlagX
327// bits which are used by RollbackAutoRTFMWrite() to use the appropriate
328// write-barrier method for the given templated type.
