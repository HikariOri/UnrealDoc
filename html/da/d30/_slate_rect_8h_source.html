329 static bool DoRectanglesIntersect( const FSlateRect& A, const FSlateRect& B )
330 {
331 // Segments A and B do not intersect when:
332 //
333 // (left) A (right)
334 // o-------------o
335 // o---o OR o---o
336 // B B
337 //
338 //
339 // We assume the A and B are well-formed rectangles.
340 // i.e. (Top,Left) is above and to the left of (Bottom,Right)
341 const bool bDoNotOverlap =
342 B.Right < A.Left || A.Right < B.Left ||
343 B.Bottom < A.Top || A.Bottom < B.Top;
344
345 return ! bDoNotOverlap;
346 }

352 }

389FSlateRect TransformRect(const TransformType& Transform, const FSlateRect& Rect)
390{
391 FVector2f TopLeftTransformed = TransformPoint(Transform, FVector2f(Rect.Left, Rect.Top));
392 FVector2f BottomRightTransformed = TransformPoint(Transform, FVector2f(Rect.Right, Rect.Bottom));
393
394 if (TopLeftTransformed.X > BottomRightTransformed.X)
395 {
396 Swap(TopLeftTransformed.X, BottomRightTransformed.X);
397 }
398 if (TopLeftTransformed.Y > BottomRightTransformed.Y)
399 {
400 Swap(TopLeftTransformed.Y, BottomRightTransformed.Y);
401 }
402 return FSlateRect(TopLeftTransformed, BottomRightTransformed);
403}
