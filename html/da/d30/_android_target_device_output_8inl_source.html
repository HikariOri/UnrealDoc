49inline uint32 FAndroidDeviceOutputReaderRunnable::Run(void)
50{
51 FString LogcatOutput;
52
53 while (StopTaskCounter.GetValue() == 0 && LogcatProcHandle.IsValid())
54 {
55 if (!FPlatformProcess::IsProcRunning(LogcatProcHandle))
56 {
57 // When user plugs out USB cable adb process stops
58 // Keep trying to restore adb connection until code that uses this object will not kill us
59 Output->Serialize(TEXT("Trying to restore connection to device..."), ELogVerbosity::Log, NAME_None);
60 FPlatformProcess::CloseProc(LogcatProcHandle);
61 if (StartLogcatProcess())
62 {
63 FPlatformProcess::Sleep(1.0f);
64 }
65 else
66 {
67 Output->Serialize(TEXT("Failed to start adb proccess"), ELogVerbosity::Log, NAME_None);
68 Stop();
69 }
70 }
71 else
72 {
73 LogcatOutput.Append(FPlatformProcess::ReadPipe(LogcatReadPipe));
74
75 if (LogcatOutput.Len() > 0)
76 {
77 TArray<FString> OutputLines;
78 LogcatOutput.ParseIntoArray(OutputLines, TEXT("\n"), false);
79
80 if (!LogcatOutput.EndsWith(TEXT("\n")))
81 {
82 // partial line at the end, do not serialize it until we receive remainder
83 LogcatOutput = OutputLines.Last();
84 OutputLines.RemoveAt(OutputLines.Num() - 1);
85 }
86 else
87 {
88 LogcatOutput.Reset();
89 }
90
91 for (int32 i = 0; i < OutputLines.Num(); ++i)
92 {
93 Output->Serialize(*OutputLines[i], ELogVerbosity::Log, NAME_None);
94 }
95 }
96
97 FPlatformProcess::Sleep(0.1f);
98 }
99 }
100
101 return 0;
102}
