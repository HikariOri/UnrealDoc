57template <typename ItemType>
58struct FTopLevenshteinResult
59{
60 ItemType Item;
61 float Score = INDEX_NONE;
62
63 bool IsSet() const { return Score != static_cast<float>(INDEX_NONE); }
64
65 void CompareAndUpdate(FStringView SearchValue, const ItemType& NewItem, FStringView NewItemValue)
66 {
67 if (SearchValue.IsEmpty() || NewItemValue.IsEmpty())
68 {
69 return;
70 }
71
72 const float WorstCase = static_cast<float>(SearchValue.Len() + NewItemValue.Len());
73 const float NormalizedDistance = 1.0f - static_cast<float>(Algo::LevenshteinDistance(SearchValue, NewItemValue)) / WorstCase;
74 if (NormalizedDistance > Score)
75 {
76 Score = NormalizedDistance;
77 Item = NewItem;
78 }
79 }
80};
