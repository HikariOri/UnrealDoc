865template<class DataLayerInstanceType, typename ...Args>
866DataLayerInstanceType* UDataLayerEditorSubsystem::CreateDataLayerInstance(AWorldDataLayers* WorldDataLayers, Args&&... InArgs)
867{
868 UDataLayerInstance* NewDataLayer = nullptr;
869
870 if (WorldDataLayers)
871 {
872 NewDataLayer = WorldDataLayers->CreateDataLayer<DataLayerInstanceType>(Forward<Args>(InArgs)...);
873 }
874
875 if (NewDataLayer != nullptr)
876 {
877 BroadcastDataLayerChanged(EDataLayerAction::Add, NewDataLayer, NAME_None);
878 }
879
880 return CastChecked<DataLayerInstanceType>(NewDataLayer);
881}
