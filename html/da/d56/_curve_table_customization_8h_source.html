121 TSharedPtr<FString> InitWidgetContent(bool bShouldModifyData)
122 {
123 TSharedPtr<FString> InitialValue;
124
125 FName RowName;
126 const FPropertyAccess::Result RowResult = RowNamePropertyHandle->GetValue( RowName );
127 RowNames.Empty();
128
129 if ( RowResult != FPropertyAccess::Fail )
130 {
132 UCurveTable* CurveTable = nullptr;
133 CurveTablePropertyHandle->GetValue((UObject*&)CurveTable);
134
135 if ( CurveTable != nullptr )
136 {
138 for ( TMap<FName, FRealCurve*>::TConstIterator Iterator(CurveTable->GetRowMap()); Iterator; ++Iterator )
139 {
141 TSharedRef<FString> RowNameItem = MakeShareable(new FString(Iterator.Key().ToString()));
142 RowNames.Add(RowNameItem);
143
145 if ( Iterator.Key() == RowName )
146 {
147 InitialValue = RowNameItem;
148 }
149 }
150 }
151
152 if (!InitialValue.IsValid() && RowResult != FPropertyAccess::MultipleValues)
153 {
154 // Only set initial value to none if it's not a multiple value case
155 InitialValue = MakeShareable(new FString(TEXT("None")));
156 }
157
158
160 if ( InitialValue.IsValid() && bShouldModifyData )
161 {
162 TArray<void*> RawData;
163
164 // This raw data access is necessary to avoid setting the value during details customization setup which would cause infinite recursion repetedly reinitializing this customization
165 RowNamePropertyHandle->AccessRawData(RawData);
166 if ( RawData.Num() == 1 )
167 {
168 FName* RawFName = (FName*)RawData[0];
169 *RawFName = **InitialValue;
170 }
171 }
172 }
173
174 return InitialValue;
175 }

247 void OnFilterTextChanged( const FText& InFilterText )
248 {
249 FString CurrentFilterText = InFilterText.ToString();
250
251 FName RowName;
252 const FPropertyAccess::Result RowResult = RowNamePropertyHandle->GetValue( RowName );
253 RowNames.Empty();
254
256 UCurveTable* CurveTable = nullptr;
257 CurveTablePropertyHandle->GetValue( ( UObject*& )CurveTable );
258
259 if( CurveTable != nullptr )
260 {
262 for( TMap<FName, FRealCurve*>::TConstIterator Iterator( CurveTable->GetRowMap() ); Iterator; ++Iterator )
263 {
265 FString RowString = Iterator.Key().ToString();
266 if( CurrentFilterText == TEXT("") || RowString.Contains(CurrentFilterText) )
267 {
268 TSharedRef<FString> RowNameItem = MakeShareable( new FString( RowString ) );
269 RowNames.Add( RowNameItem );
270 }
271 }
272 }
