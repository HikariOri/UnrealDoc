361// The purpose of a NetTraceCollector is to allow tracing of full packet content data with as low overhead as possible.
362class FNetTraceCollector
363{
364 UE_NONCOPYABLE(FNetTraceCollector)
365public:
366 // constants
367 enum { MaxNestingLevel = 32 };
368 enum { InlinedEventCount = 256 };
369 enum { InlinedObjectIdCount = 128 };
370
371 explicit FNetTraceCollector(uint32 InitialEventCount) { Events.SetNumUninitialized(InitialEventCount); Reset(); }
372 FNetTraceCollector() : FNetTraceCollector(InlinedEventCount) {};
373
374 inline void Reset();
375
376 inline FNetTracePacketContentEvent& EditEvent(uint32 Index) { return Events[Index]; }
377 inline FNetTracePacketContentEvent& EditEventNoCheck(uint32 Index) { return Events.GetData()[Index]; }
378 inline void AddReferencedObjectId(uint64 ObjectId) { ReferencedObjectIds.Add(ObjectId); }
379 inline bool CanBeginPacketContentEvent() const { return CurrentNestingLevel < (MaxNestingLevel - 1); }
380
381private:
382 friend struct FNetTrace;
383
384 // Bookkeeping
385 uint32 NestingStack[MaxNestingLevel];
386 uint32 OffsetStack[MaxNestingLevel];
387 uint32 EventCount;
388 uint32 CurrentNestingLevel : 16;
389 uint32 OffsetStackLevel : 16;
390
391 // Bunch tracking
392 uint32 LastBunchEventIndex;
393 uint32 BunchEventCount;
394 uint32 PendingBunchEventIndex;
395
396 // Events
397 typedef TArray<FNetTracePacketContentEvent, TInlineAllocator<InlinedEventCount> > FEventArray;
398 FEventArray Events;
399
400 // Object Ids referenced by events
401 TArray<uint64, TInlineAllocator<InlinedObjectIdCount>> ReferencedObjectIds;
402};
