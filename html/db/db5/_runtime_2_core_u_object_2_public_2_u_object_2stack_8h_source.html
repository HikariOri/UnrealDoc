302inline FFrame::FFrame( UObject* InObject, UFunction* InNode, void* InLocals, FFrame* InPreviousFrame, FField* InPropertyChainForCompiledIn )
303 : Node(InNode)
304 , Object(InObject)
305 , Code(InNode->Script.GetData())
306 , Locals((uint8*)InLocals)
307 , MostRecentProperty(nullptr)
308 , MostRecentPropertyAddress(nullptr)
309 , MostRecentPropertyContainer(nullptr)
310 , PreviousFrame(InPreviousFrame)
311 , OutParms(NULL)
312 , PropertyChainForCompiledIn(InPropertyChainForCompiledIn)
313 , CurrentNativeFunction(nullptr)
314 , bArrayContextFailed(false)
315 , bAbortingExecution(false)
316#if PER_FUNCTION_SCRIPT_STATS
317 , DepthCounter(0)
318#endif
319{
320#if DO_BLUEPRINT_GUARD
321 FBlueprintContextTracker::Get().ScriptStack.Push(this);
322#endif
323 PreviousTrackingFrame = PushThreadLocalTopStackFrame(this);
324
325 {
326 // we propagate bAbortingExecution to *upper* frames to avoid invoking code
327 // on top of already-aborted frames
328 if (PreviousTrackingFrame)
329 {
330 bAbortingExecution |= PreviousTrackingFrame->bAbortingExecution;
331 }
332 if (InPreviousFrame)
333 {
334 bAbortingExecution |= InPreviousFrame->bAbortingExecution;
335 }
336 }
337
338#if PER_FUNCTION_SCRIPT_STATS
339 if (InPreviousFrame)
340 {
341 DepthCounter = (InPreviousFrame->DepthCounter < MAX_uint8) ? InPreviousFrame->DepthCounter + 1 : MAX_uint8;
342 }
343#endif
344#if UE_USE_VIRTUAL_STACK_ALLOCATOR_FOR_SCRIPT_VM
345 if (InPreviousFrame == nullptr)
346 {
347 CachedThreadVirtualStackAllocator = FBlueprintContext::GetThreadSingleton()->GetVirtualStackAllocator();
348 }
349 else
350 {
351 CachedThreadVirtualStackAllocator = InPreviousFrame->CachedThreadVirtualStackAllocator;
352 }
353#endif
354}

479FORCEINLINE_DEBUGGABLE void FFrame::StepCompiledIn(void* Result, const FFieldClass* ExpectedPropertyType)
480{
481 if (Code)
482 {
483 Step(Object, Result);
484 }
485 else
486 {
487 checkSlow(ExpectedPropertyType && ExpectedPropertyType->IsChildOf(FProperty::StaticClass()));
488 checkSlow(PropertyChainForCompiledIn && PropertyChainForCompiledIn->IsA(ExpectedPropertyType));
489 FProperty* Property = (FProperty*)PropertyChainForCompiledIn;
490 PropertyChainForCompiledIn = Property->Next;
491 StepExplicitProperty(Result, Property);
492 }
493}

495template<class TProperty, typename TNativeType>
496FORCEINLINE_DEBUGGABLE TNativeType& FFrame::StepCompiledInRef(void*const TemporaryBuffer)
497{
498 MostRecentPropertyAddress = nullptr;
499 MostRecentPropertyContainer = nullptr;
500
501 if (Code)
502 {
503 Step(Object, TemporaryBuffer);
504 }
505 else
506 {
507 checkSlow(CastField<TProperty>(PropertyChainForCompiledIn) && CastField<FProperty>(PropertyChainForCompiledIn));
508 TProperty* Property = (TProperty*)PropertyChainForCompiledIn;
509 PropertyChainForCompiledIn = Property->Next;
510 StepExplicitProperty(TemporaryBuffer, Property);
511 }
512
513 return (MostRecentPropertyAddress != NULL) ? *(TNativeType*)(MostRecentPropertyAddress) : *(TNativeType*)TemporaryBuffer;
