498{
499 GENERATED_USTRUCT_BODY()
500
501
502 UPROPERTY(EditAnywhere, BlueprintReadWrite, Category=DataTableCategoryHandle)
503 TObjectPtr<const class UDataTable> DataTable = nullptr;
504
506 UPROPERTY(EditAnywhere, BlueprintReadWrite, Category=DataTableCategoryHandle)
507 FName ColumnName;
508
510 UPROPERTY(EditAnywhere, BlueprintReadWrite, Category=DataTableCategoryHandle)
511 FName RowContents;
512
514 bool IsNull() const
515 {
516 return DataTable == nullptr && ColumnName == NAME_None && RowContents == NAME_None;
517 }
518
520 template <class T>
521 void GetRows(TArray<T*>& OutRows, const FString& ContextString) const
522 {
523 OutRows.Empty();
524 if (DataTable == nullptr)
525 {
526 if (RowContents != NAME_None)
527 {
528 UE_LOGF(LogDataTable, Warning, "FDataTableCategoryHandle::GetRows : No DataTable for row %ls (%ls).", *RowContents.ToString(), *ContextString);
529 }
530
531 return;
532 }
533
534 if (ColumnName == NAME_None)
535 {
536 if (RowContents != NAME_None)
537 {
538 UE_LOGF(LogDataTable, Warning, "FDataTableCategoryHandle::GetRows : No Column selected for row %ls (%ls).", *RowContents.ToString(), *ContextString);
539 }
540
541 return;
542 }
543
544 // Find the property that matches the desired column (ColumnName)
545 FProperty* Property = DataTable->FindTableProperty(ColumnName);
546 if (Property == nullptr)
547 {
548 return;
549 }
550
551 // check each row to see if the value in the Property element is the one we're looking for (RowContents). If it is, add the row to OutRows
552 uint8* RowContentsAsBinary = (uint8*)FMemory_Alloca(Property->GetSize());
553 Property->InitializeValue(RowContentsAsBinary);
554 if (Property->ImportText_Direct(*RowContents.ToString(), RowContentsAsBinary, nullptr, PPF_None) == nullptr)
555 {
556 Property->DestroyValue(RowContentsAsBinary);
557 return;
558 }
559
560 for (auto RowIt = DataTable->GetRowMap().CreateConstIterator(); RowIt; ++RowIt)
561 {
562 uint8* RowData = RowIt.Value();
563
564 if (Property->Identical(Property->ContainerPtrToValuePtr<void>(RowData, 0), RowContentsAsBinary, PPF_None))
565 {
566 OutRows.Add((T*)RowData);
567 }
568 }
569 Property->DestroyValue(RowContentsAsBinary);
570
571 return;
572 }
