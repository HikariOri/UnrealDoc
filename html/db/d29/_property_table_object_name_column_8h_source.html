151 FString GetObjectNameAsString( const TSharedRef< IPropertyTableRow >& Row )
152 {
153 FString ObjectName;
154 FString Suffix = GetTable()->GetRootPath()->ExtendPath( Row->GetPartialPath() )->ToString();
155 if ( !Suffix.IsEmpty() )
156 {
157 Suffix = FString( TEXT("->") ) + Suffix;
158 }
159
160 const UObject* const Object = GetTable()->GetObjectPropertyNode( SharedThis( this ), Row )->GetUObject( 0 );
161 if ( Object != NULL )
162 {
163 UClass* ObjectClass = Object->GetClass();
164 const UBlueprintGeneratedClass* const BlueprintGeneratedClass = Cast<UBlueprintGeneratedClass>( ObjectClass );
165 const bool IsCDO = Object->HasAnyFlags(RF_ClassDefaultObject);
166 if ( BlueprintGeneratedClass != NULL && IsCDO )
167 {
168 ObjectName = BlueprintGeneratedClass->ClassGeneratedBy->GetFName().ToString();
169 }
170 else if ( Object->IsA( AActor::StaticClass() ) )
171 {
172 ObjectName = Cast<const AActor >( Object )->GetActorLabel();
173 }
174 else
175 {
176 ObjectName = Object->GetFName().ToString();
177 }
178 }
179
180 return ObjectName + Suffix;
181 }
