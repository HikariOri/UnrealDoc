32struct FFrameTrackSampleRef
33{
34 TSharedPtr<FFrameTrackSeries> Series;
35 TSharedPtr<FFrameTrackSample> Sample;
36
37 FFrameTrackSampleRef()
38 : Series(), Sample()
39 {
40 }
41
42 FFrameTrackSampleRef(TSharedPtr<FFrameTrackSeries> InSeries, TSharedPtr<FFrameTrackSample> InSample)
43 : Series(InSeries), Sample(InSample)
44 {
45 }
46
47 FFrameTrackSampleRef(const FFrameTrackSampleRef& Other)
48 : Series(Other.Series), Sample(Other.Sample)
49 {
50 }
51
52 FFrameTrackSampleRef& operator=(const FFrameTrackSampleRef& Other)
53 {
54 Series = Other.Series;
55 Sample = Other.Sample;
56 return *this;
57 }
58
59 void Reset()
60 {
61 Series.Reset();
62 Sample.Reset();
63 }
64
65 bool IsValid() const
66 {
67 return Series.IsValid() && Sample.IsValid();
68 }
69
70 bool Equals(const FFrameTrackSampleRef& Other) const
71 {
72 return Series == Other.Series
73 && ((Sample == Other.Sample) || (Sample.IsValid() && Other.Sample.IsValid() && Sample->Equals(*Other.Sample)));
74 }
75
76 static bool AreEquals(const FFrameTrackSampleRef& A, const FFrameTrackSampleRef& B)
77 {
78 return A.Equals(B);
79 }
