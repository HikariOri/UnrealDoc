672// Find string in string, case sensitive, requires non-alphanumeric lead-in.
673//
674template <typename T>
675const typename TCString<T>::CharType* TCString<T>::Strfind(const CharType* Str, const CharType* Find, bool bSkipQuotedChars)
676{
677 if (Find == NULL || Str == NULL)
678 {
679 return NULL;
680 }
681
682 bool Alnum = 0;
683 CharType f = *Find;
684 int32 Length = Strlen(Find++) - 1;
685 CharType c = *Str++;
686 if (bSkipQuotedChars)
687 {
688 bool bInQuotedStr = false;
689 while (c)
690 {
691 if (!bInQuotedStr && !Alnum && c == f && !Strncmp(Str, Find, Length))
692 {
693 return Str - 1;
694 }
695 Alnum = (c >= LITERAL(CharType, 'A') && c <= LITERAL(CharType, 'Z')) ||
696 (c >= LITERAL(CharType, 'a') && c <= LITERAL(CharType, 'z')) ||
697 (c >= LITERAL(CharType, '0') && c <= LITERAL(CharType, '9'));
698 if (c == LITERAL(CharType, '"'))
699 {
700 bInQuotedStr = !bInQuotedStr;
701 }
702 c = *Str++;
703 }
704 }
705 else
706 {
707 while (c)
708 {
709 if (!Alnum && c == f && !Strncmp(Str, Find, Length))
710 {
711 return Str - 1;
712 }
713 Alnum = (c >= LITERAL(CharType, 'A') && c <= LITERAL(CharType, 'Z')) ||
714 (c >= LITERAL(CharType, 'a') && c <= LITERAL(CharType, 'z')) ||
715 (c >= LITERAL(CharType, '0') && c <= LITERAL(CharType, '9'));
716 c = *Str++;
717 }
718 }
719 return NULL;
720}

722//
723// Find string in string, case insensitive, requires non-alphanumeric lead-in.
724//
725template <typename T>
726const typename TCString<T>::CharType* TCString<T>::Strifind( const CharType* Str, const CharType* Find, bool bSkipQuotedChars )
727{
728 if( Find == NULL || Str == NULL )
729 {
730 return NULL;
731 }
732
733 bool Alnum = 0;
734 CharType f = ( *Find < LITERAL(CharType, 'a') || *Find > LITERAL(CharType, 'z') ) ? (*Find) : (CharType)(*Find + LITERAL(CharType,'A') - LITERAL(CharType,'a'));
735 int32 Length = Strlen(Find++)-1;
736 CharType c = *Str++;
737
738 if (bSkipQuotedChars)
739 {
740 bool bInQuotedStr = false;
741 while( c )
742 {
743 if( c >= LITERAL(CharType, 'a') && c <= LITERAL(CharType, 'z') )
744 {
745 c += LITERAL(CharType, 'A') - LITERAL(CharType, 'a');
746 }
747 if( !bInQuotedStr && !Alnum && c==f && !Strnicmp(Str,Find,Length) )
748 {
749 return Str-1;
750 }
751 Alnum = (c>=LITERAL(CharType,'A') && c<=LITERAL(CharType,'Z')) ||
752 (c>=LITERAL(CharType,'0') && c<=LITERAL(CharType,'9'));
753 if (c == LITERAL(CharType, '"'))
754 {
755 bInQuotedStr = !bInQuotedStr;
756 }
757 c = *Str++;
758 }
759 }
760 else
761 {
762 while( c )
763 {
764 if( c >= LITERAL(CharType, 'a') && c <= LITERAL(CharType, 'z') )
765 {
766 c += LITERAL(CharType, 'A') - LITERAL(CharType, 'a');
767 }
768 if( !Alnum && c==f && !Strnicmp(Str,Find,Length) )
769 {
770 return Str-1;
771 }
772 Alnum = (c>=LITERAL(CharType,'A') && c<=LITERAL(CharType,'Z')) ||
773 (c>=LITERAL(CharType,'0') && c<=LITERAL(CharType,'9'));
774 c = *Str++;
775 }
776 }
777 return NULL;

780//
781// Finds string in string, case insensitive, requires the string be surrounded by one the specified
782// delimiters, or the start or end of the string.
783//
784template <typename T>
785const typename TCString<T>::CharType* TCString<T>::StrfindDelim(const CharType* Str, const CharType* Find, const CharType* Delim)
786{
787 if( Find == NULL || Str == NULL )
788 {
789 return NULL;
790 }
791
792 int32 Length = Strlen(Find);
793 const T* Found = Stristr(Str, Find);
794 if( Found )
795 {
796 // check if this occurrence is delimited correctly
797 if( (Found==Str || Strchr(Delim, Found[-1])!=NULL) && // either first char, or following a delim
798 (Found[Length]==LITERAL(CharType,'\0') || Strchr(Delim, Found[Length])!=NULL) ) // either last or with a delim following
799 {
800 return Found;
801 }
802
803 // start searching again after the first matched character
804 for(;;)
805 {
806 Str = Found+1;
807 Found = Stristr(Str, Find);
808
809 if( Found == NULL )
810 {
811 return NULL;
812 }
813
814 // check if the next occurrence is delimited correctly
815 if( (Strchr(Delim, Found[-1])!=NULL) && // match is following a delim
816 (Found[Length]==LITERAL(CharType,'\0') || Strchr(Delim, Found[Length])!=NULL) ) // either last or with a delim following
817 {
818 return Found;
819 }
820 }
821 }
822
823 return NULL;
824}

832template <typename T>
833const typename TCString<T>::CharType* TCString<T>::Stristr(const CharType* Str, const CharType* Find)
834{
835 // Note this implementation is not reducible to Strnistr because we need to avoid calling strlen(Str) to
836 // avoid scanning it twice
837
838 // both strings must be valid
839 if( Find == nullptr || Str == nullptr )
840 {
841 return nullptr;
842 }
843
844 // get upper-case first letter of the find string (to reduce the number of full strnicmps)
845 CharType FindInitial = TChar<CharType>::ToUpper(*Find);
846 if (!FindInitial)
847 {
848 // When searching for the empty string, always return index of the first element of Str even if Str is empty.
849 return Str;
850 }
851 // get length of find string, and increment past first letter
852 int32 Length = Strlen(Find++) - 1;
853 // get the first letter of the search string, and increment past it
854 CharType StrChar = *Str++;
855 // while we aren't at end of string...
856 while (StrChar)
857 {
858 // make sure it's upper-case
859 StrChar = TChar<CharType>::ToUpper(StrChar);
860 // if it matches the first letter of the find string, do a case-insensitive string compare for the length of the find string
861 if (StrChar == FindInitial && !Strnicmp(Str, Find, Length))
862 {
863 // if we found the string, then return a pointer to the beginning of it in the search string
864 return Str-1;
865 }
866 // go to next letter
867 StrChar = *Str++;
868 }
869
870 // if nothing was found, return nullptr
871 return nullptr;
872}

874template <typename T>
875const typename TCString<T>::CharType* TCString<T>::Strnistr(const CharType* Str, int32 InStrLen, const CharType* Find, int32 FindLen)
876{
877 if (FindLen <= 0)
878 {
879 checkf(FindLen >= 0, TEXT("Invalid FindLen: %d"), FindLen);
880 return Str;
881 }
882 if (InStrLen < FindLen)
883 {
884 checkf(InStrLen >= 0, TEXT("Invalid InStrLen: %d"), InStrLen);
885 return nullptr;
886 }
887 if (Find == nullptr || Str == nullptr)
888 {
889 return nullptr;
890 }
891
892 // get upper-case first letter of the find string (to reduce the number of full strnicmps)
893 CharType FindInitial = TChar<CharType>::ToUpper(*Find);
894 // Set FindSuffix,FindSuffixLength to the characters of Find after the first letter
895 int32 FindSuffixLength = FindLen - 1;
896 const CharType* FindSuffix = Find + 1;
897
898 // while the length of the remaining string is >= FindLen
899 const CharType* StrLastChance = Str + InStrLen - FindLen;
900 while (Str <= StrLastChance)
901 {
902 CharType StrChar = *Str++;
903
904 // make sure it's upper-case
905 StrChar = TChar<CharType>::ToUpper(StrChar);
906 // if it matches the first letter of the find string, do a case-insensitive string compare for the length of the find string
907 if (StrChar == FindInitial && !Strnicmp(Str, FindSuffix, FindSuffixLength))
908 {
909 // if we found the string, then return a pointer to the beginning of it in the search string
910 return Str - 1;
911 }
912 }
913
914 // if nothing was found, return nullptr
915 return nullptr;
916}

918template <typename T>
919const typename TCString<T>::CharType* TCString<T>::Strnstr(const CharType* Str, int32 InStrLen, const CharType* Find, int32 FindLen)
920{
921 if (FindLen <= 0)
922 {
923 checkf(FindLen >= 0, TEXT("Invalid FindLen: %d"), FindLen);
924 return Str;
925 }
926 if (InStrLen < FindLen)
927 {
928 checkf(InStrLen >= 0, TEXT("Invalid InStrLen: %d"), InStrLen);
929 return nullptr;
930 }
931 if (Find == nullptr || Str == nullptr)
932 {
933 return nullptr;
934 }
935
936 // get first letter of the find string (to reduce the number of full strncmps)
937 const CharType FindInitial = *Find;
938 // Set FindSuffix,FindSuffixLength to the characters of Find after the first letter
939 const int32 FindSuffixLength = FindLen - 1;
940 const CharType* FindSuffix = Find + 1;
941
942 // while the length of the remaining string is >= FindLen
943 const CharType* StrLastChance = Str + InStrLen - FindLen;
944
945 if constexpr (sizeof(CharType) >= 1 && sizeof(CharType) <= 2)
946 {
947 constexpr uint64 SignBit = sizeof(CharType) == 1 ? 0x80ull : 0x8000ull;
948 constexpr uint64 Mask = sizeof(CharType) == 1 ? 0xFFull : 0xFFFFull;
949 constexpr uint64 Ones64 = sizeof(CharType) == 1 ? 0x0101010101010101ull : 0x0001000100010001ull;
950 constexpr uint64 Positive64 = sizeof(CharType) == 1 ? 0x7F7F7F7F7F7F7F7Full : 0x7FFF7FFF7FFF7FFFull;
951 constexpr uint64 SignBit64 = sizeof(CharType) == 1 ? 0x8080808080808080ull : 0x8000800080008000ull;
952 constexpr uint64 CharPer64 = sizeof(uint64) / sizeof(CharType);
953 constexpr uint64 BitPerChar = sizeof(CharType) == 1 ? 8 : 16;
954
955 InStrLen -= FindSuffixLength;
956 if (InStrLen >= CharPer64)
957 {
958 // Process normally if we're unaligned
959 for (; (UPTRINT(Str) & (BitPerChar-1)) && InStrLen > 1; --InStrLen)
960 {
961 CharType StrChar = *Str++;
962
963 // if it matches the first letter of the find string, do a string compare for the length of the find string
964 if (StrChar == FindInitial && !TCString<CharType>::Strncmp(Str, FindSuffix, FindSuffixLength))
965 {
966 // if we found the string, then return a pointer to the beginning of it in the search string
967 return Str - 1;
968 }
969 }
970
971 // Broadcast the initial value in each slots.
972 const uint64 FindInitial64 = (uint64)FindInitial * Ones64;
973
974 for (; InStrLen >= (int32)CharPer64; Str += CharPer64, InStrLen -= (int32)CharPer64)
975 {
976 uint64 StrValue = *(const uint64*)Str;
977 // We want to test each slots perfect match with the initial character
978 // The trick is to avoid overflowing a slot into another one.
979 // So we XOR and NEGATE so we end up with 0xFF when the character match.
980 // Then we remove the sign bit with 0x7F and ADD one, which will end up with 0x80
981 // if all the bits are set. We also AND with 0x80 to test the sign bit separately.
982 // And then we AND those two things together that will end up with 0x80 for each
983 // slot where there is a match.
984 StrValue = ~(StrValue ^ FindInitial64);
985 StrValue = ((StrValue & Positive64) + Ones64) & (StrValue & SignBit64);
986
987 // If any slot has a match, begin analysis
988 if (StrValue)
989 {
990 for (const CharType* InnerStr = Str; StrValue; StrValue >>= BitPerChar, ++InnerStr)
991 {
992 if ((StrValue & Mask) == SignBit && !TCString<CharType>::Strncmp(InnerStr + 1, FindSuffix, FindSuffixLength))
993 {
994 return InnerStr;
995 }
996 }
997 }
998 }
999 }
1000 }
1001
1002 while (Str <= StrLastChance)
1003 {
1004 CharType StrChar = *Str++;
1005
1006 // if it matches the first letter of the find string, do a string compare for the length of the find string
1007 if (StrChar == FindInitial && !Strncmp(Str, FindSuffix, FindSuffixLength))
1008 {
1009 // if we found the string, then return a pointer to the beginning of it in the search string
1010 return Str - 1;
1011 }
1012 }
1013
1014 // if nothing was found, return nullptr
1015 return nullptr;
1016}
