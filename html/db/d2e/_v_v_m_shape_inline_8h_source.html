67inline bool VShape::VEntry::operator==(const VShape::VEntry& Other) const
68{
69 if (Type != Other.Type)
70 {
71 return false;
72 }
73 switch (Type)
74 {
75 case EFieldType::Offset:
76 return Index == Other.Index;
77 case EFieldType::FProperty:
78 case EFieldType::FPropertyVar:
79 case EFieldType::FVerseProperty:

82 {
83 ECompares Cmp = VValue::Equal(FAllocationContext(FRunningContextPromise()), Value.Get(), Other.Value.Get(),
84 [](VValue Left, VValue Right) {
85 checkSlow(!Left.IsPlaceholder());
86 checkSlow(!Right.IsPlaceholder());
87 });
88 return Cmp == ECompares::Eq;
89 }

133inline uint32 GetTypeHash(const VShape::VEntry& Field)
134{
135 switch (Field.Type)
136 {
137 case Verse::EFieldType::Offset:
138 return HashCombineFast(::GetTypeHash(static_cast<int8>(Field.Type)), ::GetTypeHash(Field.Index));
139 case Verse::EFieldType::Constant:
140 return HashCombineFast(::GetTypeHash(static_cast<int8>(Field.Type)), GetTypeHash(Field.Value.Get()));
141 case Verse::EFieldType::FProperty:
142 case Verse::EFieldType::FPropertyVar:
143 case Verse::EFieldType::FVerseProperty:
144 default:
145 VERSE_UNREACHABLE();
146 }
