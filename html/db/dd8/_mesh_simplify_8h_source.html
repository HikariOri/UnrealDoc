15FORCEINLINE uint32 HashPosition( const FVector3f& Position )
16{
17 union { float f; uint32 i; } x;
18 union { float f; uint32 i; } y;
19 union { float f; uint32 i; } z;
20
21 x.f = Position.X;
22 y.f = Position.Y;
23 z.f = Position.Z;
24
25 return Murmur32( {
26 Position.X == 0.0f ? 0u : x.i,
27 Position.Y == 0.0f ? 0u : y.i,
28 Position.Z == 0.0f ? 0u : z.i
29 } );
30}

32FORCEINLINE uint32 Cycle3( uint32 Value )
33{
34 uint32 ValueMod3 = Value % 3;
35 uint32 Value1Mod3 = ( 1 << ValueMod3 ) & 3;
36 return Value - ValueMod3 + Value1Mod3;
37}

40{
