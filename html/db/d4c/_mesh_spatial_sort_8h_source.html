129 {
130 MeshBounds.Reset();
131
132 if (MeshBoundsIn.Num() == NumInputMeshes)
133 {
134 MeshBounds.Append(MeshBoundsIn);
135 }
136 else
137 {
138 InitBounds(GetInputMesh, NumInputMeshes);
139 }
140
141 FAxisAlignedBox3d CombinedBounds;
142 for (const FAxisAlignedBox3d& Bounds : MeshBounds)
143 {
144 CombinedBounds.Contain(Bounds);
145 }
146
147 Nests.Empty();
148 SkippedMeshIndices.Empty();
149
150 int32 N = NumInputMeshes;
151 TArray<int> MeshIndices;
152 MeshIndices.SetNum(N);
153 for (int i = 0; i < N; i++)
154 {
155 MeshIndices[i] = i;
156 }
157
158 // precomputed useful stats on input meshes
159 struct FMeshInfo
160 {
161 double Volume;
162 bool bPositiveVolume;
163 };
164 TArray<FMeshInfo> Info;
165 Info.SetNum(N);
166 double DimScaleFactor = 1.0 / (CombinedBounds.MaxDim() + KINDA_SMALL_NUMBER);
167 for (int i = 0; i < N; i++)
168 {
169 double SignedVolume = TMeshQueries<TriangleMeshType>::GetVolumeNonWatertight(*GetInputMesh(i), DimScaleFactor);
170 Info[i].bPositiveVolume = SignedVolume >= 0;
171 Info[i].Volume = FMathd::Abs(SignedVolume);
172 }
173
174 MeshIndices.Sort([&Info](int Ind1, int Ind2)
175 {
176 return Info[Ind1].Volume > Info[Ind2].Volume;
177 }
178 );
179
180 TArray<int> Parent;
181 Parent.Init(-1, N);
182
183 // From largest mesh to smallest, greedily assign parents where possible
184 // (If not nesting in largest:
185 // If a mesh already has a parent, but a smaller containing parent is found, switch to the new parent)
186 for (int ParentCand = 0; ParentCand + 1 < N; ParentCand++)
187 {
188 int PIdx = MeshIndices[ParentCand];
189 const FMeshInfo& PInfo = Info[PIdx];
190 const TriangleMeshType& PMesh = *GetInputMesh(PIdx);
191 if (NestingMethod == ENestingMethod::InLargestParent && Parent[ParentCand] != -1)
192 {
193 continue; // mesh was already contained in a larger mesh
194 }
195
196 if (!PInfo.bPositiveVolume && NestingMethod != ENestingMethod::InSmallestParent && bOnlyParentPostiveVolumes)
197 {
198 // negative volume meshes can't be an 'outer shell,' so we can skip unless we're looking for nests-in-nests
199 continue;
200 }
201
202 double WindingTestSign = PInfo.bPositiveVolume ? 1 : -1;
203 TMeshAABBTree3<TriangleMeshType> Spatial(GetInputMesh(PIdx));
204 TFastWindingTree<TriangleMeshType> Winding(&Spatial);
205 for (int ChildCand = ParentCand + 1; ChildCand < N; ChildCand++)
206 {
207 int CIdx = MeshIndices[ChildCand];
208 const FMeshInfo& CInfo = Info[CIdx];
209 const TriangleMeshType& CMesh = *GetInputMesh(CIdx);
210 if (CMesh.VertexCount() == 0)
211 {
212 continue;
213 }
214 if (NestingMethod == ENestingMethod::InLargestParent)
215 {
216 // InLargestParent nesting can't have recursive nests or reassign to smaller parents, so we can skip some tests
217 if ((bOnlyNestNegativeVolumes && CInfo.bPositiveVolume) || Parent[ChildCand] != -1)
218 {
219 continue;
220 }
221 }
222 if (bOnlyCheckSingleVertexForNesting)
223 {
224 FVector3d Vert;
225
226 for (int VID = 0; VID < CMesh.MaxVertexID(); VID++)
227 {
228 if (CMesh.IsVertex(VID))
229 {
230 Vert = CMesh.GetVertex(VID);
231 break;
232 }
233 }
234
235 double WindingNumber = Winding.FastWindingNumber(Vert);
236 if (WindingNumber*WindingTestSign > .5)
237 {
238 Parent[ChildCand] = ParentCand;
239 }
240 }
241 else
242 {
243 bool bHasOutsideSample = false;
244 for (int VID = 0; VID < CMesh.MaxVertexID(); VID++)
245 {
246 if (CMesh.IsVertex(VID))
247 {
248 double WindingNumber = Winding.FastWindingNumber(CMesh.GetVertex(VID));
249 if (WindingNumber * WindingTestSign < .5)
250 {
251 bHasOutsideSample = true;
252 break;
253 }
254 }
255 }
256 if (!bHasOutsideSample)
257 {
258 Parent[ChildCand] = ParentCand;
259 }
260 }
261 }
262 }
263
264 // Build nests from parent relationship
265 TArray<bool> Taken;
266 Taken.SetNumZeroed(N);
267 for (int NestCand = 0; NestCand < N; NestCand++)
268 {
269 int NIdx = MeshIndices[NestCand];
270 if (Taken[NestCand])
271 {
272 continue;
273 }
274 if (bOnlyParentPostiveVolumes && !Info[NIdx].bPositiveVolume)
275 {
276 SkippedMeshIndices.Add(NIdx);
277 continue;
278 }
279 FMeshNesting& Nest = Nests.Emplace_GetRef();
280 Nest.OuterIndex = NIdx;
281
282 if (NestingMethod == ENestingMethod::InSmallestParent)
283 {
284 int ParentCand = Parent[NestCand];
285 if (ParentCand > -1)
286 {
287 Nest.ParentIndex = MeshIndices[ParentCand];
288 }
289 }
290 for (int HoleCand = NestCand + 1; HoleCand < N; HoleCand++)
291 {
292 int MIdx = MeshIndices[HoleCand];
293 if (Parent[HoleCand] == NestCand && (!Info[MIdx].bPositiveVolume || !bOnlyNestNegativeVolumes))
294 {
295 Nest.InnerIndices.Add(MIdx);
296 Taken[HoleCand] = true;
297 }
298 }
299 }
300 }
