48void FTableMergeTask::FillResultUsing(IUntypedTableReader& SourceTableReader) const
49{
50 const uint64 ColumnsMapCount = static_cast<uint64>(ColumnsMap.Num());
51
52 for (SourceTableReader.SetRowIndex(0); SourceTableReader.IsValid(); SourceTableReader.NextRow())

54 FImportTableRow& Row = TableC->AddRow();
55 Row.SetNumValues(ColumnsMapCount);
56
57 for (uint64 DestIndex = 0; DestIndex < ColumnsMapCount; ++DestIndex)
58 {
59 const ETableColumnType ColumnType = TableC->GetLayout().GetColumnType(DestIndex);
60
61 uint64 SourceIndex;
62 int64 Multiplier = 1;
63 if (const auto* Column = ColumnsMap[static_cast<int32>(DestIndex)].TryGet<FBoth>())
64 {
65 SourceIndex = Column->ColIndex;
66 }
67 else if (const auto* ColumnA = ColumnsMap[static_cast<int32>(DestIndex)].TryGet<Variant>())
68 {
69 SourceIndex = ColumnA->ColIndex;
70 }
71 else if (const auto* ColumnM = ColumnsMap[static_cast<int32>(DestIndex)].TryGet<FMerged>())
72 {
73 SourceIndex = ColumnM->ColIndex;
74 Multiplier = Operation == Sign::Negative ? -1 : 1;
75 }
76 else
77 {
78 switch (ColumnType)
79 {
80 case TableColumnType_Bool:
81 Row.SetValue(DestIndex, false);
82 break;
83 case TableColumnType_Int:

86 case TableColumnType_Float:

108 break;
109 case TableColumnType_Float:
