63 auto CreateBitmaskFlagsArray = [PropertyHandle]()
64 {
65 constexpr int32 BitmaskBitCount = sizeof(NumericType) << 3;
66
67 TArray<FBitmaskFlagInfo> Result;
68 Result.Empty(BitmaskBitCount);
69
70 const UEnum* BitmaskEnum = nullptr;
71 const FString& BitmaskEnumName = PropertyHandle->GetMetaData(PropertyEditorConstants::MD_BitmaskEnum);
72 if (!BitmaskEnumName.IsEmpty())
73 {
74 // @TODO: Potentially replace this with a parameter passed in from a member variable on the FProperty (e.g. FByteProperty::Enum)
75 BitmaskEnum = UClass::TryFindTypeSlow<UEnum>(BitmaskEnumName);
76 }
77
78 if (BitmaskEnum)
79 {
80 const TMap<FName, FText> EnumValueDisplayNameOverrides = PropertyEditorHelpers::GetEnumValueDisplayNamesFromPropertyOverride(PropertyHandle->GetProperty(), BitmaskEnum);
81
82 const bool bUseEnumValuesAsMaskValues = BitmaskEnum->GetBoolMetaData(PropertyEditorConstants::MD_UseEnumValuesAsMaskValuesInEditor);
83 auto AddNewBitmaskFlagLambda = [BitmaskEnum, &Result, &EnumValueDisplayNameOverrides](int32 InEnumIndex, NumericType InFlagValue)
84 {
85 Result.Emplace();
86 FBitmaskFlagInfo* BitmaskFlag = &Result.Last();
87
88 BitmaskFlag->Value = InFlagValue;
89 BitmaskFlag->DisplayName = EnumValueDisplayNameOverrides.FindRef(BitmaskEnum->GetNameByIndex(InEnumIndex));
90 if (BitmaskFlag->DisplayName.IsEmpty())
91 {
92 BitmaskFlag->DisplayName = BitmaskEnum->GetDisplayNameTextByIndex(InEnumIndex);
93 }
94 BitmaskFlag->ToolTipText = BitmaskEnum->GetToolTipTextByIndex(InEnumIndex);
95 if (BitmaskFlag->ToolTipText.IsEmpty())
96 {
97 BitmaskFlag->ToolTipText = FText::Format(LOCTEXT("BitmaskDefaultFlagToolTipText", "Toggle {0} on/off"), BitmaskFlag->DisplayName);
98 }
99 };
100
101 const TArray<FName> AllowedPropertyEnums = PropertyEditorHelpers::GetValidEnumsFromPropertyOverride(PropertyHandle->GetProperty(), BitmaskEnum);
102 const TArray<FName> DisallowedPropertyEnums = PropertyEditorHelpers::GetInvalidEnumsFromPropertyOverride(PropertyHandle->GetProperty(), BitmaskEnum);
103 // Note: This loop doesn't include (BitflagsEnum->NumEnums() - 1) in order to skip the implicit "MAX" value that gets added to the enum type at compile time.
104 for (int32 BitmaskEnumIndex = 0; BitmaskEnumIndex < BitmaskEnum->NumEnums() - 1; ++BitmaskEnumIndex)
105 {
106 bool bShouldBeHidden = BitmaskEnum->HasMetaData(TEXT("Hidden"), BitmaskEnumIndex);
107 if (!bShouldBeHidden)
108 {
109 if(AllowedPropertyEnums.Num() > 0)
110 {
111 bShouldBeHidden = AllowedPropertyEnums.Find(BitmaskEnum->GetNameByIndex(BitmaskEnumIndex)) == INDEX_NONE;
112 }
113 // If both are specified, InvalidEnumValues takes precedence
114 else if(DisallowedPropertyEnums.Num() > 0)
115 {
116 bShouldBeHidden = DisallowedPropertyEnums.Find(BitmaskEnum->GetNameByIndex(BitmaskEnumIndex)) != INDEX_NONE;
117 }
118 }
119
120 const int64 EnumValue64 = BitmaskEnum->GetValueByIndex(BitmaskEnumIndex);
121 const NumericType EnumValue = static_cast<NumericType>(EnumValue64);
122
123 if (EnumValue >= 0 && !bShouldBeHidden)
124 {
125 if (bUseEnumValuesAsMaskValues)
126 {
127 if (EnumValue64 < static_cast<int64>(TNumericLimits<NumericType>::Max()) &&FMath::IsPowerOfTwo(EnumValue64))
128 {
129 AddNewBitmaskFlagLambda(BitmaskEnumIndex, EnumValue);
130 }
131 }
132 else if (EnumValue < BitmaskBitCount)
133 {
134 AddNewBitmaskFlagLambda(BitmaskEnumIndex, TBitmaskValueHelpers<NumericType>::LeftShift(static_cast<NumericType>(1), EnumValue));
135 }
136 }
137 }
138 }
139 else
140 {
141 for (int32 BitmaskFlagIndex = 0; BitmaskFlagIndex < BitmaskBitCount; ++BitmaskFlagIndex)
142 {
143 Result.Emplace();
144 FBitmaskFlagInfo* BitmaskFlag = &Result.Last();
145
146 BitmaskFlag->Value = TBitmaskValueHelpers<NumericType>::LeftShift(static_cast<NumericType>(1), BitmaskFlagIndex);
147 BitmaskFlag->DisplayName = FText::Format(LOCTEXT("BitmaskDefaultFlagDisplayName", "Flag {0}"), FText::AsNumber(BitmaskFlagIndex + 1));
148 BitmaskFlag->ToolTipText = FText::Format(LOCTEXT("BitmaskDefaultFlagToolTipText", "Toggle {0} on/off"), BitmaskFlag->DisplayName);
149 }
150 }
151
152 return Result;
153 };

492 void OnValueChanged( NumericType NewValue )
493 {
494 if( bIsUsingSlider )
495 {
496 const TSharedRef< IPropertyHandle > PropertyHandle = PropertyEditor->GetPropertyHandle();
497
498 NumericType OrgValue(0);
499 if (PropertyHandle->GetValue(OrgValue) != FPropertyAccess::Fail)
500 {
501 // Value hasn't changed, so lets return now
502 if (OrgValue == NewValue)
503 {
504 return;
505 }
506 }
507
508 // We don't create a transaction for each property change when using the slider. Only once when the slider first is moved
509 EPropertyValueSetFlags::Type Flags = (EPropertyValueSetFlags::InteractiveChange | EPropertyValueSetFlags::NotTransactable);
510 PropertyHandle->SetValue( NewValue, Flags );
511
512 if (TypeInterface.IsValid())
513 {
514 TypeInterface->SetupFixedDisplay(NewValue);
515 }
516 }

519 void OnValueCommitted( NumericType NewValue, ETextCommit::Type CommitInfo )
520 {
521 const TSharedRef< IPropertyHandle > PropertyHandle = PropertyEditor->GetPropertyHandle();
522 NumericType OrgValue(0);
523 /* sometimes an FProperty may have been destroyed due to 2 different events invoking this handler (with the same
524 * NewValue) and the first one nullifying the current FProperty ~ in this case it's too late to not invoke the
525 * method, but we can not run the code instead */
526 if (PropertyHandle->GetProperty())
527 {
528 if (bIsUsingSlider || (PropertyHandle->GetValue(OrgValue) == FPropertyAccess::Fail || OrgValue != NewValue))
529 {
530 PropertyHandle->SetValue(NewValue);
531 LastSliderCommittedValue = NewValue;
532 }
533
534 if (TypeInterface.IsValid())
535 {
536 TypeInterface->SetupFixedDisplay(NewValue);
537 }
538 }

541 void OnUndeterminedValueCommitted( FText NewValue, ETextCommit::Type CommitType )
542 {
543 const TSharedRef<IPropertyHandle> PropertyHandle = PropertyEditor->GetPropertyHandle();
544 const FString NewValueString = NewValue.ToString();
545 TArray<FString> PerObjectValues;
546
547 // evaluate expression for each property value
548 PropertyHandle->GetPerObjectValues(PerObjectValues);
549
550 for (auto& Value : PerObjectValues)
551 {
552 NumericType OldNumericValue;
553 TTypeFromString<NumericType>::FromString(OldNumericValue, *Value);
554 const FString ReplacedNewValueString = NewValueString.Replace(*PropertyEditorConstants::DefaultUndeterminedText.ToString(), *Value);
555 TOptional<NumericType> NewNumericValue = TypeInterface->FromString(ReplacedNewValueString, OldNumericValue);
556
557 if (NewNumericValue.IsSet())
558 {
559 Value = TTypeToString<NumericType>::ToString(NewNumericValue.GetValue());
560 }
561 }
562
563 PropertyHandle->SetPerObjectValues(PerObjectValues);
564 }
