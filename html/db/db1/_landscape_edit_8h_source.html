330template<typename T>
331void FLandscapeEditDataInterface::ShrinkData(TArray<T>& Data, int32 OldMinX, int32 OldMinY, int32 OldMaxX, int32 OldMaxY, int32 NewMinX, int32 NewMinY, int32 NewMaxX, int32 NewMaxY)
332{
333 check(OldMinX <= OldMaxX && OldMinY <= OldMaxY);
334 check(NewMinX >= OldMinX && NewMaxX <= OldMaxX);
335 check(NewMinY >= OldMinY && NewMaxY <= OldMaxY);
336
337 if (NewMinX != OldMinX || NewMinY != OldMinY ||
338 NewMaxX != OldMaxX || NewMaxY != OldMaxY)
339 {
340 // if only the MaxY changes we don't need to do the moving, only the truncate
341 if (NewMinX != OldMinX || NewMinY != OldMinY || NewMaxX != OldMaxX)
342 {
343 for (int32 DestY = 0, SrcY = NewMinY - OldMinY; DestY <= NewMaxY - NewMinY; DestY++, SrcY++)
344 {
345// UE_LOGF(LogLandscape, Warning, "Dest: %d, %d = %d Src: %d, %d = %d Width = %d", 0, DestY, DestY * (1 + NewMaxX - NewMinX), NewMinX - OldMinX, SrcY, SrcY * (1 + OldMaxX - OldMinX) + NewMinX - OldMinX, (1 + NewMaxX - NewMinX));
346 T* DestData = &Data[DestY * (1 + NewMaxX - NewMinX)];
347 const T* SrcData = &Data[SrcY * (1 + OldMaxX - OldMinX) + NewMinX - OldMinX];
348 FMemory::Memmove(DestData, SrcData, (1 + NewMaxX - NewMinX) * sizeof(T));
349 }
350 }
351
352 const int32 NewSize = (1 + NewMaxY - NewMinY) * (1 + NewMaxX - NewMinX);
353 Data.RemoveAt(NewSize, Data.Num() - NewSize);
354 }
355}
