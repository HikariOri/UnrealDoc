162/*************************************************************************************************************************
163 * This fast linear allocator can be used for temporary allocations, and is best suited for allocations that are
164 * produced and consumed on different threads and within the lifetime of a frame. Although the lifetime of any
165 * individual allocation is not hard-tied to a frame (tracking is done using the FBlockHeader::NumAllocations atomic
166 * variable), the application will eventually run OOM if allocations are not cleaned up in a timely manner.
167 *
168 * There is a fast-path version of the allocator that skips AllocationHeaders by aligning the BlockHeader with the
169 * BlockSize, so that headers can easily be found by AligningDown the address of the Allocation itself.
170 * The allocator works by allocating a larger block in TLS which has a Header at the front which contains the atomic,
171 * and all allocations are than allocated from this block:
172 *
173 * ------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------
174 * | FBlockHeader(atomic counter etc.) | Alignment Waste | FAllocationHeader(size, optional) | Memory used for Allocation | Alignment Waste | FAllocationHeader(size, optional) | Memory used for Allocation | FreeSpace ...
175 * ------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------
176 *
177 * The allocator is most often used concurrently, but also supports single-threaded use cases, so it can be used for an
178 * array scratchpad.
179 */
180enum class ELinearAllocatorThreadPolicy
181{
182 ThreadSafe,
183 NotThreadSafe
184};

288 struct FTLSCleanup
289 {
290 FBlockHeader* Header = nullptr;
291
292 ~FTLSCleanup()
293 {
294 if(Header)
295 {
296 Header->NextAllocationPtr = uintptr_t(Header) + BlockAllocationTag::BlockSize;
297 const uint32 DeltaCount = UINT_MAX - Header->Num;
298
299 // on the allocating side we only need to do a single atomic to reduce contention with the deletions
300 // this will leave the atomic in a state where it only counts the number of live allocations (before it was based of UINT_MAX)
301 if (Header->NumAllocations.FetchSub(DeltaCount, std::memory_order_acq_rel) == DeltaCount)
302 {
303 //if all allocations are already freed we can reuse the Block again
304 Header->~FBlockHeader();
305 ASAN_UNPOISON_MEMORY_REGION( Header, BlockAllocationTag::BlockSize );
306 MemoryTrace_UnmarkAllocAsHeap(uint64(Header), EMemoryTraceRootHeap::SystemMemory);
307 BlockAllocationTag::Allocator::Free(Header, BlockAllocationTag::BlockSize);
308 }
309 }
310 }

313 FORCENOINLINE static void AllocateBlock(FBlockHeader*& Header)
314 {
315 if constexpr (!BlockAllocationTag::InlineBlockAllocation)
316 {
317 static_assert(BlockAllocationTag::BlockSize >= sizeof(FBlockHeader) + sizeof(FAllocationHeader));
318 uint32 BlockAlignment = SupportsFastPath ? BlockAllocationTag::BlockSize : alignof(FBlockHeader);
319 Header = new (BlockAllocationTag::Allocator::Malloc(BlockAllocationTag::BlockSize, BlockAlignment)) FBlockHeader;
320 MemoryTrace_MarkAllocAsHeap(uint64(Header), EMemoryTraceRootHeap::SystemMemory);
321 checkSlow(IsAligned(Header, BlockAlignment));
322 if constexpr (!SupportsFastPath)
323 {
324 ASAN_POISON_MEMORY_REGION( Header + 1, BlockAllocationTag::BlockSize - sizeof(FBlockHeader) );
325 }
326
327 UE_MODULE_SAFE_THREAD_LOCAL(FTLSCleanup, Cleanup);
328 new (&Cleanup) FTLSCleanup{ Header };
329 }

345 static inline void* Malloc(SIZE_T Size, uint32 Alignment)
346 {
347 checkSlow(Alignment >= 1 && FMath::IsPowerOfTwo(Alignment));
348
349 if (AutoRTFM::IsClosed())
350 {
351 void* const Pointer = AutoRTFM::Open([Size, Alignment]
352 {
353 return Malloc(Size, Alignment);
354 });
355
356 AutoRTFM::OnAbort([Pointer]
357 {
358 Free(Pointer);
359 });
360
361 return Pointer;
362 }
363
364 if constexpr (!SupportsFastPath)
365 {
366 Alignment = (Alignment < alignof(FAllocationHeader)) ? alignof(FAllocationHeader) : Alignment;
367#if IS_ASAN_ENABLED
368 // Make sure FAllocationHeader is 8 bytes aligned so poison / unpoison doesnt end up as false positive
369 Alignment = Align(Alignment, 8);
370#endif
371 }
372
373 UE_MODULE_SAFE_THREAD_LOCAL(FBlockHeader*, Header);
374 if (Header == nullptr)
375 {
376 AllocateNewBlock:
377 if constexpr (BlockAllocationTag::InlineBlockAllocation)
378 {
379 static_assert(BlockAllocationTag::BlockSize >= sizeof(FBlockHeader) + sizeof(FAllocationHeader));
380 uint32 BlockAlignment = SupportsFastPath ? BlockAllocationTag::BlockSize : alignof(FBlockHeader);
381 Header = new (BlockAllocationTag::Allocator::Malloc(BlockAllocationTag::BlockSize, BlockAlignment)) FBlockHeader;
382 MemoryTrace_MarkAllocAsHeap(uint64(Header), EMemoryTraceRootHeap::SystemMemory);
383 checkSlow(IsAligned(Header, BlockAlignment));
384 if constexpr (!SupportsFastPath)
385 {
386 ASAN_POISON_MEMORY_REGION( Header + 1, BlockAllocationTag::BlockSize - sizeof(FBlockHeader) );
387 }
388
389 UE_MODULE_SAFE_THREAD_LOCAL(FTLSCleanup, Cleanup);
390 new (&Cleanup) FTLSCleanup{ Header };
391 }
392 else
393 {
394 AllocateBlock(Header);
395 }
396 }
397
398 AllocateNewItem:
399 if constexpr (SupportsFastPath)
400 {
401 uintptr_t AlignedOffset = Align(Header->NextAllocationPtr, Alignment);
402 if (AlignedOffset + Size <= uintptr_t(Header) + BlockAllocationTag::BlockSize)
403 {
404 Header->NextAllocationPtr = AlignedOffset + Size;
405 Header->Num++;
406
407 MemoryTrace_Alloc(uint64(AlignedOffset), Size, Alignment);
408 LLM_IF_ENABLED( FLowLevelMemTracker::Get().OnLowLevelAlloc(ELLMTracker::Default, reinterpret_cast<void*>(AlignedOffset), Size, ELLMTag::LinearAllocator, LLMAllocationType) );
409 return reinterpret_cast<void*>(AlignedOffset);
410 }
411
412 // The cold path of the code starts here
413 constexpr SIZE_T HeaderSize = sizeof(FBlockHeader);
414 if constexpr (BlockAllocationTag::AllowOversizedBlocks)
415 {
416 //support for oversized Blocks
417 if (HeaderSize + Size + Alignment > BlockAllocationTag::BlockSize)
418 {
419 FBlockHeader* LargeHeader = new (BlockAllocationTag::Allocator::Malloc(HeaderSize + Size + Alignment, BlockAllocationTag::BlockSize)) FBlockHeader;
420 MemoryTrace_MarkAllocAsHeap(uint64(LargeHeader), EMemoryTraceRootHeap::SystemMemory);
421 checkSlow(IsAligned(LargeHeader, alignof(FBlockHeader)));
422
423 uintptr_t LargeAlignedOffset = Align(LargeHeader->NextAllocationPtr, Alignment);
424 LargeHeader->NextAllocationPtr = uintptr_t(LargeHeader) + HeaderSize + Size + Alignment;
425 LargeHeader->NumAllocations.Store(1, std::memory_order_release);
426
427 checkSlow(LargeAlignedOffset + Size <= LargeHeader->NextAllocationPtr);
428
429 MemoryTrace_Alloc(uint64(LargeAlignedOffset), Size, Alignment);
430 LLM_IF_ENABLED( FLowLevelMemTracker::Get().OnLowLevelAlloc(ELLMTracker::Default, reinterpret_cast<void*>(LargeAlignedOffset), Size, ELLMTag::LinearAllocator, LLMAllocationType) );
431 return reinterpret_cast<void*>(LargeAlignedOffset);
432 }
433 }
434 check(HeaderSize + Size + Alignment <= BlockAllocationTag::BlockSize);
435 }
436 else
437 {
438 uintptr_t AlignedOffset = Align(Header->NextAllocationPtr, Alignment);
439 if (AlignedOffset + Size <= uintptr_t(Header) + BlockAllocationTag::BlockSize)
440 {
441 Header->NextAllocationPtr = AlignedOffset + Size + sizeof(FAllocationHeader);
442 Header->Num++;
443
444 FAllocationHeader* AllocationHeader = reinterpret_cast<FAllocationHeader*>(AlignedOffset) - 1;
445 ASAN_UNPOISON_MEMORY_REGION( AllocationHeader, sizeof(FAllocationHeader) + Size );
446 new (AllocationHeader) FAllocationHeader(Header, Size);
447 ASAN_POISON_MEMORY_REGION( AllocationHeader, sizeof(FAllocationHeader) );
448
449 MemoryTrace_Alloc(uint64(AllocationHeader), Size + sizeof(FAllocationHeader), Alignment);
450 LLM_IF_ENABLED( FLowLevelMemTracker::Get().OnLowLevelAlloc(ELLMTracker::Default, AllocationHeader, Size + sizeof(FAllocationHeader), ELLMTag::LinearAllocator, LLMAllocationType) );
451 return reinterpret_cast<void*>(AlignedOffset);
452 }
453
454 // The cold path of the code starts here
455 constexpr SIZE_T HeaderSize = sizeof(FBlockHeader) + sizeof(FAllocationHeader);
456 if constexpr (BlockAllocationTag::AllowOversizedBlocks)
457 {
458 //support for oversized Blocks
459 if (HeaderSize + Size + Alignment > BlockAllocationTag::BlockSize)
460 {
461 FBlockHeader* LargeHeader = new (BlockAllocationTag::Allocator::Malloc(HeaderSize + Size + Alignment, alignof(FBlockHeader))) FBlockHeader;
462 MemoryTrace_MarkAllocAsHeap(uint64(LargeHeader), EMemoryTraceRootHeap::SystemMemory);
463 checkSlow(IsAligned(LargeHeader, alignof(FBlockHeader)));
464
465 uintptr_t LargeAlignedOffset = Align(LargeHeader->NextAllocationPtr, Alignment);
466 LargeHeader->NextAllocationPtr = uintptr_t(LargeHeader) + HeaderSize + Size + Alignment;
467 LargeHeader->NumAllocations.Store(1, std::memory_order_release);
468
469 checkSlow(LargeAlignedOffset + Size <= LargeHeader->NextAllocationPtr);
470 FAllocationHeader* AllocationHeader = new (reinterpret_cast<FAllocationHeader*>(LargeAlignedOffset) - 1) FAllocationHeader(LargeHeader, Size);
471 ASAN_POISON_MEMORY_REGION( AllocationHeader, sizeof(FAllocationHeader) );
472
473 MemoryTrace_Alloc(uint64(AllocationHeader), Size + sizeof(FAllocationHeader), Alignment);
474 LLM_IF_ENABLED( FLowLevelMemTracker::Get().OnLowLevelAlloc(ELLMTracker::Default, AllocationHeader, Size + sizeof(FAllocationHeader), ELLMTag::LinearAllocator, LLMAllocationType) );
475 return reinterpret_cast<void*>(LargeAlignedOffset);
476 }
477 }
478 check(HeaderSize + Size + Alignment <= BlockAllocationTag::BlockSize);
479 }
480
481 //allocation of a new Block
482 Header->NextAllocationPtr = uintptr_t(Header) + BlockAllocationTag::BlockSize;
483 const uint32 DeltaCount = UINT_MAX - Header->Num;
484
485 // on the allocating side we only need to do a single atomic to reduce contention with the deletions
486 // this will leave the atomic in a state where it only counts the number of live allocations (before it was based of UINT_MAX)
487 if (Header->NumAllocations.FetchSub(DeltaCount, std::memory_order_acq_rel) == DeltaCount)
488 {
489 //if all allocations are already freed we can reuse the Block again
490 Header->~FBlockHeader();
491 Header = new (Header) FBlockHeader;
492 ASAN_POISON_MEMORY_REGION( Header + 1, BlockAllocationTag::BlockSize - sizeof(FBlockHeader) );
493 goto AllocateNewItem;
494 }
495
496 goto AllocateNewBlock;
497 }

499 static inline void Free(void* Pointer)
500 {
501 if(Pointer != nullptr)
502 {
503 if (AutoRTFM::IsClosed())
504 {
505 AutoRTFM::OnCommit([Pointer]
506 {
507 Free(Pointer);
508 });
509
510 return;
511 }
512
513 if constexpr (SupportsFastPath)
514 {
515 MemoryTrace_Free(uint64(Pointer));
516 LLM_IF_ENABLED(FLowLevelMemTracker::Get().OnLowLevelFree(ELLMTracker::Default, Pointer, LLMAllocationType));
517 FBlockHeader* Header = reinterpret_cast<FBlockHeader*>(AlignDown(Pointer, BlockAllocationTag::BlockSize));
518
519 // this deletes complete blocks when the last allocation is freed
520 if (Header->NumAllocations.FetchSub(1, std::memory_order_acq_rel) == 1)
521 {
522 const uintptr_t NextAllocationPtr = Header->NextAllocationPtr;
523 Header->~FBlockHeader();
524 MemoryTrace_UnmarkAllocAsHeap(uint64(Header), EMemoryTraceRootHeap::SystemMemory);
525 BlockAllocationTag::Allocator::Free(Header, NextAllocationPtr - uintptr_t(Header));
526 }
527 }
528 else
529 {
530 FAllocationHeader* AllocationHeader = GetAllocationHeader(Pointer);
531 ASAN_UNPOISON_MEMORY_REGION( AllocationHeader, sizeof(FAllocationHeader) );
532 MemoryTrace_Free(uint64(AllocationHeader));
533 LLM_IF_ENABLED(FLowLevelMemTracker::Get().OnLowLevelFree(ELLMTracker::Default, AllocationHeader, LLMAllocationType));
534 FBlockHeader* Header = AllocationHeader->GetBlockHeader();
535 ASAN_POISON_MEMORY_REGION( AllocationHeader, sizeof(FAllocationHeader) + AllocationHeader->GetAllocationSize() );
536
537 // this deletes complete blocks when the last allocation is freed
538 if (Header->NumAllocations.FetchSub(1, std::memory_order_acq_rel) == 1)
539 {
540 const uintptr_t NextAllocationPtr = Header->NextAllocationPtr;
541 Header->~FBlockHeader();
542 ASAN_UNPOISON_MEMORY_REGION( Header, NextAllocationPtr - uintptr_t(Header) );
543 MemoryTrace_UnmarkAllocAsHeap(uint64(Header), EMemoryTraceRootHeap::SystemMemory);
544 BlockAllocationTag::Allocator::Free(Header, NextAllocationPtr - uintptr_t(Header));
545 }
546 }
547 }
548 }

550 AUTORTFM_OPEN static inline SIZE_T GetAllocationSize(void* Pointer)
551 {
552 if (Pointer)
553 {
554 if constexpr (SupportsFastPath)
555 {
556 return Align(uintptr_t(Pointer), BlockAllocationTag::BlockSize) - uintptr_t(Pointer);
557 }
558 else
559 {
560 FAllocationHeader* AllocationHeader = GetAllocationHeader(Pointer);
561 ASAN_UNPOISON_MEMORY_REGION( AllocationHeader, sizeof(FAllocationHeader) );
562 SIZE_T Size = AllocationHeader->GetAllocationSize();
563 ASAN_POISON_MEMORY_REGION( AllocationHeader, sizeof(FAllocationHeader) );
564 return Size;
565 }
566 }
567 return 0;
568 }

570 static inline void* Realloc(void* Old, SIZE_T Size, uint32 Alignment)
571 {
572 void* New = nullptr;
573 if(Size != 0)
574 {
575 New = Malloc(Size, Alignment);
576 SIZE_T OldSize = GetAllocationSize(Old);
577 if(OldSize != 0)
578 {
579 memcpy(New, Old, Size < OldSize ? Size : OldSize);
580 }
581 }
582 Free(Old);
583 return New;
584 }

825 inline void BulkDelete()
826 {
827 FAllocation* Allocation = Next.exchange(nullptr, std::memory_order_acquire);
828 while (Allocation != nullptr)
829 {
830 FAllocation* NextAllocation = Allocation->Next;
831 Allocation->~FAllocation();
832 TConcurrentLinearAllocator<BlockAllocationTag>::Free(Allocation);
833 Allocation = NextAllocation;
834 }

837 // In a transaction because the bulk allocator only has bulk delete, on abort we're just wasting the memory.
838 // So we run this code in the open (non-transactionally) always.
839 AUTORTFM_OPEN inline void* Malloc(SIZE_T Size, uint32 Alignment)
840 {
841 SIZE_T TotalSize = Align(sizeof(FAllocation), Alignment) + Size;
842 FAllocation* Allocation = new (TConcurrentLinearAllocator<BlockAllocationTag>::Malloc(TotalSize, FMath::Max(static_cast<uint32>(alignof(FAllocation)), Alignment))) FAllocation();
843
844 void* Alloc = Align(Allocation + 1, Alignment);
845 checkSlow(uintptr_t(Alloc) + Size - uintptr_t(Allocation) <= TotalSize);
846
847 Allocation->Next = Next.load(std::memory_order_relaxed);
848 while (!Next.compare_exchange_strong(Allocation->Next, Allocation, std::memory_order_release, std::memory_order_relaxed))
849 {
850 }
851 return Alloc;
852 }
