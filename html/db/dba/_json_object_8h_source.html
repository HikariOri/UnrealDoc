19//
20//
21//----------------------------------------------------------------------------------------------------------
22//
23// In the future, FJsonObject will only work with UE::FSharedString to store keys to values in JSON objects.
24// This is done to save memory, so that often-recurring keys in json structures can reuse the same
25// string allocations.
26//
27// In UE::FSharedString mode, the interface of FJsonObject has been reworked to always take FStringView
28// instead of FString, since storing the key string is (a) no longer guaranteed to happen, and (b) now
29// done using UE::FSharedString. As a consequence, it's no longer possible to pass a `char*` as the key,
30// meaning that you can no longer pass "foobar", you have to pass TEXT("foobar").
31//
32//----------------------------------------------------------------------------------------------------------
33//
34// WORKAROUND:
35//
36// To temporarily go back to the old FString interface, define UE_JSONOBJECT_LEGACY_STRING_KEYS=1 in your
37// build.
38//----------------------------------------------------------------------------------------------------------
39//
40// In a future release, UE_JSONOBJECT_LEGACY_STRING_KEYS will be removed, and UE::FSharedString will
41// become the only supported string key type.
42//
43//----------------------------------------------------------------------------------------------------------
44//
45//
46#if !defined(UE_JSONOBJECT_LEGACY_STRING_KEYS)
47#define UE_JSONOBJECT_LEGACY_STRING_KEYS 0
48#endif
