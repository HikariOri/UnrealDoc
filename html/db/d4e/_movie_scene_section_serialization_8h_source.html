291struct FContextAndCommands
292{
293 FContextAndCommands(IFileHandle* InHandle, const FString& InFilename)
294 : Context(InHandle, InFilename)
295 {
296
297 }
298 ~FContextAndCommands()
299 {
300 while (FSerializerCommand* Command = ReadCommands.Pop())
301 {
302 delete Command;
303 }
304
305 while (FSerializerCommand* Command = WriteCommands.Pop())
306 {
307 delete Command;
308 }
309
310 while (FSerializerCommand* Command = FreeCommands.Pop())
311 {
312 delete Command;
313 }
314 }
315
316 FGuid Guid;
317
318 FMovieSceneSerializerContext Context;
319
321 TLockFreePointerListFIFO<FSerializerCommand, 0> ReadCommands;
322
324 TLockFreePointerListFIFO<FSerializerCommand, 0> WriteCommands;
325
327 TLockFreePointerListFIFO<FSerializerCommand, 0> FreeCommands;
328
329 void Close()
330 {
331 Context.Close();
332 }
333
334 void Run()
335 {
336 // Pop and run our commands
337 TArray<FSerializerCommand*> LocalReadCommands;
338 ReadCommands.PopAll(LocalReadCommands);
339
340 TArray<FSerializerCommand*> LocalWriteCommands;
341 WriteCommands.PopAll(LocalWriteCommands);
342
343 if (LocalReadCommands.Num() > 0)
344 {
345 Context.ArReaderInner.BeginArchiving();
346
347 for (FSerializerCommand* Command : LocalReadCommands)
348 {
349 if (Command != nullptr)
350 {
351 (*Command)(Context);
352 FreeCommands.Push(Command);
353 }
354 }
355
356 Context.ArReader.Flush();
357 }
358
359 if (LocalWriteCommands.Num() > 0)
360 {
361 Context.ArWriterInner.BeginArchiving();
362
363 for (FSerializerCommand* Command : LocalWriteCommands)
364 {
365 if (Command != nullptr)
366 {
367 (*Command)(Context);
368 FreeCommands.Push(Command);
369 }
370 }
371
372 Context.ArWriter.Flush();
373 }
374 }

377 bool CheckHeader(bool bInFileExisted, TArray<FMovieSceneSessionInfo>& SessionInfos, HeaderData& InOutHeader, int64& ReadHeaderPos, FText& OutFailReason)
378 {
379
380 Context.ArReader.UsingCustomVersion(FTempCustomVersion::GUID);
381 Context.ArWriter.UsingCustomVersion(FTempCustomVersion::GUID);
382
383 if (bInFileExisted)
384 {
385 // If we are reading an existing file, check the version
386 int32 ArchiveVer = INT_MAX;
387 Context.ArReader << ArchiveVer;
388
389 if (ArchiveVer > FTempCustomVersion::LatestVersion)
390 {
391 OutFailReason = FText::Format(NSLOCTEXT("MovieSceneSerializer", "FileOpenFailedIncorrectVersion", "Cannot open file.\nFile version: {0}.\nMax supported version: {1}."), FText::AsNumber(ArchiveVer), FText::AsNumber((int32)FTempCustomVersion::LatestVersion));
392 UE_LOGF(MovieSceneSerialization, Error, "Cannot open Movie Scene Serialization cache %ls. File version: %d. Max supported version: %d", *Context.Filename, ArchiveVer, FTempCustomVersion::LatestVersion);
393 Context.Close();
394 return false;
395 }
396
397
398 int32 NumSessions = SessionInfos.Num();
399 Context.ArReader << NumSessions;
400
401 if (Context.ArReader.IsLoading())
402 {
403 SessionInfos.SetNum(NumSessions);
404 }
405
406 for (int32 SessionIndex = 0; SessionIndex < NumSessions; ++SessionIndex)
407 {
408 Context.ArReader << SessionInfos[SessionIndex];
409 }
410
411 Context.ArReader << InOutHeader;
412
413 Context.ArReader.Flush();
414 }
415 else
416 {
417 int32 ArchiveVer = Context.ArWriter.CustomVer(FTempCustomVersion::GUID);
418 Context.ArWriter << ArchiveVer;
419
420
421 int32 NumSessions = SessionInfos.Num();
422 Context.ArWriter << NumSessions;
423 for (int32 SessionIndex = 0; SessionIndex < NumSessions; ++SessionIndex)
424 {
425 Context.ArWriter << SessionInfos[SessionIndex];
426 }
427
428 Context.ArWriter << InOutHeader;
429 Context.ArWriter.Flush();
430 }
431 ReadHeaderPos = Context.ArReader.Tell();
432
433 return true;
434 }

487 {

489 TSharedPtr<FContextAndCommands<HeaderData, FrameData>> Context = MakeShared<FContextAndCommands<HeaderData, FrameData>>(InHandle, InFileName);
490 ContextMap.Add(InGuid, Context);
491 }
492
494 virtual uint32 Run() override
495 {
496 while (bRunning)
497 {
498 {
499 FScopeLock Lock(&ContextMapCriticalSection);
500

510 bRunning = false;

512 FPlatformProcess::Sleep(MovieSceneSerializationNamespace::SerializerSleepTime);

517 }

528 TSharedPtr<FContextAndCommands<HeaderData, FrameData >>* Context = ContextMap.Find(InGuid);

706{
707 if (MovieSceneSerializationNamespace::bAutoSerialize)
708 {
709 // Get absolute file path
710 FString SaveDirectory = GetLocalCaptureDir();
711 FString AbsoluteFilePath = SaveDirectory + "/" + InFilename;
712 FramesWritten = 0;
713 // Create file handle and hand off to runnable thread
714 IFileHandle* Handle = FPlatformFileManager::Get().GetPlatformFile().OpenWrite(*AbsoluteFilePath, false, true);
715 if (Handle != nullptr)
716 {
717 UE_LOGF(MovieSceneSerialization, Display, "Opened Data cache: %ls", *AbsoluteFilePath);
718
719 // Create runnable
720 if (!Runnable)
721 {
722 Runnable = new FMovieSceneSerializerRunnable<HeaderData, FrameData>();
723 }
724 Runnable->AddContext(Guid, Handle, AbsoluteFilePath);
725 Runnable->bOpenForRead = true;
726 Runnable->bOpenForWrite = true;
727 FMovieSceneSessionInfo SessionInfo(FApp::GetInstanceId(), FDateTime::UtcNow(), FNetworkVersion::GetNetworkCompatibleChangelist(), FPlatformProcess::ComputerName(),
728 FPlatformProperties::PlatformName(), FApp::GetInstanceName(), FApp::GetBuildDate());
729 SessionInfos.Emplace(SessionInfo);
730 int64 ReadHeaderPos;
731
732 if (Runnable->CheckHeader(Guid, false, SessionInfos, Header, ReadHeaderPos, OutFailReason))
733 {
734 // Create thread
735 if (!Thread)
736 {
737 FString ThreadName = (Header.SerializedType.ToString());
738 Thread = FRunnableThread::Create(Runnable, *ThreadName);
739 UE_LOGF(MovieSceneSerialization, Display, "Created Thread For Type: %ls", *ThreadName);
740 }
741
742 return true;
743 }
744 else
745 {
746 Runnable->Close(Guid);
747 /*
748 Runnable->Stop();
749 delete Runnable;
750 Runnable = nullptr;
751 */
752 return false;
753 }
754 }
755 else
756 {
757 OutFailReason = NSLOCTEXT("MovieSceneSerializer", "FileOpenFailedToCreateArchive", "Cannot open file.\nFailed to create archive.");
758
759 UE_LOGF(MovieSceneSerialization, Error, "Cannot open Movie Scene Serialization cache %ls. Failed to create archive.", *InFilename);

773template<typename HeaderData, typename FrameData>
774bool TMovieSceneSerializer<HeaderData, FrameData>::OpenForRead(const FString& AbsoluteFilePath, HeaderData& OutHeader, FText& OutFailReason)
775{
776
777 const bool bFileExists = IFileManager::Get().FileExists(*AbsoluteFilePath);
778
779 // Create file handle and hand off to runnable thread
780 IFileHandle* Handle = bFileExists ? FPlatformFileManager::Get().GetPlatformFile().OpenRead(*AbsoluteFilePath) : nullptr;
781 if (Handle != nullptr)
782 {
783 UE_LOGF(MovieSceneSerialization, Display, "Opened Movie Scene Serialization cache: %ls", *AbsoluteFilePath);
784
785 // Create runnable
786 if (!Runnable)
787 {
788 Runnable = new FMovieSceneSerializerRunnable<HeaderData, FrameData>();
789 }
790 Runnable->AddContext(Guid, Handle, AbsoluteFilePath);
791
792 Runnable->bOpenForRead = true;
793 Runnable->bOpenForWrite = false;
794 int64 ReadHeaderPos;
795
796 if (Runnable->CheckHeader(Guid,bFileExists, SessionInfos, OutHeader, ReadHeaderPos,OutFailReason))
797 {
798 HeaderData Header = OutHeader;
799
800 // Create thread
801 if (!Thread)
802 {
803 FString ThreadName = (Header.SerializedType.ToString());
804 Thread = FRunnableThread::Create(Runnable, *ThreadName);
805 UE_LOGF(MovieSceneSerialization, Display, "Created Thread For Type: %ls", *ThreadName);
806 }
807
808 // Add a command to populate the dictionaries/maps from this existing file first
809 FContextAndCommands<HeaderData,FrameData>& Context = Runnable->GetContext(Guid);
810 Context.AddReadCommand([ReadHeaderPos](FMovieSceneSerializerContext& InContext)
811 {
812 InContext.FrameIdToFrameOffset.Reset();
813 InContext.MinFrameId = 0xffffffffffffffff;
814 InContext.MaxFrameId = 0;
815
816 InContext.ArReader.Seek(ReadHeaderPos);
817
818 while (true)
819 {
820 int64 FrameHeaderPos = InContext.ArReader.Tell();
821
822 FMovieSceneSerializedFrameHeader Header;
823 InContext.ArReader << Header;
824
825 InContext.FrameIdToFrameOffset.Emplace(Header.FrameId, FrameHeaderPos);
826 InContext.MinFrameId = FMath::Min(InContext.MinFrameId, Header.FrameId);
827 InContext.MaxFrameId = FMath::Max(InContext.MaxFrameId, Header.FrameId);
828
829 if (Header.NextFrameOffset != MovieSceneSerializationNamespace::InvalidOffset && Header.NextFrameOffset < InContext.ArReader.TotalSize())
830 {
831 InContext.ArReader.Seek(Header.NextFrameOffset);
832 }
833 else
834 {
835 break;
836 }
837 }
838
839 InContext.ArReader.Seek(ReadHeaderPos);
840 });
841
842 return true;
843 }
844 else
845 {
846 /*
847 Runnable->Stop();
848 delete Runnable;
849 Runnable = nullptr;
850 */
851 return false;
852 }
853 }
854 else
855 {
856 OutFailReason = NSLOCTEXT("MovieSceneSerializer", "FileOpenFailedToCreateArchive", "Cannot open file.\nFailed to create archive.");
857
858 UE_LOGF(MovieSceneSerialization, Error, "Cannot open Movie Scene Serialization cache %ls. Failed to create archive.", *AbsoluteFilePath);
859 return false;
860 }

889template<typename HeaderData, typename FrameData>
890void TMovieSceneSerializer<HeaderData, FrameData>::GetDataRanges(TFunction<void(uint64 /*InMinFrameId*/, uint64 /*InMaxFrameId*/)> InCompletionCallback)
891{
892 if (Runnable != nullptr && Runnable->bOpenForRead)
893 {
894 FContextAndCommands<HeaderData,FrameData>& Context = Runnable->GetContext(Guid);
895 Context.AddReadCommand([InCompletionCallback](FMovieSceneSerializerContext& InContext)
896 {
897 uint64 MinFrameId = 0, MaxFrameId = 0;
898 MinFrameId = InContext.MinFrameId;
