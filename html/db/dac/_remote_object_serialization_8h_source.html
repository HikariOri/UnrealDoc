172 /*
173 * We want to be able to store "annotations" (metadata) about an Object while it's borrowed in order to
174 * execute it on the owning server when the loan is returned.
175 */
176 struct FBorrowedRpcAnnotations
177 {
178 // The data for an individual RPC call
179 struct FBorrowedSerializedRPC
180 {
182 TWeakObjectPtr<UObject> SubObject;
183
185 FName FunctionName;
186
188 TArray<uint8> Params;
189
190 friend FArchive& operator<<(FArchive& Ar, FBorrowedSerializedRPC& Value)
191 {
192 Ar << Value.SubObject;
193 Ar << Value.FunctionName;
194 Ar << Value.Params;
195
196 return Ar;
197 }
198 };
199
200 // Serialized RPC Data that were processed (not executed) while borrowed
201 TArray<FBorrowedSerializedRPC> SerializedRPCs;
202
203 // Annotation is default (empty) and removable if it has no RPCs
204 bool IsDefault() const
205 {
206 return SerializedRPCs.IsEmpty();
207 }
208
209 void Serialize(FArchive& Ar)
210 {
211 TArrayPrivateFriend::Serialize(Ar, SerializedRPCs);
212 }
213 };
