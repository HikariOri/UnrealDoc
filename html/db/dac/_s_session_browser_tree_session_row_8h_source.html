93 FText SanitizeSessionName(const TSharedPtr<ISessionInfo>& SessionInfo) const
94 {
95 const FString& SessionName = SessionInfo->GetSessionName();
96 const FString& SessionOwner = SessionInfo->GetSessionOwner();
97
98 // return name of a launched session
99 if (!SessionName.IsEmpty())
100 {
101 if (SessionInfo->GetSessionOwner() != FPlatformProcess::UserName(false))
102 {
103 return FText::Format(LOCTEXT("SessionNameFormat", "{0} - {1}"), FText::FromString(SessionName), FText::FromString(SessionOwner));
104 }
105
106 return FText::FromString(SessionName);
107 }
108
109 // generate name for a standalone session
110 TArray<TSharedPtr<ISessionInstanceInfo>> Instances;
111 SessionInfo->GetInstances(Instances);
112
113 if (Instances.Num() > 0)
114 {
115 const TSharedPtr<ISessionInstanceInfo>& FirstInstance = Instances[0];
116
117 if ((Instances.Num() == 1) && FApp::IsThisInstance(FirstInstance->GetInstanceId()))
118 {
119 return LOCTEXT("ThisApplicationSessionText", "This Application");
120 }
121 }
122
123 if (SessionInfo->GetSessionOwner() != FPlatformProcess::UserName(false))
124 {
125 return FText::Format(LOCTEXT("UnnamedSessionFormat", "Unnamed - {0}"), FText::FromString(SessionOwner));
126 }
127
128 return LOCTEXT("UnnamedSessionText", "Unnamed");
129 }

165 FText HandleBorderToolTipText() const
166 {
167 FTextBuilder ToolTipTextBuilder;
168
169 TSharedPtr<ISessionInfo> SessionInfo = Item->GetSessionInfo();
170
171 if (SessionInfo.IsValid())
172 {
173 const FCoreTexts& CoreTexts = FCoreTexts::Get();
174
175 ToolTipTextBuilder.AppendLineFormat(LOCTEXT("SessionToolTipSessionId", "Session ID: {0}"), FText::FromString(SessionInfo->GetSessionId().ToString(EGuidFormats::DigitsWithHyphensInBraces)));
176 ToolTipTextBuilder.AppendLine();
177 ToolTipTextBuilder.AppendLineFormat(LOCTEXT("SessionToolTipNumInstances", "Total Instances: {0}"), FText::AsNumber(SessionInfo->GetNumInstances()));
178 ToolTipTextBuilder.AppendLineFormat(LOCTEXT("SessionToolTipIsStandalone", "Is Standalone: {0}"), SessionInfo->IsStandalone() ? CoreTexts.Yes : CoreTexts.No);
179 ToolTipTextBuilder.AppendLine();
180 ToolTipTextBuilder.AppendLineFormat(LOCTEXT("SessionToolTipLastUpdateTime", "Last Update Time: {0}"), FText::AsDateTime(SessionInfo->GetLastUpdateTime()));
181 }
182
183 return ToolTipTextBuilder.ToText();
184 }
