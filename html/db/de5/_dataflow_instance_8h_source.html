22 template<typename T, typename TWriteElementFunction>
23 bool OverrideVariableArray(FInstancedPropertyBag& Variables, FName VariableName, const TArray<T>& Values, const TWriteElementFunction& WriteElementFunction, FGuid& OutGuid)
24 {
25 if (const FPropertyBagPropertyDesc* Desc = Variables.FindPropertyDescByName(VariableName))
26 {
27 TValueOrError<FPropertyBagArrayRef, EPropertyBagResult> MutableArrayRef = Variables.GetMutableArrayRef(*Desc);
28 if (FPropertyBagArrayRef* ArrayRef = MutableArrayRef.TryGetValue())
29 {
30 bool bSuccess = true;
31 ArrayRef->EmptyValues();
32 ArrayRef->AddValues(Values.Num());
33 for (int32 Idx = 0; Idx < Values.Num(); ++Idx)
34 {
35 bSuccess &= (WriteElementFunction(*ArrayRef, Idx, Values[Idx]) == EPropertyBagResult::Success);
36 }
37 if (bSuccess)
38 {
39 OutGuid = Desc->ID;
40 }
41 return bSuccess;
42 }
43 }
44 OutGuid.Invalidate();
45 return false;
46 }
