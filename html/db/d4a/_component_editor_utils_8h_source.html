144 // Given a template and a property, propagates a default value change to all instances (only if applicable)
145 template<typename T>
146 static void PropagateDefaultValueChange(class USceneComponent* InSceneComponentTemplate, const class FProperty* InProperty, const T& OldDefaultValue, const T& NewDefaultValue, TSet<class USceneComponent*>& UpdatedInstances, int32 PropertyOffset = INDEX_NONE)
147 {
148 TArray<UObject*> ArchetypeInstances;
149 if(InSceneComponentTemplate->HasAnyFlags(RF_ArchetypeObject))
150 {
151 InSceneComponentTemplate->GetArchetypeInstances(ArchetypeInstances);
152 for(int32 InstanceIndex = 0; InstanceIndex < ArchetypeInstances.Num(); ++InstanceIndex)
153 {
154 USceneComponent* InstancedSceneComponent = static_cast<USceneComponent*>(ArchetypeInstances[InstanceIndex]);
155 if(InstancedSceneComponent != nullptr && !UpdatedInstances.Contains(InstancedSceneComponent) && ApplyDefaultValueChange(InstancedSceneComponent, InProperty, OldDefaultValue, NewDefaultValue, PropertyOffset))
156 {
157 UpdatedInstances.Add(InstancedSceneComponent);
158 }
159 }
160 }
161 else if(UObject* Outer = InSceneComponentTemplate->GetOuter())
162 {
163 Outer->GetArchetypeInstances(ArchetypeInstances);
164 for(int32 InstanceIndex = 0; InstanceIndex < ArchetypeInstances.Num(); ++InstanceIndex)
165 {
166 USceneComponent* InstancedSceneComponent = static_cast<USceneComponent*>(FindObjectWithOuter(ArchetypeInstances[InstanceIndex], InSceneComponentTemplate->GetClass(), InSceneComponentTemplate->GetFName()));
167 if(InstancedSceneComponent != nullptr && !UpdatedInstances.Contains(InstancedSceneComponent) && ApplyDefaultValueChange(InstancedSceneComponent, InProperty, OldDefaultValue, NewDefaultValue, PropertyOffset))
168 {
169 UpdatedInstances.Add(InstancedSceneComponent);
170 }
171 }
172 }

175 // Given an instance of a template and a property, set a default value change to the instance (only if applicable)
176 template<typename T>
177 static bool ApplyDefaultValueChange(class USceneComponent* InSceneComponent, const class FProperty* InProperty, const T& OldDefaultValue, const T& NewDefaultValue, int32 PropertyOffset)
178 {
179 check(InProperty != nullptr);
180 check(InSceneComponent != nullptr);
181
182 ensureMsgf(CastField<FBoolProperty>(InProperty) == nullptr, TEXT("ApplyDefaultValueChange cannot be safely called on a bool property with a non-bool value, becuase of bitfields"));
183
184 T* CurrentValue = PropertyOffset == INDEX_NONE ? InProperty->ContainerPtrToValuePtr<T>(InSceneComponent) : (T*)((uint8*)InSceneComponent + PropertyOffset);
185 check(CurrentValue);
186
187 return ApplyDefaultValueChange(InSceneComponent, *CurrentValue, OldDefaultValue, NewDefaultValue);
188 }

190 // Bool specialization so it can properly handle bitfields
191 static bool ApplyDefaultValueChange(class USceneComponent* InSceneComponent, const class FProperty* InProperty, const bool& OldDefaultValue, const bool& NewDefaultValue, int32 PropertyOffset)
192 {
193 check(InProperty != nullptr);
194 check(InSceneComponent != nullptr);
195
196 // Only bool properties can have bool values
197 const FBoolProperty* BoolProperty = CastField<FBoolProperty>(InProperty);
198 check(BoolProperty);
199
200 uint8* CurrentValue = PropertyOffset == INDEX_NONE ? InProperty->ContainerPtrToValuePtr<uint8>(InSceneComponent) : ((uint8*)InSceneComponent + PropertyOffset);
201 check(CurrentValue);
202
203 bool CurrentBool = BoolProperty->GetPropertyValue(CurrentValue);
204 if (ApplyDefaultValueChange(InSceneComponent, CurrentBool, OldDefaultValue, NewDefaultValue, false))
205 {
206 BoolProperty->SetPropertyValue(CurrentValue, CurrentBool);
207
208 InSceneComponent->ReregisterComponent();
209
210 return true;
211 }
212
213 return false;
214 }

216 // Given an instance of a template and a current value, propagates a default value change to the instance (only if applicable)
217 template<typename T>
218 static bool ApplyDefaultValueChange(class USceneComponent* InSceneComponent, T& CurrentValue, const T& OldDefaultValue, const T& NewDefaultValue, bool bReregisterComponent = true)
219 {
220 check(InSceneComponent != nullptr);
221
222 // Propagate the change only if the current instanced value matches the previous default value (otherwise this could overwrite any per-instance override)
223 if(NewDefaultValue != OldDefaultValue && CurrentValue == OldDefaultValue)
224 {
225 // Ensure that this instance will be included in any undo/redo operations, and record it into the transaction buffer.
226 // Note: We don't do this for components that originate from script, because they will be re-instanced from the template after an undo, so there is no need to record them.
227 if (!InSceneComponent->IsCreatedByConstructionScript())
228 {
229 InSceneComponent->SetFlags(RF_Transactional);
230 InSceneComponent->Modify();
231 }
232
233 // We must also modify the owner, because we'll need script components to be reconstructed as part of an undo operation.
234 AActor* Owner = InSceneComponent->GetOwner();
235 if(Owner != nullptr)
236 {
237 Owner->Modify();
238 }
239
240 // Modify the value
241 CurrentValue = NewDefaultValue;
242
243 if (bReregisterComponent && InSceneComponent->IsRegistered())
244 {
245 // Re-register the component with the scene so that transforms are updated for display
246 if (InSceneComponent->AllowReregistration())
247 {
248 InSceneComponent->ReregisterComponent();
249 }
250 else
251 {
252 InSceneComponent->UpdateComponentToWorld();
253 }
254 }
255
256 return true;
257 }
258
259 return false;
260 }
