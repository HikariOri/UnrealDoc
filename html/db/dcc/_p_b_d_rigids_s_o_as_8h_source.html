18 // The goal of that function is to reorder a particle view per category in place without reallocation in O(N).
19 // The different categories should not overlap (example : kinematic vs dynamic vs static...)
20 template<typename CategoryFunc>
21 FORCEINLINE void ReorderParticleView(TArrayView<FPBDRigidParticleHandle*>& ParticlesView, const CategoryFunc& CategoryTrigger, int32& CategoryOffset)
22 {
23 const int32 NumParticles = ParticlesView.Num();
24 for (int32 ParticleIndex = CategoryOffset; ParticleIndex < NumParticles; ++ParticleIndex)
25 {
26 if (CategoryTrigger(ParticlesView[ParticleIndex]))
27 {
28 if (ParticleIndex != CategoryOffset)
29 {
30 Swap(ParticlesView[ParticleIndex], ParticlesView[CategoryOffset]);
31 }
32 ++CategoryOffset;
33 }
34 }
35 }
36}

59class FParticleUniqueIndicesMultithreaded: public IParticleUniqueIndices
60{
61public:
62 FParticleUniqueIndicesMultithreaded()
63 : Block(0)
64 {
65 //Note: tune this so that all allocation is done at initialization
66 AddPageAndAcquireNextId(/*bAcquireNextId = */ false);
67 }
68
69 FUniqueIdx GenerateUniqueIdx() override
70 {
71 while(true)
72 {
73 if(FUniqueIdx* Idx = FreeIndices.Pop())
74 {
75 return *Idx;
76 }
77
78 //nothing available so try to add some
79
80 if(FPlatformAtomics::InterlockedCompareExchange(&Block,1,0) == 0)
81 {
82 //we got here first so add a new page
83 FUniqueIdx RetIdx = FUniqueIdx(AddPageAndAcquireNextId(/*bAcquireNextId =*/ true));
84
85 //release blocker. Note: I don't think this can ever fail, but no real harm in the while loop
86 while(FPlatformAtomics::InterlockedCompareExchange(&Block,0,1) != 1)
87 {
88 }
89
90 return RetIdx;
91 }
92 }
93 }
94
95 void ReleaseIdx(FUniqueIdx Unique) override
96 {
97 ensure(Unique.IsValid());
98 int32 PageIdx = Unique.Idx / IndicesPerPage;
99 int32 Entry = Unique.Idx % IndicesPerPage;
100 FUniqueIdxPage& Page = *Pages[PageIdx];
101 FreeIndices.Push(&Page.Indices[Entry]);
102 }
103
104 ~FParticleUniqueIndicesMultithreaded()
105 {
106 //Make sure queue is empty, memory management of actual pages is handled automatically by TUniquePtr
107 while(FreeIndices.Pop())
108 {
109 }
110 }
111
112private:
113
114 int32 AddPageAndAcquireNextId(bool bAcquireNextIdx)
115 {
116 //Note: this should never really be called post initialization
117 TUniquePtr<FUniqueIdxPage> Page = MakeUnique<FUniqueIdxPage>();
118 const int32 PageIdx = Pages.Num();
119 int32 FirstIdxInPage = PageIdx * IndicesPerPage;
120 Page->Indices[0] = FUniqueIdx(FirstIdxInPage);
121
122 //If we acquire next id we avoid pushing it into the queue
123 for(int32 Count = bAcquireNextIdx ? 1 : 0; Count < IndicesPerPage; Count++)
124 {
125 Page->Indices[Count] = FUniqueIdx(FirstIdxInPage + Count);
126 FreeIndices.Push(&Page->Indices[Count]);
127 }
128
129 Pages.Emplace(MoveTemp(Page));
130 return bAcquireNextIdx ? FirstIdxInPage : INDEX_NONE;
131 }
132
133 static constexpr int32 IndicesPerPage = 1024;
134 struct FUniqueIdxPage
135 {
136 FUniqueIdx Indices[IndicesPerPage];
137 };
138 TArray<TUniquePtr<FUniqueIdxPage>> Pages;
139 TLockFreePointerListFIFO<FUniqueIdx,0> FreeIndices;
140 volatile int8 Block;
141};

177 void Insert(const TArrayView<FPBDRigidParticleHandle*>& ParticlesToInsert)
178 {
179 if (ParticlesToInsert.IsEmpty())
180 {
181 return;
182 }
183 TArray<bool> Contains;
184 Contains.AddZeroed(ParticlesToInsert.Num());
185
186 // TODO: Compile time check ensuring TParticle2 is derived from TParticle1?
187 int32 NextIdx = ParticleArray.Num();
188 for (int32 Idx = 0; Idx < ParticlesToInsert.Num(); ++Idx)

195 {
196 ParticleToIndex.Add(Particle, NextIdx++);

198 }
199 }
200 ParticleArray.Reserve(ParticleArray.Num() + NextIdx - ParticleArray.Num());
201 for (int32 Idx = 0; Idx < ParticlesToInsert.Num(); ++Idx)
202 {
203 if (!Contains[Idx] && HasValidType(ParticlesToInsert[Idx]))
204 {

206 ParticleArray.Add(Particle);
207 }
208 }
209 }

212 {
213 if (ParticleToIndex.Contains(Particle) == false)
214 {
215 ParticleToIndex.Add(Particle, ParticleArray.Num());
216 ParticleArray.Add(Particle);
217 }
218 }

221 void Remove(const TArrayView<FPBDRigidParticleHandle*>& ParticlesToRemove)
222 {
223 if (ParticlesToRemove.IsEmpty())
224 {
225 return;
226 }
227 for (int32 Idx = 0; Idx < ParticlesToRemove.Num(); ++Idx)
228 {
229 if (HasValidType(ParticlesToRemove[Idx]))
230 {
231 const TParticleType* Particle = static_cast<TParticleType*>(ParticlesToRemove[Idx]);
232 if (int32* IdxPtr = ParticleToIndex.Find(Particle))
233 {
234 int32 ParticleIdx = *IdxPtr;
235 ParticleArray.RemoveAtSwap(ParticleIdx, EAllowShrinking::No);
236 if (ParticleIdx < ParticleArray.Num())
237 {
238 //update swapped element with new index
239 ParticleToIndex[ParticleArray[ParticleIdx]] = ParticleIdx;
240 }
241 ParticleToIndex.Remove(Particle);
242 }
243 }
244 }
245 ParticleArray.Shrink();
246 }
247
248 void Remove(TParticleType* Particle)
249 {
250 if (int32* IdxPtr = ParticleToIndex.Find(Particle))
251 {
252 int32 Idx = *IdxPtr;
253 ParticleArray.RemoveAtSwap(Idx);
254 if (Idx < ParticleArray.Num())
255 {
256 //update swapped element with new index
257 ParticleToIndex[ParticleArray[Idx]] = Idx;
258 }
259 ParticleToIndex.Remove(Particle);
260 }

785
786 // Only allocation in the SetObjectState pipeline but we need to transfer particles->indices anyway
787 TArray<int32> ParticleIndices;
788 ParticleIndices.Init(0, RigidParticles.Num());
789
790 while (ParticleOffset < RigidParticles.Num())
791 {
792 const uint32 PreviousOffset = ParticleOffset;
793
794 // Reorder all the particles that have the same SOA as the FromSOA among the remaining ones
795 // [ParticleOffset+1...RigidParticles.Num()] and push them at the front of the array
796 Private::ReorderParticleView(RigidParticles, [&FromSOA](FPBDRigidParticleHandle* RigidParticle) -> bool
797 {
798 return RigidParticle && (RigidParticle->GeometryParticles == FromSOA);
799 }, ParticleOffset);
800
801 if (FromSOA != &ToSOA)
802 {
803 // Fill partially the particle indices that we want to move to the target SOA
804 for (int32 ViewIndex = PreviousOffset; ViewIndex < ParticleOffset; ++ViewIndex)
805 {
806 ParticleIndices[ViewIndex] = RigidParticles[ViewIndex]->ParticleIdx;
807 }
808 // Build the particle indices view used to update the SOAs
809 const int32 NumParticles = ParticleOffset - PreviousOffset;
810 TArrayView<int32> IndicesView(&ParticleIndices[PreviousOffset], NumParticles);
811
812 // For each SOA move the particles to the target SOA
813 uint32 OffsetIndex = ToSOA.Size();
814 FromSOA->MoveToOtherParticles(IndicesView, ToSOA);
815 check(ToSOA.Size() == (OffsetIndex + NumParticles))
816
817 // Update the particle indices for the removed/added particles
818 for(int32 GeometryIndex : IndicesView)
819 {
820 // Particles that have been moved with the swap should be updated with the right index
821 if (static_cast<uint32>(GeometryIndex) < FromSOA->Size())
822 {
823 FromSOA->Handle(GeometryIndex)->ParticleIdx = GeometryIndex;
824 }
825 // Particles SOA index and mask are updated
826 if(OffsetIndex < ToSOA.Size())
827 {
828 ToSOA.Handle(OffsetIndex)->ParticleIdx = OffsetIndex;
829 ToSOA.Handle(OffsetIndex)->GeometryParticles = &ToSOA;
830
831 ToSOA.ListMask(OffsetIndex) &= ~FromSOA->GetContainerListMask();
832 ToSOA.ListMask(OffsetIndex) |= ToSOA.GetContainerListMask();
833 }
834 ++OffsetIndex;
835 }
836 }
837 if (ParticleOffset < RigidParticles.Num())
838 {
839 FromSOA = RigidParticles[ParticleOffset]->GeometryParticles;
840 }
841 }
842 }
843 }
