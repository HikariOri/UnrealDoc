140 //
141 // This iterator iterates over the subresources within a view subset. eg:
142 //
143 // for (uint32 SubresourceIndex : ViewSubset)
144 // {
145 // // Action for each subresource
146 // }
147 //
148 class FIterator final
149 {
150 public:
151 FIterator(FD3D12ViewSubset const& ViewSubset, uint8 MipSlice, uint16 ArraySlice, uint8 PlaneSlice)
152 : MipMax (ViewSubset.Range.Mip.Num)
153 , ArrayMax (ViewSubset.Range.Array.Num)
154 , ArrayStride(ViewSubset.Layout.NumMips)
155 , PlaneStride(ViewSubset.Layout.NumMips * (ViewSubset.Layout.NumArraySlices - ViewSubset.Range.Array.Num))
156 {
157 MipRangeStart = D3D12CalcSubresource(
158 MipSlice,
159 ArraySlice,
160 PlaneSlice,
161 ViewSubset.Layout.NumMips,
162 ViewSubset.Layout.NumArraySlices
163 );
164 }
165
166 FIterator& operator ++ ()
167 {
168 if (++MipOffset == MipMax)
169 {
170 // Move to next array slice
171 MipOffset = 0;
172 MipRangeStart += ArrayStride;
173
174 if (++ArrayOffset == ArrayMax)
175 {
176 // Move to next plane slice
177 ArrayOffset = 0;
178 MipRangeStart += PlaneStride;
179 }
180 }
181
182 return *this;
183 }
184
185 uint32 operator * () const { return MipRangeStart + MipOffset; }
186
187 bool operator == (FIterator const& RHS) const { return *(*this) == *RHS; }
188 bool operator != (FIterator const& RHS) const { return !(*this == RHS); }
189
190 private:
191 // Constants
192 uint32 const MipMax;
193 uint32 const ArrayMax;
194 uint32 const ArrayStride;
195 uint32 const PlaneStride;
196
197 // Counters
198 uint32 MipRangeStart;
199 uint32 MipOffset = 0;
200 uint32 ArrayOffset = 0;
201 };

454template <typename TParent>
455struct FD3D12DeferredInitView : public FD3D12LinkedAdapterObject<TParent>
456{
457 void CreateViews(FRHICommandListBase& RHICmdList)
458 {
459 auto InitLambda = [this](FRHICommandListBase&)
460 {
461 for (TParent& LinkedView : *this)
462 {
463 LinkedView.CreateView();
464 }
465 };
466
467 if (RHICmdList.IsTopOfPipe())
468 {
469 // We have to defer the view initialization to the RHI thread if the resource is dynamic (and RHI threading is enabled), since dynamic resources can be renamed.
470 // Also insert an RHI thread fence to prevent parallel translate tasks running until this command has completed.
471 RHICmdList.EnqueueLambda(MoveTemp(InitLambda));
472 RHICmdList.RHIThreadFence(true);
473 }
474 else
475 {
476 // Run the command directly if we're bypassing RHI command list recording, or the buffer is not dynamic.
477 InitLambda(RHICmdList);
478 }
479 }
