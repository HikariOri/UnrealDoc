20// Utility struct for passing many arrays with designated initializer for clarity
21struct FCompiledInObjects
22{
23#if !IS_MONOLITHIC
24 // Tables of objects in other modules for resolving encoding pointers in these objects
25 TConstArrayView<TConstArrayView<UObject*>> LinkedModules;
26
27 // Objects which can be linked by other objects using the package's link id and the index of the
28 // object in this array.
29 // The following arrays may bes slices of this array produced by UHT to reduce duplication.
30 // This array and the following arrays may contain null entries if objects are declared inside #if scopes in headers
31 // This means that link indices are stable regardless of the value of those preprocessor definitions
32 TConstArrayView<UObject*> LinkableObjects;
33#endif // !IS_MONOLITHIC
34
35 // Packages that are part of this module - usually 1 for UObject modules or 2 for modules with Verse
36 TConstArrayView<UPackage*> Packages;
37
38 // Arrays of top level objects to initialize segmented by type to avoid casting.
39 // Typed as UObject* because they are slices of a single array
40 TConstArrayView<UObject*> Classes;
41 TConstArrayView<UObject*> Structs;
42 TConstArrayView<UObject*> Enums;
43 // Other types of objects (e.g. UDelegateFunction) which can be initialized by casting after
44 // classes are propery initialized.
45 TConstArrayView<UObject*> Others;
46
47 TConstArrayView<UE::CoreUObject::Private::FPartialClass*> Partials;
48
49 inline TConstArrayView<UClass*> GetClasses() const
50 {
51 return MakeArrayView(reinterpret_cast<UClass* const*>(Classes.GetData()), Classes.Num());
52 }
53 inline TConstArrayView<UScriptStruct*> GetStructs() const
54 {
55 return MakeArrayView(reinterpret_cast<UScriptStruct* const*>(Structs.GetData()), Structs.Num());
56 }
57 inline TConstArrayView<UEnum*> GetEnums() const
58 {
59 return MakeArrayView(reinterpret_cast<UEnum* const*>(Enums.GetData()), Enums.Num());
60 }
61};
