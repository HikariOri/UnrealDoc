166 template <typename OtherDataType>
167 [[nodiscard]] inline bool Contains(const TMemoryView<OtherDataType>& InView) const

180 template <typename OtherDataType>
181 [[nodiscard]] inline int32 CompareBytes(const TMemoryView<OtherDataType>& InView) const
182 {
183 const int32 Compare = Data == InView.Data ? 0 : FMemory::Memcmp(Data, InView.Data, FMath::Min(Size, InView.Size));
184 return Compare || Size == InView.Size ? Compare : Size < InView.Size ? -1 : 1;
185 }

188 template <typename OtherDataType>
189 [[nodiscard]] inline bool EqualBytes(const TMemoryView<OtherDataType>& InView) const
190 {
191 return Size == InView.Size && (Data == InView.Data || FMemory::Memcmp(Data, InView.Data, Size) == 0);
192 }

199 }

223 inline TMemoryView CopyFrom(FMemoryView InView) const
224 {
225 checkf(InView.Size <= Size, TEXT("Failed to copy from a view of %" UINT64_FMT " bytes "
226 "to a view of %" UINT64_FMT " bytes."), InView.Size, Size);
227 if (InView.Size)
228 {
229 FMemory::Memcpy(Data, InView.Data, InView.Size);
230 }
231 return RightChop(InView.Size);
232 }
