29// Run the callback in a transaction like Transact, but abort program
30// execution if the result is anything other than autortfm_committed.
31// Useful for testing.
32template <typename TFunctor>
33static UE_AUTORTFM_FORCEINLINE void Commit(AUTORTFM_IMPLICIT_ENABLE const TFunctor& Functor)
34{
35 const ETransactionResult Result = Transact(Functor);
36 AUTORTFM_TESTING_ASSERT(ETransactionResult::Committed == Result);
37}
38
39// Run the callback in a transaction like Transact, but abort program
40// execution if the result is anything other than abort.
41template <typename TFunctor>
42static UE_AUTORTFM_FORCEINLINE void Abort(AUTORTFM_IMPLICIT_ENABLE const TFunctor& Functor)
43{
44 const ETransactionResult Result = Transact(Functor);
45 AUTORTFM_TESTING_ASSERT(HasAborted(Result));
46}
