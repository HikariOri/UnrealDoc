150 // Callback for getting the text in the 'Claimed' column.
151 FText HandleClaimedText() const
152 {
153 FString ClaimUser = DeviceService->GetClaimUser();
154
155 if ((ClaimUser == FPlatformProcess::UserName(false)) && !DeviceService->IsRunning())
156 {

183 return ECheckBoxState::Unchecked;

186 // Callback for getting the enabled state of the 'Share' check box.

192 // Callback for getting the status text.

194 {
195 ITargetDevicePtr TargetDevice = DeviceService->GetDevice();
196
197 if (TargetDevice.IsValid())
198 {
199 if (TargetDevice->IsConnected())
200 {
201 return LOCTEXT("StatusConnected", "Connected");
202 }
203
204 return LOCTEXT("StatusDisconnected", "Disconnected");
205 }
206
207 return LOCTEXT("StatusUnavailable", "Unavailable");
208 }
209
210 // Callback for getting the text color.
211 FSlateColor HandleTextColorAndOpacity() const
212 {
213 if (DeviceService->CanStart())
214 {
215 return FSlateColor::UseForeground();
216 }
217
218 return FSlateColor::UseSubduedForeground();
219 }
