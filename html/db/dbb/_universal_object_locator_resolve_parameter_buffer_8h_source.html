203inline const void* FResolveParameterBuffer::FindParameterImpl(uint32 ParameterBit) const
204{
205 if (HasParameter(ParameterBit))
206 {
207 const int32 Index = GetParameterIndex(ParameterBit);
208 check(Index >= 0 && Index < 255);
209 return GetHeader(static_cast<uint8>(Index)).Resolve(Memory);
210 }
211
212 return nullptr;
213}
