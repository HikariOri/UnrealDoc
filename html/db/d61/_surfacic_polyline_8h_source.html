107 FSurfacicTolerance ComputeTolerance(const double Tolerance3D, const FSurfacicTolerance& MinToleranceIso, const int32 Index) const
108 {
109 double Distance3D = FVector::Distance(Points3D[Index], Points3D[Index + 1]);
110 if (FMath::IsNearlyZero(Distance3D, (double)DOUBLE_SMALL_NUMBER))
111 {
112 return FVectorUtil::FarawayPoint2D;
113 }
114 else
115 {
116 FSurfacicTolerance Tolerance2D = Points2D[Index] - Points2D[Index + 1];
117 return FVector2D::Max(Tolerance2D.GetAbs() * Tolerance3D / Distance3D, MinToleranceIso);
118 }
119 };
120
121 double ComputeLinearToleranceAt(const double Tolerance3D, const double MinLinearTolerance, const int32 Index) const
122 {
123 double Distance3D = FVector::Distance(Points3D[Index], Points3D[Index + 1]);
124 if (FMath::IsNearlyZero(Distance3D, (double)DOUBLE_SMALL_NUMBER))
125 {
126 return (Coordinates.Last() - Coordinates[0]) / 10.;
127 }
128 else
129 {
130 double LinearDistance = Coordinates[Index + 1] - Coordinates[Index];
131 return FMath::Max(LinearDistance / Distance3D * Tolerance3D, MinLinearTolerance);
132 }

149 {
150 int32 CoordinateCount = OutPolyline.Coordinates.Num();
151
152 TArray<int32> SegmentIndexes;
153 SegmentIndexes.Reserve(CoordinateCount);
154
155 TArray<double> SegmentCoordinates;
156 SegmentCoordinates.Reserve(CoordinateCount);
157
158
159 //for (int32 Index = 0; Index < CoordinateCount; ++Index)
160 for (const double Coordinate : OutPolyline.Coordinates)
161 {
162 const int32& Index = SegmentIndexes.Emplace_GetRef(Finder.Find(Coordinate));
163 SegmentCoordinates.Emplace(PolylineTools::SectionCoordinate(Coordinates, Index, Coordinate));
164 }
165
166 OutPolyline.Points2D.Reserve(CoordinateCount);
167 for (int32 Index = 0; Index < CoordinateCount; ++Index)
168 {
169 int32 SegmentIndex = SegmentIndexes[Index];
170 double SegmentCoordinate = SegmentCoordinates[Index];
171 OutPolyline.Points2D.Emplace(PolylineTools::LinearInterpolation(Points2D, SegmentIndex, SegmentCoordinate));
172 }
173
174 OutPolyline.Points3D.Reserve(CoordinateCount);
175 for (int32 Index = 0; Index < CoordinateCount; ++Index)
176 {
177 int32 SegmentIndex = SegmentIndexes[Index];
178 double SegmentCoordinate = SegmentCoordinates[Index];
179 OutPolyline.Points3D.Emplace(PolylineTools::LinearInterpolation(Points3D, SegmentIndex, SegmentCoordinate));
180 }
181
182 if (bWithNormals)
183 {
184 for (int32 Index = 0; Index < CoordinateCount; ++Index)
185 {
186 int32 SegmentIndex = SegmentIndexes[Index];
187 double SegmentCoordinate = SegmentCoordinates[Index];
188 OutPolyline.Normals.Emplace(PolylineTools::LinearInterpolation(Normals, SegmentIndex, SegmentCoordinate));
189 }
190 }
191
192 if (bWithTangent)
193 {
194 for (int32 Index = 0; Index < CoordinateCount; ++Index)
195 {
196 int32 SegmentIndex = SegmentIndexes[Index];
197 double SegmentCoordinate = SegmentCoordinates[Index];
198 OutPolyline.Tangents.Emplace(PolylineTools::LinearInterpolation(Tangents, SegmentIndex, SegmentCoordinate));
199 }

203 FDichotomyFinder DichotomyFinder(Coordinates);
204 int32 StartIndex = 0;
205 int32 EndIndex;
206
207 bool bUseDichotomy = false;
208 StartIndex = DichotomyFinder.Find(OutPolyline.Coordinates[0]);
209 EndIndex = DichotomyFinder.Find(OutPolyline.Coordinates.Last());
210 bUseDichotomy = PolylineTools::IsDichotomyToBePreferred(EndIndex - StartIndex, Coordinates.Num());
211
212
213 if (bUseDichotomy)
214 {
215 DichotomyFinder.StartLower = StartIndex;
216 DichotomyFinder.StartUpper = EndIndex;
217 ComputePoints(DichotomyFinder);
218 }
219 else
220 {
221 FLinearFinder LinearFinder(Coordinates, StartIndex);
222 ComputePoints(LinearFinder);

406 void RemoveComplementaryPoints(int32 Offset)
407 {
408 const int32 Count = Points2D.Num();
409
410 TBitArray<> Markers(true, Count);
411
412 for (int32 Index = Count - 1; Index >= 0; Index -= Offset)
413 {
414 Markers[Index] = false;
415 }
416
417 int32 CurrentIndex = 0;
418 for (int32 Index = 0; Index < Count; ++Index)
419 {
420 if (Markers[Index])
421 {
422 Coordinates[CurrentIndex] = Coordinates[Index];
423 Points2D[CurrentIndex] = Points2D[Index];
424 Points3D[CurrentIndex] = Points3D[Index];

427 Normals[CurrentIndex] = Normals[Index];

429 if (bWithTangent)
