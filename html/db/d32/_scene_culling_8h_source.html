232 struct FCellIndexCacheEntry
233 {
234 static constexpr uint32 SingleInstanceMask = 1u << 31;
235 static constexpr uint32 CellIndexMask = (1u << 31) - 1u;
236 static constexpr uint32 CellIndexMax = 1u << 31;
237
238 struct FItem
239 {
240 int32 NumInstances;
241 int32 CellIndex;
242 };
243
244 inline void Add(int32 CellIndex, int32 NumInstances)
245 {
246 check(uint32(CellIndex) < CellIndexMax);
247 check(!bSingleInstanceOnly || NumInstances == 1);
248
249 if (NumInstances > 1)
250 {
251 // Add RLE entry
252 Items.Add(CellIndex);
253 Items.Add(NumInstances);
254 }
255 else
256 {
257 // Mark as single-istance
258 Items.Add(uint32(CellIndex) | SingleInstanceMask);
259 }
260 }
261
265 inline void Set(int32 Index, int32 CellIndex)
266 {
267 check(bSingleInstanceOnly);
268 check(uint32(CellIndex) < CellIndexMax);
269
270 Items[Index] = uint32(CellIndex) | SingleInstanceMask;
271 }
272
273 FORCEINLINE void Reset()
274 {
275 Items.Reset();
276 }
277
278
282 FItem LoadAndStepItem(int32 &InOutItemIndex) const
283 {
284 FItem Result;
285 uint32 PackedCellIndex = Items[InOutItemIndex];
286 Result.CellIndex = int32(PackedCellIndex & FCellIndexCacheEntry::CellIndexMask);
287 Result.NumInstances = 1;
288 if ((PackedCellIndex & FCellIndexCacheEntry::SingleInstanceMask) == 0u)
289 {
290 Result.NumInstances = int32(Items[++InOutItemIndex]);
291 }
292 return Result;
293 }
294
295 bool bSingleInstanceOnly = false;
296 TArray<uint32> Items;
297 };
