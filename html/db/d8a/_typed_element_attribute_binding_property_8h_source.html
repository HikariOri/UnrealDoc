229 template <typename PropertyType>
230 PropertyType& TProperty<PropertyType>::Get(void* Object, TWeakObjectPtr<const UScriptStruct> Type) const
231 {
232 checkf(Object, TEXT("Nullptr pointer provided to Object while trying to retrieve a property value."));
233 checkf(Type.IsValid(), TEXT("Nullptr pointer provided to Type while trying to retrieve a property value."));
234 checkf(IsBound(), TEXT("Attempting to retrieve the value of object type (%s) from a property that wasn't bound."),
235 *Type->GetFName().ToString());
236 ensureMsgf(Type == ObjectTypeInfo, TEXT("Provided object type (%s) did not match bound object type (%s)."),
237 *Type->GetFName().ToString(), ObjectTypeInfo.IsValid() ? *ObjectTypeInfo->GetFName().ToString() : TEXT("<no type>"));
238
239 return *Visit([Object](auto&& Prop) { return Prop.Get(Object); }, InternalProperty);
240 }
