232// These macros are intended to allow implementing an interface with same memory footprint/performance as inheriting from
233// the abstract base class, but without mixing the interface methods with the class methods.
234//
235// It declares a member for the interface and utility functions in the host class to get the host ptr from the interface instance.
236// And thus allow declaration of the interface inside a class without requiring the interface to host a back ptr to it's owner.
237//
238//
239// Example...
240//
241// class FActorFooComponentInterface : IFooComponent
242// {
243// // Nothing but the overrides
244// virtual void OverrideSomething() override;
245// };
246//
247// class UFooComponent
248// {
249// UE_DECLARE_COMPONENT_INTERFACE(FooComponent, FActor);
250// }
251//
252// void FActorFooComponentInterface::OverrideSomething()
253// {
254// UFooComponent::GetFooComponent(this)->OverrideSomethingImplementer();
255// }
256//
257
258#define UE_DECLARE_COMPONENT_INTERFACE_INTERNAL(ThisType, ComponentInterfaceType, InterfaceType, InterfaceMember, InterfaceName)\
259 public:\
260 ComponentInterfaceType* Get##InterfaceName##Interface()\
261 {\
262 return &InterfaceMember;\
263 }\
264 const ComponentInterfaceType* Get##InterfaceName##Interface() const\
265 {\
266 return &InterfaceMember;\
267 }\
268 protected:\
269 InterfaceType InterfaceMember;\
270 \
271 static uintptr_t GetAddress(const InterfaceType* InImpl)\
272 {\
273 return reinterpret_cast<uintptr_t>(InImpl) - offsetof(ThisType, InterfaceMember);\
274 }\
275 static ThisType* Get##InterfaceName(InterfaceType* InImpl)\
276 {\
277 return reinterpret_cast<ThisType*>(GetAddress(InImpl));\
278 }\
279 static const ThisType* Get##InterfaceName(const InterfaceType* InImpl)\
280 {\
281 return reinterpret_cast<const ThisType*>(GetAddress(InImpl));\
282 }\
283 friend class InterfaceType;
284
285#define UE_DECLARE_COMPONENT_INTERFACE(ComponentName, BasePrefix)\
286 UE_DECLARE_COMPONENT_INTERFACE_INTERNAL(U##ComponentName, I##ComponentName , BasePrefix##ComponentName##Interface, ComponentName##Interface, ComponentName)
287
288#define UE_DECLARE_COMPONENT_ACTOR_INTERFACE(ComponentName)\
289 UE_DECLARE_COMPONENT_INTERFACE(ComponentName, FActor)
