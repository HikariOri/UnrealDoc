147 // corners [ (-x,-y), (x,-y), (x,y), (-x,y) ], -z, then +z
148 //
149 // 7---6 +z or 3---2 -z
150 // |\ |\ |\ |\
151 // 4-\-5 \ 0-\-1 \
152 // \ 3---2 \ 7---6
153 // \| | \| |
154 // 0---1 -z 4---5 +z
155 //
156 // @todo does this ordering make sense for UE? we are in LHS instead of RHS here
157 // if this is modified, likely need to update IndexUtil::BoxFaces and BoxFaceNormals
158
163 TVector<RealType> GetCorner(int Index) const
164 {
165 check(Index >= 0 && Index <= 7);
166 RealType dx = (((Index & 1) != 0) ^ ((Index & 2) != 0)) ? (Extents.X) : (-Extents.X);
167 RealType dy = ((Index / 2) % 2 == 0) ? (-Extents.Y) : (Extents.Y);
168 RealType dz = (Index < 4) ? (-Extents.Z) : (Extents.Z);
169 return Frame.PointAt(dx, dy, dz);
170 }

235 RealType DistanceSquared(TVector<RealType> Point) const
236 {
237 TVector<RealType> Local = Frame.ToFramePoint(Point);
238 TVector<RealType> BeyondExtents = Local.GetAbs() - Extents;
239 TVector<RealType> ClampedBeyond( // clamp negative (inside) to zero
240 FMath::Max((RealType)0, BeyondExtents.X),
241 FMath::Max((RealType)0, BeyondExtents.Y),
242 FMath::Max((RealType)0, BeyondExtents.Z));
243 return ClampedBeyond.SquaredLength();
244 }

251 RealType SignedDistance(TVector<RealType> Point) const
252 {
253 TVector<RealType> Local = Frame.ToFramePoint(Point);
254 TVector<RealType> VsExtents = Local.GetAbs() - Extents;
255 RealType MaxComponent = VsExtents.GetMax();
256 if (MaxComponent < 0) // Inside the box on every dimension
257 {
258 // The least-inside dimension gives the distance to the box surface
259 return MaxComponent;
260 }
261 else // Outside the box along at least one dimension
262 {
263 // clamp negative (inside) to zero
264 TVector<RealType> ClampedBeyond(
265 FMath::Max((RealType)0, VsExtents.X),
266 FMath::Max((RealType)0, VsExtents.Y),
267 FMath::Max((RealType)0, VsExtents.Z)
268 );
269 return ClampedBeyond.Length();
270 }

278 TVector<RealType> ClosestPoint(const TVector<RealType>& Point) const
279 {
280 TMatrix3<RealType> RotMatrix = Frame.Rotation.ToRotationMatrix();
281 TVector<RealType> FromOrigin = Point - Frame.Origin;
282 // Transform to local space
283 TVector<RealType> Local(RotMatrix.TransformByTranspose(FromOrigin));
284 // Clamp to box
285 Local.X = FMath::Clamp(Local.X, -Extents.X, Extents.X);
286 Local.Y = FMath::Clamp(Local.Y, -Extents.Y, Extents.Y);
287 Local.Z = FMath::Clamp(Local.Z, -Extents.Z, Extents.Z);
288 // Transform back
289 return Frame.Origin + RotMatrix * Local;
290 }
