19inline void MergeUniqueSortedRanges(RangeTypeA&& A, RangeTypeB&& B, RangeTypeMerged& OutMerged,
20 PredicateType Pred)
21{
22 typedef TElementType_T<RangeTypeA> ElementType;
23 typedef decltype(GetNum(A)) IndexType;
24
25 IndexType ANum = GetNum(A);
26 IndexType BNum = GetNum(B);
27 OutMerged.Reset(ANum >= BNum ? BNum : ANum);
28 const ElementType* AData = GetData(A);
29 const ElementType* BData = GetData(B);
30 const ElementType* AEnd = AData + ANum;
31 const ElementType* BEnd = BData + BNum;
32 while (AData < AEnd && BData < BEnd)
33 {
34 if (Pred(*AData, *BData))
35 {
36 OutMerged.Add(*(AData++));
37 }
38 else if (Pred(*BData, *AData))
39 {
40 OutMerged.Add(*(BData++));
41 }
42 else
43 {
44 OutMerged.Add(*(AData++));
45 ++BData;
46 }
47 }
48 while (AData < AEnd)
49 {
50 OutMerged.Add(*(AData++));
51 }
52 while (BData < BEnd)
53 {
54 OutMerged.Add(*(BData++));
55 }
56};

89template <typename RangeType, typename ModifiedIndexType, typename PredicateType>
90inline void RestoreSort(RangeType&& Range, ModifiedIndexType ModifiedIndex, PredicateType Pred)
91{
92 typedef TElementType_T<RangeType> ElementType;
93 typedef decltype(GetNum(Range)) IndexType;
94 typedef TArrayView<ElementType, IndexType> ArrayViewType;
95
96 ElementType* RangeData = GetData(Range);
97 IndexType RangeNum = GetNum(Range);
98 ArrayViewType RangeArray(RangeData, RangeNum);
99 ElementType& Modified = RangeData[ModifiedIndex];
100 if (ModifiedIndex > 0 && Pred(Modified, RangeData[ModifiedIndex - 1]))
101 {
102 ArrayViewType LowerRange = RangeArray.Left(ModifiedIndex - 1);
103 IndexType DesiredIndex = Algo::UpperBound(LowerRange, Modified, Pred);
104 ElementType Swap = MoveTemp(Modified);
105 for (IndexType SwapFromIndex = ModifiedIndex - 1; SwapFromIndex >= DesiredIndex; --SwapFromIndex)
106 {
107 RangeData[SwapFromIndex + 1] = MoveTemp(RangeData[SwapFromIndex]);
108 }
109 RangeData[DesiredIndex] = MoveTemp(Swap);
110 }
111 else if (ModifiedIndex < RangeNum - 1 && Pred(RangeData[ModifiedIndex + 1], Modified))
112 {
113 ArrayViewType UpperRange = RangeArray.RightChop(ModifiedIndex + 1);
114 IndexType DesiredIndex = ModifiedIndex + 1 + Algo::LowerBound(UpperRange, Modified, Pred);
115 ElementType Swap = MoveTemp(Modified);
116 for (IndexType SwapFromIndex = ModifiedIndex + 1; SwapFromIndex <= DesiredIndex; ++SwapFromIndex)
117 {
118 RangeData[SwapFromIndex - 1] = MoveTemp(RangeData[SwapFromIndex]);
119 }
120 RangeData[DesiredIndex] = MoveTemp(Swap);
121 }
122}
