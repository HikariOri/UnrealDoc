12// General identifiers for potential force feedback channels. These will be mapped according to the
13// platform specific implementation.
14// For example, Platform A only listens to the XXX_LARGE channels and ignores the rest, while Platform B could
15// map the XXX_LARGE to the handle motors and XXX_SMALL to the trigger motors. And others can map LEFT_SMALL to
16// its single motor.
17enum class FForceFeedbackChannelType
18{
19 LEFT_LARGE,
20 LEFT_SMALL,
21 RIGHT_LARGE,
22 RIGHT_SMALL
23};

85 {
