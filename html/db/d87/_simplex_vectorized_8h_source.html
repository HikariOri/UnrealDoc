12 FORCEINLINE_DEBUGGABLE T VectorLineSimplexFindOrigin(T* RESTRICT Simplex, int32& RESTRICT NumVerts, T& RESTRICT OutBarycentric, T* RESTRICT A, T* RESTRICT B)
13 {
14 const T& X0 = Simplex[0];
15 const T& X1 = Simplex[1];
16 const T X0ToX1 = VectorSubtract(X1, X0);
17
18 //Closest Point = (-X0 dot X1-X0) / ||(X1-X0)||^2 * (X1-X0)
19
20 const T X0ToOrigin = VectorNegate(X0);
21 const T Dot = VectorDot3(X0ToOrigin, X0ToX1);
22
23 const T IsX0 = VectorCompareGE(TVectorZero<T>(), Dot);
24
25 constexpr T OutBarycentricIfX0OrX1 = TMakeVectorRegisterConstant<T>(1, 0, 0, 0);
26 const T X0ToX1Squared = VectorDot3(X0ToX1, X0ToX1);
27 const T DotBigger = VectorCompareGE(Dot, X0ToX1Squared);
28
29 const T MinLimt = TMakeVectorRegisterConstant<T>(std::numeric_limits<FRealSingle>::min() , std::numeric_limits<FRealSingle>::min(), std::numeric_limits<FRealSingle>::min(), std::numeric_limits<FRealSingle>::min());
30 const T X0ToX1SquaredSmall = VectorCompareGE(MinLimt, X0ToX1Squared);
31 const T IsX1 = VectorBitwiseOr(DotBigger, X0ToX1SquaredSmall);
32
33 Simplex[0] = VectorSelect(IsX1, Simplex[1], Simplex[0]);
34
35 if constexpr(CalculatExtraInformation)
36 {
37 A[0] = VectorSelect(IsX1, A[1], A[0]);
38 B[0] = VectorSelect(IsX1, B[1], B[0]);
39 }
40
41 T Ratio = VectorDivide(Dot, X0ToX1Squared);
42
43 Ratio = VectorMax(Ratio, TVectorZero<T>());
44 Ratio = VectorMin(Ratio, TMakeVectorRegisterConstant<T>(1, 1, 1, 1));
45
46 T Closest = VectorMultiplyAdd(Ratio, X0ToX1, X0);
47
48 const T OneMinusRatio = VectorSubtract(TMakeVectorRegisterConstant<T>(1, 1, 1, 1), Ratio);
49 const T OutBarycentricOtherwise = VectorUnpackLo(OneMinusRatio, Ratio);
50
51 Closest = VectorSelect(IsX0, X0, VectorSelect(IsX1, X1, Closest));
52
53 const T IsX0OrX1 = VectorBitwiseOr(IsX0, IsX1);
54 NumVerts = ::VectorMaskBits(IsX0OrX1) ? 1 : NumVerts;
55
56 if constexpr (CalculatExtraInformation)
57 {
58 OutBarycentric = VectorSelect(IsX0OrX1, OutBarycentricIfX0OrX1, OutBarycentricOtherwise);
59 }
60
61 return Closest;
62 }

68 {
69 const T& A = Simplex[0];
70 const T& B = Simplex[1];
71 const T& C = Simplex[2];
72
73 const T AB = VectorSubtract(B, A);
74 const T AC = VectorSubtract(C, A);
75
76 const T TriNormal = Private::VectorCrossNoFMA(AB, AC);
77 const T TriNormal2 = VectorDot3(TriNormal, TriNormal);
78 const T MinDouble = TMakeVectorRegisterConstant<T>(std::numeric_limits<float>::min(), std::numeric_limits<float>::min(), std::numeric_limits<float>::min(), std::numeric_limits<float>::min());
79 const T AMin = VectorMultiply(A, MinDouble);
80 const T Eps2 = VectorDot3(AMin, AMin);
81 const T Eps2GENormal2 = VectorCompareGE(Eps2, TriNormal2);
82
83
84 if (::VectorMaskBits(Eps2GENormal2))
85 {
86 NumVerts = 2;
87 return VectorLineSimplexFindOrigin<T, CalculatExtraInformation>(Simplex, NumVerts, OutBarycentric, As, Bs);
88 }
89
90 // Vertex region A
91 const T AO = VectorNegate(A);
92
93 const T d1 = VectorDot3(AB, AO);
94 const T d2 = VectorDot3(AC, AO);
95
96 const T IsD1SEZero = VectorCompareGE(TVectorZero<T>(), d1);
97 const T IsD2SEZero = VectorCompareGE(TVectorZero<T>(), d2);
98 const T IsA = VectorBitwiseAnd(IsD1SEZero, IsD2SEZero);
99
100 if (::VectorMaskBits(IsA))
101 {
102 NumVerts = 1;
103 if constexpr (CalculatExtraInformation)
104 {
105 OutBarycentric = TMakeVectorRegisterConstant<T>(1, 0, 0, 0);
106 }
107 return A;
108 }
109
110 //Vertex region B
111 const T BO = VectorNegate(B);
112 const T d3 = VectorDot3(AB, BO);
113 const T d4 = VectorDot3(AC, BO);
114
115 const T IsD3GEZero = VectorCompareGE(d3, TVectorZero<T>());
116 const T IsD3GED4 = VectorCompareGE(d3, d4);
117 const T IsB = VectorBitwiseAnd(IsD3GEZero, IsD3GED4);
118
119 if (::VectorMaskBits(IsB))
120 {
121 NumVerts = 1;
122 if constexpr (CalculatExtraInformation)
123 {
124 OutBarycentric = TMakeVectorRegisterConstant<T>(1, 0, 0, 0);
125 }
126 Simplex[0] = Simplex[1];
127 if constexpr (CalculatExtraInformation)
128 {
129 As[0] = As[1];
130 Bs[0] = Bs[1];
131 }
132 return B;
133 }
134
135 // Edge AB
136 const T d1d4 = VectorMultiply(d1, d4);
137 const T vc = VectorNegateMultiplyAdd(d3, d2, d1d4);
138 const T NormalizationDenominatorAB = VectorSubtract(d1, d3);
139
140 const T IsZeroGEvc = VectorCompareGE(TVectorZero<T>(), vc);
141 const T IsD1GEZero = VectorCompareGE(d1, TVectorZero<T>());
142 const T IsZeroGED3= VectorCompareGE(TVectorZero<T>(), d3);
143 const T IsNDABGTZero = VectorCompareGT(NormalizationDenominatorAB, TVectorZero<T>());
144 const T IsAB = VectorBitwiseAnd(VectorBitwiseAnd(IsZeroGEvc, IsD1GEZero), VectorBitwiseAnd(IsZeroGED3, IsNDABGTZero));
145
146 if (::VectorMaskBits(IsAB))
147 {
148 NumVerts = 2;
149
150 const T v = VectorDivide(d1, NormalizationDenominatorAB);
151 const T OneMinusV = VectorSubtract(TMakeVectorRegisterConstant<T>(1, 1, 1, 1), v);
152 // b0 a1 a2 a3
153 if constexpr (CalculatExtraInformation)
154 {
155 OutBarycentric = VectorUnpackLo(OneMinusV, v);
156 }
157 return VectorMultiplyAdd(v, AB, A);
158 }
159
160 // Vertex C
161 const T CO = VectorNegate(C);
162 const T d5 = VectorDot3(AB, CO);
163 const T d6 = VectorDot3(AC, CO);
164 const T IsD6GEZero = VectorCompareGE(d6, TVectorZero<T>());
165 const T IsD6GED5 = VectorCompareGE(d6, d5);
166 const T IsC = VectorBitwiseAnd(IsD6GEZero, IsD6GED5);
167
168 if (::VectorMaskBits(IsC))
169 {
170 NumVerts = 1;
171 if constexpr (CalculatExtraInformation)
172 {
173 OutBarycentric = TMakeVectorRegisterConstant<T>(1, 0, 0, 0);
174 }
175
176 Simplex[0] = Simplex[2];
177 if constexpr (CalculatExtraInformation)
178 {
179 As[0] = As[2];
180 Bs[0] = Bs[2];
181 }
182 return C;
183 }
184
185 // Edge AC
186 const T d5d2 = VectorMultiply(d5, d2);
187 const T vb = VectorNegateMultiplyAdd(d1, d6, d5d2);
188 const T NormalizationDenominatorAC = VectorSubtract(d2, d6);
189
190 const T IsZeroGEvb = VectorCompareGE(TVectorZero<T>(), vb);
191 const T IsD2GEZero = VectorCompareGE(d2, TVectorZero<T>());
192 const T IsZeroGED6 = VectorCompareGE(TVectorZero<T>(), d6);
193 const T IsNDACGTZero = VectorCompareGT(NormalizationDenominatorAC, TVectorZero<T>());
194 const T IsAC = VectorBitwiseAnd(VectorBitwiseAnd(IsZeroGEvb, IsD2GEZero), VectorBitwiseAnd(IsZeroGED6, IsNDACGTZero));
195
196 if (::VectorMaskBits(IsAC))
197 {
198 const T w = VectorDivide(d2, NormalizationDenominatorAC);
199 NumVerts = 2;
200 const T OneMinusW = VectorSubtract(TMakeVectorRegisterConstant<T>(1, 1, 1, 1), w);
201 // b0 a1 a2 a3
202 if constexpr (CalculatExtraInformation)
203 {
204 OutBarycentric = VectorUnpackLo(OneMinusW, w);
205 }
206 Simplex[1] = Simplex[2];
207 if constexpr (CalculatExtraInformation)
208 {
209 As[1] = As[2];
210 Bs[1] = Bs[2];
211 }
212 return VectorMultiplyAdd(w, AC, A);
213 }
214
215 // Edge BC
216 const T d3d6 = VectorMultiply(d3, d6);
217 const T va = VectorNegateMultiplyAdd(d5, d4, d3d6);
218 const T d4MinusD3 = VectorSubtract(d4, d3);
219 const T d5MinusD6 = VectorSubtract(d5, d6);
220 const T NormalizationDenominatorBC = VectorAdd(d4MinusD3, d5MinusD6);
221
222 const T IsZeroGEva = VectorCompareGE(TVectorZero<T>(), va);
223 const T IsD4MinusD3GEZero = VectorCompareGE(d4MinusD3, TVectorZero<T>());
224 const T IsD5MinusD6GEZero = VectorCompareGE(d5MinusD6, TVectorZero<T>());
225 const T IsNDBCGTZero = VectorCompareGT(NormalizationDenominatorBC, TVectorZero<T>());
226 const T IsBC = VectorBitwiseAnd(VectorBitwiseAnd(IsZeroGEva, IsD4MinusD3GEZero), VectorBitwiseAnd(IsD5MinusD6GEZero, IsNDBCGTZero));
227
228 if (::VectorMaskBits(IsBC))
229 {
230 NumVerts = 2;
231 const T w = VectorDivide(d4MinusD3, NormalizationDenominatorBC);
232 if constexpr (CalculatExtraInformation)
233 {
234 const T OneMinusW = VectorSubtract(TMakeVectorRegisterConstant<T>(1, 1, 1, 1), w);
235 // b0 a1 a2 a3
236 OutBarycentric = VectorUnpackLo(OneMinusW, w);
237 }
238 const T CMinusB = VectorSubtract(C, B);
239 const T Result = VectorMultiplyAdd(w, CMinusB, B);
240 Simplex[0] = Simplex[1];
241 Simplex[1] = Simplex[2];
242 if constexpr (CalculatExtraInformation)
243 {
244 As[0] = As[1];
245 Bs[0] = Bs[1];
246 As[1] = As[2];
247 Bs[1] = Bs[2];
248 }
249 return Result;
250 }
251
252 // Inside triangle
253 const T denom = VectorDivide(VectorOne(), VectorAdd(va, VectorAdd(vb, vc)));
254 T v = VectorMultiply(vb, denom);
255 T w = VectorMultiply(vc, denom);
256 NumVerts = 3;
257
258 if constexpr (CalculatExtraInformation)
259 {
260 const T OneMinusVMinusW = VectorSubtract(VectorSubtract(TMakeVectorRegisterConstant<T>(1, 1, 1, 1), v), w);
261 // b0 a1 a2 a3
262 const T OneMinusVMinusW_W = VectorUnpackLo(OneMinusVMinusW, w);
263 // a0 b0 a1 b1
264 OutBarycentric = VectorUnpackLo(OneMinusVMinusW_W, v);
265 }
266
267 // We know that we are inside the triangle so we can use the projected point we calculated above.
268 // The closest point can also be derived from the barycentric coordinates, but it will contain
269 // numerical error from the determinant calculation and can cause GJK to terminate with a poor solution.
270 // (E.g., this caused jittering when walking on box with dimensions of 100000cm or more).
271 // This fix the unit test TestSmallCapsuleLargeBoxGJKRaycast_Vertical
272 // Previously was return VectorMultiplyAdd(AC, w, VectorMultiplyAdd(AB, v, A));
273 const T TriNormalOverSize2 = VectorDivide(TriNormal, TriNormal2);
274 const T SignedDistance = VectorDot3(T(A), TriNormalOverSize2);
275 return VectorMultiply(TriNormal, SignedDistance);
276 }

278 template<typename T>
279 FORCEINLINE bool VectorSignMatch(T A, T B)
280 {
281 T OneIsZero = VectorMultiply(A, B);
282 OneIsZero = ::VectorCompareEQ(OneIsZero, TVectorZero<T>());
283
284 const bool IsZero = static_cast<bool>(::VectorMaskBits(OneIsZero));
285 const int32 MaskA = ::VectorMaskBits(A);
286 const int32 MaskB = ::VectorMaskBits(B);
287 return (MaskA == MaskB) && !IsZero;
288 }
