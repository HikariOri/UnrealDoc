132 virtual bool Load(const FJsonObject& Object) override
133 {
134 EVersion Version = EVersion::Original;
135 if (!Object.TryGetNumberField(TEXT("Version"), (uint32&)Version))
136 {
137 Version = EVersion::Original;
138 }
139 if (Version > EVersion::Latest)
140 {
141 return false;
142 }
143
144 InternalName = Object.GetStringField(TEXT("InternalName"));
145 UATCommand = Object.GetStringField(TEXT("UATCommand"));
146 UATCommandLine = Object.GetStringField(TEXT("AdditionalCommandLine"));
147 bEnabled = Object.GetBoolField(TEXT("Enabled"));
148 Order = Object.GetIntegerField(TEXT("Order"));
149
150 if (Version >= EVersion::FirstVersioned)
151 {
152 Description = Object.GetStringField(TEXT("Description"));
153 }
154
155 UserTypeName = InternalName;
156 if (Version >= EVersion::AddUserTypeName)
157 {
158 UserTypeName = Object.GetStringField(TEXT("UserTypeName"));
159 }
160 return true;
161 }
