200 FTransform GetTargetTransform(const FVector& TargetOffset, FCSPose<poseType>& InPose, const FTransform& InComponentToWorld) const
201 {
202 FTransform OutTargetTransform;
203
204 auto SetComponentSpaceOffset = [](const FVector& InTargetOffset, const FTransform& LocalInComponentToWorld, FTransform& LocalOutTargetTransform)
205 {
206 LocalOutTargetTransform.SetIdentity();
207 FVector CSTargetOffset = LocalInComponentToWorld.InverseTransformPosition(InTargetOffset);
208 LocalOutTargetTransform.SetLocation(CSTargetOffset);
209 };
210
211 if (bUseSocket)
212 {
213 // this has to be done outside
214 if (SocketReference.IsValidToEvaluate())
215 {
216 FTransform SocketTransformInCS = SocketReference.GetAnimatedSocketTransform(InPose);
217
218 FVector CSTargetOffset = SocketTransformInCS.TransformPosition(TargetOffset);
219 OutTargetTransform = SocketTransformInCS;
220 OutTargetTransform.SetLocation(CSTargetOffset);
221 }
222 else
223 {
224 // if none is found, we consider this offset is world offset
225 SetComponentSpaceOffset(TargetOffset, InComponentToWorld, OutTargetTransform);
226 }
227 }
228 // if valid data is available
229 else if (BoneReference.HasValidSetup())
230 {
231 if (BoneReference.IsValidToEvaluate() &&
232 InPose.GetPose().IsValid() &&
233 ensureMsgf(InPose.GetPose().IsValidIndex(BoneReference.CachedCompactPoseIndex), TEXT("Invalid Cached Pose : Name %s(Bone Index (%d), Cached (%d))"), *BoneReference.BoneName.ToString(), BoneReference.BoneIndex, BoneReference.CachedCompactPoseIndex.GetInt()))
234 {
235 OutTargetTransform = InPose.GetComponentSpaceTransform(BoneReference.CachedCompactPoseIndex);
236 FVector CSTargetOffset = OutTargetTransform.TransformPosition(TargetOffset);
237 OutTargetTransform.SetLocation(CSTargetOffset);
238 }
239 else
240 {
241 // if none is found, we consider this offset is world offset
242 SetComponentSpaceOffset(TargetOffset, InComponentToWorld, OutTargetTransform);
243 }
244 }
245 else
246 {
247 // if none is found, we consider this offset is world offset
248 SetComponentSpaceOffset(TargetOffset, InComponentToWorld, OutTargetTransform);
249 }
250
251 return OutTargetTransform;
252 }
