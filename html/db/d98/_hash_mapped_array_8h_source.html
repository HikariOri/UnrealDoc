120 // Add an element with the specified ID to the map. Returns a reference to the newly inserted element.
121 // It is an error to insert an item twice, but this is only detected in debug builds. Use TryEmplace
122 // if you do not know whether the items already exists in the map.
123 // NOTE: since your element type will also need to contain the ID, you usually have to pass the ID twice
124 // to emplace, which is a little annoying, but shouldn't affect much.
125 template <typename... ArgsType>
126 FORCEINLINE FElementType& Emplace(const FIDType ID, ArgsType&&... Args)
127 {
128 checkSlow(Find(ID) == nullptr);
129
130 const int32 Index = Elements.Emplace(Forward<ArgsType>(Args)...);
131 const FHashType Key = FTraits::GetIDHash(ID);
132
133 HashTable.Add(Key, Index);
134
135 return Elements[Index];
136 }

138 void Remove(const FIDType ID)
139 {
140 const FHashType Key = FTraits::GetIDHash(ID);
141 for (uint32 Index = HashTable.First(Key); HashTable.IsValid(Index); Index = HashTable.Next(Index))
142 {
143 if (FTraits::GetElementID(Elements[Index]) == ID)
144 {
145 const uint32 LastElemIndex = Elements.Num() - 1;
146
147 // Remove the key/index from the hash table
148 HashTable.Remove(Key, Index);
149
150 // If this is the last element in the array (or if there's only one) then
151 // no need for fancy swaps.
152 if (Index == LastElemIndex)
153 {
154 Elements.RemoveAt(Index);
155 }
156
157 // If this is not the last element in the array, swap with the last element
158 // and fix up the hash table
159 else
160 {
161 const FIDType LastElemID = FTraits::GetElementID(Elements[LastElemIndex]);
162 const FHashType LastElemKey = FTraits::GetIDHash(LastElemID);
163
164 // Remove the last element's key/index from the hash table
165 HashTable.Remove(LastElemKey, LastElemIndex);
166
167 // Remove the element we want to remove, swapping it with the last element
168 Elements.RemoveAtSwap(Index);
169
170 // Re-add the last element's key, reusing the index of the element we just removed
171 HashTable.Add(LastElemKey, Index);
172 }
173
174 break;
175 }
176 }
177 }
