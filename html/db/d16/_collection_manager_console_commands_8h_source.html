45 void Create(const TArray<FString>& Args)
46 {
47 if ( Args.Num() < 2 )
48 {
49 UE_LOGF(LogCollectionManager, Log, "Usage: CollectionManager.Create [CollectionContainer] CollectionName CollectionType [CollectionStorageMode]");
50 return;
51 }
52
53 // Check if the last arg is a valid storage mode.
54 const ECollectionStorageMode::Type InvalidStorageMode = ECollectionStorageMode::Type(-1);
55 ECollectionStorageMode::Type StorageMode = InvalidStorageMode;
56 if (Args.Num() >= 3)
57 {
58 const FString& StorageModeStr = Args[Args.Num() - 1];
59 StorageMode = ECollectionStorageMode::FromString(*StorageModeStr, InvalidStorageMode);
60
61 // If we have four args, the last one must be the storage mode,
62 // otherwise if we have three args it may be that the first one is the collection container instead.
63 if (Args.Num() >= 4 && StorageMode == InvalidStorageMode)
64 {
65 UE_LOGF(LogCollectionManager, Warning, "Invalid collection storage mode: %ls", *StorageModeStr);
66 return;
67 }
68 }
69
70 int32 ArgIndex = 0;
71 TSharedPtr<ICollectionContainer> CollectionContainer;
72 // If we have four args, or we have three args and the last arg is not a ECollectionStorageMode, then the user should have supplied the collection container.
73 if (Args.Num() >= 4 || (Args.Num() == 3 && StorageMode == InvalidStorageMode))
74 {
75 const FString& CollectionSourceName = Args[ArgIndex++];
76 CollectionContainer = Module.Get().FindCollectionContainer(FName(CollectionSourceName, FNAME_Find));
77 if (!CollectionContainer)
78 {
79 UE_LOGF(LogCollectionManager, Warning, "Invalid collection container: %ls", *CollectionSourceName);
80 return;
81 }
82 }
83 else
84 {
85 CollectionContainer = Module.Get().GetProjectCollectionContainer();
86 }
87
88 FName CollectionName = FName(*Args[ArgIndex++]);
89
90 FString ShareStr = Args[ArgIndex++];
91 ECollectionShareType::Type ShareType;
92
93 if ( ShareStr == TEXT("LOCAL") )
94 {
95 ShareType = ECollectionShareType::CST_Local;
96 }
97 else if ( ShareStr == TEXT("PRIVATE") )
98 {
99 ShareType = ECollectionShareType::CST_Private;
100 }
101 else if ( ShareStr == TEXT("SHARED") )
102 {
103 ShareType = ECollectionShareType::CST_Shared;
104 }
105 else
106 {
107 UE_LOGF(LogCollectionManager, Warning, "Invalid collection share type: %ls", *ShareStr);
108 return;
109 }
110
111 if (StorageMode == InvalidStorageMode)
112 {
113 StorageMode = ECollectionStorageMode::Static;
114 }
115
116 if (CollectionContainer->CreateCollection(CollectionName, ShareType, StorageMode) )
117 {
118 UE_LOGF(LogCollectionManager, Log, "Collection created: %ls", *CollectionName.ToString());
119 }
120 else
121 {
122 UE_LOGF(LogCollectionManager, Warning, "Failed to create collection: %ls", *CollectionName.ToString());
123 }
124 }
