334 uint8* Add(const UScriptStruct* InScriptStruct, const FAttributeId& InAttributeId)
335 {
336 const int32 TypeIndex = FindOrAddTypeIndex(InScriptStruct);
337
338 TArray<FAttributeId>& AttributeIds = AttributeIdentifiers[TypeIndex];
339
340 const int32 ExistingAttributeIndex = AttributeIds.IndexOfByKey(InAttributeId);
341
342 // Should only add an attribute once
343 if (ensure(ExistingAttributeIndex == INDEX_NONE))
344 {
345 int32 NewIndex = INDEX_NONE;
346
347 AttributeIds.Add(InAttributeId);
348
349 UniqueTypedBoneIndices[TypeIndex].AddUnique(InAttributeId.GetIndex());
350
351 TArray<TWrappedAttribute<InAllocator>>& TypedArray = Values[TypeIndex];
352 NewIndex = TypedArray.Add(InScriptStruct);
353 TWrappedAttribute<InAllocator>& StructData = TypedArray[NewIndex];
354 InScriptStruct->InitializeStruct(StructData.template GetPtr<void>());
355
356 // Ensure arrays match in size
357 ensure(AttributeIds.Num() == TypedArray.Num());
358
359 return StructData.template GetPtr<uint8>();
360 }
361
362 return nullptr;

806 {

808 // Technically we could have avoided using const cast here if we made UniqueTypes use const UScriptStruct*
