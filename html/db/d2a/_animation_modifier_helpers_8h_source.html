83 static void EnumerateAnimationModifierTags(FStringView Tag, Func&& Callback)
84 {
85 static_assert(std::is_invocable_r_v<bool, Func, FStringView, FGuid>, "Func must be invocable (FStringView, FGuid) -> bool");
86 int Begin = 0;
87 int Mid = -1;
88 int End = -1;
89 for (size_t I = 0; I <= Tag.Len(); I++)
90 {
91 if (I == Tag.Len() || Tag[I] == UAnimationModifier::AnimationModifiersDelimiter)
92 {
93 if (I <= Begin)
94 {
95 continue; // Skip empty pair
96 }
97 End = I;
98 ensure(Mid >= Begin && End > Mid);
99 FStringView Name = Tag.SubStr(Begin, Mid - Begin);
100 FStringView Revision = Tag.SubStr(Mid + 1, End - Mid - 1);
101 FGuid RevisionGuid;
102 FGuid::Parse(FString{Revision}, RevisionGuid);
103
104 bool Break = !Invoke(Callback, Name, RevisionGuid);
105 if (Break)
106 {
107 break;
108 }
109 Begin = End + 1;
110 }
111 else if (Tag[I] == UAnimationModifier::AnimationModifiersAssignment)
112 {
113 Mid = I;
114 }
115 }
