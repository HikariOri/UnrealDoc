140struct FSkinWeightProfileStack
141{
142 static constexpr int32 MaxLayerCount = 2;
143 TStaticArray<FName, MaxLayerCount> Layers;
144
145 FSkinWeightProfileStack() :
146 Layers{InPlace, NAME_None}
147 {}
148
149 static const FSkinWeightProfileStack& GetEmpty()
150 {
151 static const FSkinWeightProfileStack EmptyStack{};
152 return EmptyStack;
153 }
154
155 // A fancy way of specifying a function with FName argument count between 1 and MaxLayerCount
156 template <
157 typename... ArgTypes
158 UE_REQUIRES((sizeof...(ArgTypes) > 0 && sizeof...(ArgTypes) <= MaxLayerCount) && UE::Core::Private::TCanBeConvertedToFromAll_V<FName, ArgTypes...>)
159 >
160 explicit FSkinWeightProfileStack(ArgTypes&&... Args)
161 : Layers{Forward<ArgTypes>(Args)...}
162 {
163 // Make sure that we mark any remaining elements are NAME_None to avoid corrupted names.
164 for (int32 LayerIndex = sizeof...(ArgTypes); LayerIndex < MaxLayerCount; ++LayerIndex)
165 {
166 Layers[LayerIndex] = NAME_None;
167 }
168 }
169
170 explicit FSkinWeightProfileStack(const FName InProfileNames[MaxLayerCount])
171 {
172 for (int32 LayerIndex = 0; LayerIndex < MaxLayerCount; ++LayerIndex)
173 {
174 Layers[LayerIndex] = InProfileNames[LayerIndex];
175 }
176 }
177
178 bool operator==(const FSkinWeightProfileStack& InProfileStack) const
179 {
180 return Layers == InProfileStack.Layers;
181 }
182
183 bool operator!=(const FSkinWeightProfileStack& InProfileStack) const
184 {
185 return Layers != InProfileStack.Layers;
186 }
187
188 FName operator[](int32 InLayerIndex) const
189 {
190 return Layers[InLayerIndex];
191 }
192
193 FName& operator[](int32 InLayerIndex)
194 {
195 return Layers[InLayerIndex];
196 }
197
198 bool IsEmpty() const
199 {
200 for (int32 LayerIndex = 0; LayerIndex < MaxLayerCount; ++LayerIndex)
201 {
202 if (!Layers[LayerIndex].IsNone())
203 {
204 return false;
205 }
206 }
207 return true;
208 }
209
210 void CopyIntoArray(FName OutProfileNames[MaxLayerCount]) const
211 {
212 for (int32 LayerIndex = 0; LayerIndex < MaxLayerCount; ++LayerIndex)
213 {
214 OutProfileNames[LayerIndex] = Layers[LayerIndex];
215 }
216 }
217
219 FSkinWeightProfileStack Normalized() const
220 {
221 FSkinWeightProfileStack NormalizedLayers;
222
223 for (int32 ReadLayer = 0, WriteLayer = 0; ReadLayer < MaxLayerCount; ++ReadLayer)
224 {
225 if (!Layers[ReadLayer].IsNone())
226 {
227 NormalizedLayers.Layers[WriteLayer++] = Layers[ReadLayer];
228 }
229 }
230 return NormalizedLayers;
231 }
232
233 FString GetUniqueId() const
234 {
235 TStringBuilder<256> IdString(InPlace, UE::String::JoinBy(Layers, [](FName InLayerName) { return InLayerName.ToString(); }, TEXT("-")));
236 return IdString.ToString();
237 }
238
239 friend uint32 GetTypeHash(const FSkinWeightProfileStack& InProfileStack)
240 {
241 uint32 Hash = 0;
242 for (int32 LayerIndex = 0; LayerIndex < MaxLayerCount; ++LayerIndex)
243 {
244 Hash = HashCombineFast(Hash, GetTypeHash(InProfileStack[LayerIndex]));
245 }
246 return Hash;
247 }
248};
