23 // There should be one value for each value in EShaderFrequency.
24 // These values are meant to be used as indices in contexts where values for different bind points can overlap (Graphics/Compute/RayTracing)
25 // like shader arrays in pipeline states or UB binding indices for Graphics (Vertex==0, Pixel==1) that can overlap with Compute (Compute==0).
26 // IMPORTANT: Adjusting these requires a full shader rebuild (ie modify the GUID in VulkanCommon.usf)

126 inline int32 GetOffsetForFrequency(EShaderFrequency Frequency)
127 {
128 // Stages that run alone or that are likely to contain more UBs start at 0
129 // The second stage follows starting at +MaxUniformBuffersPerStage
130 // The third stage returns -1 to signal descriptors are allocated from the end
131 // The sum of second and third stage must never go beyond MaxUniformBuffersPerStage
132
133 switch (Frequency)
134 {
135 case SF_Pixel: return 0;
136
137 case SF_Vertex: return MaxUniformBuffersPerStage;
138 case SF_Geometry: return -1;
139
140 case SF_Mesh: return MaxUniformBuffersPerStage;
141 case SF_Amplification: return -1;
142
143 case SF_Compute: return 0;
144
145 case SF_RayGen:
146 case SF_RayMiss:
147 case SF_RayHitGroup:
148 case SF_RayCallable: return 0;
149 default:
150 checkf(false, TEXT("Invalid shader frequency %d"), (int32)Frequency);
151 break;
152 }
153
154 return 0;
155 }

161template< class T >
162static inline void ZeroVulkanStruct(T& Struct, int32 VkStructureType)
163{
164 static_assert(!TIsPointer<T>::Value, "Don't use a pointer!");
165 static_assert(STRUCT_OFFSET(T, sType) == 0, "Assumes sType is the first member in the Vulkan type!");
166 static_assert(sizeof(T::sType) == sizeof(int32), "Assumed sType is compatible with int32!");
167 // Horrible way to coerce the compiler to not have to know what T::sType is so we can have this header not have to include vulkan.h
168 (int32&)Struct.sType = VkStructureType;
169 FMemory::Memzero(((uint8*)&Struct) + sizeof(VkStructureType), sizeof(T) - sizeof(VkStructureType));
170}
