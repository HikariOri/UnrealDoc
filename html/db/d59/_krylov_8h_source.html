196template <typename T, typename TV, typename Func1, typename Func2, typename Func3, typename Func4, typename Func5>
197void LanczosCG(
198 Func1 multiplyA,
199 Func2 dotProduct,
200 Func3 AXPY,
201 Func4 scale,
202 Func5 set,
203 TArray<TV>& x,
204 const TArray<TV>& b,
205 const int max_it,
206 const T res = T(1e-4),
207 bool check_residual = false)
208{
209 // multiplyA = [&A](TV &y, const TV& x): y = A*x;
210 // dotProduct = [](const TV& x, const TV& y): return x^Ty
211 // AXPY = [](Vector& y, T a, const TV& x): y += a*x;
212 // scale = [](Vector& y, T a): y <- a*y
213 // set = [](TV& y, const TV& x): y <- x
214 // Printing out: use
215 // (*((TV*)&b))[i]
216 // ((Vector*)&b)->operator[](i)
217
218 TArray<TV> v; v.SetNum(x.Num());
219 TArray<TV> q; q.SetNum(x.Num());
220 TArray<TV> q_old; q_old.SetNum(x.Num());
221 TArray<TV> c; c.SetNum(x.Num());
222
223 T beta = FGenericPlatformMath::Sqrt(T(dotProduct(b, b)));
224 if (beta <= res)
225 {
226 UE_LOGF(LogChaos, Warning, "Lanczos residual = %f. Lanczos converged in 0 iterations.", beta)
227 scale(x, T(0));
228 return;
229 }
230 set(q, b);
231 scale(q, T(1) / beta);
232 multiplyA(v, q);
233 T alpha = dotProduct(v, q);
234 T d = alpha;
235 T p = beta / d;
236 set(c, q);
237 set(x, c);
238 scale(x, p);
239 AXPY(v, -alpha, q);
240 T residual{};
241 TArray<TV> y; y.SetNum(x.Num());
242 for (int it = 1; it < max_it; it++)
243 {
244 beta = FGenericPlatformMath::Sqrt(T(dotProduct(v, v)));
245 if (check_residual == true)
246 {
247 multiplyA(y, x);
248 AXPY(y, T(-1), b);
249 residual = FGenericPlatformMath::Sqrt(T(dotProduct(y, y)));
250 if (residual < res)
251 {
252 UE_LOGF(LogChaos, VeryVerbose, "Ax - b residual = %g. Lanczos converged in %d iterations.", residual, it);
253 return;
254 }
255 }
256 else
257 {
258 if (beta < res)
259 {
260 UE_LOGF(LogChaos, VeryVerbose, "Lanczos residual = %g. Lanczos converged in %d iterations.", beta, it);
261 return;
262 }
263 }
264 T mu = beta / d;
265 set(q_old, q);
266 set(q, v);
267 scale(q, T(1) / beta);
268 multiplyA(v, q);
269 alpha = dotProduct(v, q);
270 T d_new = alpha - mu * mu * d;
271 p = -p * d * mu / d_new;
272 scale(c, -mu);
273 AXPY(c, T(1), q);
274 AXPY(x, p, c);
275 AXPY(v, -alpha, q);
276 AXPY(v, -beta, q_old);
277 d = d_new;
278 }
279
280 if (check_residual)
281 {
282 UE_LOGF(LogChaos, Verbose, "Lanczos used max iterations (%d). Residual = %g.", max_it, residual);
283 }
284 else
285 {
286 UE_LOGF(LogChaos, Verbose, "Lanczos used max iterations (%d).", max_it);
287 }
288}
