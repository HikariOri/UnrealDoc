65 template <typename IteratorType, typename IndexType, typename ProjectionType, typename PredicateType>
66 inline void HeapSiftDown(IteratorType Heap, IndexType Index, const IndexType Count, const ProjectionType& InProj, const PredicateType& Predicate)
67 {
68#if UE_VIRTUAL_PROJECTION_WORKAROUND
69 const ProjectionType& Proj = InProj;
70#else
71 auto&& Proj = Projection(InProj);
72#endif
73 while (!HeapIsLeaf(Index, Count))
74 {
75 const IndexType LeftChildIndex = HeapGetLeftChildIndex(Index);
76 const IndexType RightChildIndex = LeftChildIndex + 1;
77
78 IndexType MinChildIndex = LeftChildIndex;
79 if (RightChildIndex < Count)
80 {
81#if UE_VIRTUAL_PROJECTION_WORKAROUND
82 MinChildIndex = Predicate( Invoke(Proj, Heap[LeftChildIndex]), Invoke(Proj, Heap[RightChildIndex]) ) ? LeftChildIndex : RightChildIndex;
83#else
84 MinChildIndex = Predicate( Proj(Heap[LeftChildIndex]), Proj(Heap[RightChildIndex]) ) ? LeftChildIndex : RightChildIndex;
85#endif
86 }
87
88#if UE_VIRTUAL_PROJECTION_WORKAROUND
89 if (!Predicate( Invoke(Proj, Heap[MinChildIndex]), Invoke(Proj, Heap[Index]) ))
90#else
91 if (!Predicate( Proj(Heap[MinChildIndex]), Proj(Heap[Index]) ))
92#endif
93 {
94 break;
95 }
96
97 Swap(Heap[Index], Heap[MinChildIndex]);
98 Index = MinChildIndex;
99 }
100 }

113 template <class IteratorType, typename IndexType, typename ProjectionType, class PredicateType>
114 inline IndexType HeapSiftUp(IteratorType Heap, IndexType RootIndex, IndexType NodeIndex, const ProjectionType& InProj, const PredicateType& Predicate)
115 {
116#if UE_VIRTUAL_PROJECTION_WORKAROUND
117 const ProjectionType& Proj = InProj;
118#else
119 auto&& Proj = Projection(InProj);
120#endif
121 while (NodeIndex > RootIndex)
122 {
123 IndexType ParentIndex = HeapGetParentIndex(NodeIndex);
124#if UE_VIRTUAL_PROJECTION_WORKAROUND
125 if (!Predicate( Invoke(Proj, Heap[NodeIndex]), Invoke(Proj, Heap[ParentIndex]) ))
126#else
127 if (!Predicate( Proj(Heap[NodeIndex]), Proj(Heap[ParentIndex]) ))
128#endif
129 {
130 break;
131 }
132
133 Swap(Heap[NodeIndex], Heap[ParentIndex]);
134 NodeIndex = ParentIndex;
135 }
136

149 template <typename IteratorType, typename IndexType, typename ProjectionType, typename PredicateType>
150 inline void HeapifyInternal(IteratorType First, IndexType Num, ProjectionType Proj, PredicateType Predicate)
151 {
152 if constexpr (std::is_signed_v<IndexType>)
153 {
154 checkf(Num >= 0, TEXT("Algo::HeapifyInternal called with negative count"));
155 }
156
157 if (Num < 2)
158 {
159 return;
160 }
161
162 IndexType Index = HeapGetParentIndex(Num - 1);
163 for (;;)
164 {
165 HeapSiftDown(First, Index, Num, Proj, Predicate);
166 if (Index == 0)
167 {
168 return;
169 }
170 --Index;
171 }
172 }

184 void HeapSortInternal(IteratorType First, IndexType Num, ProjectionType Proj, PredicateType Predicate)
185 {
186 if constexpr (std::is_signed_v<IndexType>)
187 {

190

192 {

196 TReversePredicate< PredicateType > ReversePredicateWrapper(Predicate); // Reverse the predicate to build a max-heap instead of a min-heap
197 HeapifyInternal(First, Num, Proj, ReversePredicateWrapper);
198
199 for (IndexType Index = Num - 1; Index > 0; Index--)
200 {
201 Swap(First[0], First[Index]);
202
203 HeapSiftDown(First, (IndexType)0, Index, Proj, ReversePredicateWrapper);
