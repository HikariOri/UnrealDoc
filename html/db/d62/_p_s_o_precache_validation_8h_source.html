298 template <typename TPrecacheState>
299 bool AddStateToCache(EPSOPrecacheType PrecacheType, const TPrecacheState& PrecacheState, uint64 HashFn(const TPrecacheState&), const FMaterial* Material, int32 PSOCollectorIndex, const FVertexFactoryType* VertexFactoryType)
300 {
301 if (!IsPrecachingValidationEnabled())
302 {
303 return false;
304 }
305
306 uint64 PrecacheStateHash = HashFn(PrecacheState);
307
308 // Only update stats once per state.
309 bool bUpdateStats = false;
310 {
311 FScopeLock Lock(&StateMapLock);
312
313 FShaderStateUsage* Value = HashedStateMap.FindOrAdd(PrecacheStateHash, FShaderStateUsage());
314 if (!Value->bPrecached)
315 {
316 Value->bPrecached = true;
317 bUpdateStats = true;
318
319#if PSO_PRECACHING_TRACKING
320 Value->MaterialName = Material ? Material->GetAssetName() : FString("Unknown");
321 Value->PSOCollectorIndex = PSOCollectorIndex;
322 Value->VertexFactoryType = VertexFactoryType;
323#endif //PSO_PRECACHING_TRACKING
324 }
325 }
326
327 if (bUpdateStats)
328 {
329 Stats.PrecacheData.UpdateStats(PrecacheType, PSOCollectorIndex, VertexFactoryType);
330 }
331
332 return bUpdateStats;
333 }
