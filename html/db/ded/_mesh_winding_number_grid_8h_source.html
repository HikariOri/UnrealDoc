80 void Compute()
81 {
82 // figure out origin & dimensions
83 FAxisAlignedBox3d bounds = FastWinding->GetTree()->GetBoundingBox();
84
85 if (bounds.IsEmpty())
86 {
87 return;
88 }
89
90 float fBufferWidth = (float)BufferCells * (float)CellSize;
91 GridOrigin = (FVector3f)bounds.Min - fBufferWidth * FVector3f::One();
92 FVector3f max = (FVector3f)bounds.Max + fBufferWidth * FVector3f::One();
93 int ni = (int)((max.X - GridOrigin.X) / (float)CellSize) + 1;
94 int nj = (int)((max.Y - GridOrigin.Y) / (float)CellSize) + 1;
95 int nk = (int)((max.Z - GridOrigin.Z) / (float)CellSize) + 1;
96
97 if (ni >= 0 && nj >= 0 && nk >= 0)
98 {
99 if (ComputeMode == EComputeModes::FullGrid)
100 {
101 make_grid_dense(GridOrigin, (float)CellSize, ni, nj, nk, WindingGrid);
102 }
103 else
104 {
105 make_grid(GridOrigin, (float)CellSize, ni, nj, nk, WindingGrid);
106 }
107 }
108
109 if (!bWantMeshSDFGrid)
110 {
111 MeshSDF.Empty();
112 }
113 }

141 void make_grid(FVector3f origin, float dx, int ni, int nj, int nk, FDenseGrid3f& winding)
142 {
143 winding.Resize(ni, nj, nk);
144 winding.Assign(FMathf::MaxReal); // sentinel
145
146 // seed MWN cache
147 FastWinding->Build(false);
148
149 // Ok, because the whole idea is that the surface might have holes, we are going to
150 // compute MWN along known triangles and then propagate the computed region outwards
151 // until any MWN iso-sign-change is surrounded.
152 // To seed propagation, we compute unsigned SDF and then compute MWN for any voxels
153 // containing surface (ie w/ distance smaller than cellsize)
154
155 // compute unsigned SDF
156 MeshSDF.bComputeSigns = false;
157 MeshSDF.CancelF = CancelF;
158 MeshSDF.Compute(FastWinding->GetTree()->GetBoundingBox());
159 if (CancelF())
160 {
161 return;
162 }
163
164 FDenseGrid3f& distances = MeshSDF.Grid;
165
166 // compute MWN at surface voxels
167 double ox = (double)origin[0], oy = (double)origin[1], oz = (double)origin[2];
168 ParallelFor(nj*nk, [this, origin, dx, ni, nj, nk, &winding, &distances](int LinearJK)
169 {
170 if (CancelF())
171 {
172 return;
173 }
174 int J = LinearJK % nj, K = LinearJK / nj;
175 int StartIdx = ni * (J + nj * K);
176 for (int I = 0; I < ni; ++I)
177 {
178 int LinearIdx = StartIdx + I;
179 float dist = distances[LinearIdx];
180 // this could be tighter? but I don't think it matters...
181 if (dist < CellSize)
182 {
183 FVector3d gx((float)I * dx + origin[0], (float)J * dx + origin[1], (float)K * dx + origin[2]);
184 winding[LinearIdx] = (float)FastWinding->FastWindingNumber(gx);
185 }
186 }
187 });
188 if (CancelF())
189 {
190 return;
191 }
192
193 // Now propagate outwards from computed voxels.
194 // Current procedure is to check 26-neighbours around each 'front' voxel,
195 // and if there are any MWN sign changes, that neighbour is added to front.
196 // Front is initialized w/ all voxels we computed above
197
198 FAxisAlignedBox3i bounds = winding.Bounds();
199 bounds.Max -= FVector3i::One();
200
201 // since we will be computing new MWN values as necessary, we cannot use
202 // winding grid to track whether a voxel is 'new' or not.
203 // So, using 3D bitmap instead - is updated at end of each pass.
204 TArray<bool> bits;
205 check(winding.Num() <= MAX_int32);
206 bits.SetNumZeroed(static_cast<int32>(winding.Num()));
207
208 TArray<int> cur_front;
209 for (int32 LinearIdx = 0; LinearIdx < bits.Num(); LinearIdx++)
210 {
211 if (winding[LinearIdx] != FMathf::MaxReal)
212 {
213 cur_front.Add(LinearIdx);
214 bits[LinearIdx] = true;
215 }
216 }
217 if (CancelF())
218 {
219 return;
220 }
221
222 // Unique set of 'new' voxels to compute in next iteration.
223 TSet<int> queue;
224 FCriticalSection QueueSection;
225
226 while (true)
227 {
228 if (CancelF())
229 {
230 return;
231 }
232
233 // can process front voxels in parallel
234 bool abort = false;
235 ParallelFor(cur_front.Num(), [this, origin, dx, ni, nj, nk, &winding, &distances, &bounds, &bits, &cur_front, &queue, &QueueSection, &abort](int FrontIdx)
236 {
237 if (FrontIdx % 100 == 0)
238 {
239 abort = CancelF();
240 }
241 if (abort)
242 {
243 return;
244 }
245
246 FVector3i ijk = winding.GetCoords(cur_front[FrontIdx]);
247 float val = winding[ijk];
248
249 // check 26-neighbours to see if we have a crossing in any direction
250 for (int K = 0; K < 26; ++K) {
251 FVector3i nijk = ijk + IndexUtil::GridOffsets26[K];
252 if (bounds.Contains(nijk) == false)
253 {
254 continue;
255 }
256 checkSlow(winding.GetIndex(nijk) <= MAX_int32);
257 int LinearNIJK = static_cast<int32>(winding.GetIndex(nijk));
258 float val2 = winding[LinearNIJK];
259 if (val2 == FMathf::MaxReal) {
260 FVector3d gx((float)nijk.X * dx + origin[0], (float)nijk.Y * dx + origin[1], (float)nijk.Z * dx + origin[2]);
261 val2 = (float)FastWinding->FastWindingNumber(gx);
262 winding[LinearNIJK] = val2;
263 }
264 if (bits[LinearNIJK] == false) {
265 // this is a 'new' voxel this round.
266 // If we have a MWN-iso-crossing, add it to the front next round
267 bool crossing = (val <= WindingIsoValue && val2 > WindingIsoValue) ||
268 (val > WindingIsoValue && val2 <= WindingIsoValue);
269 if (crossing)
270 {
271 FScopeLock QueueLock(&QueueSection);
272 queue.Add(LinearNIJK);
273 }
274 }
275 }
276 });
277 if (queue.Num() == 0)
278 {
279 break;
280 }
281
282 // update known-voxels list and create front for next iteration
283 for (int LinearIdx : queue)
284 {
285 bits[LinearIdx] = true;
286 }
287 cur_front.Reset();
288 for (int idx : queue)
289 {
290 cur_front.Add(idx);
291 }
292 queue.Reset();
293 }
294
295 if (CancelF())
296 {
297 return;
298 }
299
300 // fill in the rest of the grid by propagating know MWN values
301 fill_spans(ni, nj, nk, winding);
302 }

306 void make_grid_dense(FVector3f origin, float dx, int ni, int nj, int nk, FDenseGrid3f& winding)
307 {
308 winding.Resize(ni, nj, nk);
309
310 // seed MWN cache
311 FastWinding->Build(false);
312
313 bool abort = false;
314 check(winding.Num() <= MAX_int32);
315 ParallelFor(static_cast<int32>(winding.Num()), [this, origin, dx, &winding, &abort](int LinearIdx)
316 {
317 if (LinearIdx % 100 == 0)
318 {
319 abort = CancelF();
320 }
321 if (abort)
322 {
323 return;
324 }
325 FVector3i ijk = winding.GetCoords(LinearIdx);
326 FVector3d gx((double)ijk.X * dx + origin[0], (double)ijk.Y * dx + origin[1], (double)ijk.Z * dx + origin[2]);
327 winding[LinearIdx] = (float)FastWinding->FastWindingNumber(gx);
328 });
329
330 }

333 void fill_spans(int ni, int nj, int nk, FDenseGrid3f& winding)
334 {
335 ParallelFor(nj*nk, [this, ni, nj, nk, &winding](int LinearJK)
336 {
337 int J = LinearJK % nj, K = LinearJK / nj;
338 int StartIdx = ni * (J + nj * K);
339 float last = winding[StartIdx];
340 if (last == FMathf::MaxReal)
341 {
342 last = 0;
343 }
344 for (int I = 0; I < ni; ++I) {
345 int LinearIdx = I + StartIdx;
346 if (winding[LinearIdx] == FMathf::MaxReal)
347 {
348 winding[LinearIdx] = last;
349 }
350 else
351 {
352 last = winding[LinearIdx];
353 if (last < WindingIsoValue) // propagate zeros on outside
354 {
355 last = 0;
356 }
357 }
358 }
359 });
360 }
