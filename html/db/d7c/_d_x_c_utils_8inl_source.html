82static HRESULT RetrieveDebugNameAndBlob(TRefCountPtr<IDxcResult>& CompileResult, FString& OutDebugBlobName, IDxcBlob** OutDebugBlob, DxcShaderHash& ShaderHash)
83{
84 check(CompileResult->HasOutput(DXC_OUT_SHADER_HASH));
85 {
86 TRefCountPtr<IDxcBlob> HashBlob;
87 TRefCountPtr<IDxcBlobUtf16> OutputNameUnused;
88
89 HRESULT HashResult = CompileResult->GetOutput(DXC_OUT_SHADER_HASH, __uuidof(IDxcBlob), (void**)HashBlob.GetInitReference(), OutputNameUnused.GetInitReference());
90 check(SUCCEEDED(HashResult));
91 check(HashBlob->GetBufferSize() == sizeof(DxcShaderHash));
92 ShaderHash = *reinterpret_cast<DxcShaderHash*>(HashBlob->GetBufferPointer());
93 }
94
95 HRESULT Result = S_OK;
96 if (OutDebugBlob && CompileResult->HasOutput(DXC_OUT_PDB))
97 {
98 // If the PDB blob exists, we get its name at the same time.
99 TRefCountPtr<IDxcBlobUtf16> DebugBlobName;
100 Result = CompileResult->GetOutput(DXC_OUT_PDB, __uuidof(IDxcBlob), (void**)OutDebugBlob, DebugBlobName.GetInitReference());
101 if (SUCCEEDED(Result))
102 {
103 OutDebugBlobName = DebugBlobName->GetStringPointer();
104 }
105 }
106 else
107 {
108 // If the PDB blob is not available, or if we don't need to return it, we can reconstruct the name
109 // the PDB would have used from the Shader Hash
110 TStringBuilder<128> PdbNameBuilder;
111 UE::String::BytesToHex(TConstArrayView<uint8>(ShaderHash.HashDigest, sizeof(ShaderHash.HashDigest)), PdbNameBuilder);
112 PdbNameBuilder << TEXT(".pdb");
113 OutDebugBlobName = PdbNameBuilder.ToString();
114 }
115 return Result;
116}
