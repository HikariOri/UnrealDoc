43 virtual FArchive& operator<<(FField*& InField) override
44 {
45 if (InField)
46 {
47 // Some structs (like UFunctions in their bytecode) reference properties of another UStructs.
48 // In this case we need to inspect their owner and if it's one of the objects we want to replace,
49 // replace the entire property with the one matching on the struct we want to replace it with
50 UStruct* OldOwnerStruct = InField->GetOwner<UStruct>();
51 if (OldOwnerStruct)
52 {
53 T* const* ReplaceWith = (T* const*)((const TMap<UObject*, UObject*>*)&this->ReplacementMap)->Find(OldOwnerStruct);
54 if (ReplaceWith)
55 {
56 // We want to replace the property's owner but since that would be even worse than replacing UObject's Outer
57 // we need to replace the entire property instead. We need to find the new property on the object we want to replace the Owner with
58 UStruct* NewOwnerStruct = *ReplaceWith ? CastChecked<UStruct>(*ReplaceWith) : nullptr;
59 FField* ReplaceWithField = NewOwnerStruct ? NewOwnerStruct->FindPropertyByName(InField->GetFName()) : nullptr;
60 // Do we need to verify the existence of ReplaceWithField? Theoretically it could be missing on the new version
61 // of the owner struct and in this case we still don't want to keep the stale old property pointer around so it's safer to null it
62 InField = ReplaceWithField;
63 if (this->bTrackReplacedReferences)
64 {
65 this->ReplacedReferences.FindOrAdd(OldOwnerStruct).AddUnique(this->GetSerializedProperty());
66 }
67 this->Count++;
68 }
69 // A->IsIn(A) returns false, but we don't want to NULL that reference out, so extra check here.
70 else if (OldOwnerStruct == this->SearchObject || OldOwnerStruct->IsIn(this->SearchObject))
71 {
72 bool bAlreadyAdded = false;
73 this->SerializedObjects.Add(OldOwnerStruct, &bAlreadyAdded);
74 if (!bAlreadyAdded)
75 {
76 // No recursion
77 this->PendingSerializationObjects.Add(OldOwnerStruct);
78 }
79 }
80 else if (this->bNullPrivateReferences && !OldOwnerStruct->HasAnyFlags(RF_Public))
81 {
82 checkf(false, TEXT("Can't null a reference to %s on property %s as it would be equivalent to nulling UObject's Outer."),
83 *OldOwnerStruct->GetPathName(), *InField->GetName());
84 }
85 }
86 else
87 {
88 // Just serialize the field to find any UObjects it may be referencing that we want to replace
89 InField->Serialize(*this);
90 }
91 }
92 return *this;
93 }
94};
