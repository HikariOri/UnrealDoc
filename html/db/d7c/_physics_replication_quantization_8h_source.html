79inline FQuat CanonicalizeQuatForLog(const FQuat& Q)
80{
81 // Tiebreaker chain handles the W==0 (180 degree) edge case.
82 // Beyond that just to flip when W is negative.
83 bool bFlip = false;
84 if (Q.W < 0.0) { bFlip = true; }
85 else if (Q.W == 0.0 && Q.X < 0.0) { bFlip = true; }
86 else if (Q.W == 0.0 && Q.X == 0.0 && Q.Y < 0.0) { bFlip = true; }
87 else if (Q.W == 0.0 && Q.X == 0.0 && Q.Y == 0.0 && Q.Z < 0.0) { bFlip = true; }
88 return bFlip ? FQuat(-Q.X, -Q.Y, -Q.Z, -Q.W) : Q;
89}
