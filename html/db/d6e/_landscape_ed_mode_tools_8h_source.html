29struct FNoiseParameter
30{
31 float Base;
32 float NoiseScale;
33 float NoiseAmount;
34
35 // Constructors.
36
37 FNoiseParameter()
38 {
39 }
40 FNoiseParameter(float InBase, float InScale, float InAmount) :
41 Base(InBase),
42 NoiseScale(InScale),
43 NoiseAmount(InAmount)
44 {
45 }
46
47 // Sample
48 float Sample(int32 X, int32 Y) const
49 {
50 float Noise = 0.0f;
51 X = FMath::Abs(X);
52 Y = FMath::Abs(Y);
53
54 if (NoiseScale > DELTA)
55 {
56 for (uint32 Octave = 0; Octave < 4; Octave++)
57 {
58 float OctaveShift = static_cast<float>(1 << Octave);
59 float OctaveScale = OctaveShift / NoiseScale;
60 Noise += PerlinNoise2D(X * OctaveScale, Y * OctaveScale) / OctaveShift;
61 }
62 }
63
64 return Base + Noise * NoiseAmount;
65 }
66
67 // TestGreater - Returns 1 if TestValue is greater than the parameter.
68 bool TestGreater(int32 X, int32 Y, float TestValue) const
69 {
70 float ParameterValue = Base;
71
72 if (NoiseScale > DELTA)
73 {
74 for (uint32 Octave = 0; Octave < 4; Octave++)
75 {
76 float OctaveShift = static_cast<float>(1 << Octave);
77 float OctaveAmplitude = NoiseAmount / OctaveShift;
78
79 // Attempt to avoid calculating noise if the test value is outside of the noise amplitude.
80
81 if (TestValue > ParameterValue + OctaveAmplitude)
82 return 1;
83 else if (TestValue < ParameterValue - OctaveAmplitude)
84 return 0;
85 else
86 {
87 float OctaveScale = OctaveShift / NoiseScale;
88 ParameterValue += PerlinNoise2D(X * OctaveScale, Y * OctaveScale) * OctaveAmplitude;
89 }
90 }
91 }
92
93 return TestValue >= ParameterValue;
94 }
95
96 // TestLess
97 bool TestLess(int32 X, int32 Y, float TestValue) const { return !TestGreater(X, Y, TestValue); }
98
99private:
100 static const int32 Permutations[256];
101
102 bool operator==(const FNoiseParameter& SrcNoise)
103 {
104 if ((Base == SrcNoise.Base) &&
105 (NoiseScale == SrcNoise.NoiseScale) &&
106 (NoiseAmount == SrcNoise.NoiseAmount))
107 {
108 return true;
109 }
110
111 return false;
112 }
113
114 void operator=(const FNoiseParameter& SrcNoise)
115 {
116 Base = SrcNoise.Base;
117 NoiseScale = SrcNoise.NoiseScale;
118 NoiseAmount = SrcNoise.NoiseAmount;
119 }

122 float Fade(float T) const
123 {
124 return T * T * T * (T * (T * 6 - 15) + 10);
125 }
126
127
128 float Grad(int32 Hash, float X, float Y) const
129 {
130 int32 H = Hash & 15;
131 float U = H < 8 || H == 12 || H == 13 ? X : Y,
132 V = H < 4 || H == 12 || H == 13 ? Y : 0;
133 return ((H & 1) == 0 ? U : -U) + ((H & 2) == 0 ? V : -V);
134 }
135
136 float PerlinNoise2D(float X, float Y) const
137 {
138 int32 TruncX = FMath::TruncToInt(X),
139 TruncY = FMath::TruncToInt(Y),
140 IntX = TruncX & 255,
141 IntY = TruncY & 255;
142 float FracX = X - TruncX,
143 FracY = Y - TruncY;
144
145 float U = Fade(FracX),
146 V = Fade(FracY);
147
148 int32 A = Permutations[IntX] + IntY,
149 AA = Permutations[A & 255],
150 AB = Permutations[(A + 1) & 255],
151 B = Permutations[(IntX + 1) & 255] + IntY,
152 BA = Permutations[B & 255],
153 BB = Permutations[(B + 1) & 255];
154
155 return FMath::Lerp(FMath::Lerp(Grad(Permutations[AA], FracX, FracY),
156 Grad(Permutations[BA], FracX - 1, FracY), U),
157 FMath::Lerp(Grad(Permutations[AB], FracX, FracY - 1),
158 Grad(Permutations[BB], FracX - 1, FracY - 1), U), V);
159 }
160};

168template<typename DataType>
169inline void LowPassFilter(int32 X1, int32 Y1, int32 X2, int32 Y2, FLandscapeBrushData& BrushInfo, TArray<DataType>& Data, const float DetailScale, const float ApplyRatio = 1.0f)
170{
171#if WITH_KISSFFT
172 // Low-pass filter
173 int32 FFTWidth = X2 - X1 - 1;
174 int32 FFTHeight = Y2 - Y1 - 1;
175
176 if (FFTWidth <= 1 && FFTHeight <= 1)
177 {
178 // nothing to do
179 return;
180 }
181
182 const int32 NDims = 2;
183 const int32 Dims[NDims] = { FFTHeight, FFTWidth };
184 kiss_fftnd_cfg stf = kiss_fftnd_alloc(Dims, NDims, 0, NULL, NULL),
185 sti = kiss_fftnd_alloc(Dims, NDims, 1, NULL, NULL);
186
187 kiss_fft_cpx *buf = (kiss_fft_cpx *)KISS_FFT_MALLOC(sizeof(kiss_fft_cpx) * Dims[0] * Dims[1]);
188 kiss_fft_cpx *out = (kiss_fft_cpx *)KISS_FFT_MALLOC(sizeof(kiss_fft_cpx) * Dims[0] * Dims[1]);
189
190 for (int32 Y = Y1 + 1; Y <= Y2 - 1; Y++)
191 {
192 const typename TArray<DataType>::ElementType* DataScanline = Data.GetData() + (Y - Y1) * (X2 - X1 + 1) + (0 - X1);
193 kiss_fft_cpx* BufScanline = buf + (Y - (Y1 + 1)) * Dims[1] + (0 - (X1 + 1));
194
195 for (int32 X = X1 + 1; X <= X2 - 1; X++)
196 {
197 BufScanline[X].r = DataScanline[X];
198 BufScanline[X].i = 0;
199 }
200 }
201
202 // Forward FFT
203 kiss_fftnd(stf, buf, out);
204
205 int32 CenterPos[2] = { Dims[0] >> 1, Dims[1] >> 1 };
206 for (int32 Y = 0; Y < Dims[0]; Y++)
207 {
208 float DistFromCenter;
209 for (int32 X = 0; X < Dims[1]; X++)
210 {
211 if (Y < CenterPos[0])
212 {
213 if (X < CenterPos[1])
214 {
215 // 1
216 DistFromCenter = static_cast<float>(X*X + Y*Y);
217 }
218 else
219 {
220 // 2
221 DistFromCenter = static_cast<float>((X - Dims[1])*(X - Dims[1]) + Y*Y);
222 }
223 }
224 else
225 {
226 if (X < CenterPos[1])
227 {
228 // 3
229 DistFromCenter = static_cast<float>(X*X + (Y - Dims[0])*(Y - Dims[0]));
230 }
231 else
232 {
233 // 4
234 DistFromCenter = static_cast<float>((X - Dims[1])*(X - Dims[1]) + (Y - Dims[0])*(Y - Dims[0]));
235 }
236 }
237 // High frequency removal
238 float Ratio = 1.0f - DetailScale;
239 float Dist = FMath::Min<float>((Dims[0] * Ratio)*(Dims[0] * Ratio), (Dims[1] * Ratio)*(Dims[1] * Ratio));
240 float Filter = 1.0f / (1.0f + DistFromCenter / Dist);
241 CA_SUPPRESS(6385);
242 out[X + Y*Dims[1]].r *= Filter;
243 out[X + Y*Dims[1]].i *= Filter;
244 }
245 }
246
247 // Inverse FFT
248 kiss_fftnd(sti, out, buf);
249
250 const float Scale = static_cast<float>(Dims[0] * Dims[1]);
251 const int32 BrushX1 = FMath::Max<int32>(BrushInfo.GetBounds().Min.X, X1 + 1);
252 const int32 BrushY1 = FMath::Max<int32>(BrushInfo.GetBounds().Min.Y, Y1 + 1);
253 const int32 BrushX2 = FMath::Min<int32>(BrushInfo.GetBounds().Max.X, X2);
254 const int32 BrushY2 = FMath::Min<int32>(BrushInfo.GetBounds().Max.Y, Y2);
255 for (int32 Y = BrushY1; Y < BrushY2; Y++)
256 {
257 const float* BrushScanline = BrushInfo.GetDataPtr(FIntPoint(0, Y));
258 typename TArray<DataType>::ElementType* DataScanline = Data.GetData() + (Y - Y1) * (X2 - X1 + 1) + (0 - X1);
259 const kiss_fft_cpx* BufScanline = buf + (Y - (Y1 + 1)) * Dims[1] + (0 - (X1 + 1));
260
261 for (int32 X = BrushX1; X < BrushX2; X++)
262 {
263 const float BrushValue = BrushScanline[X];
264
265 if (BrushValue > 0.0f)
266 {
267 DataScanline[X] = static_cast<DataType>(FMath::Lerp(static_cast<float>(DataScanline[X]), BufScanline[X].r / Scale, BrushValue * ApplyRatio));
268 }
269 }
270 }
271
272 // Free FFT allocation
273 KISS_FFT_FREE(stf);
274 KISS_FFT_FREE(sti);
275 KISS_FFT_FREE(buf);
276 KISS_FFT_FREE(out);
277#endif
278}

315 {
316 if (Accessor::bUseInterp)
317 {
318 // GetData alters its args, so make temp copies to avoid screwing things up
319 int32 ValidX1 = X1;
320 int32 ValidY1 = Y1;
321 int32 ValidX2 = X2;
322 int32 ValidY2 = Y2;
323 DataAccess.GetData(ValidX1, ValidY1, ValidX2, ValidY2, CachedData);
324
325 if (ValidX1 > ValidX2 || ValidY1 > ValidY2)
326 {
327 // There was no data in the requested region. GetData may nonetheless have stored extrapolated values in CachedData for the parts
328 // of the region that lie on non-existent components, discard them so that "!bIsValid" always means "the cache is empty" :
329 CachedData.Reset();
330 bIsValid = false;
331 return false;
332 }
333 }
334 else
335 {
336 DataAccess.GetDataFast(X1, Y1, X2, Y2, CachedData);
337 }

339 CachedY1 = Y1;
340 CachedX2 = X2;
341 CachedY2 = Y2;

343 // Drop a visual log to indicate the area covered by this cache region extension :

345
346 if (bCacheOriginalData)
347 {
348 OriginalData = CachedData;
349 }
350 bIsValid = true;
351 }
352 else
353 {
354 auto ExtendCache = [this, bCacheOriginalData](const int32 InX1, const int32 InY1, const int32 InX2, const int32 InY2)
355 {
356 if (Accessor::bUseInterp)
357 {
358 // GetData alters its args, so make temp copies to avoid screwing things up :
359 int32 ValidX1 = InX1;
360 int32 ValidY1 = InY1;
361 int32 ValidX2 = InX2;
362 int32 ValidY2 = InY2;

371 {

373 // Drop a visual log to indicate the area covered by this cache region extension :

375 }
376 };
377
378 // Extend the cache area if needed
379 bool bCacheExtended = false;
380 if (X1 < CachedX1)
381 {
382 ExtendCache(X1, CachedY1, CachedX1 - 1, CachedY2);
383 CachedX1 = X1;
384 bCacheExtended = true;
385 }
386
387 if (X2 > CachedX2)
388 {
389 ExtendCache(CachedX2 + 1, CachedY1, X2, CachedY2);
390 CachedX2 = X2;
391 bCacheExtended = true;
392 }

396 ExtendCache(CachedX1, Y1, CachedX2, CachedY1 - 1);

398 bCacheExtended = true;
399 }
400
401 if (Y2 > CachedY2)

405 bCacheExtended = true;
406 }
407
408 if (bCacheExtended)
409 {
410 // Drop a visual log to indicate the area covered by this cache region extension :
411 VisualizeLandscapeRegion(CachedX1, CachedY1, CachedX2, CachedY2, FColor::Red, FString::Printf(TEXT("Cache Data (X1:%i, Y1:%i, X2:%i, Y2:%i)"), CachedX1, CachedY1, CachedX2, CachedY2));
412 }
413 }
414
415 return true;

419 // Rect is half-open (exclusive), not inclusive like CacheData.
420 void CacheDataDirect(FIntRect NewDataRect, FIntRect RequestedRect, FIntRect LandscapeExtent, TConstArrayView<AccessorType> Data)
421 {
422 TRACE_CPUPROFILER_EVENT_SCOPE(TLandscapeEditCache_CacheDataDirect);
423 check(!NewDataRect.IsEmpty());
424 check(Data.Num() >= NewDataRect.Area());
425 check(LandscapeExtent.Contains(NewDataRect));
426 check(RequestedRect.Contains(NewDataRect));
427
428 // Allow caching values up to 1 pixel outside the real landscape extent.
429 FIntRect AllowedCacheExtent = LandscapeExtent;
430 AllowedCacheExtent.InflateRect(1);
431
432 FIntRect OldCachedRect(CachedX1, CachedY1, CachedX2 + 1, CachedY2 + 1);
433
434 FIntRect NewCachedRect = RequestedRect;
435 NewCachedRect.Clip(AllowedCacheExtent);
436 check(NewCachedRect.Contains(NewDataRect));
437
438 // Clamp coord to the closest valid landscape coord (exclusive extent)
439 // This replaces the complicated bUseInterp code (GetData instead of GetDataFast) for coords outside valid landscape extent. A better option
440 // might be to stop storing outside values at all and adjust coords on data lookups. That can be explored as part of removing the TMap.
441 auto ClampToLandscape = [&LandscapeExtent](FInt32Point Point) -> FInt32Point
442 {
443 return { FMath::Clamp(Point.X, LandscapeExtent.Min.X, LandscapeExtent.Max.X - 1),
444 FMath::Clamp(Point.Y, LandscapeExtent.Min.Y, LandscapeExtent.Max.Y - 1) };
445 };
446
447 int32 Stride = NewDataRect.Max.X - NewDataRect.Min.X; //Stride of the data buffer
448 CachedData.Reserve(NewCachedRect.Area());
449 for (int32 Y = NewCachedRect.Min.Y; Y < NewCachedRect.Max.Y; ++Y)
450 {
451 for (int32 X = NewCachedRect.Min.X; X < NewCachedRect.Max.X; ++X)
452 {
453 FIntPoint Coord(X, Y);
454 if (!bIsValid || !OldCachedRect.Contains(Coord))
455 {
456 FIntPoint Clamped = ClampToLandscape(Coord);
457 FIntPoint BufferRelative = Clamped - NewDataRect.Min;
458 int32 Idx = BufferRelative.Y * Stride + BufferRelative.X;
459 CachedData.Add(Coord, Data[Idx]);
460 }
461 }
462 }
463
464 bIsValid = true;
465
466 CachedX1 = NewCachedRect.Min.X;
467 CachedY1 = NewCachedRect.Min.Y;
468 CachedX2 = NewCachedRect.Max.X - 1;
469 CachedY2 = NewCachedRect.Max.Y - 1;
470
471 ensure(CachedData.Contains({ CachedX1 , CachedY1 }));
472 ensure(CachedData.Contains({ CachedX1 , CachedY2 }));
473 ensure(CachedData.Contains({ CachedX2 , CachedY1 }));
474 ensure(CachedData.Contains({ CachedX2 , CachedY2 }));

781// normal cache operation otherwise
782template<class ToolTarget>
783class FLandscapeEditLayerStackDataCache
784{
785public:
786 FLandscapeEditLayerStackDataCache(ULandscapeInfo* InLandscapeInfo, ToolTarget::CacheClass& Cache)
787 : LandscapeInfo(InLandscapeInfo)
788 , Landscape(nullptr)
789 , EditingLayerGuid()

800 {

806 void Initialize(bool bInCombinedLayerOperation)
807 {
808 check(EditingLayerGuid.IsSet()); // you must call SetCacheEditingLayer before Initialize
809 if (!bIsInitialized)
810 {
811 Landscape = LandscapeInfo ? LandscapeInfo->LandscapeActor.Get() : nullptr;
812 bCombinedLayerOperation = Landscape && bInCombinedLayerOperation && bTargetIsHeightmap;
813 if (bCombinedLayerOperation)
814 {
815 int32 I = 0;
816 for (const ULandscapeEditLayerBase* EditLayer : Landscape->GetEditLayersConst())
817 {
818 BackupLayerVisibility.Add(EditLayer->IsVisible());
819 if (EditLayer->GetGuid() == EditingLayerGuid.GetValue())
820 {
821 EditingLayerIndex = I;
822 }
823 ++I;

831 // Reads the values in the specified rectangle into Data.
832 // Returns false if there was no data in the requested region, in which case the underlying caches are left invalid and Data is left untouched.
833 [[nodiscard]] bool Read(int32 X1, int32 Y1, int32 X2, int32 Y2, TArray<typename ToolTarget::CacheClass::DataType>& Data)
834 {
835 check(bIsInitialized);
836 if (bCombinedLayerOperation)
837 {
838 TBitArray<> NewLayerVisibility;
839 int32 I = 0;
840 for (const ULandscapeEditLayerBase* EditLayer : Landscape->GetEditLayersConst())
841 {
842 NewLayerVisibility.Add((I > EditingLayerIndex) ? false : EditLayer->IsVisible());
843 ++I;
844 }
845
846 // temporarily switch to working on the final runtime data, so we can gather the combined layer data into the caches
847 FGuid PreviousLayerGUID = EditingLayerGuid.GetValue();
848 SetCacheEditingLayer(FGuid());
849
850 bool bSuccess = CacheUpToEditingLayer.GetDataAndCacheWithVisibilityOverride(X1, Y1, X2, Y2, Data, NewLayerVisibility, Landscape);
851 if (bSuccess)
852 {
853 // Now turn off visibility on the current layer in order to have the data of all bottom layers except the current one
854 NewLayerVisibility[EditingLayerIndex] = false;
855 // Both caches must succeed : Write assumes Data and BottomLayersData describe the same region
856 bSuccess = CacheBottomLayers.GetDataAndCacheWithVisibilityOverride(X1, Y1, X2, Y2, BottomLayersData, NewLayerVisibility, Landscape);
857 }
858
859 // Restore the editing layer even on failure, otherwise the caches are left pointing at the runtime data and Write would fail its check
860 SetCacheEditingLayer(PreviousLayerGUID);
861 check(PreviousLayerGUID == CacheUpToEditingLayer.DataAccess.GetEditLayer());
862
863 return bSuccess;
864 }
865 else
866 {
867 if (!CacheUpToEditingLayer.CacheData(X1, Y1, X2, Y2))
868 {
869 return false;
870 }
871 CacheUpToEditingLayer.GetCachedData(X1, Y1, X2, Y2, Data);
872 return true;
873 }
874 }

876 void Write(int32 X1, int32 Y1, int32 X2, int32 Y2, TArray<typename ToolTarget::CacheClass::DataType>& Data, ELandscapeLayerPaintingRestriction PaintingRestriction = ELandscapeLayerPaintingRestriction::None)
877 {
878 check(bIsInitialized);
879 if (bCombinedLayerOperation)
880 {
881 const ULandscapeEditLayerBase* EditLayer = Landscape->GetEditLayerConst(EditingLayerIndex);
882 check(EditLayer != nullptr);
883 check(bTargetIsHeightmap);
884 const float Alpha = EditLayer->GetAlphaForTargetType(ELandscapeToolTargetType::Heightmap);
885 const float InverseAlpha = (Alpha != 0.f) ? 1.f / Alpha : 1.f;
886 TArray<typename ToolTarget::CacheClass::DataType> DataContribution;
887 DataContribution.Empty(Data.Num());
888 DataContribution.AddUninitialized(Data.Num());
889 check(Data.Num() == BottomLayersData.Num());
890 for (int i = 0; i < Data.Num(); ++i)
891 {
892 float Contribution = (LandscapeDataAccess::GetLocalHeight(Data[i]) - LandscapeDataAccess::GetLocalHeight(BottomLayersData[i])) * InverseAlpha;
893 DataContribution[i] = static_cast<typename ToolTarget::CacheClass::DataType>(LandscapeDataAccess::GetTexHeight(Contribution));
894 }
895
896 FGuid CacheAccessorLayerGuid = CacheUpToEditingLayer.DataAccess.GetEditLayer();
897 checkf(EditingLayerGuid.GetValue() == CacheAccessorLayerGuid, TEXT("Editing Layer has changed between Initialize and Write. Was: %s (%s). Is now: %s (%s)"),
898 Landscape->GetEditLayerConst(*EditingLayerGuid) ? *(Landscape->GetEditLayerConst(*EditingLayerGuid)->GetName().ToString()) : TEXT("<unknown>"), *EditingLayerGuid->ToString(),
899 Landscape->GetEditLayerConst(CacheAccessorLayerGuid) ? *(Landscape->GetEditLayerConst(CacheAccessorLayerGuid)->GetName().ToString()) : TEXT("<unknown>"), *CacheAccessorLayerGuid.ToString());
900
901 CacheUpToEditingLayer.SetCachedData(X1, Y1, X2, Y2, Data, PaintingRestriction, false);
902 // Effectively write the contribution
903 CacheUpToEditingLayer.DataAccess.SetData(X1, Y1, X2, Y2, DataContribution.GetData(), PaintingRestriction);
904 CacheUpToEditingLayer.DataAccess.Flush();
905 }
906 else
907 {
908 CacheUpToEditingLayer.SetCachedData(X1, Y1, X2, Y2, Data, PaintingRestriction);
909 CacheUpToEditingLayer.Flush();
910 }
911 }
