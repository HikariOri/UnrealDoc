139 static TArray<FString> GetExplicitBuildTargetsForProject( const FString& InProjectFilePath, const TSet<EBuildTargetType>* InFilterTargetTypes = nullptr )
140 {
141 TArray<FString> ExplicitBuildTargetNames;
142
143 if (!InProjectFilePath.IsEmpty())
144 {
145 // ignore the filter if it's empty
146 if (InFilterTargetTypes != nullptr && InFilterTargetTypes->Num() == 0)
147 {
148 InFilterTargetTypes = nullptr;
149 }
150
151 // collect the build targets for the given project
152 TMap<EBuildTargetType,TArray<FString>> BuildTargetNamesByType;
153 const TArray<FTargetInfo>& Targets = FDesktopPlatformModule::Get()->GetTargetsForProject(InProjectFilePath);
154 for (const FTargetInfo& Target : Targets)
155 {
156 if (Target.Type != EBuildTargetType::Game && Target.Type != EBuildTargetType::Client && Target.Type != EBuildTargetType::Server)
157 {
158 continue;
159 }
160
161 if (InFilterTargetTypes != nullptr && !InFilterTargetTypes->Contains(Target.Type) )
162 {
163 continue;
164 }
165
166 BuildTargetNamesByType.FindOrAdd(Target.Type).Add(Target.Name);
167 }
168
169 // create a list of build targets that need to be specified explicitly (because there is more than one of the same type)
170 for (auto& Itr : BuildTargetNamesByType)
171 {
172 if (Itr.Value.Num() > 1)
173 {
174 ExplicitBuildTargetNames.Append(Itr.Value);
175 }
176 }
177 }
178
179 return MoveTemp(ExplicitBuildTargetNames);
180 }
