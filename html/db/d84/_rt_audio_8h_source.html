697 S24& operator = ( const int& i ) {
698 c3[0] = (i & 0x000000ff);
699 c3[1] = (i & 0x0000ff00) >> 8;
700 c3[2] = (i & 0x00ff0000) >> 16;
701 return *this;

711 int i = c3[0] | (c3[1] << 8) | (c3[2] << 16);
712 if (i & 0x800000) i |= ~0xffffff;
