1529 {
1530 for (TMultiMap<uint32, int32>::TConstKeyIterator It = HashToInstanceIndexMap.CreateConstKeyIterator(Hash); It; ++It)
1531 {
1532 TSharedStructType& Instance = Instances[It.Value()];
1533
1534 if (Instance.GetScriptStruct() == Type

1536 {
1537 return Instance;

1539 }

1542 HashToInstanceIndexMap.Add(Hash, Index);

1546 template<typename T, typename... TArgs>
1547 TSharedStructType& FindOrAdd(const uint32 Hash, TArgs&&... InArgs)
1548 {
1549 // Need to actually construct the struct to make proper comparison to possible existing instance
1550 TSharedStructType TempInstance = TSharedStructType::template Make<T>(Forward<TArgs>(InArgs)...);
1551
1552 for (TMultiMap<uint32, int32>::TConstKeyIterator It = HashToInstanceIndexMap.CreateConstKeyIterator(Hash); It; ++It)
1553 {
1554 TSharedStructType& Instance = Instances[It.Value()];
1555
1556 if (Instance.GetScriptStruct() == T::StaticStruct()
1557 && T::StaticStruct()->CompareScriptStruct(Instance.GetMemory(), TempInstance.GetMemory(), PPF_None))
1558 {
1559 return Instance;
1560 }
1561 }
1562
1563 int32 Index = Add(MoveTemp(TempInstance));
1564 HashToInstanceIndexMap.Add(Hash, Index);
1565 return Instances[Index];
1566 }
