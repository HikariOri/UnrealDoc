301 FORCENOINLINE void AtomicallySetFlags( EObjectFlags FlagsToAdd )
302 {
303 int32 OldFlags = GetFlagsInternal();
304 int32 NewFlags = OldFlags | FlagsToAdd;
305
306 // Fast path without atomics if already set
307 if (NewFlags == OldFlags)
308 {
309 return;
310 }
311
312 UE_AUTORTFM_OPEN
313 {
314 FPlatformAtomics::InterlockedOr((int32*)&ObjectFlags, FlagsToAdd);
315 };
316
317 // If we abort we undo setting the flags we just set.
318 AutoRTFM::OnAbort([this, OldFlags, FlagsToAdd]
319 {
320 int32 MaskFlags = OldFlags;
321
322 // Now just extract out the old flags that mattered (the ones we were setting).
323 MaskFlags &= FlagsToAdd;
324 // And unmask the flags we didn't mention.
325 MaskFlags |= ~FlagsToAdd;
326
327 FPlatformAtomics::InterlockedAnd((int32*)&ObjectFlags, MaskFlags);
328 });
329 }
330
334 FORCENOINLINE void AtomicallyClearFlags( EObjectFlags FlagsToClear )
335 {
336 int32 OldFlags = GetFlagsInternal();
337 int32 NewFlags = OldFlags & ~FlagsToClear;
338
339 // Fast path without atomics if already cleared
340 if (NewFlags == OldFlags)
341 {
342 return;
343 }
344
345 UE_AUTORTFM_OPEN
346 {
347 FPlatformAtomics::InterlockedAnd((int32*)&ObjectFlags, ~FlagsToClear);
348 };
349
350 // If we abort we undo clearing the flags we just unset.
351 AutoRTFM::OnAbort([this, OldFlags, FlagsToClear]
352 {
353 int32 MaskFlags = OldFlags;
354
355 // Now just extract out the old flags that mattered (the ones we were setting).
356 MaskFlags &= FlagsToClear;
357
358 FPlatformAtomics::InterlockedOr((int32*)&ObjectFlags, MaskFlags);
359 });
360 }

366 FORCEINLINE int32 GetFlagsInternal() const
367 {
368 static_assert(sizeof(int32) == sizeof(ObjectFlags), "Flags must be 32-bit for atomics.");
369
370 int32 Result = 0;
371
372 UE_AUTORTFM_OPEN
373 {
374 Result = ((std::atomic<int32>*)&ObjectFlags)->load(std::memory_order_relaxed);
375 };
376
377 return Result;
378 }
379
380 FORCEINLINE int32 GetFlagsInternal(std::memory_order InMemoryOrder) const
381 {
382 static_assert(sizeof(int32) == sizeof(ObjectFlags), "Flags must be 32-bit for atomics.");
383
384 int32 Result = 0;
385
386 UE_AUTORTFM_OPEN
387 {
388 // Make this load use memory ordering so that we can rely on looking at flags
389 // as a barrier between the thread setting a flag and a thread taking a decision
390 // depending on that flag.
391 Result = ((std::atomic<int32>*)&ObjectFlags)->load(InMemoryOrder);
392 };
393
394 return Result;
395 }
