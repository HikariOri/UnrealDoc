113 virtual bool Init() override
114 {
115 bool bInitSuccessful = false;
116
117 {
118 InitMutex.Lock();
119 ON_SCOPE_EXIT
120 {
121 InitMutex.Unlock();
122 };
123
124 Traits::CheckNotInitialized(EventLoopThreadId);
125 LoopTime = Traits::GetCurrentTime();
126 TimerManager.Init();
127 bInitSuccessful = IOManager.Init();
128 EventLoopThreadId = Traits::GetCurrentThreadId();
129 }
130
131 if (bNotifyOnInit)
132 {
133 IOManager.Notify();
134 }
135

137 }

139 virtual void RequestShutdown(FOnShutdownComplete&& OnShutdownComplete = FOnShutdownComplete()) override
140 {
141 bool bRunLocally = false;
142
143 {
144 ShutdownMutex.Lock();
145 ON_SCOPE_EXIT
146 {
147 ShutdownMutex.Unlock();
148 };
149
150 // Check if this call to RequestShutdown is the first call.
151 bool bExpectedValue = false;
152 if (bShutdownRequested.CompareExchange(bExpectedValue, true))
153 {
154 // This is the first call to RequestShutdown. Notify the request manager to wake
155 // it so that the loop will see the request.
156 ShutdownRequests.Add(MoveTemp(OnShutdownComplete));
157 TryNotify();
158 }
159 else if (!bShutdownCompleted)
160 {
161 // Shutdown has been requested, but not yet acknowledged.
162 ShutdownRequests.Add(MoveTemp(OnShutdownComplete));
163 }
164 else
165 {
166 // Loop shutdown has already completed. Run the callback locally outside of the lock.
167 bRunLocally = true;
168 }
169 }
170
171 // Loop shutdown has already completed.
172 if (bRunLocally && OnShutdownComplete)
173 {
174 OnShutdownComplete();
175 }
176 }

201 virtual void Run() override
202 {
203 QUICK_SCOPE_CYCLE_COUNTER(STAT_EventLoop_TEventLoop_Run);
204 Traits::CheckInitialized(EventLoopThreadId);
205
206 while (true)
207 {
208 const FTimespan InfiniteWait(std::numeric_limits<int64>::max());
209 FTimespan WaitTime = InfiniteWait;
210
211 if (!RunOnce(WaitTime))
212 {
213 break;
214 }
215 };

218 virtual bool RunOnce(FTimespan WaitTime) override
219 {
220 QUICK_SCOPE_CYCLE_COUNTER(STAT_EventLoop_TEventLoop_RunOnce);
221 Traits::CheckInitialized(EventLoopThreadId);
222
223 if (bShutdownCompleted)
224 {
225 // Terminated.
226 return false;
227 }
228
229 // Check if there are any running timers. The maximum wait time will then be shortened
230 // to the expiration of the next timer.
231 FTimespan TimerExpirationTime;
232 if (TimerManager.GetNextTimeout(TimerExpirationTime))
233 {
234 WaitTime = FMath::Min(WaitTime, TimerExpirationTime);
235 }
236
237 IOManager.Poll(WaitTime);
238
239 // Run timers.
240 FTimespan CurrentTime = Traits::GetCurrentTime();
241 FTimespan ElapsedTime = FMath::Max(CurrentTime - LoopTime, 0.0);
242 LoopTime = CurrentTime;
243 TimerManager.Tick(ElapsedTime);
244
245 // If any timers ran due to the above call to Tick, check whether any repeat timers are
246 // waiting to be rescheduled. Ticking with zero delta reschedules them into the heap
247 // without advancing internal time, which prevents short-interval timers from
248 // immediately re-firing in the same iteration.
249 if (TimerManager.HasPendingRepeatTimer())
250 {
251 TimerManager.Tick(FTimespan::Zero());
252 }
253
254 RunAsyncTasks();
255
256 if (Traits::IsShutdownRequested(bShutdownRequested))
257 {
258 IOManager.Shutdown();
259
260 TArray<FOnShutdownComplete> LocalShutdownRequests;
261 {
262 ShutdownMutex.Lock();
263 ON_SCOPE_EXIT
264 {
265 ShutdownMutex.Unlock();
266 };
267
268 bShutdownCompleted = true;
269 LocalShutdownRequests = MoveTemp(ShutdownRequests);
270 }
271
272 // Run post-shutdown tasks.
273 Algo::ForEach(LocalShutdownRequests, [](FOnShutdownComplete& OnShutdownComplete){
274 if (OnShutdownComplete)
275 {
276 OnShutdownComplete();
277 }
278 });
279
280 // Terminated.
281 return false;
282 }
283
284 // Continue running.
285 return true;
286 }
287
288 virtual FTimespan GetLoopTime() const override
289 {
290 Traits::CheckIsEventLoopThread(EventLoopThreadId);
291 return LoopTime;
292 }
293
294private:
295 void RunAsyncTasks()
296 {
297 FAsyncTask AsyncTask;
298 while (AsyncTasks.Dequeue(AsyncTask))
299 {
300 AsyncTask();
301 }
302 }
303
304 void TryNotify()
305 {
306 QUICK_SCOPE_CYCLE_COUNTER(STAT_TEventLoop_TryNotify);
307
308 if (Traits::IsInitialized(EventLoopThreadId))
309 {
310 // Init already completed. Notify the IO manager.
311 IOManager.Notify();
312 }
313 else
314 {
315 InitMutex.Lock();
316 ON_SCOPE_EXIT
317 {
318 InitMutex.Unlock();
319 };
320
321 if (!Traits::IsInitialized(EventLoopThreadId))
322 {
323 // Init has not yet started. Let it know to notify the IO manager of
324 // pending actions once it has completed.
325 bNotifyOnInit = true;
326 }
327 else
328 {
329 // Init competed before the lock was acquired. Notify the IO manager.
330 IOManager.Notify();
331 }
332 }
333 }
334};
