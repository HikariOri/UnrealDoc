292 void CachedBindUniformBuffer(GLuint Buffer)
293 {
294 check(FOpenGL::SupportsUniformBuffers());
295
296 VERIFY_GL_SCOPE();
297 if (ContextState.UniformBufferBound != Buffer)
298 {
299 glBindBuffer(GL_UNIFORM_BUFFER, Buffer);
300 ContextState.UniformBufferBound = Buffer;
301 }
302 }
303
304 void CachedBindBuffer(GLenum Type, GLuint Buffer)
305 {
306 VERIFY_GL_SCOPE();
307 switch (Type)
308 {
309 case GL_ARRAY_BUFFER:
310 if (ContextState.ArrayBufferBound != Buffer)
311 {
312 glBindBuffer(GL_ARRAY_BUFFER, Buffer);
313 ContextState.ArrayBufferBound = Buffer;
314 }
315 break;
316
317 case GL_ELEMENT_ARRAY_BUFFER:
318 if (ContextState.ElementArrayBufferBound != Buffer)
319 {
320 glBindBuffer(GL_ELEMENT_ARRAY_BUFFER, Buffer);
321 ContextState.ElementArrayBufferBound = Buffer;
322 }
323 break;
324
325 case GL_SHADER_STORAGE_BUFFER:
326 if (ContextState.StorageBufferBound != Buffer)
327 {
328 glBindBuffer(GL_SHADER_STORAGE_BUFFER, Buffer);
329 ContextState.StorageBufferBound = Buffer;
330 }
331 break;
332
333 case GL_PIXEL_UNPACK_BUFFER:
334 if (ContextState.PixelUnpackBufferBound != Buffer)
335 {
336 glBindBuffer(GL_PIXEL_UNPACK_BUFFER, Buffer);
337 ContextState.PixelUnpackBufferBound = Buffer;
338 }
339 break;
340
341 default:
342 checkNoEntry();
343 break;
344 }
345 }

390 FORCEINLINE_DEBUGGABLE void Lock(void* RHIBuffer, void* Buffer, uint32 ArrayIndex, uint32 MipIndex, uint32 Stride, uint32 SizeRHI, EResourceLockMode LockMode)
391 {
392//#if DO_CHECK
393 for (auto& Parms : OutstandingLocks)
394 {
395 check(Parms.RHIBuffer != RHIBuffer || Parms.MipIndex != MipIndex || Parms.ArrayIndex != ArrayIndex);
396 }
397//#endif
398 OutstandingLocks.Add(FLockParams(RHIBuffer, Buffer, ArrayIndex, MipIndex, Stride, SizeRHI, LockMode));
399 }

401 FORCEINLINE_DEBUGGABLE FLockParams Unlock(void* RHIBuffer, uint32 ArrayIndex, uint32 MipIndex)
402 {
403 for (int32 Index = 0; Index < OutstandingLocks.Num(); Index++)
