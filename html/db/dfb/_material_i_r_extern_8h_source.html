44// Set of user-provided flags used by extern instructions.
45enum class EExternFlags : uint8
46{
47 None = 0,
48
49 // Hint the IR-to-shader-code translator that any generated code does not need to get
50 // its own temporary and can be inlined as a subexpression of another expression.
51 // For instance, by turning this flag on, the generator can nest "MyFunc()" inside the + operator call as in
52 //
53 // float Result = SomeOtherValue + MyFunc();
54 //
55 // instead of evaluating MyFunc() into its own temporary as in
56 //
57 // float _14 = MyFunc();
58 // float Result = SomeOtherValue + _14;
59 //
60 // Note: This is only a hint for readability. Also, the translator will only inline the expression if it has
61 // only one user, meaning it is not shared computation used by more than one other instruction.
62 Inline = 1 << 0,
63
64 // This extern does not support analytical screen-space partial differential. Any attempt to evaluate its
65 // differential will result in a translation error generated by the Emitter.
66 NoDifferentials = 1 << 1,
67
68 //
69 ZeroDifferentials = 1 << 2,
70};

96// This concept represents the static interface of a user-specified extern instruction definition.
97// In traditional C++, this concept maps to the following interface:
98//
99// struct FMyExtern
100// {
101// // Required. Returns a description of this extern. See `FExternInfo` for more info.
102// MIR::FExternInfo GetInfo() const;
103//
104// // Optional. Invoked by the MIR module builder when this instruction is found to be
105// // necessary and needs to perform semantic analysis. The builder guarantees the
106// // following properties:
107// // - This method is invoked at most once.
108// // - This method is invoked only after all its dependencies (meaning this extern's arguments)
109// // have been analyzed successfully.
110// // - When this method is invoked, MIR graph properties have already been propagated
111// // (see MIR::EGraphProperties for more info).
112// // If your extern needs to perform validation or generate side-effect data, for example
113// // by setting state on the MaterialCompilationOutput instance provided by the Module (inside `State`),
114// // or by adding new shader environment defines, this is the method to put your logic in.
115// // If you detect any invalid usage, you can push errors to the Module using the `::AddError()` method.
116// void Analyze(FExternAnalysisContext& Context);
117//
118// // Optional. This method is invoked immediately after `Analyze`. The difference with `Analyze` is that
119// // this method is invoked once for each shader stage (vertex, pixel, compute) this extern instruction
120// // is proven to be evaluated in.
121// // If this method is not invoked for a particular stage, it means that the IR module builder could not
122// // find a case where this instruction is actually necessary to compute a needed value in that stage.
123// void AnalyzeInStage(FExternAnalysisContext& Context, EStage Stage);
124//
125// // Required. This method is invoked by the IR-to-HLSL translator to translate this extern instruction
126// // into HLSL.
127// // Example: `Printer << TEXTVIEW("Example(") << Printer.Arg(0) << TEXTVIEW(")")`
128// // For more information see `FExternPrinterHLSL`.
129// // Note: `Printer.Differential` specifies whether the original value or any analytical screen-space partial
130// // differential is being generated.
131// // Important: You should not insert any validation or analysis code into this method, as it should
132// // have all been performed in the `AnalyzeXXX` methods above. This method should contain
133// // simple and straightforward string chunk emission logic through the provided Printer.
134// void ToHLSL(FExternPrinterHLSL& Printer) const;
135//
136// // Optional. Invoked by MIR debug printers to display the parameters in your specific extern definition.
137// // In doing so, please use the "Property1=value, Property2=value" syntax for consistency with non-extern
138// // MIR instruction formatting.
139// //
140// // For instance, if your extern definition struct has `const TCHAR* ID` and a `uint32 Num` fields, you
141// // should display them as
142// //
143// // ID="Foo", Num=42
144// //
145// void EmitDebugInfo(FString& Out) const;
146// };
147//
148// Your extern definition must be trivially destructible and copyable. In other words, once constructed,
149// it must behave as inert Plain Old Data. This requirement comes from MIR itself: all MIR nodes follow this design,
150// as their bit pattern fully identifies them and their bit pattern is automatically made unique by the emitter.
151// If you need non-trivial data to be created, referenced, or destroyed, use an external manager to handle
152// its creation and lifetime.
153//
154// Note: Optional methods are not required to be defined. Not defining them is equivalent to defining them
155// and leaving their implementation empty.
156//
157// For reference, here is a minimal extern definition example:
158//
159// enum class EColorConversionExtern
160// {
161// HSV,
162// ... /* other modes */
163// };
164//
165// struct FColorConversionExtern
166// {
167// EColorConversionExtern Mode;
168//
169// MIR::FExternInfo GetInfo() const
170// {
171// return
172// {
173// .Name = TEXT("ColorConversionExtern"),
174// .Type = MIR::FType::MakeFloatVector(4),
175// .Flags = MIR::EExternFlags::Inline,
176// };
177// }
178//
179// void ToHLSL(MIR::FExternPrinterHLSL& Printer) const
180// {
181// switch (Mode)
182// {
183// case EColorConversionExtern::HSV: Printer << TEXTVIEW("RgbToHsv"); break;
184// ... /* other modes */
185// }
186// Printer << MIR::ExternTag_DD << TEXT('(') << Printer.Arg(0) << TEXT(')'); // ex: "RgbToHsvDDX(<arg0>)"
187// }
188//
189// void EmitDebugInfo(FString& Out) const
190// {
191// Out.Append(TEXTVIEW("Mode="));
192// switch (Mode)
193// {
194// case EColorConversionExtern::HSV: Out.Append(TEXTVIEW("ToHSV")); break;
195// ... /* other modes */
196// }
197// }
198//
199// // Optional. If your definition struct does not satisfy std::has_unique_object_representations_v,
200// // you must implement CopyTo to copy each member individually into Other. The emitter writes
201// // your definition into a zero-initialized buffer via CopyTo so that padding bytes stay zero,
202// // ensuring deterministic hashing. Note: the trait is also false for structs containing floats
203// // (due to +0/-0 and NaN aliasing), even if they have no padding — C++ does not offer a trait
204// // to detect padding alone, so this is the best we can do. In that case CopyTo is still
205// // required but is a straightforward member-by-member copy.
206// void CopyTo(FColorConversionExtern& Other) const
207// {
208// Other.Mode = Mode;
209// }
210// };
211//
212template <typename T>
213concept CExternDefinition = requires(const T& Self, FExternPrinterHLSL& Printer)
214{
215 requires std::is_trivially_destructible_v<T>;
216 requires std::is_trivially_copy_constructible_v<T>;
217 requires std::is_trivially_copy_assignable_v<T>;
218 { Self.GetInfo() } -> std::same_as<MIR::FExternInfo>;
219 { Self.ToHLSL(Printer) } -> std::same_as<void>;
220};

272// Helper function to create a material domain mask from a static list.
273// Use when initializing `FExternInline::MaterialDomainMask`, example:
274// MakeDomainMask(MD_Opaque, MD_DeferredDecal);
275template <typename... TMaterialDomains>
276requires (std::same_as<TMaterialDomains, EMaterialDomain> && ...)
277static constexpr uint32 MakeDomainMask(TMaterialDomains... Domains)
278{
279 return (0u | ... | (1u << static_cast<uint32>(Domains)));

282// It wraps all the data necessary to implement an extern definition through static constant data.
283// You must make sure that referenced instances of this struct are alive throughout material translation.
284// It is recommended to locally declare a static const instance of this struct and use it by emitting
285// a FExternFromInlineDecl. Example:
286//
287// static const FExternInlineDeclaration MySpecialExtern
288// {
289// .Type = MIR::FType::MakeFloatVector(3),
290// .CodeHLSL = TEXT("GetMySpecialValue($0)"),
291// .MaterialDomainMask = MIR::MakeDomainMask(MD_Opaque),
292// };
293// return Em.Extern<FExternFromInlineDecl>(&MySpecialExtern);
294//
295struct FExternInlineDeclaration
296{
297 // The type this extern value evaluates to.
298 MIR::FType Type{};
299
300 // The type this extern value's screen-space partial differential evaluate to, if defined.
301 // Note: You can leave this undefined. If you did not specify in `Flags` that this extern
302 // does not admit analytical partial differential, the emitter will reuse the non-differential
303 // `Type` instead.
304 MIR::FType DifferentialType{};
305
306 // The set of behavioral flags this extern adheres to.
307 MIR::EExternFlags Flags = MIR::EExternFlags::Inline | MIR::EExternFlags::ZeroDifferentials;
308
309 // The HLSL code snippet this extern evaluates to. This field is used when the printer
310 // `Differential` field is `Value` (the original, non-differential value).
311 // This string snippet supports various special escape patterns for convenience:
312 // - Arguments: You can use tags like "$2" as placeholders for the extern instruction
313 // argument value with index 2. For example, if this extern instruction was emitted
314 // with a single argument that happens to be a constant of value 42, the snippet
315 // `Foo(Bar, $0)` will be interpolated to `Foo(Bar, 42)`.
316 // - `<PREV>`: This tag will automatically be replaced with `Prev` while emitting code
317 // for the previous frame. This is supported since many functions in Unreal's
318 // shaders use this convention.
319 const TCHAR* CodeHLSL{};
320
321 // This works identically to `CodeHLSL`, but is used instead of it when emitting the
322 // screen-space partial differential along the X/Y axis.
323 // Note: If this is not provided (meaning it is left null), and `Flags` does not state that
324 // this extern does not support partial differential, this instruction will automatically
325 // generate a "zero" expression of the appropriate type.
326 const TCHAR* CodeHLSLDDX{};
327 const TCHAR* CodeHLSLDDY{};
328
329 // Specifies what shader frequencies this inline extern is supported in. If this instruction
330 // happens to be used in an unsupported frequency, a semantic error will be reported (during analysis).
331 EStageMask AllowedStages = EStageMask::Any;
332
333 // Specifies what material domains this extern is supported in. If this instruction happens to be
334 // used in an unsupported domain, a semantic error will be reported (during analysis).
335 // Note: Use `MIR::MakeDomainMask()` to conveniently generate a mask from a list of domains.
336 uint32 MaterialDomainMask = 0xffffffff;
337 static_assert(MD_MAX <= 32);
338
339 // Specifies the minimum EMaterialFeatureLevel required by this inline extern instruction.
340 // An error is reported if the translating material has a lower feature level.
341 uint32 MinimumFeatureLevel{};
342
343 // Array of this instruction MIR value graph properties.
344 MIR::EGraphProperties GraphProperties = MIR::EGraphProperties::None;
345};
