97 template<typename SolverParticlesOrRange>
98 FSolverVec3 GetDelta(const SolverParticlesOrRange& Particles, const int32 ConstraintIndex, const FSolverReal ExpStiffnessValue) const
99 {
100 const auto& Constraint = Constraints[ConstraintIndex];
101 const int32 i1 = Constraint[0];
102 const int32 i2 = Constraint[1];
103
104 if (Particles.InvM(i2) == (FSolverReal)0. && Particles.InvM(i1) == (FSolverReal)0.)
105 {
106 return FSolverVec3((FSolverReal)0.);
107 }
108 const FSolverReal CombinedMass = Particles.InvM(i2) + Particles.InvM(i1);
109
110 const FSolverVec3& P1 = Particles.P(i1);
111 const FSolverVec3& P2 = Particles.P(i2);
112 FSolverVec3 Direction = P1 - P2;
113 const FSolverReal Distance = Direction.SafeNormalize();
114
115 const FSolverVec3 Delta = (Distance - Dists[ConstraintIndex]) * Direction;
116 return ExpStiffnessValue * Delta / CombinedMass;
117 }

119 void CalculateRestLengths(const TConstArrayView<FSolverVec3>& Positions, const TArrayView<FSolverReal>& DistsToCompute) const
120 {

123 {
124 DistsToCompute[Index] = (Positions[Constraints[Index][0]] - Positions[Constraints[Index][1]]).Size();
125 }

129 template<int32 Valence, typename Predicate UE_REQUIRES(Valence >= 2 && Valence <= 4)>
130 TArray<TVector<int32, 2>> TrimConstraints(const TArray<TVector<int32, Valence>>& InConstraints, Predicate TrimPredicate)
131 {
132 TSet<TVec2<int32>> TrimmedConstraints;
133 TrimmedConstraints.Reserve(Valence == 2 ? InConstraints.Num() : InConstraints.Num() * Chaos::Utilities::NChooseR(Valence, 2));
134
135 for (const TVector<int32, Valence>& ConstraintV : InConstraints)
136 {
137 for (int32 i = 0; i < Valence - 1; ++i)
138 {
139 for (int32 j = i + 1; j < Valence; ++j)
140 {
141 const int32 IndexI = ConstraintV[i];
142 const int32 IndexJ = ConstraintV[j];
143
144 if (!TrimPredicate(IndexI, IndexJ))
145 {
146 TrimmedConstraints.Add(IndexI <= IndexJ ? TVec2<int32>(IndexI, IndexJ) : TVec2<int32>(IndexJ, IndexI));
147 }
148 }
149 }
150 }
151 return TrimmedConstraints.Array();
152 }
