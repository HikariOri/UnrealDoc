310 virtual void DispatchConsumerData()
311 {
312 check(IsInGameThread());
313
314 const PayloadType* Buffer = EventBuffer.Get()->GetConsumerBuffer();
315
316 // We dispatch events even when the event buffers are empty because the absence
317 // of collisions/breaks/etc is required information for some use-cases like
318 // tracking continuous collisions.
319 const bool bDispatchEmptyEvents = Private::bEventManagerDispatchEmptyEvents;
320 if (IsEventDataEmpty(Buffer) && !bDispatchEmptyEvents)
321 {
322 return;
323 }
324
325 // Unregister any handlers that should no longer be invoked
326 UnregisterDeferedHandler();
327
328 {
329 // We're going to iterate the handlers and invoke them, read lock the list to block any changes
330 UE::TReadScopeLock ScopeLock(HandlerLock);
331
332 // Dispatch process beginning, we can no longer modify the handlers until the dispatch
333 // completes, register and unregister will defer while this guard is set
334 TGuardValue<bool> ReentrancyGuard(bDispatchInProgress, true);
335
336 const TMap<IPhysicsProxyBase*, TArray<int32>>* Map = GetProxyToIndexMap(Buffer); // Use t his map to get all proxies used in the event buffer
337 // Only take this path if we have fewer Events than Handlers
338 if(Map && Map->Num() + HandlersNotInProxyOwnerMap.Num() < HandlerMap.Num())
339 {
340 if(!ProxyOwnerToHandlerMap.IsEmpty())
341 {
342 TSet<FEventHandlerPtr> UniqueHandlers;
343 UniqueHandlers.Reserve(Map->Num());
344 for(auto& KeyValue : *Map) // Only iterating over objects that are associated with events here
345 {
346 const IPhysicsProxyBase* Proxy = KeyValue.Get<0>();
347 const UObject* Owner = Proxy->GetOwner();
348 for(TMultiMap<UObject*, FEventHandlerPtr>::TConstKeyIterator It = ProxyOwnerToHandlerMap.CreateConstKeyIterator(Owner); It; ++It)
349 {
350 UniqueHandlers.Add(It.Value());
351 }
352 }
353
354 for(const FEventHandlerPtr Handler : UniqueHandlers)
355 {
356 Handler->HandleEvent(Buffer);
357 }
358 }
359
360 for(const TPair<void*, FEventHandlerPtr>& Pair : HandlersNotInProxyOwnerMap)
361 {
362 Pair.Value->HandleEvent(Buffer);
363 }
364 }
365 else
366 // This path is taken if there are fewer Handlers than events or the handler does not support GetProxyToIndexMap
367 {
368 for(const TPair<void*, FEventHandlerPtr>& Pair : HandlerMap)
369 {
370 Pair.Value->HandleEvent(Buffer);
371 }
372 }
373 }
374
375 // Register any new handlers enqueued during the dispatch process
376 RegisterDeferedHandler();
377 }
