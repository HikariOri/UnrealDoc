256 FBoundShaderStateInput AsBoundShaderState(bool bRequired = true) const
257 {
258 bool bLocalAllShaderAreLoaded = true;
259 bool bCanSkipShader = AllowSkipUnloadedShaders();
260 const bool bShaderRequired = bRequired && !bCanSkipShader;
261
262 auto GetShaderResource = [&bLocalAllShaderAreLoaded](FRHIShader* Shader) -> FRHIShader* {
263 if (Shader == nullptr)
264 {
265 bLocalAllShaderAreLoaded &= false;
266 }
267 return Shader;
268 };
269
270 FRHIVertexShader* LocalCachedVertexShader = CachedVertexShader.load(std::memory_order_acquire);
271 if (!LocalCachedVertexShader || !bAllShaderAreLoaded)
272 {
273 CachedPixelShader = PixelShaderResource ? static_cast<FRHIPixelShader*>(GetShaderResource(PixelShaderResource->GetShader(PixelShaderIndex, bShaderRequired))) : nullptr;
274#if PLATFORM_SUPPORTS_GEOMETRY_SHADERS
275 CachedGeometryShader = GeometryShaderResource ? static_cast<FRHIGeometryShader*>(GetShaderResource(GeometryShaderResource->GetShader(GeometryShaderIndex, bShaderRequired))) : nullptr;
276#endif
277#if PLATFORM_SUPPORTS_MESH_SHADERS
278 CachedMeshShader = MeshShaderResource ? static_cast<FRHIMeshShader*>(GetShaderResource(MeshShaderResource->GetShader(MeshShaderIndex, bShaderRequired))) : nullptr;
279#endif
280 LocalCachedVertexShader = VertexShaderResource ? static_cast<FRHIVertexShader*>(GetShaderResource(VertexShaderResource->GetShader(VertexShaderIndex, bShaderRequired))) : nullptr;
281
282 bAllShaderAreLoaded = bLocalAllShaderAreLoaded;
283 CachedVertexShader.store(LocalCachedVertexShader, std::memory_order_release);
284 }
285
286#if PLATFORM_SUPPORTS_MESH_SHADERS
287 if (CachedMeshShader)
288 {
289 return FBoundShaderStateInput(CachedMeshShader, nullptr, CachedPixelShader);
290 }
291 else
292#endif
293 {
294 return FBoundShaderStateInput(VertexDeclarationRHI
295 , LocalCachedVertexShader
296 , CachedPixelShader
297#if PLATFORM_SUPPORTS_GEOMETRY_SHADERS
298 , CachedGeometryShader
299#endif
300 );
301 }
302 }

1365 static inline uint32 PointerHash(const void* Key)
1366 {
1367 // Ignoring the lower 4 bits since they are likely zero anyway.
1368 // Higher bits are more significant in 64 bit builds.
1369 return static_cast<uint32>(reinterpret_cast<UPTRINT>(Key) >> 4);
1370 };

1373 {
1374 return A ^ (B + 0x9e3779b9 + (A << 6) + (A >> 2));
1375 }
1376 } HashKey;

1385 const uint32 StreamIndex = VertexInputStream.StreamIndex;
1386 const uint32 Offset = VertexInputStream.Offset;
1387
1388 uint32 Packed = (StreamIndex << 28) | Offset;
1389 HashKey.VertexStreams = FHashKey::HashCombine(HashKey.VertexStreams, Packed);
1390 HashKey.VertexBuffers = FHashKey::HashCombine(HashKey.VertexBuffers, FHashKey::PointerHash(VertexInputStream.Pointer));
1391 }

1394 HashKey.FirstIndex = FirstIndex;

1396 HashKey.NumInstances = NumInstances;
