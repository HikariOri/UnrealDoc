64 BrushAlpha.SetNumZeroed(Bounds.Area());

66
67 FIntRect GetBounds() const
68 {
69 return Bounds;
70 }
71
72 // For compatibility with older landscape code that uses inclusive bounds in 4 int32s
73 void GetInclusiveBounds(int32& X1, int32& Y1, int32& X2, int32& Y2) const
74 {
75 X1 = Bounds.Min.X;
76 Y1 = Bounds.Min.Y;
77 X2 = Bounds.Max.X - 1;
78 Y2 = Bounds.Max.Y - 1;
79 }
80
81 float* GetDataPtr(FIntPoint Position)
82 {
83 return BrushAlpha.GetData() + (Position.Y - Bounds.Min.Y) * Bounds.Width() + (Position.X - Bounds.Min.X);
84 }
85 const float* GetDataPtr(FIntPoint Position) const
86 {
87 return BrushAlpha.GetData() + (Position.Y - Bounds.Min.Y) * Bounds.Width() + (Position.X - Bounds.Min.X);
88 }
89
90 FORCEINLINE explicit operator bool() const
91 {
92 return BrushAlpha.Num() != 0;
93 }
94
95 FORCEINLINE bool operator!() const
96 {
97 return !(bool)*this;
