143template<typename PermutationVectorType>
144class TPipelineState
145{
146public:
147 TPipelineState()
148 {
149 FPlatformMemory::Memset(&Vector, 0, sizeof(Vector));
150 }
151
153 template<typename DimensionType>
154 void Set(DimensionType PermutationVectorType::*Dimension, const DimensionType& DimensionValue)
155 {
156 SIZE_T ByteOffset = GetByteOffset(Dimension);
157
158 // Make sure not updating a value of the pipeline already initialized, to ensure there is no cycle in the dependency of the different dimensions.
159 checkf(ByteOffset >= InitializedOffset, TEXT("This member of the pipeline state has already been committed."));
160
161 Vector.*Dimension = DimensionValue;
162
163 // Update the initialised offset to make sure this is not set only once.
164 InitializedOffset = ByteOffset + sizeof(DimensionType);
165 }
166
168 void Commit()
169 {
170 // Force the pipeline state to be initialized exactly once.
171 checkf(!IsCommitted(), TEXT("Pipeline state has already been committed."));
172 InitializedOffset = ~SIZE_T(0);
173 }
174
176 bool IsCommitted() const
177 {
178 return InitializedOffset == ~SIZE_T(0);
179 }
180
182 template<typename DimensionType>
183 const DimensionType& operator [](DimensionType PermutationVectorType::*Dimension) const
184 {
185 SIZE_T ByteOffset = GetByteOffset(Dimension);
186
187 checkf(ByteOffset < InitializedOffset, TEXT("This dimension has not been initialized yet."));
188
189 return Vector.*Dimension;
190 }
191
193 const PermutationVectorType* operator->() const
194 {
195 // Make sure the pipeline state is committed to catch accesses to uninitialized settings.
196 checkf(IsCommitted(), TEXT("The pipeline state needs to be fully commited before being able to reference directly the pipeline state structure."));
197 return &Vector;
198 }
199
201 const PermutationVectorType& operator * () const
202 {
203 // Make sure the pipeline state is committed to catch accesses to uninitialized settings.
204 checkf(IsCommitted(), TEXT("The pipeline state needs to be fully commited before being able to reference directly the pipeline state structure."));
205 return Vector;
206 }
207
208private:
209
210 template<typename DimensionType>
211 static SIZE_T GetByteOffset(DimensionType PermutationVectorType::*Dimension)
212 {
213 return (SIZE_T)(&(((PermutationVectorType*) 0)->*Dimension));
214 }
215
216 PermutationVectorType Vector;
217
218 SIZE_T InitializedOffset = 0;
219};
