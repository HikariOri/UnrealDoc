72 template <typename ThisType, typename FuncType>
73 static void ForEachState(ThisType* Self, FuncType InFunc, const TArray<TSubclassOf<UEditorState>>& InEditorStateTypeFilter = {})
74 {
75 // Keep track of processed states to avoid cycles
76 TMap<TSubclassOf<UEditorState>, bool> ProcessedStates;
77
78 const auto ProcessState = [&ProcessedStates, &InFunc, Self](UEditorState* StateToProcess)
79 {
80 auto ProcessStateImpl = [&ProcessedStates, &InFunc, Self](UEditorState* StateToProcess, auto& ProcessStateRef) -> bool
81 {
82 // Avoid processing dependencies twice
83 bool* bProcessedStateResult = ProcessedStates.Find(StateToProcess->GetClass());
84 if (!bProcessedStateResult)
85 {
86 bool bProcessedDependenciesSuccessfully = true;
87
88 // Process the dependencies - Stop if any error is encountered
89 for (TSubclassOf<UEditorState> DependencyType : StateToProcess->GetDependencies())
90 {
91 UEditorState* DependencyState = Self->GetState(DependencyType);
92 bProcessedDependenciesSuccessfully = DependencyState && ProcessStateRef(DependencyState, ProcessStateRef);
93
94 if (!bProcessedDependenciesSuccessfully)
95 {
96 break;
97 }
98 }
99
100 // Process the state itself
101 bool bSuccess = InFunc(StateToProcess, bProcessedDependenciesSuccessfully);
102 ProcessedStates.Add(StateToProcess->GetClass(), bSuccess);
103 return bSuccess;
104 }
105 else
106 {
107 return *bProcessedStateResult;
108 }
109 };
110
111 return ProcessStateImpl(StateToProcess, ProcessStateImpl);
112 };
113
114 for (const TObjectPtr<UEditorState>& State : Self->States)
115 {
116 if (!InEditorStateTypeFilter.IsEmpty() && !InEditorStateTypeFilter.Contains(State->GetClass()))
117 {
118 continue;
119 }
120
121 ProcessState(State.Get());
122 }
123 }
