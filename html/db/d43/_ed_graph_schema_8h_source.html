794 // Categorizes two pins into an input pin and an output pin. Returns true if successful or false if they don't make sense as such (two inputs or two outputs)
795 template<typename PinType>
796 static bool CategorizePinsByDirection(PinType* PinA, PinType* PinB, /*out*/ PinType*& InputPin, /*out*/ PinType*& OutputPin)
797 {
798 InputPin = nullptr;
799 OutputPin = nullptr;
800
801 bool bResult = false;
802 if (ensure(PinA && PinB))
803 {
804 if ((PinA->Direction == EGPD_Input) && (PinB->Direction == EGPD_Output))
805 {
806 InputPin = PinA;
807 OutputPin = PinB;
808 bResult = true;
809 }
810 else if ((PinB->Direction == EGPD_Input) && (PinA->Direction == EGPD_Output))
811 {
812 InputPin = PinB;
813 OutputPin = PinA;
814 bResult = true;
815 }
816 }
817 return bResult;
