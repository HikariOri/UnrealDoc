154// Initialize and ConditionalInitialize must be inline so that each dll calls its own address of
155// curl_set_disallowmultithreadedlock. We link curl statically into each dll using it, so every dll has its own copy
156// of curl's "global" variables, and each dll needs to set the values of its own copy.
157inline int ConditionalInitialize(FCurlInitializeData& InitializeData)
158{
159 if (InitializeData.bInitialized)
160 {
161 return 0;
162 }
163 InitializeData.bInitialized = true;
164
165 int32 CurlInitFlags = CURL_GLOBAL_ALL;
166#if WITH_SSL
167 // Override libcrypt functions to initialize memory since OpenSSL triggers multiple valgrind warnings due to this.
168 // Do this before libcurl/libopenssl/libcrypto has been inited.
169 SetCryptoMemoryHooks(InitializeData);
170
171 // Make sure SSL is loaded so that we can use the shared cert pool, and to globally initialize OpenSSL if possible
172 FSslModule& SslModule = FModuleManager::LoadModuleChecked<FSslModule>("SSL");
173 if (SslModule.GetSslManager().InitializeSsl())
174 {
175 // Do not need Curl to initialize its own SSL
176 CurlInitFlags = CurlInitFlags & ~(CURL_GLOBAL_SSL);
177 }
178#endif // #if WITH_SSL
179
180 CURLcode InitResult = curl_global_init_mem(CurlInitFlags, CurlMalloc, CurlFree, CurlRealloc, CurlStrdup, CurlCalloc);
181#if WITH_CURL_INSTRUMENTATION
182 // these functions are only implemented in our custom version of curl
183 curl_set_disallowmultithreadedlock(CurlDisallowMultithreadedLockAllocate,
184 CurlDisallowMultithreadedLockFree, CurlDisallowMultithreadedLockLock, CurlDisallowMultithreadedLockUnlock);
185 curl_set_settimetree(CurlSetTimeTree);
186#endif
187 return (int)InitResult;
188}
189
190inline void ConditionalShutdown(FCurlInitializeData& InitializeData)
191{
192 if (!InitializeData.bInitialized)
193 {
194 return;
195 }
196 InitializeData.bInitialized = false;
197
198 curl_global_cleanup();
199
200 FPlatformHttp::ShutdownPlatformCustomSsl();
201
202#if WITH_SSL
203 // Shutdown OpenSSL
204 FSslModule& SslModule = FModuleManager::LoadModuleChecked<FSslModule>("SSL");
205 SslModule.GetSslManager().ShutdownSsl();
206
207 UnsetCryptoMemoryHooks(InitializeData);
208#endif // #if WITH_SSL
209}
