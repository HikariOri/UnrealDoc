241 void Serialize(FArchive& Ar, int32 InVersion)
242 {
243 if (Ar.IsLoading() && Ar.TotalSize() < (Ar.Tell() + GetSerializedSize(InVersion)))
244 {
245 Magic = 0;
246 return;
247 }
248
249 if (Ar.IsSaving() || InVersion >= PakFile_Version_EncryptionKeyGuid)
250 {
251 Ar << EncryptionKeyGuid;
252 }
253 Ar << bEncryptedIndex;
254 Ar << Magic;
255 if (Magic != PakFile_Magic)
256 {
257 // handle old versions by failing out now (earlier versions will be attempted)
258 Magic = 0;
259 return;
260 }
261
262 Ar << Version;
263 Ar << IndexOffset;
264 Ar << IndexSize;
265 Ar << IndexHash;
266
267 if (Ar.IsLoading())
268 {
269 if (Version < PakFile_Version_IndexEncryption)
270 {
271 bEncryptedIndex = false;
272 }
273
274 if (Version < PakFile_Version_EncryptionKeyGuid)
275 {
276 EncryptionKeyGuid.Invalidate();
277 }
278 }
279
280 if (Version >= PakFile_Version_FrozenIndex && Version < PakFile_Version_PathHashIndex)
281 {
282 bool bIndexIsFrozen = false;
283 Ar << bIndexIsFrozen;
284 if (bIndexIsFrozen)
285 {
286 UE_LOGF(LogPakFile, Fatal, "PakFile was frozen with version FPakInfo::PakFile_Version_FrozenIndex, which is no longer supported. Regenerate Paks.");
287 }
288 }
289
290 if (Version < PakFile_Version_FNameBasedCompressionMethod)
291 {
292 // for old versions, put in some known names that we may have used
293 CompressionMethods.Add(NAME_Zlib);
294 CompressionMethods.Add(NAME_Gzip);
295 CompressionMethods.Add(TEXT("Oodle"));
296 }
297 else
298 {
299 // we need to serialize a known size, so make a buffer of "strings"
300 const int32 BufferSize = CompressionMethodNameLen * MaxNumCompressionMethods;
301 ANSICHAR Methods[BufferSize];
302 if (Ar.IsLoading())
303 {
304 Ar.Serialize(Methods, BufferSize);
305 for (int32 Index = 0; Index < MaxNumCompressionMethods; Index++)
306 {
307 ANSICHAR* MethodString = &Methods[Index * CompressionMethodNameLen];
308 if (MethodString[0] != 0)
309 {
310 // Make sure if the file has garbage we don't read off in to lala land.
311 MethodString[CompressionMethodNameLen-1] = 0;
312 CompressionMethods.Add(FName(MethodString));
313 }
314 }
315 }
316 else
317 {
318 // we always zero out fully what we write out so that reading in is simple
319 FMemory::Memzero(Methods, BufferSize);
320
321 for (int32 Index = 1; Index < CompressionMethods.Num(); Index++)
322 {
323 ANSICHAR* MethodString = &Methods[(Index - 1) * CompressionMethodNameLen];
324 FCStringAnsi::Strncpy(MethodString, TCHAR_TO_ANSI(*CompressionMethods[Index].ToString()), CompressionMethodNameLen);
325 }
326 Ar.Serialize(Methods, BufferSize);
327 }
328 }
329 }
330
331 int32 GetCompressionMethodIndex(FName CompressionMethod)
332 {
333 // look for existing method
334 for (uint8 Index = 0; Index < CompressionMethods.Num(); Index++)
335 {
336 if (CompressionMethods[Index] == CompressionMethod)
337 {
338 return Index;
339 }
340 }
341
342 checkf(CompressionMethod.ToString().Len() < CompressionMethodNameLen, TEXT("Compression method name, %s, is too long for pak file serialization. You can increase CompressionMethodNameLen, but then will have to handle version management."), *CompressionMethod.ToString());

344 checkf(CompressionMethods.Num() <= MaxNumCompressionMethods, TEXT("Too many unique compression methods in one pak file. You can increase MaxNumCompressionMethods, but then will have to handle version management."));
345
346 // add it if it didn't exist
347 return CompressionMethods.Add(CompressionMethod);
348 }

405struct FPakEntry
406{
407 static const uint8 Flag_None = 0x00;
408 static const uint8 Flag_Encrypted = 0x01;
409 static const uint8 Flag_Deleted = 0x02;
410
412 int64 Offset;
414 int64 Size;
416 int64 UncompressedSize;
418 uint8 Hash[20];
420 TArray<FPakCompressedBlock> CompressionBlocks;
422 uint32 CompressionBlockSize;
424 uint32 CompressionMethodIndex;
426 uint8 Flags;
428 mutable bool Verified;
429
433 FPakEntry()
434 {
435 Reset();
436 }
437
443 int64 GetSerializedSize(int32 Version) const
444 {
445 int64 SerializedSize = sizeof(Offset) + sizeof(Size) + sizeof(UncompressedSize) + sizeof(Hash);
446
447 if (Version >= FPakInfo::PakFile_Version_FNameBasedCompressionMethod)
448 {
449 SerializedSize += sizeof(CompressionMethodIndex);
450 }
451 else
452 {
453 SerializedSize += sizeof(int32); // Old CompressedMethod var from pre-fname based compression methods
454 }
455
456 if (Version >= FPakInfo::PakFile_Version_CompressionEncryption)
457 {
458 SerializedSize += sizeof(Flags) + sizeof(CompressionBlockSize);
459 if(CompressionMethodIndex != 0)
460 {
461 SerializedSize += sizeof(FPakCompressedBlock) * CompressionBlocks.Num() + sizeof(int32);
462 }
463 }
464 if (Version < FPakInfo::PakFile_Version_NoTimestamps)
465 {
466 // Timestamp
467 SerializedSize += sizeof(int64);
468 }
469 return SerializedSize;
470 }
471
475 bool operator == (const FPakEntry& B) const
476 {
477 return IndexDataEquals(B) &&
478 FMemory::Memcmp(Hash, B.Hash, sizeof(Hash)) == 0;
479 }
480
484 bool operator != (const FPakEntry& B) const
485 {
486 return !(*this == B);
487 }
488
489 bool IndexDataEquals(const FPakEntry& B) const
490 {
491 // Offset are only in the Index and so are not compared
492 // Hash is only in the payload and so are not compared
493 // Verified is only in the payload and is mutable and so is not compared
494 return Size == B.Size &&
495 UncompressedSize == B.UncompressedSize &&
496 CompressionMethodIndex == B.CompressionMethodIndex &&
497 Flags == B.Flags &&
498 CompressionBlockSize == B.CompressionBlockSize &&
499 CompressionBlocks == B.CompressionBlocks;
500 }
501
502 void Reset()
503 {
504 Offset = -1;
505 Size = 0;
506 UncompressedSize = 0;
507 FMemory::Memset(Hash, 0, sizeof(Hash));
508 CompressionBlocks.Reset();
509 CompressionBlockSize = 0;
510 CompressionMethodIndex = 0;
511 Flags = Flag_None;
512 Verified = false;
513 }
514
521 void Serialize(FArchive& Ar, int32 Version)
522 {
523 Ar << Offset;
524 Ar << Size;
525 Ar << UncompressedSize;
526 if (Version < FPakInfo::PakFile_Version_FNameBasedCompressionMethod)
527 {
528 int32 LegacyCompressionMethod;
529 Ar << LegacyCompressionMethod;
530 if (LegacyCompressionMethod == COMPRESS_None)
531 {
532 CompressionMethodIndex = 0;
533 }
534 else if (LegacyCompressionMethod & COMPRESS_ZLIB_DEPRECATED)
535 {
536 CompressionMethodIndex = 1;
537 }
538 else if (LegacyCompressionMethod & COMPRESS_GZIP_DEPRECATED)
539 {
540 CompressionMethodIndex = 2;
541 }
542 else if (LegacyCompressionMethod & COMPRESS_Custom_DEPRECATED)
543 {
544 CompressionMethodIndex = 3;
545 }
546 else
547 {
548 UE_LOGF(LogPakFile, Fatal, "Found an unknown compression type in pak file, will need to be supported for legacy files");
549 }
550 }
551 else
552 {
553 Ar << CompressionMethodIndex;
554 }
555 if (Version <= FPakInfo::PakFile_Version_Initial)
556 {
557 FDateTime Timestamp;
558 Ar << Timestamp;
559 }
560 Ar.Serialize(Hash, sizeof(Hash));
561 if (Version >= FPakInfo::PakFile_Version_CompressionEncryption)
562 {
563 if(CompressionMethodIndex != 0)
564 {
565 Ar << CompressionBlocks;
566 }
567 Ar << Flags;
568 Ar << CompressionBlockSize;
569 }
570 }

1706
1707 FPakReaderPolicy(const FPakFile& InPakFile,const FPakEntry& InPakEntry, TAcquirePakReaderFunction& InAcquirePakReader)
1708 : PakFile(InPakFile)
1709 , PakEntry(InPakEntry)
1710 , AcquirePakReader(InAcquirePakReader)
1711 {
1712 OffsetToFile = PakEntry.Offset + PakEntry.GetSerializedSize(PakFile.GetInfo().Version);
1713 }

1715 inline int64 FileSize() const
1716 {
1717 return PakEntry.Size;
1718 }
1719
1720 void Serialize(int64 DesiredPosition, void* V, int64 Length) const
1721 {
1722 FGuid EncryptionKeyGuid = PakFile.GetInfo().EncryptionKeyGuid;
1723 const constexpr int64 Alignment = (int64)EncryptionPolicy::Alignment;
1724 const constexpr int64 AlignmentMask = ~(Alignment - 1);
1725 uint8 TempBuffer[Alignment];
1726 FSharedPakReader PakReader = AcquirePakReader();
1727 if (EncryptionPolicy::AlignReadRequest(DesiredPosition) != DesiredPosition)
1728 {
1729 int64 Start = DesiredPosition & AlignmentMask;
1730 int64 Offset = DesiredPosition - Start;
1731 int64 CopySize = FMath::Min(Alignment - Offset, Length);
1732 PakReader->Seek(OffsetToFile + Start);
1733 PakReader->Serialize(TempBuffer, Alignment);
1734 EncryptionPolicy::DecryptBlock(TempBuffer, Alignment, EncryptionKeyGuid);
1735 FMemory::Memcpy(V, TempBuffer + Offset, CopySize);
1736 V = (void*)((uint8*)V + CopySize);
1737 DesiredPosition += CopySize;
1738 Length -= CopySize;
1739 check(Length == 0 || DesiredPosition % Alignment == 0);
1740 }
1741 else
1742 {
1743 PakReader->Seek(OffsetToFile + DesiredPosition);
1744 }
1745
1746 int64 CopySize = Length & AlignmentMask;
1747 PakReader->Serialize(V, CopySize);
1748 EncryptionPolicy::DecryptBlock(V, CopySize, EncryptionKeyGuid);
1749 Length -= CopySize;
1750 V = (void*)((uint8*)V + CopySize);
1751
1752 if (Length > 0)
1753 {
1754 PakReader->Serialize(TempBuffer, Alignment);
1755 EncryptionPolicy::DecryptBlock(TempBuffer, Alignment, EncryptionKeyGuid);
1756 FMemory::Memcpy(V, TempBuffer, Length);
1757 }
1758 }
