89inline FFrameTime operator*(FFrameTime InTime, const FMovieSceneTimeTransform& RHS)
90{
91 // Avoid floating point conversion when in the same time-space
92 if (RHS.TimeScale == 1.f)
93 {
94 return InTime + RHS.Offset;
95 }

98 // Special case infinite timescale (inverse of a zero-timescale) and just return the RHS.Offset

173inline FMovieSceneTimeTransform operator*(const FMovieSceneTimeTransform& LHS, const FMovieSceneTimeTransform& RHS)
174{

176 //

178 // | 0 , 1 | | 0 , 1 |
