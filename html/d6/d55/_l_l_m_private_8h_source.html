409 FInPtr(const void* InPtr)
410 : Ptr(InPtr) {
411#if PLATFORM_STORES_DATA_IN_POINTER_HIGH_BITS
412 // The values in the high 16 bits of Ptr are used to label the allocation, and are not needed to identify the pointer.
413 // No two pointers will have the same values for the lower 48 bits but different values for the upper 16. We want to use
414 // the upper 16 bits to store extra data of our own in our table that holds the pointer, so zero those bits out before using
415 // them in our AllocationMap.
416 static_assert(sizeof(void*) == 8);
417 Ptr = reinterpret_cast<const void*>(reinterpret_cast<uintptr_t>(Ptr) & 0x0000'ffff'ffff'ffff);
418#endif
419 }
420 operator const void* ()
421 {
422 return Ptr;
423 }
