622 const TVectorRegisterType<T> Norm = VectorNormalizeRotator(RegA);

624 return !VectorAnyGreaterThan(AbsNorm, VectorLoadFloat1(&Tolerance));

690{

692 Angle = FMath::Fmod(Angle, (T)360.0);

944template<typename T>
945inline T FMath::ClampAngle(T AngleDegrees, T MinAngleDegrees, T MaxAngleDegrees)
946{
947 const T MaxDelta = UE::Math::TRotator<T>::ClampAxis(MaxAngleDegrees - MinAngleDegrees) * 0.5f; // 0..180
948 const T RangeCenter = UE::Math::TRotator<T>::ClampAxis(MinAngleDegrees + MaxDelta); // 0..360
949 const T DeltaFromCenter = UE::Math::TRotator<T>::NormalizeAxis(AngleDegrees - RangeCenter); // -180..180
950
951 // maybe clamp to nearest edge
952 if (DeltaFromCenter > MaxDelta)
953 {
954 return UE::Math::TRotator<T>::NormalizeAxis(RangeCenter + MaxDelta);
955 }
956 else if (DeltaFromCenter < -MaxDelta)
957 {
958 return UE::Math::TRotator<T>::NormalizeAxis(RangeCenter - MaxDelta);
959 }
960
961 // already in range, just return it
962 return UE::Math::TRotator<T>::NormalizeAxis(AngleDegrees);
