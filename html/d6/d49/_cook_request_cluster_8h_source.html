679inline bool FRequestCluster::FQueryPlatformData::CompareExchangeAsyncQueryStatus(EAsyncQueryStatus& Expected,
680 EAsyncQueryStatus Desired)
681{
682 return AsyncQueryStatus.compare_exchange_strong(Expected, Desired,
683 // For the read operation to see whether we should set it, we need only relaxed memory order;
684 // we don't care about the values of other related variables that depend on it when deciding whether
685 // it is our turn to set it.
686 // For the write operation if we decide to set it, we need release memory order to guard reads of
687 // the variables that depend on it (e.g. CookAttachments).
688 std::memory_order_release /* success memory order */,
689 std::memory_order_relaxed /* failure memory order */
690 );
