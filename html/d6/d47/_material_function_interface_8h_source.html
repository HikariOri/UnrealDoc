203 bool GetNamedParameterOfType(const FHashedMaterialParameterInfo& ParameterInfo, ExpressionType*& Parameter, UMaterialFunctionInterface** OwningFunction = nullptr)
204 {
205 Parameter = nullptr;
206
207 if (UMaterialFunctionInterface* ParameterFunction = GetBaseFunctionInterface())
208 {
209 const UClass* TargetClass = ExpressionType::StaticClass();
210
211 auto GetExpressionParameterByNamePredicate =
212 [&ParameterInfo, &Parameter, &OwningFunction, TargetClass](UMaterialFunctionInterface* Function) -> bool
213 {
214 for (UMaterialExpression* FunctionExpression : Function->GetExpressions())
215 {
216 if (ExpressionType* ExpressionParameter = (FunctionExpression && FunctionExpression->IsA(TargetClass)) ? (ExpressionType *)FunctionExpression : nullptr)
217 {
218 if (ExpressionParameter->ParameterName == ParameterInfo.Name)
219 {
220 Parameter = ExpressionParameter;
221
222 if (OwningFunction)
223 {
224 (*OwningFunction) = Function;
225 }
226
227 return false; // found, stop iterating
228 }
229 }
230 }
231

233 };
