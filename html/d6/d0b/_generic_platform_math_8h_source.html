35 [[nodiscard]] static constexpr inline float LoadHalf(const uint16* Ptr)
36 {
37 uint16 FP16 = *Ptr;
38 constexpr uint32 shifted_exp = 0x7c00 << 13; // exponent mask after shift
39 union FP32T
40 {
41 uint32 u;
42 float f;
43 } FP32 = {}, magic = { 113 << 23 };
44
45 FP32.u = (FP16 & 0x7fff) << 13; // exponent/mantissa bits
46 uint32 exp = shifted_exp & FP32.u; // just the exponent
47 FP32.u += uint32(127 - 15) << 23; // exponent adjust
48
49 // handle exponent special cases
50 if (exp == shifted_exp) // Inf/NaN?
51 {
52 FP32.u += uint32(128 - 16) << 23; // extra exp adjust
53 }
54 else if (exp == 0) // Zero/Denormal?
55 {
56 FP32.u += 1 << 23; // extra exp adjust
57 FP32.f -= magic.f; // renormalize
58 }
59
60 FP32.u |= (FP16 & 0x8000) << 16; // sign bit
61 return FP32.f;
62 }

64 // store float to half (F16)
65 // converts with RTNE = round to nearest even
66 // values too large for F16 are stored as +-Inf
67 // https://gist.github.com/rygorous/2156668
68 // float_to_half_fast3_rtne
69 static constexpr inline void StoreHalf(uint16* Ptr, float Value)
70 {
71 union FP32T
72 {
73 uint32 u;
74 float f;
75 } FP32 = {};
76 uint16 FP16 = {};
77
78 FP32.f = Value;
79
80 constexpr FP32T f32infty = { uint32(255 << 23) };
81 constexpr FP32T f16max = { uint32(127 + 16) << 23 };
82 constexpr FP32T denorm_magic = { (uint32(127 - 15) + uint32(23 - 10) + 1) << 23 };
83 constexpr uint32 sign_mask = 0x80000000u;
84
85 uint32 sign = FP32.u & sign_mask;
86 FP32.u ^= sign;
87
88 // NOTE all the integer compares in this function can be safely
89 // compiled into signed compares since all operands are below
90 // 0x80000000. Important if you want fast straight SSE2 code
91 // (since there's no unsigned PCMPGTD).
92
93 if (FP32.u >= f16max.u) // result is Inf or NaN (all exponent bits set)
94 {
95 FP16 = (FP32.u > f32infty.u) ? 0x7e00 : 0x7c00; // NaN->qNaN and Inf->Inf
96 }
97 else // (De)normalized number or zero
98 {
99 if (FP32.u < uint32(113 << 23)) // resulting FP16 is subnormal or zero
100 {
101 // use a magic value to align our 10 mantissa bits at the bottom of
102 // the float. as long as FP addition is round-to-nearest-even this
103 // just works.
104 FP32.f += denorm_magic.f;
105
106 // and one integer subtract of the bias later, we have our final float!
107 FP16 = uint16(FP32.u - denorm_magic.u);
108 }
109 else
110 {
111 uint32 mant_odd = (FP32.u >> 13) & 1; // resulting mantissa is odd
112
113 // update exponent, rounding bias part 1
114 FP32.u += (uint32(15 - 127) << 23) + 0xfff;
115 // rounding bias part 2
116 FP32.u += mant_odd;
117 // take the bits!
118 FP16 = uint16(FP32.u >> 13);
119 }
120 }
121
122 FP16 |= sign >> 16;
123 *Ptr = FP16;
124 }

1185 template <UE::CIntegral IntType>
1186 [[nodiscard]] static inline bool AddAndCheckForOverflow(IntType A, IntType B, IntType& OutResult)
1187 {
1188 // This follows Hacker's Delight, Chapter 2-12
1189 // Signed->unsigned conversion and unsigned addition have defined behavior always
1190 typedef typename std::make_unsigned_t<IntType> UnsignedType;
1191 const UnsignedType UnsignedA = static_cast<UnsignedType>(A);
1192 const UnsignedType UnsignedB = static_cast<UnsignedType>(B);
1193 const UnsignedType UnsignedSum = UnsignedA + UnsignedB;
1194
1195 if constexpr (std::is_signed_v<IntType>)
1196 {
1197 // Check for signed overflow.
1198 // The underlying logic here is pretty simple: if A and B had opposite signs, their sum can't
1199 // overflow. If they had the same sign and the sum has the opposite value in the sign bit, we
1200 // had an overflow. (See Hacker's Delight Chapter 2-12 for more details.)
1201 constexpr UnsignedType UnsignedMSB = static_cast<UnsignedType>(std::numeric_limits<IntType>::min());
1202 if((UnsignedSum ^ UnsignedA) & (UnsignedSum ^ UnsignedB) & UnsignedMSB)
1203 {
1204 return false;
1205 }
1206 else
1207 {
1208 OutResult = A + B;
1209 return true;
1210 }
1211 }
1212 else
1213 {
1214 // Iff there was overflow, the modular sum must be less than both the operands.
1215 if (UnsignedSum >= UnsignedA)
1216 {
1217 OutResult = UnsignedSum;
1218 return true;
1219 }
1220 else
1221 {
1222 return false;
1223 }
1224 }

1234 {

1273 template <UE::CIntegral IntType>
1274 [[nodiscard]] static inline bool MultiplyAndCheckForOverflow(IntType A, IntType B, IntType& OutResult)
1275 {
1276 // Handle the case where the second factor is 0 specially (why will become clear in a minute).
1277 if (B == 0)
1278 {
1279 // Anything times 0 is 0.
1280 OutResult = 0;
1281 return true;
1282 }
1283
1284 if constexpr (std::is_signed_v<IntType>)
1285 {
1286 // The overflow check is annoying and expensive, but the basic idea is fairly simple:
1287 // reduce to an unsigned check of the absolute values. (Again the basic algorithm is
1288 // in Hacker's Delight, Chapter 2-12).
1289 //
1290 // We need the absolute value of the product to be <=MaxValue when the result is positive
1291 // (signs of factors same) and <= -MinValue = MaxValue + 1 if the result is negative
1292 // (signs of factors opposite).
1293 typedef typename std::make_unsigned_t<IntType> UnsignedType;
1294 UnsignedType UnsignedA = static_cast<UnsignedType>(A);
1295 UnsignedType UnsignedB = static_cast<UnsignedType>(B);
1296 bool bSignsDiffer = false;
1297
1298 // Determine the unsigned absolute values of A and B carefully (note we can't negate signed
1299 // A or B, because negating MinValue is UB). We can however subtract their
1300 // unsigned values from 0 if the original value was less than zero. While doing this, also
1301 // keep track of the sign parity.
1302 if (A < 0)
1303 {
1304 UnsignedA = UnsignedType(0) - UnsignedA;
1305 bSignsDiffer = !bSignsDiffer;
1306 }
1307
1308 if (B < 0)
1309 {
1310 UnsignedB = UnsignedType(0) - UnsignedB;
1311 bSignsDiffer = !bSignsDiffer;
1312 }
1313
1314 // Determine the unsigned product bound we need based on whether the signs were same or different.
1315 const UnsignedType ProductBound = UnsignedType(std::numeric_limits<IntType>::max()) + (bSignsDiffer ? 1 : 0);
1316
1317 // We're now in the unsigned case, 0 <= UnsignedA, 0 < UnsignedB (we established b != 0), and for
1318 // there not to be overflows we need
1319 // a * b <= ProductBound
1320 // <=> a <= ProductBound/b
1321 // <=> a <= floor(ProductBound/b) since a is integer
1322 if (UnsignedA <= ProductBound / UnsignedB)
1323 {
1324 OutResult = A * B;
1325 return true;
1326 }
1327 else
1328 {
1329 return false;
1330 }
1331 }
1332 else
1333 {
1334 OutResult = A * B;
1335 return OutResult / B == A;
1336 }
1337 }
