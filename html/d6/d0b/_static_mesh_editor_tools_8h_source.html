435struct FSectionLocalizer
436{
437 FSectionLocalizer(int32 InLODIndex, int32 InSectionIndex)
438 : LODIndex(InLODIndex)
439 , SectionIndex(InSectionIndex)
440 {}
441
442 bool operator==(const FSectionLocalizer& Other) const
443 {
444 return (LODIndex == Other.LODIndex && SectionIndex == Other.SectionIndex);
445 }
446
447 bool operator!=(const FSectionLocalizer& Other) const
448 {
449 return !((*this) == Other);
450 }

454};
