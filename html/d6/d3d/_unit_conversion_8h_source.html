239 template<typename OtherType>
240 [[nodiscard]] friend inline bool operator==(const FNumericUnit<NumericType>& LHS, const FNumericUnit<OtherType>& RHS)
241 {
242 if (LHS.Units != EUnit::Unspecified && RHS.Units != EUnit::Unspecified)
243 {
244 if (LHS.Units == RHS.Units)
245 {
246 return LHS.Value == RHS.Value;
247 }
248 else if (FUnitConversion::AreUnitsCompatible(LHS.Units, RHS.Units))
249 {
250 return LHS.Value == FUnitConversion::Convert(RHS.Value, RHS.Units, LHS.Units);
251 }
252 else
253 {
254 // Invalid conversion
255 return false;
256 }
257 }
258 else
259 {
260 return LHS.Value == RHS.Value;
261 }
262 }
263
264 template<typename OtherType>

266 {
267 return !(LHS == RHS);
268 }
