62inline FGuid FBuildVersionBuilder::Build() const
63{
64 FXxHash128 Hash = HashBuilder.Finalize();
65 uint32 A = uint32(Hash.HashHigh >> 32);
66 uint32 B = uint32(Hash.HashHigh);
67 uint32 C = uint32(Hash.HashLow >> 32);
68 uint32 D = uint32(Hash.HashLow);
69 // Convert to a Variant 1 Version 3 UUID, which is meant to be created
70 // by hashing the namespace and name with MD5, but this is using XXH3.
71 B = (B & ~0x0000f000) | 0x00003000; // Version 3 (MD5)
72 C = (C & ~0xc0000000) | 0x80000000; // Variant 1 (RFC 4122 UUID)
73 return FGuid(A, B, C, D);
74}
