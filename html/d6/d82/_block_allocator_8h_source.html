15// A single-threaded, bump memory allocator that sub-allocates out of blocks of
16// inline memory and once that's exhausted, from heap allocated memory.
17// All allocated memory is freed when the TBlockAllocator is destructed.
18// There is no way to free individual allocations.
19//
20// Template parameters:
21// InlineBlockDataSize - The size of the data section of the inline block.
22// DataAlignment - The maximum supported alignment of data from this allocator.
23// GrowthPercentage - The percentage to grow the block size each reallocation.
24template <size_t InlineBlockDataSize = 256, size_t DataAlignment = 16, size_t GrowthPercentage = 200>
25class AUTORTFM_INTERNAL TBlockAllocator
26{
27public:
28 // Constructor
29 TBlockAllocator() = default;
30
31 // Destructor - frees all the memory allocated by the allocator.
32 ~TBlockAllocator()
33 {
34 FreeAll();
35 }
36
37 // Frees all allocations made by the allocator.
38 void FreeAll()
39 {
40 while (Tail != &InlineBlock)
41 {
42 FBlockHeader* Prev = Tail->Prev;
43 FAllocator::Free(Tail, Tail->TotalSize());
44 Tail = Prev;
45 }
46 InlineBlock.NextAllocation = reinterpret_cast<std::byte*>(&InlineBlock) + sizeof(FBlockHeader);
47 NextBlockSize = InlineBlockDataSize * GrowthPercentage / 100;
48 }
49
50 // Allocates memory from the block allocator.
51 // Alignment must be a power of two and no larger than DataAlignment.
52 inline void* Allocate(size_t Size, size_t Alignment)
53 {
54 AUTORTFM_ASSERT(Alignment <= DataAlignment);
55
56 if (void* Allocation = Tail->TryAllocate(Size, Alignment))
57 {
58 return Allocation;
59 }
60
61 size_t NewBlockSize = std::max(NextBlockSize, Size);
62 NextBlockSize = NextBlockSize * GrowthPercentage / 100;
63 FBlockHeader* NewBlock = FBlockHeader::New(Tail, NewBlockSize);
64 Tail = NewBlock;
65
66 void* Allocation = NewBlock->TryAllocate(Size, Alignment);
67 AUTORTFM_ASSERT(Allocation);
68 return Allocation;
69 }
70
71private:
72 TBlockAllocator(TBlockAllocator&&) = delete;
73 TBlockAllocator(const TBlockAllocator&) = delete;
74 TBlockAllocator& operator=(const TBlockAllocator&) = delete;
75 TBlockAllocator& operator=(TBlockAllocator&&) = delete;
76
77 static constexpr size_t BlockAlignment = std::max<size_t>(DataAlignment, 8);
78
79 struct alignas(BlockAlignment) FBlockHeader
80 {
81 // The previous FBlock in the singly linked-list
82 FBlockHeader* const Prev = nullptr;
83 // The size of the block's data
84 size_t const BlockDataSize;
85 // Pointer to the next sub-allocation.
86 std::byte* NextAllocation = reinterpret_cast<std::byte*>(this) + sizeof(FBlockHeader);
87 // <data>
88
89 // Allocates, constructs and returns a new block from the heap-allocated memory.
90 static FBlockHeader* New(FBlockHeader* Prev, size_t BlockDataSize)
91 {
92 void* Memory = FAllocator::Allocate(sizeof(FBlockHeader) + BlockDataSize, BlockAlignment);
93 return new (Memory) FBlockHeader{Prev, BlockDataSize};
94 }
95
96 // Returns the total size in bytes of this block.
97 inline size_t TotalSize() const
98 {
99 return sizeof(FBlockHeader) + BlockDataSize;
100 }
101
102 // Attempts to sub-allocate with out of this block. Returns a pointer to
103 // the sub-allocated memory on success, or nullptr on failure.
104 inline void* TryAllocate(size_t Size, size_t Alignment)
105 {
106 std::byte* const BlockDataStart = reinterpret_cast<std::byte*>(this) + sizeof(FBlockHeader);
107 std::byte* const BlockDataEnd = BlockDataStart + BlockDataSize;
108 std::byte* const Ptr = reinterpret_cast<std::byte*>(AlignUp(reinterpret_cast<uintptr_t>(NextAllocation), Alignment));
109 std::byte* const End = Ptr + Size;
110 if (End > BlockDataEnd || /* overflow */ AUTORTFM_UNLIKELY(End < BlockDataStart))
111 {
112 return nullptr;
113 }
114 NextAllocation = End;
115 return Ptr;
116 }
117 };
118
119 static_assert(sizeof(FBlockHeader) % DataAlignment == 0);
120
121 size_t NextBlockSize = InlineBlockDataSize * GrowthPercentage / 100;
122 FBlockHeader* Tail = &InlineBlock;
123 FBlockHeader InlineBlock{/* Prev */ nullptr, InlineBlockDataSize};
124 alignas(BlockAlignment) std::byte Data[InlineBlockDataSize];
125};
