212 template<typename TObjectID>
213 class FCachedObjectTraceIDByPtr
214 {
215 public:
216
217 template<typename ObjectType, typename HashProducer>
218 TObjectID FindOrProduceTraceObjectID(const ObjectType* ObjectPtr, const HashProducer& InHashProducer, bool& OutWasAlreadyCached)
219 {
220 if (ObjectPtr == nullptr)
221 {
222 return 0;
223 }
224
225 bool bHasCachedID = true;
226 TObjectID FoundID = CachedIDsByPtr.FindOrProduce(reinterpret_cast<const void*>(ObjectPtr), [&InHashProducer, &bHasCachedID]()
227 {
228 bHasCachedID = false;
229 return InHashProducer();
230 });
231
232 if (bHasCachedID)
233 {
234 OutWasAlreadyCached = true;
235 return FoundID;
236 }
237
238 return FoundID;
239 }
240
241 template<typename ObjectType>
242 void RemoveCachedTraceObjectID(const ObjectType* Implicit)
243 {
244 if (Implicit == nullptr)
245 {
246 return;
247 }
248
249 CachedIDsByPtr.Remove(reinterpret_cast<const void*>(Implicit));
250 }
251
252 void Reset()
253 {
254 CachedIDsByPtr.Reset();
255 }
256
257 private:
258
259 static constexpr int32 StripCount = 32;
260 TStripedMap<StripCount, const void*, TObjectID> CachedIDsByPtr;
261 };

710template <typename ParticleType>
711void FChaosVisualDebuggerTrace::TraceParticlesView(const Chaos::TParticleView<ParticleType>& ParticlesView)
712{
713 if (!IsTracing())
714 {
715 return;
716 }
717
718 const FChaosVDContext* CVDContextData = FChaosVDThreadContext::Get().GetCurrentContext(EChaosVDContextType::Solver);
719 if (!ensure(CVDContextData))
720 {
721 return;
722 }
723
724 FChaosVDContext CopyContext = *CVDContextData;
725
726 ParticlesView.ParallelFor([CopyContext](auto& Particle, int32 Index)
727 {
728 CVD_SCOPE_CONTEXT(CopyContext);
729 TraceParticle(Particle.Handle());
730 });
731}
