413 // Returns the min of two pass handles. Returns null if both are null; returns the valid handle if one is null.
414 inline static TRDGHandle Min(TRDGHandle A, TRDGHandle B)

421 inline static TRDGHandle Max(TRDGHandle A, TRDGHandle B)
