95 AddResult AddRegistration(TType* (*InRegister)(ETypeConstructPhase), const TCHAR* InPackageName, const TCHAR* InName, TInfo& InInfo, const TVersion& InVersion)
96 {
97#if WITH_RELOAD
98 const FPackageAndNameKey Key = FPackageAndNameKey{ InPackageName, InName };
99 TInfo** ExistingInfo = InfoMap.Find(Key);
100
101 bool bHasChanged = !ExistingInfo || (*ExistingInfo)->ReloadVersionInfo != InVersion;
102 InInfo.ReloadVersionInfo = InVersion;
103 TType* OldSingleton = ExistingInfo ? ((*ExistingInfo)->InnerSingleton ? (*ExistingInfo)->InnerSingleton : (*ExistingInfo)->OuterSingleton) : nullptr;
104
105 bool bAdd = true;
106 if (ExistingInfo)
107 {
108 if (IReload* Reload = GetActiveReloadInterface())
109 {
110 bAdd = Reload->GetEnableReinstancing(bHasChanged);
111 }
112 if (bAdd)
113 {
114 if (!bHasChanged)
115 {
116 // With live coding, the existing might be the same as the new info.
117 // We still invoke the copy method to allow UClasses to clear the singletons.
118 UpdateSingletons(InInfo, **ExistingInfo);
119 }
120 *ExistingInfo = &InInfo;
121 }
122 }
123 else
124 {
125 InfoMap.Add(Key, &InInfo);
126 }
127 if (bAdd)
128 {
129 Registrations.Add(FRegistrant{ InRegister, InPackageName, InName, &InInfo, OldSingleton, bHasChanged });
130 }
131 return ExistingInfo == nullptr ? AddResult::New : (bHasChanged ? AddResult::ExistingChanged : AddResult::ExistingNoChange);
132#else
133 Registrations.Add(FRegistrant{ InRegister, InPackageName, InName, &InInfo });
134 return AddResult::New;
135#endif
136 }

222 template <typename FuncType>
223 void DoPendingOuterRegistrations(bool UpdateCounter, FuncType&& InOnRegistration)
224 {
225 int32 Num = Registrations.Num();
226 for (int32 Index = ProcessedRegistrations; Index < Num; ++Index)
227 {
228 TType* Object = OuterRegister(Registrations[Index]);
229 InOnRegistration(Registrations[Index].PackageName, *Object);
230 }
231 if (UpdateCounter)
232 {
233 ProcessedRegistrations = Num;

256 void ProcessChangedObjects(bool InvokeOuterRegisterFunction = false)
257 {
258#if WITH_RELOAD
259 IReload* Reload = GetActiveReloadInterface();
260 if (Reload != nullptr)
261 {
262 const TCHAR* Prefix = Reload->GetPrefix();
263 for (const FRegistrant& Registrant : Registrations)
264 {
265 if (Registrant.bHasChanged && Registrant.OldSingleton != nullptr)
266 {
267 UE_LOGF(LogClass, Log, "%ls %ls Reload.", GetRegistryName(), *Registrant.OldSingleton->GetName());
268
269 // Reset the cached class construct info.
270 Registrant.Info->InnerSingleton = nullptr;
271 Registrant.Info->OuterSingleton = nullptr;
272
273 // Perform any other processing needed for existing objects
274 ReloadProcessObject(Registrant.OldSingleton, Prefix);
275
276 // If requested, invoke the registration function to make sure it has been recreated
277 if (InvokeOuterRegisterFunction)
278 {
279 Registrant.RegisterFn(ETypeConstructPhase::Outer);
280 }
281 }
282 }
283 }
284#endif
285 }
