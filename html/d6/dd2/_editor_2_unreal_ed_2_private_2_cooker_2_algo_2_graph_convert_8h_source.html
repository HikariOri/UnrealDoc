56template <typename RangeType, typename GetKeyEdgesType>
57inline void ConvertToGraph(const RangeType& UniqueKeys, GetKeyEdgesType GetKeyEdges,
58 TArray64<FVertex>& OutGraphBuffer, TArray<TConstArrayView<FVertex>>& OutGraph,
59 EConvertToGraphOptions Options = EConvertToGraphOptions::None)
60{
61 using KeyType = TElementType_T<RangeType>;
62
63 // Map Elements to vertices in VertexOfKey
64 int32 NumVertices = GetNum(UniqueKeys);
65 TMap<KeyType, FVertex> VertexOfKey;
66 VertexOfKey.Reserve(NumVertices);
67 FVertex Vertex = 0;
68 for (const KeyType& Key : UniqueKeys)
69 {
70 FVertex& ExistingHandle = VertexOfKey.FindOrAdd(Key, InvalidVertex);
71 check(ExistingHandle == InvalidVertex);
72 if (ExistingHandle == InvalidVertex)
73 {
74 ExistingHandle = Vertex++;
75 }
76 }
77
78 TArray<TPair<int64, int32>> EdgeOffsets;
79 EdgeOffsets.Reserve(NumVertices);
80 OutGraphBuffer.Reset();
81
82 // Call GetKeyEdges on each Element, map its returnvalue to Vertices, normalize them, and store them in EdgeOffsets
83 Vertex = 0;
84 for (const KeyType& Element : UniqueKeys)
85 {
86 int64 InitialOffset = OutGraphBuffer.Num();
87 for (const KeyType& Dependency : Invoke(GetKeyEdges, Element))
88 {
89 FVertex* TargetVertex = VertexOfKey.Find(Dependency);
90 if (!TargetVertex)
91 {
92 continue;
93 }
94
95 // Normalize Step 1: remove edges to self
96 if (*TargetVertex == Vertex)
97 {
98 continue;
99 }
100 OutGraphBuffer.Add(*TargetVertex);
101 }
102
103 // Normalize Step 2: Sort
104 TArrayView<FVertex> VertexEdges = TArrayView<FVertex>(OutGraphBuffer).Mid(InitialOffset);
105 Algo::Sort(VertexEdges);
106
107 // Normalize Step 3: Remove duplicates
108 VertexEdges = VertexEdges.Left(Algo::Unique(VertexEdges));
109 OutGraphBuffer.SetNum(InitialOffset + VertexEdges.Num(), EAllowShrinking::No);
110
111 // Store the vertex's offset into GraphBuffer, for later conversion to TArrayView
112 EdgeOffsets.Emplace(InitialOffset, VertexEdges.Num());
113 ++Vertex;
114 }
115 if (EnumHasAnyFlags(Options, EConvertToGraphOptions::Shrink))
116 {
117 OutGraphBuffer.Shrink();
118 OutGraph.Empty(NumVertices);
119 }
120 else
121 {
122 OutGraph.Reset(NumVertices);
123 }
124
125 // Convert offsets to TArrayView
126 for (const TPair<int64, int32>& EdgeOffset : EdgeOffsets)
127 {
128 OutGraph.Add(TConstArrayView<FVertex>(OutGraphBuffer.GetData() + EdgeOffset.Key, EdgeOffset.Value));
129 }
130}

147template <typename RangeType, typename ProjectionType>
148inline void ConvertToSingleBufferGraph(RangeType&& Graph, TArray64<FVertex>& OutGraphBuffer,
149 TArray<TConstArrayView<FVertex>>& OutGraph, ProjectionType Proj,
150 EConvertToGraphOptions Options = EConvertToGraphOptions::None)
151{
152 typedef TElementType_T<RangeType> InEdgeRangeType;
153 typedef decltype(Proj(std::declval<const InEdgeRangeType&>())) ProjectedEdgeRangeType;
154
155 int32 NumVertices = Graph.Num();
156 int64 NumEdges = 0;
157 for (const InEdgeRangeType& VertexEdges : Graph)
158 {
159 NumEdges += GetNum(Proj(VertexEdges));
160 }
161
162 if (EnumHasAnyFlags(Options, EConvertToGraphOptions::Shrink))
163 {
164 OutGraphBuffer.Empty(NumEdges);
165 OutGraph.Empty(NumVertices);
166 }
167 else
168 {
169 OutGraphBuffer.Reset(NumEdges);
170 OutGraph.Reset(NumVertices);
171 }
172
173 FVertex* OutBufferData = OutGraphBuffer.GetData();
174 for (const InEdgeRangeType& InVertexEdges : Graph)
175 {
176 ProjectedEdgeRangeType VertexEdges = Proj(InVertexEdges);
177 int32 NumVertexEdges = GetNum(VertexEdges);
178 OutGraph.Emplace(OutBufferData + OutGraphBuffer.Num(), NumVertexEdges);
179 OutGraphBuffer.Append(GetData(VertexEdges), NumVertexEdges);
180 }
181 check(OutGraphBuffer.GetData() == OutBufferData); // We have arrayviews into OutBufferData
182}
