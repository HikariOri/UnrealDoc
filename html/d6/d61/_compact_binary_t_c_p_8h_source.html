123template <typename KeyType, typename KeyFuncs, typename Allocator>
124inline bool LoadFromCompactBinary(FCbFieldView Field, TSet<KeyType, KeyFuncs, Allocator>& OutValue)
125{
126 OutValue.Reset();
127 OutValue.Reserve(Field.AsArrayView().Num());
128 bool bOk = !Field.HasError();
129 for (const FCbFieldView& ElementField : Field)
130 {
131 KeyType Key;
132 if (LoadFromCompactBinary(ElementField, Key))
133 {
134 OutValue.Add(MoveTemp(Key));
135 }
136 else
137 {
138 bOk = false;
139 }
140 }
141 return bOk;
142}

157template <typename KeyType, typename ValueType, typename SetAllocator, typename KeyFuncs>
158inline bool LoadFromCompactBinary(FCbFieldView Field, TMap<KeyType, ValueType, SetAllocator, KeyFuncs>& OutValue)
159{
160 OutValue.Reset();
161 OutValue.Reserve(Field.AsArrayView().Num());
162 bool bOk = !Field.HasError();
163 for (const FCbFieldView& PairField : Field)
164 {
165 bOk &= PairField.IsObject();
166 KeyType Key;
167 if (LoadFromCompactBinary(PairField["K"], Key))
168 {
169 ValueType& Value = OutValue.FindOrAdd(MoveTemp(Key));
170 bOk = LoadFromCompactBinary(PairField["V"], Value) & bOk;
171 }
172 else
173 {
174 bOk = false;
175 }
176 }
177 return bOk;
178}
