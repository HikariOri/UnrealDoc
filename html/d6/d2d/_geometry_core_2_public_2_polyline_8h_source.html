502 void SmoothSubdivide(TPolyline<T,D>& NewPolyline) const
503 {
504 const T Alpha = (T)1 / (T)3;
505 const T OneMinusAlpha = (T)2 / (T)3;
506
507 int N = Vertices.Num() - 1;
508 NewPolyline.Vertices.SetNum(2*N);
509 NewPolyline.Vertices[0] = Vertices[0];
510 int k = 1;
511 for (int i = 1; i < N; ++i)
512 {
513 const VectorType& Prev = Vertices[i-1];
514 const VectorType& Cur = Vertices[i];
515 const VectorType& Next = Vertices[i+1];
516 NewPolyline.Vertices[k++] = Alpha * Prev + OneMinusAlpha * Cur;
517 NewPolyline.Vertices[k++] = OneMinusAlpha * Cur + Alpha * Next;
518 }
519 NewPolyline.Vertices[k] = Vertices[N];
520 }

540 void Simplify(T ClusterTolerance, T LineDeviationTolerance, TArray<int32>& OutNewVertexIndices)
541 {
542 const int32 VertexCount = Vertices.Num();
543 if (VertexCount < 3)
544 {
545 OutNewVertexIndices.SetNumUninitialized(VertexCount);
546 for (int32 i = 0; i < VertexCount; ++i)
547 {
548 OutNewVertexIndices[i] = i;
549 }
550 // we need at least 3 vertices to be able to simplify a line
551 return;
552 }
553
554 OutNewVertexIndices.Reset(VertexCount);
555
556 // STAGE 1. Vertex Reduction within tolerance of prior vertex cluster
557 if (ClusterTolerance > 0)
558 {
559 T ClusterToleranceSquared = ClusterTolerance * ClusterTolerance;
560 OutNewVertexIndices.Add(0); // keep the first vertex
561 for (int32 Index = 1; Index < VertexCount-1; Index++)
562 {
563 if (Geometry::DistanceSquared(Vertices[Index], Vertices[OutNewVertexIndices.Last()]) < ClusterToleranceSquared)
564 {
565 continue;
566 }
567 OutNewVertexIndices.Add(Index);
568 }
569 OutNewVertexIndices.Add(VertexCount - 1); // keep the last vertex
570 }
571 else
572 {
573 OutNewVertexIndices.SetNumUninitialized(VertexCount);
574 for (int32 i = 0; i < VertexCount; ++i)
575 {
576 OutNewVertexIndices[i] = i;
577 }
578 }
579
580 // STAGE 2. Douglas-Peucker polyline simplification
581 TArray<bool> Marked;
582 if (LineDeviationTolerance > 0 && OutNewVertexIndices.Num() >= 3)
583 {
584 Marked.SetNumZeroed(OutNewVertexIndices.Num());
585
586 // mark the first and last vertices to make sure we keep them
587 Marked[0] = true;
588 Marked.Last() = true;
589 SimplifyDouglasPeucker(LineDeviationTolerance, Vertices, OutNewVertexIndices, 0, OutNewVertexIndices.Num()-1, Marked);
590 }
591
592 // STAGE 3. Copy back values in OutNewVertexIndices
593 if (!Marked.IsEmpty()) // If we have not run Douglas-Pecker, OutNewVertexIndices contains our final result
594 {
595 // only keep the marked ones
596 int32 OutNumNewVertexIndices = 0;
597 const int32 MarkedCount = Marked.Num();
598 for (int32 Index = 0; Index < MarkedCount; ++Index)
599 {
600 if (Marked[Index])
601 {
602 OutNewVertexIndices[OutNumNewVertexIndices++] = OutNewVertexIndices[Index];
603 }
604 }
605 OutNewVertexIndices.SetNum(OutNumNewVertexIndices);
606 }
607
608 if (OutNewVertexIndices.Num() != Vertices.Num())
609 {
610 check(OutNewVertexIndices.Num() < Vertices.Num());
611 for (int32 i = 0; i < OutNewVertexIndices.Num(); ++i)
612 {
613 Vertices[i] = Vertices[OutNewVertexIndices[i]]; // This only works since OutNewVertexIndices is sorted from lowest to higest vertex index
614 }
615
616 Vertices.SetNum(OutNewVertexIndices.Num());
617 }
618 }

620private:
621 // Polygon simplification
622 // code adapted from: http://softsurfer.com/Archive/algorithm_0205/algorithm_0205.htm
623 // simplifyDP():
624 // This is the Douglas-Peucker recursive simplification routine
625 // It just marks Vertices that are part of the simplified polyline
626 // for approximating the polyline subchain v[j] to v[k].
627 // Input: tol = approximation tolerance
628 // v[] = polyline array of vertex points
629 // j,k = indices for the subchain v[j] to v[k]
630 // Output: mk[] = array of markers matching vertex array v[]
631 static void SimplifyDouglasPeucker(T Tolerance, const TArray<VectorType>& Vertices, const TArray<int32>& VertexIndices, int32 j, int32 k, TArray<bool>& Marked)
632 {
633 check(Marked.Num() == VertexIndices.Num());
634 if (k <= j + 1) // there is nothing to simplify
635 return;
636
637 // check for adequate approximation by segment S from v[j] to v[k]
638 int maxi = j; // index of vertex farthest from S
639 T maxd2 = 0; // distance squared of farthest vertex
640 T tol2 = Tolerance * Tolerance; // tolerance squared
641 SegmentType S = SegmentType(Vertices[VertexIndices[j]], Vertices[VertexIndices[k]]); // segment from v[j] to v[k]
642
643 // test each vertex v[i] for max distance from S
644 // Note: this works in any dimension (2D, 3D, ...)
645 for (int i = j + 1; i < k; i++)
646 {
647 T dv2 = S.DistanceSquared(Vertices[VertexIndices[i]]);
648 if (dv2 <= maxd2)
649 continue;
650 // v[i] is a max vertex
651 maxi = i;
652 maxd2 = dv2;
653 }
654 if (maxd2 > tol2) // error is worse than the tolerance
655 {
656 // split the polyline at the farthest vertex from S
657 Marked[maxi] = true; // mark v[maxi] for the simplified polyline
658 // recursively simplify the two subpolylines at v[maxi]
659 SimplifyDouglasPeucker(Tolerance, Vertices, VertexIndices, j, maxi, Marked); // polyline v[j] to v[maxi]
660 SimplifyDouglasPeucker(Tolerance, Vertices, VertexIndices, maxi, k, Marked); // polyline v[maxi] to v[k]
661 }
662 // else the approximation is OK, so ignore intermediate Vertices
663 return;
664 }
