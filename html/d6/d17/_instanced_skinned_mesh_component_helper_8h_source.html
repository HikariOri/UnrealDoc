31template <class T, bool bSupportHitProxies>
32FInstanceDataManagerSourceDataDesc FInstancedSkinnedMeshComponentHelper::GetComponentDesc(T& InComponent, EShaderPlatform ShaderPlatform)
33{
34 FInstanceDataManagerSourceDataDesc ComponentDesc;
35
36 ComponentDesc.PrimitiveMaterialDesc = FPrimitiveComponentHelper::GetUsedMaterialPropertyDesc(InComponent, ShaderPlatform);
37
38 FInstanceDataFlags Flags;
39 Flags.bHasPerInstanceRandom = ComponentDesc.PrimitiveMaterialDesc.bAnyMaterialHasPerInstanceRandom;
40 Flags.bHasPerInstanceCustomData = ComponentDesc.PrimitiveMaterialDesc.bAnyMaterialHasPerInstanceCustomData && InComponent.NumCustomDataFloats != 0;
41#if WITH_EDITOR
42 if constexpr (bSupportHitProxies)
43 {
44 Flags.bHasPerInstanceEditorData = GIsEditor != 0 && InComponent.bHasPerInstanceHitProxies;
45 }
46#endif
47
48 const USkinnedAsset* SkinnedAsset = InComponent.GetSkinnedAsset();
49 const UTransformProviderData* TransformProvider = InComponent.GetTransformProvider();
50 const bool bForceRefPose = UInstancedSkinnedMeshComponent::ShouldForceRefPose();
51 const bool bValidTransformProvider = !bForceRefPose && TransformProvider != nullptr && TransformProvider->IsEnabled();
52 TConstArrayView<FTransform3f> InstancePrevTransforms = InComponent.GetInstancePrevTransforms();
53
54 Flags.bHasPerInstanceHierarchyOffset = false;
55 Flags.bHasPerInstanceLocalBounds = TransformProvider ? TransformProvider->HasAnimationBounds() : false;
56 const bool bHasCPUPrevTransforms = InstancePrevTransforms.Num() > 0 && InstancePrevTransforms.Num() == InComponent.GetInstanceCount();
57 bool bHasAttachmentParent = false;
58 if constexpr (std::is_same_v<T, UInstancedSkinnedMeshComponent>)
59 {
60 bHasAttachmentParent = InComponent.HasBoneAttachmentParents();
61 }
62 Flags.bHasPerInstanceDynamicData = bHasCPUPrevTransforms || bHasAttachmentParent;
63 Flags.bHasPerInstanceSkinningData = true;
64
65 Flags.bHasPerInstanceLMSMUVBias = false;//IsStaticLightingAllowed();
66
67 ComponentDesc.Flags = Flags;
68
69 const FTransform& ComponentTransform = InComponent.GetComponentTransform();
70 const FVector3f Position(ComponentTransform.GetTranslation());
71
72 ComponentDesc.MeshBounds = SkinnedAsset ? FRenderBounds(SkinnedAsset->GetBounds()) : FRenderBounds(Position, Position);
73 ComponentDesc.NumCustomDataFloats = InComponent.NumCustomDataFloats;
74 ComponentDesc.NumInstances = InComponent.InstanceData.Num();
75
76 ComponentDesc.PrimitiveLocalToWorld = InComponent.GetRenderMatrix();
77 ComponentDesc.ComponentMobility = InComponent.GetMobility();
78
79 ComponentDesc.BuildChangeSet =
80 [
81 &InComponent,
82 ComponentTransform,
83 InstancePrevTransforms,
84 TransformProvider = bValidTransformProvider ? TransformProvider : nullptr,
85 MeshBounds = ComponentDesc.MeshBounds,
86 bHasPerInstanceLocalBounds = Flags.bHasPerInstanceLocalBounds
87 ] (FInstanceUpdateChangeSet& ChangeSet)
88 {
89 // publish data
90 ChangeSet.GetTransformWriter().Gather([&InComponent](int32 InstanceIndex) -> FRenderTransform { return FRenderTransform(InComponent.InstanceData[InstanceIndex].Transform.ToMatrixWithScale()); });
91
92 if (InstancePrevTransforms.Num() == InComponent.GetInstanceCount())
93 {
94 ChangeSet.GetPrevTransformWriter().Gather([InstancePrevTransforms](int32 InstanceIndex) -> FRenderTransform { return FRenderTransform(InstancePrevTransforms[InstanceIndex].ToMatrixWithScale()); });
95 }
96 else if (ChangeSet.Flags.bHasPerInstanceDynamicData)
97 {
98 // No CPU previous transforms, but dynamic data is required (e.g., GPU-driven bone attachments).
99 // Seed with current transforms so the first frame has no motion blur artifacts.
100 ChangeSet.GetPrevTransformWriter().Gather([&InComponent](int32 InstanceIndex) -> FRenderTransform { return FRenderTransform(InComponent.InstanceData[InstanceIndex].Transform.ToMatrixWithScale()); });
101 }
102 ChangeSet.GetCustomDataWriter().Gather(MakeArrayView(InComponent.InstanceCustomData), InComponent.NumCustomDataFloats);
103
104 if (TransformProvider)
105 {
106 ChangeSet.GetSkinningDataWriter().Gather(
107 [&InComponent, TransformProvider, ComponentTransform](int32 InstanceIndex)->uint32
108 {
109 check(InComponent.InstanceData.IsValidIndex(InstanceIndex));
110 const FSkinnedMeshInstanceData& Instance = InComponent.InstanceData[InstanceIndex];
111 return TransformProvider->GetSkinningDataOffset(InstanceIndex, ComponentTransform, Instance);
112 });
113 }
114 else
115 {
116 ChangeSet.GetSkinningDataWriter().Gather(0u);
117 }
118
119 if (TransformProvider && bHasPerInstanceLocalBounds)
120 {
121 ChangeSet.GetLocalBoundsWriter().Gather(
122 [&InComponent, TransformProvider, MeshBounds](int32 InstanceIndex) -> FRenderBounds
123 {
124 const uint32 AnimationIndex = InComponent.InstanceData[InstanceIndex].AnimationIndex;
125 FRenderBounds AnimationBounds;
126 if (TransformProvider->GetAnimationBounds(AnimationIndex, AnimationBounds))
127 {
128 return AnimationBounds;
129 }
130 return MeshBounds;
131 });
132 }
133 else
134 {
135 ChangeSet.GetLocalBoundsWriter().Gather(MeshBounds);
136 }
137
138#if WITH_EDITOR
139 if constexpr (bSupportHitProxies)
140 {
141 if (ChangeSet.Flags.bHasPerInstanceEditorData)
142 {
143 // TODO: the way hit proxies are managed seems daft, why don't we just add them when needed and store them in an array alongside the instances?
144 // this will always force us to update all the hit proxy data for every instances.
145 TArray<TRefCountPtr<HHitProxy>> HitProxies;
146 InComponent.CreateHitProxyData(HitProxies);
147 ChangeSet.SetEditorData(HitProxies, InComponent.SelectedInstances);
148 }
149 }
150#endif
151
152
153 };
154
155 return ComponentDesc;
156}

158template <class T>
159FBoxSphereBounds FInstancedSkinnedMeshComponentHelper::CalcBounds(const T& InComponent, const FTransform& LocalToWorld)
160{
161 // Helper to union parent attachment bounds for child-driven bone attachments.
162 // Expands parent bounds by child mesh extent to account for geometry extending
163 // beyond the attachment point (e.g., a rifle attached to a hand bone).
164 auto UnionParentBounds = [&InComponent](FBoxSphereBounds InBounds) -> FBoxSphereBounds
165 {
166 if constexpr (std::is_same_v<T, UInstancedSkinnedMeshComponent>)
167 {
168 if (InComponent.BoneAttachmentSockets.Num() > 0)
169 {
170 const USkinnedAsset* ChildAsset = InComponent.GetSkinnedAsset();
171 const float ChildMeshExtent = ChildAsset ? ChildAsset->GetBounds().SphereRadius : 0.0f;
172
173 for (const auto& Socket : InComponent.BoneAttachmentSockets)
174 {
175 if (Socket.Parent.IsValid())
176 {
177 FBoxSphereBounds ExpandedParentBounds = Socket.Parent->Bounds;
178 ExpandedParentBounds.SphereRadius += ChildMeshExtent;
179 ExpandedParentBounds.BoxExtent += FVector(ChildMeshExtent);
180 InBounds = InBounds + ExpandedParentBounds;
181 }
182 }
183 }
184 }
185 return InBounds;
186 };
187
188 const USkinnedAsset* SkinnedAssetPtr = InComponent.GetSkinnedAsset();
189 if (SkinnedAssetPtr && InComponent.InstanceData.Num() > 0)
190 {
191 const FMatrix BoundTransformMatrix = LocalToWorld.ToMatrixWithScale();
192
193 FBoxSphereBounds::Builder BoundsBuilder;
194
195 const bool bUseAnimationBounds = UInstancedSkinnedMeshComponent::ShouldUseAnimationBounds();
196 const UTransformProviderData* TransformProvider = InComponent.GetTransformProvider();
197 if (TransformProvider != nullptr && TransformProvider->IsEnabled() && TransformProvider->HasAnimationBounds() && !TransformProvider->IsCompiling())
198 {
199 const uint32 AnimationCount = TransformProvider->GetUniqueAnimationCount();
200
201 // Trade per sequence bounds (tighter fitting) for faster builds with high instance counts.
202 const bool bFastBuild = false;
203 if (bFastBuild)
204 {
205 FBox MergedBounds;
206
207 for (uint32 AnimationIndex = 0; AnimationIndex < AnimationCount; ++AnimationIndex)
208 {
209 FRenderBounds AnimationBounds;
210 if (TransformProvider->GetAnimationBounds(AnimationIndex, AnimationBounds))
211 {
212 MergedBounds += AnimationBounds.ToBox();
213 }
214 }
215
216 if (MergedBounds.IsValid)
217 {
218 for (int32 InstanceIndex = 0; InstanceIndex < InComponent.InstanceData.Num(); InstanceIndex++)
219 {
220 const FSkinnedMeshInstanceData& Instance = InComponent.InstanceData[InstanceIndex];
221 BoundsBuilder += MergedBounds.TransformBy(FTransform(Instance.Transform) * LocalToWorld);
222 }
223 }
224 }
225 else
226 {
227 for (int32 InstanceIndex = 0; InstanceIndex < InComponent.InstanceData.Num(); InstanceIndex++)
228 {
229 const FSkinnedMeshInstanceData& Instance = InComponent.InstanceData[InstanceIndex];
230 FRenderBounds AnimationBounds;
231 if (TransformProvider->GetAnimationBounds(Instance.AnimationIndex, AnimationBounds))
232 {
233 BoundsBuilder += AnimationBounds.ToBox().TransformBy(FTransform(Instance.Transform) * LocalToWorld);
234 }
235 }
236 }
237
238 // Only use bounds if valid, otherwise use the skinned asset bounds in ref pose.
239 if (BoundsBuilder.IsValid())
240 {
241 return UnionParentBounds(BoundsBuilder);
242 }
243 }
244
245 const FBox InstanceBounds = SkinnedAssetPtr->GetBounds().GetBox();
246 if (InstanceBounds.IsValid)
247 {
248 for (int32 InstanceIndex = 0; InstanceIndex < InComponent.InstanceData.Num(); InstanceIndex++)
249 {
250 BoundsBuilder += InstanceBounds.TransformBy(FTransform(InComponent.InstanceData[InstanceIndex].Transform) * LocalToWorld);
251 }
252
253 return UnionParentBounds(BoundsBuilder);
254 }
255 }
256
257 return UnionParentBounds(InComponent.CalcMeshBound(FVector3f::ZeroVector, false, LocalToWorld));
258}

266template <class T>
267bool FInstancedSkinnedMeshComponentHelper::IsEnabled(const T& InComponent)
268{
269 USkeletalMesh* SkeletalMeshPtr = Cast<USkeletalMesh>(InComponent.GetSkinnedAsset());
270 if (SkeletalMeshPtr && SkeletalMeshPtr->GetResourceForRendering())
271 {
272 return InComponent.GetInstanceCount() > 0;
273 }
274
275 return false;
