45enum class EPluginSizeTypes : uint8
46{
47 // The iostore chunks will be deployed to servers for downloading on-demand by the game using the Individual Asset Streaming
48 // system.
49 Streaming,
50 // The iostore chunks are written to normal iostore containers that are expected to be distributed with the game. This
51 // includes the required global iostore container.
52 Installed,
53 // The iostore chunks are written to a separate container that isn't required to be distributed with the game. This is where e.g.
54 // OptionalMips go for textures. They appear as .uptnl files in the Cooked directory (if cooking to Loose Files),
55 // The CopyBuildToStagingDirectory script manually assigns the iostore chunk to a corresponding pak chunk with the name
56 // ending in "optional", e.g. pakChunk0optional.pak/ucas/utoc/sig. The only way to catch this in UnrealPak
57 // is to parse the filename.
58 Optional,
59 // These are sidecar files for distributing EditorOnly data for Cooked Editor builds. When cooking to loose files
60 // they will contain ".o" inside their filename, e.g. "myasset.o.ubulk". They are never intended to be shipped with
61 // a game.
62 OptionalSegment,
63 COUNT
64};

67struct FPluginSizeInfo
68{
69 uint64 Sizes[EPluginSizeTypesCount] = {};
70
71 void Zero() { FMemory::Memzero(this, sizeof(*this)); }
72 void AddSizes(uint64 SizesPerType[EPluginSizeTypesCount])
73 {
74 for (uint8 Type = 0; Type < EPluginSizeTypesCount; Type++)
75 {
76 Sizes[Type] += SizesPerType[Type];
77 }
78 }
79 void Add(const FPluginSizeInfo& Other)
80 {
81 for (uint8 Type = 0; Type < EPluginSizeTypesCount; Type++)
82 {
83 Sizes[Type] += Other.Sizes[Type];
84 }
85 }
86 uint64 TotalSize() const
87 {
88 uint64 Total = 0;
89 for (uint8 Type = 0; Type < EPluginSizeTypesCount; Type++)
90 {
91 Total += Sizes[Type];
92 }
93 return Total;
94 }
95
96 friend FArchive& operator<<(FArchive& Ar, FPluginSizeInfo& SizeInfo)
97 {
98 for (uint8 i = 0; i < EPluginSizeTypesCount; i++)
99 {
100 Ar << SizeInfo.Sizes[i];
101 }
102 return Ar;
103 }
104
105 uint64& operator[](EPluginSizeTypes InType) { return Sizes[(uint8)InType]; }
106 const uint64& operator[](EPluginSizeTypes InType) const { return Sizes[(uint8)InType]; }
107};

162struct FCookMetadataPluginEntry
163{
164 FString Name;
165
166 ECookMetadataPluginType Type = ECookMetadataPluginType::Unassigned;
167
168 // These contain values pulled from the uplugin json file and hold fields that are not
169 // part of the engine FPluginDescriptor. They are for per-project values. The keys for the maps
170 // are indices in to FCookMetadataPluginHierarchy::CustomFieldEntries, where you'll also find what the type
171 // is. See the comment for CustomFieldEntries.
172 typedef TVariant<bool, FString> CustomFieldVariantType;
173 TMap<uint8, CustomFieldVariantType> CustomFields;
174
175 // The dependencies are stored in the FCookMetadataPluginHierarchy::PluginDependencies array,
176 // and this is an index into it. From there you can get a further index into PluginsEnabledAtCook
177 // to get the plugin information.
178 // Example:
179 // for (uint32 DependencyIndex = Plugin->DependencyIndexStart; DependencyIndex < Plugin->DependencyIndexEnd; DependencyIndex++)
180 // {
181 // const UE::Cook::FCookMetadataPluginEntry& DependentPlugin = PluginHierarchy.PluginsEnabledAtCook[PluginHierarchy.PluginDependencies[DependencyIndex]];
182 // }
183 //
184 uint32 DependencyIndexStart = 0;
185 uint32 DependencyIndexEnd = 0;
186
187 //
188 // Theses sizes are set during staging by unrealpak if the option in project packaging is set.
189 // To determine if they are set, check FCookMetadataState::GetSizesPresent(). Inclusive contains
190 // the size of the plugin and all of its dependencies listed in its uplugin file.
191 //
192 FPluginSizeInfo InclusiveSizes;
193 FPluginSizeInfo ExclusiveSizes;
194
195 uint32 DependencyCount() const { return DependencyIndexEnd - DependencyIndexStart; }
196
197 UE_API FText GetPluginTypeAsText() const;
198
199 // !!! If you edit this, be sure to update the upgrade paths in CookMetadata.cpp
200 friend FArchive& operator<<(FArchive& Ar, FCookMetadataPluginEntry& Entry)
201 {
202 Ar << Entry.Name << Entry.DependencyIndexStart << Entry.DependencyIndexEnd;
203 Ar << Entry.InclusiveSizes << Entry.ExclusiveSizes;
204 Ar << Entry.CustomFields << Entry.Type;
205 return Ar;
206 }
207};

209struct FCookMetadataPluginHierarchy
210{
211 // The list of plugins that were enabled during the cook that generated the FCookMetadataState
212 TArray<FCookMetadataPluginEntry> PluginsEnabledAtCook;
213
214 // The list of plugin dependencies. FCookMetadataPluginEntry::DependencyIndexStart indexes into this
215 // array.
216 TArray<uint16> PluginDependencies;
217
218 // The list of root plugins for the project as defined by the Editor.ini file.
219 TArray<uint16> RootPlugins;
220
221 // The list of custom field names. The values stored on an entry index in to this for the name.
222 // These values are copied from the plugins' json descriptor to allow for carrying through project
223 // specific data in to the metadata.
224 //
225 // To enable custom field replication, add the CookMetadataCustomPluginFields section to the relevant Editor.ini file
226 // and add to the follow arrays, as desired:
227 //
228 // BoolFields, StringFields, PerPlatformBoolFields, PerPlatformStringFields
229 //
230 // Bool and String refer to the json type, and will emit as bool or FString here. The PerPlatform arrays
231 // will check in the uplugin json for a field named PerPlatform<ArrayValue>. That field is expected to be
232 // an array of override objects. Each object is required to have the platform name and override value. E.g.:
233 //
234 // [CookMetadataCustomPluginFields]
235 // +PerPlatformBoolFields = ExampleProjectBool
236 // +StringField = ExampleProjectString
237 //
238 // Will look in the uplugin file for this:
239 //
240 // "ExampleProjectBool": true
241 // "PerPlatformExampleProjectBool": [
242 // {
243 // "Platform": "Windows",
244 // "Value": false
245 // }
246 // ]
247 // "ExampleProjectString": "AProjectString"
248 //
249 // Note that in the per platform case, the base value is not required (i.e. ExampleProjectBool above), however it
250 // provides a default value for unlisted platforms. If it does not exist, false is used for bool fields, and an empty string
251 // for string fields.
252 struct FCustomFieldEntry
253 {
254 FString Name;
255 ECookMetadataCustomFieldType Type;
256 friend FArchive& operator<<(FArchive& Ar, FCustomFieldEntry& Entry)
257 {
258 return Ar << Entry.Name << Entry.Type;
259 }
260 };
261 TArray<FCustomFieldEntry> CustomFieldEntries;
262
263 // !!! If you edit this, be sure to update the upgrade paths in CookMetadata.cpp
264 friend FArchive& operator<<(FArchive& Ar, FCookMetadataPluginHierarchy& Hierarchy)
265 {
266 Ar << Hierarchy.PluginsEnabledAtCook << Hierarchy.PluginDependencies;
267 Ar << Hierarchy.RootPlugins << Hierarchy.CustomFieldEntries;
268 return Ar;
269 }
270};
