46struct FVertexFormat
47{
48 uint8 NumTexCoords = 0;

50 bool bHasTangents : 1 = false;
51 bool bHasCurveRadius : 1 = false;

53
54 FORCEINLINE bool Matches( const FVertexFormat& Other ) const
55 {
56 return
57 NumTexCoords == Other.NumTexCoords &&
58 NumBoneInfluences == Other.NumBoneInfluences &&
59 bHasTangents == Other.bHasTangents &&
60 bHasCurveRadius == Other.bHasCurveRadius &&
61 bHasColors == Other.bHasColors;
62 }
63
64 FORCEINLINE uint8 GetCurveRadiusOffset() const
65 {
66 return 6 + ( bHasTangents ? 4 : 0 );
67 }
68
69 FORCEINLINE uint8 GetColorOffset() const
70 {
71 return GetCurveRadiusOffset() + ( bHasCurveRadius ? 4 : 0 );
72 }
73
74 FORCEINLINE uint8 GetUVOffset() const
75 {
76 return GetColorOffset() + ( bHasColors ? 4 : 0 );
77 }
78
79 FORCEINLINE uint8 GetBoneInfluenceOffset() const
80 {
81 return GetUVOffset() + (NumTexCoords * 2);
82 }
83
84 FORCEINLINE uint8 GetVertSize() const
85 {
86 return GetBoneInfluenceOffset() + (NumBoneInfluences * 2);
87 }
