16class FCompositePrimitiveShaderBase : public FGlobalShader
17{
18public:
19 static const uint32 kMSAASampleCountMax = 8;
20
21 class FSampleCountDimension : SHADER_PERMUTATION_RANGE_INT("MSAA_SAMPLE_COUNT", 1, kMSAASampleCountMax + 1);
22 class FMSAADontResolve : SHADER_PERMUTATION_BOOL("MSAA_DONT_RESOLVE");
23 using FPermutationDomain = TShaderPermutationDomain<FSampleCountDimension, FMSAADontResolve>;
24
25 static bool ShouldCompilePermutation(const FPermutationDomain& PermutationVector, const EShaderPlatform Platform)
26 {
27 const int32 SampleCount = PermutationVector.Get<FSampleCountDimension>();
28 const bool bMSAADontResolve = PermutationVector.Get<FMSAADontResolve>();
29
30 // Only use permutations with valid MSAA sample counts.
31 if (!FMath::IsPowerOfTwo(SampleCount))
32 {
33 return false;
34 }
35 if (!RHISupportsMSAA(Platform) && (SampleCount > 1 || bMSAADontResolve))
36 {
37 return false;
38 }
39
40 return IsPCPlatform(Platform);
41 }
42
43 static bool ShouldCompilePermutation(const FGlobalShaderPermutationParameters& Parameters)
44 {
45 const FPermutationDomain PermutationVector(Parameters.PermutationId);
46 return ShouldCompilePermutation(PermutationVector, Parameters.Platform);
47 }
48
49 FCompositePrimitiveShaderBase() = default;
50 FCompositePrimitiveShaderBase(const ShaderMetaType::CompiledShaderInitializerType& Initializer)
51 : FGlobalShader(Initializer)
52 {}
53};
