9SYMS_API SYMS_MsfHeaderInfo
10syms_msf_header_info_from_data_slow(SYMS_String8 data){
11 void *base = data.str;
12 SYMS_U64Range range = syms_make_u64_range(0, data.size);
13
14 // determine msf type
15 SYMS_U32 index_size = 0;
16 char magic_buffer[SYMS_MSF_MAX_MAGIC_SIZE];
17 if (syms_based_range_read(base, range, 0, SYMS_MSF20_MAGIC_SIZE, magic_buffer) &&
18 syms_memcmp(syms_msf20_magic, magic_buffer, SYMS_MSF20_MAGIC_SIZE) == 0){
19 index_size = 2;
20 }
21 else if (syms_based_range_read(base, range, 0, SYMS_MSF70_MAGIC_SIZE, magic_buffer) &&
22 syms_memcmp(syms_msf70_magic, magic_buffer, SYMS_MSF70_MAGIC_SIZE) == 0){
23 index_size = 4;
24 }
25
26 // grab parts of header we will use in syms
27 SYMS_MsfHeaderInfo result = {0};
28
29 if (index_size == 2){
30 SYMS_MsfHeader20 header = {0};
31 if (syms_based_range_read_struct(base, range, SYMS_MSF20_MAGIC_SIZE, &header)){
32 result.index_size = index_size;
33 result.block_size = header.block_size;
34 result.block_count = header.block_count;
35 result.directory_size = header.directory_size;
36 }
37 }
38 else if (index_size == 4){
39 SYMS_MsfHeader70 header = {0};
40 if (syms_based_range_read_struct(base, range, SYMS_MSF70_MAGIC_SIZE, &header)){
41 result.index_size = index_size;
42 result.block_size = header.block_size;
43 result.block_count = header.block_count;
44 result.directory_size = header.directory_size;
45 result.directory_super_map = header.directory_super_map;
46 }
47 }
48
49 return(result);
50}

55SYMS_API SYMS_MsfAccel*
56syms_msf_accel_from_data(SYMS_Arena *arena, SYMS_String8 data){
57 // NOTE(allen): Layout of directory
58 //
59 // PDB20:
60 // struct Pdb20StreamSize{
61 // U32 size;
62 // U32 unknown; // looks like kind codes or revision counters or something
63 // }
64 // struct{
65 // U32 stream_count;
66 // Pdb20StreamSize stream_sizes[stream_count];
67 // U16 stream_indices[stream_count][...];
68 // }
69 //
70 // PDB70:
71 // struct{
72 // U32 stream_count;
73 // U32 stream_sizes[stream_count];
74 // U32 stream_indices[stream_count][...];
75 // }
76
77 //- setup result
78 SYMS_MsfAccel *result = 0;
79
80 //- header
81 SYMS_MsfHeaderInfo header = syms_msf_header_info_from_data_slow(data);
82 if (header.index_size > 0){
83
84 //- directory
85 SYMS_B32 got_directory = syms_true;
86 SYMS_U8 *directory = 0;
87 {
88 SYMS_U32 directory_size = header.directory_size;
89 directory = syms_push_array(arena, SYMS_U8, directory_size);
90
91 // setup important sizes and counts
92 SYMS_U32 size_of_block_index = header.index_size;
93
94 SYMS_U64 file_size = data.size;
95 SYMS_U32 block_size = SYMS_ClampTop(header.block_size, file_size);
96
97 SYMS_U32 block_count_in_directory = SYMS_CeilIntegerDiv(directory_size, block_size);
98 SYMS_U32 directory_map_size = block_count_in_directory*size_of_block_index;
99
100 SYMS_U32 block_count_in_directory_map = SYMS_CeilIntegerDiv(directory_map_size, block_size);
101
102 SYMS_U32 block_count_in_whole_file_max = SYMS_CeilIntegerDiv(file_size, block_size);
103 SYMS_U32 block_count_in_whole_file = SYMS_ClampTop(header.block_count, block_count_in_whole_file_max);
104
105 // setup the index blocks
106 SYMS_U32 directory_super_map_dummy = 0;
107 SYMS_U32 *directory_super_map = 0;
108 SYMS_U32 directory_map_block_skip_size = 0;
109
110 if (size_of_block_index == 2){
111 directory_super_map = &directory_super_map_dummy;
112 directory_map_block_skip_size = SYMS_MSF20_MAGIC_SIZE + SYMS_MEMBER_OFFSET(SYMS_MsfHeader20, directory_map);
113 }
114 else{
115 SYMS_U64 super_map_off = SYMS_MSF70_MAGIC_SIZE + SYMS_MEMBER_OFFSET(SYMS_MsfHeader70, directory_super_map);
116 directory_super_map = (SYMS_U32*)(data.str + super_map_off);
117 }
118
119 SYMS_U32 max_index_count_in_map_block = (block_size - directory_map_block_skip_size)/size_of_block_index;
120
121 // super map: [s1, s2, s3, ...]
122 // map: s1 -> [i1, i2, i3, ...]; s2 -> [...]; s3 -> [...]; ...
123 // directory: i1 -> [data]; i2 -> [data]; i3 -> [data]; ... i1 -> [data]; ...
124
125 // for each index in super map ...
126 SYMS_U8 *out_ptr = directory;
127 SYMS_U32 *super_map_ptr = directory_super_map;
128 for (SYMS_U32 i = 0;
129 i < block_count_in_directory_map;
130 i += 1, super_map_ptr += 1){
131 SYMS_U32 directory_map_block_index = *super_map_ptr;
132 if (directory_map_block_index >= block_count_in_whole_file){
133 // TODO(allen): File Defect: (Block Index, In Super Directory, Too Large)
134 got_directory = syms_false;
135 goto read_directory_done;
136 }
137
138 SYMS_U64 directory_map_block_off = (SYMS_U64)(directory_map_block_index)*block_size;
139 SYMS_U8 *directory_map_block_base = data.str + directory_map_block_off;
140
141 // clamp index count by end of directory
142 SYMS_U32 index_count = 0;
143 {
144 SYMS_U32 directory_pos = (SYMS_U32)(out_ptr - directory);
145 SYMS_U32 remaining_size = directory_size - directory_pos;
146 SYMS_U32 remaining_map_block_count = SYMS_CeilIntegerDiv(remaining_size, block_size);
147 index_count = SYMS_ClampTop(max_index_count_in_map_block, remaining_map_block_count);
148 }
149
150 // for each index in map ...
151 SYMS_U8 *map_ptr = directory_map_block_base + directory_map_block_skip_size;
152 for (SYMS_U32 j = 0;
153 j < index_count;
154 j += 1, map_ptr += size_of_block_index){
155 SYMS_U32 directory_block_index = 0;
156 syms_memmove(&directory_block_index, map_ptr, size_of_block_index);
157 if (directory_block_index >= block_count_in_whole_file){
158 // TODO(allen): File Defect: (Block Index, In Directory, Too Large)
159 got_directory = syms_false;
160 goto read_directory_done;
161 }
162
163 SYMS_U64 directory_block_off = (SYMS_U64)(directory_block_index)*block_size;
164 SYMS_U8 *directory_block_base = data.str + directory_block_off;
165
166 // clamp copy size by end of directory
167 SYMS_U32 copy_size = 0;
168 {
169 SYMS_U32 directory_pos = (SYMS_U32)(out_ptr - directory);
170 SYMS_U32 remaining_size = directory_size - directory_pos;
171 copy_size = SYMS_ClampTop(block_size, remaining_size);
172 }
173
174 // copy block data
175 syms_memmove(out_ptr, directory_block_base, copy_size);
176 out_ptr += copy_size;
177 }
178 }
179
180 read_directory_done:;
181 }
182 SYMS_U64Range directory_range = {0, header.directory_size};
183
184 //- stream info
185 SYMS_U32 stream_count = 0;
186 SYMS_MsfAccelStreamInfo *stream_info = 0;
187 {
188 // read count
189 syms_based_range_read(directory, directory_range, 0, 4, &stream_count);
190
191 // allocate info array
192 stream_info = syms_push_array_zero(arena, SYMS_MsfAccelStreamInfo, stream_count);
193
194 // setup counts, sizes, and offsets for the directory data
195 SYMS_U32 block_size = header.block_size;
196 SYMS_U32 size_of_block_index = header.index_size;
197 SYMS_U32 size_of_stream_size_entry = 4;
198 if (size_of_block_index == 2){
199 size_of_stream_size_entry = 8;
200 }
201 SYMS_U32 all_sizes_off = 4;
202 SYMS_U32 all_indices_off = all_sizes_off + stream_count*size_of_stream_size_entry;
203
204 // iterate sizes and indices in lock step
205 SYMS_U32 size_cursor = all_sizes_off;
206 SYMS_U32 index_cursor = all_indices_off;
207 SYMS_MsfAccelStreamInfo *stream_info_ptr = stream_info;
208 for (SYMS_U32 i = 0; i < stream_count; i += 1){
209 // read stream size
210 SYMS_U32 stream_size = 0;
211 syms_based_range_read(directory, directory_range, size_cursor, 4, &stream_size);
212 if (stream_size == 0xffffffff){
213 stream_size = 0;
214 }
215
216 // compute block count
217 SYMS_U32 stream_block_count = SYMS_CeilIntegerDiv(stream_size, block_size);
218
219 // save stream info
220 stream_info_ptr->stream_indices = directory + index_cursor;
221 stream_info_ptr->size = stream_size;
222
223 // advance cursors
224 size_cursor += size_of_stream_size_entry;
225 index_cursor += stream_block_count*size_of_block_index;
226 stream_info_ptr += 1;
227 }
228 }
229
230 // fill result
231 result = syms_push_array(arena, SYMS_MsfAccel, 1);
232 result->header = header;
233 result->stream_count = stream_count;
234 result->stream_info = stream_info;
235 }
236
237 return(result);
238}

312SYMS_API SYMS_B32
313syms_msf_read(SYMS_String8 data, SYMS_MsfAccel *msf,
314 SYMS_MsfStreamNumber sn, SYMS_U32 off, SYMS_U32 size, void *out){
315 SYMS_B32 result = syms_false;
316
317 // stream info
318 SYMS_MsfStreamInfo stream_info = syms_msf_stream_info_from_sn(msf, sn);
319 if (size > 0 && off + size <= stream_info.size){
320 // copy block-by-block
321 SYMS_U64 file_size = data.size;
322 SYMS_U32 block_size = msf->header.block_size;
323 SYMS_U32 size_of_block_index = msf->header.index_size;
324 SYMS_U32 block_count_in_whole_file_max = SYMS_CeilIntegerDiv(file_size, (SYMS_U64)block_size);
325 SYMS_U32 block_count_in_whole_file = SYMS_ClampTop(msf->header.block_count, block_count_in_whole_file_max);
326 SYMS_U32 block_count_in_stream = SYMS_CeilIntegerDiv((SYMS_U64)stream_info.size, (SYMS_U64)block_size);
327
328 SYMS_U32 completed_amount = 0;
329 for (;;){
330 // offset-in-stream -> part-index
331 SYMS_U32 remaining_off = off + completed_amount;
332 SYMS_U32 src_part_index = remaining_off/block_size;
333 SYMS_ASSERT(src_part_index < block_count_in_stream);
334
335 // part-index -> block-index
336 SYMS_U32 src_block_index_off = src_part_index*size_of_block_index;
337 SYMS_U32 src_block_index = 0;
338 syms_memmove(&src_block_index, stream_info.stream_indices + src_block_index_off, size_of_block_index);
339 if (src_block_index >= block_count_in_whole_file){
340 // TODO(allen): File Defect: (Block Index, In Stream, Too Large)
341 break;
342 }
343
344 // block-index -> range-in-file
345 SYMS_U64 src_block_base = (SYMS_U64)(src_block_index)*block_size;
346 SYMS_U32 relative_off = remaining_off%block_size;
347 SYMS_U32 relative_opl = SYMS_ClampTop(relative_off + size - completed_amount, block_size);
348 SYMS_U32 contiguous_size = relative_opl - relative_off;
349 SYMS_ASSERT(src_block_base + relative_opl <= file_size);
350
351 // copy & advance
352 syms_memmove((SYMS_U8*)out + completed_amount, data.str + src_block_base + relative_off, contiguous_size);
353 completed_amount += contiguous_size;
354 if (completed_amount >= size){
355 result = syms_true;
356 break;
357 }
358 }
359 }
360
361 return(result);
362}

426SYMS_API SYMS_String8
427syms_msf_read_zstring_in_range(SYMS_Arena *arena, SYMS_String8 data, SYMS_MsfAccel *msf,
428 SYMS_MsfRange range, SYMS_U32 r_off){
429 SYMS_ArenaTemp scratch = syms_get_scratch(&arena, 1);
430
431 // build a list of string chunks on scratch
432 SYMS_String8List list = {0};
433
434 SYMS_U32 off = range.off + r_off;
435 SYMS_U32 max_off = range.off + range.size;
436
437 SYMS_MsfStreamInfo stream_info = syms_msf_stream_info_from_sn(msf, range.sn);
438 if (off < stream_info.size && off < max_off){
439 // scan block-by-block
440 SYMS_U64 file_size = data.size;
441 SYMS_U32 block_size = msf->header.block_size;
442 SYMS_U32 size_of_block_index = msf->header.index_size;
443 SYMS_U32 block_count_in_whole_file_max = SYMS_CeilIntegerDiv(file_size, block_size);
444 SYMS_U32 block_count_in_whole_file = SYMS_ClampTop(msf->header.block_count, block_count_in_whole_file_max);
445 SYMS_U32 block_count_in_stream = SYMS_CeilIntegerDiv(stream_info.size, block_size);
446
447 SYMS_U32 max_scan_amount = max_off - off;
448 SYMS_U32 scanned_amount = 0;
449 for (;;){
450 // offset-in-stream -> part-index
451 SYMS_U32 remaining_off = off + scanned_amount;
452 SYMS_U32 src_part_index = remaining_off/block_size;
453 if (src_part_index >= block_count_in_stream){
454 break;
455 }
456
457 // part-index -> block-index
458 SYMS_U32 src_block_index_off = src_part_index*size_of_block_index;
459 SYMS_U32 src_block_index = 0;
460 syms_memmove(&src_block_index, stream_info.stream_indices + src_block_index_off, size_of_block_index);
461 if (src_block_index >= block_count_in_whole_file){
462 // TODO(allen): File Defect: (Block Index, In Stream, Too Large)
463 break;
464 }
465 if (src_block_index >= block_count_in_whole_file){
466 // TODO(allen): Detected defect in the file data.
467 // (Block Index, In Stream, Too Large)
468 break;
469 }
470
471 // block-index -> range-in-file
472 SYMS_U64 src_block_base = (SYMS_U64)(src_block_index)*block_size;
473 SYMS_U32 relative_off = remaining_off%block_size;
474 SYMS_U32 remaining_max_size = max_off - remaining_off;
475 SYMS_U32 relative_opl = SYMS_ClampTop(relative_off + remaining_max_size, block_size);
476 SYMS_ASSERT(src_block_base + block_size <= file_size);
477
478 // scan
479 SYMS_U8 *start = data.str + src_block_base + relative_off;
480 SYMS_U8 *opl = start + relative_opl - relative_off;
481 SYMS_U8 *ptr = start;
482 for (;ptr < opl && *ptr != 0; ptr += 1);
483 scanned_amount += (SYMS_U64)(ptr - start);
484
485 // emit chunk
486 if (start < ptr){
487 SYMS_String8 str_range = syms_str8_range(start, ptr);
488 syms_string_list_push(scratch.arena, &list, str_range);
489 }
490
491 // check end of scan conditions
492 if (ptr < opl){
493 break;
494 }
495 if (scanned_amount >= max_scan_amount){
496 break;
497 }
498 }
499 }
500
501
502 // join into single string
503 SYMS_String8 result = syms_string_list_join(arena, &list, 0);
504 syms_release_scratch(scratch);
505 return(result);
506}
