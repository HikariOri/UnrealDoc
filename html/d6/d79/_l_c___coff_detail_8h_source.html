56 template <typename SymbolType>
57 inline uint32_t GetSectionIndex(const SymbolType* symbol)
58 {
59 // according to COFF spec, section number is a one-based (because zero is taken by UNDEFINED symbols)
60 // signed integer, and signed values like 0, -1 and -2 have special meaning. note that
61 // in cases where the index is a negative value that does not correspond to any of
62 // the special values, the section number *must* be treated as an unsigned value, which is not
63 // stated in the spec.
64 // e.g. some non-bigobj COFF files that have more than 32767 sections will have symbols with
65 // section numbers like 0x8000, 0x8001, etc. which need to be treated as unsigned values.
66 typedef typename std::make_unsigned<decltype(symbol->SectionNumber)>::type UnsignedSectionNumberType;
67
68 const uint32_t sectionIndex = static_cast<uint32_t>(static_cast<UnsignedSectionNumberType>(symbol->SectionNumber) - 1u);
69 return sectionIndex;
70 }

86 template <>
87 inline uint32_t GetAssociatedComdatSectionIndex(const IMAGE_AUX_SYMBOL_EX* symbol)
88 {
89 typedef std::make_unsigned<decltype(symbol->Section.Number)>::type UnsignedSectionNumberType;
90 typedef std::make_unsigned<decltype(symbol->Section.HighNumber)>::type UnsignedHighSectionNumberType;
91
92 const uint32_t sectionIndex = static_cast<uint32_t>(static_cast<UnsignedSectionNumberType>(symbol->Section.Number) - 1u);
93
94 // note that the high number is zero-based instead of one-based
95 const uint32_t highSectionIndex = static_cast<uint32_t>(static_cast<UnsignedHighSectionNumberType>(symbol->Section.HighNumber));
96
97 return (highSectionIndex << 16u) | sectionIndex;
98 }

108 /*
109 From the COFF spec: 5.4.2 Section Number Values
110 Normally, the Section Value field in a symbol table entry is a one-based index into the section table.
111 However, this field is a signed integer and can take negative values. The following values, less than one,
112 have special meanings.
113
114 IMAGE_SYM_UNDEFINED The symbol record is not yet assigned a section. A value of zero indicates that a reference
115 to an external symbol is defined elsewhere. A value of non-zero is a common symbol with a size that is specified
116 by the value.
117
118 IMAGE_SYM_ABSOLUTE The symbol has an absolute (non-relocatable) value and is not an address.
119
120 IMAGE_SYM_DEBUG The symbol provides general type or debugging information but does not correspond to a section.
121 Microsoft tools use this setting along with .file records (storage class FILE).
122 */
123
124 template <typename SymbolType>
125 inline bool IsUndefinedSymbol(const SymbolType* symbol)
126 {
127 return (symbol->SectionNumber == IMAGE_SYM_UNDEFINED);
128 }

143 template <typename SymbolType>
144 inline bool IsSectionSymbol(const SymbolType* symbol)
145 {
146 /*
147 From the COFF spec: 5.5.5 Auxiliary Format 5: Section Definitions
148 Follows a symbol-table record that defines a section. The auxiliary record provides information
149 about the section to which it refers.
150 */
151 if (symbol->NumberOfAuxSymbols > 0u)
152 {
153 /*
154 From the COFF spec: 5.4.4 Storage Class
155 IMAGE_SYM_CLASS_STATIC: The offset of the symbol within the section. If the Value field is zero,
156 then the symbol represents a section name.
157 */
158 if ((symbol->StorageClass == IMAGE_SYM_CLASS_STATIC) && (symbol->Value == 0u))
159 {
160 return true;
161 }
162 }
163
164 return false;
165 }

286 inline uint32_t GetArchiveMemberSize(const IMAGE_ARCHIVE_MEMBER_HEADER* header)
287 {
288 // COFF Spec, 7.2. Archive Member Headers:
289 // Field "Size" at offset 48: An ASCII decimal representation of the total size of the archive member, not including the size of the header.
290 uint32_t size = 0u;
291 sscanf_s(reinterpret_cast<const char*>(header->Size), "%" SCNu32, &size);
292
293 return size;
294 }

297 inline uint32_t PadArchiveMemberSize(uint32_t size)
298 {
299 // COFF Spec, 7.2. Archive Member Headers:
300 // "Each member header starts on the first even address after the end of the previous archive member."
301 return size + (size & 1u);
302 }
