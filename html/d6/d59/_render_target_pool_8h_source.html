43 uint32 AddRef() const override
44 {
45 return NumRefs.fetch_add(1, std::memory_order_relaxed) + 1;
46 }
47
48 uint32 Release() override
49 {
50 const uint32 OldNumRefs = NumRefs.fetch_sub(1, std::memory_order_release);
51 check(OldNumRefs > 0);
52 if (OldNumRefs == 1)
53 {
54 std::atomic_thread_fence(std::memory_order_acquire);
55 delete this;
56 }
57 return OldNumRefs - 1;
58 }
59
60 uint32 GetRefCount() const override
61 {
62 return NumRefs.load(std::memory_order_relaxed);
63 }
