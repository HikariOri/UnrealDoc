68 // Use a macro for this since it's impossible to do with a template since the function pointer that would be needed might not exist.
69 // The macro also has the added benefit that it can print the function name for better reporting.
70 // CheckArgs and CallArgs are both parenthesized argument packs. CheckArgs uses lvalue argument names for the
71 // requires() expression while CallArgs uses Forward<> for the actual call. Keeping them separate incorrect evaluated requires() as
72 // false when Forward<T> produces an rvalue for types that are not move-constructible (e.g. TFunctionRef).
73#define ExpandArgPack(...) __VA_ARGS__
74#define PrependCommaIfNeeded(...) __VA_OPT__(,) __VA_ARGS__
75#define CallFunction(ConstFunction, Capability, ReturnValue, FunctionName, CheckArgs, CallArgs) \
76 if constexpr (SupportsCapability<Capability>()) \
77 { \
78 using ImplementationType = Private::ImplementationType<Capability, ImplementationTypes...>; \
79 constexpr bool bFunctionFound = requires() \
80 { \
81 { ImplementationType:: FunctionName (Environment PrependCommaIfNeeded(ExpandArgPack CheckArgs) ) } -> UE::CSameAs< ReturnValue >; \
82 }; \
83 if constexpr (!bIsConst || (bIsConst && ConstFunction)) \
84 { \
85 static_assert(bFunctionFound, "Function '" #FunctionName "' in capability '" #Capability "' is missing from the implementation. This is required by context implementations." ); \
86 PRAGMA_DISABLE_DEPRECATION_WARNINGS \
87 return ImplementationType:: FunctionName ( Environment PrependCommaIfNeeded(ExpandArgPack CallArgs) ); \
88 PRAGMA_ENABLE_DEPRECATION_WARNINGS \
89 } \
90 else \
91 { \
92 static_assert(!bFunctionFound, "Non-const function '" #FunctionName "' in capability '" #Capability "' is implemented for a const environment." ); \
93 checkf(false, TEXT("Function '" #FunctionName "' in capability '" #Capability "' cannot be called from a const environment.")); \
94 return UE::MakeDefault< ReturnValue >(); \
95 } \
96 } \
97 else \
98 { \
99 checkf(false, TEXT("Function '" #FunctionName "' in capability '" #Capability "' is not supported by the current query context implementation.")); \
100 return UE::MakeDefault< ReturnValue >(); \
101 }

195 template<bool bIsConst, typename EnvironmentType, typename... ImplementationTypes>
196 void TQueryContextImpl<bIsConst, EnvironmentType, ImplementationTypes...>::StaticReportCompatibility(uint64 CapabilityMask)
197 {
198 uint64 MissingCapabilitiesMask = CapabilityMask & ~GetCapabilityMask();
199 if (MissingCapabilitiesMask != 0)
200 {
201 TStringBuilder<256> Builder;
202
203 while (MissingCapabilitiesMask)
204 {
205 uint64 CapabilityIndex = FMath::CountTrailingZeros64(MissingCapabilitiesMask);
206 MissingCapabilitiesMask &= MissingCapabilitiesMask - 1;
207
208 Builder.Append(TEXT("Requested query context '"));
209 Builder.Append(IContextContract::GetCapabilityName(static_cast<int32>(CapabilityIndex)).ToString());
210 Builder.Append(TEXT("' is not supported by query context implementation.\n"));
211 }
212
213 Builder.Append(TEXT("The following types are supported by this query context: \n"));
214 for (const FName& CapabilityName : StaticGetSupportedCapabilities())
215 {
216 Builder.Append(TEXT(" "));
217 Builder.Append(CapabilityName.ToString());
218 Builder.AppendChar(TEXT('\n'));
219 }
220 checkf(false, TEXT("%s"), *Builder);
221 }
222 }
