229FORCEINLINE bool IsAsyncComputeSupported(EShaderPlatform ShaderPlatform)
230{
231 // Render pass merging and async compute are mutually exclusive since they are mutually dependent on each other (i.e. we can't
232 // merge render passes that involve async compute fences, since transitions would be relocated to the ends of the merged region).
233 return GRDGAsyncCompute > 0 && !IsImmediateMode() && !IsRenderPassMergeEnabled(ShaderPlatform) && GSupportsEfficientAsyncCompute && GRHISupportsSeparateDepthStencilCopyAccess;
234}
