108// query interface to tasks info
109class ITasksProvider : public IProvider
110{
111public:
112 typedef TFunctionRef<ETaskEnumerationResult(const FTaskInfo& Task)> TaskCallback;
113
114 virtual ~ITasksProvider() = default;
115
116 // returns a task for given thread and timestamp, if any, otherwise `nullptr`
117 virtual const FTaskInfo* TryGetTask(uint32 ThreadId, double Timestamp) const = 0;
118 // returns task info for given task ID
119 virtual const FTaskInfo* TryGetTask(TaskTrace::FId TaskId) const = 0;
120
121 // returns an info about waiting for tasks completion for given thread and timestamp, if any, otherwise `nullptr`
122 virtual const FWaitingForTasks* TryGetWaiting(const TCHAR* TimerName, uint32 ThreadId, double Timestamp) const = 0;
123
124 // returns an info about ParallelFor tasks for given timer
125 virtual TArray<TaskTrace::FId> TryGetParallelForTasks(const TCHAR* TimerName, uint32 ThreadId, double StartTime, double EndTime) const = 0;
126
127 // returns the number of tasks stored in the provider
128 virtual int64 GetNumTasks() const = 0;
129
130 // Calls the callback for each task stored in the provider with CreatedTimestamp <= EndTime and FinishedTimestamp >= StartTime
131 virtual void EnumerateTasks(double StartTime, double EndTime, ETaskEnumerationOption EnumerationOption, TaskCallback Callback) const = 0;
132};
