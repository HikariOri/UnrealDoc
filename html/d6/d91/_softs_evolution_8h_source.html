298 struct TArrayRange
299 {
300 static TArrayRange AddRange(TArray<ElementType>& InArray, int32 InRangeSize)
301 {
302 TArrayRange Range;
303 Range.Offset = InArray.Num();

311 {
312 return RangeSize == 0 || (Array && Offset >= 0 && Offset + RangeSize <= Array->Num());
