140 FSymbol* FindSymbol(const FSmallStringView& View)
141 {
142 if (!View.Begin)
143 {
144 return nullptr;
145 }
146
147 // Allow resolving to the first parent aggregate symbol scope
148 // Anything beyond that is ill-defined
149 bool bIsImmediateFirstAggregate = true;
150
151 // Try inner to outer
152 for (int32 i = SymbolStack.Num() - 1; i >= 0; i--)
153 {
154 FSymbolScope& Scope = SymbolStack[i];
155
156 if (Scope.Composite)
157 {
158 if (!bIsImmediateFirstAggregate)
159 {
160 // This is a private scope
161 continue;
162 }
163
164 // We are now in "this"
165 bIsImmediateFirstAggregate = false;
166 }
167
168 if (FSymbol** It = Scope.Symbols.Find(View))
169 {
170 return *It;
171 }
172 }
173
174 return nullptr;
175 }
