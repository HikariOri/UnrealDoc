294template<typename ElementType>
295class TListThreadSafe
296{ public:
297
299 TListThreadSafe():
300 FirstElement(NULL)
301 {}
302
307 void AddElement(TList<ElementType>* Element)
308 {
309 // Link the element at the beginning of the list.
310 TList<ElementType>* LocalFirstElement;
311 do
312 {
313 LocalFirstElement = FirstElement;
314 Element->Next = LocalFirstElement;
315 }
316 while(FPlatformAtomics::InterlockedCompareExchangePointer((void**)&FirstElement,Element,LocalFirstElement) != LocalFirstElement);
317 }
318
323 TList<ElementType>* ExtractAll()
324 {
325 // Atomically read the complete list and clear the shared head pointer.
326 TList<ElementType>* LocalFirstElement;
327 do
328 {
329 LocalFirstElement = FirstElement;
330 }
331 while(FPlatformAtomics::InterlockedCompareExchangePointer((void**)&FirstElement,NULL,LocalFirstElement) != LocalFirstElement);
332 return LocalFirstElement;
333 }
334
338 void Clear()
339 {
340 while (FirstElement)
341 {
342 // Atomically read the complete list and clear the shared head pointer.
343 TList<ElementType>* Element = ExtractAll();
344
345 // Delete all elements in the local list.
346 while (Element)
347 {
348 TList<ElementType>* NextElement = Element->Next;
349 delete Element;
350 Element = NextElement;
351 };
352 };
353 }
354
355private:
356
357 TList<ElementType>* FirstElement;
358};
