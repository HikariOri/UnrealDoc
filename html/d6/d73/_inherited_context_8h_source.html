60 struct FInheritedContextExtension
61 {
62 using FCaptureFunc = void (*)(void* DataDst, void* UserData);
63 using FApplyFunc = void (*)(const void* DataSrc, void* SaveDst, void* UserData);
64 using FRestoreFunc = void (*)(const void* SaveSrc, void* UserData);
65 using FDestroyFunc = void (*)(const void* DataSrc, void* UserData);

67 struct FInterface
68 {
69 FCaptureFunc Capture; // Read current thread state into DataDst
70 FApplyFunc Apply; // Write DataSrc to thread state, save previous to SaveDst
71 FRestoreFunc Restore; // Put back previous state from SaveSrc
72 FDestroyFunc Destroy; // Destruct data in buffer (nullable for trivial types)
73 uint32 DataSize;
74 uint32 DataAlign;
75 void* UserData; // Typically pointer to the variable being propagated
76 };

78 TSharedPtr<FInterface> Impl;
79
80 bool IsValid() const { return Impl.IsValid(); }
81
82 template<typename T>
83 T& GetState() const { return *static_cast<T*>(Impl->UserData); }

95 template<typename Func>
96 FInheritedContextExtension MakeInheritedContextExtension(Func GetVar)
97 {
98 using ReturnType = decltype(GetVar());
99 static_assert(std::is_lvalue_reference_v<ReturnType>, "Accessor must return an lvalue reference (e.g. []() -> bool& { return GMyFlag; })");
100 using T = std::remove_reference_t<ReturnType>;
101 using FuncPtr = T& (*)();
102
103 static_assert(sizeof(FuncPtr) <= sizeof(void*), "Function pointer must fit in void*");
104
105 FuncPtr FnPtr = GetVar;
106
107 FInheritedContextExtension Ext;
108 Ext.Impl = MakeShared<FInheritedContextExtension::FInterface>();
109 Ext.Impl->Capture = [](void* DataDst, void* UserData)
110 {
111 auto Fn = reinterpret_cast<T& (*)()>(UserData);
112 new (DataDst) T(Fn());
113 };
114 Ext.Impl->Apply = [](const void* DataSrc, void* SaveDst, void* UserData)
115 {
116 auto Fn = reinterpret_cast<T& (*)()>(UserData);
117 new (SaveDst) T(Fn());
118 Fn() = *static_cast<const T*>(DataSrc);
119 };
120 Ext.Impl->Restore = [](const void* SaveSrc, void* UserData)
121 {
122 auto Fn = reinterpret_cast<T& (*)()>(UserData);
123 Fn() = *static_cast<const T*>(SaveSrc);
124 };
125 Ext.Impl->Destroy = [](const void* DataSrc, void* /*UserData*/)
126 {
127 static_cast<const T*>(DataSrc)->~T();
128 };
129 Ext.Impl->DataSize = sizeof(T);
130 Ext.Impl->DataAlign = alignof(T);
131 Ext.Impl->UserData = reinterpret_cast<void*>(FnPtr);
132 return Ext;
133 }

207 template<auto ApplyFn, auto RestoreFn, typename T, typename... ArgsType>
208 FInheritedContextExtension MakeStatefulInheritedContextExtension(ArgsType&&... Args)
209 {
210 using SavedState = std::decay_t<decltype(ApplyFn())>;
211
212 // The capture block stores a T* and the saved state block stores SavedState.
213 // They are separate allocations but share the same DataSize/DataAlign layout,
214 // so size must accommodate the larger of the two.
215 constexpr uint32 Size = static_cast<uint32>(sizeof(T*) > sizeof(SavedState) ? sizeof(T*) : sizeof(SavedState));
216 constexpr uint32 Align = static_cast<uint32>(alignof(T*) > alignof(SavedState) ? alignof(T*) : alignof(SavedState));
217
218 using FStatefulImpl = TStatefulContextExtensionImpl<T>;
219 auto StatefulRef = MakeShared<FStatefulImpl>(Forward<ArgsType>(Args)...);
220
221 StatefulRef->Capture = [](void* DataDst, void* UserData)
222 {
223 // Store the shared state pointer so it propagates through the chain
224 *static_cast<T**>(DataDst) = static_cast<T*>(UserData);
225 };
226 StatefulRef->Apply = [](const void* /*DataSrc*/, void* SaveDst, void* /*UserData*/)
227 {
228 // Call ApplyFn to begin scope (e.g. record start time)
229 new (SaveDst) SavedState(ApplyFn());
230 };
231 StatefulRef->Restore = [](const void* SaveSrc, void* UserData)
232 {
233 // Call RestoreFn with saved state and shared state
234 RestoreFn(*static_cast<const SavedState*>(SaveSrc), *static_cast<T*>(UserData));
235 static_cast<const SavedState*>(SaveSrc)->~SavedState();
236 };
237 StatefulRef->Destroy = nullptr; // Captured data is T* (trivially destructible)
238 StatefulRef->DataSize = Size;
239 StatefulRef->DataAlign = Align;
240 StatefulRef->UserData = &StatefulRef->State;
241
242 FInheritedContextExtension Ext;
243 Ext.Impl = TSharedPtr<FInheritedContextExtension::FInterface>(StatefulRef);
244 return Ext;
245 }
