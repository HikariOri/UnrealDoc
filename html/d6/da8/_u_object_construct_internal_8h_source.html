139template<typename UEnumClass, typename EnumParams, typename PostNewFn>
140// It is UB to call `OuterFunc` below because it can return a `UClass*` not `UObject*` (even though
141// there is a parent-child relation there). So we suppress UBSan for this function.
142UE_DISABLE_UBSAN
143void ConstructUEnumHelper(UEnum*& OutEnum, const EnumParams& Params, PostNewFn&& InPostNewFn)
144{
145 FTypeConstructFunc* OuterFunc = Params.OuterFunc;
146
147 UObject* Outer = OuterFunc ? OuterFunc(ETypeConstructPhase::Outer) : nullptr;
148
149 if (OutEnum)
150 {
151 return;
152 }
153
154 UEnumClass* NewEnum = new (EC_InternalUseOnlyConstructor, Outer, UTF8_TO_TCHAR(Params.NameUTF8), Params.ObjectFlags) UEnumClass(FObjectInitializer());
155 OutEnum = NewEnum;
156 InPostNewFn(NewEnum, Params);
157
158 TArray<TPair<FName, int64>> EnumNames;
159 EnumNames.Reserve(Params.NumEnumerators);
160 for (const FEnumeratorParam* Enumerator = Params.EnumeratorParams, *EnumeratorEnd = Enumerator + Params.NumEnumerators; Enumerator != EnumeratorEnd; ++Enumerator)
161 {
162 EnumNames.Emplace(UTF8_TO_TCHAR(Enumerator->NameUTF8), Enumerator->Value);
163 }
164
165 const UEnum::EAddMaxKeyIfMissing AddMaxKeyIfMissing = UEnum::EAddMaxKeyIfMissing::Yes;
166 NewEnum->SetEnums(EnumNames, (UEnum::ECppForm)Params.CppForm, (UEnum::EUnderlyingType)Params.UnderlyingType, Params.EnumFlags, AddMaxKeyIfMissing);
167 NewEnum->CppType = UTF8_TO_TCHAR(Params.CppTypeUTF8);
168
169 if (Params.DisplayNameFunc)
170 {
171 NewEnum->SetEnumDisplayNameFn(Params.DisplayNameFunc);
172 }
173
174#if WITH_METADATA
175 AddMetaData(NewEnum, Params.MetaDataArray, Params.NumMetaData);
176#endif
177}

180// It is UB to call `OuterFunc` below because it can return a `UClass*` not `UObject*` (even though
181// there is a parent-child relation there). So we suppress UBSan for this function.

209template<typename UClassClass, typename ClassParams, typename PostNewFn>
210// It is UB to call `SingletonFunc` below because it can return a `UClass*` not `UObject*` (even though
211// there is a parent-child relation there). So we suppress UBSan for this function.
212UE_DISABLE_UBSAN
213void ConstructUClassHelper(UClass*& OutClass, const ClassParams& Params, PostNewFn&& InPostNewFn)
214{
215 if (OutClass && (OutClass->ClassFlags & CLASS_Constructed))
216 {
217 return;
218 }
219
220 for (FTypeConstructFunc* const* SingletonFunc = Params.DependencySingletonFuncArray, *const* SingletonFuncEnd = SingletonFunc + Params.NumDependencySingletons; SingletonFunc != SingletonFuncEnd; ++SingletonFunc)
221 {
222 (*SingletonFunc)(ETypeConstructPhase::Outer);
223 }
224
225 UClass* NewClass = Params.ClassNoRegisterFunc(ETypeConstructPhase::Inner);
226 OutClass = NewClass;
227
228 if (NewClass->ClassFlags & CLASS_Constructed)
229 {
230 return;
231 }
232
233 InPostNewFn(Cast<UClassClass>(NewClass), Params);
234
235 UObjectForceRegistration(NewClass);
236
237 UClass* SuperClass = NewClass->GetSuperClass();
238 if (SuperClass)
239 {
240 NewClass->ClassFlags |= (SuperClass->ClassFlags & CLASS_Inherit);
241 }
242
243 NewClass->ClassFlags |= (EClassFlags)(Params.ClassFlags | CLASS_Constructed);
244 // Make sure the reference token stream is empty since it will be reconstructed later on
245 // This should not apply to intrinsic classes since they emit native references before AssembleReferenceTokenStream is called.
246 if ((NewClass->ClassFlags & CLASS_Intrinsic) != CLASS_Intrinsic)
247 {
248 check((NewClass->ClassFlags & CLASS_TokenStreamAssembled) != CLASS_TokenStreamAssembled);
249 NewClass->ReferenceSchema.Reset();
250 }
251 NewClass->CreateLinkAndAddChildFunctionsToMap(Params.FunctionLinkArray, Params.NumFunctions);
252
253 ConstructFProperties(NewClass, Params.PropertyArray, Params.NumProperties);
254
255 UE::CoreUObject::Private::ConstructUClassPartials(*NewClass);
256
257 if (Params.ClassConfigNameUTF8)
258 {
259 NewClass->ClassConfigName = FName(UTF8_TO_TCHAR(Params.ClassConfigNameUTF8));
260 }
261
262 NewClass->SetCppTypeInfoStatic(Params.CppClassInfo);
263
264 if (int32 NumImplementedInterfaces = Params.NumImplementedInterfaces)
265 {
266 NewClass->Interfaces.Reserve(NumImplementedInterfaces);
267 for (const FImplementedInterfaceParams* ImplementedInterface = Params.ImplementedInterfaceArray, *ImplementedInterfaceEnd = ImplementedInterface + NumImplementedInterfaces; ImplementedInterface != ImplementedInterfaceEnd; ++ImplementedInterface)
268 {
269 UClass* (*ClassFunc)(ETypeConstructPhase) = ImplementedInterface->ClassFunc;
270 UClass* InterfaceClass = ClassFunc ? ClassFunc(ETypeConstructPhase::Outer) : nullptr;
271
272 NewClass->Interfaces.Emplace(InterfaceClass, ImplementedInterface->Offset, ImplementedInterface->bImplementedByK2);
273 }
274 }
275
276#if WITH_METADATA
277 AddMetaData(NewClass, Params.MetaDataArray, Params.NumMetaData);
278#endif
279
280 NewClass->StaticLink();
281
282 NewClass->SetSparseClassDataStruct(NewClass->GetSparseClassDataArchetypeStruct());
283}

285template<typename UFunctionClass, typename FunctionParams, typename NewFn>
286// It is UB to call `OuterFunc` below because it can return a `UClass*` not `UObject*` (even though
287// there is a parent-child relation there). So we suppress UBSan for this function.
288UE_DISABLE_UBSAN
289void ConstructUFunctionHelper(UFunction*& InOutFunction, const FunctionParams& Params, UFunction** InSingletonPtr, NewFn&& InNewFn)
290{
291 FTypeConstructFunc* OuterFunc = Params.OuterFunc;
292 UFunction* (*SuperFunc)(ETypeConstructPhase) = Params.SuperFunc;
293
294 UObject* Outer = OuterFunc ? OuterFunc(ETypeConstructPhase::Outer) : nullptr;
295 UFunction* Super = SuperFunc ? SuperFunc(ETypeConstructPhase::Outer) : nullptr;
296
297 if (InOutFunction)
298 {
299 return;
300 }
301
302 FName FuncName(UTF8_TO_TCHAR(Params.NameUTF8));
303
304#if WITH_LIVE_CODING
305 // When a package is patched, it might reference a function in a class. When this happens, the existing UFunction
306 // object gets reused but the UField's Next pointer gets nulled out. This ends up terminating the function list
307 // for the class. To work around this issue, cache the next pointer and then restore it after the new instance
308 // is created. Only do this if we reuse the current instance.
309 UField* PrevFunctionNextField = nullptr;
310 UFunction* PrevFunction = nullptr;
311 if (UObject* PrevObject = StaticFindObjectFastInternal( /*Class=*/ nullptr, Outer, FuncName, EFindObjectFlags::ExactClass))
312 {
313 PrevFunction = Cast<UFunction>(PrevObject);
314 if (PrevFunction != nullptr)
315 {
316 PrevFunctionNextField = PrevFunction->Next;
317 }
318 }
319#endif
320
321 UFunction* NewFunction = InNewFn(Outer, Super, FuncName, Params);
322 check(NewFunction);
323 InOutFunction = NewFunction;
324
325#if WITH_LIVE_CODING
326 NewFunction->SingletonPtr = InSingletonPtr;
327 if (NewFunction == PrevFunction)
328 {
329 NewFunction->Next = PrevFunctionNextField;
330 }
331#endif
332
333#if WITH_METADATA
334 AddMetaData(NewFunction, Params.MetaDataArray, Params.NumMetaData);
335#endif
336 NewFunction->RPCId = Params.RPCId;
337 NewFunction->RPCResponseId = Params.RPCResponseId;
338
339 ConstructFProperties(NewFunction, Params.PropertyArray, Params.NumProperties);
340
341 NewFunction->Bind();
342 NewFunction->StaticLink();
343}
