84template <CArgTypes Type>
85Type FContext::Get(FStringView Name) const
86{
87 if (Name.StartsWith(TEXT('-')))
88 {
89 return Get(Name, Type{});
90 }
91
92 uint32 Index = FindArgument(Name);
93 if (IsSet(Index))
94 {
95 return *(Type*)(Values[Index].Buffer);
96 }
97
98 OnMissingValue(Name);
99}
