47 const GSConstraintHandle* AddSingleParticleTetrahedraConstraint(const GeometryCollection::Facades::FTetrahedralFacade& InTargetGeom, const FSolverParticles& AllParticles, const GeometryCollection::Facades::TetrahedralParticleEmbedding& InIntersection, const int32 InConstraintIndex, const Chaos::FRange& SourceRange, const Chaos::FRange& TargetRange, const FSolverReal PositionTargetStiffness)
48 {
49 GSConstraintHandle* NewConstraintPtr = new GSConstraintHandle();
50 NewConstraintPtr -> ConstraintIndex = InConstraintIndex;
51 Handles[InConstraintIndex] = NewConstraintPtr;
52
53 const FIntVector4 TargetTet = InTargetGeom.Tetrahedron[InTargetGeom.TetrahedronStart[InIntersection.GeometryIndex] + InIntersection.TetrahedronIndex] + FIntVector4(TargetRange.Start);
54
55 FGaussSeidelSpringConstraintSingleData SingleConstraintData;
56 SingleConstraintData.SingleIndices = { InIntersection.ParticleIndex + SourceRange.Start };
57 SingleConstraintData.SingleSecondIndices = {TargetTet[0], TargetTet[1], TargetTet[2], TargetTet[3]};
58 SingleConstraintData.SingleWeights = {(FSolverReal)1.};
59 SingleConstraintData.SingleSecondWeights = {InIntersection.BarycentricWeights[0], InIntersection.BarycentricWeights[1], InIntersection.BarycentricWeights[2], InIntersection.BarycentricWeights[3]};
60 SingleConstraintData.bIsAnisotropic = false;
61 SingleConstraintData.SingleNormal = TVec3<FSolverReal>((FSolverReal)0.);
62 float ConstraintStiffness = 0.f;
63 for (int32 k = 0; k < 4; k++)
64 {
65 ConstraintStiffness += SingleConstraintData.SingleSecondWeights[k] * PositionTargetStiffness * AllParticles.M(TargetTet[k]);
66 }
67 ConstraintStiffness += PositionTargetStiffness * AllParticles.M(SingleConstraintData.SingleIndices[0]);
68 SingleConstraintData.SingleStiffness = (FSolverReal)ConstraintStiffness;
69
70 ConstraintsData.SetSingleConstraint(SingleConstraintData, InConstraintIndex);
71
72 return NewConstraintPtr;

144 void ReComputeNodalWeights(const TSet<int32>& DirtyVerts)
145 {
146 for (const int32 Vert : DirtyVerts)
147 {
148 check(Vert < NodalWeights.Num());
149 NodalWeights[Vert].Init(FSolverReal(0), 6);
150 for (int32 j = 0; j < WCIncidentElements[Vert].Num(); j++)
151 {
152 const int32 ConstraintIndex = WCIncidentElements[Vert][j];
153 const int32 LocalIndex = WCIncidentElementsLocal[Vert][j];
154
155 FSolverReal Weight = FSolverReal(0);
156 if (LocalIndex >= ConstraintsData.GetIndices(ConstraintIndex).Num())
157 {
158 Weight = ConstraintsData.GetSecondWeights(ConstraintIndex)[LocalIndex - ConstraintsData.GetIndices(ConstraintIndex).Num()];
159 }
160 else
161 {
162 Weight = ConstraintsData.GetWeights(ConstraintIndex)[LocalIndex];
163 }
164
165 if (ConstraintsData.GetIsAnisotropic(ConstraintIndex))
166 {
167 for (int32 alpha = 0; alpha < 3; alpha++)
168 {
169 NodalWeights[Vert][alpha] += ConstraintsData.GetNormal(ConstraintIndex)[alpha] * ConstraintsData.GetNormal(ConstraintIndex)[alpha] * Weight * Weight * ConstraintsData.GetStiffness(ConstraintIndex);
170 }
171
172 NodalWeights[Vert][3] += ConstraintsData.GetNormal(ConstraintIndex)[0] * ConstraintsData.GetNormal(ConstraintIndex)[1] * Weight * Weight * ConstraintsData.GetStiffness(ConstraintIndex);
173 NodalWeights[Vert][4] += ConstraintsData.GetNormal(ConstraintIndex)[0] * ConstraintsData.GetNormal(ConstraintIndex)[2] * Weight * Weight * ConstraintsData.GetStiffness(ConstraintIndex);
174 NodalWeights[Vert][5] += ConstraintsData.GetNormal(ConstraintIndex)[1] * ConstraintsData.GetNormal(ConstraintIndex)[2] * Weight * Weight * ConstraintsData.GetStiffness(ConstraintIndex);
175 }
176 else
177 {
178 for (int32 alpha = 0; alpha < 3; alpha++)
179 {
180 NodalWeights[Vert][alpha] += Weight * Weight * ConstraintsData.GetStiffness(ConstraintIndex);
181 }
182 }
183 }
184 }

187 void AdjustStiffness(const TArray<const GSConstraintHandle*> ConstraintHandles, const TArray<FSolverReal>& StiffnessPerConstraint, const FSolverParticles& Particles)
188 {
189 TSet<int32> DirtyVertices;
190 for (int32 i = 0; i < ConstraintHandles.Num(); i++)
191 {
192 const GSConstraintHandle* SingleHandle = ConstraintHandles[i];
193 const FSolverReal NewStiffness = StiffnessPerConstraint[i];
194 const int32 ConstraintIndex = SingleHandle->ConstraintIndex;
195 const FGaussSeidelSpringConstraintSingleData& SingleConstraintData = ConstraintsData.GetSingleConstraintData(ConstraintIndex);
196 FSolverReal NewConstraintStiffness = (FSolverReal)0.;
197 for (int32 k = 0; k < SingleConstraintData.SingleWeights.Num(); k++)
198 {
199 NewConstraintStiffness += SingleConstraintData.SingleWeights[k] * NewStiffness * Particles.M(SingleConstraintData.SingleIndices[k]);
200 }
201 for (int32 k = 0; k < SingleConstraintData.SingleSecondWeights.Num(); k++)
202 {
203 NewConstraintStiffness += SingleConstraintData.SingleSecondWeights[k] * NewStiffness * Particles.M(SingleConstraintData.SingleSecondIndices[k]);
204 }
205 ConstraintsData.SetStiffness(ConstraintIndex, NewConstraintStiffness);
206 DirtyVertices.Append(SingleConstraintData.SingleIndices);
207 DirtyVertices.Append(SingleConstraintData.SingleSecondIndices);
208 }
209 ReComputeNodalWeights(DirtyVertices);
210 }
