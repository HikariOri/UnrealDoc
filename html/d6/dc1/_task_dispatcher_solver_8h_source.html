68 template<typename LambdaRigid, typename LambdaGCAndCluster>
69 int32 CollectPhysicsResults(LambdaRigid CollectRigidResults, LambdaGCAndCluster CollectGCAndClustersResults, int32 NumDirty, int32 DispatchBatchIndex)
70 {
71 constexpr int32 MinRigidPerTask = 50;
72 const int32 NumTasks = FMath::Max(FMath::Min(FTaskGraphInterface::Get().GetNumWorkerThreads() - 1, Chaos::MaxNumWorkers), 1);
73 const int32 NumDirtyTask = FMath::Max(FMath::Min(NumTasks, NumDirty), 1);
74 const int32 DirtyByTask = FMath::Max(FMath::DivideAndRoundUp(NumDirty, NumDirtyTask), MinRigidPerTask);
75 const int32 LocalNumRigidBatches = FMath::DivideAndRoundUp(NumDirty, DirtyByTask);
76
77 // We can have at most NumDirty overall active particles across all dirty proxy lists
78 // and in each list at most NumProxies. So we take the min, otherwise for example in a world
79 // with a few million static particles and one dynamic we'll allocate enough storage for the
80 // millions of particles despite only ever syncing one
81 for (int32 BatchIndex = 0; BatchIndex < LocalNumRigidBatches; BatchIndex++)
82 {
83 const int32 StartIndex = BatchIndex * DirtyByTask;
84 int32 EndIndex = (BatchIndex + 1) * DirtyByTask;
85 EndIndex = FMath::Min(NumDirty, EndIndex);
86
87 TArray<FSingleParticlePhysicsProxy*>& ActiveRigid = ActiveRigidArray[DispatchBatchIndex + BatchIndex];
88 ActiveRigid.Reset();
89
90 // Create Task here
91 UE::Tasks::FTask DirtyPendingTask = UE::Tasks::Launch(UE_SOURCE_LOCATION, [StartIndex, EndIndex, &ActiveRigid, CollectRigidResults]()
92 {
93 QUICK_SCOPE_CYCLE_COUNTER(STAT_CollectRigidsTasks);
94 CollectRigidResults(StartIndex, EndIndex, ActiveRigid);
95 });
96 DirtyRigidPendingTasks.Add(DirtyPendingTask);
97 }
98
99 // The GC and cluster proxies are shared among several particles
100 // all those following task will be computed sequentially to be sure not to add twice the same proxy.
101 // A fill all then merge approach could be potentially more efficient (not sure)
102 UE::Tasks::FTask DirtyPendingTask = UE::Tasks::Launch(UE_SOURCE_LOCATION, [this, NumDirty, CollectGCAndClustersResults]()
103 {
104 QUICK_SCOPE_CYCLE_COUNTER(STAT_CollectGCAndClustersTasks);
105 CollectGCAndClustersResults(0, NumDirty, ActiveGC, ActiveClusterUnions);
106 }, DirtyGCAndClusterPendingTask);
107 DirtyGCAndClusterPendingTask = DirtyPendingTask;
108
109 return LocalNumRigidBatches;
110 }
