38template<typename T>
39class TSplineInterpolationPolicy
40{
41public:
42 // Keep existing parameter-based interpolation (used by attribute channels)
43 static T Interpolate(TArrayView<const T* const> Window, float Parameter)
44 {
45 if (Window.Num() < 2)
46 {
47 return Window.Num() == 1 ? *Window[0] : T();
48 }
49 if constexpr (TIsMathInterpolable<T>::value)
50 {
51 return FMath::Lerp(*Window[0], *Window[1], Parameter);
52 }
53 else
54 {
55 // Fallback for non-math types:
56 return Parameter < 0.5f ? *Window[0] : *Window[1];
57 }
58 }
59
60 static T InterpolateCubic(TArrayView<const T* const> Window, float Parameter)
61 {
62 if (Window.Num() < 4)
63 {
64 // Fallback to linear interpolation if not enough points
65 return Interpolate(Window, Parameter);
66 }
67 if constexpr (TIsMathInterpolable<T>::value)
68 {
69 // Calculate Bernstein basis functions
70 const float t = Parameter;
71 const float OneMinusT = 1.0f - t;
72 const float t2 = t * t;
73 const float OneMinusT2 = OneMinusT * OneMinusT;
74
75 // Cubic Bernstein polynomials
76 const float Basis[4] =
77 {
78 OneMinusT * OneMinusT2, // (1-t)³
79 3.0f * t * OneMinusT2, // 3t(1-t)²
80 3.0f * t2 * OneMinusT, // 3t²(1-t)
81 t * t2 // t³
82 };
83
84 return TSplineInterpolationPolicy<T>::InterpolateWithBasis(Window, Basis);
85 }
86 else
87 {
88 // Fallback for non-math types:
89 return Parameter < 0.5f ? *Window[1] : *Window[2];
90 }
91 }
92
93 // Add window-based interpolation for spline implementations
94 static T InterpolateWithBasis(TArrayView<const T* const> Window, TArrayView<const float> Basis)
95 {
96 if constexpr (TIsMathInterpolable<T>::value)
97 {
98 T Result = T(); // Assumes T() gives a neutral value (e.g. zero vector)
99 int32 Count = FMath::Min(Window.Num(), Basis.Num());
100 for (int32 i = 0; i < Count; ++i)
101 {
102 // Assumes T supports addition and multiplication by float.
103 Result = Result + (*Window[i] * Basis[i]);
104 }
105 return Result;
106 }
107 else
108 {
109 // Fallback: simply return the first element if available.
110 return Window.Num() > 0 ? *Window[0] : T();
111 }
112 }
113
114 // Template-based n-th derivative calculation
115 template<int32 Order>
116 static T EvaluateDerivative(TArrayView<const T* const> Window, float Parameter)
117 {
118 if (Order == 0)
119 {
120 return Interpolate(Window, Parameter);
121 }
122 if constexpr (TIsMathInterpolable<T>::value)
123 {
124 // For math types, use the generic derivative helper.
125 return Math::TGenericDerivativeHelper<T, Order>::Compute(Window, Parameter);
126 }
127 else
128 {
129 // For non-math types, derivatives are not meaningful.
130 // Return a default constructed value.
131 return T();
132 }
133 }
134};

137template<>
138class TSplineInterpolationPolicy<int32>
139{
140public:
141 static int32 Interpolate(TArrayView<const int32* const> Window, float Parameter)
142 {
143 if (Window.Num() < 2)
144 {
145 return Window.Num() == 1 ? *Window[0] : 0;
146 }
147
148 // Handle boundary cases
149 if (Parameter <= 0.0f) return *Window[0];
150 if (Parameter >= 1.0f) return *Window[1];
151
152 // For intermediate values, explicitly convert to avoid implicit conversion warnings
153 const float Start = static_cast<float>(*Window[0]);
154 const float End = static_cast<float>(*Window[1]);
155 const float Result = Start + Parameter * (End - Start);
156
157 // Round to nearest integer
158 return FMath::RoundToInt(Result);
159 }
160
161 static int32 InterpolateWithBasis(TArrayView<const int32* const> Window, TArrayView<const float> Basis)
162 {
163 float Result = 0.0f;
164 for (int32 i = 0; i < FMath::Min(Window.Num(), Basis.Num()); ++i)
165 {
166 Result += static_cast<float>(*Window[i]) * Basis[i];
167 }
168 return FMath::RoundToInt(Result);
169 }
170
171 // Template-based n-th derivative calculation
172 template<int32 Order>
173 static int32 EvaluateDerivative(TArrayView<const int32* const> Window, float Parameter)
174 {
175 if (Order == 0)
176 {
177 return Interpolate(Window, Parameter);
178 }
179 return Math::TGenericDerivativeHelper<int32, Order>::Compute(Window, Parameter);
180 }

233template <typename RealType>
234class TSplineInterpolationPolicy<UE::Math::TQuat<RealType>>
235{
236 using TQuat = UE::Math::TQuat<RealType>;
237public:
238 // Parameter-based interpolation using Slerp
239 static TQuat Interpolate(TArrayView<const TQuat* const> Window, float Parameter)
240 {
241 if (Window.Num() < 2)
242 {
243 return Window.Num() == 1 ? *Window[0] : TQuat::Identity;
244 }
245 return TQuat::Slerp(*Window[0], *Window[1], static_cast<RealType>(Parameter));
246 }
247
259 static TQuat InterpolateCubic(TArrayView<const TQuat* const> Window, float Alpha)
260 {
261 if (Window.Num() < 4)
262 {
263 // Fallback if not enough points
264 return Interpolate(Window, Alpha);
265 }
266
267 return TQuat::Squad(
268 *Window[0], // P0 - start point
269 *Window[1], // C0 - out control
270 *Window[3], // P1 - end point
271 *Window[2], // C1 - in control
272 Alpha);
273 }

292 static TQuat InterpolateWithBasis(TArrayView<const TQuat* const> Window, TArrayView<const float> Basis)
293 {
294 const int32 N = FMath::Min(Window.Num(), Basis.Num());
295 if (N <= 0)
296 {
297 return TQuat::Identity;
298 }
299 if (N == 1)
300 {
301 TQuat q = *Window[0];
302 q.Normalize();
303 return q;
304 }
305 // Optimization: detect linear case (2 points with [1-t, t] weights)
306 if (N == 2)
307 {
308 const RealType W0 = static_cast<RealType>(Basis[0]);
309 const RealType W1 = static_cast<RealType>(Basis[1]);
310 const RealType Sum = W0 + W1;
311
312 if (FMath::IsNearlyEqual(Sum, static_cast<RealType>(1), static_cast<RealType>(1e-4)) &&
313 W0 >= static_cast<RealType>(0) && W1 >= static_cast<RealType>(0))
314 {
315 // Linear blend - use efficient Slerp
316 TQuat Q0 = (*Window[0]); Q0.Normalize();
317 TQuat Q1 = (*Window[1]); Q1.Normalize();
318
319 if ((Q0 | Q1) < static_cast<RealType>(0))
320 {
321 Q1 = Q1 * static_cast<RealType>(-1);
322 }
323
324 return TQuat::Slerp(Q0, Q1, W1 / Sum);
325 }
326 }
327 constexpr RealType DotAntipodalCut = static_cast<RealType>(-0.95); // treat dot ≤ -0.95 as "nearly antipodal"
328 constexpr RealType WeightBalanceTol = static_cast<RealType>(0.05); // within 5% of each other -> "balanced"
329
330 // ---------------------------------------------------------------------------
331 // Antipodal pair detection: if two high-weight samples are nearly opposite
332 // and their weights are nearly equal, we want cancellation about Identity.
333 // ---------------------------------------------------------------------------
334 bool bHasAntipodalBalancedPair = false;
335 for (int32 i = 0; i < N && !bHasAntipodalBalancedPair; ++i)
336 {
337 for (int32 j = i + 1; j < N; ++j)
338 {
339 const RealType wi = static_cast<RealType>(Basis[i]);
340 const RealType wj = static_cast<RealType>(Basis[j]);
341 if (wi >= static_cast<RealType>(0) && wj >= static_cast<RealType>(0)) // convex-only check
342 {
343 const RealType sum = wi + wj;
344 if (sum > static_cast<RealType>(UE_KINDA_SMALL_NUMBER))
345 {
346 const RealType balance = FMath::Abs(wi - wj) / sum;
347 TQuat qi = *Window[i]; qi.Normalize();
348 TQuat qj = *Window[j]; qj.Normalize();
349 const RealType dot = (qi | qj);
350
351 if (dot <= DotAntipodalCut && balance <= WeightBalanceTol)
352 {
353 bHasAntipodalBalancedPair = true;
354 break;
355 }
356 }
357 }
358 }
359 }
360
361 // ---------------------------------------------------------------------------
362 // Choose reference quaternion
363 // - If antipodal balanced pair detected: use Identity so the pair cancels.
364 // - Else: use the max-|weight| sample (your current strategy).
365 // ---------------------------------------------------------------------------
366 TQuat Qref;
367 int32 RefIdx = 0;
368
369 if (bHasAntipodalBalancedPair)
370 {
371 Qref = TQuat::Identity;
372 }
373 else
374 {
375 RealType MaxAbsW = static_cast<RealType>(0);
376 for (int32 i = 0; i < N; ++i)
377 {
378 const RealType aw = FMath::Abs(static_cast<RealType>(Basis[i]));
379 if (aw > MaxAbsW)
380 {
381 MaxAbsW = aw;
382 RefIdx = i;
383 }
384 }
385 Qref = *Window[RefIdx];
386 Qref.Normalize();
387
388 if (MaxAbsW <= static_cast<RealType>(UE_KINDA_SMALL_NUMBER))
389 {
390 return Qref; // nothing meaningful to average
391 }
392 }
393
394 const TQuat QrefInv = Qref.Inverse();
395
396 // Accumulate weighted logarithms in tangent space
397 RealType SumX = 0, SumY = 0, SumZ = 0;
398 RealType WeightSum = 0; // signed sum for nonnegative bases
399 RealType WeightAbsSum = 0; // backup for mixed-sign cases
400 bool bAllNonNegative = true;
401
402 for (int32 i = 0; i < N; ++i)
403 {
404 const RealType Weight = static_cast<RealType>(Basis[i]);
405 if (FMath::IsNearlyZero(Weight))
406 {
407 continue;
408 }
409 bAllNonNegative &= (Weight >= static_cast<RealType>(0));
410
411 TQuat Qi = (*Window[i]);
412 Qi.Normalize();
413
414 // Hemisphere alignment relative to reference
415 if ((Qref | Qi) < static_cast<RealType>(0))
416 {
417 Qi = Qi * static_cast<RealType>(-1);
418 }
419
420 // Relative rotation: Qrel = Qref^{-1} * Qi
421 const TQuat Qrel = QrefInv * Qi;
422 const TQuat V = Qrel.Log(); // tangent vector (pure imaginary)
423
424 SumX += Weight * V.X;
425 SumY += Weight * V.Y;
426 SumZ += Weight * V.Z;
427 WeightSum += Weight;
428 WeightAbsSum += FMath::Abs(Weight);
429 }
430
431 // If all weights are nonnegative, use their (signed) sum so cancellations happen naturally.
432 // If mixed signs, fall back to the absolute sum to avoid division by ~0 and keep continuity.
433 const RealType Norm = bAllNonNegative ? WeightSum : WeightAbsSum;
434 if (Norm <= static_cast<RealType>(UE_KINDA_SMALL_NUMBER))
435 {
436 return Qref;
437 }
438
439 // Map back to manifold: Qref * Exp(avg tangent)
440 const TQuat D = TQuat(SumX / Norm, SumY / Norm, SumZ / Norm, static_cast<RealType>(0)).Exp();
441 TQuat Result = Qref * D;
442 Result.Normalize();
443 return Result;
444 }

448 // Template-based n-th derivative calculation
449 template<int32 Order>
450 static TQuat EvaluateDerivative(TArrayView<const TQuat* const> Window, float Parameter)
451 {
452 if (Order == 0)
453 {
454 return Interpolate(Window, Parameter);
455 }
456
457 // First derivative - angular velocity
458 if (Order == 1)
459 {
460 // Adaptive step size based on quaternion difference
461 RealType base_h = static_cast<RealType>(0.01);
462 TQuat Q0 = Interpolate(Window, Parameter);
463 TQuat Q1base = Interpolate(Window, Parameter + static_cast<float>(base_h));
464 RealType diff = Q0.AngularDistance(Q1base);
465 RealType h = base_h * FMath::Clamp(
466 diff,
467 static_cast<RealType>(0.001),
468 static_cast<RealType>(0.1)
469 );
470
471 TQuat Q1 = Interpolate(Window, Parameter + static_cast<float>(h));
472 return (Q1 * Q0.Inverse()).Log() * (static_cast<RealType>(1) / h);
473 }
474
475 // Second derivative - angular acceleration
476 if (Order == 2)
477 {
478 RealType base_h = static_cast<RealType>(0.01);
479 TQuat Q_m1 = Interpolate(Window, Parameter - static_cast<float>(base_h));
480 TQuat Q_p1 = Interpolate(Window, Parameter + static_cast<float>(base_h));
481 RealType diff = Q_m1.AngularDistance(Q_p1);
482 RealType h = base_h * FMath::Clamp(
483 diff / static_cast<RealType>(2),
484 static_cast<RealType>(0.001),
485 static_cast<RealType>(0.1)
486 );
487
488 TQuat V0 = EvaluateDerivative<1>(Window, Parameter - static_cast<float>(h));
489 TQuat V1 = EvaluateDerivative<1>(Window, Parameter + static_cast<float>(h));
490 return (V1 * V0.Inverse()).Log() * (static_cast<RealType>(1) / (static_cast<RealType>(2) * h));
491 }
492
493 // Higher orders generally not meaningful for rotations
494 return TQuat::Identity;
495 }

498// FTransform interpolation policy
499template<>
500class TSplineInterpolationPolicy<FTransform>
501{
502public:
503 static FTransform Interpolate(TArrayView<const FTransform* const> Window, float Parameter)
504 {
505 if (Window.Num() < 2)
506 {
507 return Window.Num() == 1 ? *Window[0] : FTransform::Identity;
508 }
509
510 const FTransform& A = *Window[0];
511 const FTransform& B = *Window[1];
512
513 FTransform Result;
514 Result.SetLocation(FMath::Lerp(A.GetLocation(), B.GetLocation(), Parameter));
515 Result.SetRotation(FQuat::Slerp(A.GetRotation(), B.GetRotation(), Parameter));
516 Result.SetScale3D(FMath::Lerp(A.GetScale3D(), B.GetScale3D(), Parameter));
517
518 return Result;
519 }
520
521 static FTransform InterpolateWithBasis(TArrayView<const FTransform* const> Window, TArrayView<const float> Basis)
522 {
523 // Handle empty case
524 if (Window.Num() == 0 || Basis.Num() == 0)
525 {
526 return FTransform::Identity;
527 }
528
529 // Interpolate components separately
530 FVector Location = FVector::ZeroVector;
531 FQuat Rotation = FQuat::Identity;
532 FVector Scale = FVector(1.0f);
533
534 float TotalRotWeight = 0.0f;
535
536 // Weighted sum of locations
537 for (int32 i = 0; i < FMath::Min(Window.Num(), Basis.Num()); ++i)
538 {
539 const float Weight = Basis[i];
540 if (Weight > UE_SMALL_NUMBER)
541 {
542 Location += Window[i]->GetLocation() * Weight;
543 Scale += Window[i]->GetScale3D() * Weight;
544
545 // Handle rotation separately with proper quaternion blending
546 const FQuat& Q = Window[i]->GetRotation();
547 const double Dot = Rotation.W * Q.W + Rotation.X * Q.X + Rotation.Y * Q.Y + Rotation.Z * Q.Z;
548 Rotation += (Dot >= 0.0f ? Q : -Q) * Weight;
549 TotalRotWeight += Weight;
550 }
551 }
552
553 // Normalize the quaternion if we accumulated any weights
554 if (TotalRotWeight > UE_SMALL_NUMBER)
555 {
556 Rotation.Normalize();
557 }
558
559 return FTransform(Rotation, Location, Scale);
560 }
561
562 template<int32 Order>
563 static FTransform EvaluateDerivative(TArrayView<const FTransform* const> Window, float Parameter)
564 {
565 if (Order == 0)
566 {
567 return Interpolate(Window, Parameter);
568 }
569
570 // Split into components
571 TArray<FVector> Translations;
572 TArray<FQuat> Rotations;
573 TArray<FVector> Scales;
574
575 Translations.SetNum(Window.Num());
576 Rotations.SetNum(Window.Num());
577 Scales.SetNum(Window.Num());
578
579 for (int32 i = 0; i < Window.Num(); ++i)
580 {
581 Translations[i] = Window[i]->GetTranslation();
582 Rotations[i] = Window[i]->GetRotation();
583 Scales[i] = Window[i]->GetScale3D();
584 }
585
586 // Calculate derivatives for each component
587 TArrayView<const FVector> TransView(Translations.GetData(), Translations.Num());
588 TArrayView<const FQuat> RotView(Rotations.GetData(), Rotations.Num());
589 TArrayView<const FVector> ScaleView(Scales.GetData(), Scales.Num());
590
591 FVector TransResult = TSplineInterpolationPolicy<FVector>::EvaluateDerivative<Order>(TransView, Parameter);
592 FQuat RotResult = TSplineInterpolationPolicy<FQuat>::EvaluateDerivative<Order>(RotView, Parameter);
593 FVector ScaleResult = TSplineInterpolationPolicy<FVector>::EvaluateDerivative<Order>(ScaleView, Parameter);
594
595 return FTransform(RotResult, TransResult, ScaleResult);
596 }
