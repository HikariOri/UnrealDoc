76static bool SupportsCurveTiledRasterization(EShaderPlatform InShaderPlatform)
77{
78 return NaniteCurvesSupported()
79 && FDataDrivenShaderPlatformInfo::GetSupportsUInt64ImageAtomics(InShaderPlatform)
80 // For now, disable support for certain platforms, as the encoding & storing routines used by the curve tiled rasterization caused shader compilation issue.
81 && !FDataDrivenShaderPlatformInfo::GetIsLanguageNintendo(InShaderPlatform)
82 && !IsVulkanPlatform(InShaderPlatform)
83 && !IsMetalPlatform(InShaderPlatform);
84}

92static void CreateCurveBinningResources(FRDGBuilder& GraphBuilder, const FIntPoint& ViewResolution, uint32 InFrameIndex, bool bIsShadowView, Nanite::FCurveResources& Out)
93{
94 const uint32 TileWidth = GetNaniteCurveTileWidth();
95 const uint32 TileCapacity = GetNaniteCurveTileCapacity();
96 const FIntPoint TileCountXY = FMath::DivideAndRoundUp(ViewResolution, FIntPoint(TileWidth));
97
98 // per-frame constants
99 Out.FrameIndex = InFrameIndex;
100
101 // per-view constants
102 Out.NumTilesX = TileCountXY.X;
103 Out.NumTilesY = TileCountXY.Y;
104 Out.NumTilesA = TileCountXY.X * TileCountXY.Y;
105 Out.ViewSizeX = ViewResolution.X;
106 Out.ViewSizeY = ViewResolution.Y;
107
108 // per-view transient buffers
109 const uint32 MaxShadowViewSegmentCount = TileCapacity * FMath::DivideAndRoundUp(4096u, TileWidth) * FMath::DivideAndRoundUp(2048u, TileWidth); // ~16M Segments: capacity for 4k x 2k VSM physical pool, with all pages/tiles occupied.
110 Out.TileSegmentMaxCount = bIsShadowView ? FMath::Min(Out.NumTilesA * TileCapacity, MaxShadowViewSegmentCount) : Out.NumTilesA * TileCapacity;
111 Out.TileCount = GraphBuilder.CreateBuffer(FRDGBufferDesc::CreateStructuredDesc(sizeof(uint32), Out.NumTilesA), TEXT("Nanite.Curve.TileCount"));
112 Out.TileOffset = GraphBuilder.CreateBuffer(FRDGBufferDesc::CreateStructuredDesc(sizeof(uint32), Out.NumTilesA), TEXT("Nanite.Curve.TileOffset"));
113 Out.TileMinMaxZ = GraphBuilder.CreateBuffer(FRDGBufferDesc::CreateStructuredDesc(sizeof(uint32), 2), TEXT("Nanite.Curve.TileMinMaxZ"));
114 Out.TileSegment = GraphBuilder.CreateBuffer(FRDGBufferDesc::CreateStructuredDesc(sizeof(uint32) * 4, Out.TileSegmentMaxCount), TEXT("Nanite.Curve.TileSegment"));
115
116 // per-view transient textures
117 Out.TileHeat = GraphBuilder.CreateTexture(FRDGTextureDesc::Create2D(TileCountXY, EPixelFormat::PF_R32_FLOAT, FClearValueBinding::None, ETextureCreateFlags::ShaderResource | ETextureCreateFlags::UAV), TEXT("Nanite.Curve.TileHeat"));
118 Out.OutCoverage = GraphBuilder.CreateTexture(FRDGTextureDesc::Create2D(ViewResolution, EPixelFormat::PF_R32_FLOAT, FClearValueBinding::None, ETextureCreateFlags::ShaderResource | ETextureCreateFlags::UAV), TEXT("Nanite.Curve.OutCoverage"));
119 Out.OutVisibility = GraphBuilder.CreateTexture(FRDGTextureDesc::Create2D(ViewResolution, EPixelFormat::PF_R32_UINT, FClearValueBinding::None, ETextureCreateFlags::ShaderResource | ETextureCreateFlags::UAV), TEXT("Nanite.Curve.OutVisibility"));
120 Out.OutVisibilityW = GraphBuilder.CreateTexture(FRDGTextureDesc::Create2D(ViewResolution, EPixelFormat::PF_R32_UINT, FClearValueBinding::None, ETextureCreateFlags::ShaderResource | ETextureCreateFlags::UAV), TEXT("Nanite.Curve.OutVisibilityW"));
121
122 // Active tiles
123 Out.ActiveTileCount = GraphBuilder.CreateBuffer(FRDGBufferDesc::CreateStructuredDesc(sizeof(uint32), 1), TEXT("Nanite.Curve.ActiveTileCount"));
124 Out.ActiveTile = GraphBuilder.CreateBuffer(FRDGBufferDesc::CreateStructuredDesc(sizeof(uint32), Out.NumTilesA), TEXT("Nanite.Curve.ActiveTile"));
125 Out.ActiveTileIndirectArgs = GraphBuilder.CreateBuffer(FRDGBufferDesc::CreateIndirectDesc<FRHIDispatchIndirectParameters>(1u), TEXT("Nanite.Curve.ActiveTileIndirectArgs"));
126
127 // GroupIndex is filled during the AsyncTask
128 //Out.GroupIndex = ...
129}

491 // default parameters
492 FNaniteCurveCommonParameters DefaultCommonParams;
493 {
494 SetCurveCommonParameters(GraphBuilder, View, Resources, DefaultCommonParams);
495 }
496 FNaniteCurveBinningParameters DefaultBinningParams;
497 {

500 }
501 FNaniteCurveResolveParameters DefaultResolveParams;
502 {
503 DefaultResolveParams.CommonParameters = DefaultCommonParams;
504 DefaultResolveParams.RasterParameters = RasterContext.Parameters;
505 }
506
507 // binning
508 // i. clear tiles
509 // ii. count segments over tiles
510 // iii. calc tile heat
511 // iv. calc tile offsets
512 // v. write segments into tiles
513 {
514 RDG_EVENT_SCOPE(GraphBuilder, "Nanite::CurveBinning");
515
516 // i. clear tiles
517 {
518 auto Params = GraphBuilder.AllocParameters(&DefaultCommonParams);
519 auto Kernel = ShaderMap->GetShader<FNaniteCurveTileClear>();

532 // ii. count segments over tiles
533 {
534 auto* Params = GraphBuilder.AllocParameters<FNaniteCurveTileCount::FParameters>();

536 Params->CommonParameters = DefaultCommonParams;

540 }
541

556 }
557 );

566 FComputeShaderUtils::AddPass(

579 auto Kernel = ShaderMap->GetShader<FNaniteCurveTileIndirectArgs>();

582 GraphBuilder,
583 RDG_EVENT_NAME("CurveTileIndirectArgs"),
584 PassFlags,
585 Kernel,
586 Params,
587 FIntVector(1,1,1)
588 );
589 }
590 }
591 }
592
593 // iv. calc tile offsets
594 {
595 // minimum wave size decides group size
596 //TODO: maybe pick according to max wave size and supply [WaveSize()] in hlsl
597 int GroupSize = -1;
598 {
599 switch (GRHIMinimumWaveSize)
600 {
601 case 4: GroupSize = 16; break;
602 case 8: GroupSize = 64; break;
603 case 16: GroupSize = 256; break;
604 case 32:
605 case 64:
606 case 128: GroupSize = 1024; break;
607 default:
608 check(false);
609 }
610 }
611
612 FNaniteCurveTileOffsetScan::FPermutationDomain PermutationVectorScan;
613 FNaniteCurveTileOffsetMerge::FPermutationDomain PermutationVectorMerge;
614 {
615 PermutationVectorScan.Set<FNaniteCurveTileOffsetScan::FGroupSizeDim>(GroupSize);
616 PermutationVectorMerge.Set<FNaniteCurveTileOffsetMerge::FGroupSizeDim>(GroupSize);
617 }
618
619 auto Params = GraphBuilder.AllocParameters(&DefaultCommonParams);
620 auto KernelScan = ShaderMap->GetShader<FNaniteCurveTileOffsetScan>(PermutationVectorScan);
621 auto KernelMerge = ShaderMap->GetShader<FNaniteCurveTileOffsetMerge>(PermutationVectorMerge);
622 {
623 uint32& RefStride = Params->TileStride;
624
625 // block scan
626 for (RefStride = 1; RefStride < NumTilesA; RefStride *= GroupSize)
627 {
628 //LOG("Scan (Block) Stride %d => Tile Subset %d, Warps %d", RefTileStride, FMath::DivideAndRoundUp(NumTilesA, RefTileStride), FMath::DivideAndRoundUp(FMath::DivideAndRoundUp(NumTilesA, TileStride), WaveSize));
629 FComputeShaderUtils::AddPass(
630 GraphBuilder,
631 RDG_EVENT_NAME("CurveTileOffsetScan"),
632 PassFlags,
633 KernelScan,
634 GraphBuilder.AllocParameters(Params),// copy to instantiate changing stride
635 FComputeShaderUtils::GetGroupCount(FMath::DivideAndRoundUp(NumTilesA, RefStride), GroupSize)
636 );
637 }
638
639 // block merge
640 for (RefStride /= (GroupSize * GroupSize); RefStride >= 1; RefStride /= GroupSize)
641 {
642 //LOG("Merge (Block) Stride %d => Tile Subset %d, Warps %d", TileStride, FMath::DivideAndRoundUp(NumTilesA, RefTileStride), FMath::DivideAndRoundUp(FMath::DivideAndRoundUp(NumTilesA, RefTileStride), WaveSize));
643 FComputeShaderUtils::AddPass(
644 GraphBuilder,
645 RDG_EVENT_NAME("CurveTileOffsetMerge"),
646 PassFlags,
647 KernelMerge,
648 GraphBuilder.AllocParameters(Params),// copy to instantiate changing stride
649 FComputeShaderUtils::GetGroupCount(FMath::DivideAndRoundUp(NumTilesA, RefStride), GroupSize)
650 );
651 }
652 }
653 }
654
655 // v. write segments into tiles
656 {
657 auto* Params = GraphBuilder.AllocParameters<FNaniteCurveTileWrite::FParameters>();
658 SetCurveClusterParameters(BinningData, ClusterOffsetSWHW, IndirectArgs, Params->ClusterParameters);
659 Params->CommonParameters = DefaultCommonParams;
660 if (IsUsingVirtualShadowMap())
661 {
662 Params->VirtualShadowMap = VirtualTargetParameters;
663 }
664
665 FNaniteCurveTileWrite::FPermutationDomain PermutationVector;
666 PermutationVector.Set<FNaniteCurveTileWrite::FMultiViewDim>(bMultiView);
667 PermutationVector.Set<FNaniteCurveTileWrite::FVirtualTextureTargetDim>(IsUsingVirtualShadowMap());
668 const TShaderRef<FNaniteCurveTileWrite>& Kernel = SharedContext.ShaderMap->GetShader<FNaniteCurveTileWrite>(PermutationVector);
669 ClearUnusedGraphResources(Kernel, Params);
670
671 GraphBuilder.AddPass(
672 RDG_EVENT_NAME("CurveTileWrite"),
673 Params,
674 PassFlags,
675 [Params, &DispatchContext, Kernel](FRDGAsyncTask, FRHIComputeCommandList& RHICmdList)
676 {
677 DispatchIndirectHelper(RHICmdList, Kernel, *Params, Params->ClusterParameters.IndirectArgs, DispatchContext.CurveResources.GroupIndex);
678 }
679 );
680 }
681 }
682
683 // resolve
684 // i. resolve tiles (dedicated fine raster)
685 // ii. translate to nanite visibility
686 {
687 RDG_EVENT_SCOPE(GraphBuilder, "Nanite::CurveResolve");
688
689 // i. resolve tiles (dedicated fine raster)
690 {
691 auto Params = GraphBuilder.AllocParameters(&DefaultResolveParams);
692 auto Kernel = ShaderMap->GetShader<FNaniteCurveTileResolve>();
693 {
694 FComputeShaderUtils::AddPass(
695 GraphBuilder,
696 RDG_EVENT_NAME("CurveTileResolve"),
697 PassFlags,
698 Kernel,
699 Params,
700 Params->CommonParameters.ActiveTileIndirectArgs, 0u
701 );
702 }
703 }
704
705 // ii. translate to nanite visibility
706 {
707 FNaniteCurveTileTranslate::FPermutationDomain PermutationVector;
708 PermutationVector.Set<FNaniteCurveTileTranslate::FMultiViewDim>(bMultiView);
709 PermutationVector.Set<FNaniteCurveTileTranslate::FVirtualTextureTargetDim>(IsUsingVirtualShadowMap());
710
711 auto Params = GraphBuilder.AllocParameters(&DefaultResolveParams);
712 auto Kernel = ShaderMap->GetShader<FNaniteCurveTileTranslate>(PermutationVector);
713 {
714 FComputeShaderUtils::AddPass(
715 GraphBuilder,
716 RDG_EVENT_NAME("CurveTileTranslate"),
717 PassFlags,
718 Kernel,
719 Params,
720 Params->CommonParameters.ActiveTileIndirectArgs, 0u
721 );
722 }
723 }
