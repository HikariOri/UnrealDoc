327 virtual void SetRange(const TRange<FFrameNumber>& NewRange)
328 {
329 // Skip TryModify for objects that still need initialization (i.e. we're in the object's constructor), because modifying objects in their constructor can lead to non-deterministic cook issues.
330 bool bCanSetRange = true;
331 if (!HasAnyFlags(RF_NeedInitialization))
332 {
333 const bool bHasChanged = NewRange != SectionRange.Value;
334 bCanSetRange = !IsReadOnly() && bHasChanged && TryModify();
335 }
336
337 const bool bInvariants = ensure(
338 NewRange.GetLowerBound().IsOpen() || NewRange.GetUpperBound().IsOpen()
339 || NewRange.GetLowerBoundValue() <= NewRange.GetUpperBoundValue()
340 );
341 if (bCanSetRange && bInvariants)
342 {
343 SectionRange.Value = NewRange;
344 }
345 }

884template<typename SectionParams>
885inline FFrameNumber GetFirstLoopStartOffsetAtTrimTime(FQualifiedFrameTime TrimTime, const SectionParams& Params, FFrameNumber StartFrame, FFrameRate FrameRate)
886{
887 const float AnimPlayRate = FMath::IsNearlyZero(Params.PlayRate) ? 1.0f : Params.PlayRate;
888 const float AnimPosition = static_cast<float>((TrimTime.Time - StartFrame) / TrimTime.Rate * AnimPlayRate);
889 const float SeqLength = static_cast<float>(Params.GetSequenceLength() - FrameRate.AsSeconds(Params.StartFrameOffset + Params.EndFrameOffset) / AnimPlayRate);
890
891 FFrameNumber NewOffset = FrameRate.AsFrameNumber(FMath::Fmod(AnimPosition, SeqLength));
892 NewOffset += Params.FirstLoopStartFrameOffset;
893
894 const FFrameNumber SeqLengthInFrames = FrameRate.AsFrameNumber(SeqLength);
895 NewOffset = NewOffset % SeqLengthInFrames;
896
897 return NewOffset;
