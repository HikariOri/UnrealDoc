199// Predicate to determine if an overlap is with a certain AActor.
200struct FPredicateOverlapHasSameActor
201{
202 FPredicateOverlapHasSameActor(const AActor& Owner)
203 : MyOwnerPtr(&Owner)
204 {
205 }
206
207 bool operator() (const FOverlapInfo& Info)
208 {
209 // since we only care whether the overlaping actor is the same as MyOwnerPtr
210 // we only call GetCachedActor (which resolves the stored handle and returns the actor)
211 // rather than calling FindActor which could cause actor spawning if the entity hit
212 // is an instanced actor. We obviously don't need to spawn an actor to determine
213 // that it won't be the same as MyOwnerPtr (which has already been spawned).
214 const AActor* HitActor = Info.OverlapInfo.HitObjectHandle.GetCachedActor();
215 // MyOwnerPtr is always valid, so we don't need the IsValid() checks in the WeakObjectPtr comparison operator.
216 return HitActor && MyOwnerPtr.HasSameIndexAndSerialNumber(HitActor);
217 }
218
219private:
220 const TWeakObjectPtr<const AActor> MyOwnerPtr;
221};
222
223// Predicate to determine if an overlap is *NOT* with a certain AActor.
224struct FPredicateOverlapHasDifferentActor
225{
226 FPredicateOverlapHasDifferentActor(const AActor& Owner)
227 : MyOwnerPtr(&Owner)
228 {
229 }
230
231 bool operator() (const FOverlapInfo& Info)
232 {
233 // since we only care whether the overlaping actor is not the same as MyOwnerPtr
234 // we only call GetCachedActor (which resolves the stored handle and returns the actor)
235 // rather than calling FindActor which could cause actor spawning if the entity hit
236 // is an instanced actor. We obviously don't need to spawn an actor to determine
237 // that it won't be the same as MyOwnerPtr (which has already been spawned).
238 const AActor* HitActor = Info.OverlapInfo.HitObjectHandle.GetCachedActor();
239 // MyOwnerPtr is always valid, so we don't need the IsValid() checks in the WeakObjectPtr comparison operator.
240 return !HitActor || !MyOwnerPtr.HasSameIndexAndSerialNumber(HitActor);
241 }
242
243private:
244 const TWeakObjectPtr<const AActor> MyOwnerPtr;
245};
