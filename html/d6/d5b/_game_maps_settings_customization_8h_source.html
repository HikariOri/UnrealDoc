41 virtual void CustomizeDetails( IDetailLayoutBuilder& LayoutBuilder ) override
42 {
43 // Add extra info around 'Global Default Game Mode'
44 IDetailCategoryBuilder& DefaultModesCategory = LayoutBuilder.EditCategory("DefaultModes");
45 {
46 // Get the object that we are viewing details of. Expect to only edit one GameMapSettings object at a time!
47 TArray< TWeakObjectPtr<UObject> > ObjectsCustomized;
48 LayoutBuilder.GetObjectsBeingCustomized(ObjectsCustomized);
49 UObject* ObjectCustomized = (ObjectsCustomized.Num() > 0) ? ObjectsCustomized[0].Get() : NULL;
50 // Get name of GameMode property
51 static FName GlobalDefaultGameModeName(TEXT("GlobalDefaultGameMode"));
52
53 // Allocate customizer object
54 GameInfoModeCustomizer = MakeShareable(new FGameModeInfoCustomizer(ObjectCustomized, GlobalDefaultGameModeName));
55
56 // Then use it to customize
57 GameInfoModeCustomizer->CustomizeGameModeSetting(LayoutBuilder, DefaultModesCategory);
58 }
59 }

149 TSharedRef<SWidget> MakeMapMenu( const TSharedPtr<IPropertyHandle>& PropertyHandle )
150 {
151 FMenuBuilder MenuBuilder(true, NULL);
152
153 const FString MapFileWildCard = FString::Printf(TEXT("*%s"), *FPackageName::GetMapPackageExtension());
154 TArray<FString> MapNames;
155
156 // maps that belong to the project
157 MenuBuilder.BeginSection(NAME_None, LOCTEXT("ProjectMapsSectionHeader", "Project"));
158 {
159 IFileManager::Get().FindFilesRecursive(MapNames, *FPaths::ProjectContentDir(), *MapFileWildCard, true, false);
160 MapNames.Sort([](const FString& A, const FString& B) { return FPaths::GetBaseFilename(A) < FPaths::GetBaseFilename(B); });
161
162 for (int32 i = 0; i < MapNames.Num(); i++)
163 {
164 FUIAction Action(FExecuteAction::CreateRaw(this, &FGameMapsSettingsCustomization::HandleMapSelected, MapNames[i], PropertyHandle));
165 MenuBuilder.AddMenuEntry(
166 FText::FromString( FPaths::GetBaseFilename(MapNames[i]) ),
167 FText::FromString( MapNames[i] ),
168 FSlateIcon(),
169 Action
170 );
171 }
172 }
173 MenuBuilder.EndSection();
174
175 // maps that belong to the Engine
176 MenuBuilder.BeginSection(NAME_None, LOCTEXT("EngineMapsSectionHeader", "Engine"));
177 {
178 IFileManager::Get().FindFilesRecursive(MapNames, *FPaths::EngineContentDir(), *MapFileWildCard, true, false);
179 MapNames.Sort([](const FString& A, const FString& B) { return FPaths::GetBaseFilename(A) < FPaths::GetBaseFilename(B); });
180
181 for (int32 i = 0; i < MapNames.Num(); i++)
182 {
183 FUIAction Action(FExecuteAction::CreateRaw(this, &FGameMapsSettingsCustomization::HandleMapSelected, MapNames[i], PropertyHandle));
184 MenuBuilder.AddMenuEntry(
185 FText::FromString( FPaths::GetBaseFilename(MapNames[i]) ),
186 FText::FromString( MapNames[i] ),
187 FSlateIcon(),
188 Action
189 );
190 }
191 }
192 MenuBuilder.EndSection();
193
194 return MenuBuilder.MakeWidget();
195 }

210 // Handles getting the text color of a map text block.
211 FSlateColor HandleMapTextBoxForegroundColor( TSharedPtr<IPropertyHandle> PropertyHandle ) const
212 {
213 FString Value;
214
215 if (PropertyHandle->GetValue(Value) == FPropertyAccess::Success)
216 {
217 if (Value.IsEmpty() || IsValidMapName(Value))
218 {
219 static const FName InvertedForegroundName("InvertedForeground");
220
221 return FAppStyle::GetSlateColor(InvertedForegroundName);
222 }
223 }
224
225 return FLinearColor::Red;
226 }

232

234 {
235 return FText::FromString(Value);
236 }
237
238 return FText::GetEmpty();

254 PropertyHandle->SetValue(FString());
