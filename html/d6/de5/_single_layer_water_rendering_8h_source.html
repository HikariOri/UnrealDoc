110 PassParameters->IndirectDrawParameter = TiledScreenSpaceReflection ? TiledScreenSpaceReflection->DrawIndirectParametersBuffer : nullptr;
111
112 PassParameters->VS.ViewUniformBuffer = ViewUniformBuffer;
113 PassParameters->VS.TileListData = TiledScreenSpaceReflection ? TiledScreenSpaceReflection->TileListDataBufferSRV : nullptr;
114
115 ValidateShaderParameters(PixelShader, PassParameters->PS);
116 ClearUnusedGraphResources(PixelShader, &PassParameters->PS);
117
118 const bool bRunTiled = TiledScreenSpaceReflection != nullptr;
119 if (bRunTiled)
120 {
121 FWaterTileVS::FPermutationDomain PermutationVector;
122 TShaderMapRef<FWaterTileVS> VertexShader(GlobalShaderMap, PermutationVector);
123
124 ValidateShaderParameters(VertexShader, PassParameters->VS);
125 ClearUnusedGraphResources(VertexShader, &PassParameters->VS);
126
127 GraphBuilder.AddPass(
128 Forward<FRDGEventName>(PassName),
129 PassParameters,
130 ERDGPassFlags::Raster,
131 [PassParameters, GlobalShaderMap, Viewport, TiledScreenSpaceReflection, VertexShader, PixelShader, BlendState, RasterizerState, DepthStencilState, StencilRef](FRDGAsyncTask, FRHICommandList& RHICmdList)
132 {
133 RHICmdList.SetViewport(Viewport.Min.X, Viewport.Min.Y, 0.0f, Viewport.Max.X, Viewport.Max.Y, 1.0f);
134
135 FGraphicsPipelineStateInitializer GraphicsPSOInit;
136 FPixelShaderUtils::InitFullscreenPipelineState(RHICmdList, GlobalShaderMap, PixelShader, GraphicsPSOInit);
137
138 GraphicsPSOInit.PrimitiveType = GRHISupportsRectTopology ? PT_RectList : PT_TriangleList;
139 GraphicsPSOInit.BlendState = BlendState ? BlendState : GraphicsPSOInit.BlendState;
140 GraphicsPSOInit.RasterizerState = RasterizerState ? RasterizerState : GraphicsPSOInit.RasterizerState;
141 GraphicsPSOInit.DepthStencilState = DepthStencilState ? DepthStencilState : GraphicsPSOInit.DepthStencilState;
142 GraphicsPSOInit.BoundShaderState.VertexDeclarationRHI = GEmptyVertexDeclaration.VertexDeclarationRHI;
143 GraphicsPSOInit.BoundShaderState.VertexShaderRHI = VertexShader.GetVertexShader();
144 GraphicsPSOInit.BoundShaderState.PixelShaderRHI = PixelShader.GetPixelShader();
145
146 SetGraphicsPipelineState(RHICmdList, GraphicsPSOInit, StencilRef);
147
148 SetShaderParameters(RHICmdList, VertexShader, VertexShader.GetVertexShader(), PassParameters->VS);
149 SetShaderParameters(RHICmdList, PixelShader, PixelShader.GetPixelShader(), PassParameters->PS);
150
151 RHICmdList.DrawPrimitiveIndirect(PassParameters->IndirectDrawParameter->GetIndirectRHICallBuffer(), 0);
152 });
153 }
154 else

157 Forward<FRDGEventName>(PassName),
158 PassParameters,
159 ERDGPassFlags::Raster,

163

166
167 GraphicsPSOInit.BlendState = BlendState ? BlendState : GraphicsPSOInit.BlendState;
168 GraphicsPSOInit.RasterizerState = RasterizerState ? RasterizerState : GraphicsPSOInit.RasterizerState;
169 GraphicsPSOInit.DepthStencilState = DepthStencilState ? DepthStencilState : GraphicsPSOInit.DepthStencilState;
170
171 SetGraphicsPipelineState(RHICmdList, GraphicsPSOInit, StencilRef);
172
173 SetShaderParameters(RHICmdList, PixelShader, PixelShader.GetPixelShader(), PassParameters->PS);
174
175 FPixelShaderUtils::DrawFullscreenTriangle(RHICmdList);
176 });
