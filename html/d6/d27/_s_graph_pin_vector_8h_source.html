182 void SetNewValueHelper(EAxis::Type Axis, NumericType NewValue)
183 {
184 if (GraphPinObj->IsPendingKill())
185 {
186 return;
187 }
188
189 FVectorType NewVector = ConvertDefaultValueStringToVector();
190 const NumericType OldValue = NewVector.GetComponentForAxis(Axis);
191
192 if (OldValue == NewValue)
193 {
194 return;
195 }
196
197 NewVector.SetComponentForAxis(Axis, NewValue);
198
199 const FScopedTransaction Transaction(NSLOCTEXT("GraphEditor", "ChangeVectorPinValue", "Change Vector Pin Value"));
200 GraphPinObj->Modify();
201
202 // Create the new value string
203 FString DefaultValue = FString::Format(TEXT("{0},{1},{2}"), { NewVector.X, NewVector.Y, NewVector.Z });
204
205 //Set new default value
206 GraphPinObj->GetSchema()->TrySetDefaultValue(*GraphPinObj, DefaultValue);
207 }

209 /*
210 * @brief Converts the default string value to a value of VectorType.
211 *
212 * Example: it converts the default string value "2.00,3.00,4.00" to the corresponding 3D vector.
213 */
214 FVectorType ConvertDefaultValueStringToVector() const
215 {
216 const TArray<FString> VecComponentStrings = GetComponentArray();
217
218 // Construct the vector from the string parts.
219 FVectorType Vec = FVectorType::ZeroVector;
220 TDefaultNumericTypeInterface<NumericType> NumericTypeInterface{};
221
222 // If default string value contained a fully specified 3D vector, set the vector components, otherwise leave it zero'ed.
223 if (VecComponentStrings.Num() == 3)
224 {
225 Vec.X = NumericTypeInterface.FromString(VecComponentStrings[0], 0).Get(0);
226 Vec.Y = NumericTypeInterface.FromString(VecComponentStrings[1], 0).Get(0);
227 Vec.Z = NumericTypeInterface.FromString(VecComponentStrings[2], 0).Get(0);
228 }
229
230 return Vec;
231 }
