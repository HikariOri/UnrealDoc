113 TEventType& EmplaceProfilerEvent(TArgs&&... Args)
114 {
115 TEventType& Data = State.EventStream.Emplace<TEventType>(Forward<TArgs>(Args)...);

117 if constexpr (std::is_same_v<UE::RHI::GPUProfiler::FEvent::FBeginWork, TEventType>)
118 {
119 // Store BeginEvents in a separate array as the CPUTimestamp field needs updating at submit time.
