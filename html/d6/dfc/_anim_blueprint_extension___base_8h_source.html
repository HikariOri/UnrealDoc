211 bool IsFastPath() const
212 {
213 for(TMap<FName, FAnimNodeSinglePropertyHandler>::TConstIterator It(ServicedProperties); It; ++It)
214 {
215 const FAnimNodeSinglePropertyHandler& AnimNodeSinglePropertyHandler = It.Value();
216 for (const FPropertyCopyRecord& CopyRecord : AnimNodeSinglePropertyHandler.CopyRecords)
217 {
218 if (!CopyRecord.IsFastPath())
219 {
220 return false;
221 }
222 }
223 }
224
225 return true;
226 }
