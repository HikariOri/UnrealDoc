174 virtual bool TerminateProcess( const int64 ProcessId ) override
175 {
176#if PLATFORM_LINUX // if running natively, just terminate the local process
177 // get process path from the ProcessId
178 const int32 ReadLinkSize = 1024;
179 char ReadLinkCmd[ReadLinkSize] = { 0 };
180 FCStringAnsi::Sprintf(ReadLinkCmd, "/proc/%lld/exe", ProcessId);
181 char ProcessPath[UNIX_MAX_PATH + 1] = { 0 };
182 int32 Ret = readlink(ReadLinkCmd, ProcessPath, UE_ARRAY_COUNT(ProcessPath) - 1);
183 if (Ret != -1)
184 {
185 struct stat st;
186 uid_t euid;
187 stat(ProcessPath, &st);
188 euid = geteuid(); // get effective uid of current application, as this user is asking to kill a process
189
190 // now check if we own the process
191 if (st.st_uid == euid)
192 {
193 // terminate it (will this terminate children as well because we set their pgid?)
194 kill(ProcessId, SIGTERM);
195 sleep(2); // sleep in case process still remains then send a more strict signal
196 kill(ProcessId, SIGKILL);
197 return true;
198 }
199 }
200#else
201 // @todo: support remote termination
202 STUBBED("FLinuxTargetDevice::TerminateProcess");
203#endif // PLATFORM_LINUX
204 return false;
205 }
