31{
32protected:
33 union
34 {
35 PTRINT Size;
36 uint8* Memory;
37 };
38 FAsyncFileCallBack Callback;
39 std::atomic<bool> bDataIsReady = false;
40 std::atomic<bool> bCompleteAndCallbackCalled = false;
41 std::atomic<bool> bCompleteSync = false;
42 std::atomic<bool> bCanceled = false;
43 const bool bSizeRequest;
44 const bool bUserSuppliedMemory;
45public:
46
47 inline IAsyncReadRequest(FAsyncFileCallBack* InCallback, bool bInSizeRequest, uint8* UserSuppliedMemory)
48 : bSizeRequest(bInSizeRequest)
49 , bUserSuppliedMemory(!!UserSuppliedMemory)
50 {
51 if (InCallback)
52 {
53 Callback = *InCallback;
54 }
55 if (bSizeRequest)
56 {
57 Size = -1;
58 check(!UserSuppliedMemory && !bUserSuppliedMemory); // size requests don't do memory
59 }
60 else
61 {
62 Memory = UserSuppliedMemory;
63 check((!!Memory) == bUserSuppliedMemory);
64 }
65 INC_DWORD_STAT(STAT_AsyncFileRequests);
66 }
67
68 /* Not legal to destroy the request until it is complete. */
69 virtual ~IAsyncReadRequest()
70 {
71 // check(bCompleteAndCallbackCalled && (bSizeRequest || !Memory)); // must be complete, and if it was a read request, the memory should be gone
72 UE_CLOGF(!(bCompleteAndCallbackCalled.load(std::memory_order_relaxed) && (bSizeRequest || !Memory)),
73 LogHAL, Fatal, "IAsyncReadRequests must not be deleted until they are completed."
74 );
75 DEC_DWORD_STAT(STAT_AsyncFileRequests);
76 }
77
82 UE_FORCEINLINE_HINT bool PollCompletion()
83 {
84 // fast path to avoid a barrier until the value returns true
85 if (bCompleteAndCallbackCalled.load(std::memory_order_relaxed))
86 {
87 // Once we return true, we need a barrier to guarantee the memory
88 // writes that have happened before the release barrier can be seen.
89 // Because for all we know, PollCompletion() could be the only thing
90 // that is used to synchronize with the writing thread.
91 return bCompleteAndCallbackCalled.load(std::memory_order_acquire);
92 }
93
94 return false;
95 }
96
102 inline bool WaitCompletion(float TimeLimitSeconds = 0.0f)
103 {
104 if (PollCompletion())
105 {
106 return true;
107 }
108 WaitCompletionImpl(TimeLimitSeconds);
109 return PollCompletion();
110 }
111
116 virtual void EnsureCompletion()
117 {
118 // Default implementation is the same as WaitCompletion(0.0f) except that it skips the testing of
119 // PollCompletion. This is potentially slower because we do not early exit if PollCompletion is true, but it
120 // provides a stronger guarantee of completion because PollCompletion can sometimes return true while
121 // completion steps are still in progress.
122 WaitCompletionImpl(0.0f);
123 }
124
126 inline void Cancel()
127 {
128 if (!bCanceled.load(std::memory_order_relaxed))
129 {
130 bCanceled.store(true, std::memory_order_release);
131 bDataIsReady.store(true, std::memory_order_release);
132
133 if (!PollCompletion())
134 {
135 return CancelImpl();
136 }
137 }
138 }
139
144 inline int64 GetSizeResults()
145 {
146 check(bDataIsReady.load(std::memory_order_relaxed) && bSizeRequest);
147 return bCanceled.load(std::memory_order_relaxed) ? -1 : Size;
148 }
149
154 inline uint8* GetReadResults()
155 {
156 check(bDataIsReady.load(std::memory_order_relaxed) && !bSizeRequest);
157 uint8* Result = Memory;
158 if (bCanceled.load(std::memory_order_relaxed) && Result)
159 {
160 Result = nullptr;
161 }
162 else
163 {
164 if (Memory && !bUserSuppliedMemory)
165 {
166 ReleaseMemoryOwnershipImpl();
167 }
168
169 Memory = nullptr;
170 }
171 return Result;
172 }
173
174protected:
180 virtual void WaitCompletionImpl(float TimeLimitSeconds) = 0;
181
183 virtual void CancelImpl() = 0;
184
190 virtual void ReleaseMemoryOwnershipImpl() = 0;
191
192 void SetDataComplete()
193 {
194 bDataIsReady.store(true, std::memory_order_release);
195 if (Callback)
196 {
197 Callback(bCanceled.load(std::memory_order_relaxed), this);
198 }
199 }
200
201 void SetAllComplete()
202 {
203 bCompleteAndCallbackCalled.store(true, std::memory_order_release);
204 }
205
206 void SetComplete()
207 {
208 SetDataComplete();
209 SetAllComplete();
210 }
211};
