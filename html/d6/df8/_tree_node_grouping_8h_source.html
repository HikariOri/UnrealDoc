194template<typename Type>
195void TTreeNodeGroupingByUniqueValue<Type>::GroupNodes(const TArray<FTableTreeNodePtr>& Nodes, FTableTreeNode& ParentGroup, TWeakPtr<FTable> InParentTable, IAsyncOperationProgress& InAsyncOperationProgress) const
196{
197 TMap<Type, FTableTreeNodePtr> GroupMap;
198 FTableTreeNodePtr UnsetGroupPtr = nullptr;
199
200 ParentGroup.ClearChildren();
201
202 for (FTableTreeNodePtr NodePtr : Nodes)
203 {
204 if (InAsyncOperationProgress.ShouldCancelAsyncOp())
205 {
206 return;
207 }
208
209 if (NodePtr->IsGroup())
210 {
211 ParentGroup.AddChildAndSetParent(NodePtr);
212 continue;
213 }
214
215 FTableTreeNodePtr GroupPtr = nullptr;
216
217 const FTableColumn& Column = *GetColumn();
218 const TOptional<FTableCellValue> CellValue = Column.GetValue(*NodePtr);
219 if (CellValue.IsSet())
220 {
221 const Type Value = GetValue(CellValue.GetValue());
222
223 FTableTreeNodePtr* GroupPtrPtr = GroupMap.Find(Value);
224 if (!GroupPtrPtr)
225 {
226 const FText GroupName = Column.GetValueAsGroupingText(*NodePtr);
227 GroupPtr = CreateGroupNode(InParentTable, GroupName);
228 ParentGroup.AddChildAndSetParent(GroupPtr);
229 GroupMap.Add(Value, GroupPtr);
230 }
231 else
232 {
233 GroupPtr = *GroupPtrPtr;
234 }
235 }
236 else
237 {
238 if (!UnsetGroupPtr)
239 {
240 UnsetGroupPtr = CreateUnsetGroupNode(InParentTable);
241 ParentGroup.AddChildAndSetParent(UnsetGroupPtr);
242 }
243 GroupPtr = UnsetGroupPtr;
244 }
245
246 GroupPtr->AddChildAndSetParent(NodePtr);
247 }
248}
