86 bool ActivateConstraint(FPBDCollisionConstraint* Constraint)
87 {
88 FPBDCollisionConstraintContainerCookie& Cookie = Constraint->GetContainerCookie();
89 // Use the LastUsedEpoch to catch duplicate add attempts. This is still possible
90 // via rewind/resim and InjectCollision. Ideally we wouldn't need do update
91 // the cookie data until FCollisionConstraintAllocator::ProcessNewCollisions...
92 if (Cookie.LastUsedEpoch != CurrentEpoch)
93 {
94 Cookie.LastUsedEpoch = CurrentEpoch;
95 Cookie.ConstraintIndex = INDEX_NONE;
96 Cookie.CCDConstraintIndex = INDEX_NONE;
97
98 NewActiveConstraints.Push(Constraint);
99
100 Constraint->Activate();
101
102 return true;
103 }
104 return false;

365 void BeginDetectCollisions()
366 {
367 check(!bInCollisionDetectionPhase);
368 bInCollisionDetectionPhase = true;
369
370 // Update the tick counter
371 // NOTE: We do this here rather than in EndDetectionCollisions so that any contacts injected
372 // before collision detection count as the previous frame's collisions, e.g., from Islands
373 // that are manually awoken by modifying a particle on the game thread. This also needs to be
374 // done where we reset the Constraints array so that we can tell we have a valid index from
375 // the Epoch.
376 ++CurrentEpoch;
377
378 for (const TUniquePtr<FCollisionContextAllocator>& ContextAllocator : ContextAllocators)
379 {
380 ContextAllocator->BeginDetectCollisions(CurrentEpoch);
381 }
382
383 ResetActiveConstraints();
384
385 // Clear the collision list for this tick - we are about to rebuild them
386 ActiveConstraints.Reset();
387 ActiveCCDConstraints.Reset();
388 }

429 void AddResimConstraints(const TArray<FPBDCollisionConstraint>& InConstraints, const FCollisionContext Context)
430 {
431 for (const FPBDCollisionConstraint& SourceConstraint : InConstraints)
432 {
433 // We must keep the particles in the same order that the broadphase would generate when
434 // finding or creating the midphase. This is because Collision Constraints may have the
435 // particles in the opposite order to the midphase tha owns them.
436 FGeometryParticleHandle* Particle0 = SourceConstraint.Particle[0];
437 FGeometryParticleHandle* Particle1 = SourceConstraint.Particle[1];
438 if (ShouldSwapParticleOrder(Particle0, Particle1))
439 {
440 Swap(Particle0, Particle1);
441 }
442
443 FParticlePairMidPhase* MidPhase = Context.GetAllocator()->GetMidPhase(Particle0, Particle1, SourceConstraint.Particle[0], Context);
444
445 // We may be adding multiple constraints for the same particle pair, so we need
446 // to make sure the map is up to date in the case where we just created a new MidPhase
447 ProcessNewMidPhases();
448
449 if (MidPhase != nullptr)
450 {
451 MidPhase->InjectCollision(SourceConstraint, Context);
452 }
453 }
454
455 ProcessNewConstraints();
456 }

617 // Check the visitor flags and return true if we want to visit the constraint
618 inline bool CollisionVisitorShouldVisit(const FPBDCollisionConstraint* Constraint, const ECollisionVisitorFlags VisitFlags)
619 {
620 if (Constraint == nullptr)
621 {
622 return false;
623 }
624
625 if (!(VisitFlags & ECollisionVisitorFlags::VisitDisabled))
626 {
627 // Awkward. Sleeping constraints are considered "disabled" so if we want to
628 // visit sleeping constraints we must ignore the disabled state when asleep
629 if (!(VisitFlags & ECollisionVisitorFlags::VisitSleeping))
630 {
631 // We do NOT want to visit sleeping, so we can just check the enabled flag
632 if (!Constraint->IsEnabled())
633 {
634 return false;
635 }
636 }
637 else
638 {
639 // We do want to visit sleeping, so we only skip disabled-awake constraints
640 if (!Constraint->IsEnabled() && !Constraint->IsSleeping())
641 {
642 return false;
643 }
644 }
645 }
646
647 if (!(VisitFlags & ECollisionVisitorFlags::VisitSleeping))
648 {
649 if (Constraint->IsSleeping())
650 {
651 return false;
652 }
653 }
654
655 if (!(VisitFlags & ECollisionVisitorFlags::VisitExpired))
656 {
657 if (!Constraint->IsCurrent())
658 {
659 return false;
660 }
661 }
662
663 return true;
664 }

667 inline ECollisionVisitorResult FParticlePairMidPhase::VisitCollisions(const TLambda& Visitor, const ECollisionVisitorFlags VisitFlags)
668 {
669 if (GetMidPhaseType() == EParticlePairMidPhaseType::ShapePair)
670 {
671 FShapePairParticlePairMidPhase* This = static_cast<FShapePairParticlePairMidPhase*>(this);
672
673 for (FSingleShapePairCollisionDetector& ShapePair : This->ShapePairDetectors)
674 {
675 if (CollisionVisitorShouldVisit(ShapePair.GetConstraint(), VisitFlags))
676 {
677 if (Visitor(*ShapePair.GetConstraint()) == ECollisionVisitorResult::Stop)
678 {
679 return ECollisionVisitorResult::Stop;
680 }
681 }
682 }
683 }
684
685 if (GetMidPhaseType() == EParticlePairMidPhaseType::Generic)
686 {
687 FGenericParticlePairMidPhase* This = static_cast<FGenericParticlePairMidPhase*>(this);
688 for (auto& KVP : This->Constraints)
689 {
690 FPBDCollisionConstraintPtr& Constraint = KVP.Value;
691
692 if (CollisionVisitorShouldVisit(Constraint.Get(), VisitFlags))
693 {
694 if (Visitor(*Constraint) == ECollisionVisitorResult::Stop)
695 {
696 return ECollisionVisitorResult::Stop;
697 }
698 }
699 }
700 }
701
702 if (GetMidPhaseType() == EParticlePairMidPhaseType::SphereApproximation)
703 {
704 FSphereApproximationParticlePairMidPhase* This = static_cast<FSphereApproximationParticlePairMidPhase*>(this);
705 FPBDCollisionConstraint* Constraint = This->Constraint.Get();
706 if (Constraint != nullptr)
707 {
708 if (CollisionVisitorShouldVisit(Constraint, VisitFlags))
709 {
710 if (Visitor(*Constraint) == ECollisionVisitorResult::Stop)
711 {
712 return ECollisionVisitorResult::Stop;
713 }
714 }
715 }
716 }
717
718 return ECollisionVisitorResult::Continue;
719 }
