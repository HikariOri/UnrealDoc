161 {
162 FBufferRHIRef Buffer;
163
164 FResourceArrayInterface* RESTRICT ResourceArray = InOutResourceObject ? InOutResourceObject->GetResourceArray() : nullptr;
165 if (ResourceCount != 0)
166 {
167 Buffer = CreateRHIBufferInternal(RHICmdList, InDebugName, GetOwnerName(), ResourceCount, InBufferUsageFlags, ResourceArray, InOutResourceObject == nullptr);

169
170 // If the buffer creation emptied the resource array, delete the containing structure as well

172 {
173 delete InOutResourceObject;
174 InOutResourceObject = nullptr;
175 }
176
177 return Buffer;

445template<class ResourceType, FRenderResource::EInitPhase InInitPhase = FRenderResource::EInitPhase::Default>
446class TGlobalResource final : public ResourceType
447{
448public:
450 TGlobalResource()
451 {
452 InitGlobalResource();
453 }
454
456 template<typename... Args>
457 explicit TGlobalResource(Args... InArgs)
458 : ResourceType(InArgs...)
459 {
460 InitGlobalResource();
461 }
462
464 virtual ~TGlobalResource()
465 {
466 ReleaseGlobalResource();
467 }
468
469private:
470
474 void InitGlobalResource()
475 {
476 ResourceType::SetInitPhase(InInitPhase);
477
478 if (IsInRenderingThread())
479 {
480 // If the resource is constructed in the rendering thread, directly initialize it.
481 ((ResourceType*)this)->InitResource(FRenderResource::GetImmediateCommandList());
482 }
483 else
484 {
485 // If the resource is constructed outside of the rendering thread, enqueue a command to initialize it.
486 BeginInitResource((ResourceType*)this);
487 }
488 }
489
493 void ReleaseGlobalResource()
494 {
495 // This should be called in the rendering thread, or at shutdown when the rendering thread has exited.
496 // However, it may also be called at shutdown after an error, when the rendering thread is still running.
497 // To avoid a second error in that case we don't assert.
498#if 0
499 check(IsInRenderingThread());
500#endif
501
502 // Cleanup the resource.
503 ((ResourceType*)this)->ReleaseResource();
504 }
