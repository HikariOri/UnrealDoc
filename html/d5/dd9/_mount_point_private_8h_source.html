307inline bool FMountPoint::IsHigherOverridePriority(FMountPoint* A, FMountPoint* B)
308{
309 // Must be called within FLongPackagePathsSingleton->MountLock so that GetMountOrder is readable.
310 if ((A != nullptr) != (B != nullptr))
311 {
312 return A != nullptr;
313 }
314 if (A == nullptr)
315 {
316 return false; // equal
317 }
318 CA_ASSUME(B != nullptr);
319
320 if (A->IsMounted() != B->IsMounted())
321 {
322 return A->IsMounted();
323 }
324 return A->GetMountOrder() > B->GetMountOrder();
325}
