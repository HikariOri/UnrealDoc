158struct FChaosVDScopedTLSBufferAccessor
159{
160 FChaosVDScopedTLSBufferAccessor() : BufferRef(FChaosVDThreadContext::Get().GetTLSDataBufferRef())
161 {
162 FChaosVDThreadContext::Get().BufferAccessCounter++;
163
164 // This is close to the max amount of data we can Trace as a single trace event
165 // Some times we will go over and some times this be more than we need
166 // but keeping this amount allocated on average reduces the allocation cost when tracing.
167 // TODO: There might be a better value for this, we need to do more testing
168 constexpr uint32 MaxTraceChunkSize = TNumericLimits<uint16>::Max();
169 BufferRef.Reserve(MaxTraceChunkSize);
170 }
171
172 ~FChaosVDScopedTLSBufferAccessor()
173 {
174 // Clear the buffer on exit but maintain allocated a minimum amount of memory
175 constexpr uint32 MaxTraceChunkSize = TNumericLimits<uint16>::Max();
176 BufferRef.Reset(MaxTraceChunkSize);
177
178 FChaosVDThreadContext::Get().BufferAccessCounter--;
179 }
180
181 TArray<uint8>& BufferRef;
182};
