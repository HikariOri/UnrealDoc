123 // Barycentric interpolation of a vertex attribute
124 template <typename T>
125 T InterpolateAttribute(const FVector& InVertexA,
126 const FVector& InVertexB,
127 const FVector& InVertexC,
128 const FVector& InPointInTriangle,
129 const T& InAttributeAtVertexA,
130 const T& InAttributeAtVertexB,
131 const T& InAttributeAtVertexC,
132 const T& InDefaultValue)
133 {
134 double AreaTriangleABC = 0.5 * FVector::CrossProduct(InVertexB - InVertexA, InVertexC - InVertexA).Length();
135 double AreaTriangleABP = 0.5 * FVector::CrossProduct(InVertexA - InPointInTriangle, InVertexB - InPointInTriangle).Length();
136 double AreaTriangleACP = 0.5 * FVector::CrossProduct(InVertexA - InPointInTriangle, InVertexC - InPointInTriangle).Length();
137 double AreaTriangleBCP = 0.5 * FVector::CrossProduct(InVertexB - InPointInTriangle, InVertexC - InPointInTriangle).Length();
138
139 if (AreaTriangleABC > SMALL_NUMBER)
140 {
141 const double U = AreaTriangleBCP / AreaTriangleABC;
142 const double V = AreaTriangleACP / AreaTriangleABC;
143 const double W = AreaTriangleABP / AreaTriangleABC;
144
145 if (FMath::Abs(U + V + W - 1.0) < UE_SMALL_NUMBER)
146 {
147 return U * InAttributeAtVertexA + V * InAttributeAtVertexB + W * InAttributeAtVertexC;
148 }
149 }
150
151 return InDefaultValue;
152 }
