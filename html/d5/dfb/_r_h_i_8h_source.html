161inline FMatrix AdjustProjectionMatrixForRHI(const FMatrix& InProjectionMatrix)
162{
163 FScaleMatrix ClipSpaceFixScale(FVector(1.0f, GProjectionSignY, 1.0f - GMinClipZ));
164 FTranslationMatrix ClipSpaceFixTranslate(FVector(0.0f, 0.0f, GMinClipZ));
165 return InProjectionMatrix * ClipSpaceFixScale * ClipSpaceFixTranslate;
166}
