183 TConstArrayView<FRealSingle> GetRestAngleMapFromCollection(
184 const TMap<FString, TConstArrayView<FRealSingle>>& WeightMaps,
185 const FCollectionPropertyConstFacade& PropertyCollection)
186 {
187 const ERestAngleConstructionType ConstructionType = (ERestAngleConstructionType)GetRestAngleType(PropertyCollection, (int32)ERestAngleConstructionType::Use3DRestAngles);

189 switch (ConstructionType)
190 {
191 default:
192 case ERestAngleConstructionType::Use3DRestAngles:
193 return TConstArrayView<FRealSingle>(); // Unused
194 case ERestAngleConstructionType::FlatnessRatio:
195 return WeightMaps.FindRef(GetFlatnessRatioString(PropertyCollection, FlatnessRatioName.ToString()));
196 case ERestAngleConstructionType::ExplicitRestAngles:
197 return WeightMaps.FindRef(GetRestAngleString(PropertyCollection, RestAngleName.ToString()));
198 }
