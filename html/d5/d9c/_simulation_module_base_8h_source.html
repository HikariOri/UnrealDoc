252 class FSimulationModuleTypeContainer
253 {
254 public:
255 TSet<FName> MyTypes;
256 FName MostRecentAdd = NAME_None;
257 void AddType(FName InType)
258 {
259 MyTypes.Emplace(InType);
260 MostRecentAdd = InType;
261 }
262 bool IsSimType(FName InType) const
263 {
264 return MostRecentAdd == InType || MyTypes.Contains(InType);
265 }
266 FName GetSimType() const
267 {
268 return MostRecentAdd;
269 }
270 template <typename U>
271 static FName GetStaticSimType()
272 {
273 if constexpr (TSimModuleTypeIsRecursive_v<U>)
274 {
275 return U::RecurseSimType();
276 }
277 else
278 {
279 return TSimulationModuleTypeable<U>::StaticSimType();
280 }
281 }
282 template<typename U>
283 bool IsSimType() const
284 {
285 if constexpr (TSimModuleTypeIsRecursive_v<U>)
286 {
287 return IsSimType(U::RecurseSimType());
288 }
289 else
290 {
291 return IsSimType(TSimulationModuleTypeable<U>::StaticSimType());
292 }
293 }
294 template<typename U>
295 U* Cast()
296 {
297 if(IsSimType<std::remove_const_t<U>>())
298 {
299 return static_cast<U*>(this);
300 }
301 return nullptr;
302 }
303 template<typename U>
304 const U* Cast() const
305 {
306 if(IsSimType<std::remove_const_t<U>>())
307 {
308 return static_cast<const U*>(this);
309 }
310 return nullptr;
311 }
