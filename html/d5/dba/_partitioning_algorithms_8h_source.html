59 // Partitions the elements into two sets: left < split <= right.
60 // The return value is the size of the left split.
61 // This will swap up to MaxCount items.
62 template <typename ElementType>
63 bool PartitionElementsInPlace(TArrayView<const ElementType>& Elements, const int32 SplitAxis, const FReal SplitPosition, const int32 MaxCount, int32& Start, int32& End, int32& OutSplitIndex)
64 {
65 int32 Count = 0;
66 while (Start < End && Count < MaxCount)
67 {
68 // Move start to the right until it's greater than or equal to the pivot value.
69 while (Start < End && Elements[Start].GetCenter()[SplitAxis] < SplitPosition)
70 {
71 ++Start;
72 }
73 // Move end to the left until it's less than the pivot value.
74 while (Start < End && SplitPosition <= Elements[End - 1].GetCenter()[SplitAxis])
75 {
76 --End;
77 }
78
79 // As long as the indices haven't crossed, we know this is a valid swap.
80 if (Start < End)
81 {
82 Swap(Elements[Start], Elements[End - 1]);
83 ++Start;
84 --End;
85 ++Count;
86 }
87 }
88 OutSplitIndex = Start;
89 return Start < End;
90 }

104 // Partitions the elements into two sets using the bin indices: left < split <= right.
105 // The bin indices are not updated.
106 // The return value is the size of the left split.
107 // This will swap up to MaxCount items.
108 template <typename ElementType>
109 bool PartitionElementsInPlace(TArrayView<const ElementType>& Elements, const TArrayView<int32>& BinIndices, const int32 BestSplit, const int32 MaxCount, int32& Start, int32& End, int32& OutSplitIndex)
110 {
111 check(Elements.Num() == BinIndices.Num());
112
113 int32 Count = 0;
114 while (Start < End && Count < MaxCount)
115 {
116 // Move start to the right until it's greater than or equal to the pivot value.
117 while (Start < End && BinIndices[Start] < BestSplit)
118 {
119 ++Start;
120 }
121 // Move end to the left until it's less than the pivot value.
122 while (Start < End && BestSplit <= BinIndices[End - 1])
123 {
124 --End;
125 }
126
127 // As long as the indices haven't crossed, we know this is a valid swap.
128 if (Start < End)
129 {
130 Swap(Elements[Start], Elements[End - 1]);
131 ++Start;
132 --End;
133 ++Count;
134 }
135 }
136 OutSplitIndex = Start;
137 return Start < End;

153 // Computes an axis aligned split plane (axis + value) from the spatial median of the objects.
154 // The axis with the largest spread is picked as the split axis.
155 // The center of the centroid aabb on the split axis is picked as the split value.
156 template <typename ElementType>
157 void ComputeSplitPlaneWithSpatialMedianHeuristic(const TArrayView<const ElementType>& Elements, int32& OutSplitAxis, FReal& OutSplitPosition)
158 {
159 FAABB3 Aabb = FAABB3::EmptyAABB();
160 ComputeCentroidAabb(Elements, Aabb);
161
162 const FVec3 Center = Aabb.GetCenter();
163 const FVec3 Extents = Aabb.Extents();
164 OutSplitAxis = PickLargestAxis(Extents);
165
166 OutSplitPosition = Center[OutSplitAxis];
167 }

169 // Computes an axis aligned split plane (axis + value) by using the variance of the centroid.
170 // The axis with the largest variance on the centroid is picked as the split axis.
171 // The average of the centroids on the split axis is picked as the split value.
172 template <typename ElementType>
173 void ComputeSplitPlaneWithMedianVarianceHeuristic(const TArrayView<const ElementType>& Elements, int32& OutSplitAxis, FReal& OutSplitPosition)
174 {
175 FVec3 AverageCenter = FVec3::Zero();
176 FVec3 ScaledCenterVariance = FVec3::Zero();
177
178 int32 Count = 0;
179 ComputeScaledVariance(Elements, Count, AverageCenter, ScaledCenterVariance);
180
181 // Technically the variance is (ScaledCenterVariance/Num) and the sample variance is (ScaledCenterVariance / (Num - 1)).
182 // Since we're only picking largest, we can ignore the constant.
183 OutSplitAxis = PickLargestAxis(ScaledCenterVariance);
184 OutSplitPosition = AverageCenter[OutSplitAxis];
185 }

187 // Returns the bin index that should be used to split the data into two sets such that: left < index <= right
188 template <typename ElementType>
189 int32 ComputeSplitPlaneWithSurfaceAreaHeuristic(TArrayView<const ElementType>& Elements, const int32 BinCount, TArrayView<int32>& BinIndices)
190 {
191 FAABB3 CentroidAabb = FAABB3::EmptyAABB();
192 ComputeCentroidAabb(Elements, CentroidAabb);
193
194 const int32 SplitAxis = CentroidAabb.LargestAxis();
195
196 TArray<FAabbBin> Bins;
197 Bins.SetNum(BinCount);
198
199 FReal BinLowerBound;
200 FReal BinBoundsInvSize;
201 ComputeBoundsForBins(CentroidAabb, SplitAxis, BinLowerBound, BinBoundsInvSize);
202 ComputeBins(Elements, SplitAxis, BinLowerBound, BinBoundsInvSize, Bins, BinIndices);
203
204 TArray<FBinSplitPlane> BinSplitPlanes;
205 ComputeBinSplitPlanes(Bins, BinSplitPlanes);
206 const int32 SplitPlaneIndex = PickBestBinSplitPlane(BinSplitPlanes);
207 // If we decide that the best split plane is 2, then we have:
208 // [0, N - 1], [1, N - 2], [2, N - 3], [3, N - 4], ...
209 // Split plane 2 is [2, N - 3]. This should return the first bin index such that left < index <= right, so the result is SplitPlaneIndex(2) + 1.
210 return SplitPlaneIndex + 1;
211 }
