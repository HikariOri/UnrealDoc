246// FNetNameMapping

248// Map from a name to the number of times it's been 'created' (identical nodes create the same variable names, so they need something appended)
249struct FNetNameMapping
250{
251public:
252 // Come up with a valid, unique (within the scope of NetNameMap) name based on an existing Net object and (optional) context.
253 // The resulting name is stable across multiple calls if given the same pointer.
254 FString MakeValidName(const UEdGraphNode* Net, const FString& Context = TEXT("")) { return MakeValidNameImpl(Net, Context); }
255 FString MakeValidName(const UEdGraphPin* Net, const FString& Context = TEXT("")) { return MakeValidNameImpl(Net, Context); }
256 FString MakeValidName(const UObject* Net, const FString& Context = TEXT("")) { return MakeValidNameImpl(Net, Context); }
257
258private:
259 KISMETCOMPILER_API static FString MakeBaseName(const UEdGraphNode* Net);
260 KISMETCOMPILER_API static FString MakeBaseName(const UEdGraphPin* Net);
261 KISMETCOMPILER_API static FString MakeBaseName(const UObject* Net);
262
263 template< typename NetType >
264 FString MakeValidNameImpl(NetType Net, const FString& Context)
265 {
266 // Check to see if this net was already used to generate a name
267 if (FString* Result = NetToName.Find(Net))
268 {
269 return *Result;
270 }
271 else
272 {
273 FString BaseName = MakeBaseName(Net);
274 if (!Context.IsEmpty())
275 {
276 BaseName += FString::Printf(TEXT("_%s"), *Context);
277 }
278
279 FString NetName = GetUniqueName(MoveTemp(BaseName));
280
281 NetToName.Add(Net, NetName);
282 NameToNet.Add(NetName, Net);
283 return NetName;
284 }
285 }
286
287 FString GetUniqueName(FString NetName)
288 {
289 FNodeHandlingFunctor::SanitizeName(NetName);
290 // Scratchpad so that we can find a new postfix:
291 FString NewNetName(NetName);
292
293 int32 Postfix = 0;
294 const void** ExistingNet = NameToNet.Find(NewNetName);
295 while(ExistingNet && *ExistingNet)
296 {
297 ++Postfix;
298 // Add an integer to the base name and check if it's free:
299 NewNetName = NetName + TEXT("_") + FString::FromInt(Postfix);
300 ExistingNet = NameToNet.Find(NewNetName);
301 }
302 return NewNetName;
303 }
304
305 TMap<const void*, FString> NetToName;
306 TMap<FString, const void*> NameToNet;
307};
