189 void GetInterpolatedTriangleTangent(int32 TriangleID,
190 const UE::Math::TVector<RealType>& BaryCoords,
191 UE::Math::TVector<RealType>& TangentOut,
192 UE::Math::TVector<RealType>& BitangentOut) const
193 {
194 int32 k = TriangleID * 3;
195 if (k >= 0 && (k+2) < Tangents.Num())
196 {
197 TangentOut = Blend3(Tangents[k], Tangents[k+1], Tangents[k+2], BaryCoords.X, BaryCoords.Y, BaryCoords.Z);
198 Normalize(TangentOut);
199 BitangentOut = Blend3(Bitangents[k], Bitangents[k+1], Bitangents[k+2], BaryCoords.X, BaryCoords.Y, BaryCoords.Z);
200 Normalize(BitangentOut);
201 }
202 else
203 {
204 TangentOut = TVector<RealType>::UnitX();
205 BitangentOut = TVector<RealType>::UnitY();
206 }

266 static bool ComputeDefaultOverlayTangents(FDynamicMesh3& Mesh)
267 {
268 const FDynamicMeshAttributeSet* Attributes = Mesh.Attributes();
269 if (!Attributes)
270 {
271 return false;
272 }
273 const FDynamicMeshUVOverlay* UVOverlay = Attributes->PrimaryUV();
274 const FDynamicMeshNormalOverlay* NormalOverlay = Attributes->PrimaryNormals();
275 if (UVOverlay && NormalOverlay)
276 {
277 TMeshTangents Tangents;
278 Tangents.SetMesh(&Mesh);
279 Tangents.ComputeTriVertexTangents(NormalOverlay, UVOverlay, FComputeTangentsOptions());
280 Mesh.Attributes()->EnableTangents();
281 return Tangents.CopyToOverlays(Mesh);
282 }
283 return false;
284 }

305template<typename RealType>
306template<typename OtherRealType>
307void TMeshTangents<RealType>::CopyTriVertexTangents(const TMeshTangents<OtherRealType>& OtherMeshTangents)
308{
309 const TArray<TVector<OtherRealType>>& OtherTangents = OtherMeshTangents.GetTangents();
310 const TArray<TVector<OtherRealType>>& OtherBitangents = OtherMeshTangents.GetBitangents();
311
312 int32 Num = OtherTangents.Num();
313 check(Mesh->MaxTriangleID() * 3 == Num);
314 InitializeTriVertexTangents(false);
315 for (int32 k = 0; k < Num; ++k)
316 {
317 Tangents[k] = TVector<RealType>(OtherTangents[k]);
318 Bitangents[k] = TVector<RealType>(OtherBitangents[k]);
319 }
320 AllDegenerateTris = OtherMeshTangents.GetDegenerateTris();
321}
