178template <typename AttributeType>
179void TMeshAttributeArrayBase<AttributeType>::Insert(const int32 Index, const AttributeType& Default)
180{
181 int32 EndIndex = (Index + 1) * Extent;
182 if (EndIndex > Container.Num())
183 {
184 // If the index is off the end of the container, add as many elements as required to make it the last valid index.
185 int32 StartIndex = Container.AddUninitialized(EndIndex - Container.Num());
186 AttributeType* Data = Container.GetData() + StartIndex;
187
188 // Construct added elements with the default value passed in
189
190 while (StartIndex < EndIndex)
191 {
192 new(Data) AttributeType(Default);
193 StartIndex++;
194 Data++;
195 }
196 }
197}

199template <typename AttributeType>
200void TMeshAttributeArrayBase<AttributeType>::Remap(const TSparseArray<int32>& IndexRemap, const AttributeType& Default)
201{
202 TMeshAttributeArrayBase NewAttributeArray(Extent);
203
204 for (typename TSparseArray<int32>::TConstIterator It(IndexRemap); It; ++It)
205 {
206 const int32 OldElementIndex = It.GetIndex();
207 const int32 NewElementIndex = IndexRemap[OldElementIndex];
208
209 NewAttributeArray.Insert(NewElementIndex, Default);
210 AttributeType* DestElementBase = NewAttributeArray.GetElementBase(NewElementIndex);
211 AttributeType* SrcElementBase = GetElementBase(OldElementIndex);
212 for (uint32 Index = 0; Index < Extent; ++Index)
213 {
214 DestElementBase[Index] = MoveTemp(SrcElementBase[Index]);
215 }
216 }
217
218 Container = MoveTemp(NewAttributeArray.Container);
219}

221template <typename T>
222inline typename TEnableIf<!TIsBulkSerializable<T>::Value, FArchive>::Type& operator<<( FArchive& Ar, TMeshAttributeArrayBase<T>& Array )
223{
224 if (Ar.IsLoading() &&
225 Ar.CustomVer(FReleaseObjectVersion::GUID) != FReleaseObjectVersion::MeshDescriptionNewFormat &&
226 Ar.CustomVer(FUE5MainStreamObjectVersion::GUID) < FUE5MainStreamObjectVersion::MeshDescriptionNewFormat)
227 {
228 Array.Extent = 1;
229 }
230 else
231 {
232 Ar << Array.Extent;
233 }
234
235 // A little bit prior to skeletal meshes storing their model data as mesh description, FTransform attributes were added but marked,
236 // by default, as bulk-serializable, even though FTransform doesn't support it. Therefore, this serialization path only worked on empty
237 // FTransform attributes. However, there are still static mesh assets in the wild that contain empty FTransform attributes, and we need
238 // to be able to successfully load them -- hence this check.
239 if constexpr (std::is_same_v<T, FTransform>)
240 {
241 if (Ar.IsLoading())
242 {
243 // This version check works because saved UStaticMesh assets set this on their archive, which is then inherited by the
244 // mesh description bulk storage.
245 const FCustomVersion* PossiblySavedVersion = Ar.GetCustomVersions().GetVersion(FFortniteMainBranchObjectVersion::GUID);
246 if (PossiblySavedVersion && PossiblySavedVersion->Version < FFortniteMainBranchObjectVersion::MeshDescriptionForSkeletalMesh)
247 {
248 Array.Container.BulkSerialize( Ar );
249 return Ar;
250 }
251 }
252 }
253
254 // Serialize types which aren't bulk serializable, which need to be serialized element-by-element
255 Ar << Array.Container;
256 return Ar;
257}

259template <>
260inline TEnableIf<!TIsBulkSerializable<FName>::Value, FArchive>::Type& operator<<( FArchive& Ar, TMeshAttributeArrayBase<FName>& Array )
261{
262 if (Ar.CustomVer(FFortniteMainBranchObjectVersion::GUID) < FFortniteMainBranchObjectVersion::DynamicMeshAttributesTriangleLabels)
263 {
264 // see non-specialized implementation
265 if (Ar.IsLoading() &&
266 Ar.CustomVer(FReleaseObjectVersion::GUID) != FReleaseObjectVersion::MeshDescriptionNewFormat &&
267 Ar.CustomVer(FUE5MainStreamObjectVersion::GUID) < FUE5MainStreamObjectVersion::MeshDescriptionNewFormat)
268 {
269 Array.Extent = 1;
270 }
271 else
272 {
273 Ar << Array.Extent;
274 }
275
276 Ar << Array.Container;
277
278 return Ar;
279 }
280
281 Ar << Array.Extent;
282
283 enum class EIndexSize : uint8 { U8 = 0, U16 = 1, U32 = 2 };
284
285 TArray<FName>& Names = Array.Container;
286
287 if (Ar.IsLoading())
288 {
289 // read names count
290 uint32 NumNames = 0;
291 Ar << NumNames;
292
293 constexpr uint32 ArrayMax = static_cast<uint32>(std::numeric_limits<TArray<FName>::SizeType>::max());
294 if (!ensure(NumNames <= ArrayMax))
295 {
296 Ar.SetError();
297 return Ar;
298 }
299
300 if (NumNames == 0)
301 {
302 Names.Reset();
303 return Ar;
304 }
305
306 // ret unique names
307 uint32 NumUniqueNames = 0;
308 Ar << NumUniqueNames;
309
310 if (!ensure(NumUniqueNames <= ArrayMax))
311 {
312 Ar.SetError();
313 return Ar;
314 }
315
316 TArray<FName> UniqueNames;
317 UniqueNames.Reserve(NumUniqueNames);
318
319 for (uint32 Index = 0; Index < NumUniqueNames; ++Index)
320 {
321 FString NameStr;
322 Ar << NameStr;
323 UniqueNames.Add(FName(NameStr));
324 }
325
326 // single FName so assign the same for all triangles
327 if (NumUniqueNames == 1)
328 {
329 Names.Init(UniqueNames[0], NumNames);
330 return Ar;
331 }
332
333 // all names are unique
334 if (NumUniqueNames == NumNames)
335 {
336 Names = MoveTemp(UniqueNames);
337 return Ar;
338 }
339
340 Names.Init(NAME_None, NumNames);
341
342 // read index size
343 EIndexSize IndexSize = EIndexSize::U8;
344 Ar << IndexSize;
345
346 // read indices relative to UniqueNames
347 for (uint32 Index = 0; Index < NumNames; ++Index)
348 {
349 uint32 NameIndex = 0;
350 if (IndexSize == EIndexSize::U8)
351 {
352 uint8 v = 0;
353 Ar << v;
354 NameIndex = v;
355 }
356 else if (IndexSize == EIndexSize::U16)
357 {
358 uint16 v = 0;
359 Ar << v;
360 NameIndex = v;
361 }
362 else
363 {
364 Ar << NameIndex;
365 }
366
367 if (ensure(UniqueNames.IsValidIndex(NameIndex)))
368 {
369 Names[Index] = UniqueNames[NameIndex];
370 }
371 }
372 }
373 else
374 {
375 // write names count
376 uint32 NumNames = Names.Num();
377 Ar << NumNames;
378
379 if (NumNames == 0)
380 {
381 return Ar;
382 }
383
384 // get unique names
385 TMap<FName, uint32> NameToIndex;
386 TArray<FString> UniqueNames;
387 TArray<uint32> NameIndices;
388 NameIndices.Init(0, NumNames);
389
390 for (uint32 Index = 0; Index < NumNames; ++Index)
391 {
392 const FName Name = Names[Index];
393 if (const uint32* Found = NameToIndex.Find(Name))
394 {
395 NameIndices[Index] = *Found;
396 }
397 else
398 {
399 NameIndices[Index] = UniqueNames.Num();
400 NameToIndex.Add(Name, UniqueNames.Num());
401 UniqueNames.Emplace(Name.ToString());
402 }
403 }
404
405 // write unique count and names
406 uint32 NumUniqueNames = UniqueNames.Num();
407 Ar << NumUniqueNames;
408
409 for (uint32 Index = 0; Index < NumUniqueNames; ++Index)
410 {
411 Ar << UniqueNames[Index];
412 }
413
414 // single FName or all names are unique so nothing else to do
415 if (NumUniqueNames == 1 || NumUniqueNames == NumNames)
416 {
417 return Ar;
418 }
419
420 // choose a convenient index size and write it
421 EIndexSize IndexSize = (NumUniqueNames <= MAX_uint8) ? EIndexSize::U8 : (NumUniqueNames <= MAX_uint16) ? EIndexSize::U16 : EIndexSize::U32;
422 Ar << IndexSize;
423
424 // write indices relative to UniqueNames
425 for (uint32 Index = 0; Index < NumNames; ++Index)
426 {
427 if (IndexSize == EIndexSize::U8)
428 {
429 uint8 v = static_cast<uint8>(NameIndices[Index]);
430 Ar << v;
431 }
432 else if (IndexSize == EIndexSize::U16)
433 {
434 uint16 v = static_cast<uint16>(NameIndices[Index]);
435 Ar << v;
436 }
437 else
438 {
439 Ar << NameIndices[Index];
440 }
441 }
442 }
443
444 return Ar;
445}
