57 void UpdateColumns()
58 {
59 ClearColumns();
60
61 const TArray< TSharedRef< IPropertyTableColumn > > PropertyTableColumns = PropertyTable->GetColumns();
62
63 ColumnHeaders.Empty();
64 const TSharedRef< IPropertyTable > TableRef = PropertyTable.ToSharedRef();
65
66 for( auto ColumnIter = PropertyTableColumns.CreateConstIterator(); ColumnIter; ++ColumnIter )
67 {
68 const TSharedRef< IPropertyTableColumn > Column = *ColumnIter;
69
70 if ( !ColumnFactory->Supports( Column ) )
71 {
72 Column->SetHidden( true );
73 continue;
74 }
75
76 Column->SetHidden( false );
77 TSharedRef< SColumnHeader > ColumnHeader = ConstructColumnHeader( Column, TableRef, Style );
78 const FName ColumnId = Column->GetId();
79 ColumnHeaders.Add( ColumnId, ColumnHeader );
80
81 if ( Column->GetSizeMode() == EPropertyTableColumnSizeMode::Fixed )
82 {
83 AddColumn(
84 SHeaderRow::Column( ColumnId )
85 .SortMode( TableRef, &IPropertyTable::GetColumnSortMode, Column )
86 .HeaderComboVisibility(EHeaderComboVisibility::Never)
87 .SortPriority( TableRef, &IPropertyTable::GetColumnSortPriority, Column)
88 .OnSort( TableRef, &IPropertyTable::SortByColumnWithId )
89 .MenuContent()
90 [
91 GenerateColumnMenu( Column )
92 ]
93 .FixedWidth( Column->GetWidth() )
94 [
95 ColumnHeader
96 ]
97 );
98 }
99 else
100 {
101 TAttribute< EColumnSortMode::Type > SortMode = EColumnSortMode::None;
102 TAttribute< EColumnSortPriority::Type > SortPriority = EColumnSortPriority::Max;
103 FOnSortModeChanged OnSort;
104
105 if ( Column->CanSortBy() )
106 {
107 SortMode = TAttribute< EColumnSortMode::Type >::Create( TAttribute< EColumnSortMode::Type >::FGetter::CreateSP( TableRef, &IPropertyTable::GetColumnSortMode, Column ) );
108 SortPriority = TAttribute< EColumnSortPriority::Type >::Create(TAttribute< EColumnSortPriority::Type >::FGetter::CreateSP(TableRef, &IPropertyTable::GetColumnSortPriority, Column ) );
109 OnSort = FOnSortModeChanged::CreateSP( TableRef, &IPropertyTable::SortByColumnWithId );
110 }
111
112 AddColumn(
113 SHeaderRow::Column( ColumnId )
114 .SortMode( SortMode )
115 .HeaderComboVisibility(EHeaderComboVisibility::Never)
116 .SortPriority( SortPriority )
117 .OnSort( OnSort )
118 .MenuContent()
119 [
120 GenerateColumnMenu( Column )
121 ]
122 .FillWidth( Column, &IPropertyTableColumn::GetWidth )
123 .OnWidthChanged( Column, &IPropertyTableColumn::SetWidth )
124 [
125 ColumnHeader
126 ]
127 );
128 }
129
130 Column->OnFrozenStateChanged()->RemoveAll(this);
131 Column->OnFrozenStateChanged()->AddSP(this, &SPropertyTableHeaderRow::RegenerateColumnMenu);
132 }
133 }

150 TSharedRef< SColumnHeader > ConstructColumnHeader( const TSharedRef < class IPropertyTableColumn >& Column, const TSharedRef< class IPropertyTableUtilities >& Utilities, const FName& InStyle )
151 {
152 TSharedPtr< IPropertyTableCustomColumn > Customization;
153
154 for( auto CustomizationIter = Customizations.CreateIterator(); CustomizationIter; ++CustomizationIter )
155 {
156 auto PotentialCustomization = *CustomizationIter;
157 if ( PotentialCustomization->Supports( Column, Utilities ) )
158 {
159 Customization = PotentialCustomization;
160 break;
161 }
162 }
163
164 return ColumnFactory->CreateColumnHeaderWidget( Column, Utilities, Customization, InStyle );

167 TSharedRef< SWidget > GenerateColumnMenu( const TSharedRef< IPropertyTableColumn >& Column )
168 {
169 if ( Column->IsFrozen() )
170 {
171 return SNullWidget::NullWidget;
172 }
173
174 // Name column drop down menu
175 const bool bShouldCloseWindowAfterMenuSelection = true;
176 FMenuBuilder MenuBuilder( bShouldCloseWindowAfterMenuSelection, NULL );
177 {
178 {
179 FUIAction Action(
180 FExecuteAction::CreateSP(SharedThis(this), &SPropertyTableHeaderRow::RemoveColumn, Column),
181 FCanExecuteAction()
182 );
183
184 MenuBuilder.AddMenuEntry( LOCTEXT("RemoveColumn_Name", "Remove"), LOCTEXT("RemoveColumn_ToolTip", "Removes the column from the table view"), FSlateIcon(), Action, NAME_None, EUserInterfaceActionType::Button );
185 }
186
187 {
188 FUIAction Action(
189 FExecuteAction::CreateSP(SharedThis(this), &SPropertyTableHeaderRow::RemoveAllButThisColumn, Column),
190 FCanExecuteAction()
191 );
192
193 MenuBuilder.AddMenuEntry(LOCTEXT("RemoveAllButThisColumn_Name", "Remove All But This"), LOCTEXT("RemoveAllButThisColumn_ToolTip", "Removes all the columns from the table view except this"), FSlateIcon(), Action, NAME_None, EUserInterfaceActionType::Button);
194 }
195 }
196
197 return MenuBuilder.MakeWidget();
198 }

204 UpdateColumns();

212 void RemoveAllButThisColumn(const TSharedRef<IPropertyTableColumn> ColumnToKeep)
213 {
214 TArray<TSharedRef<IPropertyTableColumn>> ExistingColumns = PropertyTable->GetColumns();
215 for (TSharedRef<IPropertyTableColumn> ExistingColumn : ExistingColumns)
216 {
217 if (ExistingColumn->IsFrozen() || ExistingColumn == ColumnToKeep)
218 {
219 continue;
220 }
221
222 PropertyTable->RemoveColumn(ExistingColumn);
223 }
224 }
