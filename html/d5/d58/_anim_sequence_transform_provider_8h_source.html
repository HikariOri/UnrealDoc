152 void Init(uint32 InNumAnimBones)
153 {
154 NumAnimBones = static_cast<uint16>(InNumAnimBones);
155 UsedAnimBones.Init(false, InNumAnimBones);
156 AnimBoneToAnimCompact.SetNumUninitialized(InNumAnimBones);
157 FMemory::Memset(AnimBoneToAnimCompact.GetData(), 0xFF, InNumAnimBones * sizeof(uint16));
158 }
159
160 void Rebuild()
161 {
162 AnimCompactToAnimBone.Reset();
163 FMemory::Memset(AnimBoneToAnimCompact.GetData(), 0xFF, AnimBoneToAnimCompact.Num() * sizeof(uint16));
164
165 for (TConstSetBitIterator<> It(UsedAnimBones); It; ++It)
166 {
167 const uint16 AnimBone = static_cast<uint16>(It.GetIndex());
168 AnimBoneToAnimCompact[AnimBone] = static_cast<uint16>(AnimCompactToAnimBone.Num());
169 AnimCompactToAnimBone.Add(AnimBone);
170 }
171
172 NumAnimCompactBones = static_cast<uint16>(AnimCompactToAnimBone.Num());
173 }
