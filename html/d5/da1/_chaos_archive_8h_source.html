88 template <typename T, ESPMode Mode>
89 TSharedPtr<T, Mode>& ToSharedPointerHelper(TSerializablePtr<T>& Obj)
90 {
91 T* RawPtr = const_cast<T*>(Obj.Get());
92 if (FSharedPtrHolder* BaseHolder = ObjToSharedPtrHolder.FindRef(RawPtr))
93 {
94 auto ConcreteHolder = static_cast<TSharedPtrHolder<T, Mode>*>(BaseHolder);
95 return ConcreteHolder->SP;
96 }
97 else
98 {
99 auto NewHolder = new TSharedPtrHolder<T, Mode>(RawPtr);
100 TSharedPtr<T, Mode>& NewSP = NewHolder->SP;
101 ObjToSharedPtrHolder.Add((void*)RawPtr, NewHolder);
102 return NewSP;
103 }
104 }

186 template <typename T>
187 void SerializePtr(TSerializablePtr<T>& Obj)
188 {
189 bool bExists = Obj.Get() != nullptr;
190 InnerArchive << bExists;
191 if (!bExists)
192 {
193 Obj.Reset();
194 return;
195 }
196
197 if (InnerArchive.IsLoading())
198 {
199 int32 Tag;
200 InnerArchive << Tag;
201
202 if (Tag < 0)
203 {
204 InnerArchive.SetCriticalError();
205 return;
206 }
207
208 const int32 SlotsNeeded = Tag + 1 - Context->TagToObject.Num();
209 if (SlotsNeeded > 0)
210 {
211 Context->TagToObject.AddZeroed(SlotsNeeded);
212 }
213
214 if (!Context->TagToObject.IsValidIndex(Tag))
215 {
216 InnerArchive.SetCriticalError();
217 return;
218 }
219
220 if (Context->TagToObject[Tag])
221 {
222 Obj.SetFromRawLowLevel((const T*)(Context->TagToObject[Tag]));
223 }
224 else
225 {
226 StaticSerialize(Obj);
227 Context->TagToObject[Tag] = (void*)Obj.Get();
228 Context->ObjToTag.Add((void*)Obj.Get(), Tag);
229 }
230 }
231 else if (InnerArchive.IsSaving() || InnerArchive.IsCountingMemory())
232 {
233 void* ObjRaw = (void*)Obj.Get();
234 check(Context->PendingAdds.Contains(ObjRaw) == false); //catch dependency cycles. Not supported
235 if (Context->ObjToTag.Contains(ObjRaw))
236 {
237 InnerArchive << Context->ObjToTag[ObjRaw];
238 }
239 else
240 {
241 Context->PendingAdds.Add(ObjRaw);
242
243 int32 Tag = Context->TagCount++;
244 Context->ObjToTag.Add(ObjRaw, Tag);
245
246 InnerArchive << Tag;
247 StaticSerialize(Obj);
248
249 Context->PendingAdds.Remove(ObjRaw);
250 }
251 }
252 }

350 {
351 T* RawPtr = const_cast<T*>(Serializable.Get());
352 FChaosArchive& Ar = *this;

354 {
355 check(Ar.IsLoading()); //SerializationFactory should only create objects on load

357 Serializable.SetFromRawLowLevel(RawPtr);
358 }
359 else

361 check(!Ar.IsLoading()) //SerializationFactory must construct new object on load
362 }

364 RawPtr->Serialize(Ar);
