269 void AddToOpenLog(const TCHAR* Filename)
270 {
271 CriticalSection.Lock();
272 if (bLogDuplicates)
273 {
274 FString Text = FString::Printf(TEXT("\"%s\"\n"), Filename);
275 for (auto File = LogOutput.CreateIterator(); File; ++File)
276 {
277 (*File)->Write((uint8*)StringCast<ANSICHAR>(*Text).Get(), Text.Len());
278 }
279 }
280 else if (FilenameAccessMap.Find(Filename) == nullptr)
281 {
282 FilenameAccessMap.Emplace(Filename, ++OpenOrder);
283 FString Text = FString::Printf(TEXT("\"%s\" %llu\n"), Filename, OpenOrder);
284 for (auto File = LogOutput.CreateIterator(); File; ++File)
285 {
286 (*File)->Write((uint8*)StringCast<ANSICHAR>(*Text).Get(), Text.Len());
287 }
288 }
289 CriticalSection.Unlock();
290 }
291
292 void AddLabelToOpenLog(const TCHAR* LabelStr)
293 {
294 CriticalSection.Lock();
295 for (auto File = LogOutput.CreateIterator(); File; ++File)
296 {
297 FString Text = FString::Printf(TEXT("# %s\n"), LabelStr);
298 (*File)->Write((uint8*)StringCast<ANSICHAR>(*Text).Get(), Text.Len());
299 }
300 CriticalSection.Unlock();
301 }
