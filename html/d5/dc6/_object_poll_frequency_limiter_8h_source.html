64inline void FObjectPollFrequencyLimiter::SetUpdateFrequency(float InUpdateFrequency)
65{
66 if (InUpdateFrequency <= 0.0f)
67 {
68 // 0 frequency seems incorrect and would cause division by zero.
69 ensureMsgf(false, TEXT("ObjectPollFrequencyLimiter update frequency must be greater than zero"));
70 InUpdateFrequency = 1.0f;
71 }
72
73 if (InUpdateFrequency != UpdateFrequency)
74 {
75 UpdateFrequency = InUpdateFrequency;
76 ReinitFramesBetweenUpdates();
77 }
78}

91inline void FObjectPollFrequencyLimiter::SetPollFramePeriod(FInternalNetRefIndex InternalIndex, uint8 PollFramePeriod)
92{
93 MaxInternalIndex = FPlatformMath::Max(MaxInternalIndex, InternalIndex);
94
95 FramesBetweenUpdates[InternalIndex] = PollFramePeriod;
96 // Spread the polling of objects with the same frequency so that if you add lots of objects the same frame they won't be polled at the same time. The update loop decrements counters so we need to be careful with how we offset things.
97 const uint8 FrameOffset = --FrameIndexOffsets[PollFramePeriod];
98 FrameCounters[InternalIndex] = static_cast<uint8>(uint32(~(FrameIndex + FrameOffset)) % uint32(PollFramePeriod + 1U));
99}

101inline void FObjectPollFrequencyLimiter::SetPollWithObject(FInternalNetRefIndex ObjectToPollWithInternalIndex, FInternalNetRefIndex InternalIndex)
102{
103 MaxInternalIndex = FPlatformMath::Max(MaxInternalIndex, InternalIndex);
104
105 // Copy state from object to poll with
106 FramesBetweenUpdates[InternalIndex] = FramesBetweenUpdates[ObjectToPollWithInternalIndex];
107 FrameCounters[InternalIndex] = FrameCounters[ObjectToPollWithInternalIndex];
108
109 // This is really only somewhat useful if this was to be called for a root object rather than a subobject since we will call this method for subobjects in ReinitFramesBetweenUpdates.
110 // For other root objects the SetPollWithObject would have to be called manually when the update frequency changes. This call will only guarantee that we poll as often as the object to poll with, but not necessarily on the same frame.
111 Frequencies[InternalIndex] = Frequencies[ObjectToPollWithInternalIndex];
112}

114inline uint8 FObjectPollFrequencyLimiter::ConvertFrequencyToFramesBetweenUpdates(float Frequency)
115{
116 if (Frequency <= 0.0f)
117 {
118 return 0U;
119 }
120
121 uint32 FramesBetweenUpdatesForObject = static_cast<uint32>(UpdateFrequency/FPlatformMath::Max(0.001f, Frequency));
122 if (FramesBetweenUpdatesForObject > 0)
123 {
124 FramesBetweenUpdatesForObject--;
125 }
126
127 return static_cast<uint8>(FMath::Clamp<uint32>(FramesBetweenUpdatesForObject, 0U, 255U));
128}
