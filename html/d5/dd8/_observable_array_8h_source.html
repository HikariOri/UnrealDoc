332 template <UE::CIntegral CountType>
333 void RemoveAt(SizeType Index, CountType NumToRemove, EAllowShrinking AllowShrinking = EAllowShrinking::Default)
334 {
335 static_assert(!std::is_same_v<CountType, bool>, "TObservableArray::RemoveAt: unexpected bool passed as the Count argument");
336 check((NumToRemove > 0) & (Index >= 0) & (Index + NumToRemove <= Num()));
337 if (NumToRemove > 0)
338 {
339 if (NumToRemove == 1)
340 {
341 ElementType RemovedElement = MoveTemp(Array[Index]);
342 Array.RemoveAt(Index, NumToRemove, AllowShrinking);
343 ArrayChangedDelegate.Broadcast(ObservableArrayChangedArgsType::MakeRemoveAction({ &RemovedElement, 1 }, Index));
344 }
345 else
346 {
347 // Copy the items to a temporary array to call the delegate
348 FMemMark Mark(FMemStack::Get());
349 TArray<ElementType, TMemStackAllocator<>> RemovedElements;
350 RemovedElements.Reserve(NumToRemove);
351 for (SizeType RemoveIndex = 0; RemoveIndex < NumToRemove; ++RemoveIndex)
352 {
353 RemovedElements.Add(MoveTemp(Array[RemoveIndex + Index]));
354 }
355
356 Array.RemoveAt(Index, NumToRemove, AllowShrinking);
357 ArrayChangedDelegate.Broadcast(ObservableArrayChangedArgsType::MakeRemoveAction({ RemovedElements.GetData(), NumToRemove }, Index));
358 }
359 }
