39{
40 Invalid,

43
44 // Standard unary ops have three variations: normal, in place (input == output), and reuse (input == output == last output)
45 UnaryFirst = Abs,
46 UnaryNoInt = ACos, // First unary op that doesn't support integer types
47 UnaryCustom = Length, // First custom unary op with a single opcode variant instead of 3
48 UnaryLast = Length,
49
50 // Standard binary ops have seven variations: normal, in place (first input == output), reuse (first input == output == last output),
51 // plus four variations with inline constant second arguments (normal, in place, reuse, reuse with scalar constant).
52 BinaryFirst = Add,
53 BinaryNoInt = ATan2, // First binary op that doesn't support integer types
54 BinaryCustom = Cross, // First binary op with a single opcode variant instead of 7
55 BinaryLast = Dot,
56
57 TextureFirst = RuntimeVirtualTextureUniform,
58 TextureLast = TextureSize,
59};
