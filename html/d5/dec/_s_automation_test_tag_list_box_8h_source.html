160 if (AlreadyInNewTagsCaseInsensitive || (IsNew && AlreadyInExistingTagsCaseInsensitive))

211{
212 TSharedPtr<SWidget> TagListBoxWidget = MainBorder;
213
214 do
215 {
216 TagListBoxWidget = TagListBoxWidget->GetParentWidget();
217
218 if (TagListBoxWidget == nullptr)

221
222 return FReply::Handled();
223 }
224
225 } while (!TagListBoxWidget->GetTypeAsString().Equals("SAutomationTestTagListBox"));
226
227 const TSharedPtr<SAutomationTestTagListBox> TagListBox = StaticCastSharedPtr<SAutomationTestTagListBox>(TagListBoxWidget);
228
229 if (TagListBox->RemoveNewTagsItem(TagValue) == 0)
230 {
231 MarkedForDeletion = !MarkedForDeletion;
232 }
233
234 if (!IsNew)
235 {
236 MainBorder->SetBorderBackgroundColor(MarkedForDeletion ? FColorList::Red : FColorList::Black);
237 }
238 else
239 {
240 ChildSlot.DetachWidget();
241 }
242 return FReply::Handled();
243}
