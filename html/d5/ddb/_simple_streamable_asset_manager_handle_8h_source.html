66template<typename TagT>
67struct alignas(8) TSSAMAtomicHandle
68{
69public:
70 using HandleT = TSSAMHandle<TagT>;
71
72 TSSAMAtomicHandle() = default;
73
74 // A handle holder owns a single SSAM slot. Two object instances must never share it: a
75 // Release on either copy would silently invalidate the other. To enforce that:
76 // - Copy default-initializes the destination. The source is unchanged. Cloning a live
77 // handle is impossible; the duplicate behaves as a fresh, unregistered holder. This
78 // keeps containing classes (FPrimitiveSceneProxy, derived FDebugRenderSceneProxy and
79 // its 35+ subclasses) backward-compatible -- they can keep their `=default` copy ctors
80 // and the wrapper just no-ops the slot field on copy.
81 // - Move transfers the slot. Source becomes empty, destination receives the handle.
82 // Single-owner invariant preserved across the FPrimitiveSceneProxy move chain.
83 // UObject duplication (StaticDuplicateObject) uses serialization, not C++ copy, so the
84 // non-UPROPERTY handle field is naturally default-initialized on the duplicate either way.
85 TSSAMAtomicHandle(const TSSAMAtomicHandle& /*Other*/) noexcept
86 {
87 Storage.store(DefaultStorage, std::memory_order_relaxed);
88 }
89 TSSAMAtomicHandle& operator=(const TSSAMAtomicHandle& /*Other*/) noexcept
90 {
91 Storage.store(DefaultStorage, std::memory_order_relaxed);
92 return *this;
93 }
94 TSSAMAtomicHandle(TSSAMAtomicHandle&& Other) noexcept
95 {
96 const uint64 Bits = Other.Storage.exchange(DefaultStorage, std::memory_order_relaxed);
97 Storage.store(Bits, std::memory_order_relaxed);
98 }
99 TSSAMAtomicHandle& operator=(TSSAMAtomicHandle&& Other) noexcept
100 {
101 const uint64 Bits = Other.Storage.exchange(DefaultStorage, std::memory_order_relaxed);
102 Storage.store(Bits, std::memory_order_relaxed);
103 return *this;
104 }
105
106 HandleT Load(std::memory_order Order = std::memory_order_acquire) const
107 {
108 const uint64 Bits = Storage.load(Order);
109 return HandleT{ static_cast<uint32>(Bits & 0xFFFFFFFFu), static_cast<uint32>(Bits >> 32) };
110 }
111 void Store(HandleT Handle, std::memory_order Order = std::memory_order_release)
112 {
113 const uint64 Bits = static_cast<uint64>(Handle.SlotIndex)
114 | (static_cast<uint64>(Handle.Generation) << 32);
115 Storage.store(Bits, Order);
116 }
117 HandleT Exchange(HandleT Desired, std::memory_order Order = std::memory_order_acq_rel)
118 {
119 const uint64 Bits = static_cast<uint64>(Desired.SlotIndex)
120 | (static_cast<uint64>(Desired.Generation) << 32);
121 const uint64 Old = Storage.exchange(Bits, Order);
122 return HandleT{ static_cast<uint32>(Old & 0xFFFFFFFFu), static_cast<uint32>(Old >> 32) };
123 }
124
125 operator HandleT() const { return Load(); }
126 TSSAMAtomicHandle& operator=(HandleT Handle) { Store(Handle); return *this; }
127 bool operator==(HandleT Other) const { return Load() == Other; }
128 bool operator!=(HandleT Other) const { return Load() != Other; }
129
130 bool IsValid() const { return Load().IsValid(); }
131 bool IsStale() const { return Load().IsStale(); }
132 void Reset() { Store(HandleT{}); }
133
134private:
135 // Packed: low 32 bits = SlotIndex, high 32 bits = Generation. Default-constructed value
136 // matches a default-constructed TSSAMHandle (INVALID_SLOT, generation 0).
137 static constexpr uint64 DefaultStorage = static_cast<uint64>(HandleT::INVALID_SLOT);
138 std::atomic<uint64> Storage{ DefaultStorage };
139};

178template<typename HandleT>
179class TSSAMHandleTable
180{
181public:
182 HandleT Allocate()
183 {
184 UE::TUniqueLock Lock(Mutex);
185
186 uint32 SlotIndex;
187 bool bAllocatedFreshIndex = false;
188 if (FreeList.Num() > 0)
189 {
190 SlotIndex = FreeList.Pop(EAllowShrinking::No);
191 }
192 else
193 {
194 // Read under the mutex; no race with other writers. Defer publishing the new
195 // NextIndex via release-store until after the backing page is materialized so
196 // lock-free readers (GetIndex) can rely on the range check as a safety gate.
197 SlotIndex = NextIndex.load(std::memory_order_relaxed);
198 bAllocatedFreshIndex = true;
199 }
200
201 // Lazily materialize the slot's backing page + placement-new. Under the mutex there
202 // is only one materializer at a time, so plain (non-atomic) visibility within the
203 // critical section is sufficient.
204 while (NextMaterializedItem <= SlotIndex + 1)
205 {
206 SlotStorage.Alloc(1);
207 ++NextMaterializedItem;
208 }
209
210 FSSAMSlot& Slot = *SlotStorage.GetItem(SlotIndex + 1);
211 const uint32 Generation = Slot.Generation.load(std::memory_order_relaxed);
212 Slot.RegistrationIndex.store(INDEX_NONE, std::memory_order_relaxed);
213
214 if (bAllocatedFreshIndex)
215 {
216 // Release-store ordered after SlotStorage.Alloc above. Once a lock-free reader
217 // observes SlotIndex < NextIndex via acquire-load, the slot's backing page
218 // allocation is also visible -- GetItem is then guaranteed not to land on a
219 // nullptr page.
220 NextIndex.store(SlotIndex + 1, std::memory_order_release);
221 }
222 return HandleT{ SlotIndex, Generation };
223 }
224
233 int32 Release(HandleT SSAMHandle)
234 {
235 if (!SSAMHandle.IsValid()) return INDEX_NONE;
236 FSSAMSlot& Slot = *SlotStorage.GetItem(SSAMHandle.SlotIndex + 1); // +1: indexed allocator skips 0
237
238 // CAS generation to atomically guarantee only the holder of the live generation proceeds.
239 // The CAS runs outside the table mutex so concurrent Releases for different slots don't
240 // serialize on each other. If the CAS fails, the slot has already been recycled
241 // (possibly to a different handle) and we must not touch it.
242 uint32 Expected = SSAMHandle.Generation;
243 if (!Slot.Generation.compare_exchange_strong(Expected, Expected + 1, std::memory_order_acq_rel, std::memory_order_acquire))
244 {
245 return INDEX_NONE;
246 }
247
248 // Own the invalidation window between CAS (generation bumped) and the slot being
249 // returned to the free pool: no other SetIndex can succeed here (generation check
250 // fails), no other Release can win the CAS, and no Allocate can re-pick this slot
251 // (not in the free list yet). Atomic exchange captures the last live index in one op.
252 const int32 OldIndex = Slot.RegistrationIndex.exchange(INDEX_NONE, std::memory_order_acq_rel);
253
254 UE::TUniqueLock Lock(Mutex);
255 FreeList.Push(SSAMHandle.SlotIndex);
256 return OldIndex;
257 }
258
259 int32 GetIndex(HandleT SSAMHandle) const
260 {
261 if (!SSAMHandle.IsValid())
262 {
263 return INDEX_NONE;
264 }
265
266 // Defensive range check. Rejects garbage / forged / corrupted handles whose SlotIndex
267 // is outside the allocated range BEFORE any SlotStorage access. Pairs with the
268 // release-store on NextIndex in Allocate (slot's backing page is materialized before
269 // NextIndex is bumped), so observing SlotIndex < NextIndex via acquire-load guarantees
270 // the GetItem call below cannot land on an unallocated page.
271 if (SSAMHandle.SlotIndex >= NextIndex.load(std::memory_order_acquire))
272 {
273 return INDEX_NONE;
274 }
275
276 const FSSAMSlot* Slot = SlotStorage.GetItem(SSAMHandle.SlotIndex + 1);
277 if (!Slot)
278 {
279 return INDEX_NONE;
280 }
281
282 const uint32 CurrentGen = Slot->Generation.load(std::memory_order_acquire);
283 if (CurrentGen != SSAMHandle.Generation)
284 {
285 return INDEX_NONE;
286 }
287
288 return Slot->RegistrationIndex.load(std::memory_order_acquire);
289 }
290
294 void SetIndex(HandleT SSAMHandle, int32 NewIndex)
295 {
296 if (!SSAMHandle.IsValid())
297 {
298 return;
299 }
300
301 FSSAMSlot& Slot = *SlotStorage.GetItem(SSAMHandle.SlotIndex + 1);
302 const uint32 CurrentGen = Slot.Generation.load(std::memory_order_acquire);
303 if (CurrentGen != SSAMHandle.Generation)
304 {
305 return;
306 }
307
308 Slot.RegistrationIndex.store(NewIndex, std::memory_order_release);
309 }
310
311 bool IsValid(HandleT SSAMHandle) const
312 {
313 if (!SSAMHandle.IsValid())
314 {
315 return false;
316 }
317
318 const FSSAMSlot* Slot = SlotStorage.GetItem(SSAMHandle.SlotIndex + 1);
319 return Slot && Slot->Generation.load(std::memory_order_acquire) == SSAMHandle.Generation;
320 }
321
322 uint32 GetAllocatedSize() const
323 {
324 UE::TUniqueLock Lock(Mutex);
325 return static_cast<uint32>((NextIndex.load(std::memory_order_relaxed) - static_cast<uint32>(FreeList.Num())) * sizeof(FSSAMSlot));
326 }
327
328private:
329 static constexpr uint32 MaxSlots = 1u << (20 + WITH_EDITOR); // 1M handles per table (2M in editor)
330 static constexpr uint32 SlotsPerPage = 4096; // 256 pages * 4K slots
331
332 using FStorage = TLockFreeAllocOnceIndexedAllocator<FSSAMSlot, MaxSlots, SlotsPerPage>;
333
334 // SlotStorage is mutable because TLockFreeAllocOnceIndexedAllocator declares GetItem
335 // non-const even though it's semantically a read-only atomic load. Our const accessors
336 // (GetIndex, IsValid) need to call through it without forcing callers to hold a non-const
337 // table reference. Mutex is mutable so const accessors (GetAllocatedSize) can lock it.
338 mutable FStorage SlotStorage;
339 mutable UE::FMutex Mutex;
340 TArray<uint32> FreeList;
341 // Monotonic high-water mark of slot indices ever issued. Atomic so GetIndex (called
342 // lock-free, including from the render thread) can range-check incoming handles before
343 // touching SlotStorage. Updated only inside Allocate under Mutex with release-store
344 // after the slot's backing page has been materialized; readers acquire-load and gate
345 // on SlotIndex < NextIndex.
346 std::atomic<uint32> NextIndex{0};
347 uint32 NextMaterializedItem = 1; // indexed allocator skips 0
348};
