190 FText HandleDeviceIdText() const
191 {
192 if (DeviceService.IsValid())
193 {
194 ITargetDevicePtr Device = DeviceService->GetDevice();
195 if (Device.IsValid())
196 {
197 return FText::FromString(Device->GetId().ToString());
198 }
199 }
200
201 return LOCTEXT("UnknownValue", "<unknown>");
202 }
203
205 FText HandleDeviceNameText() const
206 {
207 if (DeviceService.IsValid())
208 {
209 FString DeviceName = DeviceService->GetDeviceName();
210
211 if (!DeviceName.IsEmpty())
212 {
213 return FText::FromString(DeviceName);
214 }
215 }
216
217 return LOCTEXT("UnknownValue", "<unknown>");
218 }
219
221 FText HandleIsDefaultText() const
222 {
223 if (DeviceService.IsValid())
224 {
225 ITargetDevicePtr TargetDevice = DeviceService->GetDevice();
226
227 if (TargetDevice.IsValid() && TargetDevice->IsDefault())
228 {
229 return LOCTEXT("YesText", "yes");
230 }
231
232 return LOCTEXT("NoText", "no");
233 }
234
235 return LOCTEXT("UnknownValue", "<unknown>");
236 }
237
239 FText HandleOperatingSystemText() const
240 {
241 if (DeviceService.IsValid())
242 {
243 ITargetDevicePtr TargetDevice = DeviceService->GetDevice();
244
245 if (TargetDevice.IsValid())
246 {
247 FString OSName = TargetDevice->GetOperatingSystemName();
248
249 if (!OSName.IsEmpty())
250 {
251 return FText::FromString(OSName);
252 }
253 }
254 }
255
256 return LOCTEXT("UnknownValue", "<unknown>");
257 }
258
260 const FSlateBrush* HandlePlatformIcon() const
261 {
262 if (DeviceService.IsValid())
263 {
264 const PlatformInfo::FTargetPlatformInfo* const PlatformInfo = PlatformInfo::FindPlatformInfo(DeviceService->GetDevicePlatformName());
265 if(PlatformInfo)
266 {
267 return FAppStyle::GetBrush(PlatformInfo->GetIconStyleName(EPlatformIconSize::XLarge));
268 }
269 }
270
271 return FStyleDefaults::GetNoBrush();
272 }
273
275 FText HandlePlatformNameText() const
276 {
277 if (DeviceService.IsValid())
278 {
279 FString PlatformName = DeviceService->GetDevicePlatformDisplayName();
280
281 if (!PlatformName.IsEmpty())
282 {
283 return FText::FromString(PlatformName);
284 }
285 }
286
287 return LOCTEXT("UnknownValue", "<unknown>");
288 }
289
291 FText HandleStatusText() const
292 {
293 if (DeviceService.IsValid())
294 {
295 ITargetDevicePtr TargetDevice = DeviceService->GetDevice();
296
297 if (TargetDevice.IsValid())
298 {
299 if (TargetDevice->IsConnected())
300 {
301 return LOCTEXT("StatusConnected", "Connected");
302 }
303
304 return LOCTEXT("StatusDisconnected", "Disconnected");
305 }
306
307 return LOCTEXT("StatusUnavailable", "Unavailable");
308 }
309
310 return FText::GetEmpty();
311 }
