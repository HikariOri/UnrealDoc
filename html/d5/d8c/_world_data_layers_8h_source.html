342template<class DataLayerInstanceType, typename ...CreationsArgs>
343DataLayerInstanceType* AWorldDataLayers::CreateDataLayer(CreationsArgs... InCreationArgs)
344{
345 UClass* ClassToUse = DataLayerInstanceType::StaticClass();
346 if constexpr (HasDataLayerInstanceClass<DataLayerInstanceType>::value)
347 ClassToUse = DataLayerInstanceType::GetDataLayerInstanceClass();
348 UPackage* ExternalPackage = nullptr;
349 UObject* OuterObject = this;
350 FName NewObjectName;
351
352 if (IsUsingExternalPackageDataLayerInstances())
353 {
354 check(SupportsExternalPackageDataLayerInstances());
355 const ULevel* Level = GetLevel();
356 const bool bIsTransient = (Level->IsInstancedLevel() && !Level->IsPersistentLevel()) || Level->GetPackage()->HasAnyPackageFlags(PKG_PlayInEditor);
357 check(!bIsTransient);
358
359 // We generate a globally unique name to avoid any potential clash of 2 users creating the same object
360 const FGuid NewObjectGuid = FGuid::NewGuid();
361 FString ShortName = DataLayerInstanceType::StaticClass()->GetName() + TEXT("_UID_") + NewObjectGuid.ToString(EGuidFormats::UniqueObjectGuid);
362 TStringBuilderWithBuffer<TCHAR, NAME_SIZE> GloballyUniqueObjectPath;
363 GloballyUniqueObjectPath += OuterObject->GetPathName();
364 GloballyUniqueObjectPath += TEXT(".");
365 GloballyUniqueObjectPath += ShortName;
366
367 ExternalPackage = FExternalPackageHelper::CreateExternalPackage(OuterObject, *GloballyUniqueObjectPath, FExternalPackageHelper::GetDefaultExternalPackageFlags(), GetRootExternalDataLayerAsset());
368 check(ExternalPackage);
369 NewObjectName = *ShortName;
370 }
371 else
372 {
373 NewObjectName = DataLayerInstanceType::MakeName(Forward<CreationsArgs>(InCreationArgs)...);
374 }
375
376 DataLayerInstanceType* NewDataLayer = NewObject<DataLayerInstanceType>(OuterObject, ClassToUse, NewObjectName, RF_Transactional, nullptr, /*bCopyTransientsFromClassDefaults*/false, /*InstanceGraph*/nullptr, ExternalPackage);
377 check(NewDataLayer != nullptr);
378 NewDataLayer->OnCreated(Forward<CreationsArgs>(InCreationArgs)...);
379 AddDataLayerInstance(NewDataLayer);
380
381 ResolveActorDescContainers();
382 return NewDataLayer;
383}

385template<class IdentifierType>
386void AWorldDataLayers::OverwriteDataLayerRuntimeStates(const TArray<IdentifierType>* InActiveDataLayers, const TArray<IdentifierType>* InLoadedDataLayers)
387{
388 if (GetLocalRole() == ROLE_Authority)
389 {
390 // Handle Wrapper types
391 auto IdentifierToInstance = [this](const IdentifierType& Identifier) -> const UDataLayerInstance*
392 {
393 if constexpr (std::is_same_v<IdentifierType, TSoftObjectPtr<UDataLayerAsset>>)
394 {
395 return GetDataLayerInstance(Identifier.Get());
396 }
397 else
398 {
399 return GetDataLayerInstance(Identifier);
400 }
401 };
402
403 FlushNetDormancy();
404
405 // This should get called before game starts. It doesn't send out events
406 check(!GetWorld()->bMatchStarted);
407
408 if (InActiveDataLayers)
409 {
410 ActiveDataLayerNames.Empty(InActiveDataLayers->Num());
411 for (const IdentifierType& DataLayerIdentitier : *InActiveDataLayers)
412 {
413 const UDataLayerInstance* DataLayerInstance = IdentifierToInstance(DataLayerIdentitier);
414 if (DataLayerInstance && DataLayerInstance->IsRuntime())
415 {
416 ActiveDataLayerNames.Add(DataLayerInstance->GetDataLayerFName());
417 }
418 }
419 MARK_PROPERTY_DIRTY_FROM_NAME(AWorldDataLayers, RepActiveDataLayerNames, this);
420 RepActiveDataLayerNames = ActiveDataLayerNames.Array();
421 }
422
423
424 if(InLoadedDataLayers)
425 {
426 LoadedDataLayerNames.Empty(InLoadedDataLayers->Num());
427 for (const IdentifierType& DataLayerIdentitier : *InLoadedDataLayers)
428 {
429 const UDataLayerInstance* DataLayerInstance = IdentifierToInstance(DataLayerIdentitier);
430 if (DataLayerInstance && DataLayerInstance->IsRuntime())
431 {
432 LoadedDataLayerNames.Add(DataLayerInstance->GetDataLayerFName());
433 }
434 }
435 MARK_PROPERTY_DIRTY_FROM_NAME(AWorldDataLayers, RepLoadedDataLayerNames, this);
436 RepLoadedDataLayerNames = LoadedDataLayerNames.Array();
437 }
438
439 ForEachDataLayerInstance([this](class UDataLayerInstance* DataLayerInstance)
440 {
441 if (DataLayerInstance->IsRuntime())
442 {
443 ResolveEffectiveRuntimeState(DataLayerInstance, /*bNotifyChange*/false);
444 }
445 return true;
446 });
447
448 MARK_PROPERTY_DIRTY_FROM_NAME(AWorldDataLayers, RepEffectiveActiveDataLayerNames, this);
449 RepEffectiveActiveDataLayerNames = EffectiveStates.GetReplicatedEffectiveActiveDataLayerNames().Array();
450 MARK_PROPERTY_DIRTY_FROM_NAME(AWorldDataLayers, RepEffectiveLoadedDataLayerNames, this);
451 RepEffectiveLoadedDataLayerNames = EffectiveStates.GetReplicatedEffectiveLoadedDataLayerNames().Array();
452 }
453}
