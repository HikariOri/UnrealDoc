458 FReply OnKeyDown( const FGeometry& MyGeometry, const FKeyEvent& InKeyEvent ) override
459 {
460 if (IsInteractable())
461 {
462 const EUINavigationAction NavAction = FSlateApplication::Get().GetNavigationActionFromKey(InKeyEvent);
463 const EUINavigation NavDirection = FSlateApplication::Get().GetNavigationDirectionFromKey(InKeyEvent);
464 if (EnableGamepadNavigationMode)
465 {
466 // The controller's bottom face button must be pressed once to begin manipulating the combobox's value.
467 // Navigation away from the widget is prevented until the button has been pressed again or focus is lost.
468 if (NavAction == EUINavigationAction::Accept)
469 {
470 if (bControllerInputCaptured == false)
471 {
472 // Begin capturing controller input and open the ListView
473 bControllerInputCaptured = true;
474 PlayPressedSound();
475 OnComboBoxOpening.ExecuteIfBound();
476 return SComboButton::OnButtonClicked();
477 }
478 else
479 {
480 // Set selection to the selected item on the list and close
481 bControllerInputCaptured = false;
482
483 // Re-select first selected item, just in case it was selected by navigation previously
484 TArray<OptionType> SelectedItems = ComboListView->GetSelectedItems();
485 if (SelectedItems.Num() > 0)
486 {
487 OnSelectionChanged_Internal(SelectedItems[0], ESelectInfo::Direct);
488 }
489
490 // Set focus back to ComboBox
491 FReply Reply = FReply::Handled();
492 Reply.SetUserFocus(this->AsShared(), EFocusCause::SetDirectly);
493 return Reply;
494 }
495
496 }
497 else if (NavAction == EUINavigationAction::Back || InKeyEvent.GetKey() == EKeys::BackSpace)
498 {
499 const bool bWasInputCaptured = bControllerInputCaptured;
500
501 OnMenuOpenChanged(false);
502 if (bWasInputCaptured)
503 {
504 return FReply::Handled();
505 }
506 }
507 else
508 {
509 if (bControllerInputCaptured)
510 {
511 return FReply::Handled();
512 }
513 }
514 }
515 else
516 {
517 if (NavDirection == EUINavigation::Up)
518 {
519 NullableOptionType NullableSelected = GetSelectedItem();
520 if (TListTypeTraits<OptionType>::IsPtrValid(NullableSelected))
521 {
522 OptionType ActuallySelected = TListTypeTraits<OptionType>::NullableItemTypeConvertToItemType(NullableSelected);
523 const TArrayView<const OptionType> OptionsSource = ComboListView->GetItems();
524 const int32 SelectionIndex = OptionsSource.Find(ActuallySelected);
525 if (SelectionIndex >= 1)
526 {
527 // Select an item on the prev row
528 SetSelectedItem(OptionsSource[SelectionIndex - 1]);
529 }
530 }
531
532 return FReply::Handled();
533 }
534 else if (NavDirection == EUINavigation::Down)
535 {
536 NullableOptionType NullableSelected = GetSelectedItem();
537 if (TListTypeTraits<OptionType>::IsPtrValid(NullableSelected))
538 {
539 OptionType ActuallySelected = TListTypeTraits<OptionType>::NullableItemTypeConvertToItemType(NullableSelected);
540 const TArrayView<const OptionType> OptionsSource = ComboListView->GetItems();
541 const int32 SelectionIndex = OptionsSource.Find(ActuallySelected);
542 if (SelectionIndex < OptionsSource.Num() - 1)
543 {
544 // Select an item on the next row
545 SetSelectedItem(OptionsSource[SelectionIndex + 1]);
546 }
547 }
548 return FReply::Handled();
549 }
550
551 return SComboButton::OnKeyDown(MyGeometry, InKeyEvent);
552 }
553 }
554 return SWidget::OnKeyDown(MyGeometry, InKeyEvent);
555 }

591 //** Called if the menu is closed
592 void OnMenuOpenChanged(bool bOpen)
593 {
594 if (bOpen == false)
595 {
596 bControllerInputCaptured = false;
597
598 if (TListTypeTraits<OptionType>::IsPtrValid(SelectedItem))
599 {
600 // Ensure the ListView selection is set back to the last committed selection
601 OptionType ActuallySelected = TListTypeTraits<OptionType>::NullableItemTypeConvertToItemType(SelectedItem);

604 ComboListView->RequestScrollIntoView(ActuallySelected, 0);

610 TSharedRef<SWidget> ThisRef = this->AsShared();
611 if (User.IsWidgetInFocusPath(this->ComboListView))

621 void OnSelectionChanged_Internal( NullableOptionType ProposedSelection, ESelectInfo::Type SelectInfo )
622 {
623 // Ensure that the proposed selection is different
624 if(SelectInfo != ESelectInfo::OnNavigation)
625 {
626 // Ensure that the proposed selection is different from selected
627 if (bReselectionTriggersOnSelectionChanged || ProposedSelection != SelectedItem)
628 {
629 PlaySelectionChangeSound();
630 SelectedItem = ProposedSelection;
631 OnSelectionChanged.ExecuteIfBound( ProposedSelection, SelectInfo );
632 }
633 // close combo even if user reselected item
634 this->SetIsOpen( false );
635 }
