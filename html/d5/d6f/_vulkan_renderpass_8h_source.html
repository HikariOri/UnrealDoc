506template <typename TSubpassDescriptionClass, typename TSubpassDependencyClass, typename TAttachmentReferenceClass, typename TAttachmentDescriptionClass, typename TRenderPassCreateInfoClass>
507class FVulkanRenderPassBuilder
508{
509public:
510 FVulkanRenderPassBuilder(FVulkanDevice& InDevice)
511 : Device(InDevice)
512 , CorrelationMask(0)
513 {}
514
515 void BuildCreateInfo(const FVulkanRenderTargetLayout& RTLayout)
516 {
517 uint32 NumSubpasses = 0;
518 uint32 NumDependencies = 0;
519
520 //0b11 for 2, 0b1111 for 4, and so on
521 uint32 MultiviewMask = (0b1 << RTLayout.GetMultiViewCount()) - 1;
522
523 const bool bDeferredShadingSubpass = RTLayout.GetSubpassHint() == ESubpassHint::DeferredShadingSubpass;
524 const bool bApplyFragmentShadingRate = GRHISupportsAttachmentVariableRateShading
525 && RTLayout.GetFragmentDensityAttachmentReference() != nullptr
526 && Device.GetOptionalExtensions().HasKHRFragmentShadingRate
527 && Device.GetOptionalExtensionProperties().FragmentShadingRateFeatures.attachmentFragmentShadingRate == VK_TRUE;
528 const bool bResolveDepth = GRHISupportsDepthStencilResolve &&
529 Device.GetOptionalExtensions().HasKHRDepthStencilResolve &&
530 RTLayout.GetHasDepthStencilResolve();
531
532 const bool bCustomResolveSubpass = RTLayout.GetSubpassHint() == ESubpassHint::CustomResolveSubpass;
533 const bool bDepthReadSubpass = bCustomResolveSubpass || (RTLayout.GetSubpassHint() == ESubpassHint::DepthReadSubpass);
534 const bool bHasDepthStencilAttachmentReference = (RTLayout.GetDepthAttachmentReference() != nullptr);
535
536 if (bApplyFragmentShadingRate)
537 {
538 ShadingRateAttachmentReference.SetAttachment(*RTLayout.GetFragmentDensityAttachmentReference(), VkImageAspectFlagBits::VK_IMAGE_ASPECT_COLOR_BIT);
539 FragmentShadingRateAttachmentInfo.SetReference(&ShadingRateAttachmentReference);
540 }
541
542 if (bResolveDepth)
543 {
544 DepthStencilResolveAttachmentReference.SetAttachment(*RTLayout.GetDepthStencilResolveAttachmentReference(), VkImageAspectFlagBits::VK_IMAGE_ASPECT_NONE);
545 // Using zero bit because it is always supported if the extension is supported, from spec: "The VK_RESOLVE_MODE_SAMPLE_ZERO_BIT mode
546 // is the only mode that is required of all implementations (that support the extension or support Vulkan 1.2 or higher)."
547 DepthStencilResolveSubpassDesc.SetResolveModes(VK_RESOLVE_MODE_SAMPLE_ZERO_BIT, VK_RESOLVE_MODE_SAMPLE_ZERO_BIT);
548 DepthStencilResolveSubpassDesc.SetReference(&DepthStencilResolveAttachmentReference);
549 }
550
551 // Grab (and optionally convert) attachment references.
552 uint32 NumColorAttachments = RTLayout.GetNumColorAttachments();
553 for (uint32 ColorAttachment = 0; ColorAttachment < NumColorAttachments; ++ColorAttachment)
554 {
555 ColorAttachmentReferences.Add(TAttachmentReferenceClass(RTLayout.GetColorAttachmentReferences()[ColorAttachment], 0));
556 if (RTLayout.GetResolveAttachmentReferences() != nullptr)
557 {
558 ResolveAttachmentReferences.Add(TAttachmentReferenceClass(RTLayout.GetResolveAttachmentReferences()[ColorAttachment], 0));
559 }
560 }
561
562 // CustomResolveSubpass has an additional color attachment that should not be used by main and depth subpasses
563 if (bCustomResolveSubpass && (NumColorAttachments > 1))
564 {
565 NumColorAttachments--;
566 }
567
568 uint32_t DepthInputAttachment = VK_ATTACHMENT_UNUSED;
569 VkImageLayout DepthInputAttachmentLayout = VK_IMAGE_LAYOUT_UNDEFINED;
570 VkImageAspectFlags DepthInputAspectMask = 0;
571 if (bHasDepthStencilAttachmentReference)
572 {
573 DepthStencilAttachmentReference.SetDepthStencilAttachment(*RTLayout.GetDepthAttachmentReference(), RTLayout.GetStencilAttachmentReference(), 0, Device.SupportsParallelRendering());
574
575 if (bDepthReadSubpass || bDeferredShadingSubpass)
576 {
577 DepthStencilAttachment.attachment = RTLayout.GetDepthAttachmentReference()->attachment;
578 DepthStencilAttachment.SetAspect(VK_IMAGE_ASPECT_DEPTH_BIT); // @todo?
579
580 // FIXME: checking a Depth layout is not correct in all cases
581 // PSO cache can create a PSO for subpass 1 or 2 first, where depth is read-only but that does not mean depth pre-pass is enabled
582 if (false && RTLayout.GetDepthAttachmentReference()->layout == VK_IMAGE_LAYOUT_READ_ONLY_OPTIMAL)
583 {
584 // Depth is read only and is expected to be sampled as a regular texture
585 DepthStencilAttachment.layout = VK_IMAGE_LAYOUT_READ_ONLY_OPTIMAL;
586 }
587 else
588 {
589 // lights write to stencil for culling, so stencil is expected to be writebale while depth is read-only
590 DepthStencilAttachment.layout = VK_IMAGE_LAYOUT_DEPTH_READ_ONLY_STENCIL_ATTACHMENT_OPTIMAL;
591 DepthInputAttachment = DepthStencilAttachment.attachment;
592 DepthInputAttachmentLayout = DepthStencilAttachment.layout;
593 DepthInputAspectMask = VK_IMAGE_ASPECT_DEPTH_BIT;
594 }
595 }
596 }
597
598 // main sub-pass
599 {
600 TSubpassDescriptionClass& SubpassDesc = SubpassDescriptions[NumSubpasses++];
601
602 SubpassDesc.SetColorAttachments(ColorAttachmentReferences, NumColorAttachments);
603
604 if (bHasDepthStencilAttachmentReference)
605 {
606 SubpassDesc.SetDepthStencilAttachment(&DepthStencilAttachmentReference);
607 }
608
609 if (!bDepthReadSubpass && bResolveDepth)
610 {
611 SubpassDesc.SetDepthStencilResolveAttachment(&DepthStencilResolveSubpassDesc);
612 }
613
614 if (bApplyFragmentShadingRate)
615 {
616 SubpassDesc.SetShadingRateAttachment(&FragmentShadingRateAttachmentInfo);
617 }
618 SubpassDesc.SetMultiViewMask(MultiviewMask);
619 }
620
621 // Color write and depth read sub-pass
622 if (bDepthReadSubpass)
623 {
624 TSubpassDescriptionClass& SubpassDesc = SubpassDescriptions[NumSubpasses++];
625
626 SubpassDesc.SetColorAttachments(ColorAttachmentReferences, 1);
627
628 check(RTLayout.GetDepthAttachmentReference());
629
630 // Depth as Input0
631 InputAttachments1[0].attachment = DepthInputAttachment;
632 InputAttachments1[0].layout = DepthInputAttachmentLayout;
633 InputAttachments1[0].SetAspect(DepthInputAspectMask);
634 SubpassDesc.SetInputAttachments(InputAttachments1, InputAttachment1Count);
635 // depth attachment is same as input attachment
636 SubpassDesc.SetDepthStencilAttachment(&DepthStencilAttachment);
637
638 if (bResolveDepth && !bCustomResolveSubpass)
639 {
640 SubpassDesc.SetDepthStencilResolveAttachment(&DepthStencilResolveSubpassDesc);
641 }
642
643 if (bApplyFragmentShadingRate)
644 {
645 SubpassDesc.SetShadingRateAttachment(&FragmentShadingRateAttachmentInfo);
646 }
647 SubpassDesc.SetMultiViewMask(MultiviewMask);
648
