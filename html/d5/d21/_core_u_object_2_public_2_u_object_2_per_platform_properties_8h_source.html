74template<typename _StructType, typename _ValueType, EName _BasePropertyName>
75struct TPerPlatformProperty
76{
77 typedef _ValueType ValueType;
78 typedef _StructType StructType;
79
80#if WITH_EDITOR || WITH_PREVIEW_PPX_DATA
82 _ValueType GetValueForPlatform(FName PlatformName) const
83 {
84 const _StructType* This = StaticCast<const _StructType*>(this);
85 using MapType = decltype(This->PerPlatform);
86 using KeyFuncs = typename PerPlatformProperty::Private::KeyFuncs<MapType>;
87
88 const _ValueType* Ptr = PlatformName != NAME_None ? This->PerPlatform.Find(PlatformName) : nullptr;
89 if (Ptr == nullptr)
90 {
91#if WITH_EDITOR
92 const FDataDrivenPlatformInfo& Info = FDataDrivenPlatformInfoRegistry::GetPlatformInfo(PlatformName);
93 if (Info.PlatformGroupName != NAME_None)
94 {
95 Ptr = This->PerPlatform.Find(Info.PlatformGroupName);
96 }
97#else
98 #if WITH_PREVIEW_PPX_DATA
99 if(PPXPreviewOverrides::PerPlatformGroupOverride != NAME_None)
100 {
101 Ptr = This->PerPlatform.Find(PPXPreviewOverrides::PerPlatformGroupOverride);
102 }
103 #endif
104#endif
105 }
106
107 return Ptr ? *Ptr : This->Default;
108 }
109#endif
110
111 _ValueType GetDefault() const
112 {
113 const _StructType* This = StaticCast<const _StructType*>(this);
114 return This->Default;
115 }
116
117 _ValueType GetValue() const
118 {
119#if WITH_EDITORONLY_DATA && WITH_EDITOR
120 FName PlatformName;
121 // Lookup the override preview platform info, if any
122 // @todo this doesn't set PlatformName, just a group, but GetValueForPlatform() will technically work being given a Group name instead of a platform name, so we just use it
123 if (UObject::OnGetPreviewPlatform.IsBound() && UObject::OnGetPreviewPlatform.Execute(PlatformName))
124 {
125 return GetValueForPlatform(PlatformName);
126 }
127 else
128#endif
129 {
130#if WITH_PREVIEW_PPX_DATA
131 if (PPXPreviewOverrides::PerPlatformOverride != NAME_None || PPXPreviewOverrides::PerPlatformGroupOverride != NAME_None)
132 {
133 return GetValueForPlatform(PPXPreviewOverrides::PerPlatformOverride);
134 }
135 else
136#endif
137 {
138 const _StructType* This = StaticCast<const _StructType*>(this);
139 return This->Default;
140 }
141 }
142 }
143
144 /* Load old properties that have been converted to FPerPlatformX */
145 bool SerializeFromMismatchedTag(const FPropertyTag& Tag, FArchive& Ar)
146 {
147 if (Tag.Type == _BasePropertyName)
148 {
149 _StructType* This = StaticCast<_StructType*>(this);
150 _ValueType OldValue;
151 Ar << OldValue;
152 *This = _StructType(OldValue);
153 return true;
154 }
155 return false;
156 }
157
158 /* Serialization */
159 bool Serialize(FArchive& Ar)
160 {
161 Ar << *this;
162 return true;
163 }
164
165 /* Serialization */
166 bool Serialize(FStructuredArchive::FSlot Slot)
167 {
168 Slot << *this;
169 return true;
170 }
171};
