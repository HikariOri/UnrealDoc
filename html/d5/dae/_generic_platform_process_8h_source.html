965 {

968#elif PLATFORM_CPU_ARM_FAMILY

984 static inline void YieldCycles(uint64 Cycles)
985 {
986#if PLATFORM_CPU_X86_FAMILY
987 auto ReadCycleCounter = []()
988 {
989#if defined(_MSC_VER)
990 return __rdtsc();
991#elif PLATFORM_APPLE
992 return mach_absolute_time();
993#elif __has_builtin(__builtin_readcyclecounter)
994 return __builtin_readcyclecounter();
995#else
996 // the platform with other architectures must override this to not have this function be called
997 unimplemented();
998#endif
999 };
1000
1001 uint64 start = ReadCycleCounter();
1002 //some 32bit implementations return 0 for __builtin_readcyclecounter just to be on the safe side we protect against this.
1003 Cycles = start != 0 ? Cycles : 0;
1004
1005#if PLATFORM_WINDOWS
1006 if (FPlatformMisc::HasTimedPauseCPUFeature())
1007 {
1008 uint64 PauseCycles = ReadCycleCounter() + Cycles;
1009#if defined(_MSC_VER)
1010 _tpause(0, PauseCycles);
1011#elif __has_builtin(__builtin_ia32_tpause)
1012 __builtin_ia32_tpause(0, (uint32)(PauseCycles >> 32), (uint32)PauseCycles);
1013#else
1014# error Unsupported architecture!
1015#endif
1016 }
1017 else
1018#endif
1019 {
1020 do
1021 {
1022 Yield();
1023 } while ((ReadCycleCounter() - start) < Cycles);
1024 }
1025
1026#else
1027 // We can't read cycle counter from user mode on these platform
1028 for (uint64 i = 0; i < Cycles; i++)
1029 {
1030#if PLATFORM_CPU_ARM_FAMILY
1031# if defined(_MSC_VER)
1032 __yield();
1033# else
1034 __builtin_arm_yield();
1035# endif
1036#else
1037# error "Not implemented on this platform"
1038#endif
1039 }
1040#endif
1041 }
