25 /*
26 * Struct to hold data about each platform we have encountered in the cooker.
27 * Fields in this struct persist across multiple CookByTheBook sessions.
28 * All fields on this struct are readable only from either the scheduler thread or within a ReadLock on
29 * non-scheduler thread.
30 * Some fields on this struct are SingleThreadWrite fields - written only from the scheduler thread.
31 * These are unsynchronized other than SingleThreadWrite readlocks.
32 * The remaining fields are read/write from any thread, and are TAtomics.
33 */
34 struct FPlatformData
35 {
36 FPlatformData();
37
38 /* Name of the Platform, a cache of FName(ITargetPlatform->PlatformName()) */
39 FName PlatformName;
40
46 ITargetPlatform* TargetPlatform;
47
48 /*
49 * Pointer to the platform-specific RegistryGenerator for this platform. If already constructed we can take a
50 * faster refresh path on future sessions.
51 */
52 TUniquePtr<FAssetRegistryGenerator> RegistryGenerator;
53 TUniquePtr<UE::Cook::IAssetRegistryReporter> RegistryReporter;
54 FAssetRegistrySerializationOptions AssetRegistryDevelopmentSaveOptions;
55
56 /*
57 * Whether BeginCookSandbox has been called for this platform. If we have already initialized the sandbox we
58 * can take a faster refresh path on future sessions.
59 */
60 bool bIsSandboxInitialized = false;
61
63 bool bFullBuild = false;
68 bool bAllowIncrementalResults = true;
69
74 bool bLegacyIterativeSharedBuild = false;
75
77 bool bWorkerOnSharedSandbox = false;
78
79 /*
80 * The last FPlatformTime::GetSeconds() at which this platform was requested in a CookOnTheFly request.
81 * If equal to 0, the platform was not requested in a CookOnTheFly since the last clear.
82 */
83 TAtomic<double> LastReferenceTime;
84
85 /* The count of how many active CookOnTheFly requests are currently using the Platform. */
86 TAtomic<uint32> ReferenceCount;
87 };
