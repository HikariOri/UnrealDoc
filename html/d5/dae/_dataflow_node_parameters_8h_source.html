202 template<typename T>
203 void SetData(FContextCacheKey InKey, const FProperty* InProperty, T&& InValue, const FGuid& InNodeGuid, uint32 InNodeHash, const FTimestamp& InTimestamp)
204 {
205 TUniquePtr<FContextCacheElementBase> DataStoreEntry;
206
207 if constexpr (std::is_same_v<T, FContextValue>)
208 {
209 if (const FContextCacheElementBase* CacheEntryToClone = InValue.GetCacheEntryPrivate())
210 {
211 DataStoreEntry = CacheEntryToClone->Clone(*this);
212 if (DataStoreEntry)
213 {
214 DataStoreEntry->UpdatePropertyAndNodeData(InProperty, InNodeGuid, InNodeHash, InTimestamp);
215 }
216 }
217 }
218 else
219 {
220 bool bMadeEntry = false;
221 if constexpr (TIsTArray<typename TDecay<T>::Type>::Value)
222 {
223 typedef typename TDecay<T>::Type ArrayType;
224 if constexpr (TIsUObjectPtrElement<typename ArrayType::ElementType>::Value)
225 {
226 bMadeEntry = true;
227 DataStoreEntry = MakeUnique<TContextCacheElementUObjectArray<typename ArrayType::ElementType>>(InNodeGuid, InProperty, Forward<T>(InValue), InNodeHash, InTimestamp);
228 }
229 else if constexpr (TIsReflectedStruct<ArrayType>::Value)
230 {
231 DataStoreEntry = MakeUnique<FContextCacheElementUStructArray>(InNodeGuid, InProperty, Forward<T>(InValue), InNodeHash, InTimestamp);
232 }
233 }
234 if (!bMadeEntry)
235 {
236 if constexpr (TIsUObjectPtrElement<T>::Value)
237 {
238 DataStoreEntry = MakeUnique<TContextCacheElementUObject<T>>(InNodeGuid, InProperty, Forward<T>(InValue), InNodeHash, InTimestamp);
239 }
240 else if constexpr (TIsReflectedStruct<T>::Value)
241 {
242 DataStoreEntry = MakeUnique<FContextCacheElementUStruct>(InNodeGuid, InProperty, Forward<T>(InValue), InNodeHash, InTimestamp);
243 }
244 else
245 {
246 DataStoreEntry = MakeUnique<TContextCacheElement<T>>(InNodeGuid, InProperty, Forward<T>(InValue), InNodeHash, InTimestamp);
247 }
248 }
249 }
250
251 if (DataStoreEntry)
252 {
253 SetDataImpl(InKey, MoveTemp(DataStoreEntry));
254 }
255 else
256 {
257 SetNullData(InKey, InProperty, InNodeGuid, InNodeHash, InTimestamp);
258 }
259 }
