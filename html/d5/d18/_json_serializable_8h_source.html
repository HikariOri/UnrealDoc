106template<typename T, typename CharType>
107inline bool FromJson(TArray<T>& OutArray, TStringView<CharType> JsonString)
108{
109 OutArray.Reset();
110
111 TArray<TSharedPtr<FJsonValue>> ArrayValues;
112 TSharedRef<TJsonReader<CharType>> JsonReader = TJsonReaderFactory<CharType>::CreateFromView(JsonString);
113 if (FJsonSerializer::Deserialize(JsonReader, ArrayValues))
114 {
115 for (const TSharedPtr<FJsonValue>& Value : ArrayValues)
116 {
117 TSharedPtr<FJsonObject>* ArrayEntry;
118 if (Value.IsValid() && Value->TryGetObject(ArrayEntry))
119 {
120 if (ArrayEntry && ArrayEntry->IsValid())
121 {
122 FJsonSerializerReader Serializer(*ArrayEntry);
123 OutArray.Add_GetRef(T()).Serialize(Serializer, false);
124 }
125 else
126 {
127 UE_LOGF(LogJson, Error, "Failed to parse Json from array");
128 return false;
129 }
130 }
131 }
132
133 return true;
134 }
135
136 return false;
137}

168template<typename T>
169inline void ToJson(TArray<T>& InArray, const FToJsonVariantArgs& InArgs)
170{
171 using FPrettySerializerAndWriter = TTuple<FPrettySerializer, FPrettyWriter>;
172 using FCondensedSerializerAndWriter = TTuple<FCondensedSerializer, FCondensedWriter>;
173
174 using FSerializerVariant = TVariant<FPrettySerializerAndWriter, FCondensedSerializerAndWriter>;
175
176 FSerializerVariant SerializerToUse = ::Visit(UE::Overload(
177 [](const FReturnStringArgs& ReturnStringArgs)
178 {
179 if (ReturnStringArgs.template Get<1>())
180 {
181 FPrettyWriter NewWriter = TJsonWriterFactory<>::Create(ReturnStringArgs.template Get<0>());
182 return FSerializerVariant(TInPlaceType<FPrettySerializerAndWriter>(), FPrettySerializer(NewWriter), NewWriter);
183 }
184 else
185 {
186 FCondensedWriter NewWriter = TJsonWriterFactory<TCHAR, TCondensedJsonPrintPolicy<TCHAR>>::Create(ReturnStringArgs.template Get<0>());
187 return FSerializerVariant(TInPlaceType<FCondensedSerializerAndWriter>(), FCondensedSerializer(NewWriter), NewWriter);
188 }
189 },
190 [](const FWriterVariants& WriterVariants)
191 {
192 return ::Visit(UE::Overload(
193 [](const FPrettyWriter& PrettyWriter)
194 {
195 return FSerializerVariant(TInPlaceType<FPrettySerializerAndWriter>(), FPrettySerializer(PrettyWriter), PrettyWriter);
196 },
197 [](const FCondensedWriter& CondensedWriter)
198 {
199 return FSerializerVariant(TInPlaceType<FCondensedSerializerAndWriter>(), FCondensedSerializer(CondensedWriter), CondensedWriter);
200 }
201 ), WriterVariants);
202 }
203 ), InArgs);
204
205 const bool bCloseWriter = InArgs.IsType<FReturnStringArgs>();
206
207 ::Visit([bCloseWriter, &InArray](auto& StoredSerializer)
208 {
209 StoredSerializer.template Get<0>().StartArray();
210
211 ToJson_SerializeArrayElements(InArray, StoredSerializer.template Get<0>(), false);
212
213 StoredSerializer.template Get<0>().EndArray();
214
215 if (bCloseWriter)
216 {
217 StoredSerializer.template Get<1>()->Close();
218 }
219 }, SerializerToUse);
220}
