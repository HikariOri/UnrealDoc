123inline FBufferRHIRef CreateTempMediaVertexBuffer(FRHICommandListBase& RHICmdList, float ULeft = 0.0f, float URight = 1.0f, float VTop = 0.0f, float VBottom = 1.0f)
124{
125 const FRHIBufferCreateDesc CreateDesc =
126 FRHIBufferCreateDesc::CreateVertex<FMediaElementVertex>(TEXT("TempMediaVertexBuffer"), 4)
127 .AddUsage(EBufferUsageFlags::Volatile)
128 .DetermineInitialState()
129 .SetInitActionInitializer();
130
131 TRHIBufferInitializer<FMediaElementVertex> Vertices = RHICmdList.CreateBufferInitializer(CreateDesc);
132
133 Vertices[0].Position.Set(-1.0f, 1.0f, 1.0f, 1.0f); // Top Left
134 Vertices[1].Position.Set(1.0f, 1.0f, 1.0f, 1.0f); // Top Right
135 Vertices[2].Position.Set(-1.0f, -1.0f, 1.0f, 1.0f); // Bottom Left
136 Vertices[3].Position.Set(1.0f, -1.0f, 1.0f, 1.0f); // Bottom Right
137
138 Vertices[0].TextureCoordinate.Set(ULeft, VTop);
139 Vertices[1].TextureCoordinate.Set(URight, VTop);
140 Vertices[2].TextureCoordinate.Set(ULeft, VBottom);
141 Vertices[3].TextureCoordinate.Set(URight, VBottom);
142
143 return Vertices.Finalize();
144}
