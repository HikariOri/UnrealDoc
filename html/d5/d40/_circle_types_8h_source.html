243 TVector<RealType> ClosestPoint(const TVector<RealType>& QueryPoint) const
244 {
245 const TVector<RealType>& Center = Frame.Origin;
246 TVector<RealType> Normal = Frame.GetAxis(2);
247
248 TVector<RealType> PointDelta = QueryPoint - Center;
249 TVector<RealType> DeltaInPlane = PointDelta - Normal.Dot(PointDelta)*Normal;
250 RealType OriginDist = DeltaInPlane.Length();
251 if (OriginDist > (RealType)0)
252 {
253 return Center + (Radius / OriginDist)*DeltaInPlane;
254 }
255 else // all points equidistant, use any one
256 {
257 return Frame.Origin + Radius * Frame.GetAxis(0);
258 }
259 }

270 }

273 {
