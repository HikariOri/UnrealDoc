37template <typename D3D1x_SHADER_INPUT_BIND_DESC>
38EShaderCodeResourceBindingType D3DBindDescToShaderCodeResourceBinding(const D3D1x_SHADER_INPUT_BIND_DESC& Binding)
39{
40 switch (Binding.Type)
41 {
42 case D3D_SIT_SAMPLER:
43 return EShaderCodeResourceBindingType::SamplerState;
44 case D3D_SIT_TBUFFER:
45 case D3D_SIT_CBUFFER:
46 return EShaderCodeResourceBindingType::Buffer;
47 case D3D_SIT_TEXTURE:
48 switch (Binding.Dimension)
49 {
50 case D3D_SRV_DIMENSION_BUFFER: return EShaderCodeResourceBindingType::Buffer;
51 case D3D_SRV_DIMENSION_TEXTURE2D: return EShaderCodeResourceBindingType::Texture2D;
52 case D3D_SRV_DIMENSION_TEXTURE2DARRAY: return EShaderCodeResourceBindingType::Texture2DArray;
53 case D3D_SRV_DIMENSION_TEXTURE2DMS: return EShaderCodeResourceBindingType::Texture2DMS;
54 case D3D_SRV_DIMENSION_TEXTURE3D: return EShaderCodeResourceBindingType::Texture3D;
55 case D3D_SRV_DIMENSION_TEXTURECUBE: return EShaderCodeResourceBindingType::TextureCube;
56 default:
57 return EShaderCodeResourceBindingType::Invalid;
58 }
59 case D3D_SIT_UAV_RWTYPED:
60 switch (Binding.Dimension)
61 {
62 case D3D_SRV_DIMENSION_BUFFER: return EShaderCodeResourceBindingType::RWBuffer;
63 case D3D_SRV_DIMENSION_TEXTURE2D: return EShaderCodeResourceBindingType::RWTexture2D;
64 case D3D_SRV_DIMENSION_TEXTURE2DARRAY: return EShaderCodeResourceBindingType::RWTexture2DArray;
65 case D3D_SRV_DIMENSION_TEXTURE3D: return EShaderCodeResourceBindingType::RWTexture3D;
66 case D3D_SRV_DIMENSION_TEXTURECUBE: return EShaderCodeResourceBindingType::RWTextureCube;
67 default:
68 return EShaderCodeResourceBindingType::Invalid;
69 }
70 case D3D_SIT_STRUCTURED: return EShaderCodeResourceBindingType::StructuredBuffer;
71 case D3D_SIT_UAV_RWSTRUCTURED: return EShaderCodeResourceBindingType::RWStructuredBuffer;
72 case D3D_SIT_BYTEADDRESS: return EShaderCodeResourceBindingType::ByteAddressBuffer;
73 case D3D_SIT_UAV_RWBYTEADDRESS: return EShaderCodeResourceBindingType::RWByteAddressBuffer;
74 default:
75 return EShaderCodeResourceBindingType::Invalid;
76 }
77}

79template <typename ID3D1xShaderReflection, typename D3D1x_SHADER_DESC, typename D3D1x_SHADER_INPUT_BIND_DESC,
80 typename ID3D1xShaderReflectionConstantBuffer, typename D3D1x_SHADER_BUFFER_DESC,
81 typename ID3D1xShaderReflectionVariable, typename D3D1x_SHADER_VARIABLE_DESC>
82 inline void ExtractParameterMapFromD3DShader(
83 const FShaderCompilerInput& Input,
84 const FShaderParameterParser& ShaderParameterParser,
85 uint32 BindingSpace,
86 ID3D1xShaderReflection* Reflector, const D3D1x_SHADER_DESC& ShaderDesc,
87 FD3DShaderCompileData& CompileData,
88 FShaderCompilerOutput& Output
89 )
90{
91 // Add parameters for shader resources (constant buffers, textures, samplers, etc. */
92 for (uint32 ResourceIndex = 0; ResourceIndex < ShaderDesc.BoundResources; ResourceIndex++)
93 {
94 D3D1x_SHADER_INPUT_BIND_DESC BindDesc{};
95 Reflector->GetResourceBindingDesc(ResourceIndex, &BindDesc);
96
97 if (!IsCompatibleBinding(BindDesc, BindingSpace))
98 {
99 continue;
100 }
101
102 const FString BindDescName(BindDesc.Name);
103
104 if (BindDesc.Type == D3D_SIT_CBUFFER || BindDesc.Type == D3D_SIT_TBUFFER)
105 {
106 const uint32 CBIndex = BindDesc.BindPoint;
107 ID3D1xShaderReflectionConstantBuffer* ConstantBuffer = Reflector->GetConstantBufferByName(BindDesc.Name);
108
109 D3D1x_SHADER_BUFFER_DESC CBDesc{};
110 ConstantBuffer->GetDesc(&CBDesc);
111
112 const FString ConstantBufferName(CBDesc.Name);
113
114 const bool bGlobalCB = (ConstantBufferName == TEXT("$Globals"));
115 const bool bRootConstantsCB = (ConstantBufferName == TEXT("UERootConstants"));
116 const bool bIsRootCB = (ConstantBufferName == FShaderParametersMetadata::kRootUniformBufferBindingName);
117
118 if (bGlobalCB)
119 {
120 if (Input.ShouldUseStableConstantBuffer())
121 {
122 // Each member found in the global constant buffer means it was not in RootParametersStructure or
123 // it would have been moved by ShaderParameterParser.ParseAndModify().
124 for (uint32 ConstantIndex = 0; ConstantIndex < CBDesc.Variables; ConstantIndex++)
125 {
126 ID3D1xShaderReflectionVariable* Variable = ConstantBuffer->GetVariableByIndex(ConstantIndex);
127
128 D3D1x_SHADER_VARIABLE_DESC VariableDesc{};
129 Variable->GetDesc(&VariableDesc);
130
131 if (VariableDesc.uFlags & D3D_SVF_USED)
132 {
133 const FString VariableName(VariableDesc.Name);
134
135 AddUnboundShaderParameterError(
136 Input,
137 ShaderParameterParser,
138 VariableName,
139 Output);
140 }
141 }
142 }
143 else
144 {
145 // Track all of the variables in this constant buffer.
146 for (uint32 ConstantIndex = 0; ConstantIndex < CBDesc.Variables; ConstantIndex++)
147 {
148 ID3D1xShaderReflectionVariable* Variable = ConstantBuffer->GetVariableByIndex(ConstantIndex);
149
150 D3D1x_SHADER_VARIABLE_DESC VariableDesc{};
151 Variable->GetDesc(&VariableDesc);
152
153 if (VariableDesc.uFlags & D3D_SVF_USED)
154 {
155 const FString VariableName(VariableDesc.Name);
156
157 HandleReflectedGlobalConstantBufferMember(
158 Input,
159 ShaderParameterParser,
160 VariableName,
161 CBIndex,
162 VariableDesc.StartOffset,
163 VariableDesc.Size,
164 Output
165 );
166
167 CompileData.bGlobalUniformBufferUsed = true;
168 CompileData.UsedUniformBufferSlots[CBIndex] = true;
169 }
170 }
171 }
172 }
173 else if (bRootConstantsCB)
174 {
175 // For the UERootConstants root constant CB, we want to fully skip adding it to the parameter map, or
176 // updating the used slots or num CBs (all those assume space0).
177 }
178 else if (bIsRootCB && Input.ShouldUseStableConstantBuffer())
179 {
180 if (CBIndex == FShaderParametersMetadata::kRootCBufferBindingIndex)
181 {
182 int32 ConstantBufferSize = 0;
183
184 // Track all of the variables in this constant buffer.
185 for (uint32 ConstantIndex = 0; ConstantIndex < CBDesc.Variables; ConstantIndex++)
186 {
187 ID3D1xShaderReflectionVariable* Variable = ConstantBuffer->GetVariableByIndex(ConstantIndex);
188
189 D3D1x_SHADER_VARIABLE_DESC VariableDesc{};
190 Variable->GetDesc(&VariableDesc);
191
192 if (VariableDesc.uFlags & D3D_SVF_USED)
193 {
194 const FString VariableName(VariableDesc.Name);
195
196 HandleReflectedRootConstantBufferMember(
197 Input,
198 ShaderParameterParser,
199 VariableName,
200 VariableDesc.StartOffset,
201 VariableDesc.Size,
202 Output
203 );
204
205 ConstantBufferSize = FMath::Max<int32>(ConstantBufferSize, VariableDesc.StartOffset + VariableDesc.Size);
206 }
207 }
208
209 if (ConstantBufferSize > 0)
210 {
211 HandleReflectedRootConstantBuffer(ConstantBufferSize, Output);
212
213 CompileData.bGlobalUniformBufferUsed = true;
214 CompileData.UsedUniformBufferSlots[CBIndex] = true;
215 }
216 }
217 else
218 {
219 Output.AddErrorf(
220 TEXT("Error: %s is expected to always be in the API slot %d, but is actually in slot %d."),
221 FShaderParametersMetadata::kRootUniformBufferBindingName,
222 FShaderParametersMetadata::kRootCBufferBindingIndex,
223 CBIndex
224 );
225 Output.bSucceeded = false;
226 }
227 }
228 else
229 {
230 // Track just the constant buffer itself.
231 AddShaderValidationUBSize(CBIndex, CBDesc.Size, Output);
232 HandleReflectedUniformBuffer(Input, ConstantBufferName, CBIndex, Output);
233
234 CompileData.UsedUniformBufferSlots[CBIndex] = true;
235
236 const EUniformBufferMemberReflectionReason Reason = ShouldReflectUniformBufferMembers(Input, ConstantBufferName);
237 if (Reason != EUniformBufferMemberReflectionReason::None)
238 {
239 for (uint32 ConstantIndex = 0; ConstantIndex < CBDesc.Variables; ConstantIndex++)
240 {
241 ID3D1xShaderReflectionVariable* Variable = ConstantBuffer->GetVariableByIndex(ConstantIndex);
242
243 D3D1x_SHADER_VARIABLE_DESC VariableDesc{};
244 Variable->GetDesc(&VariableDesc);
245
246 if (VariableDesc.uFlags & D3D_SVF_USED)
247 {
248 const FString VariableName(VariableDesc.Name);
249
250 HandleReflectedUniformBufferConstantBufferMember(
251 Input,
252 Reason,
253 ConstantBufferName,
254 CBIndex,
255 VariableName,
256 VariableDesc.StartOffset,
257 VariableDesc.Size,
258 Output
259 );
260 }
261 }
262 }
263 }
264
265 if (CompileData.UniformBufferNames.Num() <= (int32)CBIndex)
266 {
267 CompileData.UniformBufferNames.AddDefaulted(CBIndex - CompileData.UniformBufferNames.Num() + 1);
268 }
269 CompileData.UniformBufferNames[CBIndex] = UE::ShaderCompilerCommon::RemoveConstantBufferPrefix(ConstantBufferName);
270
271 CompileData.NumCBs = FMath::Max(CompileData.NumCBs, BindDesc.BindPoint + BindDesc.BindCount);
272 }
273 else if (BindDesc.Type == D3D_SIT_TEXTURE || BindDesc.Type == D3D_SIT_SAMPLER)
274 {
275 check(BindDesc.BindCount == 1);
276
277 // https://github.com/GPUOpen-LibrariesAndSDKs/AGS_SDK/blob/master/ags_lib/hlsl/ags_shader_intrinsics_dx11.hlsl
278 const bool bIsAMDTexExtension = (FCStringAnsi::Strcmp(BindDesc.Name, "AmdDxExtShaderIntrinsicsResource") == 0);
279 const bool bIsAMDSmpExtension = (FCStringAnsi::Strcmp(BindDesc.Name, "AmdDxExtShaderIntrinsicsSamplerState") == 0);
280 const bool bIsVendorParameter = bIsAMDTexExtension || bIsAMDSmpExtension;
281
282 const uint32 BindCount = 1;
283 const EShaderParameterType ParameterType = (BindDesc.Type == D3D_SIT_SAMPLER) ? EShaderParameterType::Sampler : EShaderParameterType::SRV;
284
285 if (bIsVendorParameter)
286 {
287 CompileData.VendorExtensions.Emplace(EGpuVendorId::Amd, 0, BindDesc.BindPoint, BindCount, ParameterType);
288 }
289 else if (ParameterType == EShaderParameterType::Sampler)
290 {
291 HandleReflectedShaderSampler(BindDescName, BindDesc.BindPoint, Output);
292 CompileData.NumSamplers = FMath::Max(CompileData.NumSamplers, BindDesc.BindPoint + BindCount);
293 }
294 else
295 {
296 EShaderCodeResourceBindingType ResourceBindingType = D3DBindDescToShaderCodeResourceBinding(BindDesc);
297 AddShaderValidationSRVType(BindDesc.BindPoint, ResourceBindingType, Output);
298
299 HandleReflectedShaderResource(BindDescName, BindDesc.BindPoint, Output);
300 CompileData.NumSRVs = FMath::Max(CompileData.NumSRVs, BindDesc.BindPoint + BindCount);
301 }
302 }
303 else if (BindDesc.Type == D3D_SIT_UAV_RWTYPED || BindDesc.Type == D3D_SIT_UAV_RWSTRUCTURED ||
304 BindDesc.Type == D3D_SIT_UAV_RWBYTEADDRESS || BindDesc.Type == D3D_SIT_UAV_RWSTRUCTURED_WITH_COUNTER ||
305 BindDesc.Type == D3D_SIT_UAV_APPEND_STRUCTURED)
306 {
307 check(BindDesc.BindCount == 1);
308
309 // https://developer.nvidia.com/unlocking-gpu-intrinsics-hlsl
310 const bool bIsNVExtension = (FCStringAnsi::Strcmp(BindDesc.Name, "g_NvidiaExt") == 0);
311
312 // https://github.com/intel/intel-graphics-compiler/blob/master/inc/IntelExtensions.hlsl
313 const bool bIsIntelExtension = (FCStringAnsi::Strcmp(BindDesc.Name, "g_IntelExt") == 0);
314
315 // https://github.com/GPUOpen-LibrariesAndSDKs/AGS_SDK/blob/master/ags_lib/hlsl/ags_shader_intrinsics_dx11.hlsl
316 const bool bIsAMDExtensionDX11 = (FCStringAnsi::Strcmp(BindDesc.Name, "AmdDxExtShaderIntrinsicsUAV") == 0);
317
318 // https://github.com/GPUOpen-LibrariesAndSDKs/AGS_SDK/blob/master/ags_lib/hlsl/ags_shader_intrinsics_dx12.hlsl
319 const bool bIsAMDExtensionDX12 = (FCStringAnsi::Strcmp(BindDesc.Name, "AmdExtD3DShaderIntrinsicsUAV") == 0);
320
321 const bool bIsVendorParameter = bIsNVExtension || bIsIntelExtension || bIsAMDExtensionDX11 || bIsAMDExtensionDX12;
322
323 // See D3DCommon.ush
324 const bool bIsDiagnosticBufferParameter = (FCStringAnsi::Strcmp(BindDesc.Name, "UEDiagnosticBuffer") == 0);
325
326 const uint32 BindCount = 1;
327 if (bIsVendorParameter)
328 {
329 const EGpuVendorId VendorId =
330 bIsNVExtension ? EGpuVendorId::Nvidia :
331 (bIsAMDExtensionDX11 || bIsAMDExtensionDX12) ? EGpuVendorId::Amd :
332 bIsIntelExtension ? EGpuVendorId::Intel :
333 EGpuVendorId::Unknown;
334
335 // g_NvidiaExt can be included in shaders compiled for different vendors due to usage of dynamic branches (e.g. UsingCLAS) to avoid extra shader permutations.
336 // Mark it optional so the shader can be created on non-NVIDIA GPUs.
337 // TODO: This could be removed once CLAS is part of the DXR spec.
338 const bool bIsOptional = bIsNVExtension;
339
340 CompileData.VendorExtensions.Emplace(VendorId, 0, BindDesc.BindPoint, BindCount, EShaderParameterType::UAV, bIsOptional);
341 }
342 else if (bIsDiagnosticBufferParameter)
343 {
344 CompileData.bDiagnosticBufferUsed = true;
345 }
346 else
347 {
348 EShaderCodeResourceBindingType ResourceBindingType = D3DBindDescToShaderCodeResourceBinding(BindDesc);
349 AddShaderValidationUAVType(BindDesc.BindPoint, ResourceBindingType, Output);
350
351 HandleReflectedShaderUAV(BindDescName, BindDesc.BindPoint, Output);
352 CompileData.NumUAVs = FMath::Max(CompileData.NumUAVs, BindDesc.BindPoint + BindCount);
353 }
354 }
355 else if (BindDesc.Type == D3D_SIT_STRUCTURED || BindDesc.Type == D3D_SIT_BYTEADDRESS)
356 {
357 check(BindDesc.BindCount == 1);
358
359 EShaderCodeResourceBindingType ResourceBindingType = D3DBindDescToShaderCodeResourceBinding(BindDesc);
360 AddShaderValidationSRVType(BindDesc.BindPoint, ResourceBindingType, Output);
361
362 HandleReflectedShaderResource(BindDescName, BindDesc.BindPoint, Output);
363
364 // https://learn.microsoft.com/en-us/windows/win32/api/d3d12shader/ns-d3d12shader-d3d12_shader_input_bind_desc
365 // If the shader resource is a structured buffer, the field contains the stride of the type in bytes
366 if (BindDesc.Type == D3D_SIT_STRUCTURED)
367 {
368 UpdateStructuredBufferStride(Input, BindDescName, BindDesc.BindPoint, BindDesc.NumSamples, Output);
369 }
370
371 CompileData.NumSRVs = FMath::Max(CompileData.NumSRVs, BindDesc.BindPoint + 1);
372 }
373 else if (BindDesc.Type == D3D_SIT_RTACCELERATIONSTRUCTURE)
374 {
375 // Acceleration structure resources are treated as SRVs.
376 check(BindDesc.BindCount == 1);
377
378 EShaderCodeResourceBindingType ResourceBindingType = D3DBindDescToShaderCodeResourceBinding(BindDesc);
379 AddShaderValidationSRVType(BindDesc.BindPoint, ResourceBindingType, Output);
380
381 HandleReflectedShaderResource(BindDescName, BindDesc.BindPoint, Output);
382 CompileData.NumSRVs = FMath::Max(CompileData.NumSRVs, BindDesc.BindPoint + 1);
383 }
384 }
385
386 CompileData.NumInstructions = ShaderDesc.InstructionCount;
387}
388
389// Validate that we are not going over to maximum amount of resource bindings support by the default root signature on DX12
390// Currently limited for hard-coded root signature setup (see: FD3D12Adapter::StaticGraphicsRootSignature)
391// In theory this limitation is only required for DX12, but we don't want a shader to compile on DX11 while not working on DX12.
392// (DX11 has an API limit on 128 SRVs, 16 Samplers, 8 UAVs and 14 CBs but if you go over these values then the shader won't compile)
393inline bool ValidateResourceCounts(const FD3DShaderCompileData& CompileData, TArray<FShaderCompilerError>& OutErrors)
394{
395 const bool bTooManySRVs = !CompileData.bBindlessEnabled && CompileData.NumSRVs > CompileData.MaxSRVs;
396 const bool bTooManyUAVs = !CompileData.bBindlessEnabled && CompileData.NumUAVs > CompileData.MaxUAVs;
397 const bool bTooManySamplers = !CompileData.bBindlessEnabled && CompileData.NumSamplers > CompileData.MaxSamplers;
398 const bool bTooManyCBs = CompileData.NumCBs > CompileData.MaxCBs;
399
400 if (bTooManySRVs || bTooManySamplers || bTooManyUAVs || bTooManyCBs)
401 {
402 if (bTooManySRVs)
403 {
404 OutErrors.Add(FString::Printf(TEXT("Shader is using too many SRVs: %d (only %d supported)"), CompileData.NumSRVs, CompileData.MaxSRVs));
405 }
406
407 if (bTooManySamplers)
408 {
409 OutErrors.Add(FString::Printf(TEXT("Shader is using too many Samplers: %d (only %d supported)"), CompileData.NumSamplers, CompileData.MaxSamplers));
410 }
411
412 if (bTooManyUAVs)
413 {
414 OutErrors.Add(FString::Printf(TEXT("Shader is using too many UAVs: %d (only %d supported)"), CompileData.NumUAVs, CompileData.MaxUAVs));
415 }
416
417 if (bTooManyCBs)
418 {
419 OutErrors.Add(FString::Printf(TEXT("Shader is using too many Constant Buffers: %d (only %d supported)"), CompileData.NumCBs, CompileData.MaxCBs));
420 }
421
422 return false;
423 }
424
425 return true;
426}

428inline FShaderCodePackedResourceCounts InitPackedResourceCounts(const FD3DShaderCompileData& CompileData)
429{
430 FShaderCodePackedResourceCounts PackedResourceCounts{};
431
432 if (CompileData.bGlobalUniformBufferUsed)
433 {
434 EnumAddFlags(PackedResourceCounts.UsageFlags, EShaderResourceUsageFlags::GlobalUniformBuffer);
435 }
436
437 if (CompileData.bBindlessEnabled)
438 {
439 EnumAddFlags(PackedResourceCounts.UsageFlags, EShaderResourceUsageFlags::BindlessResources);
440 EnumAddFlags(PackedResourceCounts.UsageFlags, EShaderResourceUsageFlags::BindlessSamplers);
441 }
442
443 if (CompileData.bDiagnosticBufferUsed)
444 {
445 EnumAddFlags(PackedResourceCounts.UsageFlags, EShaderResourceUsageFlags::DiagnosticBuffer);
446 }
447
448 PackedResourceCounts.NumSamplers = static_cast<uint8>(CompileData.NumSamplers);
449 PackedResourceCounts.NumSRVs = static_cast<uint8>(CompileData.NumSRVs);
450 PackedResourceCounts.NumCBs = static_cast<uint8>(CompileData.NumCBs);
451 PackedResourceCounts.NumUAVs = static_cast<uint8>(CompileData.NumUAVs);
452
453 return PackedResourceCounts;
454}

456template <typename TBlob>
457inline void GenerateFinalOutput(
458 TRefCountPtr<TBlob>& CompressedData,
459 const FShaderCompilerInput& Input,
460 ED3DShaderModel ShaderModel,
461 bool bProcessingSecondTime,
462 FD3DShaderCompileData& CompileData,
463 const FShaderCodePackedResourceCounts& PackedResourceCounts,
464 FShaderCompilerOutput& Output,
465 TFunction<void(FMemoryWriter&)> PostSRTWriterCallback,
466 TFunction<void(FShaderCode&)> AddOptionalDataCallback)
467{
468 const uint32 NumBindlessResources = CompileData.bBindlessEnabled ? Output.ParameterMap.CountParametersOfType(EShaderParameterType::BindlessSRV) : 0;
469 const uint32 NumBindlessSamplers = CompileData.bBindlessEnabled ? Output.ParameterMap.CountParametersOfType(EShaderParameterType::BindlessSampler) : 0;
470
471 // Build the SRT for this shader.
472 FShaderResourceTable SRT;
473
474 TArray<uint8> UniformBufferNameBytes;
475
476 {
477 // Build the generic SRT for this shader.
478 FShaderCompilerResourceTable GenericSRT;
479 BuildResourceTableMapping(Input.Environment.ResourceTableMap, Input.Environment.UniformBufferMap, CompileData.UsedUniformBufferSlots, Output.ParameterMap, GenericSRT);
480
481 // Ray generation shaders rely on a different binding model that aren't compatible with global uniform buffers.
482 if (Input.Target.Frequency != SF_RayGen)
483 {
484 CullGlobalUniformBuffers(Input.Environment.UniformBufferMap, Output.ParameterMap);
485 }
486
487 if (CompileData.UniformBufferNames.Num() < GenericSRT.ResourceTableLayoutHashes.Num())
488 {
489 CompileData.UniformBufferNames.AddDefaulted(GenericSRT.ResourceTableLayoutHashes.Num() - CompileData.UniformBufferNames.Num());
490 }
491
492 for (int32 Index = 0; Index < GenericSRT.ResourceTableLayoutHashes.Num(); ++Index)
493 {
494 if (GenericSRT.ResourceTableLayoutHashes[Index] != 0 && CompileData.UniformBufferNames[Index].Len() == 0)
495 {
496 for (const auto& KeyValue : Input.Environment.UniformBufferMap)
497 {
498 const FUniformBufferEntry& UniformBufferEntry = KeyValue.Value;
499
500 if (UniformBufferEntry.LayoutHash == GenericSRT.ResourceTableLayoutHashes[Index])
501 {
502 CompileData.UniformBufferNames[Index] = KeyValue.Key;
503 break;
504 }
505 }
506 }
507 }
508
509 FMemoryWriter UniformBufferNameWriter(UniformBufferNameBytes);
510 UniformBufferNameWriter << CompileData.UniformBufferNames;
511
512 UE::ShaderCompilerCommon::BuildShaderResourceTable(GenericSRT, SRT);
513 }
514
515 if (Input.Environment.CompilerFlags.Contains(CFLAG_ForceRemoveUnusedInterpolators) && Input.Target.Frequency == SF_Pixel && Input.bCompilingForShaderPipeline && bProcessingSecondTime)
516 {
517 Output.bSupportsQueryingUsedAttributes = true;
518 Output.UsedAttributes = CompileData.ShaderInputs;
519 }
520
521 // Generate the final Output
522 FMemoryWriter Ar(Output.ShaderCode.GetWriteAccess(), true);
523 Ar << SRT;
524
525 PostSRTWriterCallback(Ar);
526
527 Ar.Serialize(CompressedData->GetBufferPointer(), CompressedData->GetBufferSize());
528
529 // Append data that is generate from the shader code and assist the usage, mostly needed for DX12
530 {
531 Output.ShaderCode.AddOptionalData(PackedResourceCounts);
532 Output.ShaderCode.AddOptionalData(FShaderCodeUniformBuffers::Key, UniformBufferNameBytes.GetData(), UniformBufferNameBytes.Num());
533 AddOptionalDataCallback(Output.ShaderCode);
534 }
535
536 // Append the shader binding layout hash used for validation
537 {
538 uint32 ShaderBindingLayoutHash = Input.Environment.RHIShaderBindingLayout.GetHash();
539
540 TArray<uint8> WriterBytes;
541 FMemoryWriter Writer(WriterBytes);
542 Writer << ShaderBindingLayoutHash;
543 if (WriterBytes.Num() > 0)
544 {
545 Output.ShaderCode.AddOptionalData(FShaderCodeShaderResourceTableDataDesc::Key, WriterBytes.GetData(), WriterBytes.Num());
546 }
547 }
548
549 // Append information about optional hardware vendor extensions
550 if (CompileData.VendorExtensions.Num() > 0)
551 {
552 TArray<uint8> WriterBytes;
553 FMemoryWriter Writer(WriterBytes);
554 Writer << CompileData.VendorExtensions;
555 if (WriterBytes.Num() > 0)
556 {
557 Output.ShaderCode.AddOptionalData(FShaderCodeVendorExtension::Key, WriterBytes.GetData(), WriterBytes.Num());
558 }
559 }
560
561 if (Input.Environment.CompilerFlags.Contains(CFLAG_ExtraShaderData))
562 {
563 Output.ShaderCode.AddOptionalData(FShaderCodeName::Key, TCHAR_TO_UTF8(*Input.GenerateShaderName()));
564 }
565
566 Output.SerializeShaderCodeValidation();
567 Output.SerializeShaderDiagnosticData();
568
569 // Set the number of instructions.
570 Output.NumInstructions = CompileData.NumInstructions;
571
572 Output.NumTextureSamplers = PackedResourceCounts.NumSamplers;
573
574 // Pass the target through to the output.
575 Output.Target = Input.Target;
576
577 // SRV Limits
578 {
579 if (CompileData.bBindlessEnabled)
580 {
581 Output.AddStatistic<uint32>(TEXT("Bindless Resources"), NumBindlessResources);
582 Output.AddStatistic<uint32>(TEXT("Bindless Samplers"), NumBindlessSamplers);
583 }
584 else
585 {
586 Output.AddStatistic<uint32>(TEXT("Resources Used"), CompileData.NumSRVs);
587 Output.AddStatistic<uint32>(TEXT("Resource Limit"), CompileData.MaxSRVs);
588
589 Output.AddStatistic<uint32>(TEXT("RW Resources Used"), CompileData.NumUAVs);
590 Output.AddStatistic<uint32>(TEXT("RW Resource Limit"), CompileData.MaxUAVs);
591
592 Output.AddStatistic<uint32>(TEXT("Samplers Used"), CompileData.NumSamplers);
593 Output.AddStatistic<uint32>(TEXT("Sampler Limit"), CompileData.MaxSamplers);
594 }
595 }
596}

614template<ShaderCompilerType LambdaPlatform, typename TShaderCompileLambdaType,
615 typename ID3D1xShaderReflection, typename D3D1x_SHADER_DESC, typename D3D1x_SIGNATURE_PARAMETER_DESC>
616bool RemoveUnusedInterpolators(
617 const FShaderCompilerInput& Input,
618 const FString& PreprocessedShaderSource,
619 const FString& EntryPointName,
620 const FShaderParameterParser& ShaderParameterParser,
621 const TCHAR* ShaderProfile,
622 const ED3DShaderModel ShaderModel,
623 const bool bProcessingSecondTime,
624 FD3DShaderCompileData& CompileData,
625 TRefCountPtr<ID3D1xShaderReflection> Reflector,
626 TShaderCompileLambdaType ShaderCompileLambda,
627 FShaderCompilerOutput& Output,
628 bool& CompileResult)
629{
630 if (Input.Target.Frequency == SF_Pixel)
631 {
632 if (Reflector)
633 {
634 // Read the constant table description.
635 D3D1x_SHADER_DESC ShaderDesc;
636 Reflector->GetDesc(&ShaderDesc);
637
638 bool bFoundUnused = false;
639 for (uint32 Index = 0; Index < ShaderDesc.InputParameters; ++Index)
640 {
641 // VC++ horrible hack: Runtime ESP checks get confused and fail for some reason calling Reflector->GetInputParameterDesc() (because it comes from another DLL?)
642 // so "guard it" using the middle of an array; it's been confirmed NO corruption is really happening.
643 D3D1x_SIGNATURE_PARAMETER_DESC ParamDescs[3];
644 D3D1x_SIGNATURE_PARAMETER_DESC& ParamDesc = ParamDescs[1];
645 Reflector->GetInputParameterDesc(Index, &ParamDesc);
646 if (ParamDesc.SystemValueType == D3D_NAME_UNDEFINED)
647 {
648 if (ParamDesc.ReadWriteMask != 0)
649 {
650 FString SemanticName(ParamDesc.SemanticName);
651
652 CompileData.ShaderInputs.AddUnique(SemanticName);
653
654 // Add the number (for the case of TEXCOORD)
655 FString SemanticIndexName = FString::Printf(TEXT("%s%d"), *SemanticName, ParamDesc.SemanticIndex);
656 CompileData.ShaderInputs.AddUnique(SemanticIndexName);
657
658 // Add _centroid
659 CompileData.ShaderInputs.AddUnique(SemanticName + TEXT("_centroid"));
660 CompileData.ShaderInputs.AddUnique(SemanticIndexName + TEXT("_centroid"));
661 }
662 else
663 {
664 bFoundUnused = true;
665 }
666 }
667 else
668 {
669 // if (ParamDesc.ReadWriteMask != 0)
670 {
671 // Keep system values
672 CompileData.ShaderInputs.AddUnique(FString(ParamDesc.SemanticName));
673 }
674 }
675 }
676
677 if constexpr (LambdaPlatform == ShaderCompilerType::DXC)
678 {
679 //Seems like there is bug in DXC compiler which prevents SV_Coverage semantic
680 //from being detected as Input Parameter in the shader reflection data.
681 //Other system interpolators are fine...
682 //Can compile GizmoMaterial to repro the issue for example: -run=CookShadersCommandlet -targetPlatform=Windows -material=GizmoMaterial
683 //Always adding SV_Coverage as Used interpolator to mitigate this issue.
684 CompileData.ShaderInputs.AddUnique(TEXT("SV_Coverage"));
685 }
686
687 if (Input.Environment.CompilerFlags.Contains(CFLAG_ForceRemoveUnusedInterpolators) && Input.bCompilingForShaderPipeline && bFoundUnused && !bProcessingSecondTime)
688 {
689 // Rewrite the source removing the unused inputs so the bindings will match.
690 // We may need to do this more than once if unused inputs change after the removal. Ie. for complex shaders, what can happen is:
691 // pass1 detects that input A is not used, but input B and C are. Input A is removed, and we recompile (pass2). After the recompilation, we see that Input B is now also unused in pass2
692 // (it became simpler and the compiler could see through that).
693 // Since unused inputs are passed to the next stage, that will cause us to generate a vertex shader that does not output B, but our pixel shader will still be expecting B on input,
694 // as it was rewritten based on the pass1 results.
695
696 const FShaderCompilerOutput OriginalOutput = Output;
697 const int kMaxReasonableAttempts = 64;
698 for (int32 Attempt = 0; Attempt < kMaxReasonableAttempts; ++Attempt)
699 {
700 if (UE::ShaderCompilerCommon::RemoveUnusedInputs(Input, Output, PreprocessedShaderSource, Input.EntryPointName, CompileData.ShaderInputs))
701 {
702 CompileResult = ShaderCompileLambda(Input, Output.ModifiedShaderSource, Output.ModifiedEntryPointName, ShaderParameterParser, ShaderProfile, ShaderModel, true, Output);
703 if (!CompileResult)
704 {
705 // if we failed to compile the shader, propagate the error up
706 return true;
707 }
708
709 // check if the ShaderInputs changed - if not, we're done here
710 if (Output.UsedAttributes.Num() == CompileData.ShaderInputs.Num())
711 {
712 return true;
713 }
714
715 // second pass cannot use more attributes than previously
716 if (Output.UsedAttributes.Num() > CompileData.ShaderInputs.Num())
717 {
718 UE_LOGF(LogD3DShaderCompiler, Warning, "Second pass had more used attributes (%d) than first pass (%d)", Output.UsedAttributes.Num(), CompileData.ShaderInputs.Num());
719 FShaderCompilerError NewError(FString::Printf(TEXT("Second pass had more used attributes (%d) than first pass (%d)"), Output.UsedAttributes.Num(), CompileData.ShaderInputs.Num()));
720 Output = OriginalOutput;
721 Output.Errors.Emplace(MoveTemp(NewError));
722 break;
723 }
724
725 // if we're about to run out of attempts, report
726 if (Attempt >= kMaxReasonableAttempts - 1)
727 {
728 UE_LOGF(LogD3DShaderCompiler, Warning, "Unable to determine unused inputs after %d attempts (last number of used attributes: %d, previous step:%d)!",
729 Attempt + 1,
730 Output.UsedAttributes.Num(),
731 CompileData.ShaderInputs.Num());
732 FShaderCompilerError NewError(FString::Printf(TEXT("Unable to determine unused inputs after %d attempts (last number of used attributes: %d, previous step:%d)!"),
733 Attempt + 1,
734 Output.UsedAttributes.Num(),
735 CompileData.ShaderInputs.Num()));
736 Output = OriginalOutput;
737 Output.Errors.Emplace(MoveTemp(NewError));
738 break;
739 }
740
741 CompileData.ShaderInputs = Output.UsedAttributes;
742 Output = OriginalOutput;
743 // go around to remove newly identified unused inputs
744 }
745 }
746 }
747 }
748 }
749
750 return false;
751}
