22void SQPerfComparisonHelper(const FString& TestName, FPhysTestSerializer& Serializer, bool bEnsureOnMismatch = false)
23{
24 using namespace Chaos;
25 uint32 PhysXSum = 0;
26 uint32 ChaosSum = 0;
27 double NumIterations = bHasPhysX ? 100 : 10000;
28 FPendingSpatialDataQueue Empty;
29 //double NumIterations = 1000000;
30
31 const FSQCapture& CapturedSQ = *Serializer.GetSQCapture();
32 switch (CapturedSQ.SQType)
33 {
34 case FSQCapture::ESQType::Raycast:
35 {
36 ISpatialAccelerationCollection<FAccelerationStructureHandle, FReal, 3>* Accelerator = nullptr;
37 Serializer.GetChaosData()->UpdateExternalAccelerationStructure_External(Accelerator, Empty);
38 FChaosSQAccelerator SQAccelerator(*Accelerator);
39 for (double i = 0; i < NumIterations; ++i)
40 {
41 auto ChaosHitBuffer = MakeUnique<ChaosInterface::FSQHitBuffer<ChaosInterface::FRaycastHit>>();
42 uint32 StartTime = FPlatformTime::Cycles();
43 SQAccelerator.Raycast(CapturedSQ.StartPoint, CapturedSQ.Dir, CapturedSQ.DeltaMag, *ChaosHitBuffer, CapturedSQ.OutputFlags.HitFlags, CapturedSQ.QueryFilterData, *CapturedSQ.FilterCallback);
44 ChaosSum += FPlatformTime::Cycles() - StartTime;
45 }
46 break;
47 }
48 case FSQCapture::ESQType::Sweep:
49 {
50 ISpatialAccelerationCollection<FAccelerationStructureHandle, FReal, 3>* Accelerator = nullptr;
51 Serializer.GetChaosData()->UpdateExternalAccelerationStructure_External(Accelerator, Empty);
52 FChaosSQAccelerator SQAccelerator(*Accelerator);
53 for (double i = 0; i < NumIterations; ++i)
54 {
55 auto ChaosHitBuffer = MakeUnique<ChaosInterface::FSQHitBuffer<ChaosInterface::FSweepHit>>();
56 uint32 StartTime = FPlatformTime::Cycles();
57 SQAccelerator.Sweep(*CapturedSQ.ChaosImplicitGeometry, CapturedSQ.StartTM, CapturedSQ.Dir, CapturedSQ.DeltaMag, *ChaosHitBuffer, CapturedSQ.OutputFlags.HitFlags, CapturedSQ.QueryFilterData, *CapturedSQ.FilterCallback);
58 ChaosSum += FPlatformTime::Cycles() - StartTime;
59 }
60 break;
61 }
62 case FSQCapture::ESQType::Overlap:
63 {
64 ISpatialAccelerationCollection<FAccelerationStructureHandle, FReal, 3>* Accelerator = nullptr;
65 Serializer.GetChaosData()->UpdateExternalAccelerationStructure_External(Accelerator, Empty);
66 FChaosSQAccelerator SQAccelerator(*Accelerator);
67 for (double i = 0; i < NumIterations; ++i)
68 {
69 auto ChaosHitBuffer = MakeUnique<ChaosInterface::FSQHitBuffer<ChaosInterface::FOverlapHit>>();
70 uint32 StartTime = FPlatformTime::Cycles();
71 SQAccelerator.Overlap(*CapturedSQ.ChaosImplicitGeometry, CapturedSQ.StartTM, *ChaosHitBuffer, CapturedSQ.QueryFilterData, *CapturedSQ.FilterCallback);
72 ChaosSum += FPlatformTime::Cycles() - StartTime;
73 }
74 break;
75 }
76 }
77
78 float ChaosMs = FPlatformTime::ToMilliseconds(ChaosSum);
79 float ChaosAvgUs = (ChaosMs * 1000) / NumIterations;
80
81 float PhysXMs = FPlatformTime::ToMilliseconds(PhysXSum);
82 float PhysXAvgUs = (PhysXMs * 1000) / NumIterations;
83
84
85 if (bHasPhysX)
86 {
87 UE_LOGF(LogPhysicsCore, Warning, "Perf Test:%ls\nPhysX:%f(us), Chaos:%f(us)", *TestName, PhysXAvgUs, ChaosAvgUs);
88 }
89 else
90 {
91 UE_LOGF(LogPhysicsCore, Warning, "Perf Test:%ls\nChaos:%f(us), Total:%f(ms)", *TestName, ChaosAvgUs, ChaosMs);
92 //UE_LOGF(LogPhysicsCore, Warning, "Perf Test:%ls\nChaos:%f(us)", *TestName, AvgChaos);
93 }
94}
