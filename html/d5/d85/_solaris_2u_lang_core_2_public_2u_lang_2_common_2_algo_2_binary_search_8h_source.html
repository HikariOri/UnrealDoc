27 SizeType LowerBoundInternal(RangeValueType* First, const SizeType Num, const PredicateValueType& Value, ProjectionType Projection, SortPredicateType SortPredicate)
28 {

30 SizeType Start = 0;

32 SizeType Size = Num;

34 // With this method, if Size is even it will do one more comparison than necessary, but because Size can be predicted by the CPU it is faster in practice

38 Size = Size / 2;
