109inline void SetupNonUniformHelper(FVector InScale3D, double& OutMinScale, double& OutMinScaleAbs, FVector& OutScale3DAbs)
110{
111 // if almost zero, set min scale
112 // @todo fixme
113 if (InScale3D.IsNearlyZero())
114 {
115 // set min scale
116 InScale3D = FVector(0.1f);
117 }
118
119 OutScale3DAbs = InScale3D.GetAbs();
120 OutMinScaleAbs = OutScale3DAbs.GetMin();
121
122 OutMinScale = FMath::Max3(InScale3D.X, InScale3D.Y, InScale3D.Z) < 0.f ? -OutMinScaleAbs : OutMinScaleAbs; //if all three values are negative make minScale negative
123
124 if (FMath::IsNearlyZero(OutMinScale))
125 {
126 // only one of them can be 0, we make sure they have mini set up correctly
127 OutMinScale = 0.1f;
128 OutMinScaleAbs = 0.1f;
129 }
130}

136 OutMinScale = static_cast<float>(OutMinScaleD); // LWC_TODO: Precision loss?
137 OutMinScaleAbs = static_cast<float>(OutMinScaleAbsD);

142inline bool CalcMeshNegScaleCompensation(const FVector& InScale3D, FTransform& OutTransform)
143{
144 OutTransform = FTransform::Identity;
145
146 if (InScale3D.Y > 0.f)
147 {
148 if (InScale3D.Z > 0.f)
149 {
150 // no rotation needed
151 }
152 else
153 {
154 // y pos, z neg
155 OutTransform.SetRotation(FQuat(FVector(0.0f, 1.0f, 0.0f), UE_PI));
156 //OutTransform.q = PxQuat(PxPi, PxVec3(0,1,0));
157 }
158 }
159 else
160 {
161 if (InScale3D.Z > 0.f)
162 {
163 // y neg, z pos
164 //OutTransform.q = PxQuat(PxPi, PxVec3(0,0,1));
165 OutTransform.SetRotation(FQuat(FVector(0.0f, 0.0f, 1.0f), UE_PI));
166 }
167 else
168 {
169 // y neg, z neg
170 //OutTransform.q = PxQuat(PxPi, PxVec3(1,0,0));
171 OutTransform.SetRotation(FQuat(FVector(1.0f, 0.0f, 0.0f), UE_PI));
172 }
173 }
174
175 // Use inverted mesh if determinant is negative
176 return (InScale3D.X * InScale3D.Y * InScale3D.Z) < 0.f;
177}
