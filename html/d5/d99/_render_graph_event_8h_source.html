203//
204// Main RDG scope class.
205//
206// A tree of these scopes is created by the render thread as the RenderGraph is built.
207// Each scope type implementation uses the following functions, which are called during different RDG phases:
208//
209// Constructor / ImmediateEnd() - Render thread timeline. Called once, either side of scoped graph building work.
210//
211// BeginCPU / EndCPU - Parallel threads. Called during RDG pass lambdas execution. Scopes may be
212// entered / exited multiple times depending on parallel pass set bucketing.
213//
214// BeginGPU / EndGPU - Parallel threads. Called once for each GPU pipeline the scope covers.
215// Used for inserting commands on the RHICmdList. The command list passed to
216// Begin / End may be different in each, depending on parallel pass set bucketing.
217//
218struct FRDGScope
219{
220 FRDGScope* const Parent;
221 FRDGPass* CPUFirstPass = nullptr;
222 FRDGPass* CPULastPass = nullptr;
223 TRHIPipelineArray<FRDGPass*> GPUFirstPass { InPlace, nullptr };
224 TRHIPipelineArray<FRDGPass*> GPULastPass { InPlace, nullptr };

226 template <typename... TTypes>
227 class TStorage
228 {
229 typedef TVariant<FEmptyVariantState, TTypes...> TImpl;
230 TImpl Impl;
231
232 public:
233 template <typename TCallback>
234 void Dispatch(TCallback&& Callback)
235 {
236 size_t Index = Impl.GetIndex();
237 check(Index > 0);
238 ((Index == Impl.template IndexOfType<TTypes>() ? Callback(Impl.template Get<TTypes>()),0 : 0), ...);
239 }
240
241 template <typename TScopeType, typename... TArgs>
242 void Emplace(TArgs&&... Args)
243 {
244 Impl.template Emplace<TScopeType>(Forward<TArgs>(Args)...);
245 }
246
247 template <typename TScopeType>
248 static constexpr SIZE_T GetTypeIndex()
249 {
250 return TImpl::template IndexOfType<TScopeType>();
251 }
252
253 template <typename TScopeType>
254 TScopeType* Get()
255 {
256 return Impl.GetIndex() == Impl.template IndexOfType<TScopeType>()
257 ? &Impl.template Get<TScopeType>()
258 : nullptr;
259 }
260
261 template <typename TScopeType>
262 TScopeType const* Get() const
263 {
264 return const_cast<TStorage&>(*this).Get<TScopeType>();
265 }
266 };

405 // Returns an RHI breadcrumb initializer appropriate for the given varargs.
406 #define RDG_BREADCRUMB_DESC_FORWARD_VALUES(StaticName, FormatString, GPUStatArgs) \
407 [&](UE::RHI::Breadcrumbs::Private::FRDGFormatTag&& Tag, auto&&... Values) -> auto \
408 { \
409 using namespace UE::RHI::Breadcrumbs::Private; \
410 if constexpr (std::is_same_v<TRemoveCVRef<decltype(FormatString)>, FRDGEventName>) \
411 { \
412 /* Single FRDGEventName */ \
413 static_assert(sizeof...(Values) == 0, "Unexpected additional arguments"); \
414 return RHI_BREADCRUMB_DESC_FORWARD_VALUES(StaticName, FForceNoSprintf(), GPUStatArgs)( \
415 std::forward<decltype(FormatString)>(FormatString) \
416 ); \
417 } \
418 else if constexpr (TIsValidArgs<decltype(Values)...>::Value) \
419 { \
420 /* Varargs are compatible with RHI breadcrumbs. Forward the args directly. */ \
421 return RHI_BREADCRUMB_DESC_FORWARD_VALUES(StaticName, FormatString, GPUStatArgs)( \
422 std::forward<decltype(Values)>(Values)... \
423 ); \
424 } \
425 else \
426 { \
427 /* Args are not compatible with RHI breadcrumbs. Do the snprintf now */ \
428 return RHI_BREADCRUMB_DESC_COPY_VALUES(StaticName, FForceNoSprintf(), GPUStatArgs)( \
429 std::forward_as_tuple( \
430 Tag \
431 , FormatString \
432 , std::forward<decltype(Values)>(Values)... \
433 ) \
434 ); \
435 } \
436 }
