95struct FInterruptionContext
96{
97public:
98 typedef TFunction<bool(void)> ShouldExitEarlyCallbackType;
99
100 FInterruptionContext() = default;
101 explicit FInterruptionContext(double InTickStartTime, double InMaxRunningTime)
102 : TickStartTime(InTickStartTime)
103 , MaxRunningTime(InMaxRunningTime)
104 {
105 }
106
107 FInterruptionContext(double InTickStartTime, double InMaxRunningTime, ShouldExitEarlyCallbackType& Callback)
108 : TickStartTime(InTickStartTime)
109 , MaxRunningTime(InMaxRunningTime)
110 , EarlyExitCallback(Callback) {}
111
112 void SetEarlyExitCallback(const ShouldExitEarlyCallbackType& InCallback)
113 {
114 EarlyExitCallback = InCallback;
115 }
116 void SetUnlimitedTickTime()
117 {
118 TickStartTime = -1.;
119 MaxRunningTime = -1.;
120 }
121 void SetLimitedTickTime(double InTickStartTime, double InMaxRunningTime)
122 {
123 TickStartTime = InTickStartTime;
124 MaxRunningTime = InMaxRunningTime;
125 }
126 double GetTickStartTime() const { return TickStartTime; }
127 bool IsTimeSlicingEnabled() const { return TickStartTime > 0; }
128 bool WasInterrupted() const { return OutInterrupted; }
129 bool ShouldExitEarly();
130 void RequestEarlyExit() { OutInterrupted = true; }
131
132private:
133 // A negative value disables time slicing
134 double TickStartTime = -1.;
135 // The maximum time should allow before interruption. If TickStartTime is negative, this is ignored
136 double MaxRunningTime = -1.;
137 // If provided, this is always checked
138 ShouldExitEarlyCallbackType EarlyExitCallback;
139 // True if we ran out of time, EarlyExitCallback returned true, or RequestEarlyExit() was called
140 bool OutInterrupted = false;
141};
