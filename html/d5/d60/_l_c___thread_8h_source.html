92template <typename F, typename... Args>
93inline Thread::Handle Thread::CreateFromFunction(const char* threadName, unsigned int stackSize, F ptrToFunction, Args&&... args)
94{
95 // a helper lambda that calls the given functions with the provided arguments.
96 // because this lambda needs to capture, it cannot be converted to a function pointer implicitly,
97 // and therefore cannot be used as a thread function directly.
98 // however, we can solve this by another indirection.
99 auto captureLambda = [threadName, ptrToFunction, args...]() -> Thread::ReturnValue
100 {
101 Thread::Current::SetName(threadName);
102
103 return (*ptrToFunction)(args...);
104 };
105
106 // helper to make the following easier to read
107 typedef decltype(captureLambda) CaptureLambdaType;
108
109 // the lambda with captures will be called in the thread about to be created, hence it
110 // must be allocated on the heap.
111 CaptureLambdaType* lambdaOnHeap = new CaptureLambdaType(captureLambda);
112
113 // here's the trick: we generate another capture-less lambda that has the same signature as a thread function,
114 // and internally cast the given object to its original type, calling the lambda with captures from within
115 // this capture-less lambda.
116 auto capturelessLambda = [](void* lambdaContext) -> unsigned int
117 {
118 CaptureLambdaType* lambdaOnHeap = static_cast<CaptureLambdaType*>(lambdaContext);
119 const Thread::ReturnValue result = (*lambdaOnHeap)();
120 delete lambdaOnHeap;
121
122 return +result;
123 };
124
125 Thread::Function threadFunction = capturelessLambda;
126 Thread::Handle handle = Thread::Create(stackSize, threadFunction, lambdaOnHeap);
127
128 return handle;
129}
