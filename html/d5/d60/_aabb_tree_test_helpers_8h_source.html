65 // Adds a new node the the node list that is the parent of the given two nodes.
66 // This automatically computes the aabb of the parent node from the two children and links them child/parent indices up.
67 inline int32 BuildAndAppendNode(TArray<FAabbTreeNode>& Nodes, const int32 LeftIndex, const int32 RightIndex)
68 {
69 const int32 ParentIndex = Nodes.Emplace();
70
71 FAabbTreeNode& LeftNode = Nodes[LeftIndex];
72 FAabbTreeNode& RightNode = Nodes[RightIndex];
73 FAabbTreeNode& ParentNode = Nodes[ParentIndex];
74 ParentNode.Left = LeftIndex;
75 ParentNode.Right = RightIndex;
76 ParentNode.Aabb = AabbTreeAlgorithm::Union(RightNode.Aabb, LeftNode.Aabb);
77 LeftNode.Parent = ParentIndex;
78 RightNode.Parent = ParentIndex;
79#ifdef UE_ENABLE_CHAOS_AABBTREE_VISUALIZER
80 ParentNode.DebugNodeList = &Nodes;
81#endif
82 return ParentIndex;
83 }

85 inline int32 BuildSimpleTree(TArray<FAabbTreeNode>& Nodes)
86 {
87 if (Nodes.IsEmpty())
88 {
89 return INDEX_NONE;
90 }
91
92 // Build the initial list of nodes to start grouping (the leaves)
93 TArray<int32> NodeIndices;
94 for (int32 I = 0; I < Nodes.Num(); ++I)
95 {
96 NodeIndices.Add(I);
97 }
98
99 // Do a simple grouping of pairs until there's one node left
100 while (NodeIndices.Num() != 1)
101 {
102 TArray<int32> NewNodeIndices;
103 for (int32 I = 1; I < NodeIndices.Num(); I += 2)
104 {
105 int32 NewNodeIndex = BuildAndAppendNode(Nodes, NodeIndices[I - 1], NodeIndices[I]);
106 NewNodeIndices.Add(NewNodeIndex);
107 }
108 // If there's a remaining node, just carry it over to the next level
109 if (NodeIndices.Num() % 2 == 1)
110 {
111 NewNodeIndices.Add(NodeIndices.Last());
112 }
113 NodeIndices = NewNodeIndices;
114 }
115 return Nodes.Num() - 1;
116 }
