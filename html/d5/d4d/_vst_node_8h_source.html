35template<class NodeType, bool AllowNull, class AllocatorType, typename... AllocatorArgsType>
36struct TNodePtrG final
37{
38private:
39 // Store as non-const internally so Deallocate works with void*.
40 // Const-correctness is enforced through the public interface (operator->, operator*, Get).
41 using TMutableNodeType = std::remove_const_t<NodeType>;
42 static_assert(std::is_empty_v<AllocatorType>, "TNodePtrG only supports stateless allocators");
43
44 TMutableNodeType* _Object;
45
46 // Construct from raw pointer, incrementing its reference count.
47 // Used only by New() and MakeSharedFromThis().
48 explicit TNodePtrG(TMutableNodeType* InObject) : _Object(InObject)
49 {
50 if (InObject) { InObject->Reference(); }
51 }
52
53 void ReleaseObject()
54 {
55 if (_Object->Dereference())
56 {
57 _Object->~TMutableNodeType();
58 AllocatorType::Deallocate(_Object);
59 }
60 }

1376 // Conditional flow control with failure context in test condition.
1377 //
1378 // Children are Clause block nodes in the following order:
1379 // - if_identifier ]
1380 // - condition |- Repeating
1381 // - [then_body] ]
1382 // - [else_body] -- Optional last node
1383 //
1384 // If Identifier blocks - must be present and is simply an empty Clause meant to
1385 // hold information about comments surrounding the 'if' identifier in the
1386 // original source code
1387 // Condition blocks - must be present (they cannot be omitted - there must be at
1388 // least 1) and they must have 1 or more expressions where 1 or more of the
1389 // expressions can fail. Any local variables within it's top scope are made
1390 // available to any immediately following then block.
1391 // Then blocks - are optional (they can be omitted) and when present must follow a
1392 // conditional block and may have zero (can be empty) or more expressions.
1393 // Else block - is optional (it can be omitted) and when present must follow a
1394 // conditional or then block (it cannot be the only block), must be the last
1395 // block and it may have zero (can be empty) or more expressions.
1396 //
1397 // Chained `else if` are automatically flattened into a single FlowIf node with
1398 // multiple condition/[then] block pairs followed by an optional else block.
1399 //
1400 // An `if` may be used as an expression with a result if all the control flow paths
1401 // have a common result - i.e. it will have a result if every condition is paired
1402 // with a then block and there is an ending else block and all the then blocks and
1403 // else block have a most common result type.
