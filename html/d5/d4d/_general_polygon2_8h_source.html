90 bool AddHole(TPolygon2<T> Hole, bool bCheckContainment = true, bool bCheckOrientation = true)
91 {
92 if (bCheckContainment)
93 {
94 if (!Outer.Contains(Hole))
95 {
96 return false;
97 }
98
99 for (const TPolygon2<T>& ExistingHole : Holes)
100 {
101 if (Hole.Overlaps(ExistingHole))
102 {
103 return false;
104 }
105 }
106 }
107
108
109 if (bCheckOrientation)
110 {
111 bool bHoleIsClockwise = Hole.IsClockwise();
112 if (bOuterIsCW == bHoleIsClockwise)
113 {
114 return false;
115 }
116 }
117
118 Holes.Add(Hole);
119 return true;
120 }

316 double DistanceSquared(TVector2<T> p, int &iHoleIndex, int &iNearSeg, double &fNearSegT) const
317 {
318 iNearSeg = iHoleIndex = -1;
319 fNearSegT = TMathUtil<T>::MaxReal;
320 double dist = Outer.DistanceSquared(p, iNearSeg, fNearSegT);
321 for (int i = 0; i < Holes.Num(); ++i )
322 {
323 int seg; double segt;
324 double holedist = Holes[i].DistanceSquared(p, seg, segt);
325 if (holedist < dist)
326 {
327 dist = holedist;
328 iHoleIndex = i;
329 iNearSeg = seg;
330 fNearSegT = segt;
331 }
332 }
333 return dist;
334 }
