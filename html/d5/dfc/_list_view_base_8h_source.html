451 TSharedRef<ITableRow> HandleGenerateRow(ItemType Item, const TSharedRef<STableViewBase>& OwnerTable)
452 {
453 TSubclassOf<UUserWidget> DesiredEntryClass = GetDesiredEntryClassForItem(Item);
454
455 UUserWidget& EntryWidget = OnGenerateEntryWidgetInternal(Item, DesiredEntryClass, OwnerTable);
456
457 // Combine the desired entry padding with the padding the widget wants natively on the CDO.
458 const FMargin DefaultPadding = EntryWidget.GetClass()->GetDefaultObject<UUserWidget>()->GetPadding();
459 EntryWidget.SetPadding(DefaultPadding + GetDesiredEntryPadding(Item));
460
461 TSharedPtr<SWidget> CachedWidget = EntryWidget.GetCachedWidget();
462 CachedWidget->SetCanTick(true); // this is a hack to force ticking to true so selection works (which should NOT require ticking! but currently does)
463 return StaticCastSharedPtr<SObjectTableRow<ItemType>>(CachedWidget).ToSharedRef();
464 }
