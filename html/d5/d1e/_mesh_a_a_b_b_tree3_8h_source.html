235 // bEarlyStop causes the traversal to stop as soon as any triangle closer than NearestDistSqr is found
236 // @return true if an early stop triangle was found
237 template<bool bEarlyStop = false>
238 bool find_nearest_tri(int IBox, const FVector3d& P, double& NearestDistSqr, int& TID, const FQueryOptions& Options) const
239 {
240 int idx = BoxToIndex[IBox];
241 if (idx < TrianglesEnd)
242 { // triangle-list case, array is [N t1 t2 ... tN]
243 int num_tris = IndexList[idx];
244 for (int i = 1; i <= num_tris; ++i)
245 {
246 int ti = IndexList[idx + i];
247 if (Options.TriangleFilterF != nullptr && Options.TriangleFilterF(ti) == false)
248 {
249 continue;
250 }
251 double fTriDistSqr = TMeshQueries<TriangleMeshType>::TriDistanceSqr(*Mesh, ti, P);
252 if (fTriDistSqr < NearestDistSqr)
253 {
254 NearestDistSqr = fTriDistSqr;
255 TID = ti;
256 if constexpr (bEarlyStop)
257 {
258 return true;
259 }
260 }
261 }
262 }
263 else
264 { // internal node, either 1 or 2 child boxes
265 int iChild1 = IndexList[idx];
266 if (iChild1 < 0)
267 { // 1 child, descend if nearer than cur min-dist
268 iChild1 = (-iChild1) - 1;
269 double fChild1DistSqr = BoxDistanceSqr(iChild1, P);
270 if (fChild1DistSqr <= NearestDistSqr)
271 {
272 bool bFoundEarly = find_nearest_tri<bEarlyStop>(iChild1, P, NearestDistSqr, TID, Options);
273 if (bEarlyStop && bFoundEarly)
274 {
275 return true;
276 }
277 }
278 }
279 else
280 { // 2 children, descend closest first
281 iChild1 = iChild1 - 1;
282 int iChild2 = IndexList[idx + 1] - 1;
283
284 double fChild1DistSqr = BoxDistanceSqr(iChild1, P);
285 double fChild2DistSqr = BoxDistanceSqr(iChild2, P);
286 if (fChild1DistSqr < fChild2DistSqr)
287 {
288 if (fChild1DistSqr < NearestDistSqr)
289 {
290 bool bFoundEarly1 = find_nearest_tri<bEarlyStop>(iChild1, P, NearestDistSqr, TID, Options);
291 if (bEarlyStop && bFoundEarly1)
292 {
293 return true;
294 }
295 if (fChild2DistSqr < NearestDistSqr)
296 {
297 bool bFoundEarly2 = find_nearest_tri<bEarlyStop>(iChild2, P, NearestDistSqr, TID, Options);
298 if (bEarlyStop && bFoundEarly2)
299 {
300 return true;
301 }
302 }
303 }
304 }
305 else
306 {
307 if (fChild2DistSqr < NearestDistSqr)
308 {
309 bool bFoundEarly1 = find_nearest_tri<bEarlyStop>(iChild2, P, NearestDistSqr, TID, Options);
310 if (bEarlyStop && bFoundEarly1)
311 {
312 return true;
313 }
314 if (fChild1DistSqr < NearestDistSqr)
315 {
316 bool bFoundEarly2 = find_nearest_tri<bEarlyStop>(iChild1, P, NearestDistSqr, TID, Options);
317 if (bEarlyStop && bFoundEarly2)
318 {
319 return true;
320 }
321 }
322 }
323 }
324 }
325 }
326 return false;
327 }

538 template<typename RayType = FRay3d>
539 void FindHitTriangle(
540 int IBox, const RayType& Ray, double& NearestT, int& TID, FVector3d& BaryCoords,
541 const FQueryOptions& Options = FQueryOptions()) const
542 {
543 int idx = BoxToIndex[IBox];
544 if (idx < TrianglesEnd)
545 { // triangle-list case, array is [N t1 t2 ... tN]
546 FTriangle3d Triangle;
547 int num_tris = IndexList[idx];
548 for (int i = 1; i <= num_tris; ++i)
549 {
550 int ti = IndexList[idx + i];
551 if (Options.TriangleFilterF != nullptr && Options.TriangleFilterF(ti) == false)
552 {
553 continue;
554 }
555
556 Mesh->GetTriVertices(ti, Triangle.V[0], Triangle.V[1], Triangle.V[2]);
557 TIntrRay3Triangle3<double, RayType> Query(Ray, Triangle);
558 if (Query.Find())
559 {
560 if (Query.RayParameter < NearestT)
561 {
562 NearestT = Query.RayParameter;
563 TID = ti;
564 BaryCoords = Query.TriangleBaryCoords;
565 }
566 }
567 }
568 }
569 else
570 { // internal node, either 1 or 2 child boxes
571 double e = FMathd::ZeroTolerance;
572
573 int iChild1 = IndexList[idx];
574 if (iChild1 < 0)
575 { // 1 child, descend if nearer than cur min-dist
576 iChild1 = (-iChild1) - 1;
577 double fChild1T = box_ray_intersect_t(iChild1, Ray);
578 if (fChild1T <= NearestT + e)
579 {
580 FindHitTriangle(iChild1, Ray, NearestT, TID, BaryCoords, Options);
581 }
582 }
583 else
584 { // 2 children, descend closest first
585 iChild1 = iChild1 - 1;
586 int iChild2 = IndexList[idx + 1] - 1;
587
588 double fChild1T = box_ray_intersect_t(iChild1, Ray);
589 double fChild2T = box_ray_intersect_t(iChild2, Ray);
590 if (fChild1T < fChild2T)
591 {
592 if (fChild1T <= NearestT + e)
593 {
594 FindHitTriangle(iChild1, Ray, NearestT, TID, BaryCoords, Options);
595 if (fChild2T <= NearestT + e)
596 {
597 FindHitTriangle(iChild2, Ray, NearestT, TID, BaryCoords, Options);
598 }
599 }
600 }
601 else
602 {
603 if (fChild2T <= NearestT + e)
604 {
605 FindHitTriangle(iChild2, Ray, NearestT, TID, BaryCoords, Options);
606 if (fChild1T <= NearestT + e)
607 {
608 FindHitTriangle(iChild1, Ray, NearestT, TID, BaryCoords, Options);
609 }
610 }
611 }
612 }

732 // TODO: RayTriangleTest does most of the work of Query.Find(); test if it's faster to just do Query.Find() directly.

886 virtual void TreeTraversalImpl(
887 int IBox, int Depth, FTreeTraversal& Traversal, const FQueryOptions& Options
888 ) const
889 {
890 int idx = BoxToIndex[IBox];
891
892 if (idx < TrianglesEnd)
893 {
894 Traversal.BeginBoxTrianglesF(IBox, Depth);
895

897 int n = IndexList[idx];

899 {
900 int ti = IndexList[idx + i];
901 if (Options.TriangleFilterF != nullptr && Options.TriangleFilterF(ti) == false)
902 {
903 continue;
904 }
905 Traversal.NextTriangleF(ti);
906 }
907
908 Traversal.EndBoxTrianglesF(IBox);
909 }
910 else
911 {
912 int i0 = IndexList[idx];
913 if (i0 < 0)
914 {
915 // negative index means we only have one 'child' Box to descend into
916 i0 = (-i0) - 1;
917 if (Traversal.NextBoxF(GetBox(i0), Depth + 1))
918 {
919 TreeTraversalImpl(i0, Depth + 1, Traversal, Options);
920 }
921 }
922 else
923 {
924 // positive index, two sequential child Box indices to descend into
925 i0 = i0 - 1;
926 if (Traversal.NextBoxF(GetBox(i0), Depth + 1))
927 {
928 TreeTraversalImpl(i0, Depth + 1, Traversal, Options);
929 }
930
931 int i1 = IndexList[idx + 1] - 1;
932 if (Traversal.NextBoxF(GetBox(i1), Depth + 1))
933 {
934 TreeTraversalImpl(i1, Depth + 1, Traversal, Options);
935 }
936 }

1394 int SplitTriSetMidpoint(
1395 TArray<int>& Triangles,
1396 TArray<FVector3d>& Centers,
1397 int IStart, int ICount, int Depth, int MinTriCount,
1398 FBoxesSet& Tris, FBoxesSet& Nodes, FAxisAlignedBox3d& Box)
1399 {
1400 Box = (Triangles.Num() > 0) ?
1401 FAxisAlignedBox3d::Empty() : FAxisAlignedBox3d(FVector3d::Zero(), 0.0);
1402 int IBox = -1;
1403
1404 if (ICount <= MinTriCount)
1405 {
1406 // append new Triangles Box
1407 IBox = Tris.IBoxCur++;
1408 Tris.BoxToIndex.InsertAt(Tris.IIndicesCur, IBox);
1409
1410 Tris.IndexList.InsertAt(ICount, Tris.IIndicesCur++);
1411 for (int i = 0; i < ICount; ++i)
1412 {
1413 Tris.IndexList.InsertAt(Triangles[IStart + i], Tris.IIndicesCur++);
1414 Box.Contain(TMeshQueries<TriangleMeshType>::GetTriBounds(*Mesh, Triangles[IStart + i]));
1415 }
1416
1417 Tris.BoxCenters.InsertAt(Box.Center(), IBox);
1418 Tris.BoxExtents.InsertAt(Box.Extents(), IBox);
1419
1420 return -(IBox + 1);
1421 }
1422
1423 //compute interval along an axis and find midpoint
1424 int axis = GetSplitAxis(Depth, Box);
1425 FInterval1d interval = FInterval1d::Empty();
1426 for (int i = 0; i < ICount; ++i)
1427 {
1428 interval.Contain(Centers[IStart + i][axis]);
1429 }
1430 double midpoint = interval.Center();
1431
1432 int n0, n1;
1433 if (interval.Length() > FMathd::ZeroTolerance)
1434 {
1435 // we have to re-sort the Centers & Triangles lists so that Centers < midpoint
1436 // are first, so that we can recurse on the two subsets. We walk in from each side,
1437 // until we find two out-of-order locations, then we swap them.
1438 int l = 0;
1439 int r = ICount - 1;
1440 while (l < r)
1441 {
1442 // Float rounding can collapse midpoint onto an extremum, in which case
1443 // every value matches one of the conditions; the scans must be bounded.
1444 while (l < ICount && Centers[IStart + l][axis] <= midpoint)
1445 {
1446 l++;
1447 }
1448 while (r >= 0 && Centers[IStart + r][axis] > midpoint)
1449 {
1450 r--;
1451 }
1452 if (l >= r)
1453 {
1454 break; //done!
1455 //swap
1456 }
1457 FVector3d tmpc = Centers[IStart + l];
1458 Centers[IStart + l] = Centers[IStart + r];
1459 Centers[IStart + r] = tmpc;
1460 int tmpt = Triangles[IStart + l];
1461 Triangles[IStart + l] = Triangles[IStart + r];
1462 Triangles[IStart + r] = tmpt;
1463 }
1464
1465 n0 = l;
1466 n1 = ICount - n0;
1467 if (n0 == 0 || n0 == ICount)
1468 {
1469 // Partition degenerated to a single side; fall back to half-split.
1470 n0 = ICount / 2;
1471 n1 = ICount - n0;
1472 }
1473 checkSlow(n0 >= 1 && n1 >= 1);
1474 }
1475 else
1476 {
1477 // interval is near-empty, so no point trying to do sorting, just split half and half
1478 n0 = ICount / 2;
1479 n1 = ICount - n0;
1480 }
1481
1482 // create child boxes
1483 FAxisAlignedBox3d box1;
1484 int child0 = SplitTriSetMidpoint(Triangles, Centers, IStart, n0, Depth + 1, MinTriCount, Tris, Nodes, Box);
1485 int child1 = SplitTriSetMidpoint(Triangles, Centers, IStart + n0, n1, Depth + 1, MinTriCount, Tris, Nodes, box1);
1486 Box.Contain(box1);
1487
1488 // append new Box
1489 IBox = Nodes.IBoxCur++;
1490 Nodes.BoxToIndex.InsertAt(Nodes.IIndicesCur, IBox);
1491
1492 Nodes.IndexList.InsertAt(child0, Nodes.IIndicesCur++);
1493 Nodes.IndexList.InsertAt(child1, Nodes.IIndicesCur++);
1494

1502 void find_nearest_triangles(
1503 int iBox, TMeshAABBTree3& OtherTree, const TFunction<FVector3d(const FVector3d&)>& TransformF,
1504 int oBox, int depth, double &nearest_sqr, FIndex2i &nearest_pair,
1505 const FQueryOptions& Options, const FQueryOptions& OtherTreeOptions
1506 ) const
1507 {
1508 int idx = BoxToIndex[iBox];
1509 int odx = OtherTree.BoxToIndex[oBox];
1510
1511 if (idx < TrianglesEnd && odx < OtherTree.TrianglesEnd)
1512 {
1513 // ok we are at triangles for both trees, do triangle-level testing
1514 FTriangle3d Tri, otri;
1515 int num_tris = IndexList[idx], onum_tris = OtherTree.IndexList[odx];
1516
1517 FDistTriangle3Triangle3d dist;
1518
1519 // outer iteration is "other" tris that need to be transformed (more expensive)
1520 for (int j = 1; j <= onum_tris; ++j)
1521 {
1522 int tj = OtherTree.IndexList[odx + j];
1523 if (OtherTreeOptions.TriangleFilterF != nullptr && OtherTreeOptions.TriangleFilterF(tj) == false)
1524 {
1525 continue;
1526 }
1527 OtherTree.Mesh->GetTriVertices(tj, otri.V[0], otri.V[1], otri.V[2]);
1528 if (TransformF != nullptr)
1529 {
1530 otri.V[0] = TransformF(otri.V[0]);
1531 otri.V[1] = TransformF(otri.V[1]);
1532 otri.V[2] = TransformF(otri.V[2]);
1533 }
1534 dist.SetTriangle(0, otri);
1535
1536 // inner iteration over "our" triangles
1537 for (int i = 1; i <= num_tris; ++i)
1538 {
1539 int ti = IndexList[idx + i];
1540 if (Options.TriangleFilterF != nullptr && Options.TriangleFilterF(ti) == false)
1541 {
1542 continue;
1543 }
1544 Mesh->GetTriVertices(ti, Tri.V[0], Tri.V[1], Tri.V[2]);
1545 dist.SetTriangle(1, Tri);
1546 double dist_sqr = dist.GetSquared();
1547 if (dist_sqr < nearest_sqr)
1548 {
1549 nearest_sqr = dist_sqr;
1550 nearest_pair = FIndex2i(ti, tj);
1551 }
1552 }
1553 }
1554
1555 return;
1556 }
1557
1558 // we either descend "our" tree or the other tree
1559 // - if we have hit triangles on "our" tree, we have to descend other
1560 // - if we hit triangles on "other", we have to descend ours
1561 // - otherwise, we alternate at each depth. This produces wider
1562 // branching but is significantly faster (~10x) for both hits and misses
1563 bool bDescendOther = (idx < TrianglesEnd || depth % 2 == 0);
1564 if (bDescendOther && odx < OtherTree.TrianglesEnd)
1565 {
1566 bDescendOther = false; // can't
1567 }
1568
1569 if (bDescendOther)
1570 {
1571 // ok we reached triangles on our side but we need to still reach triangles on
1572 // the other side, so we descend "their" children
1573 FAxisAlignedBox3d bounds = GetBox(iBox);
1574
1575 int oChild1 = OtherTree.IndexList[odx];
1576 if (oChild1 < 0) // 1 child, descend if nearer than cur min-dist
1577 {
1578 oChild1 = (-oChild1) - 1;
1579 FAxisAlignedBox3d oChild1Box = OtherTree.GetBox(oChild1, TransformF);
1580 if (oChild1Box.DistanceSquared(bounds) < nearest_sqr)
1581 {
1582 find_nearest_triangles(iBox, OtherTree, TransformF, oChild1, depth + 1, nearest_sqr, nearest_pair, Options, OtherTreeOptions);
1583 }
1584 }
1585 else // 2 children
1586 {
1587 oChild1 = oChild1 - 1;
1588 int oChild2 = OtherTree.IndexList[odx + 1] - 1;
1589
1590 FAxisAlignedBox3d oChild1Box = OtherTree.GetBox(oChild1, TransformF);
1591 FAxisAlignedBox3d oChild2Box = OtherTree.GetBox(oChild2, TransformF);
1592
1593 // descend closer box first
1594 double d1Sqr = oChild1Box.DistanceSquared(bounds);
1595 double d2Sqr = oChild2Box.DistanceSquared(bounds);
1596 if (d2Sqr < d1Sqr)
1597 {
1598 if (d2Sqr < nearest_sqr)
1599 {
1600 find_nearest_triangles(iBox, OtherTree, TransformF, oChild2, depth + 1, nearest_sqr, nearest_pair, Options, OtherTreeOptions);
1601 }
1602 if (d1Sqr < nearest_sqr)
1603 {
1604 find_nearest_triangles(iBox, OtherTree, TransformF, oChild1, depth + 1, nearest_sqr, nearest_pair, Options, OtherTreeOptions);
1605 }
1606 }
1607 else
1608 {
1609 if (d1Sqr < nearest_sqr)
1610 {
1611 find_nearest_triangles(iBox, OtherTree, TransformF, oChild1, depth + 1, nearest_sqr, nearest_pair, Options, OtherTreeOptions);
1612 }
1613 if (d2Sqr < nearest_sqr)
1614 {
1615 find_nearest_triangles(iBox, OtherTree, TransformF, oChild2, depth + 1, nearest_sqr, nearest_pair, Options, OtherTreeOptions);
1616 }
1617 }
1618 }
1619 }
1620 else
1621 {
1622 // descend our tree nodes if they intersect w/ current bounds of other tree
1623 FAxisAlignedBox3d oBounds = OtherTree.GetBox(oBox, TransformF);
1624
1625 int iChild1 = IndexList[idx];
1626 if (iChild1 < 0) // 1 child, descend if nearer than cur min-dist
1627 {
1628 iChild1 = (-iChild1) - 1;
1629 if (box_box_distsqr(iChild1, oBounds) < nearest_sqr)
1630 {
1631 find_nearest_triangles(iChild1, OtherTree, TransformF, oBox, depth + 1, nearest_sqr, nearest_pair, Options, OtherTreeOptions);
1632 }
1633 }
1634 else // 2 children
1635 {
1636 iChild1 = iChild1 - 1;
1637 int iChild2 = IndexList[idx + 1] - 1;
1638
1639 // descend closer box first
1640 double d1Sqr = box_box_distsqr(iChild1, oBounds);
1641 double d2Sqr = box_box_distsqr(iChild2, oBounds);
1642 if (d2Sqr < d1Sqr)
1643 {
1644 if (d2Sqr < nearest_sqr)
1645 {
1646 find_nearest_triangles(iChild2, OtherTree, TransformF, oBox, depth + 1, nearest_sqr, nearest_pair, Options, OtherTreeOptions);
1647 }
1648 if (d1Sqr < nearest_sqr)
1649 {
1650 find_nearest_triangles(iChild1, OtherTree, TransformF, oBox, depth + 1, nearest_sqr, nearest_pair, Options, OtherTreeOptions);
1651 }
1652 }
1653 else
1654 {
1655 if (d1Sqr < nearest_sqr)
1656 {
1657 find_nearest_triangles(iChild1, OtherTree, TransformF, oBox, depth + 1, nearest_sqr, nearest_pair, Options, OtherTreeOptions);
1658 }
1659 if (d2Sqr < nearest_sqr)
1660 {
1661 find_nearest_triangles(iChild2, OtherTree, TransformF, oBox, depth + 1, nearest_sqr, nearest_pair, Options, OtherTreeOptions);
1662 }
1663 }
1664 }
1665 }

1709 const FQueryOptions& Options
1710 ) const

1712 int idx = BoxToIndex[iBox];

1714 {

1716 int num_tris = IndexList[idx];
1717 for (int i = 1; i <= num_tris; ++i)
1718 {

1729 }
1730 }

1741

1745 iChild1 = iChild1 - 1;

1904 // helper for find_self_intersections that checks intersections between triangles from separate boxes
1905 bool find_self_intersections_acrossboxes(
1906 int Box1, int Box2, MeshIntersection::FIntersectionsQueryResult* Result,
1907 bool bIgnoreTopoConnected, int depth,
1908 TFunctionRef<bool(FIntrTriangle3Triangle3d&)> IntersectFn,
1909 const FQueryOptions& Options
1910 ) const
1911 {
1912 bool bFound = false;
1913 if (Box1 < 0)
1914 {
1915 return false;
1916 }
1917
1918 int Box1Idx = BoxToIndex[Box1];
1919 int Box2Idx = Box2 > -1 ? BoxToIndex[Box2] : -1; // Box2Idx allowed to be invalid
1920
1921 // at leaf
1922 if (Box1Idx < TrianglesEnd && Box2Idx < TrianglesEnd)
1923 {
1924 if (Box2 == -1)
1925 {
1926 return false;
1927 }
1928
1929 for (int IdxA = Box1Idx + 1, Box1End = Box1Idx + 1 + IndexList[Box1Idx]; IdxA < Box1End; IdxA++)
1930 {
1931 int TID_A = IndexList[IdxA];
1932 bFound = find_tri_tri_intersections(TID_A, Box2Idx + 1, Box2Idx + 1 + IndexList[Box2Idx], Result, bIgnoreTopoConnected, IntersectFn, Options) || bFound;
1933 if (!Result && bFound)
1934 {
1935 return true;
1936 }
1937 }
1938 return bFound;
1939 }
1940
1941 // alternate which box we descend, while still making sure Box1 is not a leaf
1942 // (the alternating part is just a heuristic that I guess may help performance; it's not necessary)
1943 if (Box1Idx < TrianglesEnd || (Box2Idx >= TrianglesEnd && depth % 2 == 1))
1944 {
1945 Swap(Box1, Box2);
1946 Swap(Box1Idx, Box2Idx);
1947 }
1948
1949
1950 // if Box2 is present, we *only* care about the intersection of Box1's children w/ Box2
1951 // if Box2 is not present, we handle all the other cases (descending into the children of box1, and checking if the children intersect)
1952 if (Box2 > -1)
1953 {

1955
1956 // Descend through Box1
1957 int iChild1 = IndexList[Box1Idx];
1958 if (iChild1 < 0) // 1 child
1959 {
1960 iChild1 = (-iChild1) - 1;
1961

1965 if (!Result && bFound)
1966 {
1967 return true;
1968 }
1969 }
1970 }
1971 else // 2 children
1972 {
1973 iChild1 = iChild1 - 1;
1974 int iChild2 = IndexList[Box1Idx + 1] - 1;
1975
1976 if (Box2 > -1 && box_box_intersect(iChild1, Box2Box))
1977 {
1978 bFound = find_self_intersections_acrossboxes(iChild1, Box2, Result, bIgnoreTopoConnected, depth + 1, IntersectFn, Options) || bFound;
1979 if (!Result && bFound)
1980 {
1981 return true;
1982 }
1983 }
1984 if (Box2 > -1 && box_box_intersect(iChild2, Box2Box))
1985 {
1986 bFound = find_self_intersections_acrossboxes(iChild2, Box2, Result, bIgnoreTopoConnected, depth + 1, IntersectFn, Options) || bFound;
1987 if (!Result && bFound)
1988 {
1989 return true;
1990 }
1991 }
1992 }
1993 }
1994 else // Box2 is not present
1995 {
1996 // Descend through Box1
1997 int iChild1 = IndexList[Box1Idx];
1998 if (iChild1 < 0) // 1 child
1999 {
2000 iChild1 = (-iChild1) - 1;
2001
2002 bFound = find_self_intersections_acrossboxes(iChild1, -1, Result, bIgnoreTopoConnected, depth + 1, IntersectFn, Options) || bFound;
2003 if (!Result && bFound)
2004 {
2005 return true;

2011 int iChild2 = IndexList[Box1Idx + 1] - 1;

2024 FAxisAlignedBox3d Child2Box = GetBox(iChild2);
2025
2026 if (box_box_intersect(iChild1, Child2Box))

2028 bFound = find_self_intersections_acrossboxes(iChild1, iChild2, Result, bIgnoreTopoConnected, depth + 1, IntersectFn, Options) || bFound;
2029 if (!Result && bFound)
2030 {
2031 return true;
2032 }
2033 }
2034 }
2035 }

2037 return bFound;

2040 // helper for find_self_intersections that intersects a single triangle vs a range of triangles
2041 bool find_tri_tri_intersections(int TID_A, int IdxRangeStart, int IdxRangeEnd,
2042 MeshIntersection::FIntersectionsQueryResult* Result,
2043 bool bIgnoreTopoConnected,

2046 ) const

2050 return false;

2055 Mesh->GetTriVertices(TID_A, TriA.V[0], TriA.V[1], TriA.V[2]);
2056 FIntrTriangle3Triangle3d Intr;

2113 static void AddTriTriIntersectionResult(const FIntrTriangle3Triangle3d& Intr, int TID_A, int TID_B, MeshIntersection::FIntersectionsQueryResult& Result)
2114 {
2115 if (Intr.Quantity == 1)
2116 {
2117 Result.Points.Add(MeshIntersection::FPointIntersection{ {TID_A, TID_B}, Intr.Points[0] });
2118 }
2119 else if (Intr.Quantity == 2)
2120 {
2121 Result.Segments.Add(MeshIntersection::FSegmentIntersection{ {TID_A, TID_B}, {Intr.Points[0], Intr.Points[1]} });
2122 }
2123 else if (Intr.Quantity > 2)
2124 {
2125 if (Intr.Type == EIntersectionType::MultiSegment)
2126 {
2127 Result.Segments.Add(MeshIntersection::FSegmentIntersection{ {TID_A, TID_B}, {Intr.Points[0], Intr.Points[1]} });
2128 Result.Segments.Add(MeshIntersection::FSegmentIntersection{ {TID_A, TID_B}, {Intr.Points[2], Intr.Points[3]} });
