7//
8// RHI Descriptor Range - Experimental feature, these types may completely change in the future.
9//
10// Abstraction around allocating a contiguous range of descriptors and loading them in shaders. The descriptor range will manage updating the platform specific
11// bindless descriptor heap with its resources. When bound to a shader, the offset into the bindless descriptor heap will be available in the shader.
12//
13// Type Mask
14// Due to limitations on certain platforms, not all resource/view types can be in the same descriptor range. The types that can be used together are exposed
15// via 'GRHIGlobals.DescriptorTypeGroups'. Each element in the DescriptorTypeGroups array corresponds to a set of descriptor types (via mask) that can be
16// used together. Given how nuanced some of the descriptor types are, this may create friction in some uses of descriptor ranges.
17//
18// To make type mask support easier, consider using different descriptor ranges for read-only vs read-write resources.
19//
20// Resource Lifetimes
21// The Descriptor Range will keep the resources assigned to it alive until the descriptor range is destroyed.
22//
23// Support checks for this feature:
24// Shader support: FDataDrivenShaderPlatformInfo::GetSupportsDescriptorRange
25// Shader define: PLATFORM_SUPPORTS_DESCRIPTOR_RANGE
26// Runtime: GRHIGlobals.bSupportsDescriptorRange
27// (These must be checked in addition to the normal bindless checks)
28//
29// Creation:
30// FRHIDescriptorRangeDesc Desc{ Count, TypeMask };
31// TArray<FRHIDescriptorResource> Resources = ...;
32// FDescriptorRangeRHIRef DescriptorRange = RHICmdList.CreateDescriptorRange(Desc, Resources);
33//
34// Updating:
35// TArray<FRHIDescriptorResource> Resources = ...;
36// RHICmdList.UpdateDescriptorRange(DescriptorRange, Offset, Resources);
37//
38// Shader binding:
39// SHADER_PARAMETER_DESCRIPTOR_RANGE(FDescriptorRange, DescriptorRange)
40//
41// Shader access:
42// Texture2D MyTexture = GetDescriptorRangeSRV(Texture2D, DescriptorRange, Offset);
43//
44
45struct FRHIDescriptorRangeDesc
46{
47 // Number of descriptors in the range.
48 uint32 Count = 0;
49
50 // Mask of different descriptor types in the range. Types cannot cross groups exposed in GRHIGlobals.DescriptorTypeGroups
51 ERHIDescriptorTypeMask TypeMask = ERHIDescriptorTypeMask::None;
52};
