68 FVertexInputStream() :
69 StreamIndex(0),
70 Offset(0),
71 bStreamSourceSlot(0),
72 VertexBuffer(nullptr)
73 {}
74
75 FVertexInputStream(uint32 InStreamIndex, uint32 InOffset, TYPE_OF_NULLPTR)
76 : StreamIndex(InStreamIndex), Offset(InOffset), bStreamSourceSlot(0), VertexBuffer(nullptr)
77 {
78 // Verify no overflow
79 checkSlow(InStreamIndex == StreamIndex && InOffset == Offset);
80 }
81
82 FVertexInputStream(uint32 InStreamIndex, uint32 InOffset, FRHIStreamSourceSlot* InStreamSourceSlot)
83 : StreamIndex(InStreamIndex), Offset(InOffset), bStreamSourceSlot(1), StreamSourceSlot(InStreamSourceSlot)
84 {
85 // Verify no overflow
86 checkSlow(InStreamIndex == StreamIndex && InOffset == Offset);
87 }
88
89 FVertexInputStream(uint32 InStreamIndex, uint32 InOffset, FRHIBuffer* InVertexBuffer)
90 : StreamIndex(InStreamIndex), Offset(InOffset), bStreamSourceSlot(0), VertexBuffer(InVertexBuffer)
91 {
92 // Verify no overflow
93 checkSlow(InStreamIndex == StreamIndex && InOffset == Offset);
94 }
95
96 inline bool operator==(const FVertexInputStream& rhs) const
97 {
98 if (StreamIndex != rhs.StreamIndex ||
99 Offset != rhs.Offset ||
100 Pointer != rhs.Pointer)
101 {
102 return false;
103 }
104
105 return true;
106 }
107
108 inline bool operator!=(const FVertexInputStream& rhs) const
109 {
110 return !(*this == rhs);
111 }
