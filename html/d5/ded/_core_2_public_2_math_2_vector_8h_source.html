1338template<typename T>
1339inline TVector<T> TVector<T>::RotateAngleAxisRad(const T AngleRad, const TVector<T>& Axis) const
1340{
1341 T S, C;
1342 FMath::SinCos(&S, &C, AngleRad);
1343
1344 const T XX = Axis.X * Axis.X;
1345 const T YY = Axis.Y * Axis.Y;
1346 const T ZZ = Axis.Z * Axis.Z;
1347
1348 const T XY = Axis.X * Axis.Y;
1349 const T YZ = Axis.Y * Axis.Z;
1350 const T ZX = Axis.Z * Axis.X;
1351
1352 const T XS = Axis.X * S;
1353 const T YS = Axis.Y * S;
1354 const T ZS = Axis.Z * S;
1355
1356 const T OMC = 1.f - C;
1357
1358 return TVector<T>(
1359 (OMC * XX + C) * X + (OMC * XY - ZS) * Y + (OMC * ZX + YS) * Z,
1360 (OMC * XY + ZS) * X + (OMC * YY + C) * Y + (OMC * YZ - XS) * Z,
1361 (OMC * ZX - YS) * X + (OMC * YZ + XS) * Y + (OMC * ZZ + C) * Z
1362 );
1363}

2192template<typename T>
2193T TVector<T>::EvaluateBezier(const TVector<T>* ControlPoints, int32 NumPoints, TArray<TVector<T>>& OutPoints)
2194{

2196 check(NumPoints >= 2);
2197
2198 // var q is the change in t between successive evaluations.
2199 const T q = 1.f / (T)(NumPoints - 1); // q is dependent on the number of GAPS = POINTS-1
2200
2201 // recreate the names used in the derivation
2202 const TVector<T>& P0 = ControlPoints[0];
2203 const TVector<T>& P1 = ControlPoints[1];
2204 const TVector<T>& P2 = ControlPoints[2];
2205 const TVector<T>& P3 = ControlPoints[3];
2206
2207 // coefficients of the cubic polynomial that we're FDing -
2208 const TVector<T> a = P0;
2209 const TVector<T> b = 3 * (P1 - P0);
2210 const TVector<T> c = 3 * (P2 - 2 * P1 + P0);
2211 const TVector<T> d = P3 - 3 * P2 + 3 * P1 - P0;
2212
2213 // initial values of the poly and the 3 diffs -
2214 TVector<T> S = a; // the poly value
2215 TVector<T> U = b * q + c * q * q + d * q * q * q; // 1st order diff (quadratic)
2216 TVector<T> V = 2 * c * q * q + 6 * d * q * q * q; // 2nd order diff (linear)
2217 TVector<T> W = 6 * d * q * q * q; // 3rd order diff (constant)
2218
2219 // Path length.
2220 T Length = 0;
2221
2222 TVector<T> OldPos = P0;
2223 OutPoints.Add(P0); // first point on the curve is always P0.
2224
2225 for (int32 i = 1; i < NumPoints; ++i)
2226 {
2227 // calculate the next value and update the deltas
2228 S += U; // update poly value
2229 U += V; // update 1st order diff value
2230 V += W; // update 2st order diff value
2231 // 3rd order diff is constant => no update needed.
2232
2233 // Update Length.
2234 Length += TVector<T>::Dist(S, OldPos);
2235 OldPos = S;
2236
2237 OutPoints.Add(S);
2238 }

2241 return Length;
