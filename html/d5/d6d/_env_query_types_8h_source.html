94 Constant
95 // What other curve shapes should be supported? At first I was thinking we'd have parametric (F*V^P + C), but
96 // many versions of that curve would violate the [0, 1] output range which I think we should preserve. So instead
97 // I think we should define these by "curve shape". I'm not sure if we need to allow full tweaks to the curves,
98 // such as supporting other "Exponential" curves (positive even powers). However, I think it's likely that we'll
99 // want to support "smooth LERP" / S-shaped curve of the form 2x^3 - 3x^2, and possibly a "sideways" version of
100 // the same S-curve. We also might want to allow "Sine" curves, basically adjusted to match the range and then
101 // simply offset by some amount to allow a peak or valley in the middle or on the ends. (Four Sine options are
102 // probably sufficient.) I'm not sure if Sine is really needed though, so probably we should only add it if
103 // there's a need identified. One other curve shape we might want is "Square Root", which might optionally
104 // support any positive fractional power (if we also supported any positive even number for an "Exponential"
105 // type.
106 };

482struct FEnvQueryItem
483{
485 float Score;
486
488 int32 DataOffset:31;
489
491 uint32 bIsDiscarded:1;
492
493 inline bool IsValid() const { return DataOffset >= 0 && !bIsDiscarded; }
494 inline void Discard() { bIsDiscarded = true; }
495
496 bool operator<(const FEnvQueryItem& Other) const
497 {
498 // sort by validity
499 if (IsValid() != Other.IsValid())
500 {
501 // self not valid = less important
502 return !IsValid();
503 }
504
505 // sort by score if not equal. As last resort sort by DataOffset to achieve stable sort.
506 return Score != Other.Score ? Score < Other.Score : DataOffset < Other.DataOffset;
507 }
508
509 FEnvQueryItem() : Score(0.0f), DataOffset(-1), bIsDiscarded(false) {}
510 FEnvQueryItem(int32 InOffset) : Score(0.0f), DataOffset(InOffset), bIsDiscarded(false) {}
511};
