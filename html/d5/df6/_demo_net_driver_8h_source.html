80 UPROPERTY()

82
83 void CountBytes(FArchive& Ar) const
84 {
85 if (FRepState const * const LocalRepState = RepState.Get())
86 {
87 Ar.CountBytes(sizeof(FRepState), sizeof(FRepState));
88 LocalRepState->CountBytes(Ar);
89 }
90
91 SubObjRepState.CountBytes(Ar);
92 for (const auto& SubObjRepStatePair : SubObjRepState)
93 {
94 SubObjRepStatePair.Key.CountBytes(Ar);
95
96 if (FRepState const * const LocalRepState = SubObjRepStatePair.Value.Get())
97 {
98 const SIZE_T SizeOfRepState = sizeof(FRepState);
99 Ar.CountBytes(SizeOfRepState, SizeOfRepState);
100 LocalRepState->CountBytes(Ar);
101 }
102 }
103
104 ObjReferences.CountBytes(Ar);
105 }
106};
