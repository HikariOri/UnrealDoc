692 void Lerp(const TBaseBlendedCurve<AllocatorA, ElementTypeA>& A, const TBaseBlendedCurve<AllocatorB, ElementTypeB>& B, float Alpha)
693 {
694 CURVE_PROFILE_CYCLE_COUNTER(TBaseBlendedCurve_Lerp);
695
696 if (!FAnimWeight::IsRelevant(FMath::Abs(Alpha)))
697 {
698 // if blend is all the way for child1, then just copy its bone atoms
699 Override(A);
700 }
701 else if (!FAnimWeight::IsRelevant(FMath::Abs(Alpha - 1.0f)))
702 {
703 // if blend is all the way for child2, then just copy its bone atoms
704 Override(B);
705 }
706 else
707 {
708 // Combine using Lerp. Result is a merged set of curves in 'this'.

710 {
711 OutResult.Value = FMath::Lerp(InElement0.Value, InElement1.Value, Alpha);
712 OutResult.Flags = InElement0.Flags | InElement1.Flags;
713 });
