66 FReal PhiWithNormal(const FVec3& x, FVec3& OutNormal) const
67 {
68 const FVec3f OriginToX = x - Origin;
69 const FRealSingle DistanceAlongMajorAxis = FMath::Clamp(FVec3f::DotProduct(OriginToX, MajorAxis), -HalfLength, HalfLength);
70 const FRealSingle DistanceAlongMinorAxis = FMath::Clamp(FVec3f::DotProduct(OriginToX, MinorAxis), -HalfWidth, HalfWidth);
71 const FVec3f ClosestPoint = Origin + MajorAxis * DistanceAlongMajorAxis + MinorAxis * DistanceAlongMinorAxis;
72 const FRealSingle Radius = (HalfLength > UE_SMALL_NUMBER) ? FMath::Lerp(Radius2, Radius1, .5f*(DistanceAlongMajorAxis / HalfLength) + .5f) : FMath::Max(Radius1, Radius2);
73 OutNormal = (x - ClosestPoint);
74 return OutNormal.SafeNormalize() - Radius;
75 }
