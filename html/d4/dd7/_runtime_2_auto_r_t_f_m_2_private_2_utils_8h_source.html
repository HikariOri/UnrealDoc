19// methods or functions to prevent AutoRTFM closed function(s) from being

130// Rounds Value down to the next multiple of Alignment.
131// Alignment must be a power of two.
132template <typename T>
133inline constexpr T AlignDown(T Value, T Alignment)
134{
135 static_assert(std::is_integral_v<T>);
136 return Value & ~(Alignment - 1);
137}
138
139// Rounds Value up to the next multiple of Alignment.
140// Alignment must be a power of two.
141template <typename T>
142inline constexpr T AlignUp(T Value, T Alignment)
143{
144 static_assert(std::is_integral_v<T>);
145 return (Value + Alignment - 1) & ~(Alignment - 1);
146}
147
148// Rounds Value down to the next multiple of Multiple.
149// Unlike AlignDown(), Multiple does not need to be a power of two.
150template <typename T>
151inline constexpr T RoundDown(T Value, T Multiple)
152{
153 static_assert(std::is_integral_v<T>);
154 return (Value / Multiple) * Multiple;
155}
156
157// Rounds Value up to the next multiple of Multiple.
158// Unlike AlignUp(), Multiple does not need to be a power of two.
159template <typename T>
160inline constexpr T RoundUp(T Value, T Multiple)
161{
162 static_assert(std::is_integral_v<T>);
163 return RoundDown(Value + Multiple - 1, Multiple);
164}

169inline constexpr T Log2(T Value)
170{
171 constexpr size_t NumBits = sizeof(Value) * 8;
172 return Value > 0 ? NumBits - __builtin_clzg(Value) - 1 : 0;

175// RAII helper to assign a value to a variable for the lifetime of the
176// FScopedAssignment, restoring the original value when the FScopedAssignment is
177// destructed.
178template <typename T>
179struct TScopedAssignment
180{
181 inline TScopedAssignment(T& Variable, T NewValue) : Variable(Variable), OldValue(Variable)
182 {
183 Variable = NewValue;
184 }
185
186 inline ~TScopedAssignment()
187 {
188 Variable = OldValue;
189 }
190
191private:
192 TScopedAssignment(const TScopedAssignment&) = delete;
193 TScopedAssignment(TScopedAssignment&&) = delete;
194 TScopedAssignment& operator=(const TScopedAssignment&) = delete;
195 TScopedAssignment& operator=(TScopedAssignment&&) = delete;
196 T& Variable;
197 T OldValue;
198};

325// This is all a bit funky - but for good reason! We want `AUTORTFM_ASSERT`
326// to be as close to *zero* cost if the assert wouldn't trigger.
327// We will always pay the cost of the unlikely branch, but we need
328// to make the body of taking the assert happen in another function
329// so as not to affect codegen, register allocation, and stack
330// reservation. But we also want the `AUTORTFM_ASSERT` to give us accurate
331// `__FILE__` and `__LINE__` information for the line that it was
332// triggered on. So what we do is have a lambda at the line that
333// actually does the assert (but crucially gets the correct file
334// and line numbers), but then pass this to another function
335// `DoAssert` which has a special calling convention that makes
336// the caller as optimal as possible (at the expense of the callee).
337#define AUTORTFM_ASSERT(exp) \
338 if (AUTORTFM_UNLIKELY(!(exp))) \
339 { \
340 auto Lambda = [] \
341 { \
342 AUTORTFM_FATAL("AUTORTFM_ASSERT(%s) failure", #exp); \
343 }; \
344 AutoRTFM::DoAssert(Lambda); \
345 }
346
347// For places where zero cost is truly important, AUTORTFM_ASSERT_DEBUG will
348// disappear entirely in Shipping and Test builds. We still evaluate the expression,
349// on the off-chance that we are given an expression with side effects, and rely on
350// the compiler to optimize away a dead expression.
351#if AUTORTFM_BUILD_SHIPPING || AUTORTFM_BUILD_TEST
352#define AUTORTFM_ASSERT_DEBUG(exp) ((void)AUTORTFM_LIKELY(static_cast<bool>(exp)))
353#else
354#define AUTORTFM_ASSERT_DEBUG(exp) \
355 if (AUTORTFM_UNLIKELY(!(exp))) \
356 { \
357 auto Lambda = [] \
358 { \
359 AUTORTFM_FATAL("AUTORTFM_ASSERT_DEBUG(%s) failure", #exp); \
360 }; \
361 AutoRTFM::DoAssert(Lambda); \
362 }
363#endif
364
365// Same funkiness as AUTORTFM_ASSERT, except that it doesn't cause a fatal error.
366// TODO: Maybe upgrade the Info -> Warning?
367#define AUTORTFM_EXPECT(exp) \
368 if (AUTORTFM_UNLIKELY(!(exp))) \
369 { \
370 auto Lambda = [] \
371 { \
372 ::AutoRTFM::LogWithCallstack(autortfm_log_info, "AUTORTFM_EXPECT(%s) failure", #exp); \
373 }; \
374 AutoRTFM::DoExpect(Lambda); \
375 }
