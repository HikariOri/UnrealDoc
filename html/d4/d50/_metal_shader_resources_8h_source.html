154inline FArchive& FMetalShaderBindings::Serialize(FArchive& Ar, FShaderResourceTable& SRT)
155{

158 Ar << ConstantBuffers;

161 if (ArgumentBuffers)

163 Ar << ArgumentBufferMasks;
