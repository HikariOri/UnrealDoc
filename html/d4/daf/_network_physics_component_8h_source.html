301 virtual int32 ReceiveNewData(Chaos::FBaseRewindHistory& NewData, const int32 FrameOffset, bool CompareDataForRewind = false, const bool bImportant = false, int32 TryInjectAtFrame = INDEX_NONE) override
302 {
303 TNetRewindHistory& NetNewData = static_cast<TNetRewindHistory&>(NewData);
304
305 int32 RewindFrame = INDEX_NONE;
306 DataType* ReceiveData = &NetNewData.GetAndLoadEarliestData();
307
308 for (int32 Itr = 0; Itr < NetNewData.GetHistorySize(); Itr++)
309 {
310 DataType* NextReceiveData = NetNewData.GetAndLoadNextIncrementalData();
311
312 ReceiveData->bImportant = bImportant;
313 ReceiveData->bReceivedData = true; // Received data is marked to differentiate from locally predicted data
314
315 ReceiveData->LocalFrame = ReceiveData->ServerFrame - FrameOffset;
316
317 if (ShouldRecordReceivedDataOnFrame(*ReceiveData, NextReceiveData))
318 {
319 if (CompareDataForRewind && ReceiveData->LocalFrame > RewindFrame && TriggerRewindFromNewData(*ReceiveData))
320 {
321 RewindFrame = ReceiveData->LocalFrame;
322 }
323
324 Super::RecordData(ReceiveData->LocalFrame, ReceiveData);
325 }
326
327 if (NextReceiveData)
328 {
329 ReceiveData = NextReceiveData;
330 }
331 else
332 {
333 // Record a copy of the last data at specified frame if the history doesn't have data for that frame yet
334 if (TryInjectAtFrame > Super::GetLatestFrame())
335 {
336#if DEBUG_NETWORK_PHYSICS
337 UE_LOGF(LogChaos, Log, "SERVER | PT | Input Buffer Empty, Injecting Received Input at frame %d || LocalFrame = %d || ServerFrame = %d || bDataAltered = %d || Data: %ls"
338 , TryInjectAtFrame, ReceiveData.LocalFrame, ReceiveData.ServerFrame, ReceiveData.bDataAltered, *ReceiveData.DebugData());
339#endif
340
341 DataType InjectData = *ReceiveData;
342 InjectData.bReceivedData = false;
343 InjectData.bDataAltered = true;
344 InjectData.LocalFrame = TryInjectAtFrame;
345 InjectData.ServerFrame = TryInjectAtFrame + FrameOffset;
346
347 Super::RecordData(TryInjectAtFrame, &InjectData);
348 }
349
350 break;
351 }
352 }
353
354 return RewindFrame;

359 virtual bool ShouldRecordReceivedDataOnFrame(const DataType& ReceivedData, DataType* NextReceivedData = nullptr)
360 {
361 if (ReceivedData.LocalFrame < 0)
362 {
363 return false;
364 }
365
366 // Get the cached data at the index slot for the received data
367 Super::LoadData(ReceivedData.LocalFrame);
368 DataType& Data = Super::DataHistory[Super::CurrentIndex];
369
370 if (Data.LocalFrame < ReceivedData.LocalFrame)
371 {
372 // Allow recording the received data if it's newer than the already recorded data on this index
373 return true;
374 }
375 else if (!Data.bReceivedData && Data.LocalFrame == ReceivedData.LocalFrame)
376 {
377 // If the data exist but is not marked as received and marked as altered, the server has produced this input via extrapolation or interpolation, don't overwrite it with data from the client.
378 if (Data.bDataAltered)
379 {
380 // If we have a newer received data (since we receive multiple data at the same time) merge this data into the newer which can either get recorded or injected at the front of the input buffer on the server.
381 if (NextReceivedData != nullptr)
382 {
383 NextReceivedData->MergeData(ReceivedData);
384 NextReceivedData->bDataAltered = true;
385 }
386
387 // Mark the already cached data as received so that we don't perform this merge again when receiving redundant inputs / states
388 Data.bReceivedData = true;
389 return false;
390 }
391 else
392 {
393 // Allow recording the received data if we already have data for the same frame cached, not marked as altered, meaning it was predicted data on the client
394 return true;
395 }
396 }
397 return false;
398 }

404 {
405 return !NewData.CompareData(Super::DataHistory[Super::CurrentIndex]);
406 }
407
408 return false;

417 virtual void NetSerialize(FArchive& Ar, UPackageMap* InPackageMap, TUniqueFunction<void(void* Data, const int32 DataIndex)> DataSetupFunction) override
418 {
419 bool bOneEntry = Super::NumFrames == 1;
420 if (Ar.IsSaving())
421 {
422 Ar.SerializeBits(&bOneEntry, 1);
423
424 if (!bOneEntry)
425 {
426 uint32 NumFramesUnsigned = static_cast<uint32>(Super::NumFrames);
427
428 constexpr uint32 NumBitsNumFramesUnsigned = FMath::CeilLogTwo(NetworkPhysicsComponentConstants::MaxNumberOfElementsToNetwork);
429 Ar.SerializeBits(&NumFramesUnsigned, NumBitsNumFramesUnsigned);
430
431 Super::NumFrames = static_cast<int32>(NumFramesUnsigned);
432 }
433 else
434 {
435 Super::NumFrames = 1;
436 }
437
438 if (Super::NumFrames > GetMaxArraySize() || Super::NumFrames <= 0)
439 {
440 UE_LOGF(LogChaos, Warning, "TNetRewindHistory: serialized array of size %d exceeds maximum size %d.", Super::NumFrames, GetMaxArraySize());
441 Ar.SetError();
442 return;
443 }
444
445 // Note that we use a for-loop to iterate over all entries instead of relying on GetAndLoadNextIncrementalData because the data can have Frame 0 which makes incremental not work.
446 DataType* LastData = nullptr;
447 DataType* CurrentData = &Super::GetAndLoadEarliestData();
448 for (int32 Itr = 0; Itr < Super::GetHistorySize(); Itr++)
449 {
450 // Set the implementation component pointer and stateful delta serialization source
451 if (DataSetupFunction)
452 {
453 DataSetupFunction(CurrentData, Itr);
454 }
455
456 // Set the internal delta serialization source (between data entries in the collection)
457 if constexpr (bLegacyData)
458 {
459 if (LastData)
460 {
461 CurrentData->SetDeltaSourceData(LastData);
462 }
463 }
464
465 NetSerializeData(*CurrentData, Ar, InPackageMap);
466
467 // Get next incremental data
468 if (DataType* NextData = Super::GetAndLoadNextData())
469 {
470 LastData = CurrentData;
471 CurrentData = NextData;
472 }
473 else
474 {
475 ensureMsgf(false, TEXT("TNetRewindHistory: Serialization failed to get the next data."));
476 }
477 }
478
479 for (DataType& Data : Super::DataHistory)
480 {
481 // Clear delta source and strong pointer to implementation component after serialization
482 if constexpr (bLegacyData)
483 {
484 Data.ClearImplementationComponent();
485 Data.ClearDeltaSourceData();
486 }
487 }
488 }
489 else // IsLoading
490 {
491 Ar.SerializeBits(&bOneEntry, 1);
492
493 if (!bOneEntry)
494 {
495 uint32 NumFramesUnsigned = static_cast<uint32>(Super::NumFrames);
496
497 constexpr uint32 NumBitsNumFramesUnsigned = FMath::CeilLogTwo(NetworkPhysicsComponentConstants::MaxNumberOfElementsToNetwork);
498 Ar.SerializeBits(&NumFramesUnsigned, NumBitsNumFramesUnsigned);
499
500 Super::NumFrames = static_cast<int32>(NumFramesUnsigned);
501 }
502 else
503 {
504 Super::NumFrames = 1;
505 }
506
507 if (Super::NumFrames > GetMaxArraySize())
508 {
509 UE_LOGF(LogChaos, Warning, "TNetRewindHistory: serialized array of size %d exceeds maximum size %d.", Super::NumFrames, GetMaxArraySize());
510 Ar.SetError();
511 return;
512 }
513
514 Super::DataHistory.SetNum(Super::NumFrames);
515
516 for (int32 I = 0; I < Super::DataHistory.Num(); I++)
517 {
518 DataType& Data = Super::DataHistory[I];
519
520 // Set the implementation component pointer and stateful delta serialization source
521 if (DataSetupFunction)
522 {
523 DataSetupFunction(&Data, I);
524 }
525
526 // Set the internal delta serialization source (between data entries in the collection)
527 if constexpr (bLegacyData)
528 {
529 if (I > 0)
530 {
531 Data.SetDeltaSourceData(&Super::DataHistory[I - 1]);
532 }
533 }
534
535 NetSerializeData(Data, Ar, InPackageMap);
536 }
537
538 for (DataType& Data : Super::DataHistory)
539 {
540 // Clear delta source and strong pointer to implementation component after serialization
541 if constexpr (bLegacyData)
542 {
543 Data.ClearImplementationComponent();
544 Data.ClearDeltaSourceData();
545 }
546 }
547 }
548
549 Super::Initialize();
550 }

1275 {
1276 int32 FrameIndex = INDEX_NONE;
1277 int32 EntryIndex = INDEX_NONE;
1278 int32 LocalFrame = INDEX_NONE;
1279 };
1280
1281 TArray<FPendingReplicationLocation, TInlineAllocator<16>> PendingReplicationLocations;
1282
1283 for (int32 FrameIndex = 0; FrameIndex < FrameHistory.Num(); ++FrameIndex)
1284 {
1285 const FActionHistoryFrameData& FrameData = FrameHistory[FrameIndex];
1286
1287 if (FrameData.LocalFrame == INDEX_NONE)
1288 {
1289 continue;
1290 }
1291
1292 for (int32 EntryIndex = 0; EntryIndex < FrameData.Entries.Num(); ++EntryIndex)
1293 {
1294 const FActionHistoryEntry& Entry = FrameData.Entries[EntryIndex];
1295
1296 if (Entry.bHasBeenQueuedForReplication == true)
1297 {
1298 continue;
1299 }
1300 if (bOnlyActiveActions == true && Entry.bActive == false)
1301 {
1302 continue;
1303 }
1304 if (Entry.ActionInstance.IsValid() == false)
1305 {
1306 continue;
1307 }
1308
1309 FPendingReplicationLocation& Location = PendingReplicationLocations.Emplace_GetRef();
1310 Location.FrameIndex = FrameIndex;
1311 Location.EntryIndex = EntryIndex;
1312 Location.LocalFrame = FrameData.LocalFrame;
1313 }
1314 }
1315
1316 // The circular buffer storage order is not chronological, sort by local frame, this makes delta serialization work better when multiple actions are being replicated
1317 PendingReplicationLocations.Sort([](const FPendingReplicationLocation& A, const FPendingReplicationLocation& B)
1318 {
1319 if (A.LocalFrame != B.LocalFrame)
1320 {
1321 return A.LocalFrame < B.LocalFrame;
1322 }
1323 if (A.FrameIndex != B.FrameIndex)
1324 {
1325 return A.FrameIndex < B.FrameIndex;
1326 }
1327 return A.EntryIndex < B.EntryIndex;
1328 });
1329
1330 for (const FPendingReplicationLocation& Location : PendingReplicationLocations)
1331 {
1332 FActionHistoryFrameData& FrameData = FrameHistory[Location.FrameIndex];
1333 FActionHistoryEntry& Entry = FrameData.Entries[Location.EntryIndex];
1334
1335 if (Entry.bReliable)
1336 {
1337 OutReliableActions.Add(Entry.ActionInstance);
1338 }
1339 else
1340 {
1341 OutUnreliableActions.Add(Entry.ActionInstance);
1342 }
1343
1344 if (bMarkAsQueuedForReplication == true)
1345 {
1346 Entry.bHasBeenQueuedForReplication = true;
1347 }
1348 }
