504 // Called only when we KNOW we are going to do a realloc increasing by 1.
505 // In this case, we know that max == num and can simplify things in a very
506 // hot location in the code.
507 // This returns the old ArrayMax in order to save a register clobber/reload.
508 template <uint32 Flags, typename AllocatorInstanceType>
509 UE_FORCEINLINE_HINT TAllocatorSizeType_T<AllocatorInstanceType> ReallocGrow1_DoAlloc_Impl(
510 uint32 ElementSize,
511 uint32 ElementAlignment,
512 AllocatorInstanceType& AllocatorInstance,
513 TAllocatorSizeType_T<AllocatorInstanceType>& ArrayMax
514 )
515 {
516 using SizeType = TAllocatorSizeType_T<AllocatorInstanceType>;
517 using USizeType = std::make_unsigned_t<SizeType>;
518
519 const USizeType UOldMax = (USizeType)ArrayMax;
520 const USizeType UNewNum = UOldMax + 1U;
521 const SizeType OldMax = (SizeType)UOldMax;
522 const SizeType NewNum = (SizeType)UNewNum;
523
524 // This should only happen when we've underflowed or overflowed SizeType
525 if (NewNum < OldMax)
526 {
527 OnInvalidArrayNum((unsigned long long)UNewNum);
528 }
529
530 SizeType NewMax;
531 if constexpr (!!(Flags & 1)) // TAllocatorTraits<AllocatorType>::SupportsElementAlignment
532 {
533 NewMax = AllocatorInstance.CalculateSlackGrow(NewNum, OldMax, ElementSize, ElementAlignment);
534 AllocatorInstance.ResizeAllocation(UOldMax, NewMax, ElementSize, ElementAlignment);
535 }
536 else
537 {
538 NewMax = AllocatorInstance.CalculateSlackGrow(NewNum, OldMax, ElementSize);
539 AllocatorInstance.ResizeAllocation(UOldMax, NewMax, ElementSize);
540 }
541 ArrayMax = NewMax;
542 #if UE_ENABLE_ARRAY_SLACK_TRACKING
543 if constexpr (!!(Flags & 2)) // TAllocatorTraits<AllocatorType>::SupportsSlackTracking
544 {
545 AllocatorInstance.SlackTrackerLogNum(NewNum);
546 }
547 #endif
548
549 return OldMax;
550 }

575 // This should be used for repeated growing operations when reallocations are to be amortized over multiple inserts.
576 template <uint32 Flags, typename AllocatorInstanceType>
577 FORCENOINLINE TAllocatorSizeType_T<AllocatorInstanceType> ReallocGrow(
578 uint32 ElementSize,
579 uint32 ElementAlignment,
580 TAllocatorSizeType_T<AllocatorInstanceType> Count,
581 AllocatorInstanceType& AllocatorInstance,
582 TAllocatorSizeType_T<AllocatorInstanceType>& ArrayNum,
583 TAllocatorSizeType_T<AllocatorInstanceType>& ArrayMax
584 )
585 {
586 using SizeType = TAllocatorSizeType_T<AllocatorInstanceType>;
587 using USizeType = std::make_unsigned_t<SizeType>;
588
589 const USizeType UCount = (USizeType)Count;
590 const USizeType UOldNum = (USizeType)ArrayNum;
591 const USizeType UOldMax = (USizeType)ArrayMax;
592 const USizeType UNewNum = UOldNum + UCount;
593 const SizeType OldNum = (SizeType)UOldNum;
594 const SizeType OldMax = (SizeType)UOldMax;
595 const SizeType NewNum = (SizeType)UNewNum;
596
597 checkSlow((OldNum >= 0) & (OldMax >= OldNum) & (Count >= 0)); // & for one branch
598
599 ArrayNum = NewNum;
600
601#if DO_GUARD_SLOW
602 if (UNewNum > UOldMax)
603#else
604 // SECURITY - This check will guard against negative counts too, in case the checkSlow above is compiled out.
605 // However, it results in slightly worse code generation.
606 if (UCount > UOldMax - UOldNum)
607#endif
608 {
609 // This should only happen when we've underflowed or overflowed SizeType
610 if (NewNum < OldNum)
611 {
612 OnInvalidArrayNum((unsigned long long)UNewNum);
613 }
614 SizeType NewMax;
615 if constexpr (!!(Flags & 1)) // TAllocatorTraits<AllocatorType>::SupportsElementAlignment
616 {
617 NewMax = AllocatorInstance.CalculateSlackGrow(NewNum, OldMax, ElementSize, ElementAlignment);
618 AllocatorInstance.ResizeAllocation(UOldNum, NewMax, ElementSize, ElementAlignment);
619 }
620 else
621 {
622 NewMax = AllocatorInstance.CalculateSlackGrow(NewNum, OldMax, ElementSize);
623 AllocatorInstance.ResizeAllocation(UOldNum, NewMax, ElementSize);
624 }
625 ArrayMax = NewMax;
626#if UE_ENABLE_ARRAY_SLACK_TRACKING
627 if constexpr (!!(Flags & 2)) // TAllocatorTraits<AllocatorType>::SupportsSlackTracking
628 {
629 AllocatorInstance.SlackTrackerLogNum(NewNum);
630 }
631#endif
632 }
633
634 return OldNum;
635 }

1044 template <typename FromArrayType, typename ToArrayType>
1045 static UE_FORCEINLINE_HINT void MoveOrCopy(ToArrayType& ToArray, FromArrayType& FromArray, SizeType PrevMax)
1046 {
1047 if constexpr (UE::Core::Private::CanMoveTArrayPointersBetweenArrayTypes<FromArrayType, ToArrayType>())
1048 {
1049 // Move
1050
1051 static_assert(std::is_same_v<TArray, ToArrayType>, "MoveOrCopy is expected to be called with the current array type as the destination");
1052
1053 using FromAllocatorType = typename FromArrayType::AllocatorType;
1054 using ToAllocatorType = typename ToArrayType::AllocatorType;
1055
1056 if constexpr (TCanMoveBetweenAllocators<FromAllocatorType, ToAllocatorType>::Value)
1057 {
1058 ToArray.AllocatorInstance.template MoveToEmptyFromOtherAllocator<FromAllocatorType>(FromArray.AllocatorInstance);
1059 }
1060 else
1061 {
1062 ToArray.AllocatorInstance.MoveToEmpty(FromArray.AllocatorInstance);
1063 }
1064
1065 ToArray .ArrayNum = (SizeType)FromArray.ArrayNum;
1066 ToArray .ArrayMax = (SizeType)FromArray.ArrayMax;
1067
1068 // Ensure the destination container could hold the source range (when the allocator size types shrink)
1069 if constexpr (sizeof(USizeType) < sizeof(typename FromArrayType::USizeType))
1070 {
1071 if (ToArray.ArrayNum != FromArray.ArrayNum || ToArray.ArrayMax != FromArray.ArrayMax)
1072 {
1073 // Cast to USizeType first to prevent sign extension on negative sizes, producing unusually large values.
1074 UE::Core::Private::OnInvalidArrayNum((unsigned long long)(USizeType)ToArray.ArrayNum);
1075 }
1076 }
1077
1078 FromArray.ArrayNum = 0;
1079 FromArray.ArrayMax = FromArray.AllocatorInstance.GetInitialCapacity();
1080
1081 FromArray.SlackTrackerNumChanged();
1082 ToArray.SlackTrackerNumChanged();
1083 }
1084 else
1085 {
1086 // Copy
1087
1088 ToArray.CopyToEmpty(FromArray.GetData(), FromArray.Num(), PrevMax);
1089 }
1090 }

1832 void BulkSerialize(FArchive& Ar, bool bForcePerElementSerialization = false)
1833 {
1834 constexpr int32 ElementSize = sizeof(ElementType);
1835 // Serialize element size to detect mismatch across platforms.
1836 int32 SerializedElementSize = ElementSize;
1837 Ar << SerializedElementSize;
1838
1839 if (bForcePerElementSerialization
1840 || (Ar.IsSaving() // if we are saving, we always do the ordinary serialize as a way to make sure it matches up with bulk serialization
1841 && !Ar.IsCooking() // but cooking and transacting is performance critical, so we skip that
1842 && !Ar.IsTransacting())
1843 || Ar.IsByteSwapping() // if we are byteswapping, we need to do that per-element
1844 )
1845 {
1846 Ar << *this;
1847 }
1848 else
1849 {
1850 CountBytes(Ar);
1851 if (Ar.IsLoading())
1852 {
1853 // Basic sanity checking to ensure that sizes match.
1854 if (!ensure(SerializedElementSize == ElementSize))
1855 {
1856 Ar.SetError();
1857 return;
1858 }
1859
1860 // Serialize the number of elements, block allocate the right amount of memory and deserialize

1903 UE_FORCEINLINE_HINT SizeType AddUninitialized()
1904 {
1905 // Begin sensitive code! Single element additions/insertions get inlined _everywhere_ and thus
1906 // punch above their weight with respect to exe size, so we pay a lot of attention to every instruction
1907 // here until we get in to the NOINLINE functions that actually handle the growth.
1908
1909 // Modulo some register allocation and pointer offsets, inlined sites on x64 for the "Tiny" path should look roughly like:
1910 //
1911 // mov eax, dword ptr [array + 8h] // load arraynum
1912 // cmp eax, dword ptr [array + 0Ch] // load arraymax
1913 // jne [to ArrayNum++]
1914 // mov ecx, immediate representing size/alignment
1915 // lea r8, [array + 0Ch]] // array max
1916 // lea rdx, [array] // allocator instance
1917 // call
1918 // lea ecx, [rax+1] // ArrayNum++
1919 // mov dword ptr [array + 8h], ecx // Save ArrayNum
1920 //
1921
1922 // Single cmp, which we can assume because we are adding a single element.
1923 if (ArrayNum == ArrayMax)
1924 {
1925 // Both branches here write the return in to ArrayNum. This is because the function call clobbers the registers
1926 // and if we assign as part of the return into something we need, the compiler doesn't have to reload the data
1927 // into the clobbered register.
1928
1929 // When we can pack size and alignment into a single 16 bit load, we save a parameter setup instruction
1930 // for the function call. The 16 bit part is load-bearing for arm codegen.
1931 // PVS believes this is overwrought because technically sizeof() <= 255 implies alignof <= 255. I think this is far more
1932 // clear and costs nothing.
1933 if constexpr (sizeof(ElementType) <= 255 && alignof(ElementType) <= 255) // -V590
1934 {
1935 // Note that the realloc functions are templated ONLY on allocator instance so they are not duplicated
1936 // in the code for every type!
1937 ArrayNum = UE::Core::Private::ReallocGrow1_DoAlloc_Tiny<UE::Core::Private::GetAllocatorFlags<AllocatorType>()>(sizeof(ElementType) | (alignof(ElementType) << 8), AllocatorInstance, ArrayMax);
1938 }
1939 else
1940 {
1941 ArrayNum = UE::Core::Private::ReallocGrow1_DoAlloc<UE::Core::Private::GetAllocatorFlags<AllocatorType>()>(sizeof(ElementType), alignof(ElementType), AllocatorInstance, ArrayMax);
1942 }
1943 }
1944 // End sensitive code!
1945
1946 SizeType OldArrayNum = ArrayNum;
1947 ArrayNum++;
1948 return OldArrayNum;
1949 }
