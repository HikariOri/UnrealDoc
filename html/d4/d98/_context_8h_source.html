37 // Begins an unscoped transaction.
38 // It is common for a transaction to be started and then immediately aborted
39 // without actually being used. As a performance optimization,
40 // StartTransaction() increments NumDeferredTransactions, which represents
41 // the number of "deferred" transactions at the top of the transaction

43 // the first call to MaterializeDeferredTransactions() or

134 inline void EnteringStaticLocalInitializer()
135 {
136 if (AUTORTFM_LIKELY(GetStatus() == EContextStatus::Idle))
137 {
138 return;
139 }
140
141 if (Status == EContextStatus::OnTrack)
142 {
143 AUTORTFM_ASSERT(0 == StackLocalInitializerDepth);
144 Status = EContextStatus::InStaticLocalInitializer;
145 StackLocalInitializerDepth++;
146 }
147 else if (Status == EContextStatus::InStaticLocalInitializer)
148 {
149 StackLocalInitializerDepth++;
150 }
151 }

153 inline void LeavingStaticLocalInitializer()
154 {
155 if (AUTORTFM_LIKELY(GetStatus() == EContextStatus::Idle))
156 {
157 return;
158 }
159
160 AUTORTFM_ASSERT(Status != EContextStatus::OnTrack);
161
162 if (Status == EContextStatus::InStaticLocalInitializer)
163 {
164 StackLocalInitializerDepth--;
165
166 if (0 == StackLocalInitializerDepth)
167 {
168 Status = EContextStatus::OnTrack;
169 }
170 }
