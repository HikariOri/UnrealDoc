299 virtual void SetGeometryImpl(const int32 Index, const FImplicitObjectPtr& InGeometry) override
300 {
301 // We hit these checks if there's a call to modify geometry without first clearing constraints
302 // on the particle (e.g., PBDRigidsEvolutionGBF::InvalidateParticle)
303 check(MParticleCollisions[Index].Num() == 0);
304 //check(MParticleConstraints[Index].Num() == 0);
305
306 TSimpleGeometryParticles<T, d>::SetGeometryImpl(Index, InGeometry);
307
308 UpdateShapesArray(Index);
309
310 MHasBounds[Index] = (InGeometry && InGeometry->HasBoundingBox());
311 if (MHasBounds[Index])
312 {
313 MLocalBounds[Index] = TAABB<T, d>(InGeometry->BoundingBox());
314
315 // Update the threshold we use to determine when to enable CCD. This is based on the bounds.
316 UpdateCCDAxisThreshold(Index);
317
318 // Update the world-space stat of all the shapes - must be called after UpdateShapesArray
319 // world space inflated bounds needs to take expansion into account - this is done in integrate for dynamics anyway, so
320 // this computation is mainly for statics
321 UpdateWorldSpaceState(Index, TRigidTransform<FReal, 3>(GetX(Index), GetR(Index)), FVec3(0));
322 }
323 }
324
325 void UpdateCCDAxisThreshold(const int32 Index)
326 {
327 // NOTE: We get empty bounds (as opposed to no bounds) if we have Geometry that is an empty Union
328 if (!MHasBounds[Index] || MLocalBounds[Index].IsEmpty())
329 {
330 MCCDAxisThreshold[Index] = FVec3(0);
331 return;
332 }
333
334 if (CVars::CCDAxisThresholdMode == 0)
335 {
336 // Use object extents as CCD axis threshold
337 MCCDAxisThreshold[Index] = MLocalBounds[Index].Extents();
338 }
339 else if (CVars::CCDAxisThresholdMode == 1)
340 {
341 // Use thinnest object extents as all axis CCD thresholds
342 MCCDAxisThreshold[Index] = FVec3(MLocalBounds[Index].Extents().GetMin());
343 }
344 else
345 {
346 // Find minimum shape bounds thickness on each axis
347 FVec3 ThinnestBoundsPerAxis = MLocalBounds[Index].Extents();
348 for (const TUniquePtr<FPerShapeData>& Shape : ShapesArray(Index))
349 {
350 // Only sim-enabled shapes should ever be swept with CCD, so make sure the
351 // sim-enabled flag is on for each shape before considering it's min bounds
352 // for CCD extents.
353 if (Shape->GetSimEnabled() && (CVars::bCCDAxisThresholdUsesProbeShapes || !Shape->GetIsProbe()))
354 {
355 const FImplicitObjectRef Geometry = Shape->GetGeometry();
356 if (Geometry && Geometry->HasBoundingBox())
357 {
358 const TVector<T, d> ShapeExtents = Geometry->BoundingBox().Extents();
359 TVector<T, d>& CCDAxisThreshold = MCCDAxisThreshold[Index];
360 for (int32 AxisIndex = 0; AxisIndex < d; ++AxisIndex)
361 {
362 ThinnestBoundsPerAxis[AxisIndex] = FMath::Min(ShapeExtents[AxisIndex], ThinnestBoundsPerAxis[AxisIndex]);
363 }
364 }
365 }
366 }
367
368 if (CVars::CCDAxisThresholdMode == 2)
369 {
370 // On each axis, use the thinnest shape bound on that axis
371 MCCDAxisThreshold[Index] = ThinnestBoundsPerAxis;
372 }
373 else if (CVars::CCDAxisThresholdMode == 3)
374 {
375 // Find the thinnest shape bound on any axis and use this for all axes
376 MCCDAxisThreshold[Index] = FVec3(ThinnestBoundsPerAxis.GetMin());
377 }
378 }

444
445 void UpdateWorldSpaceState(const int32 Index, const FRigidTransform3& WorldTransform, const FVec3& BoundsExpansion)
446 {
447 const FShapesArray& Shapes = ShapesArray(Index);
448
449 // If we have no shapes we are have a point bounds at our local origin
450 FAABB3 WorldBounds = (Shapes.Num() > 0) ? FAABB3::EmptyAABB() : FAABB3::ZeroAABB();
451
452 // NOTE: Individual shape bounds are not expanded. We only require that the particle bounds is expanded because that is used
453 // in the broadphase. The midphase is what requires the shape bounds and it also has all the information required to expand the shape bounds as needed.
454 for (const auto& Shape : Shapes)
455 {
456 Shape->UpdateWorldSpaceState(WorldTransform);
457 WorldBounds.GrowToInclude(Shape->GetWorldSpaceShapeBounds());
458 }
459
460 MWorldSpaceInflatedBounds[Index] = TAABB<T, d>(WorldBounds).ThickenSymmetrically(BoundsExpansion);
461 ensureMsgf(MWorldSpaceInflatedBounds[Index].IsValid(), TEXT("Invalid world space inflated bounds. %s, BoundsExpansion(%s)"),
462 *MWorldSpaceInflatedBounds[Index].ToString(), *BoundsExpansion.ToString());
463 }
464
465 void UpdateWorldSpaceStateSwept(const int32 Index, const FRigidTransform3& WorldTransform, const FVec3& BoundsExpansion, const FVec3& DeltaX)
466 {
467 // Update the bounds of all shapes (individual shape bounds are not expanded) and accumulate the net bounds
468 UpdateWorldSpaceState(Index, WorldTransform, BoundsExpansion);
469
470 // Apply the swept bounds delta
471 MWorldSpaceInflatedBounds[Index].GrowByVector(DeltaX);
472 ensureMsgf(MWorldSpaceInflatedBounds[Index].IsValid(), TEXT("Invalid world space inflated bounds. %s, BoundsExpansion(%s), DeltaX(%s)"),
473 *MWorldSpaceInflatedBounds[Index].ToString(), *BoundsExpansion.ToString(), *DeltaX.ToString());
474 }

569 virtual void Serialize(FChaosArchive& Ar) override
570 {
571 LLM_SCOPE(ELLMTag::ChaosParticles);
572 TSimpleGeometryParticles<T, d>::Serialize(Ar);
573
574 Ar.UsingCustomVersion(FPhysicsObjectVersion::GUID);
575 if (Ar.CustomVer(FPhysicsObjectVersion::GUID) >= FPhysicsObjectVersion::PerShapeData)
576 {
577 Ar << MShapesArray;
578 }
579
580 if (Ar.CustomVer(FPhysicsObjectVersion::GUID) >= FPhysicsObjectVersion::SerializeGTGeometryParticles)
581 {
582 SerializeGeometryParticleHelper(Ar, this);
583 }
584
585 Ar.UsingCustomVersion(FExternalPhysicsCustomObjectVersion::GUID);
586 if (Ar.CustomVer(FExternalPhysicsCustomObjectVersion::GUID) >= FExternalPhysicsCustomObjectVersion::SerializeParticleBounds)
587 {
588 TBox<FReal, 3>::SerializeAsAABBs(Ar, MLocalBounds);
589 TBox<FReal, 3>::SerializeAsAABBs(Ar, MWorldSpaceInflatedBounds);
590 Ar << MHasBounds;
591
592 if (Ar.CustomVer(FExternalPhysicsCustomObjectVersion::GUID) < FExternalPhysicsCustomObjectVersion::SerializeShapeWorldSpaceBounds)
593 {
594 for (int32 Idx = 0; Idx < MShapesArray.Num(); ++Idx)
595 {
596 UpdateWorldSpaceState(Idx, FRigidTransform3(GetX(Idx), GetR(Idx)), FVec3(0));
597 }
598 }
599 }
600 else
601 {
602 //just assume all bounds come from geometry (technically wrong for pbd rigids with only sample points, but backwards compat is not that important right now)
603 for (int32 Idx = 0; Idx < GetAllGeometry().Num(); ++Idx)
604 {
605 MHasBounds[Idx] = GetGeometry(Idx) && GetGeometry(Idx)->HasBoundingBox();
606 if (MHasBounds[Idx])
607 {
608 MLocalBounds[Idx] = TAABB<T, d>(GetGeometry(Idx)->BoundingBox());
609 //ignore velocity too, really just trying to get something reasonable)
610 UpdateWorldSpaceState(Idx, FRigidTransform3(GetX(Idx), GetR(Idx)), FVec3(0));
611 }
612 }
613 }
614
615 if (Ar.CustomVer(FExternalPhysicsCustomObjectVersion::GUID) < FExternalPhysicsCustomObjectVersion::SpatialIdxSerialized)
616 {
617 MSpatialIdx.AddZeroed(GetAllGeometry().Num());
618 }
619 else
620 {
621 Ar << MSpatialIdx;
622 }
623
624 if (Ar.CustomVer(FExternalPhysicsCustomObjectVersion::GUID) < FExternalPhysicsCustomObjectVersion::SerializeHashResult)
625 {
626 //no longer care about hash so don't read it and don't do anything
627 }
628 }
