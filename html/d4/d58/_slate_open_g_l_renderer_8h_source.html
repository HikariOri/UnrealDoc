125 FMatrix CreateProjectionMatrix( uint32 Width, uint32 Height ) const
126 {
127 // Create ortho projection matrix
128 const float Left = 0.0f;
129 const float Right = Left+Width;
130 const float Top = 0.0f;
131 const float Bottom = Top+Height;
132 const float ZNear = -100.0f;
133 const float ZFar = 100.0f;
134
135 return FMatrix( FPlane(2.0f/(Right-Left), 0, 0, 0 ),
136 FPlane(0, 2.0f/(Top-Bottom), 0, 0 ),
137 FPlane(0, 0, 1/(ZNear-ZFar), 0 ),
138 FPlane((Left+Right)/(Left-Right), (Top+Bottom)/(Bottom-Top), ZNear/(ZNear-ZFar), 1 ) );
139
140 }
