260 inline FVector VRandCone( FVector const& Dir, float ConeHalfAngleRad ) const
261 {
262 if (ConeHalfAngleRad > 0.f)
263 {
264 float const RandU = FRand();
265 float const RandV = FRand();
266
267 // Get spherical coords that have an even distribution over the unit sphere
268 // Method described at http://mathworld.wolfram.com/SpherePointPicking.html
269 float Theta = 2.f * UE_PI * RandU;
270 float Phi = FMath::Acos((2.f * RandV) - 1.f);
271
272 // restrict phi to [0, ConeHalfAngleRad]
273 // this gives an even distribution of points on the surface of the cone
274 // centered at the origin, pointing upward (z), with the desired angle
275 Phi = FMath::Fmod(Phi, ConeHalfAngleRad);
276
277 // get axes we need to rotate around
278 FMatrix const DirMat = FRotationMatrix(Dir.Rotation());
279 // note the axis translation, since we want the variation to be around X
280 FVector const DirZ = DirMat.GetUnitAxis( EAxis::X );
281 FVector const DirY = DirMat.GetUnitAxis( EAxis::Y );
282
283 FVector Result = Dir.RotateAngleAxis(Phi * 180.f / UE_PI, DirY);
284 Result = Result.RotateAngleAxis(Theta * 180.f / UE_PI, DirZ);
285
286 // ensure it's a unit vector (might not have been passed in that way)
287 Result = Result.GetSafeNormal();
288
289 return Result;
290 }
291 else
292 {
293 return Dir.GetSafeNormal();
294 }
295 }

305 inline FVector VRandCone( FVector const& Dir, float HorizontalConeHalfAngleRad, float VerticalConeHalfAngleRad ) const
306 {
307 if ( (VerticalConeHalfAngleRad > 0.f) && (HorizontalConeHalfAngleRad > 0.f) )
308 {
309 float const RandU = FRand();
310 float const RandV = FRand();
311
312 // Get spherical coords that have an even distribution over the unit sphere
313 // Method described at http://mathworld.wolfram.com/SpherePointPicking.html
314 float Theta = 2.f * UE_PI * RandU;
315 float Phi = FMath::Acos((2.f * RandV) - 1.f);
316
317 // restrict phi to [0, ConeHalfAngleRad]
318 // where ConeHalfAngleRad is now a function of Theta
319 // (specifically, radius of an ellipse as a function of angle)
320 // function is ellipse function (x/a)^2 + (y/b)^2 = 1, converted to polar coords
321 float ConeHalfAngleRad = FMath::Square(FMath::Cos(Theta) / VerticalConeHalfAngleRad) + FMath::Square(FMath::Sin(Theta) / HorizontalConeHalfAngleRad);
322 ConeHalfAngleRad = FMath::Sqrt(1.f / ConeHalfAngleRad);
323
324 // clamp to make a cone instead of a sphere
325 Phi = FMath::Fmod(Phi, ConeHalfAngleRad);
326
327 // get axes we need to rotate around
328 FMatrix const DirMat = FRotationMatrix(Dir.Rotation());
329 // note the axis translation, since we want the variation to be around X
330 FVector const DirZ = DirMat.GetUnitAxis( EAxis::X );
331 FVector const DirY = DirMat.GetUnitAxis( EAxis::Y );
332
333 FVector Result = Dir.RotateAngleAxis(Phi * 180.f / UE_PI, DirY);
334 Result = Result.RotateAngleAxis(Theta * 180.f / UE_PI, DirZ);
335
336 // ensure it's a unit vector (might not have been passed in that way)
337 Result = Result.GetSafeNormal();
338
339 return Result;
340 }
341 else
342 {
343 return Dir.GetSafeNormal();
344 }
345 }
