204 NumericType GetValueType(const FString& InString) const
205 {
206 static_assert(std::is_floating_point_v<NumericType>);
207
208 if constexpr (std::is_same_v<float, NumericType>)
209 {
210 return FCString::Atof(*InString);
211 }
212 else if constexpr (std::is_same_v<double, NumericType>)
213 {
214 return FCString::Atod(*InString);
215 }
216 else
217 {
218 return NumericType{};
219 }
220 }
