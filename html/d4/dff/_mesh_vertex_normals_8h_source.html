29 // This is a workaround for some platforms not supporting C++20's fetch_add for floats
30 // UE platforms should be on C++20, so hopefully we can remove this method in the future and just use .fetch_add directly
31 template<typename AtomicFloatType = std::atomic<float>>
32 inline void AtomicFloatFetchAdd(AtomicFloatType& Value, float ToAdd)
33 {
34 constexpr bool bHasAtomicFloatFetchAdd = requires(AtomicFloatType & AtomicFloat, float Param)
35 {
36 AtomicFloat.fetch_add(Param, std::memory_order_relaxed);
37 };
38 if constexpr (bHasAtomicFloatFetchAdd)
39 {
40 Value.fetch_add(ToAdd, std::memory_order_relaxed);
41 }
42 else
43 {
44 // Manual implementation of fetch_add
45 float Old = Value.load();
46 while (!Value.compare_exchange_weak(Old, Old + ToAdd, std::memory_order_relaxed)) {}
47 }
48 }

50 // This is a workaround for some platforms not supporting C++20's std::atomic_ref
51 // UE platforms should be on C++20, so hopefully we can remove this method in the future and just use std::atomic_ref directly
52 // Atomic_Ref is a slight performance improvement for platforms with weak memory order such as ARM as we can avoid a full barrier with
53 // relaxed ordering here.
54 template<typename IntType = int>
55 inline void AtomicIncrement(IntType& Value, std::memory_order MemoryOrder)
56 {
57#if defined(__cpp_lib_atomic_ref)
58 std::atomic_ref<IntType>(Value).fetch_add(1, MemoryOrder);
59#else
60 FPlatformAtomics::InterlockedIncrement(&Value);
61#endif

106 template<typename MeshType, typename NormalsArrayType>
107 void ComputeVertexNormals(const MeshType& Mesh, NormalsArrayType& OutNormals, bool bRequireDeterministicNormals = true)
108 {
109 TRACE_CPUPROFILER_EVENT_SCOPE(GeometryCore::MeshNormals::RecomputeNormals);
110 check(Mesh.MaxVertexID() == OutNormals.Num());
111
112 if (bRequireDeterministicNormals)
113 {
114 TArray<UE::TConsumeAllMpmcQueue<TPair<int, FVector3f>, ParallelHelpers::FVertexNormalLinearAllocator>> VertexNormalsToSum;
115 VertexNormalsToSum.SetNum(Mesh.MaxVertexID());
116
117 {
118 TRACE_CPUPROFILER_EVENT_SCOPE(GeometryCore::MeshNormals::Tri_Normal_Parallel);
119 ParallelFor(Mesh.MaxTriangleID(), [&](int TriangleID)
120 {
121 if (!Mesh.IsTriangle(TriangleID))
122 {
123 return;
124 }
125
126 const FIndex3i Triangle = Mesh.GetTriangle(TriangleID);
127 FVector3d A, B, C;
128 Mesh.GetTriVertices(TriangleID, A, B, C);
129 double TriArea;
130 const FVector3f TriNormal = FVector3f(VectorUtil::NormalArea(A, B, C, TriArea));
131 FVector3f TriWeights = FVector3f(VectorUtil::TriangleInternalAngles(A, B, C) * TriArea);
132
133 for (int VertIndex = 0; VertIndex < 3; ++VertIndex)
134 {
135 const int VertexID = Triangle[VertIndex];
136 const int ElementID = 3 * VertexID;
137
138 const float VertexWeight = TriWeights[VertIndex];
139 const FVector3f Normal = TriNormal * VertexWeight;
140
141 VertexNormalsToSum[VertexID].ProduceItem(TPair<int, FVector3f>(TriangleID, Normal));
142 }
143 });
144 }
145 {
146 TRACE_CPUPROFILER_EVENT_SCOPE(GeometryCore::MeshNormals::Average);
147 ParallelFor(Mesh.MaxVertexID(), [&](int VertexID)
148 {
149 if (!Mesh.IsVertex(VertexID))
150 {
151 return;
152 }
153
154 TArray<TPair<int, FVector3f>, TInlineAllocator<8>> VertexNormals;
155 VertexNormalsToSum[VertexID].ConsumeAllLifo([&](TPair<int, FVector3f>&& Item) { VertexNormals.Add(Item); });
156 VertexNormals.Sort([](const TPair<int, FVector3f>& A, const TPair<int, FVector3f>& B)
157 {
158 return A.Key < B.Key;
159 });
160
161 const FVector3f Normal = Algo::Accumulate(VertexNormals, FVector3f::ZeroVector,
162 [](FVector3f Result, const TPair<int, FVector3f>& Normal) -> FVector3f
163 {
164 return Result + Normal.Value;
165 }
166 );
167
168 OutNormals[VertexID] = FVector3f(Normalized(Normal));
169 });
170 }
171 }
172 else // Non-deterministic fast path which uses unordered float operations which can produce slightly different roundings
173 {
174 TArray<std::atomic<float>> VertexNormals;
175 VertexNormals.SetNumZeroed(Mesh.MaxVertexID() * 3);
176
177 {
178 TRACE_CPUPROFILER_EVENT_SCOPE(GeometryCore::MeshNormals::Tri_Normal_Parallel);
179 ParallelFor(Mesh.MaxTriangleID(), [&](int TriangleID)
180 {
181 if (!Mesh.IsTriangle(TriangleID))
182 {
183 return;
184 }
185
186 const FIndex3i Triangle = Mesh.GetTriangle(TriangleID);
187 FVector3d A, B, C;
188 Mesh.GetTriVertices(TriangleID, A, B, C);
189 double TriArea;
190 const FVector3f TriNormal = FVector3f(VectorUtil::NormalArea(A, B, C, TriArea));
191
192 FVector3f TriWeights = FVector3f(VectorUtil::TriangleInternalAngles(A, B, C) * TriArea);
193
194 for (int VertIndex = 0; VertIndex < 3; ++VertIndex)
195 {
196 const int VertexID = Triangle[VertIndex];
197 const int ElementID = 3 * VertexID;
198
199 const float VertexWeight = TriWeights[VertIndex];
200 const FVector3f Normal = TriNormal * VertexWeight;
201
202 for (int FloatIndex = 0; FloatIndex < 3; ++FloatIndex)
203 {
204 ParallelHelpers::AtomicFloatFetchAdd(VertexNormals[ElementID + FloatIndex], Normal[FloatIndex]);
205 }
206 }
207 });
208 }
209 {
210 TRACE_CPUPROFILER_EVENT_SCOPE(GeometryCore::MeshNormals::Average);
211 ParallelFor(Mesh.MaxVertexID(), [&](int VertexID)
212 {
213 if (!Mesh.IsVertex(VertexID))
214 {
215 return;
216 }
217
218 const FVector3f Normal(VertexNormals[VertexID * 3], VertexNormals[VertexID * 3 + 1], VertexNormals[VertexID * 3 + 2]);
219
220 OutNormals[VertexID] = Normalized(Normal);
221 });
222 }
223 }
224 }

233 template<typename MeshType, typename NormalsArrayType>
234 void ComputeWeldedVertexNormals(const MeshType& Mesh, TFunctionRef<int32(int32)> VertexIDWeldMap, NormalsArrayType& OutNormals, bool bRequireDeterministicNormals = true)
235 {
236 // Minimal mesh wrapper struct to forward triangle IDs to welded IDs
237 struct FWeldMesh
238 {
239 FWeldMesh(const MeshType& InMesh, const TFunctionRef<int32(int32)>& InVertexIDWeldMap) : Mesh(&InMesh), VertexIDWeldMap(&InVertexIDWeldMap){}
240
241 int32 MaxVertexID() const
242 {
243 return Mesh->MaxVertexID();
244 }
245 int32 MaxTriangleID() const
246 {
247 return Mesh->MaxTriangleID();
248 }
249 bool IsTriangle(int32 TID) const
250 {
251 return Mesh->IsTriangle(TID);
252 }
253 bool IsVertex(int32 VID) const
254 {
255 return Mesh->IsVertex(VID);
256 }
257 void GetTriVertices(int32 TID, FVector3d& A, FVector3d& B, FVector3d& C) const
258 {
259 return Mesh->GetTriVertices(TID, A, B, C);
260 }
261 FIndex3i GetTriangle(int32 TID) const
262 {
263 FIndex3i Tri = Mesh->GetTriangle(TID);
264 Tri.A = (*VertexIDWeldMap)(Tri.A);
265 Tri.B = (*VertexIDWeldMap)(Tri.B);
266 Tri.C = (*VertexIDWeldMap)(Tri.C);
267 return Tri;
268 }
269 private:
270 const MeshType* Mesh;
271 const TFunctionRef<int32(int32)>* VertexIDWeldMap;
272 };
273 FWeldMesh WeldMesh(Mesh, VertexIDWeldMap);
274 ComputeVertexNormals(WeldMesh, OutNormals, bRequireDeterministicNormals);
275 // Copy welded IDs back out
276 ParallelFor(Mesh.MaxVertexID(), [&OutNormals, &VertexIDWeldMap](int32 Idx)
277 {
278 int32 WeldIdx = VertexIDWeldMap(Idx);
279 if (WeldIdx > INDEX_NONE && WeldIdx != Idx)
280 {
281 OutNormals[Idx] = OutNormals[WeldIdx];
282 }
283 });
284 }
