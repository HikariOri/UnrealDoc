167 // structure to do an amortized update of a set of update elements in a (conceptual) array
168 struct FAmortizedUpdate
169 {
170 // the first index to update this tick (inclusive)
171 int32 FirstIndex = 0;
172
173 // the last index to update this tick (exclusive)
174 int32 LastIndex = 0;
175
176 // Call at the beginning of the update tick to set up how many updates you want to run this tick
177 // UpdateElementCount is the total number of update elements (to wrap when we hit the end of the possibility space)
178 void StartUpdateTick(int32 UpdateElementCount, int32 MaxUpdatesThisFrame)
179 {
180 check(MaxUpdatesThisFrame >= 0);
181 if (MaxUpdatesThisFrame >= UpdateElementCount)
182 {
183 // if we are updating more than what we have, then just update everything
184 FirstIndex = 0;
185 LastIndex = UpdateElementCount;
186 }
187 else
188 {
189 FirstIndex = LastIndex;
190 if (FirstIndex >= UpdateElementCount)
191 {
192 FirstIndex = 0;
193 }
194 check(FirstIndex >= 0);
195 LastIndex = FirstIndex + MaxUpdatesThisFrame;
196 }
197 }
198
199 // return true if the given element should update
200 bool ShouldUpdate(int32 UpdateElementIndex)
201 {
202 if ((UpdateElementIndex >= FirstIndex) && (UpdateElementIndex < LastIndex))
203 {
204 return true;
205 }
206 return false;
207 }
208
209 // call if an update element is deleted from the array, to update the valid ranges
210 void HandleDeletion(int32 DeletedUpdateElementIndex)
211 {
212 if (DeletedUpdateElementIndex < FirstIndex)
213 {
214 FirstIndex--;
215 }
216 if (DeletedUpdateElementIndex < LastIndex)
217 {
218 LastIndex--;
219 }
220
221 }
222 };
