100/*-----------------------------------------------------------------------------
101 * TCallTraits
102 *
103 * Same call traits as boost, though not with as complete a solution.
104 *
105 * The main member to note is ParamType, which specifies the optimal
106 * form to pass the type as a parameter to a function.
107 *
108 * Has a small-value optimization when a type is a POD type and as small as a pointer.
109-----------------------------------------------------------------------------*/
110
114template <typename T>
115struct TCallTraitsBase
116{
117private:
118 static constexpr bool PassByValue = ((sizeof(T) <= sizeof(void*)) && std::is_trivial_v<T> && std::is_standard_layout_v<T>) || std::is_arithmetic_v<T> || std::is_pointer_v<T>;
119
120public:
121 using ValueType = T;
122 using Reference = T&;
123 using ConstReference = const T&;
124 using ParamType = typename TCallTraitsParamTypeHelper<T, PassByValue>::ParamType;
125 using ConstPointerType = typename TCallTraitsParamTypeHelper<T, PassByValue>::ConstParamType;
126};

217template <typename T, typename Arg>
218struct TIsBitwiseConstructible
219{
220 static_assert(
221 !std::is_reference_v<T > &&
222 !std::is_reference_v<Arg>,
223 "TIsBitwiseConstructible is not designed to accept reference types");
224
225 static_assert(
226 std::is_same_v<T, typename std::remove_cv_t<T >> &&
227 std::is_same_v<Arg, typename std::remove_cv_t<Arg>>,
228 "TIsBitwiseConstructible is not designed to accept qualified types");
229
230 // Assume no bitwise construction in general
231 static constexpr bool Value = false;
232};

234template <typename T>
235struct TIsBitwiseConstructible<T, T>
236{
237 // Ts can always be bitwise constructed from itself if it is trivially copyable.
238 static constexpr bool Value = std::is_trivially_copy_constructible_v<T>;
239};

241template <typename T, typename U>
242struct TIsBitwiseConstructible<const T, U> : TIsBitwiseConstructible<T, U>
243{
244 // Constructing a const T is the same as constructing a T
245};

247// Const pointers can be bitwise constructed from non-const pointers.
248// This is not true for pointer conversions in general, e.g. where an offset may need to be applied in the case
249// of multiple inheritance, but there is no way of detecting that at compile-time.
250template <typename T>
251struct TIsBitwiseConstructible<const T*, T*>
252{
253 // Constructing a const T is the same as constructing a T
254 static constexpr bool Value = true;
255};

257// Unsigned types can be bitwise converted to their signed equivalents, and vice versa.
258// (assuming two's-complement, which we are)
