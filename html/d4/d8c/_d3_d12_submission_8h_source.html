70//
71// A sync point is a logical point on a GPU queue's timeline that can be awaited by other queues, or the CPU.
72// These are used throughout the RHI as a way to abstract the underlying D3D12 fences. The submission thread
73// manages the underlying fences and signaled values, and reports completion to the relevant sync points via
74// an FGraphEvent.
75//
76// Sync points are one-shot, meaning they represent a single timeline point, and are released after use, via ref-counting.
77// Use FD3D12SyncPoint::Create() to make a new sync point and hold a reference to it via a FD3D12SyncPointRef object.
78//
79class FD3D12SyncPoint final : public FRefCountedObject
80{
81 friend FD3D12DynamicRHI;
82 friend FD3D12Queue;

91 FGraphEventRef GraphEvent;

93 FD3D12SyncPoint(ED3D12SyncPointType Type, FName InDebugName)
94#if RHI_USE_SYNC_POINT_DEBUG_NAME
95 :DebugName(InDebugName)
96#endif

99 {
100 GraphEvent = FGraphEvent::CreateGraphEvent();
101 }
102 }

104public:
105 static FD3D12SyncPointRef Create(ED3D12SyncPointType Type, FName InDebugName)
106 {
107 LLM_SCOPE_BYNAME(TEXT("RHIMisc/CreateSyncPoint"));
108 return new FD3D12SyncPoint(Type, InDebugName);
109 }
110
111 bool IsComplete() const
112 {
113 checkf(GraphEvent, TEXT("This sync point was not created with a CPU event. Cannot check completion on the CPU."));
114 return GraphEvent->IsComplete();
115 }
116
117 void Wait() const;
118
119 FGraphEvent* GetGraphEvent() const
120 {
121 checkf(GraphEvent, TEXT("This sync point was not created with a CPU event."));
122 return GraphEvent;
123 }
124
125 ED3D12SyncPointType GetType() const
126 {
127 return GraphEvent != nullptr
128 ? ED3D12SyncPointType::GPUAndCPU
129 : ED3D12SyncPointType::GPUOnly;
130 }
131
132 void* operator new(size_t Size)
133 {

136 void* Memory = MemoryPool.Pop();

138 {
139 Memory = FMemory::Malloc(sizeof(FD3D12SyncPoint), alignof(FD3D12SyncPoint));
140 }
141 return Memory;
142 }
143
144 void operator delete(void* Pointer)
145 {
146 MemoryPool.Push(Pointer);
147 }
