7syms_arena_def_alloc__sized(SYMS_U64 res, SYMS_U64 cmt){
8 SYMS_ASSERT(SYMS_ARENA_HEADER_SIZE < cmt && cmt <= res);

11 if (memory != 0){
12 if (syms_mem_commit(memory, cmt)){
13 result = (SYMS_Arena*)memory;
14 result->prev = 0;
15 result->current = result;
16 result->base_pos = 0;
17 result->pos = SYMS_ARENA_HEADER_SIZE;
18 result->cmt = cmt;
19 result->cap = res;
20 result->align = 8;
21 result->dev = 0;
22 }
23 else{
24 syms_mem_release(memory, res);
25 }
26 }
27 return(result);
28}

48SYMS_OVERRIDE_FUNC void*
49syms_arena_def_push(SYMS_DefArena *arena, SYMS_U64 size){
50 void *result = 0;
51
52 // get new pos
53 SYMS_Arena *current = arena->current;
54 SYMS_U64 pos_aligned = SYMS_AlignPow2(current->pos, arena->align);
55 SYMS_U64 pos_new = pos_aligned + size;
56
57 // check cap
58 if (current->cap < pos_new){
59
60 // normal growth path
61 SYMS_Arena *new_block = 0;
62 if (size < SYMS_ARENA_VERY_LARGE){
63 new_block = syms_arena_def_alloc();
64 }
65 // "very large" growth path
66 else{
67 SYMS_U64 size_aligned = SYMS_AlignPow2(size + SYMS_ARENA_HEADER_SIZE, (4 << 10));
68 new_block = syms_arena_def_alloc__sized(size_aligned, size_aligned);
69 }
70
71 // connect block to chain
72 if (new_block != 0){
73 new_block->base_pos = current->base_pos + current->cap;
74 SYMS_StackPush_N(arena->current, new_block, prev);
75
76 // recompute the new pos
77 current = new_block;
78 pos_aligned = SYMS_AlignPow2(current->pos, arena->align);
79 pos_new = pos_aligned + size;
80 }
81 }
82
83 // cap big enough?
84 if (current->cap >= pos_new){
85
86 // cmt too small?
87 if (current->cmt < pos_new){
88
89 // get new cmt
90 SYMS_U64 cmt_new_unclamped = SYMS_AlignPow2(pos_new, SYMS_ARENA_COMMIT_SIZE);
91 SYMS_U64 cmt_new = SYMS_ClampTop(cmt_new_unclamped, current->cap);
92 SYMS_U64 cmt_size = cmt_new - current->cmt;
93
94 // try commit
95 if (syms_mem_commit((SYMS_U8*)current + current->cmt, cmt_size)){
96 current->cmt = cmt_new;
97 }
98 }
99
100 // cmt big enough?
101 if (current->cmt >= pos_new){
102
103 // get result & advance pos
104 result = (SYMS_U8*)current + pos_aligned;
105 current->pos = pos_new;
106 }
107 }
108
109 SYMS_ASSERT(arena->current != 0);
110
111 // dev
112#if SYMS_ENABLE_DEV_ARENA
113 syms_arena_dev_push__impl(arena, size, result);
114#endif
115
116 return(result);
117}

126SYMS_OVERRIDE_FUNC void
127syms_arena_def_pop_to(SYMS_DefArena *arena, SYMS_U64 pos_unclamped){
128 SYMS_U64 big_pos = SYMS_ClampBot(SYMS_ARENA_HEADER_SIZE, pos_unclamped);
129
130 // unroll the chain
131 SYMS_Arena *current = arena->current;
132 for (SYMS_Arena *prev = 0;
133 current->base_pos >= big_pos;
134 current = prev){
135 prev = current->prev;
136 syms_mem_release(current, current->cap);
137 }
138 arena->current = current;
139
140 // fix the pos
141 {
142 SYMS_U64 pos_unclamped = big_pos - current->base_pos;
143 SYMS_U64 pos = SYMS_ClampBot(SYMS_ARENA_HEADER_SIZE, pos_unclamped);
144 if (pos < current->pos){
145 current->pos = pos;
146 }
147 }
148
149 SYMS_ASSERT(arena->current != 0);
150
151 // dev
152#if SYMS_ENABLE_DEV_ARENA
153 syms_arena_dev_pop_to__impl(arena, pos_unclamped);
154#endif
155}
