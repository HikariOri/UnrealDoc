331 virtual void SaveAsset_Execute() override
332 {
333 TArray<FDelegateHandle> ValidatorHandles;
334 SaveAssetValidators.GetKeys(ValidatorHandles);
335
336 for (const FDelegateHandle& Handle : ValidatorHandles)
337 {
338 const FAddSaveAssetValidator* Validator = SaveAssetValidators.Find(Handle);
339 if (Validator && Validator->IsBound())
340 {
341 FString Err;
342 if (!Validator->Execute(Err))
343 {
344 FMessageDialog::Open(
345 EAppMsgType::Ok,
346 FText::FromString(Err)
347 );
348 return;
349 }
350 }
