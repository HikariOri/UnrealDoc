18struct FTabPayload_UObject : public FTabPayload
19{
20public:
21 // Create a new payload wrapper for a UObject
22 static TSharedRef<FTabPayload_UObject> Make(const UObject* DocumentID)
23 {
24 return MakeShareable(new FTabPayload_UObject(const_cast<UObject*>(DocumentID)));
25 }
26
27 // Helper method to get the payload object as a specific type
28 // (checks both that the payload is a FTabPayload_UObject, and that the type is correct)
29 template <typename CastType>
30 static CastType* CastChecked(TSharedPtr<FTabPayload> Payload)
31 {
32 check((Payload.IsValid()) && (Payload->PayloadType == NAME_Object));
33
34 UObject* UntypedObject = StaticCastSharedPtr<FTabPayload_UObject>(Payload)->DocumentID.Get(true);
35 return ::CastChecked<CastType>(UntypedObject);
36 }
37
38 // Determine if another payload is the same as this one
39 virtual bool IsEqual(const TSharedRef<FTabPayload>& OtherPayload) const override
40 {
41 if (OtherPayload->PayloadType == PayloadType)
42 {
43 return this->DocumentID.HasSameIndexAndSerialNumber(StaticCastSharedRef<FTabPayload_UObject>(OtherPayload)->DocumentID);
44 }
45
46 return false;
47 }
48
49 virtual bool IsValid() const override
50 {
51 return DocumentID.IsValid();
52 }
53
54 virtual ~FTabPayload_UObject() = default;
55
56private:
57 // Buried constructor. Use Make instead!
58 FTabPayload_UObject(UObject* InDocumentID)
59 : FTabPayload(NAME_Object)
60 , DocumentID(InDocumentID)
61 {
62 }
63
64private:
65 // The object that is the real payload
66 TWeakObjectPtr<UObject> DocumentID;
67};

83 // FWorkflowTabFactory public interface

85 {
86 BaseClass* TypedDocumentID = FTabPayload_UObject::CastChecked<BaseClass>(Info.Payload);
87 return CreateTabBodyForObject(Info, TypedDocumentID);
88 }
89
90 virtual const FSlateBrush* GetTabIcon(const FWorkflowTabSpawnInfo& Info) const override
91 {
92 BaseClass* TypedDocumentID = FTabPayload_UObject::CastChecked<BaseClass>(Info.Payload);
93 return GetTabIconForObject(Info, TypedDocumentID);
94 }
95
96 virtual bool IsPayloadSupported(TSharedRef<FTabPayload> Payload) const override
97 {
98 if (Payload->PayloadType == NAME_Object && Payload->IsValid())
99 {
100 UObject* DocumentID = FTabPayload_UObject::CastChecked<UObject>(Payload);
101 return SupportsObjectType(DocumentID);
102 }
103 return false;
104 }
105
106 virtual bool IsPayloadValid(TSharedRef<FTabPayload> Payload) const override
107 {
108 if (Payload->PayloadType == NAME_Object)
109 {
110 return Payload->IsValid();
111 }
112 return false;
113 }
114 // End of FWorkflowTabFactory public interface
115
116protected:

121 }

123 // Creates the label for the tab

125 {
126 check(Info.Payload.IsValid());

128 return ConstructTabNameForObject(TypedDocumentID);

130 }

132 // Methods to override in subclasses that make life easier for UObject-based documents
133 virtual TAttribute<FText> ConstructTabNameForObject(BaseClass* DocumentID) const=0;
134 virtual TSharedRef<SWidget> CreateTabBodyForObject(const FWorkflowTabSpawnInfo& Info, BaseClass* DocumentID) const=0;
135 virtual const FSlateBrush* GetTabIconForObject(const FWorkflowTabSpawnInfo& Info, BaseClass* DocumentID) const=0;
136};
