55// Generic error serializer for TArray. Routes through CallSerializeForError on each element so
56// element types that customize via SerializeForError, or fall back through SerializeForLog or
57// operator<<, all work uniformly.
58template <typename ValueType, typename AllocatorType UE_REQUIRES(TModels_V<UE::CSerializableForError, ValueType>)>
59inline void SerializeForError(FCbWriter& Writer, const TArray<ValueType, AllocatorType>& Array)
60{
61 Writer.BeginArray();
62 for (const ValueType& Element : Array)
63 {
64 UE::CallSerializeForError(Writer, Element);
65 }
66 Writer.EndArray();
67}

79 constexpr uint64 CompileTimeHashString(const UTF8CHAR* String)
80 {
81 // Based on djb2
82 uint64 Result = 5381;
83 while (char Ch = *String++)
84 {
85 Result = ((Result << 5) + Result) + Ch;
86 }
87 return Result;
88 }

94 template<typename T>
95 concept CErrorPayload = requires(FCbWriter& Writer, const T& ConstRef)
96 {
97 // We need to be able to move or copy the data into the IErrorDetails
98 requires std::is_move_constructible_v<T> || std::is_copy_constructible_v<T>;
99 // We need to be able to identify the type from the IErrorDetails for reinterpet-casting
100 { T::DeclarationHash } -> UE::CConvertibleTo<uint64>;
101 // We need to be able to write the type to compact binary for logging / sending over network.
102 // SerializeForError has a default that routes to SerializeForLog (and from there to operator<<),
103 // so any type that already supports either of those is automatically usable.
104 { SerializeForError(Writer, ConstRef) };
105 };

120 template<typename T>
121 concept CErrorField = requires(FCbWriter& Writer, const T& ConstRef)
122 {
123 // We must be able to write the type into compact binary. The SerializeForError default
124 // covers any type with SerializeForLog (and therefore any type with operator<< into FCbWriter).
125 { SerializeForError(Writer, ConstRef) };
126 };

133 template<typename T>
134 concept CErrorDeclaration = requires (const T& t, T::PayloadType&& p)
135 {
136 // Stored payload type
137 typename T::PayloadType;
138 // Which must conform to CErrorPayload
139 requires CErrorPayload<typename T::PayloadType>;
140 // Whether or not this is an empty payload, i.e. there is no per-instance data for this error type
141 { T::bPayloadHasFields } -> UE::CConvertibleTo<bool>;
142 // Hash of declaration name to look up error payload for runtime casting
143 { T::DeclarationHash } -> UE::CConvertibleTo<uint64>;
144 // Declared error code
145 { T::ErrorCode } -> UE::CConvertibleTo<int32>;
146 // Declared error code cannot be 0
147 requires T::ErrorCode != 0;
148 // Declared module name
149 { T::StaticModuleName } -> UE::CConvertibleTo<FUtf8StringView>;
150 // DeclaredErrorName
151 { T::StaticErrorName } -> UE::CConvertibleTo<FUtf8StringView>;
152 // Declared full name with module and error name
153 { T::StaticFullErrorName } -> UE::CConvertibleTo<FUtf8StringView>;
154 // Declare format string to print error & payload
155 { T::FormatString } -> UE::CConvertibleTo<FText>;
156 // Factory method taking payload type to return error
157 { t(MoveTemp(p)) } -> UE::CConvertibleTo<UE::UnifiedError::FError>;
158 };

443 // For errors declared with UE_DECLARE_ERROR that have no payload fields, there is a single
444 // shared TMandatoryErrorDetails instance used by every FError of that type — it has no
445 // per-instance data to store. This helper returns that instance (constructed on first call).
446 //
447 // For errors with payload fields, this helper is never called at runtime (FError operator()
448 // takes the fields-branch in an if-constexpr) and returns nullptr. The default-construction
449 // of PayloadType is guarded by the same if-constexpr so payload types are not required to
450 // be default-constructible.
451 template<CErrorDeclaration ErrorDeclarationType>
452 const FMandatoryErrorDetails* GetSharedStaticErrorDetails()
453 {
454 if constexpr (!ErrorDeclarationType::bPayloadHasFields)
455 {
456 static TMandatoryErrorDetails<ErrorDeclarationType> Details{typename ErrorDeclarationType::PayloadType{}};
457 return &Details;
458 }
459 else
460 {
461 return nullptr;
462 }
463 }
