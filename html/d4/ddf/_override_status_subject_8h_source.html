240 // returns a common for a given predicate (or an unset TOptional). each object in the subject
241 // is interrogated for using the predicate and can return a TOptional indicating success.
242 // if the value matches for all objects the value will be returned as a set TOptional, otherwise
243 // an unset TOptional will be returned.
244 template<typename ValueType, typename ObjectType = UObject>
245 TOptional<ValueType> GetCommonValue(TFunction<TOptional<ValueType>(const FOverrideStatusObjectHandle<ObjectType>&)> InGetValuePerObjectPredicate) const
246 {
247 TOptional<ValueType> Result;
248 for(int32 Index = 0; Index < Num(); Index++)
249 {
250 if(const FOverrideStatusObjectHandle Handle = GetHandle<ObjectType>(Index))
251 {
252 const TOptional<ValueType> SingleValue = InGetValuePerObjectPredicate(Handle);
253 if(SingleValue.IsSet())
254 {
255 if(Result.IsSet())
256 {
257 if(Result.GetValue() != SingleValue)
258 {
259 Result.Reset();
260 break;
261 }
262 }
263 else
264 {
265 Result = SingleValue;
266 }
267 }
268 }
269 }
270 return Result;
271 }
