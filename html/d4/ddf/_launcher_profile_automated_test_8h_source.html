141 enum class EVersion : uint32
142 {

153 virtual bool Load(const FJsonObject& Object) override
154 {
155 EVersion Version = EVersion::Original;
156 if (!Object.TryGetNumberField(TEXT("Version"), (uint32&)Version))
157 {
158 Version = EVersion::Original;
159 }
160 if (Version > EVersion::Latest)
161 {
162 return false;
163 }
164
165 InternalName = Object.GetStringField(TEXT("InternalName"));
166 UATCommand = Object.GetStringField(TEXT("UATCommand"));
167 Tests = Object.GetStringField(TEXT("Tests"));
168 UATCommandLine = Object.GetStringField(TEXT("AdditionalCommandLine"));
169 bEnabled = Object.GetBoolField(TEXT("Enabled"));
170
171 if (Version >= EVersion::FirstVersioned)
172 {
173 Description = Object.GetStringField(TEXT("Description"));
174 Order = Object.GetIntegerField(TEXT("Order"));
175 }
176 else
177 {
178 Object.TryGetNumberField(TEXT("Order"), Order);
179 }
180 Order = FMath::Max(1, Order);
181
182 UserTypeName = InternalName;
183 if (Version >= EVersion::AddUserTypeName)
184 {
185 UserTypeName = Object.GetStringField(TEXT("UserTypeName"));
186 }
187
188 return true;
189 }
