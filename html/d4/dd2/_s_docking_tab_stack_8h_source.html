382struct FTabMatcher
383{
384 FTabMatcher( const FTabId& InTabId, ETabState::Type InTabState = static_cast<ETabState::Type>(ETabState::ClosedTab | ETabState::OpenedTab | ETabState::SidebarTab), const bool InTreatIndexNoneAsWildcard = true )
385 : TabIdToMatch( InTabId )
386 , RequiredTabState( InTabState )
387 , TreatIndexNoneAsWildcard( InTreatIndexNoneAsWildcard )
388 {
389 }
390
391 bool operator()(const FTabManager::FTab& Candidate) const
392 {
393 return
394 ((Candidate.TabState & RequiredTabState) != 0) &&
395 (Candidate.TabId.TabType == TabIdToMatch.TabType) &&
396 // INDEX_NONE is treated as a wildcard
397 ((TreatIndexNoneAsWildcard && TabIdToMatch.InstanceId == INDEX_NONE) || TabIdToMatch.InstanceId == Candidate.TabId.InstanceId);;
398 }
399
400 FTabId TabIdToMatch;
401 ETabState::Type RequiredTabState;
402 bool TreatIndexNoneAsWildcard;
