130template <typename T>
131FString DumpSnapshotPerfData(const TCurvesSnapshotPerfInfo<T> InData, ECurvesSnapshotPerfFlags InFlags)
132{
133 TStringBuilder<256> Buffer;
134 bool bNeedsSeparator = false;
135 const auto AddSeparator = [&bNeedsSeparator, &Buffer]
136 {
137 if (!bNeedsSeparator)
138 {
139 bNeedsSeparator = true;
140 }
141 else
142 {
143 Buffer << TEXT(", ");
144 }
145 };
146
147 if (EnumHasAnyFlags(InFlags, ECurvesSnapshotPerfFlags::TotalAllocatedSize))
148 {
149 AddSeparator();
150 Buffer << TEXT("Total Size: ") << *FText::AsMemory(InData.GetTotalAllocatedSize()).ToString();
151 }
152
153 if (EnumHasAnyFlags(InFlags, ECurvesSnapshotPerfFlags::Keys))
154 {
155 AddSeparator();
156 Buffer << InData.NumKeys() << TEXT(" keys (") << *FText::AsMemory(InData.GetAllocatedSize_Keys()).ToString() << TEXT(")");
157 }
158
159 if (EnumHasAnyFlags(InFlags, ECurvesSnapshotPerfFlags::Keys))
160 {
161 AddSeparator();
162 Buffer << InData.NumKeys() << TEXT(" positions (") << *FText::AsMemory(InData.GetAllocatedSize_Positions()).ToString() << TEXT(")");
163 }
164
165 if (EnumHasAnyFlags(InFlags, ECurvesSnapshotPerfFlags::Keys))
166 {
167 AddSeparator();
168 Buffer << InData.NumKeys() << TEXT(" attributes (") << *FText::AsMemory(InData.GetAllocatedSize_Attributes()).ToString() << TEXT(")");
169 }
170
171 return FString(Buffer.ToString());
172}
