106 struct FFreeBlock
107 {
108 inline FFreeBlock(uint32 InPageSize, uint16 InBinSize, EBlockCanary InCanary)
109 : BinSize(InBinSize)
110 , CanaryAndForkState(InCanary)
111 , NextFreeBlockIndex(InvalidNextFreeBlock)
112 {
113 NumFreeBins = uint16(InPageSize / InBinSize);
114 }
115
116 FORCEINLINE uint32 GetNumFreeBins() const
117 {
118 return NumFreeBins;
119 }
120
121 inline void* AllocateBin()
122 {
123 --NumFreeBins;
124 // allocate high-to-low: offset 0 (the header location) is returned last
125 return (uint8*)this + NumFreeBins * BinSize;
126 }
127
128 FORCEINLINE FFreeBlock* GetNextFreeBlock() const
129 {
130 if (NextFreeBlockIndex == InvalidNextFreeBlock)
131 {
132 return nullptr;
133 }
134 uint8* BlockBase = (uint8*)AlignDown(this, UE_MB4_LARGE_ALLOC);
135 return (FFreeBlock*)(BlockBase + NextFreeBlockIndex * UE_MBC_MIN_BIN_SIZE);
136 }
137
138 FORCEINLINE void SetNextFreeBlock(FFreeBlock* Next)
139 {
140 if (Next)
141 {
142 uint8* BlockBase = (uint8*)AlignDown(this, UE_MB4_LARGE_ALLOC);
143 NextFreeBlockIndex = uint16((uint8*)Next - BlockBase) / UE_MBC_MIN_BIN_SIZE;
144 }
145 else
146 {
147 NextFreeBlockIndex = InvalidNextFreeBlock;
148 }
149 }
150
151 //constexpr static uint32 MaxBitsForBinSize = 32 - FMath::CountLeadingZeros(UE_MB4_MAX_SMALL_POOL_SIZE));
152 constexpr static uint32 MaxBitsForFreeBins = 32 - FMath::CountLeadingZeros(UE_MB4_LARGE_ALLOC / UE_MBC_MIN_BIN_SIZE);
153 constexpr static uint32 InvalidNextFreeBlock = 1u << MaxBitsForFreeBins;
154 static_assert(MaxBitsForFreeBins + 1 <= 16, "NextFreeBlockIndex should fit in 16 bits");
155
156 uint16 BinSize; // Size of the bins
157 EBlockCanary CanaryAndForkState; // Canary value; two valid values when fork support is active
158 uint16 NumFreeBins; // Number of consecutive free bins here, at least 1
159 uint16 NextFreeBlockIndex; // Bin index within 64KB block of next recycled FFreeBlock, or InvalidNextFreeBlock
160 };

230 FORCEINLINE uint8 LookupBlockPoolEntry(const void* Ptr) const
231 {
232 const uint64 Addr = uint64(AlignDown(Ptr, UE_MB4_LARGE_ALLOC));
233 const uint8* L2 = BlockPoolL1Table[Addr >> 32];

240 // Returns the validated side-table entry for a small-pool allocation,
241 // or UnregisteredBlock if Ptr is a large OS allocation (64KB-aligned and unregistered) or null.
242 // Calls CanaryFail if Ptr looks like a small-pool pointer but has no registered entry.
243 uint8 GetSmallBinEntry(const void* Ptr) const;

245 FORCEINLINE bool IsOSAllocation(const void* Ptr) const
246 {
247#if UE_USE_VERYLARGEPAGEALLOCATOR && !PLATFORM_UNIX && !PLATFORM_ANDROID
248 // With the very large page allocator, small pool blocks come from managed 2MB pages
249 // and must be distinguished from large OS allocations via IsSmallBlockAllocation.
250 // The side table provides the same distinction but we keep both checks for correctness.
251 return !CachedOSPageAllocator.IsSmallBlockAllocation(Ptr) && IsAligned(Ptr, UE_MB4_LARGE_ALLOC) && LookupBlockPoolEntry(Ptr) == UnregisteredBlock;
252#else
253 return IsAligned(Ptr, UE_MB4_LARGE_ALLOC) && LookupBlockPoolEntry(Ptr) == UnregisteredBlock;
254#endif
255 }
