12 FORCEINLINE float DiffusionProfile(float RadiusInMm, float A, float S, float L)
13 {
14 if (L <= 0.0f)
15 {
16 return 0.0f;
17 }
18
19 const float Inv8Pi = 1.0f / (8 * UE_PI);
20
21 float D = 1.0f / S;
22 float R = RadiusInMm / L;
23 float NegRbyD = -R / D;
24 float RrDotR = A * FMath::Max((FMath::Exp(NegRbyD) + FMath::Exp(NegRbyD / 3.0f)) / (D * L) * Inv8Pi, 0.0f);
25 return RrDotR;
26 }

28 FORCEINLINE float TransmissionProfile(float RadiusInMm, float A, float S, float L)
29 {
30 if (L <= 0.0f)
31 {
32 return 0.0f;
33 }

35 //integration from RadiusInMm to infty
36 return 0.25f * A * (FMath::Exp(-S * RadiusInMm / L) + 3.0f * FMath::Exp(-S * RadiusInMm / (3.0f * L)));
37 }
