17template <typename VFrameCallback, typename NativeFrameCallback>
18void FNativeFrame::WalkCallstackFrames(const FOp* InPC, VFrame* InFrame, VTask* InTask, VFrameCallback FrameCallback, NativeFrameCallback NativeCallback) const
19{
20 for (const FNativeFrame* NativeFrame = this; NativeFrame; NativeFrame = NativeFrame->PreviousNativeFrame)
21 {
22 while (InFrame)
23 {
24 if (InPC != &StopInterpreterSentry)
25 {
26 FrameCallback(InPC, InFrame);
27 }
28
29 if (InTask && InFrame == InTask->RootFrame.Get())
30 {
31 // InTask-spawn boundary: pivot to parent InTask's YieldFrame.
32 InPC = InTask->YieldPC;
33 InFrame = InTask->YieldFrame.Get();
34 InTask = InTask->YieldTask.Get();
35 }
36 else
37 {
38 InPC = InFrame->CallerPC;
39 InFrame = InFrame->CallerFrame.Get();
40 }
41 }
42
43 NativeCallback(NativeFrame);
44
45 InPC = NativeFrame->CallerPC;
46 InFrame = NativeFrame->CallerFrame;
47 InTask = NativeFrame->Task;
48 }
49}
