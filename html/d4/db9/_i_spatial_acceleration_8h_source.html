635 {
636 const int32 Idx = GetUniqueIdx(Key).Idx;
637 if(Idx >= Entries.Num())
638 {
639 const int32 NumToAdd = Idx + 1 - Entries.Num();
640 Entries.AddDefaulted(NumToAdd);
641#if CHAOS_SERIALIZE_OUT
642 KeysToSerializeOut.AddDefaulted(NumToAdd);
643#endif
644 }
645
646 ensure(Entries[Idx].bSet == false); //element already added
647 Entries[Idx].bSet = true;
648
649#if CHAOS_SERIALIZE_OUT
650 KeysToSerializeOut[Idx] = Key;
651#endif
