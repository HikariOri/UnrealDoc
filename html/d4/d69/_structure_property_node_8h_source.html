90 void GetOwnerPackages(TArray<UPackage*>& OutPackages) const
91 {
92 if (StructProvider)
93 {
94 // Walk up until we find the objects that contain this struct property to get the packages
95 const FPropertyNode* Parent = this;
96
97 while(Parent)
98 {
99 const FComplexPropertyNode* ComplexParent = Parent->FindComplexParent();
100 if (!ensureMsgf(ComplexParent != nullptr, TEXT("Expected to find a complex parent")))
101 {
102 return;
103 }
104
105 if (const FObjectPropertyNode* ObjectNode = ComplexParent->AsObjectNode())
106 {
107 for (int32 ObjectIndex = 0, ObjectCount = ObjectNode->GetNumObjects(); ObjectIndex < ObjectCount; ++ObjectIndex)
108 {
109 const UPackage* Package = ObjectNode->GetUPackage(ObjectIndex);
110 OutPackages.Add(const_cast<UPackage*>(Package));
111 }
112 break;
113 }
114
115 if (const FStructurePropertyNode* StructNode = ComplexParent->AsStructureNode())
116 {
117 if (StructNode->StructProvider && StructNode->StructProvider->IsPropertyIndirection())
118 {
119 // Skip this and keep walking up to the next complex parent. This is assumed to be a struct that is pointed
120 // to be a parent indirection.
121 // Note: in InstancedStruct case, will re-enter GetOwnerPackages during EnumerateInstances if calling GetInstances here.
122 // We want to avoid that otherwise it can cause poor performance as recursion branches into multiple recursions on each level.
123 Parent = ComplexParent->GetParentNode();
124 Parent = Parent ? Parent->FindComplexParent() : nullptr;
125 if (Parent == nullptr)
126 {
127 // If no owning property, then there is no package.
128 // Returning null to match instance count.
129 OutPackages.Add(nullptr);
130 break;
131 }
132 }
133 else
134 {
135 if (StructNode->StructProvider)
136 {
137 TArray<TSharedPtr<FStructOnScope>> Instances;
138 StructNode->StructProvider->GetInstances(Instances, StructNode->WeakCachedBaseStruct.Get());
139 for (int32 Index = 0, ObjectCount = Instances.Num(); Index < ObjectCount; ++Index)
140 {
141 UPackage* Package = Instances[Index]->GetPackage();
142 OutPackages.Add(Package);
143 }
144 }
145 break;
146 }
147 }
148 }
149 }

208 virtual int32 GetInstancesNum() const override
209 {
210 // Can't get instance count directly from standalone structures, need to walk to the next parent that is an object
211 // to get number of instances from that
212 // Note: This uses iteration over StructProvider->GetInstances and counting returned instances since InstancedStructProvider
213 // calls GetInstancesNum from within it's implementation of GetInstances, leading to exponential recursion up the
214 // property tree.
215 const FComplexPropertyNode* CurrentNode = this;
216
217 while(CurrentNode)
218 {
219 if (const FObjectPropertyNode* ObjectNode = CurrentNode->AsObjectNode())
220 {
221 // Found owning UObject
222 const int32 ObjectCount = ObjectNode->GetInstancesNum();
223 return ObjectCount;
224 }
225 else if (const FStructurePropertyNode* StructNode = CurrentNode->AsStructureNode())
226 {
227 if (TSharedPtr<IStructureDataProvider> TempStructProvider = StructNode->GetStructProvider())
228 {
229 if (TempStructProvider->IsPropertyIndirection())
230 {
231 // If the struct provider is marked as property indirection, it is assumed that it handles indirection between it's
232 // parent property, and some data inside that property (e.g. FInstancedStruct).
233 const FPropertyNode* ParentNode = CurrentNode->GetParentNode();
234 CurrentNode = ParentNode->FindComplexParent();
235 }
236 else
237 {
238 TArray<TSharedPtr<FStructOnScope>> Instances;
239 TempStructProvider->GetInstances(Instances, StructNode->WeakCachedBaseStruct.Get());
240 return Instances.Num();
241 }
242 }
243 else
244 {
245 return 0;
246 }
247 }
248 }
249 return 0;
