248public:
249 FAndroid_MultiTargetPlatformControls(bool bIsClient, ITargetPlatformSettings* TargetPlatformSettings) : FAndroidTargetPlatformControls(bIsClient, TargetPlatformSettings, TEXT("Multi"))
250 {

252
253 // set up all of the multiple formats together into this one
254 void LoadFormats(TArray<FAndroidTargetPlatformControls*> SingleFormatTPs)
255 {
256 // sort formats by priority so higher priority formats are packaged (and thus used by the device) first
257 // note that we passed this by value, not ref, so we can sort it
258 SingleFormatTPs.Sort([](const FAndroidTargetPlatformControls& A, const FAndroidTargetPlatformControls& B)
259 {
260 float PriorityA = 0.f;
261 float PriorityB = 0.f;
262 FString VariantA = A.GetAndroidVariantName().Replace(TEXT("Client"), TEXT(""));
263 FString VariantB = B.GetAndroidVariantName().Replace(TEXT("Client"), TEXT(""));
264 GConfig->GetFloat(TEXT("/Script/AndroidRuntimeSettings.AndroidRuntimeSettings"), *(FString(TEXT("TextureFormatPriority_")) + VariantA), PriorityA, GEngineIni);
265 GConfig->GetFloat(TEXT("/Script/AndroidRuntimeSettings.AndroidRuntimeSettings"), *(FString(TEXT("TextureFormatPriority_")) + VariantB), PriorityB, GEngineIni);
266 return PriorityA > PriorityB;
267 });
268
269 FormatTargetPlatforms.Empty();
270 FormatTargetString = TEXT("");
271
272 TSet<FString> SeenFormats;
273
274 // Load the TargetPlatform module for each format
275 for (FAndroidTargetPlatformControls* SingleFormatTP : SingleFormatTPs)
276 {
277 // only use once each
278 if (SeenFormats.Contains(SingleFormatTP->GetAndroidVariantName()))
279 {
280 continue;
281 }
282 SeenFormats.Add(SingleFormatTP->GetAndroidVariantName());
283
284 bool bEnabled = false;
285 FString SettingsName = FString(TEXT("bMultiTargetFormat_")) + *SingleFormatTP->GetAndroidVariantName();
286 GConfig->GetBool(TEXT("/Script/AndroidRuntimeSettings.AndroidRuntimeSettings"), *SettingsName, bEnabled, GEngineIni);
287 if (bEnabled)
288 {
289 if (FormatTargetPlatforms.Num())
290 {
291 FormatTargetString += TEXT(",");
292 }
293 FormatTargetString += SingleFormatTP->GetAndroidVariantName();
294 FormatTargetPlatforms.Add(SingleFormatTP);
295 }
296 }
297
298 // @todo do we need this with DisplayName below?
299 PlatformInfo::UpdatePlatformDisplayName(TEXT("Android_Multi"), DisplayName());
300 }
301
302 virtual float GetVariantPriority() const override
303 {
304 // lowest priority so specific variants are chosen first
305 return (IsClientOnly() ? 0.25f : 0.5f);
306 }
307
308#if WITH_ENGINE
309 virtual void GetTextureFormats(const UTexture* Texture, TArray< TArray<FName> >& OutFormats) const;
310#endif
311 virtual FText DisplayName() const override;
