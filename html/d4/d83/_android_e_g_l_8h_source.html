280void VerifyEGLResult(EGLint ErrorCode, const TCHAR* Msg1, const TCHAR* Msg2, const TCHAR* Filename, uint32 Line)
281{
282 if (ErrorCode != EGL_SUCCESS)
283 {
284 static const TCHAR* EGLErrorStrings[] =
285 {
286 TEXT("EGL_NOT_INITIALIZED"),
287 TEXT("EGL_BAD_ACCESS"),
288 TEXT("EGL_BAD_ALLOC"),
289 TEXT("EGL_BAD_ATTRIBUTE"),
290 TEXT("EGL_BAD_CONFIG"),
291 TEXT("EGL_BAD_CONTEXT"),
292 TEXT("EGL_BAD_CURRENT_SURFACE"),
293 TEXT("EGL_BAD_DISPLAY"),
294 TEXT("EGL_BAD_MATCH"),
295 TEXT("EGL_BAD_NATIVE_PIXMAP"),
296 TEXT("EGL_BAD_NATIVE_WINDOW"),
297 TEXT("EGL_BAD_PARAMETER"),
298 TEXT("EGL_BAD_SURFACE"),
299 TEXT("EGL_CONTEXT_LOST"),
300 TEXT("UNKNOWN EGL ERROR")
301 };
302
303 uint32 ErrorIndex = FMath::Min<uint32>(ErrorCode - EGL_SUCCESS, UE_ARRAY_COUNT(EGLErrorStrings) - 1);
304 UE_LOGF(LogRHI, Warning, "%ls(%u): %ls%ls failed with error %ls (0x%x)",
305 Filename, Line, Msg1, Msg2, EGLErrorStrings[ErrorIndex], ErrorCode);
306 check(0);
307 }
