49 template <typename T, typename ProjectionType, typename PredicateType>
50 void StableSortInternal(T* First, int32 Num, ProjectionType Projection, PredicateType Predicate)
51 {
52 int32 SubgroupStart = 0;
53
54 if constexpr (MinMergeSubgroupSize > 1)
55 {
56 if constexpr (MinMergeSubgroupSize > 2)
57 {
58 // First pass with simple bubble-sort.
59 do
60 {
61 int32 GroupEnd = SubgroupStart + MinMergeSubgroupSize;
62 if (Num < GroupEnd)
63 {
64 GroupEnd = Num;
65 }
66 do
67 {
68 for (int32 It = SubgroupStart; It < GroupEnd - 1; ++It)
69 {
70 if (Invoke(Predicate, Invoke(Projection, First[It + 1]), Invoke(Projection, First[It])))
71 {
72 Swap(First[It], First[It + 1]);
73 }
74 }
75 GroupEnd--;
76 }
77 while (GroupEnd - SubgroupStart > 1);
78
79 SubgroupStart += MinMergeSubgroupSize;
80 }
81 while (SubgroupStart < Num);
82 }
83 else
84 {
85 for (int32 Subgroup = 0; Subgroup < Num; Subgroup += 2)
86 {
87 if (Subgroup + 1 < Num && Invoke(Predicate, Invoke(Projection, First[Subgroup + 1]), Invoke(Projection, First[Subgroup])))
88 {
89 Swap(First[Subgroup], First[Subgroup + 1]);
90 }
91 }
92 }
93 }
94
95 int32 SubgroupSize = MinMergeSubgroupSize;
96 while (SubgroupSize < Num)
97 {
98 SubgroupStart = 0;
99 do
100 {
101 int32 MergeNum = SubgroupSize << 1;
102 if (Num - SubgroupStart < MergeNum)
103 {
104 MergeNum = Num - SubgroupStart;
105 }
106
107 Merge(First + SubgroupStart, SubgroupSize, MergeNum, Projection, Predicate);
108 SubgroupStart += SubgroupSize << 1;
109 }
110 while (SubgroupStart < Num);
111
112 SubgroupSize <<= 1;
113 }
114 }
