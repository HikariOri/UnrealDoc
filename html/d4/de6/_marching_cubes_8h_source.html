256 Pos.X = Bounds.Min.X + CubeSize * IJK.X;

276 //
277 // corner and edge hash functions, these pack the coordinate
278 // integers into 16-bits, so max of 65536 in any dimension.
279 //
280
281
282 int64 corner_hash(const FVector3i& Idx)
283 {
284 return ((int64)Idx.X&0xFFFF) | (((int64)Idx.Y&0xFFFF) << 16) | (((int64)Idx.Z&0xFFFF) << 32);
285 }
286 int64 corner_hash(int X, int Y, int Z)
287 {
288 return ((int64)X & 0xFFFF) | (((int64)Y & 0xFFFF) << 16) | (((int64)Z & 0xFFFF) << 32);
289 }
290
291 const int64 EDGE_X = int64(1) << 60;
292 const int64 EDGE_Y = int64(1) << 61;
293 const int64 EDGE_Z = int64(1) << 62;
294
295 int64 edge_hash(const FVector3i& Idx1, const FVector3i& Idx2)
296 {
297 if ( Idx1.X != Idx2.X )
298 {
299 int xlo = FMath::Min(Idx1.X, Idx2.X);
300 return corner_hash(xlo, Idx1.Y, Idx1.Z) | EDGE_X;
301 }
302 else if ( Idx1.Y != Idx2.Y )
303 {
304 int ylo = FMath::Min(Idx1.Y, Idx2.Y);
305 return corner_hash(Idx1.X, ylo, Idx1.Z) | EDGE_Y;
306 }
307 else
308 {
309 int zlo = FMath::Min(Idx1.Z, Idx2.Z);
310 return corner_hash(Idx1.X, Idx1.Y, zlo) | EDGE_Z;
311 }
312 }

320 const int64 NumEdgeVertexSections = 64;
321 TArray<TMap<int64, int>> EdgeVertexSections;
322 TArray<FCriticalSection> EdgeVertexSectionLocks;
323
324 int FindVertexID(int64 hash)
325 {
326 int32 SectionIndex = (int32)(hash % (NumEdgeVertexSections - 1));
327 FScopeLock Lock(&EdgeVertexSectionLocks[SectionIndex]);
328 int* Found = EdgeVertexSections[SectionIndex].Find(hash);
329 return (Found != nullptr) ? *Found : IndexConstants::InvalidID;
330 }
331
332 int AppendOrFindVertexID(int64 hash, TVector<double> Pos)
333 {
334 int32 SectionIndex = (int32)(hash % (NumEdgeVertexSections - 1));
335 FScopeLock Lock(&EdgeVertexSectionLocks[SectionIndex]);
336 int* FoundVID = EdgeVertexSections[SectionIndex].Find(hash);
337 if (FoundVID != nullptr)
338 {
339 return *FoundVID;
340 }
341 int NewVID = append_vertex(Pos, hash);
342 EdgeVertexSections[SectionIndex].Add(hash, NewVID);
343 return NewVID;
344 }
345
346
347 int edge_vertex_id(const FVector3i& Idx1, const FVector3i& Idx2, double F1, double F2)
348 {
349 int64 hash = edge_hash(Idx1, Idx2);
350
351 int foundvid = FindVertexID(hash);
352 if (foundvid != IndexConstants::InvalidID)
353 {
354 return foundvid;
355 }
356
357 // ok this is a bit messy. We do not want to lock the entire hash table
358 // while we do find_iso. However it is possible that during this time we
359 // are unlocked we have re-entered with the same edge. So when we
360 // re-acquire the lock we need to check again that we have not already
361 // computed this edge, otherwise we will end up with duplicate vertices!
362
363 TVector<double> pa = TVector<double>::Zero(), pb = TVector<double>::Zero();
364 corner_pos(Idx1, pa);
365 corner_pos(Idx2, pb);
366 TVector<double> Pos = TVector<double>::Zero();
367 find_iso(pa, pb, F1, F2, Pos);
368
369 return AppendOrFindVertexID(hash, Pos);
370 }

379 //
380 // store corner values in pre-allocated grid that has
381 // FMathf::MaxReal as sentinel.
382 // (note this is float grid, not double...)
383 //
384
385 FBlockedDenseGrid3f BlockedCornerValuesGrid;
386
387 double corner_value_grid_parallel(const FVector3i& Idx)
388 {
389 // note: it's possible to have a race here, where multiple threads might both
390 // GetValue, see that the value is invalid, and compute and set it. Since Implicit(V)
391 // is (intended to be) determinstic, they will compute the same value, so this doesn't cause an error,
392 // it just wastes a bit of computation time. Since it is common for multiple corners to be
393 // in the same grid-block, and locking is on the block level, it is (or was in some testing)
394 // better to not lock the entire block while Implicit(V) computed, at the cost of
395 // some wasted evals in some cases.
396
397 float CurrentValue = BlockedCornerValuesGrid.GetValueThreadSafe(Idx.X, Idx.Y, Idx.Z);
398 if (CurrentValue != FMathf::MaxReal)
399 {
400 return (double)CurrentValue;
401 }
402
403 TVector<double> V = corner_pos(Idx);
404 CurrentValue = (float)Implicit(V);
405
406 BlockedCornerValuesGrid.SetValueThreadSafe(Idx.X, Idx.Y, Idx.Z, CurrentValue);
407
408 return (double)CurrentValue;
409 }
410 double corner_value_grid(const FVector3i& Idx)
411 {
412 if (bParallelCompute)
413 {
414 return corner_value_grid_parallel(Idx);
415 }
416
417 float CurrentValue = BlockedCornerValuesGrid.GetValue(Idx.X, Idx.Y, Idx.Z);
418 if (CurrentValue != FMathf::MaxReal)
419 {
420 return (double)CurrentValue;
421 }
422
423 TVector<double> V = corner_pos(Idx);
424 CurrentValue = (float)Implicit(V);
425
426 BlockedCornerValuesGrid.SetValue(Idx.X, Idx.Y, Idx.Z, CurrentValue);
427
428 return (double)CurrentValue;
429 }

563 polygonize_cell(Cell, vertTArray);

566 shift_cell_x(Cell, XIdx);
567 polygonize_cell(Cell, vertTArray);
568 }
569 }

615 void generate_continuation(TArrayView<const FVector3d> Seeds)
616 {
617 FGridCell Cell;
618 int vertTArray[12];
619
620 BlockedDoneCells.Reset(CellDimensions.X, CellDimensions.Y, CellDimensions.Z, 0);
621
622 TArray<FVector3i> stack;
623
624 for (FVector3d seed : Seeds)
625 {
626 FVector3i seed_idx = cell_index(seed);
627 if (!BlockedDoneCells.IsValidIndex(seed_idx) || BlockedDoneCells.GetValue(seed_idx.X, seed_idx.Y, seed_idx.Z) == 1)
628 {
629 continue;
630 }
631 stack.Add(seed_idx);
632 BlockedDoneCells.SetValue(seed_idx.X, seed_idx.Y, seed_idx.Z, 1);
633
634 while ( stack.Num() > 0 )
635 {
636 FVector3i Idx = stack[stack.Num()-1];
637 stack.RemoveAt(stack.Num()-1);
638 if (CancelF())
639 {
640 return;
641 }
642
643 initialize_cell(Cell, Idx);
644 if ( polygonize_cell(Cell, vertTArray) )
645 { // found crossing
646 for ( FVector3i o : IndexUtil::GridOffsets6 )
647 {
648 FVector3i nbr_idx = Idx + o;
649 if (GridBounds.Contains(nbr_idx) && BlockedDoneCells.GetValue(nbr_idx.X, nbr_idx.Y, nbr_idx.Z) == 0)
650 {
651 stack.Add(nbr_idx);
652 BlockedDoneCells.SetValue(nbr_idx.X, nbr_idx.Y, nbr_idx.Z, 1);
653 }
654 }
655 }
656 }
657 }

666 void generate_continuation_parallel(TArrayView<const FVector3d> Seeds)
667 {
668 // Parallel marching cubes based on continuation (ie surface-following / front propagation)
669 // can have quite poor multithreaded performance depending on the ordering of the region-growing.
670 // For example processing each seed point in parallel can result in one thread that
671 // takes significantly longer than others, if the seed point distribution is such
672 // that a large part of the surface is only reachable from one seed (or gets
673 // "cut off" by a thin area, etc). So we want to basically do front-marching in
674 // parallel passes. However this can result in a large number of very short passes
675 // if the front ends up with many small regions, etc. So, in the implementation below,
676 // each "seed cell" is allowed to process up to N neighbour cells before terminating, at
677 // which point any cells remaining on the active front are added as seed cells for the next pass.
678 // This seems to provide good utilization, however more profiling may be needed.
679 // (In particular, if the active cell list is large, some blocks of the ParallelFor
680 // may still end up doing much more work than others)
681
682
683 // set this flag so that append vertex/triangle operations will lock the mesh
684 parallel_mesh_access = true;
685
686 // maximum number of cells to process in each ParallelFor iteration
687 static constexpr int MaxNeighboursPerActiveCell = 100;
688
689 // list of active cells on the MC front to process in the next pass
690 TArray<FVector3i> ActiveCells;
691
692 // initially push list of seed-point cells onto the ActiveCells list
693 for (FVector3d Seed : Seeds)
694 {
695 FVector3i seed_idx = cell_index(Seed);
696 if ( BlockedDoneCells.IsValidIndex(seed_idx) && set_cell_if_not_done(seed_idx) )
697 {
698 ActiveCells.Add(seed_idx);
699 }
700 }
701
702 // new active cells will be accumulated in each parallel-pass
703 TArray<FVector3i> NewActiveCells;
704 FCriticalSection NewActiveCellsLock;
705
706 while (ActiveCells.Num() > 0)
707 {
708 // process all active cells
709 ParallelFor(ActiveCells.Num(), [&](int32 Idx)
710 {
711 FVector3i InitialCellIndex = ActiveCells[Idx];
712 if (CancelF())
713 {
714 return;
715 }
716
717 FGridCell TempCell;
718 int TempArray[12];
719
720 // we will process up to MaxNeighboursPerActiveCell new cells in each ParallelFor iteration
721 TArray<FVector3i, TInlineAllocator<64>> LocalStack;
722 LocalStack.Add(InitialCellIndex);
723 int32 CellsProcessed = 0;
724
725 while (LocalStack.Num() > 0 && CellsProcessed++ < MaxNeighboursPerActiveCell)
726 {
727 FVector3i CellIndex = LocalStack.Pop(EAllowShrinking::No);
728
729 initialize_cell(TempCell, CellIndex);
730 if (polygonize_cell(TempCell, TempArray))
731 {
732 // found crossing
733 for (FVector3i GridOffset : IndexUtil::GridOffsets6)
734 {
735 FVector3i NbrCellIndex = CellIndex + GridOffset;
736 if (GridBounds.Contains(NbrCellIndex))
737 {
738 if (set_cell_if_not_done(NbrCellIndex) == true)
739 {
740 LocalStack.Add(NbrCellIndex);
741 }
742 }
743 }
744 }
745 }
746
747 // if stack is not empty, ie hit MaxNeighboursPerActiveCell, add remaining cells to next-pass Active list
748 if (LocalStack.Num() > 0)
749 {
750 NewActiveCellsLock.Lock();
751 NewActiveCells.Append(LocalStack);
752 NewActiveCellsLock.Unlock();
753 }
754
755 });
756
757 ActiveCells.Reset();
758 if (NewActiveCells.Num() > 0)
759 {
760 Swap(ActiveCells, NewActiveCells);
761 }
762 }
763
764 parallel_mesh_access = false;
765 }

793 bool polygonize_cell(FGridCell& Cell, int VertIndexArray[])
794 {
795 // construct bits of index into edge table, where bit for each
796 // corner is 1 if that value is < isovalue.
797 // This tell us which edges have sign-crossings, and the int value
798 // of the bitmap is an index into the edge and triangle tables
799 int cubeindex = 0, Shift = 1;
800 for (int i = 0; i < 8; ++i)
801 {
802 if (Cell.f[i] < IsoValue)
803 {
804 cubeindex |= Shift;
805 }
806 Shift <<= 1;
807 }
808
809 // no crossings!
810 if (EdgeTable[cubeindex] == 0)
811 {
812 return false;
813 }
814
815 // check each bit of value in edge table. If it is 1, we
816 // have a crossing on that edge. Look up the indices of this
817 // edge and find the intersection point along it
818 Shift = 1;
819 TVector<double> pa = TVector<double>::Zero(), pb = TVector<double>::Zero();
820 for (int i = 0; i <= 11; i++)
821 {
822 if ((EdgeTable[cubeindex] & Shift) != 0)
823 {
824 int a = EdgeIndices[i][0], b = EdgeIndices[i][1];
825 VertIndexArray[i] = edge_vertex_id(Cell.i[a], Cell.i[b], Cell.f[a], Cell.f[b]);
826 }
827 Shift <<= 1;
828 }
829
830 int64 CellHash = corner_hash(Cell.i[0]);
831
832 // now iterate through the set of triangles in TriTable for this cube,
833 // and emit triangles using the vertices we found.
834 int tri_count = 0;
835 for (uint64 tris = TriTable[cubeindex]; tris != 0; tris >>= 12)
836 {
837 int ta = int(tris & 0xf);
838 int tb = int((tris >> 4) & 0xf);
839 int tc = int((tris >> 8) & 0xf);
840 int a = VertIndexArray[ta], b = VertIndexArray[tb], c = VertIndexArray[tc];
841
842 // if a corner is within tolerance of isovalue, then some triangles
843 // will be degenerate, and we can skip them w/o resulting in cracks (right?)
844 // !! this should never happen anymore...artifact of old hashtable impl
845 if (!ensure(a != b && a != c && b != c))
846 {
847 continue;
848 }
849
850 append_triangle(a, b, c, CellHash);
851 tri_count++;
852 }
853
854 return (tri_count > 0);
855 }

973 void find_iso(const TVector<double>& P1, const TVector<double>& P2, double ValP1, double ValP2, TVector<double>& PIso)
974 {
975 // Ok, this is a bit hacky but seems to work? If both isovalues
976 // are the same, we just return the midpoint. If one is nearly zero, we can
977 // but assume that's where the surface is. *However* if we return that point exactly,
978 // we can get nonmanifold vertices, because multiple fans may connect there.
979 // Since FDynamicMesh3 disallows that, it results in holes. So we pull
980 // slightly towards the other point along this edge. This means we will get
981 // repeated nearly-coincident vertices, but the mesh will be manifold.
982 const double dt = 0.999999;
983 if (FMath::Abs(ValP1 - ValP2) < 0.00001)
984 {
985 PIso = (P1 + P2) * 0.5;
986 return;
987 }
988 if (FMath::Abs(IsoValue - ValP1) < 0.00001)
989 {
990 PIso = dt * P1 + (1.0 - dt) * P2;
991 return;
992 }
993 if (FMath::Abs(IsoValue - ValP2) < 0.00001)
994 {
995 PIso = (dt) * P2 + (1.0 - dt) * P1;
996 return;
997 }
998
999 // Note: if we don't maintain min/max order here, then numerical error means
1000 // that hashing on point x/y/z doesn't work
1001 TVector<double> a = P1, b = P2;
1002 double fa = ValP1, fb = ValP2;
1003 if (ValP2 < ValP1)
1004 {
1005 a = P2; b = P1;
1006 fb = ValP1; fa = ValP2;
1007 }
1008
1009 // converge on root
1010 if (RootMode == ERootfindingModes::Bisection)
1011 {
1012 for (int k = 0; k < RootModeSteps; ++k)
1013 {
1014 PIso.X = (a.X + b.X) * 0.5; PIso.Y = (a.Y + b.Y) * 0.5; PIso.Z = (a.Z + b.Z) * 0.5;
1015 double mid_f = Implicit(PIso);
1016 if (mid_f < IsoValue)
1017 {
1018 a = PIso; fa = mid_f;
1019 }
1020 else
1021 {
1022 b = PIso; fb = mid_f;
1023 }
1024 }
1025 PIso = Lerp(a, b, 0.5);
1026
1027 }
1028 else
1029 {
1030 double mu = 0;
1031 if (RootMode == ERootfindingModes::LerpSteps)
1032 {
1033 for (int k = 0; k < RootModeSteps; ++k)
1034 {
1035 mu = FMathd::Clamp((IsoValue - fa) / (fb - fa), 0.0, 1.0);
1036 PIso.X = a.X + mu * (b.X - a.X);
1037 PIso.Y = a.Y + mu * (b.Y - a.Y);
1038 PIso.Z = a.Z + mu * (b.Z - a.Z);
1039 double mid_f = Implicit(PIso);
1040 if (mid_f < IsoValue)
1041 {
1042 a = PIso; fa = mid_f;
1043 }
1044 else
1045 {
1046 b = PIso; fb = mid_f;
1047 }
1048 }
1049 }
1050
1051 // final lerp
1052 mu = FMathd::Clamp((IsoValue - fa) / (fb - fa), 0.0, 1.0);
1053 PIso.X = a.X + mu * (b.X - a.X);
1054 PIso.Y = a.Y + mu * (b.Y - a.Y);
1055 PIso.Z = a.Z + mu * (b.Z - a.Z);
1056 }
1057 }
