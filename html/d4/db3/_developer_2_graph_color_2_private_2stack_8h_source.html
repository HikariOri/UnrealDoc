105inline int sp__Push(stackP theStack, int a)
106{
107 if (theStack->size >= theStack->capacity)
108 return NOTOK;
109
110 theStack->S[theStack->size++] = a;

114inline int sp__Push2(stackP theStack, int a, int b)
115{
116 if (theStack->size + 1 >= theStack->capacity)
117 return NOTOK;
118
119 theStack->S[theStack->size++] = a;
120 theStack->S[theStack->size++] = b;
121 return OK;
122}

124inline int sp__Pop(stackP theStack, int *pA)
125{

127 return NOTOK;
128
129 *pA = theStack->S[--theStack->size];
130 return OK;
131}

133inline int sp__Pop2(stackP theStack, int *pA, int *pB)
134{
135 if (theStack->size <= 1)
136 return NOTOK;
137
138 *pB = theStack->S[--theStack->size];
139 *pA = theStack->S[--theStack->size];
140
141 return OK;
142}
