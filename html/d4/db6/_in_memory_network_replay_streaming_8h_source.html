18/* Holds all data about an entire replay */
19struct FInMemoryReplay
20{
21 struct FCheckpoint
22 {
23 FCheckpoint() :
24 TimeInMS(0),
25 StreamByteOffset(0) {}
26
27 TArray<uint8> Data;
28 uint32 TimeInMS;
29 uint32 StreamByteOffset;
30
31 void Reset()
32 {
33 Data.Reset();
34 TimeInMS = 0;
35 StreamByteOffset = 0;
36 }
37 };
38
40 struct FStreamChunk
41 {
42 FStreamChunk() : StartIndex(0), TimeInMS(0) {}
43
44 int32 StartIndex;
45 uint32 TimeInMS;
46 TArray<uint8> Data;
47 };
48
49 FInMemoryReplay() :
50 NetworkVersion(0)
51 {
52 // Add the first stream chunk, this one will contain the stream before the first checkpoint is written. */
53 StreamChunks.Emplace();
54 }
55
56 TArray<uint8> Header;
57 TArray<FStreamChunk> StreamChunks;
58 TArray<uint8> Metadata;
59 TArray<FCheckpoint> Checkpoints;
60 FNetworkReplayStreamInfo StreamInfo;
61 uint32 NetworkVersion;
62
63 int64 TotalStreamSize() const
64 {
65 int64 TotalSize = sizeof(FInMemoryReplay);
66
67 TotalSize += Header.GetAllocatedSize();
68 TotalSize += StreamChunks.GetAllocatedSize();
69
70 for (const FStreamChunk& Chunk : StreamChunks)
71 {
72 TotalSize += Chunk.Data.GetAllocatedSize();
73 }
74
75 TotalSize += Metadata.GetAllocatedSize();
76 TotalSize += Checkpoints.GetAllocatedSize();
77
78 for (const FCheckpoint& Checkpoint : Checkpoints)
79 {
80 TotalSize += Checkpoint.Data.GetAllocatedSize();
81 }
82
83 return TotalSize;
84 }
85};
