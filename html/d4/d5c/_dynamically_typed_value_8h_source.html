78 // A value stored in some uninterpreted memory and a pointer to a type that contains methods to interpret it.
79 struct FDynamicallyTypedValue
80 {
81 static COREUOBJECT_API FDynamicallyTypedValueType& NullType();
82
83 FDynamicallyTypedValue()
84 {
85 // We need `this` to be in a special state such that if we've created this within a transaction
86 // the GC will be able to destroy the value. To do that it needs a very specific null state,
87 // which is what `InitializeToNull` gives us. So run this in the open, but disable any memory
88 // validation on the value as its fine if we've written to this in the open, then write to it in
89 // the closed. Those closed writes will be undone, reverting the value back to the specific null
90 // state, which the GC can handle.
91 UE_AUTORTFM_OPEN_NO_VALIDATION { InitializeToNull(); };
92 }
93
94 // Use a delegated constructor so that the value is nulled correctly before initialized from the copy.
95 FDynamicallyTypedValue(const FDynamicallyTypedValue& Copyee) : FDynamicallyTypedValue() { InitializeFromCopy(Copyee); }
96
97 // Use a delegated constructor so that the value is nulled correctly before initialized from the move.
98 FDynamicallyTypedValue(FDynamicallyTypedValue&& Movee) : FDynamicallyTypedValue() { InitializeFromMove(MoveTemp(Movee)); }
99
100 ~FDynamicallyTypedValue() { Deinit(); }
101
102 FDynamicallyTypedValue& operator=(const FDynamicallyTypedValue& Copyee)
103 {
104 if (this != &Copyee)
105 {
106 Deinit();
107 InitializeFromCopy(Copyee);
108 }
109 return *this;
110 }
111 FDynamicallyTypedValue& operator=(FDynamicallyTypedValue&& Movee)
112 {
113 if (this != &Movee)
114 {
115 Deinit();
116 InitializeFromMove(MoveTemp(Movee));
117 }
118 return *this;
119 }
120
121 // Returns a pointer to the value's data.
122 const void* GetDataPointer() const { return IsInline() ? &InlineData : HeapData; }
123 void* GetDataPointer() { return IsInline() ? &InlineData : HeapData; }
124
125 // Returns the value's type.
126 FDynamicallyTypedValueType& GetType() const
127 {
128 return Type ? *Type : NullType();
129 }
130
131 // Sets the value to the null state.
132 void SetToNull()
133 {
134 Deinit();
135 InitializeToNull();
136 }
137
138 // Sets the value to the initial value of a type.
139 void InitializeAsType(FDynamicallyTypedValueType& NewType)
140 {
141 check(&NewType != nullptr);
142 Deinit();
143 Type = &NewType;
144 AllocateData();
145 Type->InitializeValue(GetDataPointer());
146 MarkTypeReachableIfIncrementalReachabilityPending();
147 }
148
149#if WITH_VERSE_VM
150 ::Verse::VValue ToVValue(::Verse::FAllocationContext Context) const
151 {
152 return Type->ToVValue(Context, GetDataPointer());
153 }
154#endif
155
156 // Returns hash of the underlying FDynamicallyTypedValue's value. Added to allow for FDynamicallyTypedValue to be used as TMap keys.
157 [[nodiscard]] friend uint32 GetTypeHash(const FDynamicallyTypedValue& DynamicallyTypedValue)
158 {
159 return DynamicallyTypedValue.GetType().GetValueHash(DynamicallyTypedValue.GetDataPointer());
160 }
161
162 // Traverses the data based on the type of value that is stored and invokes ObjectFunc for each UObject that is found.
163 void ForEachSubobject(TFunctionRef<void(const UObject*)> ObjectFunc) const
164 {
165 Type->ForEachSubobject(GetDataPointer(), ObjectFunc);
166 }

168 private:
169 void MarkTypeReachableIfIncrementalReachabilityPending()
170 {
171 struct FTypeReferenceCollector final : public FReferenceCollector
172 {
173 bool IsIgnoringArchetypeRef() const override { return false; }
174 bool IsIgnoringTransient() const override { return false; }
175 void HandleObjectReference(UObject*& InObject, const UObject* InReferencingObject, const FProperty* InReferencingProperty) override
176 {
177 if (InObject)
178 {
179 UE::GC::MarkAsReachable(InObject);
180 }
181 }
182 };
183
184 if (UE::GC::GIsIncrementalReachabilityPending && Type)
185 {
186 // nb: this is done to simulate a write barrier for this type, which
187 // enables it to behave properly with incremental gc.
188 FTypeReferenceCollector Collector;
189 Type->MarkReachable(Collector);
190 }
191 }
192
193 FDynamicallyTypedValueType* Type;
194
195 // Store pointer-sized or smaller values inline, heap allocate all others.
196 union
197 {
198 UPTRINT InlineData;
199 void* HeapData;
200 };
201
202 // Initialize this value from the primordial state to the null state.
203 void InitializeToNull()
204 {
205 Type = &NullType();
206 HeapData = nullptr;
207 MarkTypeReachableIfIncrementalReachabilityPending();
208 }
209 // Deinitializes this value back to the primordial state.
210 void Deinit()
211 {
212 if (!Type)
213 {
214 return;
215 }
216 Type->DestroyValue(GetDataPointer());
217 FreeData();
218 Type = nullptr;
219 }
220 // Copies the data from another value to this one, which is assumed to be in the primordial state.
221 void InitializeFromCopy(const FDynamicallyTypedValue& Copyee)
222 {
223 Type = Copyee.Type;
224 AllocateData();
225 Type->InitializeValueFromCopy(GetDataPointer(), Copyee.GetDataPointer());
226 MarkTypeReachableIfIncrementalReachabilityPending();
227 }
228 // Moves the data from another value to this one, which is assumed to be in the primordial state.
229 // The source value is set to the null state.
230 void InitializeFromMove(FDynamicallyTypedValue&& Movee)
231 {
232 // Simply copy the type and data from the source value.
233 // This assumes that the data is trivially relocatable.
234 Type = Movee.Type;
235 InlineData = Movee.InlineData;
236 MarkTypeReachableIfIncrementalReachabilityPending();
237
238 // Reset the source value to null.
239 Movee.InitializeToNull();
240 }
241
242 // Whether the value's data is stored in InlineData or in the memory pointed to by HeapData.
243 bool IsInline() const
244 {
245 return Type->GetNumBytes() <= sizeof(UPTRINT)
246 && Type->GetMinAlignmentLogTwo() <= UE_FORCE_CONSTEVAL(FMath::ConstExprCeilLogTwo(alignof(UPTRINT)));
247 }
248
249 // Allocates heap memory for the value if it uses it.
250 void AllocateData()
251 {
252 if (IsInline())
253 {
254 // Ensure that the data is zeroed in the inline case to avoid spurious static analysis
255 // errors about passing a reference to uninitialized data to InitializeValueFromCopy.
256 InlineData = 0;
257 }
258 else
259 {
260 HeapData = FMemory::Malloc(Type->GetNumBytes(), Type->GetMinAlignment());
261 }
262 }
263
264 // Frees heap memory for the value if it uses it.
265 void FreeData()
266 {
267 if (!IsInline())
268 {
269 FMemory::Free(HeapData);
270 HeapData = nullptr;
271 }
272 else if (::AutoRTFM::IsClosed())
273 {
274 // Assign to InlineData if we're in a closed transaction.
275 // This is done to ensure that the value of InlineData is
276 // recorded in the transaction. This is important as some
277 // code paths destruct then re-construct the
278 // FDynamicallyTypedValue, and the constructor calls
279 // InitializeToNull() in the open (without recording the initial
280 // values). This can result in the FDynamicallyTypedValue being
281 // nulled without a write record to restore the original value.
282 InlineData = 0;
283 }
284 }
285 };
