257 if constexpr (bUsingIntrusiveUnsetState)

306 [[nodiscard]] friend constexpr bool operator==(const TOptional& Lhs, const TOptional& Rhs)
307 {

328 void Serialize(FArchive& Ar)
329 {
330 bool bOptionalIsSet = IsSet();
331 if (Ar.IsLoading())
332 {
333 bool bOptionalWasSaved = false;
334 Ar << bOptionalWasSaved;
335 if (bOptionalWasSaved)
336 {
337 if (!bOptionalIsSet)
338 {
339 Emplace();
340 }
341 Ar << GetValue();
342 }
343 else
344 {
345 Reset();
346 }
347 }
348 else
349 {
350 Ar << bOptionalIsSet;
351 if (bOptionalIsSet)
352 {
353 Ar << GetValue();
354 }
355 }
