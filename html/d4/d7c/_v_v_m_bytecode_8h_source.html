84inline bool MightFallThrough(EOpcode Opcode)
85{
86 if (IsTerminal(Opcode))
87 {
88 return false;
89 }
90
91 if (!IsBranch(Opcode))
92 {
93 return true;
94 }
95
96 switch (Opcode)
97 {
98 case EOpcode::Jump:
99 case EOpcode::Switch:
100 case EOpcode::EndTask:
101 case EOpcode::Yield:
102 return false;
103 case EOpcode::JumpIfInitialized:
104 case EOpcode::LtFastFail:
105 case EOpcode::LteFastFail:
106 case EOpcode::GtFastFail:
107 case EOpcode::GteFastFail:
108 case EOpcode::EqFastFail:
109 case EOpcode::NeqFastFail:
110 case EOpcode::ArrayIndexFastFail:
111 case EOpcode::TypeCastFastFail:
112 case EOpcode::QueryFastFail:
113 case EOpcode::CanFastAppendToArrayFastFail:
114 case EOpcode::CreateField:
115 case EOpcode::CreateFieldICValueObjectConstant:
116 case EOpcode::CreateFieldICValueObjectField:
117 case EOpcode::CreateFieldICNativeStruct:
118 case EOpcode::CreateFieldICUObject:
119 case EOpcode::JumpIfDefaultSubObject:
120 return true;
121 default:
122 break;
123 }
124
125 VERSE_UNREACHABLE();
126 return false;

205struct FValueOperand
206{
207 static constexpr uint32 UNINITIALIZED = INT32_MAX;
208
209 union
210 {
211 uint32 Index;
212 FRegisterIndex Register;
213 FConstantIndex Constant;
214 };
215
216 static_assert(sizeof(Index) == sizeof(Register));
217
218 FValueOperand()
219 : Index(UNINITIALIZED)
220 {
221 }
222
223 FValueOperand(FRegisterIndex Register)
224 : Index(Register.Index)
225 {
226 check(Register.Index < UNINITIALIZED);
227 check(IsRegister());
228 }
229 FValueOperand(FConstantIndex Constant)
230 : Index{~Constant.Index}
231 {
232 check(Constant.Index <= UNINITIALIZED);
233 check(IsConstant());
234 }
235
236 V_FORCEINLINE bool IsRegister() const { return Index < UNINITIALIZED; }
237 V_FORCEINLINE bool IsConstant() const { return UNINITIALIZED < Index; }
238 V_FORCEINLINE explicit operator bool() const { return Index != UNINITIALIZED; }
239
240 V_FORCEINLINE FRegisterIndex& AsRegister()
241 {
242 checkSlow(IsRegister());
243 return Register;
244 }
245
246 V_FORCEINLINE FConstantIndex AsConstant() const
247 {
248 checkSlow(IsConstant());
249 return FConstantIndex{~Index};
250 }
251
252 template <typename TVisitor>
253 friend void Visit(TVisitor& Visitor, FValueOperand& Value)
254 {
255 Visitor.Visit(Value.Index, TEXT("Index"));
256 }
257};
