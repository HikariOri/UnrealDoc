11constexpr int32 SubpixelBits = 8;
12constexpr int32 SubpixelSamples = 1 << SubpixelBits;

14FORCEINLINE int32 EdgeC( const FIntVector2& Edge, const FIntVector2& Vert, int32 SubpixelDilate = 0 )
15{
16 int64 ex = Edge.X;
17 int64 ey = Edge.Y;
18 int64 vx = Vert.X;
19 int64 vy = Vert.Y;
20
21 // Half-edge constants
22 // 24.16 fixed point
23 int64 C = ey * vx - ex * vy;
24
25 // Correct for fill convention
26 // Top left rule for CCW
27 C -= ( Edge.Y < 0 || ( Edge.Y == 0 && Edge.X > 0 ) ) ? 0 : 1;
28
29 // Dilate edges
30 C += ( FMath::Abs( Edge.X ) + FMath::Abs( Edge.Y ) ) * SubpixelDilate;
31
32 // Step in pixel increments
33 // Low bits would always be the same and thus don't matter when testing sign.
34 // 24.8 fixed point
35 return int32( C >> SubpixelBits );
36};

38// Fixed point rasterization
39struct FTriangle
40{
41 FIntVector2 Vert0;
42 FIntVector2 Vert1;
43 FIntVector2 Vert2;
44
45 FIntVector2 MinPixel;
46 FIntVector2 MaxPixel;
47
48 FIntVector2 Edge01;
49 FIntVector2 Edge12;
50 FIntVector2 Edge20;
51
52 int32 C0;
53 int32 C1;
54 int32 C2;
55
56 bool bBackFace;
57
58 FTriangle( const FVector3f Verts[3], FIntVector2 ScissorMin, FIntVector2 ScissorMax, int32 SubpixelDilate = 0 )
59 {
60 // 24.8 fixed point
61 Vert0 = RoundToInt( FVector2f( Verts[0] ) * SubpixelSamples );
62 Vert1 = RoundToInt( FVector2f( Verts[1] ) * SubpixelSamples );
63 Vert2 = RoundToInt( FVector2f( Verts[2] ) * SubpixelSamples );
64
65 // Bounding rect
66 FIntVector2 MinSubpixel = Min3( Vert0, Vert1, Vert2 );
67 FIntVector2 MaxSubpixel = Max3( Vert0, Vert1, Vert2 );
68
69 MinSubpixel -= SubpixelDilate;
70 MaxSubpixel += SubpixelDilate;
71
72 // Round to nearest pixel
73 MinPixel = ( ( MinSubpixel + (SubpixelSamples / 2) - 1 ) ) / SubpixelSamples;
74 MaxPixel = ( ( MaxSubpixel + (SubpixelSamples / 2) - 1 ) ) / SubpixelSamples;
75
76 // Scissor
77 MinPixel = Max( MinPixel, ScissorMin );
78 MaxPixel = Min( MaxPixel, ScissorMax );
79
80 // Rebase off MinPixel with half pixel offset
81 // 12.8 fixed point
82 // Max triangle size = 2047x2047 pixels
83 const FIntVector2 BaseSubpixel = MinPixel * SubpixelSamples + (SubpixelSamples / 2);
84 Vert0 -= BaseSubpixel;
85 Vert1 -= BaseSubpixel;
86 Vert2 -= BaseSubpixel;
87
88 // 12.8 fixed point
89 Edge01 = Vert0 - Vert1;
90 Edge12 = Vert1 - Vert2;
91 Edge20 = Vert2 - Vert0;
92
93 // 24.16 fixed point
94 int64 DetXY = Edge01.Y * Edge20.X - Edge01.X * Edge20.Y;
95 bBackFace = DetXY >= 0;
96 if( bBackFace )
97 {
98 // Swap winding order
99 Edge01 *= -1;
100 Edge12 *= -1;
101 Edge20 *= -1;
102 }
103
104 C0 = EdgeC( Edge12, Vert1, SubpixelDilate );
105 C1 = EdgeC( Edge20, Vert2, SubpixelDilate );
106 C2 = EdgeC( Edge01, Vert0, SubpixelDilate );
107 }
108
109 inline bool IsCovered( int32 x, int32 y ) const
110 {
111 x -= MinPixel.X;
112 y -= MinPixel.Y;
113
114 int32 CX0 = C0 - x * Edge12.Y + y * Edge12.X;
115 int32 CX1 = C1 - x * Edge20.Y + y * Edge20.X;
116 int32 CX2 = C2 - x * Edge01.Y + y * Edge01.X;
117
118 return ( CX0 | CX1 | CX2 ) >= 0;
119 }
120
121 template< typename FFunc >
122 void ForAllCovered( FFunc&& Func ) const
123 {
124 int32 CY0 = C0;
125 int32 CY1 = C1;
126 int32 CY2 = C2;
127
128 for( int32 y = MinPixel.Y; y < MaxPixel.Y; y++ )
129 {
130 int32 CX0 = CY0;
131 int32 CX1 = CY1;
132 int32 CX2 = CY2;
133
134 for( int32 x = MinPixel.X; x < MaxPixel.X; x++ )
135 {
136 if( ( CX0 | CX1 | CX2 ) >= 0 )
137 Func( x, y );
138
139 CX0 -= Edge12.Y;
140 CX1 -= Edge20.Y;
141 CX2 -= Edge01.Y;
142 }
143
144 CY0 += Edge12.X;
145 CY1 += Edge20.X;
146 CY2 += Edge01.X;
147 }
148 }
149
150 inline FVector3f GetBarycentrics( int32 x, int32 y ) const
151 {
152 FIntVector2 p = ( FIntVector2(x,y) - MinPixel ) * SubpixelSamples;
153 FVector2f p0( Vert0 - p );
154 FVector2f p1( Vert1 - p );
155 FVector2f p2( Vert2 - p );
156
157 // Not perspective correct
158 FVector3f Barycentrics(
159 (float)Edge12.Y * p1.X - (float)Edge12.X * p1.Y,
160 (float)Edge20.Y * p2.X - (float)Edge20.X * p2.Y,
161 (float)Edge01.Y * p0.X - (float)Edge01.X * p0.Y );
162 Barycentrics /= Barycentrics[0] + Barycentrics[1] + Barycentrics[2];
163
164 return Barycentrics;
165 }
166};

168inline bool IsCovered( const FVector2f& Edge, const FVector2f& Vert, const FVector2f& Center, const FVector2f& Extent )
169{
170 FVector2f Point = Vert - Center;
171 float Barycentric = Edge.Y * Point.X - Edge.X * Point.Y;
172
173 Barycentric += FMath::Abs( Edge.Y ) * Extent.X;
174 Barycentric += FMath::Abs( Edge.X ) * Extent.Y;
175
176 // Correct for fill convention
177 // Top left rule for CCW
178 bool bTopLeft = Edge.Y < 0.0f || ( Edge.Y == 0.0f && Edge.X > 0.0f );
179 bool bIsInside = Barycentric > 0.0f || ( Barycentric == 0.0f && bTopLeft );
180 return bIsInside;
181}

184struct FTriangle3f
185{
186 FVector3f Vert0;
187 FVector3f Vert1;
188 FVector3f Vert2;

190 FVector3f Min;

194 FVector3f Edge12;
195 FVector3f Edge20;

197 FVector4f Plane;

199 inline FTriangle3f() {}
200 inline FTriangle3f( const FVector3f Verts[3] )
201 {
202 Vert0 = Verts[0];
203 Vert1 = Verts[1];
204 Vert2 = Verts[2];
205
206 Min = Min3( Vert0, Vert1, Vert2 );
207 Max = Max3( Vert0, Vert1, Vert2 );
208
209 Edge01 = Vert0 - Vert1;
210 Edge12 = Vert1 - Vert2;
211 Edge20 = Vert2 - Vert0;
212
213 FVector3f Normal = Edge01 ^ Edge20;
214 Plane = FVector4f( Normal, -( Normal | Vert0 ) );
215 }
216
217 inline FVector3f GetBarycentrics( float x, float y ) const
218 {
219 FVector2f p( x, y );
220 FVector2f p0 = FVector2f( Vert0 ) - p;
221 FVector2f p1 = FVector2f( Vert1 ) - p;
222 FVector2f p2 = FVector2f( Vert2 ) - p;
223
224 // Not perspective correct
225 FVector3f Barycentrics(
226 Edge12.Y * p1.X - Edge12.X * p1.Y,
227 Edge20.Y * p2.X - Edge20.X * p2.Y,
228 Edge01.Y * p0.X - Edge01.X * p0.Y );
229 Barycentrics /= Barycentrics[0] + Barycentrics[1] + Barycentrics[2];
230
231 return Barycentrics;
232 }
233
234 inline FVector3f GetBarycentrics( const FVector3f& Point ) const
235 {
236 FVector3f v0 = -Edge01;
237 FVector3f v1 = Edge20;
238 FVector3f v2 = Point - Vert0;
239
240 float d00 = v0 | v0;
241 float d01 = v0 | v1;
242 float d11 = v1 | v1;
243 float d20 = v2 | v0;
244 float d21 = v2 | v1;
245 float rcp = 1.0f / ( d00 * d11 - d01 * d01 );
246
247 FVector3f Barycentrics;
248 Barycentrics.Y = ( d11 * d20 - d01 * d21 ) * rcp;
249 Barycentrics.Z = ( d00 * d21 - d01 * d20 ) * rcp;
250 Barycentrics.X = 1.0f - Barycentrics.Y - Barycentrics.Z;
251 return Barycentrics;
252 }

254 inline bool IsCovered( const FVector2f& Center, const FVector2f& Extent ) const
255 {
256 float Sign = Plane.Z >= 0.0f ? 1.0f : -1.0f;
257 return
258 Rasterizer::IsCovered( Sign * FVector2f( Edge12 ), FVector2f( Vert1 ), Center, Extent ) &&
259 Rasterizer::IsCovered( Sign * FVector2f( Edge20 ), FVector2f( Vert2 ), Center, Extent ) &&
260 Rasterizer::IsCovered( Sign * FVector2f( Edge01 ), FVector2f( Vert0 ), Center, Extent );
261 }

263 inline bool IsCovered( int32 x, int32 y, float PixelExtent = 0.0f ) const

265 return IsCovered( FVector2f( (float)x + 0.5f, (float)y + 0.5f ), FVector2f( PixelExtent ) );
266 }

268 inline bool IsCovered( FVector3f Origin, FVector3f Direction, FVector2f Time )
269 {
270 // Muller-Trumbore ray triangle intersect
271 FVector3f Origin0 = Origin - Vert0;
272 FVector3f Dirx20 = Direction ^ Edge20;
273
274 float Det = -( Edge01 | Dirx20 );
275 if( FMath::Abs( Det ) < 1e-8f )
276 return false;
277 float InvDet = 1.0f / Det;
278
279 float V = InvDet * ( Origin0 | Dirx20 );
280 float W = InvDet * ( Direction | ( Edge01 ^ Origin0 ) );
281 float t = InvDet * ( Edge20 | ( Edge01 ^ Origin0 ) );
282
283 if( V < 0.0f || V > 1.0f )
284 return false;
285 if( W < 0.0f || V + W > 1.0f )
286 return false;
287 if( t < Time[0] || t > Time[1] )
288 return false;
289 return true;
290 }

292 inline FVector3f GetDepthPlane() const
293 {
294 /*
295 Solve for v.z
296 n | (v - p) = 0;
297 (n|v) - (n|p) = 0;
298 (n.xy|v.xy) + n.z*z - (n|p) = 0;
299 -(n.xy|v.xy)/n.z + (n|p)/n.z = v.z;
300 */
301 return FVector3f( -Plane.X, -Plane.Y, -Plane.W ) / Plane.Z;
302 }
303
304 inline FTriangle3f Swizzle( int32 X, int32 Y, int32 Z ) const
305 {
306 FTriangle3f TriXYZ;
307 TriXYZ.Vert0 = FVector3f( Vert0[X], Vert0[Y], Vert0[Z] );
308 TriXYZ.Vert1 = FVector3f( Vert1[X], Vert1[Y], Vert1[Z] );
309 TriXYZ.Vert2 = FVector3f( Vert2[X], Vert2[Y], Vert2[Z] );
310
311 TriXYZ.Min = FVector3f( Min[X], Min[Y], Min[Z] );
312 TriXYZ.Max = FVector3f( Max[X], Max[Y], Max[Z] );
313
314 TriXYZ.Edge01 = FVector3f( Edge01[X], Edge01[Y], Edge01[Z] );
315 TriXYZ.Edge12 = FVector3f( Edge12[X], Edge12[Y], Edge12[Z] );
316 TriXYZ.Edge20 = FVector3f( Edge20[X], Edge20[Y], Edge20[Z] );
317
318 TriXYZ.Plane = FVector4f( Plane[X], Plane[Y], Plane[Z], Plane.W );
319
320 return TriXYZ;
321 }
322};

326template< typename FWritePixel >
327void RasterizeTri( const FVector3f Verts[3], const FIntRect& ScissorRect, uint32 SubpixelDilate, bool bBackFaceCull, FWritePixel&& WritePixel )
328{
329 Rasterizer::FTriangle Tri( Verts, ScissorRect.Min, ScissorRect.Max, SubpixelDilate );
330
331 // Cull when no pixels covered
332 if( Tri.MinPixel.X >= Tri.MaxPixel.X ||
333 Tri.MinPixel.Y >= Tri.MaxPixel.Y )
334 return;
335
336 if( Tri.bBackFace && bBackFaceCull )
337 return;
338
339 Tri.ForAllCovered(
340 [&]( int32 x, int32 y )
341 {
342 FVector3f Barycentrics = Tri.GetBarycentrics( x, y );
343
344 float Depth =
345 Verts[0].Z * Barycentrics[0] +
346 Verts[1].Z * Barycentrics[1] +
347 Verts[2].Z * Barycentrics[2];
348
349 WritePixel( x, y, Depth, Barycentrics );
350 } );
351}

454{

457 FMath::Abs( Tri.Plane.X ),
458 FMath::Abs( Tri.Plane.Y ),
