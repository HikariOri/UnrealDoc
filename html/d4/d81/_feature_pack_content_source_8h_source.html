106 FString GetFeaturePackNameForLevel(EFeaturePackDetailLevel InLevel, bool bLevelRequired = false)
107 {
108 check(DetailLevels.Num()>0); // We need at least one detail level defined
109 int32 Index = DetailLevels.Find(InLevel);
110 FString DetailString;
111 if( Index != INDEX_NONE)
112 {
113 UEnum::GetValueAsString(TEXT("/Script/AddContentDialog.EFeaturePackDetailLevel"), InLevel, DetailString);
114 }
115 else
116 {
117 check(bLevelRequired==false); // The level is REQUIRED and we don't have it !
118 // If we didn't have the requested level, use the first
119 UEnum::GetValueAsString(TEXT("/Script/AddContentDialog.EFeaturePackDetailLevel"), DetailLevels[0], DetailString);
120 }
121 FString NameString = MountName+DetailString + TEXT(".upack");
122 return NameString;
123 }
