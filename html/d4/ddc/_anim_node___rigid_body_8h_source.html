67 // Global multipler on the effects of simulation space movement. Must be in range [0, 1]. If WorldAlpha = 0.0, the system is disabled and the simulation will
68 // be fully local (i.e., world-space actor movement and rotation does not affect the simulation). When WorldAlpha = 1.0 the simulation effectively acts as a
69 // world-space sim, but with the ability to apply limits using the other parameters.
70 UPROPERTY(EditAnywhere, BlueprintReadWrite, Category = Settings, meta = (ClampMin = "0.0", ClampMax = "1.0"))
71 float WorldAlpha;
72
73 // Multiplier on the Z-component of velocity and acceleration that is passed to the simulation. Usually from 0.0 to 1.0 to
74 // reduce the effects of jumping and crouching on the simulation, but it can be higher than 1.0 if you need to exaggerate this motion for some reason.
75 // Should probably have been called "WorldAlphaScaleZ".
76 UPROPERTY(EditAnywhere, BlueprintReadWrite, Category = Settings, meta = (ClampMin = "0.0"))
77 float VelocityScaleZ;
78
79 // A muliplier to control how much of the simulation space movement is used to calculate the drag forces from Linear/Angular Damping in the Physics Asset.
80 // When DampingAlpha=1.0, Damping drag forces are equivalent to a world-space simulation. This is similar to air resistance.
81 // When DampingAlpha=0.0, Damping drag forces depend only on local-space body velocity and not on the simulation space velocity.
82 // It can be useful to set this to zero so that the Linear/Angular Damping settings on the BodyInstances do not contribute to air resistance.
83 // Air resistance can be re-added in a controlled way using the ExternalLinearDrag setting below.
84 UPROPERTY(EditAnywhere, BlueprintReadWrite, Category = Settings, meta = (ClampMin = "0.0"))
85 float DampingAlpha;
86
87 // A clamp on the effective world-space velocity that is passed to the simulation. Units are cm/s. The default value effectively means "unlimited". It is not usually required to
88 // change this but you would reduce this to limit the effects of drag on the bodies in the simulation (if you have bodies that have LinearDrag set to non-zero in the physics asset).
89 // Expected values in this case would be somewhat less than the usual velocities of your object which is commonly a few hundred for a character.
90 UPROPERTY(EditAnywhere, BlueprintReadWrite, Category = Settings, meta = (ClampMin = "0.0"))
91 float MaxLinearVelocity;
92
93 // A clamp on the effective world-space angular velocity that is passed to the simulation. Units are radian/s, so a value of about 6.0 is one rotation per second.
94 // The default value effectively means "unlimited". You would reduce this (and MaxAngularAcceleration) to limit how much bodies "fly out" when the actor spins on the spot.
95 // This is especially useful if you have characters than can rotate very quickly and you would probably want values around or less than 10 in this case.
96 UPROPERTY(EditAnywhere, BlueprintReadWrite, Category = Settings, meta = (ClampMin = "0.0"))
97 float MaxAngularVelocity;
98
99 // A clamp on the effective world-space acceleration that is passed to the simulation. Units are cm/s/s. The default value effectively means "unlimited".
100 // This property is used to stop the bodies of the simulation flying out when suddenly changing linear speed. It is useful when you have characters than can
101 // changes from stationary to running very quickly such as in an FPS. A common value for a character might be in the few hundreds.
102 UPROPERTY(EditAnywhere, BlueprintReadWrite, Category = Settings, meta = (ClampMin = "0.0"))
103 float MaxLinearAcceleration;
104
105 // A clamp on the effective world-space angular accleration that is passed to the simulation. Units are radian/s/s. The default value effectively means "unlimited".
106 // This has a similar effect to MaxAngularVelocity, except that it is related to the flying out of bodies when the rotation speed suddenly changes. Typical limist for
107 // a character might be around 100.
108 UPROPERTY(EditAnywhere, BlueprintReadWrite, Category = Settings, meta = (ClampMin = "0.0"))
109 float MaxAngularAcceleration;
110
111#if WITH_EDITORONLY_DATA
112 UPROPERTY(meta = (DeprecatedProperty, DeprecationMessage = "ExternalLinearDrag is deprecated. Please use ExternalLinearDragV instead."))
113 float ExternalLinearDrag_DEPRECATED;
114#endif
115
116 // Additional linear drag applied to each body based on total body velocity. This is in addition to per-body linear damping in the physics asset (but see DampingAlpha to control that).
117 // (NOTE: The "V" suffix is to differentiate from the deprecated float property of the same name. It means "Vector" and not "Velocity").
118 //
119 // NOTE: ExternalLinearDragV is in simulation space, so if the RB AnimNode is set to Bone Space the ExternalLinearDragV.Z will be the drag in the
120 // Up direction of the selected bone.
121 UPROPERTY(EditAnywhere, BlueprintReadWrite, Category = Settings)
122 FVector ExternalLinearDragV;
123
124 // Additional velocity that is added to the component velocity so the simulation acts as if the actor is moving at speed, even when stationary.
125 // Vector is in world space. Units are cm/s. Could be used for a wind effects etc. Typical values are similar to the velocity of the object or effect,
126 // and usually around or less than 1000 for characters/wind.
127 UPROPERTY(EditAnywhere, BlueprintReadWrite, Category = Settings)
128 FVector ExternalLinearVelocity;
129
130 // Additional angular velocity that is added to the component angular velocity. This can be used to make the simulation act as if the actor is rotating
131 // even when it is not. E.g., to apply physics to a character on a podium as the camera rotates around it, to emulate the podium itself rotating.
132 // Vector is in world space. Units are rad/s.
133 UPROPERTY(EditAnywhere, BlueprintReadWrite, Category = Settings)
134 FVector ExternalAngularVelocity;
135

602 switch (Space)
603 {
604 case ESimulationSpace::ComponentSpace: return ComponentToWorld;

606 case ESimulationSpace::BaseBoneSpace: return BaseBoneTM * ComponentToWorld;

611inline FVector WorldVectorToSpaceNoScale(
612 ESimulationSpace Space, const FVector& WorldDir, const FTransform& ComponentToWorld, const FTransform& BaseBoneTM)
613{
614 switch (Space)
615 {
616 case ESimulationSpace::ComponentSpace: return ComponentToWorld.InverseTransformVectorNoScale(WorldDir);
617 case ESimulationSpace::WorldSpace: return WorldDir;
618 case ESimulationSpace::BaseBoneSpace:
619 return BaseBoneTM.InverseTransformVectorNoScale(ComponentToWorld.InverseTransformVectorNoScale(WorldDir));
620 default: return FVector::ZeroVector;
621 }
622}
