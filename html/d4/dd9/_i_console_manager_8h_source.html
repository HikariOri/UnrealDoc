81 ECVF_ReadOnly = 0x4,

103 ECVF_Scalability = 0x40,

106 ECVF_ScalabilityGroup = 0x80,

116
117 /* Cvars with this flag will modify the Shader Keystring for Desktop Platforms*/
118 ECVF_DesktopShaderChange = 0x800,
119
120 /* CVars with this flag will be excluded from the device profile previews. */
121 ECVF_ExcludeFromPreview = 0x1000,
122
131 ECVF_SaveForNextBoot = 0x2000,

135 /* Set flags */
136 ECVF_SetFlagMask = 0x00ff0000,

138 // Use to set a cvar without calling all cvar sinks. Much faster, but potentially unsafe. Use only if you know the particular cvar/setting does not require a sink call
139 ECVF_Set_NoSinkCall_Unsafe = 0x00010000,

145 ECVF_Set_ReplaceExistingTag = 0x00040000,

150 ECVF_SetByMask = 0xff000000,
151
152 // The ECVF_SetBy flags are sorted in override order (weak to strong), the value is not serialized. It only affects the override behavior when calling Set()
153
154 // Lowest priority (default after console variable creation)
155 ECVF_SetByConstructor = 0x00000000,
156 // Reserved value for SETBY_ERROR. Please don't use.
157 // SETBY_ERROR = 0x01000000
158 // Set by scalability groups from Scalability.ini (lower priority than game settings so it's easier to override partially). THIS IS AN ARRAY TYPE
159 ECVF_SetByScalability = 0x02000000,
160 // Default priority for engine-level game user settings, platform-specific settings will override this
161 ECVF_SetByGameSetting = 0x03000000,
162 // Set by project settings UI or specific sections in ini file (higher priority than game setting to allow enforcing some settings for this project)
163 ECVF_SetByProjectSetting = 0x04000000,
164 // Used by the [ConsoleVariables] section of Engine.ini as well as FSystemSettings
165 ECVF_SetBySystemSettingsIni = 0x05000000,
166 // Dyanmically loaded/unloaded plugins, used with the History concept to restore cvars on plugin unload. THIS IS AN ARRAY TYPE meaning multiple vales with this SetBy are stored in the history.
167 ECVF_SetByPluginLowPriority = 0x06000000,

169 ECVF_SetByDeviceProfile = 0x07000000,

171 ECVF_SetByPluginHighPriority = 0x08000000,
172 // User settable game overrides, used for GameUserSettings fields that need to override device specific settings
173 ECVF_SetByGameOverride = 0x09000000,
174 // Set by local consolevariables.ini, mostly used for testing multiple projects
175 ECVF_SetByConsoleVariablesIni = 0x0A000000,
176 // Set by hotfix. THIS IS AN ARRAY TYPE meaning multiple vales with this SetBy are stored in the history
177 ECVF_SetByHotfix = 0x0B000000,
178 // Set for previewing in editor. THIS IS AN ARRAY TYPE meaning multiple vales with this SetBy are stored in the history
179 ECVF_SetByPreview = 0x0C000000,
180 // Used by some command line parameters, others use the Console priority instead
181 ECVF_SetByCommandline = 0x0D000000,
182 // Used for high priority temporary debugging or operation modes
183 ECVF_SetByCode = 0x0E000000,
184 // Used for temporarily replacing a variable. THIS IS AN ARRAY TYPE meaning multiple vales with this SetBy are stored in the history
185 ECVF_SetByTemp = 0x0F000000,
186 // Highest priority used via editor UI or or game/editor interactive console
187 ECVF_SetByConsole = 0x10000000,
