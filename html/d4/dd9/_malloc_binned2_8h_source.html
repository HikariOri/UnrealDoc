113 struct FFreeBlock
114 {
115 inline FFreeBlock(uint32 InPageSize, uint16 InBinSize)
116 : NextFreeBlock(0)
117 {
118 NumFreeBins = InPageSize / InBinSize;
119 if (NumFreeBins * InBinSize + sizeof(FFreeBlock) > InPageSize)
120 {
121 NumFreeBins--;
122 }
123 check(NumFreeBins * InBinSize + sizeof(FFreeBlock) <= InPageSize);
124 }
125
126 FORCEINLINE uint32 GetNumFreeBins() const
127 {
128 return NumFreeBins;
129 }
130
131 FORCEINLINE FFreeBlock* GetNextFreeBlock() const
132 {
133 return (FFreeBlock*)NextFreeBlock;
134 }
135
136 FORCEINLINE void SetNextFreeBlock(FFreeBlock* Next)
137 {
138 NextFreeBlock = uint64(Next);
139 }
140
141 inline void* AllocateBin(uint32 BinSize)
142 {
143 --NumFreeBins;
144 // this is a pointer to a whole 64KB block
145 if (IsAligned(this, UE_MB2_LARGE_ALLOC))
146 {
147 return (uint8*)this + UE_MB2_LARGE_ALLOC - (NumFreeBins + 1) * BinSize;
148 }
149
150 // this is a pointer to a single bin within the 64KB block
151 return (uint8*)this + (NumFreeBins * BinSize);
152 }
153
154 constexpr static uint32 MaxBitsForFreeBins = 32 - FMath::CountLeadingZeros((UE_MB2_LARGE_ALLOC - 16) / 16); // max number of free bins in the 64kb page - 4095, fits into 12 bits
155 static_assert(MaxBitsForFreeBins <= 16, "NextFreeBlockIndex should fit in 16 bits");
156
157 uint64 NumFreeBins : 16; // Number of consecutive free bins here, at least 1
158 uint64 NextFreeBlock : 48; // Next free block in another pool
159 };
