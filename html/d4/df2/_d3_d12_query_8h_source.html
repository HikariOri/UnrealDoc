151inline void FD3D12QueryLocation::CopyResultTo(void* Dst) const
152{
153 check(Dst);
154 check(Index < Heap->NumQueries);
155 check(Heap->ResultPtr);
156
157 void const* Src = Heap->ResultPtr + Index * Heap->GetResultSize();
158 FMemory::Memcpy(Dst, Src, Heap->GetResultSize());
159}

161template <typename TValueType>
162inline TValueType FD3D12QueryLocation::GetResult() const
163{
164 check(sizeof(TValueType) >= Heap->GetResultSize());
165
166 TValueType Value;
167 CopyResultTo(&Value);
168 return Value;
169}
