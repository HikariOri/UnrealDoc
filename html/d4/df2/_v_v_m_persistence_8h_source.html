26template <typename TArg>
27TOptional<TArg> NameToShortName(TArg PropertyName)
28{
29 int32 Index;
30 // Remove everything up to and including the 'x' of "__verse_0x".
31 if (!PropertyName.FindChar('x', Index))
32 {
33 return {};
34 }
35 PropertyName.RightChopInline(Index + 1);
36 // Remove the hash followed by '_' that follows "__verse_0x".
37 if (!PropertyName.FindChar('_', Index))
38 {
39 return {};
40 }
41 return PropertyName.RightChop(Index + 1);
42}
