356 template <typename FunctorType>
357 typename TDecay<FunctorType>::Type* Bind(FunctorType&& InFunc)
358 {
359 if (!IsBound(InFunc))
360 {
361 return nullptr;
362 }
363
364#if ULANG_FUNCTION_USES_INLINE_STORAGE
365 constexpr bool bUseInline = sizeof(TStorageOwnerTypeT<FunctorType, bUnique, false>) <= ULANG_FUNCTION_INLINE_SIZE;
366#else
367 constexpr bool bUseInline = false;
368#endif
369
370 using OwnedType = TStorageOwnerTypeT<FunctorType, bUnique, !bUseInline>;
371
372 void* NewAlloc;
373#if ULANG_FUNCTION_USES_INLINE_STORAGE
374 if constexpr (bUseInline)
375 {
376 NewAlloc = &InlineAllocation;
377 }
378 else
379#endif
380 {
381 NewAlloc = GetSystemParams()._HeapMalloc(sizeof(OwnedType)); // , alignof(OwnedType));
382 HeapAllocation = NewAlloc;
383 }
384
385 ULANG_CA_ASSUME(NewAlloc);
386 auto* NewOwned = new (NewAlloc) OwnedType(ForwardArg<FunctorType>(InFunc));
387 return &NewOwned->Obj;
388 }

391 template <typename T, bool bOnHeap>
392 void* TFunction_CopyableOwnedObject<T, bOnHeap>::CloneToEmptyStorage(void* UntypedStorage) const
393 {
394 TFunctionStorage<false>& Storage = *(TFunctionStorage<false>*)UntypedStorage;
395
396 void* NewAlloc;
397 #if ULANG_FUNCTION_USES_INLINE_STORAGE
398 if /* constexpr */ (!bOnHeap)
399 {
400 NewAlloc = &Storage.InlineAllocation;
401 }
402 else
403 #endif
404 {
405 NewAlloc = GetSystemParams()._HeapMalloc(sizeof(TFunction_CopyableOwnedObject)); // , alignof(TFunction_CopyableOwnedObject));
406 Storage.HeapAllocation = NewAlloc;
407 ULANG_CA_ASSUME(NewAlloc);
408 }
409
410 auto* NewOwned = new (NewAlloc) TFunction_CopyableOwnedObject(this->Obj);
411
412 return &NewOwned->Obj;
413 }

556 TFunctionRefBase(FunctorType&& InFunc)
557 {
558 if (auto* Binding = Storage.Bind(ForwardArg<FunctorType>(InFunc)))
559 {
560 using DecayedFunctorType = typename TRemovePointer<decltype(Binding)>::Type;
561
562 Callable = &TFunctionRefCaller<DecayedFunctorType, Ret (ParamTypes...)>::Call;
563
564 #if ULANG_ENABLE_TFUNCTIONREF_VISUALIZATION
565 // We placement new over the top of the same object each time. This is illegal,
566 // but it ensures that the vptr is set correctly for the bound type, and so is
567 // visualizable. We never depend on the state of this object at runtime, so it's
568 // ok.
569 new ((void*)&DebugPtrStorage) TDebugHelper<DecayedFunctorType>;
570 DebugPtrStorage.Ptr = (void*)Binding;
571 #endif
572 }
