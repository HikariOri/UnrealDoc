303 [[nodiscard]] inline TMatrix<T> ToMatrixWithScale() const
304 {
305
306 TMatrix<T> OutMatrix;
307 TransformVectorRegister DiagonalsXYZ;
308 TransformVectorRegister Adds;
309 TransformVectorRegister Subtracts;
310
311 ToMatrixInternal( DiagonalsXYZ, Adds, Subtracts );
312 const TransformVectorRegister DiagonalsXYZ_W0 = VectorSet_W0(DiagonalsXYZ);
313
314 // OutMatrix.M[0][0] = (1.0f - (yy2 + zz2)) * Scale.X; // Diagonal.X
315 // OutMatrix.M[0][1] = (xy2 + wz2) * Scale.X; // Adds.X
316 // OutMatrix.M[0][2] = (xz2 - wy2) * Scale.X; // Subtracts.Z
317 // OutMatrix.M[0][3] = 0.0f; // DiagonalsXYZ_W0.W
318 const TransformVectorRegister AddX_DC_DiagX_DC = VectorShuffle(Adds, DiagonalsXYZ_W0, 0, 0, 0, 0);
319 const TransformVectorRegister SubZ_DC_DiagW_DC = VectorShuffle(Subtracts, DiagonalsXYZ_W0, 2, 0, 3, 0);
320 const TransformVectorRegister Row0 = VectorShuffle(AddX_DC_DiagX_DC, SubZ_DC_DiagW_DC, 2, 0, 0, 2);
321
322 // OutMatrix.M[1][0] = (xy2 - wz2) * Scale.Y; // Subtracts.X
323 // OutMatrix.M[1][1] = (1.0f - (xx2 + zz2)) * Scale.Y; // Diagonal.Y
324 // OutMatrix.M[1][2] = (yz2 + wx2) * Scale.Y; // Adds.Y
325 // OutMatrix.M[1][3] = 0.0f; // DiagonalsXYZ_W0.W
326 const TransformVectorRegister SubX_DC_DiagY_DC = VectorShuffle(Subtracts, DiagonalsXYZ_W0, 0, 0, 1, 0);
327 const TransformVectorRegister AddY_DC_DiagW_DC = VectorShuffle(Adds, DiagonalsXYZ_W0, 1, 0, 3, 0);
328 const TransformVectorRegister Row1 = VectorShuffle(SubX_DC_DiagY_DC, AddY_DC_DiagW_DC, 0, 2, 0, 2);
329
330 // OutMatrix.M[2][0] = (xz2 + wy2) * Scale.Z; // Adds.Z
331 // OutMatrix.M[2][1] = (yz2 - wx2) * Scale.Z; // Subtracts.Y
332 // OutMatrix.M[2][2] = (1.0f - (xx2 + yy2)) * Scale.Z; // Diagonals.Z
333 // OutMatrix.M[2][3] = 0.0f; // DiagonalsXYZ_W0.W
334 const TransformVectorRegister AddZ_DC_SubY_DC = VectorShuffle(Adds, Subtracts, 2, 0, 1, 0);
335 const TransformVectorRegister Row2 = VectorShuffle(AddZ_DC_SubY_DC, DiagonalsXYZ_W0, 0, 2, 2, 3);
336
337 VectorStoreAligned(Row0, &(OutMatrix.M[0][0]));
338 VectorStoreAligned(Row1, &(OutMatrix.M[1][0]));
339 VectorStoreAligned(Row2, &(OutMatrix.M[2][0]));
340
341 // OutMatrix.M[3][0] = Translation.X;
342 // OutMatrix.M[3][1] = Translation.Y;
343 // OutMatrix.M[3][2] = Translation.Z;
344 // OutMatrix.M[3][3] = 1.0f;
345 const TransformVectorRegister Row3 = VectorSet_W1(Translation);
346 VectorStoreAligned(Row3, &(OutMatrix.M[3][0]));
347
348 return OutMatrix;
349 }

427 inline void Blend(const TTransform<T>& Atom1, const TTransform<T>& Atom2, float Alpha)
428 {
429#if !(UE_BUILD_SHIPPING || UE_BUILD_TEST) && WITH_EDITORONLY_DATA
430 // Check that all bone atoms coming from animation are normalized
431 check( Atom1.IsRotationNormalized() );
432 check( Atom2.IsRotationNormalized() );
433#endif
434
435 if( FMath::Abs(Alpha) <= ZERO_ANIMWEIGHT_THRESH )
436 {
437 // if blend is all the way for child1, then just copy its bone atoms
438 (*this) = Atom1;
439 }
440 else if( FMath::Abs(Alpha - 1.0f) <= ZERO_ANIMWEIGHT_THRESH )
441 {
442 // if blend is all the way for child2, then just copy its bone atoms
443 (*this) = Atom2;
444 }
445 else
446 {
447 // Simple linear interpolation for translation and scale.
448 TransformVectorRegister BlendWeight = VectorLoadFloat1(&Alpha);
449
450 Translation = FMath::Lerp<TransformVectorRegister>(Atom1.Translation, Atom2.Translation, BlendWeight);
451 Scale3D = FMath::Lerp<TransformVectorRegister>(Atom1.Scale3D, Atom2.Scale3D, BlendWeight);
452
453 TransformVectorRegister VRotation = VectorLerpQuat(Atom1.Rotation, Atom2.Rotation, BlendWeight);
454
455 // ..and renormalize
456 Rotation = VectorNormalizeQuaternion(VRotation);
457
458 DiagnosticCheckNaN_All(); // MR
459 }
460 }

463 inline void BlendWith(const TTransform<T>& OtherAtom, float Alpha)
464 {
465#if !(UE_BUILD_SHIPPING || UE_BUILD_TEST) && WITH_EDITORONLY_DATA
466 // Check that all bone atoms coming from animation are normalized
467 check( IsRotationNormalized() );
468 check( OtherAtom.IsRotationNormalized() );
469#endif
470
471 if( Alpha > ZERO_ANIMWEIGHT_THRESH )
472 {
473 if( Alpha >= 1.f - ZERO_ANIMWEIGHT_THRESH )
474 {
475 // if blend is all the way for child2, then just copy its bone atoms
476 (*this) = OtherAtom;
477 }
478 else
479 {
480 // Simple linear interpolation for translation and scale.
481 TransformVectorRegister BlendWeight = VectorLoadFloat1(&Alpha);
482 Translation = FMath::Lerp<TransformVectorRegister>(Translation, OtherAtom.Translation, BlendWeight);
483
484 Scale3D = FMath::Lerp<TransformVectorRegister>(Scale3D, OtherAtom.Scale3D, BlendWeight);
485
486 TransformVectorRegister VRotation = VectorLerpQuat(Rotation, OtherAtom.Rotation, BlendWeight);
487
488 // ..and renormalize
489 Rotation = VectorNormalizeQuaternion(VRotation);
490
491 DiagnosticCheckNaN_All();
492 }
493 }

1163 inline static void BlendFromIdentityAndAccumulate(TTransform<T>& FinalAtom, const TTransform<T>& SourceAtom, const ScalarRegister& BlendWeightScalar)
1164 {
1165 const TransformVectorRegister BlendWeight(BlendWeightScalar.Value);
1166 const TransformVectorRegister Const0001 = GlobalVectorConstants::Float0001;
1167 const TransformVectorRegister ConstNegative0001 = VectorSubtract(VectorZero(), Const0001);
1168 const TransformVectorRegister VOneMinusAlpha = VectorSubtract(VectorOne(), BlendWeight);
1169 const TransformVectorRegister DefaultScale = GlobalVectorConstants::Float1110;
1170
1171 // Blend rotation
1172 // To ensure the 'shortest route', we make sure the dot product between the both rotations is positive.
1173 // const float Bias = (|A.B| >= 0 ? 1 : -1)
1174 // BlendedAtom.Rotation = (B * Alpha) + (A * (Bias * (1.f - Alpha)));
1175 // BlendedAtom.Rotation.QuaternionNormalize();
1176 // Note: A = (0,0,0,1), which simplifies things a lot; only care about sign of B.W now, instead of doing a dot product
1177 const TransformVectorRegister RotationB = SourceAtom.Rotation;
1178
1179 const TransformVectorRegister QuatRotationDirMask = VectorCompareGE(RotationB, VectorZero());
1180 const TransformVectorRegister BiasTimesA = VectorSelect(QuatRotationDirMask, Const0001, ConstNegative0001);
1181 const TransformVectorRegister RotateBTimesWeight = VectorMultiply(RotationB, BlendWeight);
1182 const TransformVectorRegister UnnormalizedRotation = VectorMultiplyAdd(BiasTimesA, VOneMinusAlpha, RotateBTimesWeight);
1183
1184 // Normalize blended rotation ( result = (Q.Q >= 1e-8) ? (Q / |Q|) : (0,0,0,1) )
1185 const TransformVectorRegister BlendedRotation = VectorNormalizeSafe(UnnormalizedRotation, Const0001);
1186
1187 // FinalAtom.Rotation = BlendedAtom.Rotation * FinalAtom.Rotation;
1188 FinalAtom.Rotation = VectorQuaternionMultiply2(BlendedRotation, FinalAtom.Rotation);
1189
1190 // Blend translation and scale
1191 // BlendedAtom.Translation = Lerp(Zero, SourceAtom.Translation, Alpha);
1192 // BlendedAtom.Scale = Lerp(0, SourceAtom.Scale, Alpha);
1193 const TransformVectorRegister Zero = VectorZero();
1194 const TransformVectorRegister BlendedTranslation = FMath::Lerp<TransformVectorRegister>(Zero, SourceAtom.Translation, BlendWeight);
1195 const TransformVectorRegister BlendedScale3D = FMath::Lerp<TransformVectorRegister>(Zero, SourceAtom.Scale3D, BlendWeight);
1196
1197 // Apply translation and scale to final atom
1198 // FinalAtom.Translation += BlendedAtom.Translation
1199 // FinalAtom.Scale *= BlendedAtom.Scale
1200 FinalAtom.Translation = VectorAdd( FinalAtom.Translation, BlendedTranslation );
1201 FinalAtom.Scale3D = VectorMultiply( FinalAtom.Scale3D, VectorAdd(DefaultScale, BlendedScale3D));
1202 checkSlow( FinalAtom.IsRotationNormalized() );
1203 }

1300 inline void ToMatrixInternal(TransformVectorRegister& OutDiagonals, TransformVectorRegister& OutAdds, TransformVectorRegister& OutSubtracts) const
1301 {
1302#if !(UE_BUILD_SHIPPING || UE_BUILD_TEST) && WITH_EDITORONLY_DATA
1303 // Make sure Rotation is normalized when we turn it into a matrix.
1304 check( IsRotationNormalized() );
1305#endif
1306
1307 TransformVectorRegister One;
1308 if constexpr (std::is_same_v<T, double>)
1309 {
1310 One = VectorOneDouble();
1311 }
1312 else
1313 {
1314 One = VectorOne();
1315 }
1316
1317 const TransformVectorRegister MyRotation = Rotation; // Load from persistent value, to avoid repeated loads.
1318 const TransformVectorRegister RotationX2Y2Z2 = VectorAdd(MyRotation, MyRotation); // x2, y2, z2
1319 const TransformVectorRegister RotationXX2YY2ZZ2 = VectorMultiply(RotationX2Y2Z2, MyRotation); // xx2, yy2, zz2
1320
1321 // The diagonal terms of the rotation matrix are:
1322 // (1 - (yy2 + zz2)) * scale
1323 // (1 - (xx2 + zz2)) * scale
1324 // (1 - (xx2 + yy2)) * scale
1325 const TransformVectorRegister yy2_xx2_xx2 = VectorSwizzle(RotationXX2YY2ZZ2, 1, 0, 0, 0);
1326 const TransformVectorRegister zz2_zz2_yy2 = VectorSwizzle(RotationXX2YY2ZZ2, 2, 2, 1, 0);
1327 const TransformVectorRegister DiagonalSum = VectorAdd(yy2_xx2_xx2, zz2_zz2_yy2);
1328 const TransformVectorRegister Diagonals = VectorSubtract(One, DiagonalSum);
1329 OutDiagonals = VectorMultiply(Diagonals, Scale3D);
1330
1331 // Grouping the non-diagonal elements in the rotation block by operations:
1332 // ((x*y2,y*z2,x*z2) + (w*z2,w*x2,w*y2)) * scale.xyz and
1333 // ((x*y2,y*z2,x*z2) - (w*z2,w*x2,w*y2)) * scale.yxz
1334 // Rearranging so the LHS and RHS are in the same order as for +
1335 // ((x*y2,y*z2,x*z2) - (w*z2,w*x2,w*y2)) * scale.yxz
1336
1337 // RotBase = x*y2, y*z2, x*z2
1338 // RotOffset = w*z2, w*x2, w*y2
1339 const TransformVectorRegister x_y_x = VectorSwizzle(MyRotation, 0, 1, 0, 0);
1340 const TransformVectorRegister y2_z2_z2 = VectorSwizzle(RotationX2Y2Z2, 1, 2, 2, 0);
1341 const TransformVectorRegister RotBase = VectorMultiply(x_y_x, y2_z2_z2);
1342
1343 const TransformVectorRegister w_w_w = VectorReplicate(MyRotation, 3);
1344 const TransformVectorRegister z2_x2_y2 = VectorSwizzle(RotationX2Y2Z2, 2, 0, 1, 0);
1345 const TransformVectorRegister RotOffset = VectorMultiply(w_w_w, z2_x2_y2);
1346
1347 // Adds = (RotBase + RotOffset)*Scale3D : (x*y2 + w*z2) * Scale3D.X , (y*z2 + w*x2) * Scale3D.Y, (x*z2 + w*y2) * Scale3D.Z
1348 // Subtracts = (RotBase - RotOffset)*Scale3DYZX : (x*y2 - w*z2) * Scale3D.Y , (y*z2 - w*x2) * Scale3D.Z, (x*z2 - w*y2) * Scale3D.X
1349 const TransformVectorRegister Adds = VectorAdd(RotBase, RotOffset);
1350 OutAdds = VectorMultiply(Adds, Scale3D);
1351 const TransformVectorRegister Scale3DYZXW = VectorSwizzle( Scale3D, 1, 2, 0, 3);
1352 const TransformVectorRegister Subtracts = VectorSubtract(RotBase, RotOffset);
1353 OutSubtracts = VectorMultiply(Subtracts , Scale3DYZXW);
1354 }

1423 [[nodiscard]] inline TTransform<T> InverseFast() const
1424 {
1425 // Inverse QST (A) = QST (~A)
1426 // Since A*~A = Identity,
1427 // A(P) = Q(A)*S(A)*P*-Q(A) + T(A)
1428 // ~A(A(P)) = Q(~A)*S(~A)*(Q(A)*S(A)*P*-Q(A) + T(A))*-Q(~A) + T(~A) = Identity
1429 // Q(~A)*Q(A)*S(~A)*S(A)*P*-Q(A)*-Q(~A) + Q(~A)*S(~A)*T(A)*-Q(~A) + T(~A) = Identity
1430 // [Q(~A)*Q(A)]*[S(~A)*S(A)]*P*-[Q(~A)*Q(A)] + [Q(~A)*S(~A)*T(A)*-Q(~A) + T(~A)] = I
1431
1432 // Identity Q = (0, 0, 0, 1) = Q(~A)*Q(A)
1433 // Identity Scale = 1 = S(~A)*S(A)
1434 // Identity Translation = (0, 0, 0) = [Q(~A)*S(~A)*T(A)*-Q(~A) + T(~A)]
1435
1436 // Q(~A) = Q(~A)
1437 // S(~A) = 1.f/S(A)
1438 // T(~A) = - (Q(~A)*S(~A)*T(A)*Q(A))
1439 checkSlow(IsRotationNormalized());
1440 checkSlow(VectorAnyGreaterThan(VectorAbs(Scale3D), GlobalVectorConstants::SmallNumber));
1441
1442 // Invert the scale
1443 const TransformVectorRegister InvScale = VectorSet_W0(GetSafeScaleReciprocal(VectorSet_W1(Scale3D), ScalarRegister(GlobalVectorConstants::SmallNumber)));
1444
1445 // Invert the rotation
1446 const TransformVectorRegister InvRotation = VectorQuaternionInverse(Rotation);
1447
1448 // Invert the translation
1449 const TransformVectorRegister ScaledTranslation = VectorMultiply(InvScale, Translation);
1450 const TransformVectorRegister t2 = VectorQuaternionRotateVector(InvRotation, ScaledTranslation);
1451 const TransformVectorRegister InvTranslation = VectorSet_W0(VectorNegate(t2));
1452
1453 return TTransform<T>(InvRotation, InvTranslation, InvScale);
1454 }
