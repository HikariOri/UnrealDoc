7/* Compact hash table has two distinct features to keep it small
8 1. The Index type adapts to the number of elements, so a smaller table will use a smaller type
9 2. Holes in the table are patched up so all slots up to count are guaranteed to be valid
10
11 For performance reasons we only reset the HashTable part of the lookup table since the indexes will be unused until added.
12 It's the users responsibility to maintain correct item count
13 !!Important!! It's also the users reponsibility to move items from the end of the list into their new spots when removing items.
14
15 You are be able to maintain different views (hashes) of the same data since hole patching is deterministic between two tables.
16 This can be used to allow searching against the same data using different fields as names
17*/
18
19/* Helper to write code against specific types resolves by number of elements. Usage:
20 #define UE_COMPACTHASHTABLE_EXECUTEBYTYPE(Type) // Code
21 UE_COMPACTHASHTABLE_CALLBYTYPE(NumElements)
22 #undef UE_COMPACTHASHTABLE_EXECUTEBYTYPE
23*/
24#define UE_COMPACTHASHTABLE_CALLBYTYPE(NextIndexCount) \
25 switch (UE::Core::CompactHashTable::GetTypeSize(NextIndexCount)) { \
26 case 1: { UE_COMPACTHASHTABLE_EXECUTEBYTYPE(uint8); } break; \
27 case 2: { UE_COMPACTHASHTABLE_EXECUTEBYTYPE(uint16); } break; \
28 case 4: { UE_COMPACTHASHTABLE_EXECUTEBYTYPE(uint32); } break; \
29 default: checkNoEntry(); }

33 // The reference to the given index in chain of a given hash key and return a pointer to it
34 template<typename IndexType>
35 inline IndexType* FindLookupEntry(const uint32 Index, const uint32 Key, IndexType* HashData, const uint32 HashCount, IndexType* NextIndexData, const uint32 NextIndexCount, const TCHAR* Context)
36 {
37 const uint32 HashIndex = Key & (HashCount - 1);
38
39 for (IndexType* Lookup = &HashData[HashIndex]; *Lookup < NextIndexCount; Lookup = &NextIndexData[*Lookup])
40 {
41 if (*Lookup == Index)
42 {
43 return Lookup;
44 }
45 }
46
47 checkf(false, TEXT("Failed to find lookup entry from set from: %s."), Context); // This usually indicates that something changed in the entry that affected the hash so it no longer matches up.
48 return nullptr; // will crash
49 }
50}

90 /* Calculate the size of the hash table from the number of elements in the set */
91 [[nodiscard]] inline constexpr uint32 GetHashCount(uint32 NumElements)
92 {
93 if (!NumElements)
94 {
95 return 0;
96 }
97 if (NumElements < 8)
98 {
99 return 4;
100 }
101
102 // Always use the power of 2 smaller than the current size to prioritize size over speed just a little bit
103 return FGenericPlatformMath::RoundUpToPowerOfTwo(NumElements / 2 + 1); // 255 == 128, 256 == 256
104 }

130 /* Do a full search for an existing element in the table given a function to compare if a found element is what you're looking for */
131 template<typename IndexType, typename PredicateType>
132 [[nodiscard]] inline uint32 Find(uint32 Key, const IndexType* HashData, const uint32 HashCount, const IndexType* NextIndexData, const uint32 NextIndexCount, const PredicateType& Predicate)
133 {
134 for (IndexType ElementIndex = HashData[Key & (HashCount - 1)]; ElementIndex != (IndexType)INDEX_NONE; ElementIndex = NextIndexData[ElementIndex])
135 {
136 if (Predicate(ElementIndex))
137 {
138 // Return the first match, regardless of whether the set has multiple matches for the key or not.
139 return ElementIndex;
140 }

157 /* Remove an element from the list, move the last element into the now empty slot
158 If the item to remove is the last element then the last element's key will be ignored (you can skip calculating it if it's expensive)
159 */
160 template<typename IndexType>
161 inline uint32 Remove(const uint32 Index, const uint32 Key, const uint32 LastIndex, uint32 OptLastKey, IndexType* HashData, const uint32 HashCount, IndexType* NextIndexData, const uint32 NextIndexCount)
162 {
163 checkSlow(LastIndex < NextIndexCount && Index <= LastIndex);
164
165 // Next = Next->Next
166 IndexType* LookupEntry = UE::Core::CompactHashTable::Private::FindLookupEntry<IndexType>(Index, Key, HashData, HashCount, NextIndexData, NextIndexCount, TEXT("Remove initial lookup"));
167 *LookupEntry = NextIndexData[Index];
168
169 if (Index != LastIndex)
170 {
171 // Remap the last element into the now empty spot
172 IndexType* LastLookupEntry = UE::Core::CompactHashTable::Private::FindLookupEntry<IndexType>(LastIndex, OptLastKey, HashData, HashCount, NextIndexData, NextIndexCount, TEXT("Remove last element lookup"));
173 *LastLookupEntry = (IndexType)Index;
174
175 // Move the last elements next index into the new slot as well
176 NextIndexData[Index] = NextIndexData[LastIndex];
177 }
178
179 // Return whatever next is at the original lookup entry that we fixed up. If it happened to be the last index the the previous block would have fixed it up for us
180 return IndexTypeToSizeType(*LookupEntry);
181 }

183 /* Remove an element from the list, shift all indexes down to preserve the order of elements in a list
184 This is a very expensive operation so it should only be used if absolutely necessary (i.e. generally only for user facing data)
185 */
186 template<typename IndexType>
187 inline uint32 RemoveStable(const uint32 Index, const uint32 Key, IndexType* HashData, const uint32 HashCount, IndexType* NextIndexData, const uint32 NextIndexCount)
188 {
189 checkSlow(Index < NextIndexCount);
190
191 constexpr IndexType InvalidIndex = (IndexType)INDEX_NONE;
192
193 // If TestIndex is valid and above Index just remove 1 so it will reference the new location that elements should get relocated to
194 auto FixIndex = [Index](IndexType& TestIndex)
195 {
196 if (TestIndex != InvalidIndex && TestIndex > Index)
197 {
198 --TestIndex;
199 }
200 };
201
202 IndexType* LookupEntry = UE::Core::CompactHashTable::Private::FindLookupEntry<IndexType>(Index, Key, HashData, HashCount, NextIndexData, NextIndexCount, TEXT("Remove stable"));
203 *LookupEntry = NextIndexData[Index];
204 FixIndex(*LookupEntry);
205
206 // Cache return value here since the memory location could move if it was a 'next index' that's located after where Index is
207 const uint32 NextIndex = IndexTypeToSizeType(*LookupEntry);
208
209 // For the hash indexes, just decrement any that are bigger than the removed element
210 for (uint32 HashIndex = 0; HashIndex < HashCount; ++HashIndex)
211 {
212 FixIndex(HashData[HashIndex]);
213 }
214
215 // Decrement values for all next index elements that are before the removed element
216 for (uint32 NextIndexIndex = 0; NextIndexIndex < Index; ++NextIndexIndex)
217 {
218 FixIndex(NextIndexData[NextIndexIndex]);
219 }
220
221 // Move AND Decrement values for all next index elements that are after the removed element
222 for (uint32 NextIndexIndex = Index + 1; NextIndexIndex < NextIndexCount; ++NextIndexIndex)
223 {
224 // Not using FixIndex so we can use the else
225 if (NextIndexData[NextIndexIndex] != InvalidIndex && NextIndexData[NextIndexIndex] > Index)
226 {
227 NextIndexData[NextIndexIndex - 1] = NextIndexData[NextIndexIndex] - 1;
228 }
229 else
230 {
231 NextIndexData[NextIndexIndex - 1] = NextIndexData[NextIndexIndex];
232 }
233 }
234
235 return NextIndex;
236 }
