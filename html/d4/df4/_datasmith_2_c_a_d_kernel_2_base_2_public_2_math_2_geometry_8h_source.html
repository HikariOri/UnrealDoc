32CADKERNEL_API inline FVector2d ComputeCircumCircleCenter(const FVector2d& InPoint0, const FVector2d& InPoint1, const FVector2d& InPoint2)
33{
34 FVector2d Segment_P0_P1 = InPoint1 - InPoint0;
35 FVector2d Segment_P0_P2 = InPoint2 - InPoint0;
36
37 // D = 2(BuCv - BvCu)
38 double D = 2. * Segment_P0_P1 ^ Segment_P0_P2;
39 if (FMath::IsNearlyZero(D, SMALL_NUMBER_SQUARE))
40 {
41 return FVector2d::ZeroVector;
42 }
43
44 // CenterU = 1/D * (Cv.|B|.|B| - By.|C|.|C|) = 1/D * CBv ^ SquareNorms
45 // CenterV = 1/D * (Bu.|B|.|B| - Cu.|C|.|C|) = -1/D * SquareNorms ^ CBu
46 double SquareNormB = Segment_P0_P1.SquaredLength();
47 double SquareNormC = Segment_P0_P2.SquaredLength();
48 double CenterU = (SquareNormB * Segment_P0_P2.Y - SquareNormC * Segment_P0_P1.Y) / D;
49 double CenterV = (SquareNormC * Segment_P0_P1.X - SquareNormB * Segment_P0_P2.X) / D;
50
51 return FVector2d(CenterU, CenterV) + InPoint0;
52}

54CADKERNEL_API inline FVector ComputeCircumCircleCenter(const FVector& Point0, const FVector& Point1, const FVector& Point2)
55{
56 FVector Test = Point1 - Point0;
57 FVector AxisZ = (Point1 - Point0) ^ (Point2 - Point0);
58 AxisZ.Normalize();
59
60 FVector AxisX = Point1 - Point0;
61 AxisX.Normalize();
62 FVector AxisY = AxisZ ^ AxisX;
63
64 FMatrixH Matrix(Point0, AxisX, AxisY, AxisZ);
65 FMatrixH MatrixInverse = Matrix;
66 MatrixInverse.Inverse();
67
68 FVector2d D2Point1 = FVectorUtil::FromVector(MatrixInverse * Point1);
69 FVector2d D2Point2 = FVectorUtil::FromVector(MatrixInverse * Point2);
70
71 double D = 2. * D2Point1 ^ D2Point2;
72 if (FMath::IsNearlyZero(D, SMALL_NUMBER_SQUARE))
73 {
74 return FVector::ZeroVector;
75 }
76
77 double SquareNormB = D2Point1.SquaredLength();
78 double SquareNormC = D2Point2.SquaredLength();
79
80 double CenterU = (SquareNormB * D2Point2.Y - SquareNormC * D2Point1.Y) / D;
81 double CenterV = (SquareNormC * D2Point1.X - SquareNormB * D2Point2.X) / D;
82
83 FVector Center2D(CenterU, CenterV, 0);
84 return Matrix * Center2D;
85}

88CADKERNEL_API inline FVector2d ComputeCircumCircleCenterAndSquareRadius(const FVector2d& InPoint0, const FVector2d& InPoint1, const FVector2d& InPoint2, double& OutSquareRadius)
89{

115template<class PointType>
116inline PointType ProjectPointOnSegment(const PointType& Point, const PointType& InSegmentA, const PointType& InSegmentB, double& OutCoordinate, bool bRestrictCoodinateToInside = true)
117{
118 PointType Segment = InSegmentB - InSegmentA;
119
120 double SquaredLength = Segment | Segment;
121
122 if (SquaredLength <= 0.0)
123 {
124 OutCoordinate = 0.0;
125 return InSegmentA;
126 }
127 else
128 {
129 PointType APoint = Point - InSegmentA;
130 OutCoordinate = (APoint | Segment) / SquaredLength;
131
132 if (bRestrictCoodinateToInside)
133 {
134 if (OutCoordinate < 0.0)
135 {
136 OutCoordinate = 0.0;
137 return InSegmentA;
138 }
139
140 if (OutCoordinate > 1.0)
141 {
142 OutCoordinate = 1.0;
143 return InSegmentB;
144 }
145 }
146
147 PointType ProjectedPoint = Segment * OutCoordinate;
148 ProjectedPoint += InSegmentA;
149 return ProjectedPoint;
150 }
151}

319 virtual inline PointType ProjectPoint(const PointType& InPoint, FVector2d& OutCoordinate)
320 {
321 PointType Segment01 = Point1 - Point0;
322 PointType Segment02 = Point2 - Point0;
323 double SquareLength01 = Segment01.SquaredLength();
324 double SquareLength02 = Segment02.SquaredLength();
325 double Seg01Seg02 = Segment01 | Segment02;
326 double Det = SquareLength01 * SquareLength02 - FMath::Square(Seg01Seg02);
327
328 int32 SideIndex;
329 // If the 3 points are aligned
330 if (FMath::IsNearlyZero(Det))
331 {
332 double MaxLength = SquareLength01;
333 SideIndex = Side01;
334 if (SquareLength02 > MaxLength)
335 {
336 MaxLength = SquareLength02;
337 SideIndex = Side20;
338 }
339
340 PointType Segment12 = Point2 - Point1;
341 if (Segment12.SquaredLength() > MaxLength)
342 {
343 SideIndex = Side12;
344 }
345 }
346 else
347 {
348 // Resolve
349 PointType Segment1Point = InPoint - Point0;
350 double Segment1PointSegment01 = Segment1Point | Segment01;
351 double Segment1PointSegment02 = Segment1Point | Segment02;
352
353 OutCoordinate.X = ((Segment1PointSegment01 * SquareLength02) - (Segment1PointSegment02 * Seg01Seg02)) / Det;
354 OutCoordinate.Y = ((Segment1PointSegment02 * SquareLength01) - (Segment1PointSegment02 * Seg01Seg02)) / Det;
355
356 // tester la solution pour choisir parmi 4 possibilites
357 if (OutCoordinate.X < 0.0)
358 {
359 // the project point is on the segment 02
360 SideIndex = Side20;
361 }
362 else if (OutCoordinate.Y < 0.0)
363 {
364 // the project point is on the segment 01
365 SideIndex = Side01;
366 }
367 else if ((OutCoordinate.X + OutCoordinate.Y) > 1.0)
368 {
369 // the project point is on the segment 12
370 SideIndex = Side12;
371 }
372 else {
373 // the project point is inside the Segment
374 Segment01 = Segment01 * OutCoordinate.X;
375 Segment02 = Segment02 * OutCoordinate.Y;
376 PointType ProjectedPoint = Segment01 + Segment02;
377 ProjectedPoint = ProjectedPoint + Point0;
378 return ProjectedPoint;
379 }
380 }
381
382 // projects the point on the nearest side
383 PointType ProjectedPoint;
384 double SegmentCoordinate;
385 switch (SideIndex)
386 {
387 case Side01:
388 ProjectedPoint = ProjectPointOnSegment<PointType>(InPoint, Point0, Point1, SegmentCoordinate);
389 OutCoordinate.X = SegmentCoordinate;
390 OutCoordinate.Y = 0.0;
391 break;
392 case Side20:
393 ProjectedPoint = ProjectPointOnSegment<PointType>(InPoint, Point0, Point2, SegmentCoordinate);
394 OutCoordinate.X = 0.0;
395 OutCoordinate.Y = SegmentCoordinate;
396 break;
397 case Side12:
398 ProjectedPoint = ProjectPointOnSegment<PointType>(InPoint, Point1, Point2, SegmentCoordinate);
399 OutCoordinate.X = 1.0 - SegmentCoordinate;
400 OutCoordinate.Y = SegmentCoordinate;
401 break;
402 }
403 return ProjectedPoint;
404 }

443inline FVector2d FindIntersectionOfSegments2D(const FSegment2D& SegmentAB, const FSegment2D& SegmentCD, double& OutABIntersectionCoordinate)
444{
445 const FVector2d AB = SegmentAB[1] - SegmentAB[0];
446 const FVector2d DC = SegmentCD[0] - SegmentCD[1];
447 const FVector2d AC = SegmentCD[0] - SegmentAB[0];
448
449 double ParallelCoef = DC ^ AB;
450 if (FMath::IsNearlyZero(ParallelCoef))
451 {
452 const double SquareAB = AB | AB;
453 double CCoordinate = (AB | AC) / SquareAB;
454
455 const FVector2d AD = SegmentCD[1] - SegmentAB[0];
456 double DCoordinate = (AB | AD) / SquareAB;
457
458 if (CCoordinate >= -DOUBLE_KINDA_SMALL_NUMBER && CCoordinate <= 1 + DOUBLE_KINDA_SMALL_NUMBER)
459 {
460 if (DCoordinate >= -DOUBLE_KINDA_SMALL_NUMBER && DCoordinate <= 1 + DOUBLE_KINDA_SMALL_NUMBER)
461 {
462 OutABIntersectionCoordinate = (DCoordinate + CCoordinate) * 0.5;
463 return (SegmentCD[0] + SegmentCD[1]) * 0.5;
464 }
465
466 CCoordinate = FMath::Clamp(CCoordinate, 0., 1.);
467 OutABIntersectionCoordinate = CCoordinate;
468 return SegmentCD[0];
469 }
470 else if (DCoordinate >= -DOUBLE_KINDA_SMALL_NUMBER && DCoordinate <= 1 + DOUBLE_KINDA_SMALL_NUMBER)
471 {
472 DCoordinate = FMath::Clamp(DCoordinate, 0., 1.);
473 OutABIntersectionCoordinate = DCoordinate;
474 return SegmentCD[1];
475 }
476 else
477 {
478 OutABIntersectionCoordinate = 0.5;
479 return (SegmentAB[0] + SegmentAB[1]) * 0.5;
480 }
481 }
482
483 OutABIntersectionCoordinate = (DC ^ AC) / ParallelCoef;
484 OutABIntersectionCoordinate = FMath::Clamp(OutABIntersectionCoordinate, 0., 1.);
485
486 return SegmentAB[0] + AB * OutABIntersectionCoordinate;
487}
