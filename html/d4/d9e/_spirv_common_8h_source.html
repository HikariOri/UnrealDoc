30 TSpirvForwardIterator(T* InPtr, bool bPointsAtHeader = false) :
31 Ptr(InPtr)
32 {
33 // Ignore null pointers so that empty SPIR-V modules can still use begin()/end() loops with zero iterations.
34 if (bPointsAtHeader && Ptr != nullptr)
35 {
36 // Validate header of SPIR-V module
37 const uint32 MagicNumberValue = Ptr[0];
38 check(MagicNumberValue == SpvMagicNumber);
39
40 //const uint32 Version = Ptr[1];
41 //const uint32 Generator = Ptr[2];
42 //const uint32 Bound = Ptr[3];
43 const uint32 ZeroCheckValue = Ptr[4];
44 checkf(ZeroCheckValue == 0, TEXT("redundancy check for SPIR-V module failed: WORD[4] is non-zero"));
45
46 // Move pointer to first instruction
47 Ptr += 5;
48 }
49 }
50
52 SpvOp Opcode() const
53 {
54 return static_cast<SpvOp>(*Ptr & SpvOpCodeMask);
55 }
56
58 uint32 WordCount() const
59 {
60 return (*Ptr >> SpvWordCountShift) & SpvOpCodeMask;
61 }
62
64 uint32 Operand(int32 WordOffset) const
65 {
66 return Ptr[WordOffset];
67 }
68
70 template <typename TDst>
71 TDst OperandAs(int32 WordOffset) const
72 {
73 return *reinterpret_cast<const TDst*>(&(Ptr[WordOffset]));
74 }
75
77 const ANSICHAR* OperandAsString(int32 WordOffset) const
78 {
79 return reinterpret_cast<const ANSICHAR*>(&(Ptr[WordOffset]));
80 }
81
83 T* operator * () const
84 {
85 return Ptr;
86 }
87
89 TSpirvForwardIterator& operator ++ ()
90 {
91 const uint32 Words = WordCount();
92 checkf(Words > 0, TEXT("Invalid SPIR-V instruction with word count of zero"));
93 Ptr += Words;
94 return *this;
95 }
96
98 TSpirvForwardIterator operator ++ (int)
99 {
100 TSpirvForwardIterator Current(*this);
101 this->operator++();
102 return Current;
103 }
104
106 bool operator == (const TSpirvForwardIterator& Other) const
107 {
108 return Ptr == Other.Ptr;
109 }
110
112 bool operator != (const TSpirvForwardIterator& Other) const
113 {
114 return Ptr != Other.Ptr;
115 }
116
117private:
118 T* Ptr;
119};
