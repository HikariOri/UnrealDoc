38inline bool VArrayBase::IsInBounds(const VInt& Index, const uint32 Bounds) const
39{
40 if (Index.IsInt64())
41 {
42 const int64 IndexInt64 = Index.AsInt64();
43 return (IndexInt64 >= 0) && (IndexInt64 < Bounds);
44 }
45 else
46 {
47 // Array maximum size is limited to the maximum size of a unsigned 32-bit integer.
48 // So even if it's a `VHeapInt`, if it fails the `IsInt64` check, it is definitely out-of-bounds.
49 return false;
50 }

91template <EWriteMode WriteMode /* = EWriteMode::Default */>
92inline void VArrayBase::ConvertDataToVValues(FAllocationContext Context, uint32 NewCapacity)
93{
94 if (GetArrayType() != EArrayType::VValue)
95 {
96 uint32 Num = this->Num();
97 VBuffer NewBuffer = VBuffer(Context, Num, NewCapacity, EArrayType::VValue);
98 for (uint32 Index = 0; Index < Num; ++Index)
99 {
100 new (&NewBuffer.GetData<TWriteBarrier<VValue>>()[Index]) TWriteBarrier<VValue>(Context, GetValue(Index));
101 }
102
103 // If an append is rolled back, the buffer must remain legible to the GC.
104 FMemory::Memzero(NewBuffer.GetData<VValue>() + Num, (NewCapacity - Num) * Verse::ByteLength(EArrayType::VValue));
105
106 // We need to see the store to ArrayType/Num/all the VValues before the GC
107 // sees the buffer itself.
108 SetBufferWithStoreBarrier<WriteMode>(Context, NewBuffer);
109 }
110}

112template <EWriteMode WriteMode /* = EWriteMode::NonTransactional */>
113inline void VArrayBase::SetValue(FAllocationContext Context, uint32 Index, VValue Value)
114{
115 static_assert(WriteMode != EWriteMode::Default, "Use EWriteMode::NonTransactional or EWriteMode::Transactional");
116 checkSlow(Index < Capacity());
117 EArrayType ArrayType = GetArrayType();
118 if (ArrayType == EArrayType::VValue)
119 {
120 SetVValue<WriteMode>(Context, Index, Value);
121 }
122 else if (ArrayType != DetermineArrayType(Value))
123 {
124 ConvertDataToVValues<WriteMode>(Context, Capacity());
125 SetVValue<WriteMode>(Context, Index, Value);
126 }
127 else
128 {
129 switch (ArrayType)
130 {
131 case EArrayType::Int32:
132 SetInt32<WriteMode>(Index, Value.AsInt32());
133 break;
134 case EArrayType::Char8:
135 SetChar<WriteMode>(Index, Value.AsChar());
136 break;
137 case EArrayType::Char32:
138 SetChar32<WriteMode>(Index, Value.AsChar32());
139 break;
140 case EArrayType::VValue:
141 case EArrayType::None:
142 default:
143 V_DIE("Unhandled EArrayType (%u) encountered!", static_cast<uint32>(ArrayType));
144 }
145 }
146}
