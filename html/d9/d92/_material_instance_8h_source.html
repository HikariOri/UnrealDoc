548template<class T>
549bool CompareValueArraysByExpressionGUID(const TArray<T>& InA, const TArray<T>& InB)
550{
551 if (InA.Num() != InB.Num())
552 {
553 return false;
554 }
555 if (!InA.Num())
556 {
557 return true;
558 }
559 TArray<T> AA(InA);
560 TArray<T> BB(InB);
561 AA.Sort([](const T& A, const T& B) { return B.ExpressionGUID < A.ExpressionGUID; });
562 BB.Sort([](const T& A, const T& B) { return B.ExpressionGUID < A.ExpressionGUID; });
563 return AA == BB;
564}
