188 /*
189 * [EXPERIMENTAL] Apply a momentumless correction to the particle transform, usually as a result of a server correction.
190 * This will shift the particle by the supplied delta and handle updating of friction anchors or anything else that might prevent or undo the shift.
191 * If bApplyToConnectedBodies is true, any particle attached by a joint with locked linear limits will also get moved.
192 * NOTE: must be called prior to Integrate() to be effective.
193 * NOTE: be careful with bApplyToConnectedBodies - only one particle in the connected graph should have ApplyParticleTransformCorrectionDelta called on

279 // Resets VSmooth value to something plausible based on external forces to prevent object from going back to sleep if it was just impulsed.
280 template <bool bPersistent>
281 void ResetVSmoothFromForces(TPBDRigidParticleHandleImp<FReal, 3, bPersistent>& Particle)
282 {
283 const FReal SmoothRate = FMath::Clamp(CVars::SmoothedPositionLerpRate, 0.0f, 1.0f);
284
285 // Reset VSmooth to something roughly in the same direction as what V will be after integration.
286 // This is temp fix, if this is only re-computed after solve, island will get incorrectly put back to sleep even if it was just impulsed.
287 FReal FakeDT = (FReal)1. / (FReal)30.;
288 if (Particle.LinearImpulseVelocity().IsNearlyZero() == false || Particle.Acceleration().IsNearlyZero() == false)
289 {
290 const FVec3 PredictedLinearVelocity = Particle.GetV() + Particle.Acceleration() * FakeDT + Particle.LinearImpulseVelocity();
291 Particle.VSmooth() =FMath::Lerp(Particle.VSmooth(), PredictedLinearVelocity, SmoothRate);
292 }
293 if (Particle.AngularImpulseVelocity().IsNearlyZero() == false || Particle.AngularAcceleration().IsNearlyZero() == false)
294 {
295 const FVec3 PredictedAngularVelocity = Particle.GetW() + Particle.AngularAcceleration() * FakeDT + Particle.AngularImpulseVelocity();
296 Particle.WSmooth() = FMath::Lerp(Particle.WSmooth(), PredictedAngularVelocity, SmoothRate);
297 }
298 }
