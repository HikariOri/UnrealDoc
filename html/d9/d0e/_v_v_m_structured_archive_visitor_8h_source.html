55 template <typename T>
56 void Visit(TWriteBarrier<T>& Value, const TCHAR* ElementName)
57 {
58 if constexpr (TWriteBarrier<T>::bIsVValue)
59 {
60 T ScratchValue = Value.Get();
61 Visit(ScratchValue, ElementName);
62 if (IsLoading() || IsModifying())
63 {
64 Value.Set(Context, ScratchValue);
65 }
66 }
67 else if constexpr (!TWriteBarrier<T>::bIsAux)
68 {
69 VCell* ScratchValue = Value.Get();
70 Visit(ScratchValue, ElementName);
71 if (IsLoading() || IsModifying())
72 {
73 Value.Set(Context, ScratchValue ? &ScratchValue->StaticCast<T>() : nullptr);
74 }
75 }
76 else
77 {
78 static_assert(bAlwaysFalse<T>, "Cannot serialize TAux");
79 }
80 }
