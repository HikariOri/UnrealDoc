104 // Create a circular arc with given center, radius, and angle range
105 static TPolyBezierSpline<ValueType> CreateCircleArc(
106 const ValueType& Center,
107 float Radius,
108 float StartAngle,
109 float EndAngle,
110 int32 NumSegments = 4)
111 {
112 // Ensure angles are in the right order
113 if (EndAngle < StartAngle)
114 {
115 float Temp = StartAngle;
116 StartAngle = EndAngle;
117 EndAngle = Temp;
118 }
119
120 // Ensure reasonable segment count
121 NumSegments = FMath::Max(1, NumSegments);
122
123 // Calculate angle per segment
124 float AngleSpan = EndAngle - StartAngle;
125 float AnglePerSegment = AngleSpan / static_cast<float>(NumSegments);
126
127 // For large angles per segment, increase segment count for better approximation
128 if (AnglePerSegment > UE_PI/4)
129 {
130 NumSegments = FMath::CeilToInt(AngleSpan / (UE_PI/4));
131 AnglePerSegment = AngleSpan / static_cast<float>(NumSegments);
132 }
133
134 // Pre-compute all points along the arc for exact positioning
135 TArray<ValueType> Points;
136 Points.SetNum(NumSegments + 1);
137
138 for (int32 i = 0; i <= NumSegments; i++)
139 {
140 float Angle = StartAngle + (static_cast<float>(i) * AnglePerSegment);
141 Points[i] = Center + ValueType(
142 Radius * FMath::Cos(Angle),
143 Radius * FMath::Sin(Angle),
144 0);
145 }
146
147 // Calculate first segment with precise tangents
148 ValueType P0 = Points[0];
149 ValueType P3 = Points[1];
150
151 // Get normalized direction vectors from center to points
152 ValueType Dir0 = Math::GetSafeNormal(P0 - Center);
153 ValueType Dir3 = Math::GetSafeNormal(P3 - Center);
154
155 // Calculate perpendicular vectors (normalized)
156 ValueType Perp0 = ValueType(-Dir0.Y, Dir0.X, 0);
157 ValueType Perp3 = ValueType(-Dir3.Y, Dir3.X, 0);
158
159 // Calculate precise tangent scale
160 float TangentScale = Radius * (4.0f / 3.0f) *
161 FMath::Tan(AnglePerSegment / 4.0f);
162
163 // Create control points with exact perpendicular tangents
164 ValueType P1 = P0 + Perp0 * TangentScale;
165 ValueType P2 = P3 - Perp3 * TangentScale;
166
167 TPolyBezierSpline<ValueType> Result(
168 FAnchor{P0, P0, P1},
169 FAnchor{P2, P3, P3},
170 EParameterizationPolicy::Uniform
171 );
172
173 // Add remaining segments with carefully calculated control points
174 for (int32 i = 1; i < NumSegments; i++)
175 {
176 P0 = Points[i];
177 P3 = Points[i+1];
178
179 // Calculate exact perpendicular vectors for this segment
180 Dir0 = Math::GetSafeNormal(P0 - Center);
181 Dir3 = Math::GetSafeNormal(P3 - Center);
182 Perp0 = ValueType(-Dir0.Y, Dir0.X, 0);
183 Perp3 = ValueType(-Dir3.Y, Dir3.X, 0);
184
185 // Create control points with exact perpendicular tangents
186 P1 = P0 + Perp0 * TangentScale;
187 P2 = P3 - Perp3 * TangentScale;
188
189 const int32 NewAnchorIdx = Result.AppendAnchor(FAnchor { P2, P3, P3 }, EParameterizationPolicy::Uniform);
190 const int32 OldAnchorIdx = NewAnchorIdx - 1;
191 Result.SetHandleOutOfAnchor(OldAnchorIdx, P1);
192 }
193
194 return Result;
195 }

217 // Create an ellipse with X and Y radii
218 static TPolyBezierSpline<ValueType> CreateEllipse(
219 const ValueType& Center,
220 float RadiusX,
221 float RadiusY,
222 int32 NumSegments = 4)
223 {
224 // Ensure at least 4 segments for good quality
225 if (NumSegments < 4) NumSegments = 4;
226
227 // Calculate angle per segment
228 const float AnglePerSegment = 2.0f * UE_PI / static_cast<float>(NumSegments);
229
230 // Start with first point at angle 0
231 float CurrentAngle = 0.0f;
232 ValueType P0 = Center + ValueType(RadiusX * FMath::Cos(CurrentAngle),
233 RadiusY * FMath::Sin(CurrentAngle), 0);
234
235 // Calculate next point
236 CurrentAngle += AnglePerSegment;
237 ValueType P3 = Center + ValueType(RadiusX * FMath::Cos(CurrentAngle),
238 RadiusY * FMath::Sin(CurrentAngle), 0);
239
240 // Calculate correct tangent scale for this angle
241 const float TangentScale = (4.0f/3.0f) * FMath::Tan(AnglePerSegment / 4.0f);
242
243 // Calculate tangent vectors (scaled appropriately for ellipse)
244 ValueType Tangent0 = ValueType(-RadiusX * FMath::Sin(0.0f),
245 RadiusY * FMath::Cos(0.0f), 0) * TangentScale;
246 ValueType Tangent3 = ValueType(-RadiusX * FMath::Sin(CurrentAngle),
247 RadiusY * FMath::Cos(CurrentAngle), 0) * TangentScale;
248
249 ValueType P1 = P0 + Tangent0;
250 ValueType P2 = P3 - Tangent3;
251
252 // Create the spline with the first segment
253 TPolyBezierSpline<ValueType> Result(
254 FAnchor{P0, P0, P1},
255 FAnchor{P2, P3, P3},
256 EParameterizationPolicy::Uniform
257 );
258
259 // Add remaining segments
260 for (int32 i = 1; i < NumSegments; ++i)
261 {
262 P0 = P3; // Start from previous endpoint
263 Tangent0 = Tangent3; // Reuse previous tangent
264
265 // Calculate next endpoint
266 CurrentAngle += AnglePerSegment;
267 P3 = Center + ValueType(RadiusX * FMath::Cos(CurrentAngle),
268 RadiusY * FMath::Sin(CurrentAngle), 0);
269
270 // Calculate tangent at next point
271 Tangent3 = ValueType(-RadiusX * FMath::Sin(CurrentAngle),
272 RadiusY * FMath::Cos(CurrentAngle), 0) * TangentScale;
273
274 // Calculate control points
275 P1 = P0 + Tangent0;
276 P2 = P3 - Tangent3;
277
278 // Add the segment
279 const int32 NewAnchorIdx = Result.AppendAnchor(P2, P3, P3);
280 const int32 OldAnchorIdx = NewAnchorIdx - 1;
281 Result.SetHandleOutOfAnchor(OldAnchorIdx, P1);
282 }
283
284 // Set as closed loop
285 Result.SetClosedLoop(true);
286
287 // Apply consistent parameterization
288 Result.Reparameterize(EParameterizationPolicy::Centripetal);
289
290 return Result;
291 }

344 float FindNearestOnSegment(const ValueType& Point, int32 SegmentIndex, float& OutSquaredDistance) const
345 {
346 if (!IsValidSegmentIndex(SegmentIndex))
347 {
348 OutSquaredDistance = TNumericLimits<float>::Max();
349 return 0.0f;
350 }
351
352 // Get Bezier control points for this segment
353 TStaticArray<ValueType, 4> Coeffs = {};
354
355 const ValueType* P0Ptr = nullptr;
356 const ValueType* P1Ptr = nullptr;
357 const ValueType* P2Ptr = nullptr;
358 const ValueType* P3Ptr = nullptr;
359
360 if (!IsClosingSegment(SegmentIndex))
361 {
362 P0Ptr = &Base::GetValue(P0Idx(SegmentIndex)); // A_s
363 P1Ptr = &Base::GetValue(P1Idx(SegmentIndex)); // Out_s
364 P2Ptr = &Base::GetValue(P2Idx(SegmentIndex)); // In_{s+1}
365 P3Ptr = &Base::GetValue(P3Idx(SegmentIndex)); // A_{s+1}
366 }
367 else
368 {
369 const int32 NumAnchors = GetNumAnchorPoints();
370
371 // Closing segment: A_last → A_0
372 P0Ptr = &Base::GetValue(Close_P0Idx(NumAnchors)); // A_last
373 P1Ptr = &Base::GetValue(Close_P1Idx(NumAnchors)); // Out_last
374 P2Ptr = &Base::GetValue(Close_P2Idx()); // In_0
375 P3Ptr = &Base::GetValue(Close_P3Idx()); // A_0
376 }
377
378 const ValueType& P0 = *P0Ptr;
379 const ValueType& P1 = *P1Ptr;
380 const ValueType& P2 = *P2Ptr;
381 const ValueType& P3 = *P3Ptr;
382
383 // Setup coefficients in standard Bezier polynomial form relative to test point
384 Coeffs[0] = P0 - Point; // constant term
385 Coeffs[1] = (P1 - P0) * 3; // linear coefficient
386 Coeffs[2] = (P2 - P1*2 + P0) * 3; // quadratic coefficient
387 Coeffs[3] = P3 - P2*3 + P1*3 - P0; // cubic coefficient
388
389 const float LocalT = Math::FindNearestPoint_Cubic(MakeArrayView(Coeffs), 0.0f, 1.0f, OutSquaredDistance);
390 return MapLocalSegmentParameterToGlobal(SegmentIndex, LocalT);
391 }

393 float PointDistanceSqToSegmentBBox(const ValueType &Point, int32 SegmentIndex) const
394 {
395 // Decide whether this is the synthetic closing segment
396
397 // Get the control points
398 const ValueType* P0Ptr = nullptr;
399 const ValueType* P1Ptr = nullptr;
400 const ValueType* P2Ptr = nullptr;
401 const ValueType* P3Ptr = nullptr;
402
403 if (!IsClosingSegment(SegmentIndex))
404 {
405 P0Ptr = &Base::GetValue(P0Idx(SegmentIndex));
406 P1Ptr = &Base::GetValue(P1Idx(SegmentIndex));
407 P2Ptr = &Base::GetValue(P2Idx(SegmentIndex));
408 P3Ptr = &Base::GetValue(P3Idx(SegmentIndex));
409 }
410 else
411 {
412 const int32 NumAnchors = GetNumAnchorPoints();
413
414 P0Ptr = &Base::GetValue(Close_P0Idx(NumAnchors)); // A_last
415 P1Ptr = &Base::GetValue(Close_P1Idx(NumAnchors)); // Out_last
416 P2Ptr = &Base::GetValue(Close_P2Idx()); // In_0
417 P3Ptr = &Base::GetValue(Close_P3Idx()); // A_0
418 }
419
420 const ValueType& P0 = *P0Ptr;
421 const ValueType& P1 = *P1Ptr;
422 const ValueType& P2 = *P2Ptr;
423 const ValueType& P3 = *P3Ptr;
424
425 // For cubics, convex hull is defined by control points, so use them to compute BBox.
426 ValueType BBoxMin = P0.ComponentMin(P1);
427 ValueType BBoxMax = P0.ComponentMax(P1);
428
429 BBoxMin = BBoxMin.ComponentMin(P2);
430 BBoxMax = BBoxMax.ComponentMax(P2);
431
432 BBoxMin = BBoxMin.ComponentMin(P3);
433 BBoxMax = BBoxMax.ComponentMax(P3);
434
435 // Get point on Bbox nearest to input point and compute distance sq
436 ValueType PtOnBox = BBoxMin.ComponentMax(BBoxMax.ComponentMin(Point));
437 return float(Math::SizeSquared(PtOnBox - Point));
438 }

440 virtual float FindNearest(const ValueType& Point, float& OutSquaredDistance) const override
441 {
442 const int32 NumSegments = GetNumberOfSegments();
443 float BestDistSq;
444 float BestParam = FindNearestOnSegment(Point, 0, BestDistSq);
445
446 float SegmentDistSq;
447 if (Private::UseFindNearestUseSplineCulling())
448 {
449 for (int32 i = 1; i < NumSegments; ++i)
450 {
451 // For cubic spline: compute AABB and distance to point.
452 // If distance > best result distance then reject segment. Saves a lot of computation (see FindNearestOnSegment)
453 //
454 if (PointDistanceSqToSegmentBBox(Point, i) >= BestDistSq)
455 {
456 continue;
457 }
458
459 const float SplineParam = FindNearestOnSegment(Point, i, SegmentDistSq);
460 if (SegmentDistSq < BestDistSq)
461 {
462 BestDistSq = SegmentDistSq;
463 BestParam = SplineParam;
464 }
465
466 if (FMath::IsNearlyZero(BestDistSq))
467 {
468 break;
469 }
470 }
471 }
472 else
473 {
474 for (int32 i = 1; i < NumSegments; ++i)
475 {
476 const float SplineParam = FindNearestOnSegment(Point, i, SegmentDistSq);
477 if (SegmentDistSq < BestDistSq)
478 {
479 BestDistSq = SegmentDistSq;
480 BestParam = SplineParam;
481 }
482 }
483 }
484 OutSquaredDistance = BestDistSq;
485
486 return BestParam;
487 }

495 template<int32 Order>
496 ValueType EvaluateDerivative(float Parameter) const
497 {
498 FWindow Window;
499 FWindowStore Scratch;
500 FindWindow(Parameter, Window, Scratch);
501
502 // If FindWindow fails, it will return an array of nullptr. InterpolateWindow assumes validity of elements.
503 if (!Math::FSplineValidation::IsValidWindow<ValueType, Base::WindowSize>(Window) ||
504 !Math::FSplineValidation::IsValidParameter(Parameter))
505 {
506 return Math::FSplineValidation::GetDefaultValue<ValueType>();
507 }
508
509 // Order 0 is just regular evaluation
510 if constexpr (Order == 0)
511 {
512 return InterpolateWindow(Window, Parameter);
513 }
514
515 // Convert global parameter to segment information
516 float LocalT;
517 int32 SegmentIndex = FindIndexForParameter(Parameter, LocalT);
518
519 if (SegmentIndex == INDEX_NONE)
520 {
521 return Math::FSplineValidation::GetDefaultValue<ValueType>();
522 }
523
524 const float SegmentScale = GetSegmentParameterRange(SegmentIndex).Length();
525
526 // Compute the derivative with proper scaling
527 return Math::TBezierDerivativeCalculator<ValueType, Order>::Compute(
528 Window, LocalT, SegmentScale);
529 }

552 if (SegmentIndex < 0 || SegmentIndex >= NumSegments || Base::PairKnots.Num() == 0 || !Base::PairKnots.IsValidIndex(SegmentIndex))

557 SegmentRange.Min = Base::PairKnots[SegmentIndex].Value;
558

560 {
561 // Open: knots are standard boundaries, so we still use i+1.
562 if (SegmentIndex + 1 < Base::PairKnots.Num())
563 {
564 SegmentRange.Max = Base::PairKnots[SegmentIndex + 1].Value;
565 }
566 return SegmentRange;
567 }
568
569 // Closed: PairKnots holds starts of each segment; the final segment
570 // closes the loop at Min + LoopKnotDelta.
571 if (SegmentIndex < NumSegments - 1)
572 {
573 SegmentRange.Max = Base::PairKnots[SegmentIndex + 1].Value;
574 }
575 else
576 {
577 SegmentRange.Max = SegmentRange.Min + LoopKnotDelta;
578 }

662 virtual int32 SetParameter(int32 Index, float NewParameter) override
663 {
664 const int32 NumAnchors = GetNumAnchorPoints();
665 const int32 NumSegments = GetNumberOfSegments();
666
667 // Cannot set parameters for empty splines, fail.
668 if (NumAnchors <= 0)
669 {
670 return INDEX_NONE;
671 }
672
673 // Trivial case, no possible reordering or numerical issues.
674 if (NumAnchors == 1)
675 {
676 if (Index == 0)
677 {
678 Base::PairKnots[Index].Value = NewParameter;
679 return 0;
680 }
681 else
682 {
683 return INDEX_NONE;
684 }
685 }
686
687 // Setting the virtual closing knot.
688 if (Index == NumAnchors && HasClosingSegment())
689 {
690 // No reordering when manipulating the closing knot, fail.
691 if (NewParameter <= Base::PairKnots.Last().Value)
692 {
693 return INDEX_NONE;
694 }
695
696 LoopKnotDelta = NewParameter - Base::PairKnots.Last().Value;
697 return Index;
698 }
699
700 // Because we special case the closing point above, we can trivially bounds check here.
701 if (Index < 0 || Index >= NumAnchors)
702 {
703 return INDEX_NONE;
704 }
705
706 const int32 SegmentIndex = Index == 0 ? 0 : Index - 1;
707 const int32 LocalPointIndex = Index == 0 ? 0 : 3;
708
709 float OldParameter = Base::PairKnots[Index].Value;
710
711 // Early exit if parameter isn't changing
712 if (OldParameter == NewParameter)
713 {
714 return Index;
715 }
716
717 UE_LOGF(LogSpline, Verbose, "\t");
718 UE_LOGF(LogSpline, Verbose, "Before SetParameter(%d, %f):", Index, NewParameter);
719 Base::Dump();
720
721 // Prevent reuse of an existing knot.
722 const float DesiredNewParameter = NewParameter;
723 FKnotSearchCache SearchCache{ Base::PairKnots };
724 const float ActualNewParameter = Base::GetNearestAvailableKnotValue(DesiredNewParameter, SearchCache);
725
726 const float DesiredToOldDistance = FMath::Abs(DesiredNewParameter - OldParameter);
727 const float DesiredToActualNewDistance = FMath::Abs(DesiredNewParameter - ActualNewParameter);
728
729 if (DesiredToActualNewDistance > DesiredToOldDistance)
730 {
731 // If our nearest valid knot is further away from the desired value than our current value is, we need to just keep our current value
732 return Index;
733 }
734
735 int32 ReturnIndex = Index;
736 Base::SetKnot(Index, NewParameter);
737
738 // Fix the knot vector by flipping segments which have inverted parameter ranges until it is valid again.
739 if (NewParameter < OldParameter)
740 {
741 int32 LeftSegmentIndex = Index - 1;
742
743 // Repeated left flips
744 while (LeftSegmentIndex >= 0 && GetSegmentParameterRange(LeftSegmentIndex).Min > GetSegmentParameterRange(LeftSegmentIndex).Max)
745 {
746 FlipSegment(LeftSegmentIndex);
747 LeftSegmentIndex--;
748 ReturnIndex--;
749 }
750 }
751 else if (NewParameter > OldParameter)
752 {
753 int32 RightSegmentIndex = Index;
754
755 while (RightSegmentIndex <= NumSegments - 1 && !IsClosingSegment(RightSegmentIndex) && GetSegmentParameterRange(RightSegmentIndex).Min > GetSegmentParameterRange(RightSegmentIndex).Max)
756 {
757 FlipSegment(RightSegmentIndex);
758 RightSegmentIndex++;
759 ReturnIndex++;
760 }
761 }
762
763 UE_LOGF(LogSpline, Verbose, "\t");
764 UE_LOGF(LogSpline, Verbose, "After SetParameter(%d, %f):", Index, NewParameter);
765 Base::Dump();
766
767 return ReturnIndex;
768 }

770 void FlipSegment(int32 Segment)
771 {
772 const int32 NumSegments = GetNumberOfSegments();
773 const int32 NumAnchors = GetNumAnchorPoints();
774 if (Segment < 0 || Segment >= NumSegments || NumAnchors < 2)
775 {
776 return;
777 }
778
779 UE_LOGF(LogSpline, Verbose, "\t");
780 UE_LOGF(LogSpline, Verbose, "Before FlipSegment(%d):", Segment);
781 Base::Dump();
782
783 // 1) Flip this segment's Bezier window: P0<->P3, P1<->P2
784 {
785 int32 Idx0, Idx1, Idx2, Idx3;
786 GetBezierIndices(Segment, Idx0, Idx1, Idx2, Idx3);
787
788 ValueType P0 = Base::GetValue(Idx0);
789 ValueType P1 = Base::GetValue(Idx1);
790 ValueType P2 = Base::GetValue(Idx2);
791 ValueType P3 = Base::GetValue(Idx3);
792
793 Base::SetValue(Idx0, P3);
794 Base::SetValue(Idx1, P2);
795 Base::SetValue(Idx2, P1);
796 Base::SetValue(Idx3, P0);
797 }
798
799 // 2) Fix left neighbor (Segment-1): keep its out-tangent but move its end point to match
800 if (Segment > 0 || (HasClosingSegment() && Segment == 0))
801 {
802 const int32 LeftSeg =
803 (Segment > 0) ? (Segment - 1) : (NumSegments - 1); // wrap in closed case
804
805 int32 ThisIdx0, ThisIdx1, ThisIdx2, ThisIdx3;
806 int32 LeftIdx0, LeftIdx1, LeftIdx2, LeftIdx3;
807
808 GetBezierIndices(Segment, ThisIdx0, ThisIdx1, ThisIdx2, ThisIdx3);
809 GetBezierIndices(LeftSeg, LeftIdx0, LeftIdx1, LeftIdx2, LeftIdx3);
810
811 const ValueType ThisP0 = Base::GetValue(ThisIdx0);
812 const ValueType LeftP2 = Base::GetValue(LeftIdx2);
813 const ValueType LeftP3 = Base::GetValue(LeftIdx3);
814
815 const ValueType Delta = LeftP3 - LeftP2; // original out-tangent
816 const ValueType NewEnd = ThisP0;
817 const ValueType NewIn = NewEnd - Delta;
818
819 Base::SetValue(LeftIdx3, NewEnd);
820 Base::SetValue(LeftIdx2, NewIn);
821 }
822
823 // 3) Fix right neighbor (Segment+1): keep its in-tangent but move its start point to match
824 if (Segment < NumSegments - 1 || IsClosingSegment(Segment))
825 {
826 const int32 RightSeg =
827 (Segment < NumSegments - 1) ? (Segment + 1) : 0; // wrap in closed case
828
829 int32 ThisIdx0, ThisIdx1, ThisIdx2, ThisIdx3;
830 int32 RightIdx0, RightIdx1, RightIdx2, RightIdx3;
831
832 GetBezierIndices(Segment, ThisIdx0, ThisIdx1, ThisIdx2, ThisIdx3);
833 GetBezierIndices(RightSeg, RightIdx0, RightIdx1, RightIdx2, RightIdx3);
834
835 const ValueType ThisP3 = Base::GetValue(ThisIdx3);
836 const ValueType RightP0 = Base::GetValue(RightIdx0);
837 const ValueType RightP1 = Base::GetValue(RightIdx1);
838
839 const ValueType Delta = RightP0 - RightP1; // original in-tangent
840 const ValueType NewStart = ThisP3;
841 const ValueType NewOut = NewStart - Delta;
842
843 Base::SetValue(RightIdx0, NewStart);
844 Base::SetValue(RightIdx1, NewOut);
845 }
846
847 // 4) Swap knot intervals for the flipped segment & its neighbor interval
848 Base::SwapKnots(Segment, Segment + 1);
849
850 UE_LOGF(LogSpline, Verbose, "\t");
851 UE_LOGF(LogSpline, Verbose, "After FlipSegment(%d):", Segment);
852 Base::Dump();
853 }

855 virtual int32 FindIndexForParameter(float Parameter, float& OutLocalParam) const override
856 {
857 const int32 NumSegments = GetNumberOfSegments();
858 const UE::Geometry::FInterval1f ParameterSpace = GetParameterSpace();
859
860 if (ParameterSpace.IsEmpty())
861 {
862 return INDEX_NONE;
863 }
864
865 // Left end: snap onto first segment start
866 if (Parameter <= ParameterSpace.Min)
867 {
868 OutLocalParam = 0.0f;
869 return 0;
870 }
871
872 // Right end: snap onto last segment end
873 if (Parameter >= ParameterSpace.Max)
874 {
875 OutLocalParam = 1.0f;
876 return NumSegments - 1;
877 }
878
879 const float LastKnot = Base::PairKnots.Last().Value;
880
881 // We can now assume Parameter is contained by ParameterSpace, no need for clamping.
882
883 // Closed loop: handle the synthetic closing segment [LastKnot, LastKnot + LoopKnotDelta]
884 if (Parameter >= LastKnot && ensure(HasClosingSegment()))
885 {
886 const float t = (Parameter - LastKnot) / FMath::Max(ParameterSpace.Max - LastKnot, UE_SMALL_NUMBER);
887 OutLocalParam = FMath::Clamp(t, 0.0f, 1.0f);
888 return NumSegments - 1;
889 }
890
891 int32 SegmentIndex = INDEX_NONE;
892
893 // todo: make CVar
894 constexpr bool bEnableSegmentCaching = true;
895 if (bEnableSegmentCaching && LastEvaluatedSegment != INDEX_NONE && LastEvaluatedSegment < Base::PairKnots.Num() - 1)
896 {
897 if (Parameter < Base::PairKnots[LastEvaluatedSegment + 1].Value && Parameter >= Base::PairKnots[LastEvaluatedSegment].Value)
898 {
899 // cache hit!
900 SegmentIndex = LastEvaluatedSegment;
901 }
902 }
903
904 if (SegmentIndex == INDEX_NONE)
905 {
906 // cache miss!
907 int32 SegmentLow = 0;
908 int32 SegmentHigh = Base::PairKnots.Num() - 2;
909 SegmentIndex = SegmentLow + (SegmentHigh - SegmentLow) / 2;
910
911 while (SegmentLow <= SegmentHigh)
912 {
913 int32 Mid = SegmentLow + (SegmentHigh - SegmentLow) / 2;
914 const float& StartParam = Base::PairKnots[Mid].Value;
915 const float& EndParam = Base::PairKnots[Mid + 1].Value;
916
917 if (Parameter >= StartParam && Parameter < EndParam)
918 {
919 SegmentIndex = Mid;
920 break;
921 }
922 else if (Parameter < StartParam)
923 {
924 SegmentHigh = Mid - 1;
925 }
926 else
927 {
928 SegmentLow = Mid + 1;
929 }
930 }
931 }
932
933 // Update the cache.
934 LastEvaluatedSegment = SegmentIndex;
935
936 // Calculate local parameter
937 float SegmentLength = Base::PairKnots[SegmentIndex + 1].Value - Base::PairKnots[SegmentIndex].Value;
938 if (SegmentLength > UE_SMALL_NUMBER)
939 {
940 OutLocalParam = (Parameter - Base::PairKnots[SegmentIndex].Value) / SegmentLength;
941 }
942 else
943 {
944 OutLocalParam = 0.0f;
945 }
946
947 return SegmentIndex;
948 }

1166 int32 InsertAnchor(const FAnchor& InAnchor, float Parameter)
1167 {
1168 if (GetNumAnchorPoints() <= 0)
1169 {
1170 return PrependAnchor(InAnchor, Parameter);
1171 }
1172
1173 const FInterval1f ParameterSpace = GetParameterSpace();
1174
1175 if (!ensureAlwaysMsgf(!ParameterSpace.IsEmpty(), TEXT("Parameter space is empty but we do not have an empty spline!")))
1176 {
1177 return INDEX_NONE;
1178 }
1179 else if (Parameter < ParameterSpace.Min)
1180 {
1181 return PrependAnchor(InAnchor, Parameter);
1182 }
1183 else if (Parameter > ParameterSpace.Max)
1184 {
1185 return AppendAnchor(InAnchor, Parameter);
1186 }
1187
1188 float LocalT;
1189 const int32 SegmentIndex = FindIndexForParameter(Parameter, LocalT);
1190
1191 if (SegmentIndex < 0 || SegmentIndex >= GetNumberOfSegments())
1192 {
1193 // If Parameter is contained by ParameterSpace (confirmed by Prepend/Append special casing above)
1194 // but the parameter maps to an invalid segment, something is wrong that we can't handle nicely.
1195 return INDEX_NONE;
1196 }
1197
1198 // Segment S is bounded by anchors S and S+1, except for the closing segment which is bounded by anchors S and 0.
1199 // We will special case insertion on the closing segment as this is simply an append operation.
1200 if (IsClosingSegment(SegmentIndex))
1201 {
1202 return AppendAnchor(InAnchor, Parameter);
1203 }
1204
1205 // We may now assume anchors SegmentIndex and SegmentIndex + 1 are valid without bounds checking.
1206 // In order to insert onto segment S, we must insert at the index S + 1 so that the new anchor is the new endpoint of segment S.
1207
1208 FInterval1f SegmentRange = GetSegmentParameterRange(SegmentIndex);
1209
1210 if (Parameter == SegmentRange.Min)
1211 {
1212 Parameter = Param::NextDistinct(Parameter);
1213 }
1214 else if (Parameter == SegmentRange.Max)
1215 {
1216 Parameter = Param::PrevDistinct(Parameter);
1217 }
1218
1219 // Bail out if we lost usable range, this should not happen.
1220 if (Parameter <= SegmentRange.Min || Parameter >= SegmentRange.Max)
1221 {
1222 ensureAlways(false);
1223 return INDEX_NONE;
1224 }
1225
1226 const int32 InsertIndex = SegmentIndex + 1;
1227 Base::InsertValue(AnchorBase(InsertIndex) + 0, InAnchor.InHandle);
1228 Base::InsertValue(AnchorBase(InsertIndex) + 1, InAnchor.Anchor);
1229 Base::InsertValue(AnchorBase(InsertIndex) + 2, InAnchor.OutHandle);
1230
1231 // We always use Degree as the multiplicity because this is always an interior knot.
1232 const int32 KnotIndex = Base::InsertKnot(FKnot(Parameter, Degree));
1233 ensureAlways(InsertIndex == KnotIndex);
1234
1235 return InsertIndex;
1236 }

1263 // Split a segment using De Casteljau
1264 bool SplitSegment(int32 SegmentIndex, float SegmentT)
1265 {
1266 if (SegmentIndex < 0 || SegmentIndex >= GetNumberOfSegments())
1267 {
1268 return false;
1269 }
1270
1271 FInterval1f SegmentRange = GetSegmentParameterRange(SegmentIndex);
1272 float Parameter = SegmentRange.Interpolate(SegmentT);
1273
1274 if (Parameter == SegmentRange.Min)
1275 {
1276 Parameter = Param::NextDistinct(Parameter);
1277 }
1278 else if (Parameter == SegmentRange.Max)
1279 {
1280 Parameter = Param::PrevDistinct(Parameter);
1281 }
1282
1283 // Bail out if we lost usable range
1284 if (Parameter <= SegmentRange.Min || Parameter >= SegmentRange.Max)
1285 {
1286 return false;
1287 }
1288
1289 ValueType Position = Base::Evaluate(Parameter);
1290
1291 FWindow Window;
1292 if (!BuildRawWindowForSegment(SegmentIndex, Window))
1293 {
1294 // If we cannot build a window for segment index, something is wrong and we cannot handle it nicely.
1295 return false;
1296 }
1297
1298 ValueType P0 = *Window[0];
1299 ValueType P1 = *Window[1];
1300 ValueType P2 = *Window[2];
1301 ValueType P3 = *Window[3];
1302
1303 // Use De Casteljau algorithm to split the curve
1304 const float t = SegmentT;
1305 const float mt = 1.0f - t;
1306
1307 // Calculate split control points
1308 ValueType Q0 = P0;
1309 ValueType Q1 = P0 * mt + P1 * t;
1310 ValueType Q2 = (P0 * mt + P1 * t) * mt + (P1 * mt + P2 * t) * t;
1311 ValueType Q3 = Position; // Use provided position for the split point
1312
1313 ValueType R0 = Position; // Use provided position for the split point
1314 ValueType R1 = (P1 * mt + P2 * t) * mt + (P2 * mt + P3 * t) * t;
1315 ValueType R2 = P2 * mt + P3 * t;
1316 ValueType R3 = P3;
1317
1318 // Fixup endpoints of segment being split.
1319 TPair<int32, int32> SegmentAnchors = GetSegmentAnchors(SegmentIndex);
1320 SetHandleOutOfAnchor(SegmentAnchors.Key, Q1);
1321 SetHandleInOfAnchor(SegmentAnchors.Value, R2);
1322
1323 // Insert anchor, splitting the segment.
1324 InsertAnchor(FAnchor { Q2, Q3, R1 }, Parameter);
1325
1326 return true;
1327 }

1330 {
1331 const int32 NumAnchors = GetNumAnchorPoints();
1332 const int32 NumSegments = GetNumberOfSegments();
1333

1335 {

1337 }

1353 // start multiplicity = 4, end multiplicity = 4, each interior boundary = 3

1360 // all boundaries behave like "interior" → multiplicity = 3
1361 // There are NumSegments+1 boundaries including the wrap
1362 // Total = 3 * (NumSegments + 1)

1364 }

1511 return;
1512 }
1513
1514 default:
1515 break;
1516 }

1518 // Legacy layout (actual):
1519 // [ P0, C0_0, C1_0, P1,
1520 // P1_Duplicate, C0_1, C1_1, P2,
1521 // P2_Duplicate, C0_2, C1_2, P3,
1522 // ...
1523 // P_{n-1}_Duplicate, C0_{n-1}, C1_{n-1}, Pn ]
1524
1525 // We handle cases 0-3 above
1526 checkf(NumValues >= 4,
1527 TEXT("Legacy PolyBezier must have at least 1 segment (4 values minimum)"));
1528
1529 checkf(NumValues % 4 == 0,
1530 TEXT("Legacy PolyBezier layout mismatch (NumValues = %d)"), NumValues);
1531
1532 const bool bClosed = Base::IsClosedLoop();
1533
1534 const int32 NumSegments = NumValues / 4;
1535
1536 const TArray<ValueType>& Legacy = Base::Values;
1537 // For open: anchors = NumSegments + 1
1538 // For closed: anchors = NumSegments (closing segment is virtual)
1539 const int32 NumAnchors = bClosed ? NumSegments : (NumSegments + 1);
1540
1541 TArray<ValueType> NewValues;
1542 NewValues.SetNumUninitialized(NumAnchors * 3);
1543
1544 // Anchor positions P_j
1545 auto IndexP = [](int32 j) -> int32
1546 {
1547 // P0 is at index 0.
1548 // For j > 0, P_j is the end of the (j-1)-th segment.
1549 return (j == 0) ? 0 : (3 + 4 * (j - 1));
1550 };
1551
1552 // Outgoing handle C0_j for segment j
1553 auto IndexC0 = [](int32 segIdx) -> int32
1554 {
1555 // Segment segIdx starts at 4*segIdx:
1556 // [ P_j_or_dup, C0_j, C1_j, P_{j+1} ]
1557 return 1 + 4 * segIdx;
1558 };
1559
1560 // Incoming handle C1_j for segment j
1561 auto IndexC1 = [](int32 segIdx) -> int32
1562 {
1563 return 2 + 4 * segIdx;
1564 };
1565
1566 for (int32 AnchorIdx = 0; AnchorIdx < NumAnchors; ++AnchorIdx)
1567 {
1568 ValueType In(ForceInitToZero);
1569 ValueType A(ForceInitToZero);
1570 ValueType Out(ForceInitToZero);
1571
1572 // A_j = P_j
1573 const int32 PIndex = IndexP(AnchorIdx);
1574 check(Legacy.IsValidIndex(PIndex));
1575 A = Legacy[PIndex];
1576
1577 // In_j
1578 if (!bClosed)
1579 {
1580 if (AnchorIdx == 0)
1581 {
1582 // No incoming segment, start flat.
1583 In = A;
1584 }
1585 else
1586 {
1587 // From previous segment's C1
1588 const int32 PrevSeg = AnchorIdx - 1;
1589 const int32 C1Index = IndexC1(PrevSeg);
1590 check(Legacy.IsValidIndex(C1Index));
1591 In = Legacy[C1Index];
1592 }
1593 }
1594 else
1595 {
1596 // Closed: every anchor has an incoming segment.
1597 const int32 PrevSeg = (AnchorIdx == 0) ? (NumSegments - 1) : (AnchorIdx - 1);
1598 const int32 C1Index = IndexC1(PrevSeg);
1599 check(Legacy.IsValidIndex(C1Index));
1600 In = Legacy[C1Index];
1601 }
1602
1603 // Out_j
1604 if (!bClosed)
1605 {
1606 if (AnchorIdx < NumSegments)
1607 {
1608 const int32 C0Index = IndexC0(AnchorIdx);
1609 check(Legacy.IsValidIndex(C0Index));
1610 Out = Legacy[C0Index];
1611 }
1612 else
1613 {
1614 // Last anchor has no outgoing segment in open curves.
1615 Out = A;
1616 }
1617 }
1618 else
1619 {
1620 // Closed: every anchor has outgoing segment j
1621 const int32 Seg = AnchorIdx;
1622 const int32 C0Index = IndexC0(Seg);
1623 check(Legacy.IsValidIndex(C0Index));
1624 Out = Legacy[C0Index];
1625 }
1626
1627 const int32 NewBase = AnchorIdx * 3;
1628 NewValues[NewBase + 0] = In;
1629 NewValues[NewBase + 1] = A;
1630 NewValues[NewBase + 2] = Out;
1631 }
1632
1633 Base::Values = MoveTemp(NewValues);
1634
1635 // Fix up knot representation for closed loops.
1636 // Legacy data uses an explicit closing knot in PairKnots; the new
1637 // representation uses LoopKnotDelta instead.
1638 if (bClosed && Base::PairKnots.Num() > 2)
1639 {
1640 const int32 LastIdx = Base::PairKnots.Num() - 1;
1641 const float ClosingKnot = Base::PairKnots[LastIdx].Value;
1642 const float PrevKnot = Base::PairKnots[LastIdx - 1].Value;
1643
1644 // Store the closing interval length and drop the closing knot.
1645 LoopKnotDelta = ClosingKnot - PrevKnot;
1646
1647 Base::PairKnots.RemoveAt(LastIdx, 1, EAllowShrinking::Yes);
1648 }
1649
1650 // Values changed (and potentially PairKnots layout), so flat-knot cache is dirty.
1651 Base::MarkFlatKnotsCacheDirty();
1652 }

1659 void GenerateDistanceKnotsForBezier(EParameterizationPolicy Mode)
1660 {
1661 TRACE_CPUPROFILER_EVENT_SCOPE(GenerateDistanceKnotsForBezier);
1662
1663 // 3-tuple layout:
1664 // Values[3*i + 0] = In_i
1665 // Values[3*i + 1] = A_i
1666 // Values[3*i + 2] = Out_i
1667
1668 const int32 NumAnchors = GetNumAnchorPoints();
1669
1670 if (NumAnchors <= 0)
1671 {
1672 Base::PairKnots.Reset();
1673 Base::MarkFlatKnotsCacheDirty();
1674 return;
1675 }
1676 else if (NumAnchors == 1)
1677 {
1678 // If we have a single anchor, our number of expected knots is itself the multiplicity of a single distinct knot.
1679 Base::PairKnots = { FKnot(0.f, GetExpectedNumKnots()) };
1680 Base::MarkFlatKnotsCacheDirty();
1681 return;
1682 }
1683
1684 // If we have > 1 anchors (guaranteed by the above branches), we ought to have at least 1 segment.
1685 const int32 NumSegments = GetNumberOfSegments();
1686 ensureAlways(NumSegments > 0);
1687
1688 TArray<float> SegmentLengths;
1689 SegmentLengths.Reserve(NumSegments);
1690
1691 // For a Bezier spline, the control points don't all lie on the curve
1692 // So we need to compute chord lengths between segment endpoints only.
1693 // Segment i goes from A_i to A_{i+1} (open) or A_{(i+1)%N} (closed).
1694 const bool bClosed = HasClosingSegment();
1695
1696 for (int32 i = 0; i < NumSegments; ++i)
1697 {
1698 const int32 StartAnchorIdx = i;
1699 const int32 EndAnchorIdx = bClosed
1700 ? (i + 1) % NumAnchors
1701 : (i + 1); // for open, NumSegments == NumAnchors - 1, so this is always valid
1702
1703 const ValueType& P0 = Base::Values[AnchorBase(StartAnchorIdx) + 1]; // A_i
1704 const ValueType& P3 = Base::Values[AnchorBase(EndAnchorIdx) + 1]; // A_{i+1} or A_0 (closed)
1705
1706 SegmentLengths.Add(ComputeSegmentLength(P0, P3, Mode));
1707 }
1708
1709 // Now create a knot vector that respects:
1710 // 1. Bezier segment structure (knot multiplicity)
1711 // 2. Proportional chord/centripetal lengths
1712
1713 // This will only return Val if Offset is exactly 0.f. Otherwise, it is guaranteed to change in the direction of Offset.
1714 // We sacrifice a bit of accuracy for the assumption that Val will actually change.
1715 auto SafeAdd = [](float Val, float Offset) -> float
1716 {
1717 if (Offset == 0.f) return Val;
1718
1719 const float Result = Val + Offset;
1720 if (Result != Val) return Result;
1721
1722 // Guaranteed one-step progress even under FTZ/DAZ:
1723 return UE::Geometry::Spline::Param::Step(
1724 Val, Offset > 0.f ? UE::Geometry::Spline::Param::EDir::Right
1725 : UE::Geometry::Spline::Param::EDir::Left);
1726 };
1727
1728 // Reset pair knots
1729 Base::PairKnots.Reset();
1730
1731 FValidKnotSearchParams SearchParams;
1732 SearchParams.bSearchLeft = false; // we are appending repeatedly, we want insertion order to be predictable (always go after conflicting knots).
1733
1734 FKnotSearchCache SearchCache{ Base::PairKnots };
1735
1736 const auto InsertKnotAndUpdateCache = [this, &SearchCache, &SearchParams](uint32 InMultiplicity)
1737 {
1738 const float KnotValue = Base::GetNearestAvailableKnotValue(SearchParams, SearchCache);
1739 SearchCache.KnotValuesSet.Add(Param::NormalizeKey(KnotValue));
1740 Base::InsertKnot(FKnot(KnotValue, InMultiplicity));
1741 };
1742
1743 constexpr int32 EndMult = Degree + 1;
1744 constexpr int32 InteriorMult = Degree;
1745 if (!bClosed)
1746 {
1747 // OPEN: start mult=4, interior mult=3, end mult=4
1748 // Total knots = 4 + (NumSegments - 1) * 3 + 4
1749
1750 // Add start knot with multiplicity EndMult = 4
1751 SearchParams.DesiredParameter = 0.f;
1752 InsertKnotAndUpdateCache(EndMult);
1753
1754 // Calculate accumulated distances for internal knots
1755 float AccumLength = 0.0f;
1756 for (int32 i = 0; i < NumSegments - 1; ++i)
1757 {
1758 if (i < SegmentLengths.Num())
1759 {
1760 AccumLength = SafeAdd(AccumLength, SegmentLengths[i]);
1761
1762 // Use accumulated length as the desired knot value
1763 SearchParams.DesiredParameter = AccumLength;
1764
1765 // Interior knots at segment boundaries with multiplicity 3
1766 InsertKnotAndUpdateCache(InteriorMult);
1767 }
1768 }
1769
1770 // Final end knot with multiplicity 4
1771 SearchParams.DesiredParameter = SafeAdd(
1772 AccumLength,
1773 SegmentLengths.Num() > 0 ? SegmentLengths.Last() : 0.0f);
1774
1775 InsertKnotAndUpdateCache(EndMult);
1776 }
1777 else
1778 {
1779 // CLOSED:
1780 // All boundaries behave like interior ones, multiplicity = InteriorMult = 3.
1781 // We create one boundary per segment (at its start) and store the
1782 // total loop length in LoopKnotDelta instead of inserting a
1783 // separate closing knot.
1784
1785 float AccumLength = 0.0f;
1786
1787 // Boundaries 0 .. NumSegments-1 at starts of each segment
1788 for (int32 BoundaryIdx = 0; BoundaryIdx < NumSegments; ++BoundaryIdx)
1789 {
1790 SearchParams.DesiredParameter = AccumLength;
1791 InsertKnotAndUpdateCache(InteriorMult);
1792
1793 if (BoundaryIdx < SegmentLengths.Num())
1794 {
1795 // If this is the closing segment (last one), remember its param length
1796 if (BoundaryIdx == NumSegments - 1)
1797 {
1798 LoopKnotDelta = SegmentLengths[BoundaryIdx];
1799 }
1800
1801 AccumLength = SafeAdd(AccumLength, SegmentLengths[BoundaryIdx]);
1802 }
1803 }
1804 if (NumSegments == 0 || SegmentLengths.Num() == 0)
1805 {
1806 LoopKnotDelta = 0.0f;
1807 }
1808 }
1809 // --------------------------------------------------------------------
1810
1811 // Mark flat knots cache as dirty
1812 Base::MarkFlatKnotsCacheDirty();
1813 }
