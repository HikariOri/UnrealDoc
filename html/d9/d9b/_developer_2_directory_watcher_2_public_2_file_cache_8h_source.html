100 friend FArchive& operator<<(FArchive& Ar, FDirectoryState& State)
101 {
102 Ar.UsingCustomVersion(FFileCacheCustomVersion::Key);
103
104 // Ignore any old versions to data to ensure that we generate a new cache
105 if (Ar.CustomVer(FFileCacheCustomVersion::Key) >= FFileCacheCustomVersion::IncludeFileHash)
