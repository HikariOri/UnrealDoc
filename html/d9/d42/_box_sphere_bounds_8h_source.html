425template<typename T, typename TExtent>
426inline TBoxSphereBounds<T, TExtent>::TBoxSphereBounds( const TVector<T>* Points, uint32 NumPoints )
427{
428 TBox<T> BoundingBox(ForceInit);
429
430 // find an axis aligned bounding box for the points.
431 for (uint32 PointIndex = 0; PointIndex < NumPoints; PointIndex++)
432 {
433 BoundingBox += Points[PointIndex];
434 }
435
436 TVector<T> LocalExtent;
437 BoundingBox.GetCenterAndExtents(Origin, LocalExtent);
438 BoxExtent = TVector<TExtent>(LocalExtent);
439
440 // using the center of the bounding box as the origin of the sphere, find the radius of the bounding sphere.
441 TExtent SquaredSphereRadius = 0;
442
443 for (uint32 PointIndex = 0; PointIndex < NumPoints; PointIndex++)
444 {
445 SquaredSphereRadius = FMath::Max<TExtent>(SquaredSphereRadius, (Points[PointIndex] - Origin).SizeSquared()); // LWC_TODO: Precision loss
446 }
447
448 SphereRadius = FMath::Sqrt(SquaredSphereRadius);
449
450 DiagnosticCheckNaN();
451}

453template<typename T, typename TExtent>
454inline TBoxSphereBounds<T, TExtent> TBoxSphereBounds<T, TExtent>::operator+( const TBoxSphereBounds<T, TExtent>& Other ) const
455{
456 TBox<T> BoundingBox(ForceInit);
457
458 BoundingBox += (this->Origin - this->BoxExtent);
459 BoundingBox += (this->Origin + this->BoxExtent);
460 BoundingBox += (Other.Origin - Other.BoxExtent);
461 BoundingBox += (Other.Origin + Other.BoxExtent);
462
463 // build a bounding sphere from the bounding box's origin and the radii of A and B.
464 TBoxSphereBounds<T, TExtent> Result(BoundingBox);
465
466 Result.SphereRadius = FMath::Min<TExtent>(Result.SphereRadius, FMath::Max<TExtent>((Origin - Result.Origin).Size() + SphereRadius, (Other.Origin - Result.Origin).Size() + Other.SphereRadius));
467 Result.DiagnosticCheckNaN();
468
469 return Result;
470}

497template<typename T, typename TExtent>
498TBoxSphereBounds<T, TExtent> TBoxSphereBounds<T, TExtent>::TransformBy(const TMatrix<T>& M) const
499{
500#if ENABLE_NAN_DIAGNOSTIC
501 if (M.ContainsNaN())
502 {
503 logOrEnsureNanError(TEXT("Input Matrix contains NaN/Inf! %s"), *M.ToString());
504 }
505#endif
506
507 TBoxSphereBounds<T> Result;
508
509 const TVectorRegisterType<T> VecOrigin = VectorLoadFloat3(&Origin);
510 const TVectorRegisterType<T> VecExtent = VectorLoadFloat3(&BoxExtent);
511
512 const TVectorRegisterType<T> m0 = VectorLoadAligned(M.M[0]);
513 const TVectorRegisterType<T> m1 = VectorLoadAligned(M.M[1]);
514 const TVectorRegisterType<T> m2 = VectorLoadAligned(M.M[2]);
515 const TVectorRegisterType<T> m3 = VectorLoadAligned(M.M[3]);
516
517 TVectorRegisterType<T> NewOrigin = VectorMultiply(VectorReplicate(VecOrigin, 0), m0);
518 NewOrigin = VectorMultiplyAdd(VectorReplicate(VecOrigin, 1), m1, NewOrigin);
519 NewOrigin = VectorMultiplyAdd(VectorReplicate(VecOrigin, 2), m2, NewOrigin);
520 NewOrigin = VectorAdd(NewOrigin, m3);
521
522 TVectorRegisterType<T> NewExtent = VectorAbs(VectorMultiply(VectorReplicate(VecExtent, 0), m0));
523 NewExtent = VectorAdd(NewExtent, VectorAbs(VectorMultiply(VectorReplicate(VecExtent, 1), m1)));
524 NewExtent = VectorAdd(NewExtent, VectorAbs(VectorMultiply(VectorReplicate(VecExtent, 2), m2)));
525
526 VectorStoreFloat3(NewExtent, &(Result.BoxExtent.X));
527 VectorStoreFloat3(NewOrigin, &(Result.Origin.X));
528
529 TVectorRegisterType<T> MaxRadius = VectorMultiply(m0, m0);
530 MaxRadius = VectorMultiplyAdd(m1, m1, MaxRadius);
531 MaxRadius = VectorMultiplyAdd(m2, m2, MaxRadius);
532 MaxRadius = VectorMax(VectorMax(MaxRadius, VectorReplicate(MaxRadius, 1)), VectorReplicate(MaxRadius, 2));
533 Result.SphereRadius = FMath::Sqrt(VectorGetComponent(MaxRadius, 0)) * SphereRadius;
534
535 // For non-uniform scaling, computing sphere radius from a box results in a smaller sphere.
536 T const BoxExtentMagnitude = FMath::Sqrt(VectorDot3Scalar(NewExtent, NewExtent));
537 Result.SphereRadius = FMath::Min(Result.SphereRadius, BoxExtentMagnitude);
538
539 Result.DiagnosticCheckNaN();
540 return TBoxSphereBounds<T, TExtent>(Result);
541}

550TBoxSphereBounds<T, TExtent> TBoxSphereBounds<T, TExtent>::TransformBy(const TTransform<T>& M) const
