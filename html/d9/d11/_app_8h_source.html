138 static FString GetName()
139 {
140 FString ExecutableName = FPlatformProcess::ExecutableName();
141
142 int32 ChopIndex = ExecutableName.Find(TEXT("-"));
143
144 if (ExecutableName.FindChar(TCHAR('-'), ChopIndex))
145 {
146 return ExecutableName.Left(ChopIndex);
147 }
148
149 if (ExecutableName.FindChar(TCHAR('.'), ChopIndex))
150 {
151 return ExecutableName.Left(ChopIndex);
152 }
153
154 return ExecutableName;
155 }
