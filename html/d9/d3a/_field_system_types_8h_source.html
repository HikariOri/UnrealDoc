237TArray<EFieldPhysicsType> GetFieldTargetTypes(EFieldOutputType OutputType)
238{
239 TArray<EFieldPhysicsType> PhysicsTypes;
240 switch (OutputType)
241 {
242 case Field_Output_Vector:

244 PhysicsTypes = { EFieldPhysicsType::Field_LinearForce,
245 EFieldPhysicsType::Field_LinearVelocity,
246 EFieldPhysicsType::Field_AngularVelociy,

249 EFieldPhysicsType::Field_InitialLinearVelocity,
250 EFieldPhysicsType::Field_InitialAngularVelocity,
251 EFieldPhysicsType::Field_LinearImpulse };
252 break;
253 }
254 case Field_Output_Scalar:
255 {
256 PhysicsTypes = { EFieldPhysicsType::Field_ExternalClusterStrain,
257 EFieldPhysicsType::Field_Kill,
258 EFieldPhysicsType::Field_DisableThreshold,
259 EFieldPhysicsType::Field_SleepingThreshold,
260 EFieldPhysicsType::Field_InternalClusterStrain,
261 EFieldPhysicsType::Field_DynamicConstraint };
262 break;
263 }
264 case Field_Output_Integer:
265 {
266 PhysicsTypes = { EFieldPhysicsType::Field_DynamicState,
267 EFieldPhysicsType::Field_ActivateDisabled,
268 EFieldPhysicsType::Field_CollisionGroup,
269 EFieldPhysicsType::Field_PositionAnimated,
270 EFieldPhysicsType::Field_PositionStatic };
271 break;
272 }
273 default:
274 break;
275 }
276 return MoveTemp(PhysicsTypes);
277}

279inline
280EFieldOutputType GetFieldTargetIndex(const TArray<EFieldPhysicsType>& VectorTypes,
281 const TArray<EFieldPhysicsType>& ScalarTypes,
282 const TArray<EFieldPhysicsType>& IntegerTypes,
283 const EFieldPhysicsType FieldTarget, int32& TargetIndex)
284{
285 EFieldOutputType OutputType = EFieldOutputType::Field_Output_Max;
286
287 TargetIndex = VectorTypes.Find(FieldTarget);
288 if (TargetIndex == INDEX_NONE)
289 {
290 TargetIndex = ScalarTypes.Find(FieldTarget);
291 if (TargetIndex == INDEX_NONE)
292 {
293 TargetIndex = IntegerTypes.Find(FieldTarget);
294 if (TargetIndex != INDEX_NONE)
295 {
296 OutputType = EFieldOutputType::Field_Output_Integer;
297 }
298 }
299 else
300 {
301 OutputType = EFieldOutputType::Field_Output_Scalar;
302 }
303 }
304 else
305 {
306 OutputType = EFieldOutputType::Field_Output_Vector;
307 }
308 return OutputType;
309}
