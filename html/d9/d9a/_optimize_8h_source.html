133{
134 // Simplex
135 FVector2f sX[3];
136 float sY[3];
137
138 for( int32 i = 0; i < MaxIter; i++ )
139 {
140 // Sort
141 if( sY[1] < sY[2] )
142 {
143 Swap( sX[1], sX[2] );
144 Swap( sY[1], sY[2] );
145 }
146 if( sY[0] < sY[1] )
147 {
148 Swap( sX[0], sX[1] );
149 Swap( sY[0], sY[1] );
150 }
151
152 if( 2.0f * FMath::Abs( sY[0] - sY[2] ) < ( FMath::Abs( sY[0] ) + FMath::Abs( sY[2] ) ) * Tolerance )
153 break;
154
155 FVector2f ReflectedX = sX[0] + sX[1] - sX[2];
156 float ReflectedY = Func( ReflectedX );
157
158 if( ReflectedY > sY[0] )
159 {
160 FVector2f ExpandedX = 2.0f * ReflectedX - 0.5f * ( sX[0] + sX[1] );
161 float ExpandedY = Func( ExpandedX );
162
163 if( ExpandedY > ReflectedY )
164 {
165 sX[2] = ExpandedX;
166 sY[2] = ExpandedY;
167 }
168 else
169 {
170 sX[2] = ReflectedX;
171 sY[2] = ReflectedY;
172 }
173 }
174 else if( ReflectedY > sY[1] )
175 {
176 sX[2] = ReflectedX;
177 sY[2] = ReflectedY;
178 }
179 else
180 {
181 FVector2f ContractedX = 0.25f * ( sX[0] + sX[1] ) + 0.5f * sX[2];

193 sX[2] = 0.5f * ( sX[0] + sX[2] );
