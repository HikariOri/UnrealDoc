230template<typename T>
231void CalculateFrameTM(
232 bool& bNeedToUpdateFrameTM, FTransform& FrameTM,
233 const int32 SampleKey, const T& AnalysisProperties, const UAnimSequence& Animation)
234{
235 if (bNeedToUpdateFrameTM)
236 {
237 FrameTM.SetIdentity();
238 if (AnalysisProperties->Space != EAnalysisSpace::World)
239 {
240 FTransform SpaceBoneOffset;
241 FName SpaceBoneName;
242 if (GetBoneInfo(Animation, AnalysisProperties->SpaceBoneSocket, SpaceBoneOffset, SpaceBoneName))
243 {
244 FTransform SpaceBoneTM = GetBoneTransform(Animation, SampleKey, SpaceBoneName);
245 FrameTM = SpaceBoneOffset * SpaceBoneTM;
246 }
247 }
248
249 bNeedToUpdateFrameTM = (
250 AnalysisProperties->Space == EAnalysisSpace::Changing ||
251 AnalysisProperties->Space == EAnalysisSpace::Moving);
252 }
253}

259template<typename T>
260void GetFrameDirs(
261 FVector& FrameFacingDir, FVector& FrameUpDir, FVector& FrameRightDir,
262 const FTransform& FrameTM, const T& AnalysisProperties)
263{
264 FrameFacingDir = GetAxisFromTM(FrameTM, AnalysisProperties->CharacterFacingAxis);
265 FrameUpDir = GetAxisFromTM(FrameTM, AnalysisProperties->CharacterUpAxis);
266 FrameRightDir = FVector::CrossProduct(FrameUpDir, FrameFacingDir);
267}

273template<typename FunctionType>
274static bool CalculateComponentSampleValue(
275 double& Result,
276 const FunctionType& Fn,
277 const UBlendSpace& BlendSpace,
278 const ULinearAnalysisProperties* AnalysisProperties,
279 const UAnimSequence& Animation,
280 const float RateScale)
281{
282 FVector Value = FVector::ZeroVector;
283 int32 ComponentIndex = (int32)AnalysisProperties->FunctionAxis;
284 if (Fn(Value, BlendSpace, AnalysisProperties, Animation, RateScale))
285 {
286 Result = Value | BlendSpaceAnalysis::GetAxisFromTM(FTransform::Identity, AnalysisProperties->FunctionAxis);
287 return true;
288 }
289 return false;
290}

296template<typename FunctionType>
297static bool CalculateComponentSampleValue(
298 double& Result,
299 const FunctionType& Fn,
300 const UBlendSpace& BlendSpace,
301 const UEulerAnalysisProperties* AnalysisProperties,
302 const UAnimSequence& Animation,
303 const float RateScale)
304{
305 FVector Value;
306 int32 ComponentIndex = (int32)AnalysisProperties->FunctionAxis; // Roll, Pitch, Yaw -> 0, 1, 2
307 if (Fn(Value, BlendSpace, AnalysisProperties, Animation, RateScale))
308 {
309 Result = Value[ComponentIndex];
310 return true;
311 }
312 return false;
