192 virtual bool DoPayloadsExist(TArrayView<const FIoHash> PayloadIds, TArray<bool>& OutResults)
193 {
194 // This is the default implementation that just calls ::DoesExist on each FIoHash in the
195 // array, one at a time.
196 // Backends may override this with their own implementations if it can be done with less
197 // overhead by performing the check on the entire batch instead.
198
199 OutResults.SetNum(PayloadIds.Num());
200
201 for (int32 Index = 0; Index < PayloadIds.Num(); ++Index)
202 {
203 OutResults[Index] = DoesPayloadExist(PayloadIds[Index]);
204 }
205
206 return true;
207 }
