136 {
137 FSequencerFilterData& FilterData = GetFilterInterface().GetFilterData();
138
139 UObject* const BoundObject = FilterData.ResolveTrackBoundObject(GetSequencer(), InItem);
140 if (!BoundObject)
141 {
142 return false;
143 }
144
145 if (BoundObject->IsA(InComponentType::StaticClass()))
146 {
147 return true;
148 }
149
150 if (BoundObject->IsA(AActor::StaticClass()))
151 {
152 const AActor* const Actor = CastChecked<const AActor>(BoundObject);
153 if (Actor->FindComponentByClass(InComponentType::StaticClass()))
154 {
155 return true;
156 }
157 }
158
159 return false;
160 }
