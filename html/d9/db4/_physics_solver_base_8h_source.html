411 template <typename TSimCallbackObjectType, typename... ConstructorArgTypes>
412 inline TSimCallbackObjectType* CreateAndRegisterSimCallbackObject_External(ConstructorArgTypes&&... ConstructorArgs)
413 {
414 auto NewCallbackObject = new TSimCallbackObjectType(Forward<ConstructorArgTypes>(ConstructorArgs)...);
415 //If at least one callback is on frozen GT we mark it as such
416 //Note this is never cleaned up (to avoid race conditions of register/unregister).
417 //Expectation is that there's typically just one of these on frozen gt that triggers all the fixed tick on gt (maybe two if another system wants direct control)
418 //This means we don't expect to optimize for the case where a gt callback goes away half way through sim
419 if (NewCallbackObject->HasOption(ESimCallbackOptions::RunOnFrozenGameThread))
420 {
421 bSolverHasFrozenGameThreadCallbacks = true;
422 }
423
424 RegisterSimCallbackObject_External(NewCallbackObject);
425 NewCallbackObject->PostInitialize_External();
426 return NewCallbackObject;
427 }

559 //Used as helper for GT to go from unique idx back to gt particle
560 //If GT deletes a particle, this function will return null (that's a good thing when consuming async outputs as GT may have already deleted the particle we care about)
561 //Note: if the physics solver has been advanced after the particle was freed on GT, the index may have been freed and reused.
562 //In this case instead of getting a nullptr you will get an unrelated (wrong) GT particle
563 //Because of this we keep the index alive for as long as the async callback can lag behind. This way as long as you immediately consume the output, you will always be sure the unique index was not released.
564 //Practically the flow should always be like this:
565 //advance the solver and trigger callbacks. Callbacks write to outputs. Consume the outputs on GT and use this function _before_ advancing solver again
566 FGeometryParticle* UniqueIdxToGTParticle_External(const FUniqueIdx& UniqueIdx) const
567 {
568 FGeometryParticle* Result = nullptr;
569 if (ensure(UniqueIdx.Idx < UniqueIdxToGTParticles.Num())) //asking for particle on index that has never been allocated
570 {
571 Result = UniqueIdxToGTParticles[UniqueIdx.Idx];
572 }
573
574 return Result;
575 }
