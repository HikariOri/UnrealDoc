24template< typename T >
25inline T DifferenceOfProducts( T a, T b, T c, T d )
26{
27 T cd = c * d;
28 T Error = std::fma( -c, d, cd );
29 T ab_cd = std::fma( a, b, -cd );
30 return ab_cd + Error;
31}

33// Knuth 1974
34template< typename T >
35T TwoSumError( T a, T b )
36{
37 T x = a + b;
38 T z = x - b;
39 return ( a - (x - z) ) + (b - z);
40}

42template< typename T >
43T TwoProdError( T a, T b )
44{
45 T x = a * b;
46 return std::fma( a, b, -x );
47}

109#if USE_FMA
110 /*
111 T zz = z * v.z;
112 T Error = std::fma( z, v.z, -zz );
113 T yy_zz = std::fma( y, v.y, zz );
114 T xx_yy_zz = std::fma( x, v.z, yy_zz );
115 return xx_yy_zz + Error;
116 */
117
118 T xx = x * v.x;
119 T yy = y * v.y;
120 T zz = z * v.z;
121
122 T Dot, DotError;
123 DotError = TwoProdError( x, v.x );
124 Dot = xx;
125 DotError += TwoProdError( y, v.y ) + TwoSumError( Dot, yy );
126 Dot += yy;
127 DotError += TwoProdError( z, v.z ) + TwoSumError( Dot, zz );
128 Dot += zz;
129
130 return Dot + DotError;
131#else
132 return x * v.x + y * v.y + z * v.z;
133#endif
134 }

139#if USE_FMA

142 Result.z = DifferenceOfProducts( x, v.y, y, v.x );
143#else
144 Result.x = y * v.z - z * v.y;
145 Result.y = z * v.x - x * v.z;
146 Result.z = x * v.y - y * v.x;
147#endif
148

150 }

185inline FEdgeQuadric::FEdgeQuadric( const QVec3 p0, const QVec3 p1, const QVec3 FaceNormal, const float Weight )
186{
187 const QVec3 p01 = p1 - p0;
188
189 n = p01 ^ FaceNormal;
190
191 const QScalar Length = sqrt( n | n );
192 if( Length < (QScalar)SMALL_NUMBER )
193 {
194 Zero();
195 return;
196 }
197 else
198 {
199 n.x /= Length;
200 n.y /= Length;
201 n.z /= Length;
202 }
203
204 a = Weight * sqrt( p01 | p01 );
205
206 nxx = a * n.x * n.x;
207 nyy = a * n.y * n.y;
208 nzz = a * n.z * n.z;
209
210 nxy = a * n.x * n.y;
211 nxz = a * n.x * n.z;
212 nyz = a * n.y * n.z;
213}

300inline void FQuadric::Add( const FEdgeQuadric& RESTRICT EdgeQuadric, const FVector3f& Point )
301{
302 const QVec3 p0( Point );
303
304 const QScalar Dist = -( EdgeQuadric.n | p0 );
305
306 nxx += EdgeQuadric.nxx;
307 nyy += EdgeQuadric.nyy;
308 nzz += EdgeQuadric.nzz;
309
310 nxy += EdgeQuadric.nxy;
311 nxz += EdgeQuadric.nxz;
312 nyz += EdgeQuadric.nyz;
313
314 QScalar aDist = EdgeQuadric.a * Dist;
315 //dn += aDist * EdgeQuadric.n;
316 //d2 += aDist * Dist;
317
318 dn += EdgeQuadric.a * -p0 - aDist * EdgeQuadric.n;
319 d2 += EdgeQuadric.a * (p0 | p0) - aDist * Dist;
320}
