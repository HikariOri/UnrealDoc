49struct FShaderCodeLibraryPipeline
50{
51 FShaderHash Shaders[SF_NumGraphicsFrequencies];
52 mutable uint32 Hash;
53
55 RENDERCORE_API void Initialize(const FShaderPipeline* Pipeline);
56
57 FShaderCodeLibraryPipeline() : Hash(0) {}
58
59 friend bool operator ==(const FShaderCodeLibraryPipeline& A,const FShaderCodeLibraryPipeline& B)
60 {
61 for (uint32 Frequency = 0u; Frequency < SF_NumGraphicsFrequencies; ++Frequency)
62 {
63 if (A.Shaders[Frequency] != B.Shaders[Frequency])
64 {
65 return false;
66 }
67 }
68 return true;
69 }
70
71 friend uint32 GetTypeHash(const FShaderCodeLibraryPipeline &Key)
72 {
73 if(!Key.Hash)
74 {
75 for (uint32 Frequency = 0u; Frequency < SF_NumGraphicsFrequencies; ++Frequency)
76 {
77 Key.Hash = HashCombine(Key.Hash, GetTypeHash(Key.Shaders[Frequency]));
78 }
79 }
80 return Key.Hash;
81 }
82
84 RENDERCORE_API void GetPipelineHash(FShaderHash& Output);
85
86 friend FArchive& operator<<( FArchive& Ar, FShaderCodeLibraryPipeline& Info )
87 {
88 for (uint32 Frequency = 0u; Frequency < SF_NumGraphicsFrequencies; ++Frequency)
89 {
90 Ar << Info.Shaders[Frequency];
91 }
92 return Ar << Info.Hash;
93 }
94};
