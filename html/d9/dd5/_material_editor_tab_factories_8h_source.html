44 virtual TAttribute<FText> ConstructTabNameForObject(UEdGraph* DocumentID) const override
45 {
46 auto GetGraphDisplayName = [](UEdGraph* Graph)
47 {
48 if (Graph)
49 {
50 if (const UEdGraphSchema* Schema = Graph->GetSchema())
51 {
52 FGraphDisplayInfo Info;
53 Schema->GetGraphDisplayInformation(*Graph, /*out*/ Info);
54
55 return Info.DisplayName;
56 }
57 else
58 {
59 // if we don't have a schema, we're dealing with a malformed (or incomplete graph)...
60 // possibly in the midst of some transaction - here we return the object's outer path
61 // so we can at least get some context as to which graph we're referring
62 return FText::FromString(Graph->GetPathName());
63 }
64 }
65
66 return LOCTEXT("UnknownGraphName", "UNKNOWN");
67 };
68
69 return TAttribute<FText>::Create(TAttribute<FText>::FGetter::CreateLambda(GetGraphDisplayName, DocumentID));
70 }
