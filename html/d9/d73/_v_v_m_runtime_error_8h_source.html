99#define RAISE_VERSE_RUNTIME_ERROR_FORMAT(Diagnostic, FormatString, ...) \
100 do \
101 { \
102 const FText MessageAsText(FText::FromString(FString::Printf(FormatString, ##__VA_ARGS__))); \
103 ::Verse::RaiseVerseRuntimeError(Diagnostic, MessageAsText); \

107#define RETURN_RAISE_VERSE_RUNTIME_ERROR_UNLESS(Cond, ErrCode, ErrStr, ...) \
108 do \
109 { \
110 if (!(Cond)) \
111 { \
112 RAISE_VERSE_RUNTIME_ERROR_FORMAT(::Verse::ERuntimeDiagnostic::ErrCode, ErrStr, ##__VA_ARGS__); \
113 return; \
114 } \
