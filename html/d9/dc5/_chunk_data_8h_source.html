173 // Some helper constants for dealing with chunks that are full of one single byte, usually padding.
174 namespace PaddingChunk
175 {
176 // The A, B, and C components of a chunk Guid indicating that this is a padding chunk. D would be the actual byte padded with.
177 static const int32 ChunkIdA = 0x00000001;
178 static const int32 ChunkIdB = 0x00000000;
179 static const int32 ChunkIdC = 0x00000000;
180 // The size of the chunk we use to save out, which would allow a legacy client to actually use one.
181 static const uint32 ChunkSize = LegacyFixedChunkWindow;
186 FORCEINLINE bool IsPadding(const FGuid& Guid)
187 {
188 return Guid.A == ChunkIdA && Guid.B == ChunkIdB && Guid.C == ChunkIdC && Guid.D >= 0 && Guid.D <= 255;
189 }
190
194 FORCEINLINE uint8 GetPaddingByte(const FGuid& Guid)
195 {
196 check(IsPadding(Guid));
197 return Guid.D;
198 }
199
203 FORCEINLINE FGuid MakePaddingGuid(uint8 Byte)
204 {
205 return FGuid(ChunkIdA, ChunkIdB, ChunkIdC, Byte);
206 }
207 }
