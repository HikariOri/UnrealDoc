58template<class T, unsigned int MaxTotalItems, unsigned int ItemsPerPage>
59class TLockFreeAllocOnceIndexedAllocator
60{
61 enum
62 {
63 MaxBlocks = (MaxTotalItems + ItemsPerPage - 1) / ItemsPerPage
64 };
65public:
66
67 [[nodiscard]] TLockFreeAllocOnceIndexedAllocator()
68 {
69 NextIndex.Increment(); // skip the null ptr
70 for (uint32 Index = 0; Index < MaxBlocks; Index++)
71 {
72 Pages[Index] = nullptr;
73 }
74 }
75
76 inline uint32 Alloc(uint32 Count = 1)
77 {
78 uint32 FirstItem = NextIndex.Add(Count);
79 if (FirstItem + Count > MaxTotalItems)
80 {
81 LockFreeLinksExhausted(MaxTotalItems);
82 }
83 for (uint32 CurrentItem = FirstItem; CurrentItem < FirstItem + Count; CurrentItem++)
84 {
85 ::new (GetRawItem(CurrentItem)) T();
86 }
87 return FirstItem;
88 }
89 [[nodiscard]] inline T* GetItem(uint32 Index)
90 {
91 if (!Index)
92 {
93 return nullptr;
94 }
95 uint32 BlockIndex = Index / ItemsPerPage;
96 uint32 SubIndex = Index % ItemsPerPage;
97 checkLockFreePointerList(Index < (uint32)NextIndex.GetValue() && Index < MaxTotalItems && BlockIndex < MaxBlocks && Pages[BlockIndex].load(std::memory_order_relaxed));
98 return Pages[BlockIndex].load(std::memory_order_relaxed) + SubIndex;
99 }
100
101private:
102 [[nodiscard]] void* GetRawItem(uint32 Index)
103 {
104 uint32 BlockIndex = Index / ItemsPerPage;
105 uint32 SubIndex = Index % ItemsPerPage;
106 checkLockFreePointerList(Index && Index < (uint32)NextIndex.GetValue() && Index < MaxTotalItems && BlockIndex < MaxBlocks);
107
108 if (!Pages[BlockIndex].load(std::memory_order_relaxed))
109 {
110 T* NewBlock = (T*)LockFreeAllocLinks(ItemsPerPage * sizeof(T));
111 checkLockFreePointerList(IsAligned(NewBlock, alignof(T)));
112 T* ExpectedValue = nullptr;
113 if (!Pages[BlockIndex].compare_exchange_strong(ExpectedValue, NewBlock))
114 {
115 // we lost discard block
116 checkLockFreePointerList(Pages[BlockIndex].load(std::memory_order_relaxed) && Pages[BlockIndex].load(std::memory_order_relaxed) != NewBlock);
117 LockFreeFreeLinks(ItemsPerPage * sizeof(T), NewBlock);
118 }
119 else
120 {
121 checkLockFreePointerList(Pages[BlockIndex].load(std::memory_order_relaxed));
122 }
123 }
124 return (void*)(Pages[BlockIndex].load(std::memory_order_relaxed) + SubIndex);
125 }
126
127 alignas(PLATFORM_CACHE_LINE_SIZE) FThreadSafeCounter NextIndex;
128 alignas(PLATFORM_CACHE_LINE_SIZE) std::atomic<T*> Pages[MaxBlocks];
129};

136MS_ALIGN(8)
137struct FIndexedPointer
138{
139 // no constructor, intentionally. We need to keep the ABA double counter in tact
140
141 // This should only be used for FIndexedPointer's with no outstanding concurrency.
142 // Not recycled links, for example.
143 void Init()
144 {
145 static_assert(((MAX_LOCK_FREE_LINKS - 1) & MAX_LOCK_FREE_LINKS) == 0, "MAX_LOCK_FREE_LINKS must be a power of two");
146 Ptrs.store(0, std::memory_order_relaxed);
147 }
148 inline void SetAll(uint32 Ptr, uint64 CounterAndState)
149 {
150 checkLockFreePointerList(Ptr < MAX_LOCK_FREE_LINKS && CounterAndState < (uint64(1) << (64 - MAX_LOCK_FREE_LINKS_AS_BITS)));
151 Ptrs.store(uint64(Ptr) | (CounterAndState << MAX_LOCK_FREE_LINKS_AS_BITS), std::memory_order_relaxed);
152 }
153
154 [[nodiscard]] UE_FORCEINLINE_HINT uint32 GetPtr() const
155 {
156 return uint32(Ptrs.load(std::memory_order_relaxed) & (MAX_LOCK_FREE_LINKS - 1));
157 }
158
159 UE_FORCEINLINE_HINT void SetPtr(uint32 To)
160 {
161 SetAll(To, GetCounterAndState());
162 }
163
164 [[nodiscard]] UE_FORCEINLINE_HINT uint64 GetCounterAndState() const
165 {
166 return (Ptrs.load(std::memory_order_relaxed) >> MAX_LOCK_FREE_LINKS_AS_BITS);
167 }
168
169 UE_FORCEINLINE_HINT void SetCounterAndState(uint64 To)
170 {
171 SetAll(GetPtr(), To);
172 }
173
174 inline void AdvanceCounterAndState(const FIndexedPointer &From, uint64 TABAInc)
175 {
176 SetCounterAndState(From.GetCounterAndState() + TABAInc);
177 if (UNLIKELY(GetCounterAndState() < From.GetCounterAndState()))
178 {
179 // this is not expected to be a problem and it is not expected to happen very often. When it does happen, we will sleep as an extra precaution.
180 LockFreeTagCounterHasOverflowed();
181 }
182 }
183
184 template<uint64 TABAInc>
185 [[nodiscard]] UE_FORCEINLINE_HINT uint64 GetState() const
186 {
187 return GetCounterAndState() & (TABAInc - 1);
188 }
189
190 template<uint64 TABAInc>
191 inline void SetState(uint64 Value)
192 {
193 checkLockFreePointerList(Value < TABAInc);
194 SetCounterAndState((GetCounterAndState() & ~(TABAInc - 1)) | Value);
195 }
196
197 inline void AtomicRead(const FIndexedPointer& Other)
198 {
199 checkLockFreePointerList(IsAligned(&Ptrs, 8) && IsAligned(&Other.Ptrs, 8));
200 Ptrs.store(Other.Ptrs.load(std::memory_order_acquire), std::memory_order_relaxed);
201 TestCriticalStall();
202 }
203
204 inline bool InterlockedCompareExchange(const FIndexedPointer& Exchange, const FIndexedPointer& Comparand)
205 {
206 TestCriticalStall();
207 uint64 Expected = Comparand.Ptrs.load(std::memory_order_relaxed);
208 return Ptrs.compare_exchange_strong(Expected, Exchange.Ptrs.load(std::memory_order_relaxed), std::memory_order_acq_rel, std::memory_order_relaxed);
209 }
210
211 [[nodiscard]] UE_FORCEINLINE_HINT bool operator==(const FIndexedPointer& Other) const
212 {
213 return Ptrs.load(std::memory_order_relaxed) == Other.Ptrs.load(std::memory_order_relaxed);
214 }
215 [[nodiscard]] UE_FORCEINLINE_HINT bool operator!=(const FIndexedPointer& Other) const
216 {
217 return Ptrs.load(std::memory_order_relaxed) != Other.Ptrs.load(std::memory_order_relaxed);
218 }
219
220private:
221 std::atomic<uint64> Ptrs;
222
223} GCC_ALIGN(8);

262template<int TPaddingForCacheContention, uint64 TABAInc = 1>
263class FLockFreePointerListLIFORoot
264{
265 UE_NONCOPYABLE(FLockFreePointerListLIFORoot)
266
267 typedef FLockFreeLinkPolicy::TDoublePtr TDoublePtr;
268 typedef FLockFreeLinkPolicy::TLink TLink;
269 typedef FLockFreeLinkPolicy::TLinkPtr TLinkPtr;
270
271public:
272 [[nodiscard]] inline FLockFreePointerListLIFORoot()
273 {
274 // We want to make sure we have quite a lot of extra counter values to avoid the ABA problem. This could probably be relaxed, but eventually it will be dangerous.
275 // The question is "how many queue operations can a thread starve for".
276 static_assert(MAX_TagBitsValue / TABAInc >= (1 << 23), "risk of ABA problem");
277 static_assert((TABAInc & (TABAInc - 1)) == 0, "must be power of two");
278 Reset();
279 }
280
281 void Reset()
282 {
283 Head.Init();
284 }
285
286 void Push(TLinkPtr Item)
287 {
288 while (true)
289 {
290 TDoublePtr LocalHead;
291 LocalHead.AtomicRead(Head);
292 TDoublePtr NewHead;
293 NewHead.AdvanceCounterAndState(LocalHead, TABAInc);
294 NewHead.SetPtr(Item);
295 FLockFreeLinkPolicy::DerefLink(Item)->SingleNext.store(LocalHead.GetPtr(), std::memory_order_relaxed);
296 if (Head.InterlockedCompareExchange(NewHead, LocalHead))
297 {
298 break;
299 }
300 }
301 }
302
303 bool PushIf(TFunctionRef<TLinkPtr(uint64)> AllocateIfOkToPush)
304 {
305 static_assert(TABAInc > 1, "method should not be used for lists without state");
306 while (true)
307 {
308 TDoublePtr LocalHead;
309 LocalHead.AtomicRead(Head);
310 uint64 LocalState = LocalHead.GetState<TABAInc>();
311 TLinkPtr Item = AllocateIfOkToPush(LocalState);
312 if (!Item)
313 {
314 return false;
315 }
316
317 TDoublePtr NewHead;
318 NewHead.AdvanceCounterAndState(LocalHead, TABAInc);
319 FLockFreeLinkPolicy::DerefLink(Item)->SingleNext.store(LocalHead.GetPtr(), std::memory_order_relaxed);
320 NewHead.SetPtr(Item);
321 if (Head.InterlockedCompareExchange(NewHead, LocalHead))
322 {
323 break;
324 }
325 }
326 return true;
327 }
328
329
330 TLinkPtr Pop()
331 {
332 TLinkPtr Item = 0;
333 while (true)
334 {
335 TDoublePtr LocalHead;
336 LocalHead.AtomicRead(Head);
337 Item = LocalHead.GetPtr();
338 if (!Item)
339 {
340 break;
341 }
342 TDoublePtr NewHead;
343 NewHead.AdvanceCounterAndState(LocalHead, TABAInc);
344 TLink* ItemP = FLockFreeLinkPolicy::DerefLink(Item);
345 NewHead.SetPtr(ItemP->SingleNext.load(std::memory_order_relaxed));
346 if (Head.InterlockedCompareExchange(NewHead, LocalHead))
347 {
348 ItemP->SingleNext.store(0, std::memory_order_relaxed);
349 break;
350 }
351 }
352 return Item;
353 }
354
355 TLinkPtr PopAll()
356 {
357 TLinkPtr Item = 0;
358 while (true)
359 {
360 TDoublePtr LocalHead;
361 LocalHead.AtomicRead(Head);
362 Item = LocalHead.GetPtr();
363 if (!Item)
364 {
365 break;
366 }
367 TDoublePtr NewHead;
368 NewHead.AdvanceCounterAndState(LocalHead, TABAInc);
369 NewHead.SetPtr(0);
370 if (Head.InterlockedCompareExchange(NewHead, LocalHead))
371 {
372 break;
373 }
374 }
375 return Item;
376 }
377
378 TLinkPtr PopAllAndChangeState(TFunctionRef<uint64(uint64)> StateChange)
379 {
380 static_assert(TABAInc > 1, "method should not be used for lists without state");
381 TLinkPtr Item = 0;
382 while (true)
383 {
384 TDoublePtr LocalHead;
385 LocalHead.AtomicRead(Head);
386 Item = LocalHead.GetPtr();
387 TDoublePtr NewHead;
388 NewHead.AdvanceCounterAndState(LocalHead, TABAInc);
389 NewHead.SetState<TABAInc>(StateChange(LocalHead.GetState<TABAInc>()));
390 NewHead.SetPtr(0);
391 if (Head.InterlockedCompareExchange(NewHead, LocalHead))
392 {
393 break;
394 }
395 }
396 return Item;
397 }
398
399 [[nodiscard]] UE_FORCEINLINE_HINT bool IsEmpty() const
400 {
401 return !Head.GetPtr();
402 }

406 TDoublePtr LocalHead;
407 LocalHead.AtomicRead(Head);
408 return LocalHead.GetState<TABAInc>();

445 bool PushIf(T* InPayload, TFunctionRef<bool(uint64)> OkToPush)
446 {
447 TLinkPtr Item = 0;
448
449 auto AllocateIfOkToPush = [&OkToPush, InPayload, &Item](uint64 State)->TLinkPtr
450 {
451 if (OkToPush(State))
452 {
453 if (!Item)
454 {
455 Item = FLockFreeLinkPolicy::AllocLockFreeLink();
456 FLockFreeLinkPolicy::DerefLink(Item)->Payload.store(InPayload, std::memory_order_relaxed);
457 }
458 return Item;
459 }
460 return 0;
461 };
462 if (!RootList.PushIf(AllocateIfOkToPush))
463 {
464 if (Item)
465 {
466 // we allocated the link, but it turned out that the list was closed
467 FLockFreeLinkPolicy::FreeLockFreeLink(Item);
468 }
469 return false;
470 }
471 return true;
472 }

476 {
477 TLinkPtr Item = RootList.Pop();
478 T* Result = nullptr;
479 if (Item)
480 {
481 Result = (T*)FLockFreeLinkPolicy::DerefLink(Item)->Payload.load(std::memory_order_relaxed);
482 FLockFreeLinkPolicy::FreeLockFreeLink(Item);
483 }
484 return Result;
485 }

651 NewHead.AdvanceCounterAndState(LocalHead, TABAInc);

656 }

689template<class T, int TPaddingForCacheContention, int NumPriorities>
690class FStallingTaskQueue
691{
692 UE_NONCOPYABLE(FStallingTaskQueue)
693
694 typedef FLockFreeLinkPolicy::TDoublePtr TDoublePtr;
695 typedef FLockFreeLinkPolicy::TLink TLink;
696 typedef FLockFreeLinkPolicy::TLinkPtr TLinkPtr;
697public:
698 [[nodiscard]] FStallingTaskQueue()
699 {
700 MasterState.Init();
701 }
702 int32 Push(T* InPayload, uint32 Priority)
703 {
704 checkLockFreePointerList(Priority < NumPriorities);
705 TDoublePtr LocalMasterState;
706 LocalMasterState.AtomicRead(MasterState);
707 PriorityQueues[Priority].Push(InPayload);
708 TDoublePtr NewMasterState;
709 NewMasterState.AdvanceCounterAndState(LocalMasterState, 1);
710 int32 ThreadToWake = FindThreadToWake(LocalMasterState.GetPtr());
711 if (ThreadToWake >= 0)
712 {
713 NewMasterState.SetPtr(TurnOffBit(LocalMasterState.GetPtr(), ThreadToWake));
714 }
715 else
716 {
717 NewMasterState.SetPtr(LocalMasterState.GetPtr());
718 }
719 while (!MasterState.InterlockedCompareExchange(NewMasterState, LocalMasterState))
720 {
721 LocalMasterState.AtomicRead(MasterState);
722 NewMasterState.AdvanceCounterAndState(LocalMasterState, 1);
723 ThreadToWake = FindThreadToWake(LocalMasterState.GetPtr());
724#if 0
725 // This block of code is supposed to avoid starting the task thread if the queues are empty.
726 // There WAS a silly bug here. In rare cases no task thread is woken up.
727 // That bug has been fixed, but I don't think we really need this code anyway.
728 // Without this block, it is possible that we do a redundant wake-up, but for task threads, that can happen anyway.
729 // For named threads, the rare redundant wakeup seems acceptable.
730 if (ThreadToWake >= 0)
731 {
732 bool bAny = false;
733 for (int32 Index = 0; !bAny && Index < NumPriorities; Index++)
734 {
735 bAny = !PriorityQueues[Index].IsEmpty();
736 }
737 if (!bAny) // if there is nothing in the queues, then don't wake anyone
738 {
739 ThreadToWake = -1;
740 }
741 }
742#endif
743 if (ThreadToWake >= 0)
744 {
745 NewMasterState.SetPtr(TurnOffBit(LocalMasterState.GetPtr(), ThreadToWake));
746 }
747 else
748 {

750 }

755 [[nodiscard]] T* Pop(int32 MyThread, bool bAllowStall)
756 {
757 check(MyThread >= 0 && MyThread < FLockFreeLinkPolicy::MAX_BITS_IN_TLinkPtr);
758
759 while (true)
760 {
761 TDoublePtr LocalMasterState;
762 LocalMasterState.AtomicRead(MasterState);
763 //checkLockFreePointerList(!TestBit(LocalMasterState.GetPtr(), MyThread) || !FPlatformProcess::SupportsMultithreading()); // you should not be stalled if you are asking for a task
764 for (int32 Index = 0; Index < NumPriorities; Index++)
765 {
766 T *Result = PriorityQueues[Index].Pop();
767 if (Result)
768 {
769 while (true)
770 {
771 TDoublePtr NewMasterState;
772 NewMasterState.AdvanceCounterAndState(LocalMasterState, 1);
773 NewMasterState.SetPtr(LocalMasterState.GetPtr());
774 if (MasterState.InterlockedCompareExchange(NewMasterState, LocalMasterState))
775 {
776 return Result;
777 }
778 LocalMasterState.AtomicRead(MasterState);
779 checkLockFreePointerList(!TestBit(LocalMasterState.GetPtr(), MyThread) || !FPlatformProcess::SupportsMultithreading()); // you should not be stalled if you are asking for a task
780 }
781 }
782 }
783 if (!bAllowStall)
784 {
785 break; // if we aren't stalling, we are done, the queues are empty
786 }
787 {
788 TDoublePtr NewMasterState;
789 NewMasterState.AdvanceCounterAndState(LocalMasterState, 1);
790 NewMasterState.SetPtr(TurnOnBit(LocalMasterState.GetPtr(), MyThread));
791 if (MasterState.InterlockedCompareExchange(NewMasterState, LocalMasterState))
792 {
793 break;
794 }
795 }
796 }
797 return nullptr;
798 }

803 {
804 int32 Result = -1;
805 UPTRINT Test = UPTRINT(Ptr);
806 if (Test)
807 {
808 Result = 0;
809 while (!(Test & 1))
810 {
811 Test >>= 1;
812 Result++;
813 }
814 }
815 return Result;
816 }
