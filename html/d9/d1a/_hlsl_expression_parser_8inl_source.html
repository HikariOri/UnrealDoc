567 EParseResult ParseGeneralType(FHlslScanner& Scanner, int32 TypeFlags, FSymbolScope* SymbolScope, FLinearAllocator* Allocator, AST::FTypeSpecifier** OutSpecifier)
568 {
569 auto* Token = Scanner.PeekToken();
570
571 // Handle types with namespaces
572 {
573 auto* Token1 = Scanner.PeekToken(1);
574 auto* Token2 = Scanner.PeekToken(2);
575 TStringBuilder<256> TypeString;
576 if (Token && Token->Token == EHlslToken::Identifier && Token1 && Token1->Token == EHlslToken::ColonColon && Token2 && Token2->Token == EHlslToken::Identifier && SymbolScope)
577 {
578 auto* Namespace = SymbolScope->GetGlobalScope();
579 do
580 {
581 Token = Scanner.PeekToken();
582 check(Scanner.MatchToken(EHlslToken::Identifier));
583 auto* OuterNamespace = Token;
584 check(Scanner.MatchToken(EHlslToken::ColonColon));
585 auto* InnerOrType = Scanner.PeekToken();
586 if (!InnerOrType)
587 {
588 Scanner.SourceError(*FString::Printf(TEXT("Expecting identifier for type '%s'!"), InnerOrType ? *InnerOrType->String : TEXT("null")));
589 return ParseResultError();
590 }
591
592 Namespace = Namespace->FindNamespace(*OuterNamespace->String);
593 if (!Namespace)
594 {
595 Scanner.SourceError(*FString::Printf(TEXT("Unknown namespace '%s'!"), *TypeString));
596 return ParseResultError();
597 }
598 TypeString += OuterNamespace->String;
599 TypeString += TEXT("::");
600 auto* Token3 = Scanner.PeekToken(1);
601 if (Token3 && Token3->Token == EHlslToken::ColonColon)
602 {
603 continue;
604 }
605 else
606 {
607 if (InnerOrType && FChar::IsAlpha(InnerOrType->String[0]))
608 {
609 Scanner.Advance();
610 if (FSymbolScope::FindType(Namespace, InnerOrType->String, false))
611 {
612 auto* Type = new(Allocator) AST::FTypeSpecifier(Allocator, Token->SourceInfo);
613 Type->TypeName = Allocator->Strdup(TypeString.ToView());
614 *OutSpecifier = Type;
615 return EParseResult::Matched;
616 }
617 else
618 {
619 Scanner.SourceError(*FString::Printf(TEXT("Unknown type '%s'!"), *TypeString));
620 return ParseResultError();
621 }
622 }
623 break;
624 }
625 }
626 while (Token);
627 }
628 }
629
630 int32 ExtraQualifiers = 0;
631 bool bPrecise = false;
632 if (Scanner.MatchToken(EHlslToken::Precise))
633 {
634 Token = Scanner.GetCurrentToken();
635 ExtraQualifiers |= EEQ_PRECISE;
636 }
637
638 if ((TypeFlags & ETF_UNORM) == ETF_UNORM && Token && Token->String.Len() == 5)
639 {
640 if (!FCString::Strcmp(*Token->String, TEXT("unorm")))
641 {
642 ExtraQualifiers |= EEQ_UNORM;
643 Scanner.Advance();
644 Token = Scanner.GetCurrentToken();
645 }
646 else if (!FCString::Strcmp(*Token->String, TEXT("snorm")))
647 {
648 ExtraQualifiers |= EEQ_SNORM;
649 Scanner.Advance();
650 Token = Scanner.GetCurrentToken();
651 }
652 }
653
654 auto Result = ParseGeneralTypeFromToken(Token, TypeFlags, ExtraQualifiers, SymbolScope, Allocator, OutSpecifier);
655 if (Result == EParseResult::Matched)
656 {
657 Scanner.Advance();
658 return EParseResult::Matched;
659 }
660
661 if (Result == EParseResult::Error && (TypeFlags & ETF_ERROR_IF_NOT_USER_TYPE))
662 {
663 Scanner.SourceError(*FString::Printf(TEXT("Unknown type '%s'!"), Token ? *Token->String : TEXT("<null>")));
664 return Result;
665 }
666
667 return EParseResult::NotMatched;
668 }

670 EParseResult ParseFullType(FHlslScanner& Scanner, int32 TypeFlags, int32 TemplateTypeFlags, FSymbolScope* SymbolScope, FLinearAllocator* Allocator, AST::FFullySpecifiedType** OutFullySpecifiedType)
671 {
672 AST::FTypeSpecifier* TypeSpecifier = nullptr;
673 EParseResult Result = ParseGeneralType(Scanner, TypeFlags, SymbolScope, Allocator, &TypeSpecifier);
674 if (Result != EParseResult::Matched)
675 {
676 return Result;
677 }
678
679 // OutFullySpecifiedType could already be initialized
680 AST::FFullySpecifiedType* FullType = *OutFullySpecifiedType ? *OutFullySpecifiedType : new(Allocator) AST::FFullySpecifiedType(Allocator, TypeSpecifier->SourceInfo);
681 FullType->Specifier = TypeSpecifier;
682
683 if (Scanner.MatchToken(EHlslToken::Lower))
684 {
685 AST::FTypeSpecifier* ElementTypeSpecifier = nullptr;
686 EParseResult InnerResult = ParseGeneralType(Scanner, TemplateTypeFlags, SymbolScope, Allocator, &ElementTypeSpecifier);
687 if (InnerResult != EParseResult::Matched)
688 {
689 Scanner.SourceError(TEXT("Expected type!"));
690 return ParseResultError();
691 }
692
693 FullType->Specifier->InnerType = ElementTypeSpecifier->TypeName;
694
695 if (Scanner.MatchToken(EHlslToken::Comma))
696 {
697 auto* Integer = Scanner.GetCurrentToken();
698 if (!Scanner.MatchIntegerLiteral())
699 {
700 Scanner.SourceError(TEXT("Expected constant!"));
701 return ParseResultError();
702 }
703 FullType->Specifier->TextureMSNumSamples = FCString::Atoi(*Integer->String);
704 }
705

707 {
708 Scanner.SourceError(TEXT("Expected '>'!"));
709 return ParseResultError();
710 }
711 }
712

714
715 return EParseResult::Matched;

861 {
862 bool bFoundAny = false;
863 bool bTryAgain = true;
864 AST::FExpression*& PrevExpression = *InOutExpression;

866 {
867 auto* Token = Scanner.GetCurrentToken();
868 AST::EOperators Operator = AST::EOperators::Plus;

871 {
872 case EHlslToken::LeftSquareBracket:

874 Scanner.Advance();

877 if (Result != EParseResult::Matched)

884 {
885 Scanner.SourceError(TEXT("Expected ']'!"));
886 return ParseResultError();
887 }
888
889 auto* ArrayIndexExpression = new(Allocator) AST::FBinaryExpression(Allocator, AST::EOperators::ArrayIndex, PrevExpression, ArrayIndex, Token->SourceInfo);
890 PrevExpression = ArrayIndexExpression;
891 bFoundAny = true;
892 }
893 break;
894 case EHlslToken::Dot:

896 Scanner.Advance();

899 {
900 Scanner.SourceError(TEXT("Expected identifier for member or swizzle!"));
901 return ParseResultError();

903 auto* FieldExpression = new(Allocator) AST::FUnaryExpression(Allocator, AST::EOperators::FieldSelection, PrevExpression, Token->SourceInfo);
904 FieldExpression->Identifier = AST::FIdentifier::New(Allocator, Identifier->String);
905 PrevExpression = FieldExpression;
906 bFoundAny = true;
907 }
908 break;

911 Scanner.Advance();

921 {
922 AST::FTypeSpecifier* ElementTypeSpecifier = nullptr;

959 if (Result != EParseResult::Matched)
960 {
961 Scanner.SourceError(TEXT("Expected ')'!"));
962 return ParseResultError();
963 }
964
965 PrevExpression = FunctionCall;
966
967 bFoundAny = true;
968 }
969
970
971 }
972 break;
973 case EHlslToken::PlusPlus:
974 {
975 Scanner.Advance();

977 PrevExpression = IncExpression;
978 bFoundAny = true;
979 }
980 break;
981 case EHlslToken::MinusMinus:
982 {
983 Scanner.Advance();
984 auto* DecExpression = new(Allocator) AST::FUnaryExpression(Allocator, AST::EOperators::PostDec, PrevExpression, Token->SourceInfo);
985 PrevExpression = DecExpression;
986 bFoundAny = true;
987 }
988 break;
989 case EHlslToken::Question:
990 {
991 Scanner.Advance();

993 if (ComputeExpr(Scanner, 0, /*Info,*/ SymbolScope, (ExpressionFlags | EEF_ALLOW_ASSIGNMENT), Allocator, &Left, nullptr) != EParseResult::Matched)
994 {
995 Scanner.SourceError(TEXT("Expected expression!"));
996 return ParseResultError();

999 {
1000 Scanner.SourceError(TEXT("Expected ':'!"));

1005 {
1006 Scanner.SourceError(TEXT("Expected expression!"));
1007 return ParseResultError();

1010 auto* Ternary = new(Allocator) AST::FExpression(Allocator, AST::EOperators::Conditional, nullptr, Left, Right, Token->SourceInfo);
1011 *OutTernaryExpression = Ternary;
1012 bFoundAny = true;
1013 bTryAgain = false;
1014 }
1015 break;
1016 case EHlslToken::Lower:
1017 {
1018 if (SymbolScope && SymbolScope->ScopeType == FSymbolScope::EType::Statement)
1019 {
1020 const FHlslToken* Token1 = Scanner.PeekToken(1);
1021
1022 AST::FTypeSpecifier* TypeSpecifier = nullptr;
1023 if (ParseGeneralTypeFromToken(Token1, ETF_BUILTIN_NUMERIC | ETF_USER_TYPES, 0, SymbolScope, Allocator, &TypeSpecifier) == EParseResult::Matched )
1024 {

1026

1028

1030 {
1031 const auto* CurrentToken = Scanner.PeekToken();
1032 if (CurrentToken->Token == EHlslToken::Greater)
1033 {
1034 Scanner.Advance();
1035 bFoundAny = true;
1036 break;
1037 }
1038
1039 if (CurrentToken->Token == EHlslToken::Comma)
1040 {
1041 Scanner.Advance();
1042 }
1043 else
1044 {
1045 AST::FTypeSpecifier* ElementTypeSpecifier = nullptr;
1046 EParseResult InnerResult = ParseGeneralType(Scanner, ETF_BUILTIN_NUMERIC | ETF_USER_TYPES, SymbolScope, Allocator, &ElementTypeSpecifier);
1047 if (InnerResult != EParseResult::Matched)
1048 {
1049 Scanner.SourceError(TEXT("Expected numeric or user type!"));
1050 return ParseResultError();
1051 }
1052
1053 AST::FExpression* CurrentExpression = new(Allocator) AST::FExpression(Allocator, AST::EOperators::Identifier, CurrentToken->SourceInfo);
1054 ExpressionList->Expressions.Add(CurrentExpression);
1055 }
1056 }
1057
1058 if (!bFoundAny)
1059 {
1060 Scanner.SourceError(TEXT("Expected >"));
1061 return ParseResultError();
1062 }
1063
1064 PrevExpression = ExpressionList;
1065 }
1066 else
1067 {
1068 bTryAgain = false;
1069 }
1070 }
1071 else
1072 {
1073 bTryAgain = false;
1074 }
1075
1076 }
1077 break;
1078 default:
1079 bTryAgain = false;
1080 break;
1081 }
1082 }
1083
1084 *InOutExpression = PrevExpression;
1085 return bFoundAny ? EParseResult::Matched : EParseResult::NotMatched;
1086 }

1395 EParseResult ComputeExpr(FHlslScanner& Scanner, int32 MinPrec, /*FInfo& Info,*/ FSymbolScope* SymbolScope, int32 ExpressionFlags, FLinearAllocator* Allocator, AST::FExpression** OutExpression, AST::FExpression** OutTernaryExpression)
1396 {

1398
1399 auto OriginalToken = Scanner.GetCurrentTokenIndex();
1400 //FInfoIndentScope Scope(Info);
1401 /*
1402 // Precedence Climbing
1403 // http://eli.thegreenplace.net/2012/08/02/parsing-expressions-by-precedence-climbing
1404 compute_expr(min_prec):
1405 result = compute_atom()
1406
1407 while cur token is a binary operator with precedence >= min_prec:
1408 prec, assoc = precedence and associativity of current token
1409 if assoc is left:
1410 next_min_prec = prec + 1
1411 else:
1412 next_min_prec = prec
1413 rhs = compute_expr(next_min_prec)
1414 result = compute operator(result, rhs)
1415
1416 return result
1417 */
1418 //Info.PrintWithTabs(FString::Printf(TEXT("Compute Expr %d\n"), MinPrec));
1419 AST::FExpression* TernaryExpression = nullptr;
1420 auto Result = ComputeAtom(Scanner, /*Info,*/ SymbolScope, ExpressionFlags, Allocator, OutExpression, &TernaryExpression);
1421 if (Result != EParseResult::Matched)
1422 {
1423 return Result;
1424 }
1425 check(*OutExpression);
1426 do
1427 {
