761 void CondCheckCountQuota()
762 {
763 // 64 RPC interval
764 if ((RPCIntervalCounter & 0x3F) == 0)
765 {
766 FTickScope& TickScope = GetTickScope();
767
768 if (HasHitQuota_Count(CounterPerPeriodHistory, TickScope.FrameCounter))
769 {
770 TickScope.UpdateSeverity(*this, ERPCDoSSeverityUpdate::Escalate, ERPCDoSEscalateReason::CountLimit);
771 }
772 }
773 }

780 void CondCheckTimeQuota(double TimeSeconds)
781 {
782 if (TimeSeconds > NextTimeQuotaCheck)
783 {
784 FTickScope& TickScope = GetTickScope();
785
786 NextTimeQuotaCheck = TimeSeconds + TimeQuotaCheckInterval;
787
788 if (HasHitQuota_Time(CounterPerPeriodHistory, TickScope.FrameCounter))
789 {
790 TickScope.UpdateSeverity(*this, ERPCDoSSeverityUpdate::Escalate, ERPCDoSEscalateReason::TimeLimit);
791 }
792 }
