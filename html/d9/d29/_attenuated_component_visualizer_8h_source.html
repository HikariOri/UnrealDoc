102 }
103 break;

110 DrawWireSphereAutoSides(PDI, Translation, InnerRadiusColor, ShapeDetails.Extents.X, SDPG_World);
111 }
112 else
113 {
114 DrawWireSphereAutoSides(PDI, Translation, OuterRadiusColor, ShapeDetails.Extents.X, SDPG_World);
115 }
116 break;
117
118 default:
119 check(false);
120 }
121 }
122 }
123 }
