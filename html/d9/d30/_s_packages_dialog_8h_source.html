312 FString GetCheckedOutByString() const
313 {
314 ISourceControlProvider& SourceControlProvider = ISourceControlModule::Get().GetProvider();
315 FSourceControlStatePtr SourceControlState = SourceControlProvider.GetState(Package, EStateCacheUsage::Use);
316
317 FString CheckedOutBy;
318 if (SourceControlState.IsValid())
319 {
320 SourceControlState->IsCheckedOutOther(&CheckedOutBy);
321
322 // If not checked out in this branch, see if checked out or modified in other branches
323 if (!CheckedOutBy.Len())
324 {
325 // Check for modification first, as this is a more important state than only checked out
326 if (SourceControlState->IsModifiedInOtherBranch())
327 {
328 FString HeadBranch, HeadAction;
329 int32 HeadCL;
330 SourceControlState->GetOtherBranchHeadModification(HeadBranch, HeadAction, HeadCL);
331
332 FNumberFormattingOptions NoCommas;
333 NoCommas.UseGrouping = false;
334 return FString::Format(TEXT("Modified in {0} CL:{1} ({2})"), { HeadBranch, FText::AsNumber(HeadCL, &NoCommas).ToString(), HeadAction });
335 }
336
337 if (SourceControlState->IsCheckedOutInOtherBranch())
338 {
339 return SourceControlState->GetOtherUserBranchCheckedOuts();
340 }
341 }
342 }
343
344 return CheckedOutBy;
