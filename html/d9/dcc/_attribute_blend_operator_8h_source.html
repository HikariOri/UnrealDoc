35 template <typename Type>
36 inline typename TEnableIf<TAttributeTypeTraits<Type>::IsBlendable, void>::Type Blend_Internal(const FAttributeBlendData& BlendData, FStackAttributeContainer* OutAttributes) const
37 {
38 BlendData.ForEachAttributeSet<AttributeType>([OutAttributes](typename FAttributeBlendData::template TAttributeSetIterator<AttributeType>& It) -> void
39 {
40 const FAttributeId& Identifier = It.GetIdentifier();
41 const ECustomAttributeBlendType BlendType = Attributes::GetAttributeBlendType(Identifier);
42
43 AttributeType* OutAttributePtr = OutAttributes->FindOrAdd<AttributeType>(Identifier);
44 AttributeType& OutAttribute = (*OutAttributePtr);
45
46 if (BlendType == ECustomAttributeBlendType::Override)
47 {
48 // Not iterating over each attribute, but just picking highest weighted value
49 OutAttribute = It.GetHighestWeightedValue();
50 }
51 else
52 {
53 // Iterate over each attribute while accumulating final value according to provided weightings
54 while (It.Next())
55 {
56 const AttributeType& Attribute = It.GetValue();
57 const float AttributeWeight = It.GetWeight();
58
59 if (It.GetIndex() == 0)
60 {
61 OutAttribute = Attribute;
62 OutAttribute = Attribute.Multiply(AttributeWeight);
63 }
64 else
65 {
66 OutAttribute.Accumulate(Attribute, AttributeWeight, It.GetAdditiveType());
67 }
68 }
69
70 if constexpr (UE::Anim::TAttributeTypeTraits<AttributeType>::RequiresNormalization)
71 {
72 OutAttribute.Normalize();
73 }
74 }
75 });
76
77 BlendData.ForEachUniqueAttribute<AttributeType>([OutAttributes](typename FAttributeBlendData::template TSingleIterator<AttributeType>& It) -> void
78 {
79 while (It.Next())
80 {
81 const AttributeType& Attribute = It.GetValue();
82 const float AttributeWeight = It.GetWeight();
83 const FAttributeId& Identifier = It.GetIdentifier();
84 const ECustomAttributeBlendType BlendType = Attributes::GetAttributeBlendType(Identifier);
85
86 AttributeType* OutAttributePtr = OutAttributes->FindOrAdd<AttributeType>(Identifier);
87 AttributeType& OutAttribute = (*OutAttributePtr);
88
89 // Choose between weighted (blended) assignment vs override
90 if (BlendType == ECustomAttributeBlendType::Override)
91 {
92 OutAttribute = Attribute;
93 }
94 else
95 {
96 OutAttribute.Interpolate(Attribute, AttributeWeight);
97 }
98
99 if constexpr (UE::Anim::TAttributeTypeTraits<AttributeType>::RequiresNormalization)
100 {
101 OutAttribute.Normalize();
102 }
103 }

132 inline typename TEnableIf<TAttributeTypeTraits<Type>::IsBlendable, void>::Type BlendPerBone_Internal(const FAttributeBlendData& BlendData, FStackAttributeContainer* OutAttributes) const
133 {
134 BlendData.ForEachAttributeSet<AttributeType>([this, OutAttributes](typename FAttributeBlendData::template TAttributeSetIterator<AttributeType>& It) -> void
135 {
136 const FAttributeId& Identifier = It.GetIdentifier();
137 const ECustomAttributeBlendType BlendType = Attributes::GetAttributeBlendType(Identifier);
138
139 const int32 ExistingIndex = OutAttributes->IndexOfByKey<AttributeType>(Identifier);
140 AttributeType* OutAttributePtr = OutAttributes->FindOrAdd<AttributeType>(Identifier);
141 AttributeType& OutAttribute = *OutAttributePtr;
142
143 if (BlendType == ECustomAttributeBlendType::Override)
144 {
145 // Not iterating over each attribute, but just picking highest weighted value
146 OutAttribute = It.GetHighestBoneWeightedValue();
147 }
148 else
149 {
150 while (It.Next())
151 {
152 const float AttributeWeight = It.GetBoneWeight();
153 const AttributeType& Attribute = It.GetValue();
154 if (FAnimWeight::IsFullWeight(AttributeWeight))
155 {
156 OutAttribute = Attribute;
157 break;
158 }
159 else
160 {
161 if (It.GetIndex() == 0 && ExistingIndex == INDEX_NONE)
162 {
163 OutAttribute = Attribute.Multiply(AttributeWeight);
164 }
165 else
166 {
167 OutAttribute.Accumulate(Attribute, AttributeWeight, It.GetAdditiveType());
168 }
169 }
170 }
171
172 if constexpr (UE::Anim::TAttributeTypeTraits<AttributeType>::RequiresNormalization)
173 {
174 OutAttribute.Normalize();
175 }
176 }
177 });

180 {
181 while (It.Next())
182 {
183 const FAttributeId& Identifier = It.GetIdentifier();
184 const int32 ExistingIndex = OutAttributes->IndexOfByKey<AttributeType>(Identifier);
185
186 const float AttributeWeight = It.GetBoneWeight();
187 AttributeType* OutAttributePtr = OutAttributes->FindOrAdd<AttributeType>(Identifier);
188 AttributeType& OutAttribute = *OutAttributePtr;
189
190 const ECustomAttributeBlendType BlendType = Attributes::GetAttributeBlendType(Identifier);
191 if (BlendType == ECustomAttributeBlendType::Override)
192 {
193 OutAttribute = It.GetValue();
194 }
195 else
196 {
197 const AttributeType& Attribute = It.GetValue();
198 if (ExistingIndex == INDEX_NONE)
199 {
200 OutAttribute.Interpolate(Attribute, AttributeWeight);
201 }
202 else
203 {
204 OutAttribute.Accumulate(Attribute, AttributeWeight, It.GetAdditiveType());
205 }
206 }
