469 static bool IsVertexProgrammable(const FMaterialShaderParameters& MaterialParameters, bool bHWRasterShader)
470 {
471 const bool bPixelProgrammable = IsPixelProgrammable(MaterialParameters);
472 const bool bHasVertexUVs = bPixelProgrammable && (MaterialParameters.bHasVertexInterpolator || MaterialParameters.NumCustomizedUVs > 0);
473 const bool bHasTessellation = (!bHWRasterShader && MaterialParameters.bIsTessellationEnabled);
474 return MaterialParameters.bHasVertexPositionOffsetConnected || bHasVertexUVs || bHasTessellation || MaterialParameters.bHasMaterialCacheOutput || MaterialParameters.bHasFirstPersonInterpolation;
475 }
476
477 static bool IsVertexProgrammable(uint32 MaterialBitFlags)
478 {
479 return (MaterialBitFlags & NANITE_MATERIAL_VERTEX_PROGRAMMABLE_FLAGS);
480 }
481
482 static bool IsPixelProgrammable(const FMaterialShaderParameters& MaterialParameters)
483 {
484 return MaterialParameters.bIsMasked || MaterialParameters.bHasPixelDepthOffsetConnected;
485 }
486
487 static bool IsPixelProgrammable(uint32 MaterialBitFlags)
488 {
489 return (MaterialBitFlags & NANITE_MATERIAL_PIXEL_PROGRAMMABLE_FLAGS);
490 }
491
492 static bool ShouldCompileProgrammablePermutation(
493 const FMaterialShaderParameters& MaterialParameters,
494 bool bPermutationVertexProgrammable,
495 bool bPermutationPixelProgrammable,
496 bool bHWRasterShader)
497 {
498 if (MaterialParameters.bIsDefaultMaterial)
499 {
500 return true;
501 }
502
503 // Custom materials should compile only the specific combination that is actually used
504 // TODO: The status of material attributes on the FMaterialShaderParameters is determined without knowledge of any static
505 // switches' values, and therefore when true could represent the set of materials that both enable them and do not. We could
506 // isolate a narrower set of required shaders if FMaterialShaderParameters reflected the status after static switches are
507 // applied.
508 //return IsVertexProgrammable(MaterialParameters, bHWRasterShader) == bPermutationVertexProgrammable &&
509 // IsPixelProgrammable(MaterialParameters) == bPermutationPixelProgrammable;
510 return (IsVertexProgrammable(MaterialParameters, bHWRasterShader) || !bPermutationVertexProgrammable) &&
511 (IsPixelProgrammable(MaterialParameters) || !bPermutationPixelProgrammable) &&
512 (bPermutationVertexProgrammable || bPermutationPixelProgrammable);
513 }
514
515
516 static bool ShouldCompilePixelPermutation(const FMaterialShaderPermutationParameters& Parameters)
517 {
518 // Always compile default material as the fast opaque "fixed function" raster path
519 bool bValidMaterial = Parameters.MaterialParameters.bIsDefaultMaterial;
520
521 // Compile this pixel shader if it requires programmable raster
522 if (Parameters.MaterialParameters.bIsUsedWithNanite && FNaniteMaterialShader::IsPixelProgrammable(Parameters.MaterialParameters))
523 {
524 bValidMaterial = true;
525 }
526
527 return
528 DoesPlatformSupportNanite(Parameters.Platform) &&
529 Parameters.MaterialParameters.MaterialDomain == MD_Surface &&
530 bValidMaterial;
531 }
532
533 static bool ShouldCompileVertexPermutation(const FMaterialShaderPermutationParameters& Parameters)
534 {
535 // Always compile default material as the fast opaque "fixed function" raster path
536 bool bValidMaterial = Parameters.MaterialParameters.bIsDefaultMaterial;
537
538 // Compile this vertex shader if it requires programmable raster
539 static const bool bHWRasterShader = true; // all vertex permutations are HWRaster
540 if (Parameters.MaterialParameters.bIsUsedWithNanite &&
541 FNaniteMaterialShader::IsVertexProgrammable(Parameters.MaterialParameters, bHWRasterShader))
542 {
543 bValidMaterial = true;
544 }
545
546 return
547 DoesPlatformSupportNanite(Parameters.Platform) &&
548 Parameters.MaterialParameters.MaterialDomain == MD_Surface &&
549 bValidMaterial;
550 }
551
552 static bool ShouldCompileComputePermutation(const FMaterialShaderPermutationParameters& Parameters)
553 {
554 // Always compile default material as the fast opaque "fixed function" raster path
555 bool bValidMaterial = Parameters.MaterialParameters.bIsDefaultMaterial;
556
557 // Compile this compute shader if it requires programmable raster
558 static const bool bHWRasterShader = false; // all compute permutations are SWRaster
559 if (Parameters.MaterialParameters.bIsUsedWithNanite &&
560 (IsVertexProgrammable(Parameters.MaterialParameters, bHWRasterShader) || IsPixelProgrammable(Parameters.MaterialParameters)))
561 {
562 bValidMaterial = true;
563 }
564
565 return
566 DoesPlatformSupportNanite(Parameters.Platform) &&
567 Parameters.MaterialParameters.MaterialDomain == MD_Surface &&
568 DoesMaterialQualitySupportNanite(Parameters.Platform, Parameters.MaterialParameters.QualityLevel) &&
569 bValidMaterial;
570 }

932 inline uint32 GetPipelineHash() const
933 {
934 // Ignoring the lower 4 bits since they are likely zero anyway.
935 // Higher bits are more significant in 64 bit builds.
936 uint64 PipelineHash = uint64(reinterpret_cast<UPTRINT>(MaterialProxy) >> 4);
937
938 // Combine with shader and UB pointers so recompiled shaders / recreated UBs create new entries
939 PipelineHash = CityHash128to64({ PipelineHash, reinterpret_cast<UPTRINT>(ComputeShader) });
940 PipelineHash = CityHash128to64({ PipelineHash, reinterpret_cast<UPTRINT>(WorkGraphShader) });
941 PipelineHash = CityHash128to64({ PipelineHash, reinterpret_cast<UPTRINT>(MaterialUB) });
942
943 // Combine shader flags hash and material hash
944 PipelineHash = CityHash128to64({ PipelineHash, ShadingFlagsHash });
945
946 // Combine with bound target mask
947 PipelineHash = CityHash128to64({ PipelineHash, BoundTargetMask });
948
949 // Combine with shader bindings hash
950 PipelineHash = CityHash128to64({ PipelineHash, ShaderBindingsHash });
951
952 return HashCombineFast(uint32(PipelineHash & 0xFFFFFFFF), uint32((PipelineHash >> 32) & 0xFFFFFFFF));
953 }
