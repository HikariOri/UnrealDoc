117 void GetTransformedTriangle(const int32 TriangleIndex, const FRigidTransform3& Transform, FTriangle& OutTriangle, int32& OutVertexIndex0, int32& OutVertexIndex1, int32& OutVertexIndex2) const
118 {
119 // Convert the triangle index into a cell index and extract the cell row/column
120 // Reverse of FaceIndex0 = CellIndex * 2 + 0|1;
121 const int32 CellIndex = TriangleIndex / 2;
122 const int32 CellRowIndex = CellIndex / (GeomData.NumCols - 1);
123 const int32 CellColIndex = CellIndex % (GeomData.NumCols - 1);
124
125 // Calculate the vertex indices for the 4 cell corners
126 const int32 CellVertexIndex = CellRowIndex * GeomData.NumCols + CellColIndex;
127 const int32 VertexIndex0 = CellVertexIndex;
128 const int32 VertexIndex1 = CellVertexIndex + 1;
129 const int32 VertexIndex2 = CellVertexIndex + GeomData.NumCols;
130 const int32 VertexIndex3 = CellVertexIndex + GeomData.NumCols + 1;
131
132 // Get the vertices
133 const FVec3 Vertex0 = Transform.TransformPosition(GeomData.GetPointScaled(VertexIndex0));
134 const FVec3 Vertex1 = Transform.TransformPosition(GeomData.GetPointScaled(VertexIndex1));
135 const FVec3 Vertex2 = Transform.TransformPosition(GeomData.GetPointScaled(VertexIndex2));
136 const FVec3 Vertex3 = Transform.TransformPosition(GeomData.GetPointScaled(VertexIndex3));
137
138 // Set the output triangle which depends on winding (begative scales) and which of the two triangles in the cell we want
139 const bool bStandardWinding = ((GeomData.Scale.X * GeomData.Scale.Y * GeomData.Scale.Z) >= FReal(0));
140 const bool bIsFirstTriangle = ((TriangleIndex & 1) == 0);
141 if (bIsFirstTriangle)
142 {
143 if (bStandardWinding)
144 {
145 OutTriangle = FTriangle(Vertex0, Vertex1, Vertex3);
146 OutVertexIndex0 = VertexIndex0;
147 OutVertexIndex1 = VertexIndex1;
148 OutVertexIndex2 = VertexIndex3;
149 }
150 else
151 {
152 OutTriangle = FTriangle(Vertex0, Vertex3, Vertex1);
153 OutVertexIndex0 = VertexIndex0;
154 OutVertexIndex1 = VertexIndex3;
155 OutVertexIndex2 = VertexIndex1;
156 }
157 }
158 else
159 {
160 if (bStandardWinding)
161 {
162 OutTriangle = FTriangle(Vertex0, Vertex3, Vertex2);
163 OutVertexIndex0 = VertexIndex0;
164 OutVertexIndex1 = VertexIndex3;
165 OutVertexIndex2 = VertexIndex2;
166 }
167 else
168 {
169 OutTriangle = FTriangle(Vertex0, Vertex2, Vertex3);
170 OutVertexIndex0 = VertexIndex0;
171 OutVertexIndex1 = VertexIndex2;
172 OutVertexIndex2 = VertexIndex3;
173 }
174 }

185 template<typename TriangleVisitor>
186 void VisitTriangles(const FAABB3& QueryBounds, const FRigidTransform3& QueryTransform, const TriangleVisitor& Visitor) const
187 {
188 FBounds2D GridQueryBounds;
189 GridQueryBounds.Min = FVec2(QueryBounds.Min()[0], QueryBounds.Min()[1]);
190 GridQueryBounds.Max = FVec2(QueryBounds.Max()[0], QueryBounds.Max()[1]);
191
192 // @todo(chaos): could we do the 3D bounds test here?
193 TArray<TVec2<int32>> Intersections;
194 GetGridIntersections(GridQueryBounds, Intersections);
195
196 const bool bStandardWinding = ((GeomData.Scale.X * GeomData.Scale.Y * GeomData.Scale.Z) >= FReal(0));
197
198 FVec3 Points[4];
199 FTriangle Triangles[2];
200
201 for (const TVec2<int32>& Cell : Intersections)
202 {
203 const int32 CellVertexIndex = Cell[1] * GeomData.NumCols + Cell[0]; // First vertex in cell
204 const int32 CellIndex = Cell[1] * (GeomData.NumCols - 1) + Cell[0];
205
206 // Check for holes and skip checking if we'll never collide
207 if (GeomData.MaterialIndices.IsValidIndex(CellIndex) && GeomData.MaterialIndices[CellIndex] == TNumericLimits<uint8>::Max())
208 {
209 continue;
210 }
211
212 // The triangle is solid so proceed to test it
213 FAABB3 CellBounds;
214 GeomData.GetPointsAndBoundsScaled(CellVertexIndex, Points, CellBounds);
215
216 if (CellBounds.Intersects(QueryBounds))
217 {
218 // Transform points into the space of the query
219 // @todo(chaos): duplicate work here when we overlap lots of cells. We could generate the transformed verts for the full query once...
220 Points[0] = QueryTransform.TransformPositionNoScale(Points[0]);
221 Points[1] = QueryTransform.TransformPositionNoScale(Points[1]);
222 Points[2] = QueryTransform.TransformPositionNoScale(Points[2]);
223 Points[3] = QueryTransform.TransformPositionNoScale(Points[3]);
224
225 // @todo(chaos): utility function for this
226 const int32 VertexIndex0 = CellVertexIndex;
227 const int32 VertexIndex1 = CellVertexIndex + 1;
228 const int32 VertexIndex2 = CellVertexIndex + GeomData.NumCols;
229 const int32 VertexIndex3 = CellVertexIndex + GeomData.NumCols + 1;
230
231 // Generate contacts if overlapping
232 const int32 FaceIndex0 = CellIndex * 2 + 0;
233 const int32 FaceIndex1 = CellIndex * 2 + 1;
234
235 if (bStandardWinding)
236 {
237 Triangles[0] = FTriangle(Points[0], Points[1], Points[3]);
238 Triangles[1] = FTriangle(Points[0], Points[3], Points[2]);
239 Visitor(Triangles[0], FaceIndex0, VertexIndex0, VertexIndex1, VertexIndex3);
240 Visitor(Triangles[1], FaceIndex1, VertexIndex0, VertexIndex3, VertexIndex2);
241 }
242 else
243 {
244 Triangles[0] = FTriangle(Points[0], Points[3], Points[1]);
245 Triangles[1] = FTriangle(Points[0], Points[2], Points[3]);
246 Visitor(Triangles[0], FaceIndex0, VertexIndex0, VertexIndex3, VertexIndex1);
247 Visitor(Triangles[1], FaceIndex1, VertexIndex0, VertexIndex2, VertexIndex3);
248 }
249 }
250 }

277 // Does the mesh-space QueryBounds overlap the bounds of any cells in the heightfield?
278 // @todo(chaos): we can return as soon as we overlap any cell so maybe use a
279 // custom function rather than GetBoundsScaled. Also we only care about Z overlap.
280 bool IsOverlappingAnyCells(const FAABB3& QueryBounds) const
281 {
282 // Top-level bounds check
283 if (!BoundingBox().Intersects(QueryBounds))
284 {
285 return false;
286 }
287
288 // Find all the cells that overlap in the X/Y plane
289 FVec2 Scale2D(GeomData.Scale[0], GeomData.Scale[1]);
290 const FBounds2D FlatBounds = GetFlatBounds();
291
292 FBounds2D GridQueryBounds;
293 GridQueryBounds.Min = FVec2(QueryBounds.Min()[0], QueryBounds.Min()[1]);
294 GridQueryBounds.Max = FVec2(QueryBounds.Max()[0], QueryBounds.Max()[1]);
295 GridQueryBounds = FBounds2D::FromPoints(FlatBounds.Clamp(GridQueryBounds.Min) / Scale2D, FlatBounds.Clamp(GridQueryBounds.Max) / Scale2D);
296
297 FAABBVectorized QueryBoundsSimd = FAABBVectorized(QueryBounds);
298
299 // We want to capture the first cell (delta == 0) as well
300 TVec2<int32> FirstCell = FlatGrid.Cell(GridQueryBounds.Min);
301 TVec2<int32> LastCell = FlatGrid.Cell(GridQueryBounds.Max);
302
303 FAABBVectorized RegionBounds;
304 GeomData.GetBoundsScaled(FirstCell, LastCell - FirstCell, RegionBounds);
305 return RegionBounds.Intersects(QueryBoundsSimd);
306 }

959
960 bool ClipLine(FVec2& InOutStart, FVec2& InOutEnd) const
961 {
962
963 // Test we don't need to clip at all, quite likely with a heightfield so optimize for it.
964 const bool bStartInside = IsInside(InOutStart);
965 const bool bEndInside = IsInside(InOutEnd);
966 if(bStartInside && bEndInside)
967 {
968 return true;
969 }
970
971 const FVec2 Dir = InOutEnd - InOutStart;
972
973 // Tiny ray not inside so must be outside
974 if(Dir.SizeSquared() < 1e-4)
975 {
976 return false;
977 }
978
979 bool bPerpendicular[2];
980 FVec2 InvDir;
981 for(int Axis = 0; Axis < 2; ++Axis)
982 {
983 bPerpendicular[Axis] = Dir[Axis] == 0;
984 InvDir[Axis] = bPerpendicular[Axis] ? 0 : 1 / Dir[Axis];
985 }
986
987
988
989 if(bStartInside)
990 {
991 const FReal TimeToExit = ComputeTimeToExit(InOutStart,InvDir);
992 InOutEnd = InOutStart + Dir * TimeToExit;
993 return true;
994 }
995
996 if(bEndInside)
997 {
998 const FReal TimeToExit = ComputeTimeToExit(InOutEnd,-InvDir);
999 InOutStart = InOutEnd - Dir * TimeToExit;
1000 return true;
1001 }
1002
1003 //start and end outside, need to see if we even intersect
1004 FReal TimesToEnter[2] = {TNumericLimits<FReal>::Max(),TNumericLimits<FReal>::Max()};
1005 FReal TimesToExit[2] = {TNumericLimits<FReal>::Max(),TNumericLimits<FReal>::Max()};
1006
1007 for(int Axis = 0; Axis < 2; ++Axis)
1008 {
1009 if(bPerpendicular[Axis])
1010 {
1011 if(InOutStart[Axis] >= Min[Axis] && InOutStart[Axis] <= Max[Axis])
1012 {
1013 TimesToEnter[Axis] = 0;
1014 }
1015 }
1016 else
1017 {
1018 if(Dir[Axis] > 0)
1019 {
1020 if(InOutStart[Axis] <= Max[Axis])
1021 {
1022 TimesToEnter[Axis] = FMath::Max<FReal>(Min[Axis] - InOutStart[Axis], 0) * InvDir[Axis];
1023 TimesToExit[Axis] = (Max[Axis] - InOutStart[Axis]) * InvDir[Axis];
1024 }
1025 }
1026 else if(Dir[Axis] < 0)
1027 {
1028 if(InOutStart[Axis] >= Min[Axis])
1029 {
1030 TimesToEnter[Axis] = FMath::Max<FReal>(InOutStart[Axis] - Max[Axis],0) * InvDir[Axis];
1031 TimesToExit[Axis] = (InOutStart[Axis] - Min[Axis]) * InvDir[Axis];
1032 }
1033 }
1034 }
1035 }
1036
1037 const FReal TimeToEnter = FMath::Max(FMath::Abs(TimesToEnter[0]),FMath::Abs(TimesToEnter[1]));
1038 const FReal TimeToExit = FMath::Min(FMath::Abs(TimesToExit[0]),FMath::Abs(TimesToExit[1]));
1039
1040 if(TimeToExit < TimeToEnter)
1041 {
1042 //no intersection
1043 return false;
1044 }
1045
1046 InOutEnd = InOutStart + Dir * TimeToExit;
1047 InOutStart = InOutStart + Dir * TimeToEnter;
1048 return true;
1049 }

1052 //This helper assumes Start is inside the min/max box and uses InvDir to compute how long it takes to exit
1053 FReal ComputeTimeToExit(const FVec2& Start,const FVec2& InvDir) const
1054 {
1055 FReal Times[2] ={TNumericLimits<FReal>::Max(),TNumericLimits<FReal>::Max()};
