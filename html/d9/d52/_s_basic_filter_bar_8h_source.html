851 void SetFilterCheckState(const TSharedPtr<FFilterBase<FilterType>>& InFilter, ECheckBoxState InCheckState)
852 {
853 if (!InFilter || InCheckState == ECheckBoxState::Undetermined)
854 {
855 return;
856 }
857
858 // Check if the filter is already checked.
859 TSharedRef<FFilterBase<FilterType>> Filter = InFilter.ToSharedRef();
860 bool FrontendFilterChecked = IsFrontendFilterInUse(Filter);
861
862 if (InCheckState == ECheckBoxState::Checked && !FrontendFilterChecked)
863 {
864 AddFilterToBar(Filter)->SetEnabled(true); // Pin a filter widget on the UI and activate the filter. Same behaviour as FrontendFilterClicked()
865 }
866 else if (InCheckState == ECheckBoxState::Unchecked && FrontendFilterChecked)
867 {
