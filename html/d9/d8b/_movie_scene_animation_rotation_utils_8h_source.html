13// Euler decomposition order for rotation matching. When matching specific
14// rotation axes, the matched axis should be decomposed first to avoid
15// gimbal lock artifacts on the axes we care about.
16enum class ERotationOrder
17{
18 XYZ, XZY, YXZ, YZX, ZXY, ZYX
19};
20
21// Choose the decomposition order that puts the matched axis first.
22// Yaw (Z) is the most common match axis, so YXZ is the default when matching yaw.
23inline ERotationOrder FindBestRotationOrder(bool bMatchRoll, bool bMatchPitch, bool bMatchYaw)
24{
25 if (bMatchYaw)
26 {
27 return ERotationOrder::YXZ;
28 }
29 if (bMatchPitch)
30 {
31 return ERotationOrder::YZX;
32 }
33 return ERotationOrder::XYZ;
34}

36inline FQuat QuatFromEuler(const FVector& XYZAnglesInDegrees, ERotationOrder RotationOrder)
37{
38 double X = FMath::DegreesToRadians(XYZAnglesInDegrees.X);
39 double Y = FMath::DegreesToRadians(XYZAnglesInDegrees.Y);
40 double Z = FMath::DegreesToRadians(XYZAnglesInDegrees.Z);
41
42 double CosX = FMath::Cos(X * 0.5);
43 double CosY = FMath::Cos(Y * 0.5);
44 double CosZ = FMath::Cos(Z * 0.5);
45
46 double SinX = FMath::Sin(X * 0.5);
47 double SinY = FMath::Sin(Y * 0.5);
48 double SinZ = FMath::Sin(Z * 0.5);
49
50 switch (RotationOrder)
51 {
52 case ERotationOrder::XYZ:
53 return FQuat(SinX * CosY * CosZ - CosX * SinY * SinZ,
54 CosX * SinY * CosZ + SinX * CosY * SinZ,
55 CosX * CosY * SinZ - SinX * SinY * CosZ,
56 CosX * CosY * CosZ + SinX * SinY * SinZ);
57
58 case ERotationOrder::XZY:
59 return FQuat(SinX * CosY * CosZ + CosX * SinY * SinZ,
60 CosX * SinY * CosZ + SinX * CosY * SinZ,
61 CosX * CosY * SinZ - SinX * SinY * CosZ,
62 CosX * CosY * CosZ - SinX * SinY * SinZ);
63
64 case ERotationOrder::YXZ:
65 return FQuat(SinX * CosY * CosZ - CosX * SinY * SinZ,
66 CosX * SinY * CosZ + SinX * CosY * SinZ,
67 CosX * CosY * SinZ + SinX * SinY * CosZ,
68 CosX * CosY * CosZ - SinX * SinY * SinZ);
69
70 case ERotationOrder::YZX:
71 return FQuat(SinX * CosY * CosZ - CosX * SinY * SinZ,
72 CosX * SinY * CosZ - SinX * CosY * SinZ,
73 CosX * CosY * SinZ + SinX * SinY * CosZ,
74 CosX * CosY * CosZ + SinX * SinY * SinZ);
75
76 case ERotationOrder::ZXY:
77 return FQuat(SinX * CosY * CosZ + CosX * SinY * SinZ,
78 CosX * SinY * CosZ - SinX * CosY * SinZ,
79 CosX * CosY * SinZ - SinX * SinY * CosZ,
80 CosX * CosY * CosZ + SinX * SinY * SinZ);
81
82 case ERotationOrder::ZYX:
83 return FQuat(SinX * CosY * CosZ + CosX * SinY * SinZ,
84 CosX * SinY * CosZ - SinX * CosY * SinZ,
85 CosX * CosY * SinZ + SinX * SinY * CosZ,
86 CosX * CosY * CosZ - SinX * SinY * SinZ);
87 }
88
89 return FQuat::Identity;
90}

92inline FVector EulerFromQuat(const FQuat& Rotation, ERotationOrder RotationOrder)
93{
94 double X = Rotation.X;
95 double Y = Rotation.Y;
96 double Z = Rotation.Z;
97 double W = Rotation.W;
98 double X2 = X * 2.0;
99 double Y2 = Y * 2.0;
100 double Z2 = Z * 2.0;
101 double XX2 = X * X2;
102 double XY2 = X * Y2;
103 double XZ2 = X * Z2;
104 double YX2 = Y * X2;
105 double YY2 = Y * Y2;
106 double YZ2 = Y * Z2;
107 double ZX2 = Z * X2;
108 double ZY2 = Z * Y2;
109 double ZZ2 = Z * Z2;
110 double WX2 = W * X2;
111 double WY2 = W * Y2;
112 double WZ2 = W * Z2;
113
114 FVector AxisX, AxisY, AxisZ;
115 AxisX.X = (1.0 - (YY2 + ZZ2));
116 AxisY.X = (XY2 + WZ2);
117 AxisZ.X = (XZ2 - WY2);
118 AxisX.Y = (XY2 - WZ2);
119 AxisY.Y = (1.0 - (XX2 + ZZ2));
120 AxisZ.Y = (YZ2 + WX2);
121 AxisX.Z = (XZ2 + WY2);
122 AxisY.Z = (YZ2 - WX2);
123 AxisZ.Z = (1.0 - (XX2 + YY2));
124
125 FVector Result = FVector::ZeroVector;
126
127 if (RotationOrder == ERotationOrder::XYZ)
128 {
129 Result.Y = FMath::Asin(-FMath::Clamp<double>(AxisZ.X, -1.0, 1.0));
130
131 if (FMath::Abs(AxisZ.X) < 1.0 - SMALL_NUMBER)
132 {
133 Result.X = FMath::Atan2(AxisZ.Y, AxisZ.Z);
134 Result.Z = FMath::Atan2(AxisY.X, AxisX.X);
135 }
136 else
137 {
138 Result.X = 0.0;
139 Result.Z = FMath::Atan2(-AxisX.Y, AxisY.Y);
140 }
141 }
142 else if (RotationOrder == ERotationOrder::XZY)
143 {
144
145 Result.Z = FMath::Asin(FMath::Clamp<double>(AxisY.X, -1.0, 1.0));
146
147 if (FMath::Abs(AxisY.X) < 1.0 - SMALL_NUMBER)
148 {
149 Result.X = FMath::Atan2(-AxisY.Z, AxisY.Y);
150 Result.Y = FMath::Atan2(-AxisZ.X, AxisX.X);
151 }
152 else
153 {
154 Result.X = 0.0;
155 Result.Y = FMath::Atan2(AxisX.Z, AxisZ.Z);
156 }
157 }
158 else if (RotationOrder == ERotationOrder::YXZ)
159 {
160 Result.X = FMath::Asin(FMath::Clamp<double>(AxisZ.Y, -1.0, 1.0));
161
162 if (FMath::Abs(AxisZ.Y) < 1.0 - SMALL_NUMBER)
163 {
164 Result.Y = FMath::Atan2(-AxisZ.X, AxisZ.Z);
165 Result.Z = FMath::Atan2(-AxisX.Y, AxisY.Y);
166 }
167 else
168 {
169 Result.Y = 0.0;
170 Result.Z = FMath::Atan2(AxisY.X, AxisX.X);
171 }
172 }
173 else if (RotationOrder == ERotationOrder::YZX)
174 {
175 Result.Z = FMath::Asin(-FMath::Clamp<double>(AxisX.Y, -1.0, 1.0));
176
177 if (FMath::Abs(AxisX.Y) < 1.0 - SMALL_NUMBER)
178 {
179 Result.X = FMath::Atan2(AxisZ.Y, AxisY.Y);
180 Result.Y = FMath::Atan2(AxisX.Z, AxisX.X);
181 }
182 else
183 {
184 Result.X = FMath::Atan2(-AxisY.Z, AxisZ.Z);
185 Result.Y = 0.0;
186 }
187 }
188 else if (RotationOrder == ERotationOrder::ZXY)
189 {
190 Result.X = FMath::Asin(-FMath::Clamp<double>(AxisY.Z, -1.0, 1.0));
191
192 if (FMath::Abs(AxisY.Z) < 1.0 - SMALL_NUMBER)
193 {
194 Result.Y = FMath::Atan2(AxisX.Z, AxisZ.Z);
195 Result.Z = FMath::Atan2(AxisY.X, AxisY.Y);
196 }
197 else
198 {
199 Result.Y = FMath::Atan2(-AxisZ.X, AxisX.X);
200 Result.Z = 0.0;
201 }
202 }
203 else if (RotationOrder == ERotationOrder::ZYX)
204 {
205 Result.Y = FMath::Asin(FMath::Clamp<double>(AxisX.Z, -1.0, 1.0));
206
207 if (FMath::Abs(AxisX.Z) < 1.0 - SMALL_NUMBER)
208 {
209 Result.X = FMath::Atan2(-AxisY.Z, AxisZ.Z);
210 Result.Z = FMath::Atan2(-AxisX.Y, AxisX.X);
211 }
212 else
213 {
214 Result.X = FMath::Atan2(AxisZ.Y, AxisY.Y);
215 Result.Z = 0.0;
216 }
217 }
218
219 return Result * 180.0 / DOUBLE_PI;
220}
