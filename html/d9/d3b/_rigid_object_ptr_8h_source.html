14 // A Context-Specific Smart Pointer to a physics object.
15 //
16 // This is a wrapper around a smart pointer implementation object like TRigidBodyPtrImpl,
17 // TRigidBodyContainerPtrImpl, etc. It provides operator-> to give it const and non-const pointer
18 // semantics and behaviour.
19 //
20 // May be "null". Check IsValid() (or cast to bool) before dereferencing.
21 //
22 // Cannot be copied or stored - use Handles for that.
23 //
24 // NOTE: Much like a regular pointer, a "valid" pointer may still reference deleted memory if the
25 // object is destroyed after the Handle is Pinned. Do not use pointers, even to check for validity,
26 // after the underlying object has been destroyed. If the underlying object may have been destroyed,
27 // you should re-Pin the Handle to obtain a new pointer.
28 //
29 template <typename PointerImplType>
30 class TRigidObjectPtr
31 {
32 public:
33 using FPointerImpl = PointerImplType;
34 using FContext = FPointerImpl::FContext;
35 using FInterface = FPointerImpl::FInterface;
36 using FHandle = FPointerImpl::FHandle;
37
38 UE_NONCOPYABLE(TRigidObjectPtr);
39
40 TRigidObjectPtr() = default;
41 UE_INTERNAL TRigidObjectPtr(FInterface* InObject)
42 : Impl(InObject)
43 {
44 }
45
46 // Downcasting. Will fail to compile for attempts to cast to non-derived types.
47 // Will generate "invalid" pointer in the runtime when InPtr is not an instance of right type.
48 // Note: Also make sure we don't allow conversion to different context types.
49 template<
50 typename OtherPointerImplType
51 UE_REQUIRES(std::is_base_of_v<typename OtherPointerImplType::FInterface, FInterface>&& std::is_same_v<typename OtherPointerImplType::FContext, FContext>)
52 >
53 UE_INTERNAL TRigidObjectPtr(const TRigidObjectPtr<OtherPointerImplType>& InPtr)
54 : Impl(InPtr.Get())
55 {
56 }
57
58 ~TRigidObjectPtr() = default;
59
60 // Reset the pointer (it will now be Invalid)
61 void Reset()
62 {
63 Impl.Reset();
64 }
65
66 // Has this pointer been initialized with a valid body pointer?
67 // NOTE: This will still return true if the underlying object is destroyed after the Handle is Pinned.
68 // If this is a possibility, the Handle should be re-Pinned.
69 bool IsValid() const
70 {
71 return Impl.IsValid();
72 }
73
74 // Auto cast to bool to support using in if statements
75 // e.g., if (TRigidBodyPtr<FRigidContextGameRW> Body = BodyHandle.Pin(Context)) { ... }
76 operator bool() const
77 {
78 return IsValid();
79 }
80
81 // Extract the handle
82 FHandle ToHandle() const
83 {
84 return Impl.GetHandle();
85 }
86
87 // Cast to a handle
88 operator FHandle() const
89 {
90 return ToHandle();
91 }
92
93 // Const pointer semantics
94 const FPointerImpl* operator->() const
95 {
96 CheckIsValid();
97 return &Impl;
98 }
99
100 // Non-const pointer semantics
101 FPointerImpl* operator->() const requires (FContext::bWriteEnabled)
102 {
103 CheckIsValid();
104 return &Impl;
105 }
106
107 // Extract the object interface (internal use only)
108 UE_INTERNAL FInterface* Get() const
109 {
110 return Impl.Get();
111 }
112
113 UE_INTERNAL void Init(FInterface* InObject)
114 {
115 Impl = FPointerImpl(InObject);
116 }
117
118 private:
119 void CheckIsValid() const
120 {
121 UE_RIGIDPHYSICS_CHECKF(IsValid(), TEXT("Ptr is null. Check IsValid() before dereferencing"));
122 }
123
124 // Mutable to support const TRigidObjectPtr to an non-const object
125 // i.e., this wrapper cannot change, but the pointee is mutabale
126 // if the wrapper was created from a mutable context. This avoids
127 // an undefined behaviour compiler warning.
128 mutable FPointerImpl Impl = nullptr;
129 };
130} // namespace UE::Physics
