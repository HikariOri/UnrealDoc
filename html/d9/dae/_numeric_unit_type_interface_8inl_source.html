20FString TNumericUnitTypeInterface<NumericType>::ToString(const NumericType& Value) const
21{

58template<typename NumericType>
59TOptional<NumericType> TNumericUnitTypeInterface<NumericType>::FromString(const FString& InString, const NumericType& InExistingValue)
60{
61 if (UnderlyingUnits == EUnit::Unspecified)
62 {
63 return TDefaultNumericTypeInterface<NumericType>::FromString(InString, InExistingValue);
64 }
65
66 // Get displayed units from the user, if set, otherwise the fixed units, or finally the underlying units
67 EUnit DefaultUnits = UserDisplayUnits.IsSet()
68 ? UserDisplayUnits.GetValue()
69 : FixedDisplayUnits.IsSet()
70 ? FixedDisplayUnits.GetValue()
71 : UnderlyingUnits;
72
73 // Always parse in as a double, to allow for input of higher-order units with decimal numerals into integral types (eg, inputting 0.5km as 500m)
74 TValueOrError<FNumericUnit<double>, FText> NewValue = FNumericUnit<double>::TryParseExpression( *InString, DefaultUnits, InExistingValue );
75 if (NewValue.IsValid())
76 {
77 // Convert the number into the correct units
78 EUnit SourceUnits = NewValue.GetValue().Units;
79 if (SourceUnits == EUnit::Unspecified)
80 {
81 // Use the user overridden input units if set
82 if (UserDisplayUnits.IsSet())
83 {
84 SourceUnits = UserDisplayUnits.GetValue();
85 }
86 // Use the default supplied input units
87 else if (FixedDisplayUnits.IsSet())
88 {
89 SourceUnits = FixedDisplayUnits.GetValue();
90 }
91 }
92 double ConvertedValue = FUnitConversion::Convert(NewValue.GetValue().Value, SourceUnits, UnderlyingUnits);
93 return FMath::Clamp((NumericType)ConvertedValue, TNumericLimits<NumericType>::Lowest(), TNumericLimits<NumericType>::Max());
94 }
95
96 return TOptional<NumericType>();
97}
