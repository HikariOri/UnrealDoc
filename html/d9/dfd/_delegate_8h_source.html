272 // Given a char array as a template argument, returns a new char array that is the substring after the ::
273 template <TCharArray CharArray>
274 constexpr auto GetTrimmedMemberFunctionName()
275 {
276 // Returns the substring after the ::
277 auto FindFuncName = [](const char* Str)
278 {
279 while (Str[0] != ':' || Str[1] != ':')
280 {
281 ++Str;
282 }
283 return Str + 2;
284 };
285
286 // Inline version of StrLen that is also constexpr
287 auto StrLen = [](const char* Str)
288 {
289 SIZE_T Result = 0;
290 while (*Str != 0)
291 {
292 ++Str;
293 ++Result;
294 }
295 return Result;
296 };
297
298 constexpr const char* FuncName = FindFuncName(CharArray.Str);
299 constexpr SIZE_T FuncNameArrayLen = StrLen(FuncName) + 1;
300 TCharArray<FuncNameArrayLen> Result;
301 for (SIZE_T Index = 0; Index != FuncNameArrayLen; ++Index)
302 {
303 Result.Str[Index] = FuncName[Index];
304 }
305 return Result;
306 }
307
308 template <TCharArray CharArray>
309 constexpr inline auto TrimmedMemberFunctionName = GetTrimmedMemberFunctionName<CharArray>();
310}
