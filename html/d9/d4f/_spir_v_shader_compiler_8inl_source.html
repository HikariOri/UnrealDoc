171static FString GetBindlessUBNameFromHeap(const FString& HeapName)
172{

221static bool HasDerivatives(const FSpirv& Spirv)
222{
223 for (FSpirvConstIterator Iter = Spirv.begin(); Iter != Spirv.end(); ++Iter)
224 {
225 switch (Iter.Opcode())
226 {
227 case SpvOpCapability:
228 {
229 const uint32 Capability = Iter.Operand(1);
230 if ((Capability == SpvCapabilityComputeDerivativeGroupLinearNV) ||
231 (Capability == SpvCapabilityComputeDerivativeGroupQuadsNV))
232 {
233 return true;
234 }
235 }
236 break;
237
238 case SpvOpExtension:
239 case SpvOpEntryPoint:
240 // By the time we've reached extensions/entrypoints, we're done listing capabilities
241 return false;
242
243 default:
244 break;
245 }
246 }
247 return false;
248}

278static void BuildShaderOutput(
279 SpirvShaderCompilerSerializedOutput& SerializedOutput,
280 FShaderCompilerOutput& ShaderOutput,
281 const FSpirvShaderCompilerInternalState& InternalState,
282 const FSpirvReflectBindings& SpirvReflectBindings,
283 const FString& DebugName,
284 TBitArray<>& UsedUniformBufferSlots
285)
286{
287 auto ParseNumber = []<typename T>(const T * Str, bool bEmptyIsZero = false)
288 {
289 check(Str);
290
291 uint32 Num = 0;
292
293 int32 Len = 0;
294 // Find terminating character
295 for (int32 Index = 0; Index < 128; Index++)
296 {
297 if (Str[Index] == 0)
298 {
299 Len = Index;
300 break;
301 }
302 }
303
304 if (Len == 0)
305 {
306 if (bEmptyIsZero)
307 {
308 return 0u;
309 }
310 else
311 {
312 check(0);
313 }
314 }
315
316 // Find offset to integer type
317 int32 Offset = -1;
318 for (int32 Index = 0; Index < Len; Index++)
319 {
320 if (*(Str + Index) >= '0' && *(Str + Index) <= '9')
321 {
322 Offset = Index;
323 break;
324 }
325 }
326
327 // Check if we found a number
328 check(Offset >= 0);
329
330 Str += Offset;
331
332 while (*(Str) && *Str >= '0' && *Str <= '9')
333 {
334 Num = Num * 10 + *Str++ - '0';
335 }
336
337 return Num;
338 };
339
340 const FShaderCompilerInput& ShaderInput = InternalState.Input;
341 const EShaderFrequency Frequency = InternalState.GetShaderFrequency();
342 FVulkanShaderHeader& Header = SerializedOutput.Header;
343
344 Header.SpirvCRC = SerializedOutput.SpirvCRC;
345 Header.RayTracingPayloadType = ShaderInput.Environment.GetCompileArgument(TEXT("RT_PAYLOAD_TYPE"), 0u);
346 Header.RayTracingPayloadSize = ShaderInput.Environment.GetCompileArgument(TEXT("RT_PAYLOAD_MAX_SIZE"), 0u);
347
348 // :todo-jn: Hash entire SPIRV for now, could eventually be removed since we use ShaderKeys
349 Header.CodeHash = FShaderHash::HashBuffer(SerializedOutput.Spirv.Data.GetData(), SerializedOutput.Spirv.GetByteSize());
350
351
352 // Flattens the array dimensions of the interface variable (aka shader attribute), e.g. from float4[2][3] -> float4[6]
353 auto FlattenAttributeArrayDimension = [](const SpvReflectInterfaceVariable& Attribute, uint32 FirstArrayDim = 0)
354 {
355 uint32 FlattenedArrayDim = 1;
356 for (uint32 ArrayDimIndex = FirstArrayDim; ArrayDimIndex < Attribute.array.dims_count; ++ArrayDimIndex)
357 {
358 FlattenedArrayDim *= Attribute.array.dims[ArrayDimIndex];
359 }
360 return FlattenedArrayDim;
361 };
362
363
364 // Only process input attributes for vertex shaders.
365 if (Frequency == SF_Vertex)
366 {
367 static const FString AttributePrefix = TEXT("ATTRIBUTE");
368
369 for (const SpvReflectInterfaceVariable* Attribute : SpirvReflectBindings.InputAttributes)
370 {
371 if (CrossCompiler::FShaderConductorContext::IsIntermediateSpirvOutputVariable(Attribute->name))
372 {
373 continue;
374 }
375
376 if (!Attribute->semantic)
377 {
378 continue;
379 }
380
381 const FString InputAttrName(Attribute->semantic);
382 if (InputAttrName.StartsWith(AttributePrefix))
383 {
384 const uint32 AttributeIndex = ParseNumber(*InputAttrName + AttributePrefix.Len(), /*bEmptyIsZero:*/ true);
385 const uint32 FlattenedArrayDim = FlattenAttributeArrayDimension(*Attribute);
386 for (uint32 Index = 0; Index < FlattenedArrayDim; ++Index)
387 {
388 const uint32 BitIndex = (AttributeIndex + Index);
389 Header.InOutMask |= (1u << BitIndex);
390 }
391 }
392 }
393 }
394
395 // Only process output attributes for pixel shaders.
396 if (Frequency == SF_Pixel)
397 {
398 static const FString TargetPrefix = "SV_Target";
399
400 for (const SpvReflectInterfaceVariable* Attribute : SpirvReflectBindings.OutputAttributes)
401 {
402 // Only depth writes for pixel shaders must be tracked.
403 if (Attribute->built_in == SpvBuiltInFragDepth)
404 {
405 const uint32 BitIndex = (CrossCompiler::FShaderBindingInOutMask::DepthStencilMaskIndex);
406 Header.InOutMask |= (1u << BitIndex);
407 }
408 else
409 {
410 // Only targets for pixel shaders must be tracked.
411 const FString OutputAttrName(Attribute->semantic);
412 if (OutputAttrName.StartsWith(TargetPrefix))
413 {
414 const uint32 TargetIndex = ParseNumber(*OutputAttrName + TargetPrefix.Len(), /*bEmptyIsZero:*/ true);
415
416 const uint32 FlattenedArrayDim = FlattenAttributeArrayDimension(*Attribute);
417 for (uint32 Index = 0; Index < FlattenedArrayDim; ++Index)
418 {
419 const uint32 BitIndex = (TargetIndex + Index);
420 Header.InOutMask |= (1u << BitIndex);
421 }
422 }
423 }
424 }
425 }
426
427 // Build the SRT for this shader.
428 {
429 checkf(Header.UniformBufferInfos.Num() == (UsedUniformBufferSlots.FindLast(true) + 1),
430 TEXT("Some of the Uniform Buffers containing constants weren't flag as in-use. This might lead to duplicate indices being assigned."));
431
432 FShaderCompilerResourceTable CompilerSRT;
433 if (!BuildResourceTableMapping(ShaderInput.Environment.ResourceTableMap, ShaderInput.Environment.UniformBufferMap, UsedUniformBufferSlots, ShaderOutput.ParameterMap, CompilerSRT))
434 {
435 ShaderOutput.AddErrorf(TEXT("Internal error on BuildResourceTableMapping."));
436 return;
437 }
438 UE::ShaderCompilerCommon::BuildShaderResourceTable(CompilerSRT, SerializedOutput.ShaderResourceTable);
439
440 // The previous step also added resource only UBs starting at the first free slot in UsedUniformBufferSlots
441 // We need to add the hashes for their layouts in the same slots of our UniformBufferInfos in the header
442 {
443 const int32 NumUBSlots = CompilerSRT.MaxBoundResourceTable + 1;
444 if (Header.UniformBufferInfos.Num() < NumUBSlots)
445 {
446 Header.UniformBufferInfos.SetNumZeroed(NumUBSlots);
447 }
448
449 TArray<FStringView> UBParameterNames = ShaderOutput.ParameterMap.GetAllParameterNamesOfType(EShaderParameterType::UniformBuffer);
450 for (const FStringView& ParameterName : UBParameterNames)
451 {
452 TOptional<FParameterAllocation> Allocation = ShaderOutput.ParameterMap.FindParameterAllocation(ParameterName);
453 check(Allocation.IsSet());
454 const uint32 UniformBufferIndex = Allocation.GetValue().BufferIndex;
455
456 FVulkanShaderHeader::FUniformBufferInfo& UniformBufferInfo = Header.UniformBufferInfos[UniformBufferIndex];
457 UniformBufferInfo.bHasResources = 1;
458
459 const bool bIsRootParamStructure = (ParameterName == FShaderParametersMetadata::kRootUniformBufferBindingName) && ShaderInput.RootParametersStructure;
460 if (bIsRootParamStructure)
461 {
462 check(UniformBufferIndex == FShaderParametersMetadata::kRootCBufferBindingIndex);
463 const uint32 UBLayoutHash = CompilerSRT.ResourceTableLayoutHashes[UniformBufferIndex];
464
465 checkf(!UBLayoutHash || (UBLayoutHash == ShaderInput.RootParametersStructure->GetLayoutHash()),
466 TEXT("Resource table layout hash for RootParametersStructure (0x%08X) should be unset (0x0) or identical to shader input (0x%08X)!"),
467 UBLayoutHash, ShaderInput.RootParametersStructure->GetLayoutHash());
468
469 CompilerSRT.ResourceTableLayoutHashes[UniformBufferIndex] = ShaderInput.RootParametersStructure->GetLayoutHash();
470 }
471 else
472 {
473 const uint32 UBLayoutHash = CompilerSRT.ResourceTableLayoutHashes[UniformBufferIndex];
474 checkf(!UniformBufferInfo.LayoutHash || (UniformBufferInfo.LayoutHash == UBLayoutHash),
475 TEXT("Existing layout hash (0x%08X) should be unset (resource only UB) or identical to resource table (0x%08X)!"),
476 UniformBufferInfo.LayoutHash, UBLayoutHash);
477 UniformBufferInfo.LayoutHash = UBLayoutHash;
478 }
479 }
480 }
481 }
482
483 ShaderOutput.bSucceeded = true;
484
485 // guard disassembly of SPIRV code on bExtractShaderSource setting since presumably this isn't that cheap.
486 // this roughly will maintain existing behaviour, except the debug usf will be this version of the code
487 // instead of the output of preprocessing if this setting is enabled (which is probably fine since this is only
488 // ever set in editor)
489 if (ShaderInput.ExtraSettings.bExtractShaderSource)
490 {
491 TArray<ANSICHAR> AssemblyText;
492 if (CrossCompiler::FShaderConductorContext::Disassemble(CrossCompiler::EShaderConductorIR::Spirv, SerializedOutput.Spirv.GetByteData(), SerializedOutput.Spirv.GetByteSize(), AssemblyText))
493 {
494 ShaderOutput.ModifiedShaderSource = FString(AssemblyText.GetData());
495 }
496 }
497 if (ShaderInput.ExtraSettings.OfflineCompilerPath.Len() > 0)
498 {
499 if (InternalState.SupportsOfflineCompiler())
500 {
501 CompileShaderOffline(ShaderInput, ShaderOutput, (const ANSICHAR*)SerializedOutput.Spirv.GetByteData(), SerializedOutput.Spirv.GetByteSize(), true, SerializedOutput.SpirvEntryPointName);
502 }
503 }
504
505 // Ray generation shaders rely on a different binding model that aren't compatible with global uniform buffers.
506 if (!InternalState.IsRayTracingShader())
507 {
508 CullGlobalUniformBuffers(ShaderInput.Environment.UniformBufferMap, ShaderOutput.ParameterMap);
509 }
510
511#if VULKAN_ENABLE_BINDING_DEBUG_NAMES
512 Header.DebugName = DebugName;
513#else
514 if (ShaderInput.Environment.CompilerFlags.Contains(CFLAG_ExtraShaderData))
515 {
516 Header.DebugName = ShaderInput.GenerateShaderName();
517 }
518#endif
519}

524static void GatherSpirvReflectionBindings(
525 spv_reflect::ShaderModule& Reflection,
526 FSpirvReflectBindings& OutBindings,
527 TSet<FString>& OutBindlessUB,
528 const FSpirvShaderCompilerInternalState& InternalState)
529{
530 // Change descriptor set numbers
531 TArray<SpvReflectDescriptorSet*> DescriptorSets;
532 uint32 NumDescriptorSets = 0;
533
534 // If bindless is supported, then offset the descriptor set to fit the bindless heaps at the beginning
535 const EShaderFrequency ShaderFrequency = InternalState.GetShaderFrequency();
536 const uint32 StageIndex = (uint32)ShaderStage::GetStageForFrequency(ShaderFrequency);
537 const uint32 DescSetNo = InternalState.bSupportsBindless ? VulkanBindless::MaxNumSets + StageIndex : StageIndex;
538
539 SpvReflectResult SpvResult = Reflection.EnumerateDescriptorSets(&NumDescriptorSets, nullptr);
540 check(SpvResult == SPV_REFLECT_RESULT_SUCCESS);
541 if (NumDescriptorSets > 0)
542 {
543 DescriptorSets.SetNum(NumDescriptorSets);
544 SpvResult = Reflection.EnumerateDescriptorSets(&NumDescriptorSets, DescriptorSets.GetData());
545 check(SpvResult == SPV_REFLECT_RESULT_SUCCESS);
546
547 for (const SpvReflectDescriptorSet* DescSet : DescriptorSets)
548 {
549 Reflection.ChangeDescriptorSetNumber(DescSet, DescSetNo);
550 }
551 }
552
553 OutBindings.GatherInputAttributes(Reflection);
554 OutBindings.GatherOutputAttributes(Reflection);
555 OutBindings.GatherDescriptorBindings(Reflection);
556
557 // Storage buffers always occupy a UAV binding slot, so move all SBufferSRVs into the SBufferUAVs array
558 OutBindings.SBufferUAVs.Append(OutBindings.SBufferSRVs);
559 OutBindings.SBufferSRVs.Empty();
560
561 // Change indices of input attributes by their name suffix. Only in the vertex shader stage, "ATTRIBUTE" semantics have a special meaning for shader attributes.
562 if (ShaderFrequency == SF_Vertex)
563 {
564 OutBindings.AssignInputAttributeLocationsBySemanticIndex(Reflection, CrossCompiler::FShaderConductorContext::GetIdentifierTable().InputAttribute);
565 }
566
567 // Patch resource heaps descriptor set numbers
568 if (InternalState.bSupportsBindless)
569 {
570 // Move the bindless heap to its dedicated descriptor set and remove it from our regular binding arrays
571 auto MoveBindlessHeaps = [&](TArray<SpvReflectDescriptorBinding*>& BindingArray, const TCHAR* HeapPrefix, uint32 BinldessDescSetNo, int32& InOutNumBindlessEntries)
572 {
573 InOutNumBindlessEntries += BindingArray.Num();
574
575 for (int32 Index = BindingArray.Num() - 1; Index >= 0; --Index)
576 {
577 const SpvReflectDescriptorBinding* pBinding = BindingArray[Index];
578 const FString BindingName(pBinding->name);
579 if (BindingName.StartsWith(HeapPrefix))
580 {
581 const uint32 Binding = 0; // single bindless heap per descriptor set
582 Reflection.ChangeDescriptorBindingNumbers(pBinding, Binding, BinldessDescSetNo);
583 BindingArray.RemoveAtSwap(Index);
584 }
585 }
586 };
587
588 // Remove sampler heaps from binding arrays
589 MoveBindlessHeaps(OutBindings.Samplers, FShaderParameterParser::kBindlessSamplerArrayPrefix, VulkanBindless::BindlessSamplerSet, OutBindings.NumBindlessSamplers);
590
591 // Remove resource heaps from binding arrays
592 MoveBindlessHeaps(OutBindings.SBufferUAVs, FShaderParameterParser::kBindlessUAVArrayPrefix, VulkanBindless::BindlessStorageBufferSet, OutBindings.NumBindlessResources);
593 MoveBindlessHeaps(OutBindings.SBufferUAVs, FShaderParameterParser::kBindlessSRVArrayPrefix, VulkanBindless::BindlessStorageBufferSet, OutBindings.NumBindlessResources); // try with both prefixes, they were merged earlier
594 MoveBindlessHeaps(OutBindings.TextureSRVs, FShaderParameterParser::kBindlessSRVArrayPrefix, VulkanBindless::BindlessSampledImageSet, OutBindings.NumBindlessResources);
595 MoveBindlessHeaps(OutBindings.TextureUAVs, FShaderParameterParser::kBindlessUAVArrayPrefix, VulkanBindless::BindlessStorageImageSet, OutBindings.NumBindlessResources);
596 MoveBindlessHeaps(OutBindings.TextureUAVs, FShaderParameterParser::kBindlessSRVArrayPrefix, VulkanBindless::BindlessStorageImageSet, OutBindings.NumBindlessResources); // try with both prefixes, R64 SRV textures are read as storage images
597 MoveBindlessHeaps(OutBindings.TBufferSRVs, FShaderParameterParser::kBindlessSRVArrayPrefix, VulkanBindless::BindlessUniformTexelBufferSet, OutBindings.NumBindlessResources);
598 MoveBindlessHeaps(OutBindings.TBufferUAVs, FShaderParameterParser::kBindlessUAVArrayPrefix, VulkanBindless::BindlessStorageTexelBufferSet, OutBindings.NumBindlessResources);
599 MoveBindlessHeaps(OutBindings.AccelerationStructures, FShaderParameterParser::kBindlessSRVArrayPrefix, VulkanBindless::BindlessAccelerationStructureSet, OutBindings.NumBindlessResources);
600 MoveBindlessHeaps(OutBindings.UniformBuffers, FShaderParameterParser::kBindlessCBVArrayPrefix, VulkanBindless::BindlessUniformBufferSet, OutBindings.NumBindlessResources);
601
602 // Move uniform buffers to the correct set
603 {
604 const uint32 BindingOffset = (StageIndex * VulkanBindless::MaxUniformBuffersPerStage);
605 for (int32 Index = OutBindings.UniformBuffers.Num() - 1; Index >= 0; --Index)
606 {
607 const SpvReflectDescriptorBinding* pBinding = OutBindings.UniformBuffers[Index];
608 const FString BindingName(pBinding->name);
609 if (BindingName.StartsWith(kBindlessCBPrefix))
610 {
611 check(InternalState.bUseBindlessUniformBuffer || InternalState.bUseStaticUniformBufferBindings);
612 Reflection.ChangeDescriptorBindingNumbers(pBinding, 0, VulkanBindless::BindlessUniformBufferSet);
613 const FString BindlessUBName = GetBindlessUBNameFromHeap(BindingName);
614 if (!InternalState.PushConstantUBs.Contains(BindlessUBName))
615 {
616 checkf(InternalState.AllBindlessUBs.Contains(BindlessUBName), TEXT("Bindless Uniform Buffer was found in SPIRV but not tracked in internal state"));
617 OutBindlessUB.Add(BindlessUBName);
618 }
619 OutBindings.UniformBuffers.RemoveAtSwap(Index);
620 }
621 else
622 {
623 Reflection.ChangeDescriptorBindingNumbers(pBinding, BindingOffset + pBinding->binding, VulkanBindless::BindlessSingleUseUniformBufferSet);
624 }
625 }
626
627 checkf(OutBindings.UniformBuffers.Num() <= VulkanBindless::MaxUniformBuffersPerStage, TEXT("Too many UB bindings in shader."));
628 }
629 }

632static uint32 CalculateSpirvInstructionCount(FSpirv& Spirv)
633{
634 // Count instructions inside functions
635 bool bInsideFunction = false;
636 uint32 ApproxInstructionCount = 0;
637 for (FSpirvConstIterator Iter = Spirv.cbegin(); Iter != Spirv.cend(); ++Iter)
638 {
639 switch (Iter.Opcode())
640 {
641
642 case SpvOpFunction:
643 {
644 check(!bInsideFunction);
645 bInsideFunction = true;
646 }
647 break;
648
649 case SpvOpFunctionEnd:
650 {
651 check(bInsideFunction);
652 bInsideFunction = false;
653 }
654 break;
655
656 case SpvOpLabel:
657 case SpvOpAccessChain:
658 case SpvOpSelectionMerge:
659 case SpvOpCompositeConstruct:
660 case SpvOpCompositeInsert:
661 case SpvOpCompositeExtract:
662 // Skip a few ops that show up often but don't result in much work on their own
663 break;
664
665 default:
666 {
667 if (bInsideFunction)
668 {
669 ++ApproxInstructionCount;
670 }
671 }
672 break;
673
674 }
675 }
676 check(!bInsideFunction);
677
678 return ApproxInstructionCount;

764 TEXT("Bindless shaders should only have uniform buffers."));

766 const FString ResourceName(Binding->name);

793 case VK_DESCRIPTOR_TYPE_STORAGE_BUFFER:
794 case VK_DESCRIPTOR_TYPE_STORAGE_IMAGE:
795 HandleReflectedShaderUAV(ResourceName, BindingTypeCount, ReflectionSlot, 1, Output);

800 case VK_DESCRIPTOR_TYPE_UNIFORM_TEXEL_BUFFER:

809 if (GlobalSamplerType == FVulkanShaderHeader::EGlobalSamplerType::Invalid)
810 {
811 HandleReflectedShaderSampler(ResourceName, ReflectionSlot, Output);

814 {
815 FVulkanShaderHeader::FGlobalSamplerInfo& GlobalSamplerInfo = SerializedOutput.Header.GlobalSamplerInfos.AddZeroed_GetRef();

869 );

875 }

878 check(BindingTypeCount == ReflectionSlot);

969 }
970
971 Output.Target = InternalState.Input.Target;
972
973 // Overwrite updated SPIRV code
974 SerializedOutput.Spirv.Data = TArray<uint32>(Reflection.GetCode(), Reflection.GetCodeSize() / 4);
975
976 // After we've copied the SPIRV module's copy of the SPIRV, use it to evaluate specialization constants
977 SerializedOutput.Header.SpecializationConstantData = UE::ShaderCompilerCommon::ProcessSpirvSpecializationConstants(InternalState.Input, CompilerContext, Reflection, Bindings, Output);
978
979 // We have to strip out most debug instructions (except OpName) for Vulkan mobile
980 if (InternalState.ShouldStripReflect())
981 {
982 const char* OptArgs[] = { "--strip-reflect", "-O"};

984 {
985 Output.AddErrorf(TEXT("Failed to strip debug instructions from SPIR-V module"));

987 }
988 }
989
990 // For Android run an additional pass to patch spirv to be compatible across drivers
991 if (InternalState.IsAndroid())
992 {
993 const char* OptArgs[] = {
994 "--android-driver-patch",
995 // FORT-733360: Some Adreno drivers have bugs for interpolators, which are arrays,
996 // hence we need to get rid of them.
997 "--adv-interface-variable-scalar-replacement=skip-matrices"
998 };
999 if (!CompilerContext.OptimizeSpirv(SerializedOutput.Spirv.Data, OptArgs, UE_ARRAY_COUNT(OptArgs)))
1000 {
1001 Output.AddErrorf(TEXT("Failed to apply driver patches for Android"));
1002 return false;
1003 }
1004 }
1005
1006 // :todo-jn: We don't store the CRC of each member of the hit group, leave the entrypoint untouched on the extra modules
1007 if (InternalState.HasMultipleEntryPoints() && (InternalState.HitGroupShaderType != FSpirvShaderCompilerInternalState::EHitGroupShaderType::ClosestHit))
1008 {
1009 SerializedOutput.SpirvEntryPointName = "main_00000000_00000000";

1011 else
1012 {
1013 SerializedOutput.SpirvEntryPointName = PatchSpirvEntryPointWithCRC(SerializedOutput.Spirv, SerializedOutput.SpirvCRC);
1014 }
1015
1016 Output.NumInstructions = CalculateSpirvInstructionCount(SerializedOutput.Spirv);
1017
1018 BuildShaderOutput(
1019 SerializedOutput,
1020 Output,
1021 InternalState,
1022 Bindings,
1023 InternalState.GetDebugName(),
1024 UsedUniformBufferSlots
1025 );
1026
1027 if (InternalState.bDebugDump)
1028 {
1029 FString SPVExt(InternalState.GetSPVExtension());

1031
1032 // Write meta data to debug output file and write SPIR-V dump in binary and text form
1033 DumpDebugShaderBinary(InternalState.Input, SerializedOutput.Spirv.GetByteData(), SerializedOutput.Spirv.GetByteSize(), SPVExt);
1034 DumpDebugShaderDisassembledSpirv(InternalState.Input, SerializedOutput.Spirv.GetByteData(), SerializedOutput.Spirv.GetByteSize(), SPVASMExt);
1035 }
1036
1037 return true;

1040// Replaces OpImageFetch with OpImageRead for 64bit samplers
1041static void Patch64bitSamplers(FSpirv& Spirv)
1042{
1043 uint32_t ULongSampledTypeId = 0;
1044 uint32_t LongSampledTypeId = 0;
1045
1046 TArray<uint32_t, TInlineAllocator<2>> ImageTypeIDs;
1047 TArray<uint32_t, TInlineAllocator<2>> LoadedIDs;
1048
1049
1050 // Count instructions inside functions
1051 for (FSpirvIterator Iter = Spirv.begin(); Iter != Spirv.end(); ++Iter)
1052 {
1053 switch (Iter.Opcode())
1054 {
1055
1056 case SpvOpTypeInt:
1057 {
1058 // Operands:
1059 // 1 - Result Id
1060 // 2 - Width specifies how many bits wide the type is
1061 // 3 - Signedness: 0 indicates unsigned
1062
1063 const uint32_t IntWidth = Iter.Operand(2);
1064 if (IntWidth == 64)
1065 {
1066 const uint32_t IntSignedness = Iter.Operand(3);
1067 if (IntSignedness == 1)
1068 {
1069 check(LongSampledTypeId == 0);
1070 LongSampledTypeId = Iter.Operand(1);
1071 }
1072 else
1073 {
1074 check(ULongSampledTypeId == 0);
1075 ULongSampledTypeId = Iter.Operand(1);
1076 }
1077 }
1078 }
1079 break;
1080
1081 case SpvOpTypeImage:
1082 {
1083 // Operands:
1084 // 1 - Result Id
1085 // 2 - Sampled Type is the type of the components that result from sampling or reading from this image type
1086 // 3 - Dim is the image dimensionality (Dim).
1087 // 4 - Depth : 0 indicates not a depth image, 1 indicates a depth image, 2 means no indication as to whether this is a depth or non-depth image
1088 // 5 - Arrayed : 0 indicates non-arrayed content, 1 indicates arrayed content
1089 // 6 - MS : 0 indicates single-sampled content, 1 indicates multisampled content
1090 // 7 - Sampled : 0 indicates this is only known at run time, not at compile time, 1 indicates used with sampler, 2 indicates used without a sampler (a storage image)
1091 // 8 - Image Format
1092
1093 if ((Iter.Operand(7) == 1) && (Iter.Operand(6) == 0) && (Iter.Operand(5) == 0))
1094 {
1095 // Patch the node info and the SPIRV
1096 const uint32_t SampledTypeId = Iter.Operand(2);
1097 const uint32_t WithoutSampler = 2;
1098 if (SampledTypeId == LongSampledTypeId)
1099 {
1100 uint32* CurrentOpPtr = *Iter;
1101 CurrentOpPtr[7] = WithoutSampler;
1102 CurrentOpPtr[8] = (uint32_t)SpvImageFormatR64i;
1103 ImageTypeIDs.Add(Iter.Operand(1));
1104 }
1105 else if (SampledTypeId == ULongSampledTypeId)
1106 {
1107 uint32* CurrentOpPtr = *Iter;
1108 CurrentOpPtr[7] = WithoutSampler;
1109 CurrentOpPtr[8] = (uint32_t)SpvImageFormatR64ui;
1110 ImageTypeIDs.Add(Iter.Operand(1));
1111 }
1112 }
1113 }
1114 break;
1115
1116 case SpvOpLoad:
1117 {
1118 // Operands:
1119 // 1 - Result Type Id
1120 // 2 - Result Id
1121 // 3 - Pointer
1122
1123 // Find loaded images of this type
1124 if (ImageTypeIDs.Find(Iter.Operand(1)) != INDEX_NONE)
1125 {
1126 LoadedIDs.Add(Iter.Operand(2));
1127 }
1128 }
1129 break;
1130
1131 case SpvOpImageFetch:
1132 {
1133 // Operands:
1134 // 1 - Result Type Id
1135 // 2 - Result Id
1136 // 3 - Image Id
1137 // 4 - Coordinate
1138 // 5 - Image Operands
1139
1140 // If this is one of the modified images, patch the node and the SPIRV.
1141 if (LoadedIDs.Find(Iter.Operand(3)) != INDEX_NONE)
1142 {
1143 const uint32_t OldWordCount = Iter.WordCount();
1144 const uint32_t NewWordCount = 5;
1145 check(OldWordCount >= NewWordCount);
1146 const uint32_t EncodedOpImageRead = (NewWordCount << 16) | ((uint32_t)SpvOpImageRead & 0xFFFF);
1147 uint32* CurrentOpPtr = *Iter;
1148 (*CurrentOpPtr) = EncodedOpImageRead;
1149
1150 // Remove unsupported image operands (mostly force LOD 0)
1151 const uint32_t NopWordCount = 1;
1152 const uint32_t EncodedOpNop = (NopWordCount << 16) | ((uint32_t)SpvOpNop & 0xFFFF);
1153 for (uint32_t ImageOperandIndex = NewWordCount; ImageOperandIndex < OldWordCount; ++ImageOperandIndex)
1154 {
1155 CurrentOpPtr[ImageOperandIndex] = EncodedOpNop;
1156 }
1157 }
1158 }
1159 break;
1160
1161 default:
1162 break;
1163 }
1164 }
1165}

1222// Quick and dirty way to get the location of the entrypoint in the source

1224static FString ParseEntrypointDecl(FShaderSource::FViewType PreprocessedShader, FStringView Entrypoint)
1225{
1226 FShaderSource::FStringType EntrypointConverted(Entrypoint);
1227 auto SkipWhitespace = [&](int32& Index)
1228 {
1229 while (FChar::IsWhitespace(PreprocessedShader[Index]))
1230 {
1231 ++Index;
1232 }
1233 };
1234
1235 auto EraseDebugLines = [](FString& EntryPointDecl)
1236 {
1237 int32 HashIndex;
1238 while (EntryPointDecl.FindChar(TEXT('#'), HashIndex))
1239 {
1240 while ((HashIndex < EntryPointDecl.Len()) && (!FChar::IsLinebreak(EntryPointDecl[HashIndex])))
1241 {
1242 EntryPointDecl[HashIndex] = TEXT(' ');
1243 ++HashIndex;
1244 }
1245 }
1246 };

1250 // Go through all the case sensitive matches in the source
1251 int32 EntrypointIndex = PreprocessedShader.Find(EntrypointConverted);
1252 check(EntrypointIndex != INDEX_NONE);
1253 while (EntrypointIndex != INDEX_NONE)
1254 {
1255 // This should be the beginning of a new word
1256 if ((EntrypointIndex == 0) || !FChar::IsWhitespace(PreprocessedShader[EntrypointIndex - 1]))
1257 {
1258 EntrypointIndex = PreprocessedShader.Find(EntrypointConverted, EntrypointIndex + 1);
1259 continue;
1260 }
1261
1262 // The next thing after the entrypoint should its parameters
1263 // White space is allowed, so skip any that is found
1264
1265 int32 ParamsStart = EntrypointIndex + Entrypoint.Len();
1266 SkipWhitespace(ParamsStart);
1267 if (PreprocessedShader[ParamsStart] != '(')
1268 {
1269 EntrypointIndex = PreprocessedShader.Find(EntrypointConverted, ParamsStart);
1270 continue;
1271 }
1272
1273 int32 ParamsEnd = PreprocessedShader.Find(ANSITEXTVIEW(")"), ParamsStart + 1);
1274 check(ParamsEnd != INDEX_NONE);
1275 if (ParamsEnd == INDEX_NONE)
1276 {
1277 // Suspicious
1278 EntrypointIndex = PreprocessedShader.Find(EntrypointConverted, ParamsStart);
1279 continue;
1280 }
1281
1282 // Make sure to grab everything up to the function content
1283
1284 int32 DeclEnd = ParamsEnd + 1;
1285 while (PreprocessedShader[DeclEnd] != '{' && (PreprocessedShader[DeclEnd] != ';'))
1286 {
1287 ++DeclEnd;
1288 }
1289 if (PreprocessedShader[DeclEnd] != '{')
1290 {
1291 EntrypointIndex = PreprocessedShader.Find(EntrypointConverted, DeclEnd);
1292 continue;
1293 }
1294
1295 // Now back up to pick up the return value, the attributes and everything else that can come with it, like "[numthreads(1,1,1)]"
1296
1297 int32 DeclBegin = EntrypointIndex - 1;
1298 while ( (DeclBegin > 0) && (PreprocessedShader[DeclBegin] != ';') && (PreprocessedShader[DeclBegin] != '}'))
1299 {
1300 --DeclBegin;
1301 }
1302 ++DeclBegin;
1303
1304 EntryPointDecl = FString::ConstructFromPtrSize(&PreprocessedShader[DeclBegin], DeclEnd - DeclBegin);
1305 EraseDebugLines(EntryPointDecl);
1306 EntryPointDecl.TrimStartAndEndInline();
1307 break;
1308 }
1309
1310 return EntryPointDecl;
1311}

1313static uint8 ParseWaveSize(
1314 const FSpirvShaderCompilerInternalState& InternalState,
1315 FShaderSource::FViewType PreprocessedShader
1316 )
1317{
1318 uint8 WaveSize = 0;
1319 if (!InternalState.IsRayTracingShader())
1320 {
1321 const FString EntrypointDecl = ParseEntrypointDecl(PreprocessedShader, InternalState.GetEntryPointName());
1322
1323 const FString WaveSizeMacro(TEXT("VULKAN_WAVESIZE("));
1324 int32 WaveSizeIndex = EntrypointDecl.Find(*WaveSizeMacro, ESearchCase::CaseSensitive);
1325 while (WaveSizeIndex != INDEX_NONE)
1326 {
1327 const int32 StartNumber = WaveSizeIndex + WaveSizeMacro.Len();
1328 const int32 EndNumber = EntrypointDecl.Find(TEXT(")"), ESearchCase::CaseSensitive, ESearchDir::FromStart, StartNumber);
1329 check(EndNumber != INDEX_NONE);
1330
1331 FString WaveSizeValue = FString::ConstructFromPtrSize(&EntrypointDecl[StartNumber], EndNumber - StartNumber);
1332 WaveSizeValue.RemoveSpacesInline();
1333 if (WaveSizeValue != TEXT("N")) // skip the macro decl
1334 {
1335 float FloatResult = 0.0;
1336 if (FMath::Eval(WaveSizeValue, FloatResult))
1337 {
1338 checkf((FloatResult >= 0.0f) && (FloatResult < (float)MAX_uint8), TEXT("Specified wave size is too large for 8bit uint!"));
1339 WaveSize = static_cast<uint8>(FloatResult);
1340
1341 }
1342 else
1343 {
1344 check(WaveSizeValue.IsNumeric());
1345 const int32 ConvertedWaveSize = FCString::Atoi(*WaveSizeValue);
1346 checkf((ConvertedWaveSize > 0) && (ConvertedWaveSize < MAX_uint8), TEXT("Specified wave size is too large for 8bit uint!"));
1347 WaveSize = (uint8)ConvertedWaveSize;
1348 }
1349 break;
1350 }
1351
1352 WaveSizeIndex = EntrypointDecl.Find(*WaveSizeMacro, ESearchCase::CaseSensitive, ESearchDir::FromStart, EndNumber);
1353 }
1354 }
1355
1356 // Take note of preferred wave size flag if none was specified in HLSL
1357 if ((WaveSize == 0) && InternalState.Input.Environment.CompilerFlags.Contains(CFLAG_Wave32))
1358 {
1359 WaveSize = 32;
1360 }
1361
1362 return WaveSize;
1363}

1365static bool CompileWithShaderConductor(
1366 const FSpirvShaderCompilerInternalState& InternalState,
1367 FShaderSource::FViewType PreprocessedShader,
1368 SpirvShaderCompilerSerializedOutput& SerializedOutput,
1369 FShaderCompilerOutput& Output
1370)
1371{
1372 const FShaderCompilerInput& Input = InternalState.Input;
1373
1374 CrossCompiler::FShaderConductorContext CompilerContext;
1375
1376 // Inject additional macro definitions to circumvent missing features: external textures
1377 FShaderCompilerDefinitions AdditionalDefines;
1378
1379 TArray<FString> ExtraDxcArgs;
1380 if (InternalState.IsSM6())
1381 {
1382 ExtraDxcArgs.Add(TEXT("-fvk-allow-rwstructuredbuffer-arrays"));
1383 }
1384
1385 // Fix issues when reading matrices directly for ByteAddrBuffer
1386 // By default the compiler will emit column-major loads and this flag makes sure to revert to the original behavior of row-major.
1387 ExtraDxcArgs.Add(TEXT("-fspv-use-legacy-buffer-matrix-order"));
1388
1389 // Load shader source into compiler context
1390 CompilerContext.LoadSource(PreprocessedShader, Input.VirtualSourceFilePath, InternalState.GetEntryPointName(), InternalState.GetShaderFrequency(), &AdditionalDefines, &ExtraDxcArgs);
1391
1392 // Initialize compilation options for ShaderConductor
1393 CrossCompiler::FShaderConductorOptions Options(Input.Environment.CompilerFlags);
1394 Options.TargetEnvironment = InternalState.GetMinimumTargetEnvironment();
1395
1396 // VK_EXT_scalar_block_layout is required by raytracing and by Nanite (so expect it to be present in SM6/Vulkan_1_3)
1397 Options.bDisableScalarBlockLayout = !(InternalState.IsRayTracingShader() || InternalState.IsSM6());
1398
1399 if (InternalState.IsRayTracingShader() || InternalState.IsSM6())
1400 {
1401 // Use SM 6.6 as the baseline for Vulkan SM6 shaders
1402 Options.ShaderModel.Major = 6;
1403 Options.ShaderModel.Minor = 6;
1404 }
1405
1406 if (Input.Environment.CompilerFlags.Contains(CFLAG_AllowRealTypes))
1407 {
1408 Options.bEnable16bitTypes = true;
1409 }
1410
1411 if (InternalState.bDebugDump)
1412 {
1413 SpirvCreateDXCCompileBatchFiles(CompilerContext, InternalState, Options);
1414 }
1415
1416 // Before the shader rewritter removes all traces of it, pull any WAVESIZE directives from the shader source
1417 SerializedOutput.Header.WaveSize = ParseWaveSize(InternalState, PreprocessedShader);
1418
1419 // Compile HLSL source to SPIR-V binary
1420 if (!CompilerContext.CompileHlslToSpirv(Options, SerializedOutput.Spirv.Data))
1421 {
1422 CompilerContext.FlushErrors(Output.Errors);

1424 }
1425
1426 // If this shader samples R64 image formats, they need to get converted to STORAGE_IMAGE
1427 // todo-jnmo: Scope this with a CFLAG if it affects compilation times
1428 Patch64bitSamplers(SerializedOutput.Spirv);
1429
1430 // Build shader output and binding table
1431 Output.bSucceeded = BuildShaderOutputFromSpirv(CompilerContext, InternalState, SerializedOutput, Output);
1432
1433 // Flush warnings
1434 CompilerContext.FlushErrors(Output.Errors);
1435
1436 // Return code reflection if requested for shader analysis
1437 if (Input.Environment.CompilerFlags.Contains(CFLAG_OutputAnalysisArtifacts) && Output.bSucceeded)
1438 {
1439 const TArray<uint32>& SpirvData = SerializedOutput.Spirv.Data;
1440 FGenericShaderStat ShaderReflection;
1441 if (CrossCompiler::FShaderConductorContext::Disassemble(CrossCompiler::EShaderConductorIR::Spirv, SpirvData.GetData(), SpirvData.Num() * SpirvData.GetTypeSize(), ShaderReflection))
1442 {
1443 ShaderReflection.StatName = FName(FString::Printf(TEXT("%s (%s)"), *ShaderReflection.StatName.ToString(), *InternalState.Input.EntryPointName));
1444 Output.ShaderStatistics.Add(MoveTemp(ShaderReflection));
1445 }
1446 }
1447
1448 return true;
1449}

1561// :todo-jn: TEMPORARY EXPERIMENT - will eventually move into preprocessing step
1562static TArray<FString> ConvertUBToBindless(FString& PreprocessedShaderSource,
1563 EBindlessUniformBufferType BindlessUniformBufferType = EBindlessUniformBufferType::ShaderRecord,
1564 TArrayView<FString> UBNames = {})
1565{
1566 // Fill a map so we pull our bindless sampler/resource indices from the right struct
1567 // :todo-jn: Do we not have the layout somewhere instead of calculating offsets? there must be a better way...
1568 auto GenerateNewDecl = [BindlessUniformBufferType](const int32 CBIndex, const FString& Members, const FString& CBName)
1569 {
1570 const FString PrefixedCBName = FString::Printf(TEXT("%s%d_%s"), *SpirvShaderCompiler::kBindlessCBPrefix, CBIndex, *CBName);
1571 const FString BindlessCBType = PrefixedCBName + TEXT("_Type");
1572 const FString BindlessCBHeapName = PrefixedCBName + SpirvShaderCompiler::kBindlessHeapSuffix;
1573 const FString PaddingName = FString::Printf(TEXT("%s_Padding"), *CBName);
1574 const TCHAR* BindlessIndexName =
1575 (BindlessUniformBufferType == EBindlessUniformBufferType::ShaderRecord) ?
1576 TEXT("VulkanHitGroupSystemParameters.BindlessUniformBuffers") :
1577 TEXT("VulkanPushConstants.StaticUniformBufferBindings");
1578
1579 FString CBDecl;
1580 CBDecl.Reserve(Members.Len() * 3); // start somewhere approx less bad
1581
1582 // Declare the struct
1583 CBDecl += TEXT("struct ") + BindlessCBType + TEXT(" \n{\n") + Members + TEXT("\n};\n");
1584
1585 // Declare the safetype and bindless array for this cb
1586 CBDecl += FString::Printf(TEXT("ConstantBuffer<%s> %s[];\n"), *BindlessCBType, *BindlessCBHeapName);
1587
1588 // Now bring in the CB
1589 CBDecl += FString::Printf(TEXT("static const %s %s = %s[%s[%d]];\n"),
1590 *BindlessCBType, *PrefixedCBName, *BindlessCBHeapName, BindlessIndexName, CBIndex);
1591
1592 // Now create a global scope var for each value (as the cbuffer would provide) to patch in seemlessly with the rest of the code
1593 uint32 MemberOffset = 0;
1594 const TCHAR* MemberSearchPtr = *Members;
1595 const uint32 LastMemberSemicolonIndex = Members.Find(TEXT(";"), ESearchCase::CaseSensitive, ESearchDir::FromEnd, -1);
1596 check(LastMemberSemicolonIndex != INDEX_NONE);
1597 const TCHAR* LastMemberSemicolon = &Members[LastMemberSemicolonIndex];
1598
1599 do
1600 {
1601 const TCHAR* MemberTypeStartPtr = nullptr;
1602 const TCHAR* MemberTypeEndPtr = nullptr;
1603 ParseHLSLTypeName(MemberSearchPtr, MemberTypeStartPtr, MemberTypeEndPtr);
1604 const FString MemberTypeName = FString::ConstructFromPtrSize(MemberTypeStartPtr, MemberTypeEndPtr - MemberTypeStartPtr);
1605
1606 FString MemberName;
1607 MemberSearchPtr = ParseHLSLSymbolName(MemberTypeEndPtr, MemberName);
1608 check(MemberName.Len() > 0);
1609
1610 if (MemberName.StartsWith(PaddingName))
1611 {
1612 while (*MemberSearchPtr && *MemberSearchPtr != ';')
1613 {
1614 MemberSearchPtr++;
1615 }
1616 }
1617 else
1618 {
1619 // Skip over trailing tokens and pick up arrays
1620 FString ArrayDecl;
1621 while (*MemberSearchPtr && *MemberSearchPtr != ';')
1622 {
1623 if (*MemberSearchPtr == '[')
1624 {
1625 ArrayDecl.AppendChar(*MemberSearchPtr);
1626
1627 MemberSearchPtr++;
1628 while (*MemberSearchPtr && *MemberSearchPtr != ']')
1629 {
1630 ArrayDecl.AppendChar(*MemberSearchPtr);
1631 MemberSearchPtr++;
1632 }
1633
1634 ArrayDecl.AppendChar(*MemberSearchPtr);
1635 }
1636
1637 MemberSearchPtr++;
1638 }
1639
1640 CBDecl += FString::Printf(TEXT("static const %s %s%s = %s.%s;\n"), *MemberTypeName, *MemberName, *ArrayDecl, *PrefixedCBName, *MemberName);
1641 }
1642
1643 MemberSearchPtr++;
1644
1645 } while (MemberSearchPtr < LastMemberSemicolon);
1646
1647 return CBDecl;
1648 };
1649
1650 // replace "cbuffer" decl with a struct filled from bindless constant buffer
1651 TArray<FString> BindlessUBs;
1652 if (!UBNames.IsEmpty())
1653 {
1654 BindlessUBs.SetNum(UBNames.Num());
1655 }
1656
1657 // Patch ConstantBuffers
1658 {
1659 const FString UniformBufferDeclIdentifier = TEXT("ConstantBuffer");
1660
1661 int32 SearchIndex = PreprocessedShaderSource.Find(UniformBufferDeclIdentifier, ESearchCase::CaseSensitive, ESearchDir::FromStart, -1);
1662 while (SearchIndex != INDEX_NONE)
1663 {
1664 const TCHAR* TemplateBeginPtr = FCString::Strstr(&PreprocessedShaderSource[SearchIndex + UniformBufferDeclIdentifier.Len()], TEXT("<"));
1665 const TCHAR* TemplateEndPtr = FCString::Strstr(TemplateBeginPtr + 1, TEXT(">"));
1666
1667 FString TemplateName;
1668 const TCHAR* TemplateNameEndPtr = ParseHLSLSymbolName(TemplateBeginPtr + 1, TemplateName);
1669 check(TemplateName.Len() > 0);
1670
1671 FString CBName;
1672 const TCHAR* CBNameEndPtr = ParseHLSLSymbolName(TemplateEndPtr + 1, CBName);
1673 check(CBName.Len() > 0);
1674
1675
1676 int32 CBIndex = INDEX_NONE;
1677
1678 // Don't patch the VulkanHitGroupSystemParameters system buffer
1679 if (CBName != TEXT("VulkanHitGroupSystemParameters"))
1680 {
1681 if (UBNames.IsEmpty())
1682 {
1683 CBIndex = BindlessUBs.Add(CBName);
1684 check(CBIndex < 16);
1685 }
1686 else
1687 {
1688 CBIndex = UBNames.Find(*CBName);
1689 if (CBIndex != INDEX_NONE)
1690 {
1691 BindlessUBs[CBIndex] = CBName;
1692 }
1693 }
1694 }
1695
1696 const TCHAR* EndDeclPtr = FCString::Strstr(CBNameEndPtr, TEXT(";"));
1697 const int32 EndDeclIndex = EndDeclPtr - (*PreprocessedShaderSource);
1698
1699 int32 NextStartIndex = EndDeclIndex;
1700 if (CBIndex != INDEX_NONE)
1701 {
1702 FString NewDecl;
1703
1704 // Start with the bindless heap
1705 const FString BindlessCBHeapName = FString::Printf(TEXT("%s%d_%s%s"),
1706 *SpirvShaderCompiler::kBindlessCBPrefix, CBIndex, *CBName, *SpirvShaderCompiler::kBindlessHeapSuffix);
1707 NewDecl += FString::Printf(TEXT("ConstantBuffer<%s> %s[]; "), *TemplateName, *BindlessCBHeapName);
1708
1709 // Now bring in the CB
1710 const TCHAR* BindlessIndexName =
1711 (BindlessUniformBufferType == EBindlessUniformBufferType::ShaderRecord) ?
1712 TEXT("VulkanHitGroupSystemParameters.BindlessUniformBuffers") :
1713 TEXT("VulkanPushConstants.StaticUniformBufferBindings");
1714 NewDecl += FString::Printf(TEXT("static const %s %s = %s[%s[%d]];"),
1715 *TemplateName, *CBName, *BindlessCBHeapName, BindlessIndexName, CBIndex);
1716
1717 const int32 OldDeclLen = EndDeclIndex - SearchIndex + 1;
1718 PreprocessedShaderSource.RemoveAt(SearchIndex, OldDeclLen, EAllowShrinking::No);
1719 PreprocessedShaderSource.InsertAt(SearchIndex, NewDecl);
1720
1721 NextStartIndex = SearchIndex + NewDecl.Len();
1722 }
1723
1724 SearchIndex = PreprocessedShaderSource.Find(UniformBufferDeclIdentifier, ESearchCase::CaseSensitive, ESearchDir::FromStart, NextStartIndex);
1725 }
1726 }
1727
1728 // Patch cbuffers
1729 {
1730 const FString UniformBufferDeclIdentifier = TEXT("cbuffer");
1731
1732 int32 SearchIndex = PreprocessedShaderSource.Find(UniformBufferDeclIdentifier, ESearchCase::CaseSensitive, ESearchDir::FromStart, -1);
1733 while (SearchIndex != INDEX_NONE)
1734 {
1735 FString StructName;
1736 const TCHAR* StructNameEndPtr = ParseHLSLSymbolName(&PreprocessedShaderSource[SearchIndex + UniformBufferDeclIdentifier.Len()], StructName);
1737 check(StructName.Len() > 0);
1738
1739 const TCHAR* OpeningBracePtr = FCString::Strstr(&PreprocessedShaderSource[SearchIndex + UniformBufferDeclIdentifier.Len()], TEXT("{"));
1740 check(OpeningBracePtr);
1741 const TCHAR* ClosingBracePtr = FindMatchingClosingBrace(OpeningBracePtr + 1);
1742 check(ClosingBracePtr);
1743 const int32 ClosingBraceIndex = ClosingBracePtr - (*PreprocessedShaderSource);
1744
1745 int32 CBIndex;
1746 if (UBNames.IsEmpty())
1747 {
1748 CBIndex = BindlessUBs.Add(StructName);
1749 check(CBIndex < 16);
1750 }
1751 else
1752 {
1753 CBIndex = UBNames.Find(*StructName);
1754 if (CBIndex != INDEX_NONE)
1755 {
1756 BindlessUBs[CBIndex] = StructName;
1757 }
1758 }
1759
1760 int32 NextStartIndex = ClosingBraceIndex;
1761 if (CBIndex != INDEX_NONE)
1762 {
1763 const FString Members = FString::ConstructFromPtrSize(OpeningBracePtr + 1, ClosingBracePtr - OpeningBracePtr - 1);
1764 const FString NewDecl = GenerateNewDecl(CBIndex, Members, StructName);
1765
1766 const int32 OldDeclLen = ClosingBraceIndex - SearchIndex + 1;
1767 PreprocessedShaderSource.RemoveAt(SearchIndex, OldDeclLen, EAllowShrinking::No);
1768 PreprocessedShaderSource.InsertAt(SearchIndex, NewDecl);
1769
1770 NextStartIndex = SearchIndex + NewDecl.Len();
1771 }
1772
1773 SearchIndex = PreprocessedShaderSource.Find(UniformBufferDeclIdentifier, ESearchCase::CaseSensitive, ESearchDir::FromStart, NextStartIndex);
1774 }
1775 }
1776
1777 return BindlessUBs;
1778}

1780static bool CompileShaderGroup(
1781 FSpirvShaderCompilerInternalState& InternalState,
1782 const FShaderSource::FStringType& OriginalPreprocessedShaderSource,
1783 FShaderCompilerOutput& MergedOutput
1784)
1785{
1786 checkf(InternalState.bSupportsBindless && InternalState.bUseBindlessUniformBuffer, TEXT("Ray tracing requires full bindless in Vulkan."));
1787
1788 // Compile each one of the shader modules separately and create one big blob for the engine
1789 auto CompilePartialExport = [&OriginalPreprocessedShaderSource, &InternalState, &MergedOutput](
1790 FSpirvShaderCompilerInternalState::EHitGroupShaderType HitGroupShaderType,
1791 const TCHAR* PartialFileExtension,
1792 SpirvShaderCompilerSerializedOutput& PartialSerializedOutput)
1793 {
1794 InternalState.HitGroupShaderType = HitGroupShaderType;
1795
1796 FShaderCompilerOutput TempOutput;
1797 const bool bIsClosestHit = (HitGroupShaderType == FSpirvShaderCompilerInternalState::EHitGroupShaderType::ClosestHit);
1798 FShaderCompilerOutput& PartialOutput = bIsClosestHit ? MergedOutput : TempOutput;
1799
1800 FShaderSource::FViewType OrigSourceView(OriginalPreprocessedShaderSource);
1801 FShaderSource PartialPreprocessedShaderSource(OrigSourceView);
1802 UE::ShaderCompilerCommon::RemoveDeadCode(PartialPreprocessedShaderSource, InternalState.GetEntryPointName(), PartialOutput.Errors);
1803
1804 if (InternalState.bDebugDump)
1805 {
1806 DumpDebugShaderText(InternalState.Input, PartialPreprocessedShaderSource.GetView().GetData(), *FString::Printf(TEXT("%s.hlsl"), PartialFileExtension));
1807 }
1808
1809 const bool bPartialSuccess = SpirvShaderCompiler::CompileWithShaderConductor(InternalState, PartialPreprocessedShaderSource.GetView(), PartialSerializedOutput, PartialOutput);
1810
1811 if (!bIsClosestHit)
1812 {
1813 MergedOutput.NumInstructions = FMath::Max(MergedOutput.NumInstructions, PartialOutput.NumInstructions);
1814 MergedOutput.NumTextureSamplers = FMath::Max(MergedOutput.NumTextureSamplers, PartialOutput.NumTextureSamplers);
1815 MergedOutput.Errors.Append(MoveTemp(PartialOutput.Errors));
1816 }
1817
1818 return bPartialSuccess;
1819 };

1821 bool bSuccess = false;
