70 virtual int32 GetProcessSnapshot( TArray<FTargetDeviceProcessInfo>& OutProcessInfos ) override
71 {
72 // enumerate processes
73 HANDLE ProcessSnapshot = ::CreateToolhelp32Snapshot(TH32CS_SNAPPROCESS, 0);
74
75 if (ProcessSnapshot != INVALID_HANDLE_VALUE)
76 {
77 PROCESSENTRY32 ProcessEntry;
78 ProcessEntry.dwSize = sizeof(PROCESSENTRY32);
79
80 if (::Process32First(ProcessSnapshot, &ProcessEntry))
81 {
82 do
83 {
84 // only include processes that the user has permission to query
85 HANDLE ProcessHandle = ::OpenProcess(PROCESS_QUERY_INFORMATION, false, ProcessEntry.th32ProcessID);
86
87 if (ProcessHandle != NULL)
88 {
89 FTargetDeviceProcessInfo ProcessInfo;
90
91 // get process details
92 ProcessInfo.Id = ProcessEntry.th32ProcessID;
93 ProcessInfo.Name = ProcessEntry.szExeFile;
94 ProcessInfo.ParentId = ProcessEntry.th32ParentProcessID;
95 ProcessInfo.UserName = TEXT("-");
96
97 // get process user name
98 HANDLE TokenHandle;
99
100 if (::OpenProcessToken(ProcessHandle, TOKEN_QUERY, &TokenHandle))
101 {
102 DWORD UserTokenSize;
103
104 ::GetTokenInformation(TokenHandle, TokenUser, NULL, 0, &UserTokenSize);
105
106 if (::GetLastError() == ERROR_INSUFFICIENT_BUFFER)
107 {
108 PTOKEN_USER UserToken = reinterpret_cast<PTOKEN_USER>(new BYTE[UserTokenSize]);
109
110 if (UserToken != NULL)
111 {
112 if (::GetTokenInformation(TokenHandle, TokenUser, UserToken, UserTokenSize, &UserTokenSize))
113 {
114 WCHAR DomainName[256];
115 DWORD DomainNameLength = 256;
116
117 WCHAR UserName[256];
118 DWORD UserNameLength = 256;
119
120 SID_NAME_USE SidType;
121
122 if (::LookupAccountSid(NULL, UserToken->User.Sid, UserName, &UserNameLength, DomainName, &DomainNameLength, &SidType))
123 {
124 ProcessInfo.UserName = UserName;
125 }
126 else
127 {
128 ProcessInfo.UserName = TEXT("SYSTEM");
129 }
130 }
131 }
132 else
133 {
134 DWORD LastError = ::GetLastError();
135 }
136 }
137
138 ::CloseHandle(TokenHandle);
139 }
140
141 ::CloseHandle(ProcessHandle);
142
143 OutProcessInfos.Add(ProcessInfo);
144 }
145 }
146 while (::Process32Next(ProcessSnapshot, &ProcessEntry));
147 }
148
149 ::CloseHandle(ProcessSnapshot);
150 }
151
152 // get thread details
153 HANDLE ThreadSnapshot = ::CreateToolhelp32Snapshot(TH32CS_SNAPTHREAD, 0);
154
155 if (ThreadSnapshot != INVALID_HANDLE_VALUE)
156 {
157 THREADENTRY32 ThreadEntry;
158 ThreadEntry.dwSize = sizeof(THREADENTRY32);
159
160 if (::Thread32First(ThreadSnapshot, &ThreadEntry))
161 {
162 do
163 {
164 for (int32 ProcessInfoIndex = 0; ProcessInfoIndex < OutProcessInfos.Num(); ++ProcessInfoIndex)
165 {
166 FTargetDeviceProcessInfo& ProcessInfo = OutProcessInfos[ProcessInfoIndex];
167
168 if (ProcessInfo.Id == ThreadEntry.th32OwnerProcessID)
169 {
170 FTargetDeviceThreadInfo ThreadInfo;
171
172 ThreadInfo.ExitCode = 0;
173 ThreadInfo.Id = ThreadEntry.th32ThreadID;
174 ThreadInfo.StackSize = 0;
175 ThreadInfo.State = ETargetDeviceThreadStates::Unknown;
176 ThreadInfo.WaitState = ETargetDeviceThreadWaitStates::Unknown;
177
178 ProcessInfo.Threads.Add(ThreadInfo);
179
180 break;
181 }
182 }
183 }
184 while (::Thread32Next(ThreadSnapshot, &ThreadEntry));
185
186 ::CloseHandle(ThreadSnapshot);
187 }
188 }
189
190 return OutProcessInfos.Num();
191 }

295 bool AdjustShutdownPrivileges( )
296 {
297 HANDLE TokenHandle;
298 TOKEN_PRIVILEGES Token;
299
300 if (!::OpenProcessToken(GetCurrentProcess(), TOKEN_ADJUST_PRIVILEGES | TOKEN_QUERY, &TokenHandle))
301 {
302 return false;
303 }
304
305 ::LookupPrivilegeValue(NULL, SE_SHUTDOWN_NAME, &Token.Privileges[0].Luid);
306
307 if (Token.Privileges[0].Attributes == SE_PRIVILEGE_ENABLED)
308 {
309 return true;
310 }
311
312 Token.PrivilegeCount = 1;
313 Token.Privileges[0].Attributes = SE_PRIVILEGE_ENABLED;
314
315 ::AdjustTokenPrivileges(TokenHandle, 0, &Token, 0, (PTOKEN_PRIVILEGES)NULL, 0);
316
317 return (GetLastError() == ERROR_SUCCESS);
318 }
