116FORCEINLINE FUtf8StringView FCbValue::AsString(ECbFieldError* OutError, FUtf8StringView Default) const
117{
118 const UTF8CHAR* const Chars = static_cast<const UTF8CHAR*>(Data);
119 uint32 ValueSizeByteCount;
120 const uint64 ValueSize = ReadVarUInt(Chars, ValueSizeByteCount);
121
122 if (OutError)
123 {
124 if (ValueSize >= (uint64(1) << 31))
125 {
126 *OutError = ECbFieldError::RangeError;
127 return Default;
128 }
129 *OutError = ECbFieldError::None;
130 }
131
132 return FUtf8StringView(Chars + ValueSizeByteCount, int32(ValueSize));
133}

135FORCEINLINE uint64 FCbValue::AsInteger(UE::CompactBinary::Private::FIntegerParams Params, ECbFieldError* OutError, uint64 Default) const
136{
137 // A shift of a 64-bit value by 64 is undefined so shift by one less because magnitude is never zero.
138 const uint64 OutOfRangeMask = uint64(-2) << (Params.MagnitudeBits - 1);
139 const uint64 IsNegative = uint8(Type) & 1;
140
141 uint32 MagnitudeByteCount;
142 const uint64 Magnitude = ReadVarUInt(Data, MagnitudeByteCount);
143 const uint64 Value = Magnitude ^ -int64(IsNegative);
144
145 if (OutError)
146 {
147 const uint64 IsInRange = (!(Magnitude & OutOfRangeMask)) & ((!IsNegative) | Params.IsSigned);
148 *OutError = IsInRange ? ECbFieldError::None : ECbFieldError::RangeError;
149
150 const uint64 UseValueMask = -int64(IsInRange);
151 return (Value & UseValueMask) | (Default & ~UseValueMask);
152 }
153
154 return Value;
155}
