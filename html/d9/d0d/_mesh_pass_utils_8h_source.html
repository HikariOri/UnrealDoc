84 EShaderFrequency Frequency = Shader->GetFrequency();

99 Shader->GetShaderBindings(Scene, FeatureLevel, PrimitiveSceneProxy, MaterialRenderProxy, Material, ShaderElementData, SingleShaderBindings);
100 }
101
102 ShaderBindings.Finalize(&MeshProcessorShaders);
