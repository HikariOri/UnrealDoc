12 // Class fragments use to composite a class that forwards its calls to the query context contract.
13 // If the forwarder contains any virtual calls, the forwarder won't be fully inlined and will add an additional pointer dereference
14 // to each call. To avoid this, the forwarder doesn't directly inherit from the capability interfaces.
15
16 struct FQueryContext {};
17 struct FConstQueryContext {};
18
19 template<bool bIsConst, typename... Capabilities> struct TForwarder
20 {
21 static_assert(sizeof...(Capabilities) < 0,
22 "Specialization for query context forwards missing. The most likely reason is the addition of a new query "
23 "context capability that wasn't specialized.");
24 };
