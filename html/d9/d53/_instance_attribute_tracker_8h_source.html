15template <typename ElementType>
16struct FBitManip
17{
18 // Utility to, at compile time, replicate the given bit mask 'Mask' as many times as MaskSize inside sizeof(ElementType)*8.
19 // e.g. turning Mask=0x1 into 0x11111111 if MaskSize=4 & ElementType is uint32
20 template <ElementType Mask, uint32 MaskSize>
21 struct FRepMask
22 {
23 template <uint32 Step>
24 struct FIterator;
25 template <>
26 struct FIterator<0u>
27 {
28 static constexpr ElementType Result = Mask;
29 };
30 template <uint32 Step>
31 struct FIterator
32 {
33 static constexpr ElementType Result = (Mask << (Step * MaskSize)) | FIterator<Step - 1u>::Result;
34
35 };
36
37 static constexpr uint32 NumSteps = (sizeof(ElementType) * 8u / MaskSize) - 1u;
38 static constexpr ElementType Result = FIterator<NumSteps>::Result;
39 };
40
43 static inline uint32 CountTrailingZeros(ElementType Value);
44};

129 inline void MarkRemoved(FPrimitiveInstanceId Id)
130 {
131 FirstRemovedIdIndex = FMath::Min(FirstRemovedIdIndex, Id.GetAsIndex());
132 // Lazy allocate the memory so there is nothing to move around in case nothing is removed
133 if (Id.GetAsIndex() >= Removed.Num())
134 {
135 // TBitArray does a lot of work for each added bit, first clearing the slack redundantly and then also clearing the bit even though it was already cleared because the slack was cleared...
136 // so we pad the allocation request with at least a dword at a time.
137 int32 RemovedPadded = FMath::DivideAndRoundUp(Id.GetAsIndex() + 1, 32) * 32;
138 Removed.SetNum(RemovedPadded, false);
139 }
140 Removed[Id.GetAsIndex()] = true;
141 }

147 inline void RemoveAtSwap(FPrimitiveInstanceId Id, int32 Index, int32 MaxInstanceIndex)
148 {
149 Validate();
150
151 SetNum(FMath::Max(Index, MaxInstanceIndex));
152
153 MarkRemoved(Id);
154
155 // book-keeping
156 DecChangedCounts(Index);
157
158 // Remove the tracked state bits
159 int32 LastIndex = MaxIndex - 1;
160 if (Index != LastIndex)
161 {
162 // Move tracked & tag as moved.
163 ElementType PrevFlags = GetFlags(LastIndex);
164 if ((((FToBit<EFlag::IndexChanged>::Bit | FToBit<EFlag::Added>::Bit)) & PrevFlags) == 0u)
165 {
166 check(NumChanged[uint32(EFlag::IndexChanged)] >= 0);
167 NumChanged[uint32(EFlag::IndexChanged)] += 1;
168 }
169 SetMaskTo(Index, FToBit<EFlag::IndexChanged>::Bit | PrevFlags);
170 FirstChangedIndex = FMath::Min(FirstChangedIndex, Index);
171 LastChangedIndex = FMath::Max(LastChangedIndex, Index);
172 }
173 // clear the last one so that it is all zero after the last mask (important for when we do implicit add, as we do at the moment).
174 ClearLastElement();
175 MaxIndex -= 1;
176
177 Validate();
178 }

184 inline void RemoveAt(FPrimitiveInstanceId Id, int32 Index, int32 MaxInstanceIndex)
185 {
186 Validate();
187
188 SetNum(FMath::Max(Index, MaxInstanceIndex));
189
190 MarkRemoved(Id);
191
192 DecChangedCounts(Index);
193
194 // Remove the tracked state bits
195 int32 LastIndex = MaxIndex - 1;
196 if (Index != LastIndex)
197 {
198 int32 FirstElementIndex = Index / MasksPerElement;
199 int32 LastElementIndex = LastIndex / MasksPerElement;
200
201 ElementType CurrentElement = Data[FirstElementIndex];
202 ElementType NextElement = LoadCountAndMask(FirstElementIndex + 1, FirstElementIndex != LastElementIndex);
203 // 1. Move up to "Num" bits to cover the removed item, shifting in zero.
204 {
205 uint32 ElementSubIndex = (Index % MasksPerElement);
206 // Count mask changes only in the moved part (not including the removed or any previous).
207 uint32 NumMasksInElement = FMath::Min(MasksPerElement, MaxIndex - FirstElementIndex * MasksPerElement);
208 CountMaskChanges(CurrentElement, ElementSubIndex + 1, NumMasksInElement);
209 ElementType CurrentElementResult = (CurrentElement >> uint32(EFlag::Num)) | IndexChangedElementMask;
210 if (ElementSubIndex != 0u)
211 {
212 uint32 ElementSubShift = ElementSubIndex * uint32(EFlag::Num);
213 ElementType Mask = ~ElementType(0) << ElementSubShift;
214 // we shifted some as we shouldn't, put them back
215 CurrentElementResult = (CurrentElementResult & Mask) | (CurrentElement & (~Mask));
216 }
217 // Move the first item from the next step
218 CurrentElementResult |= NextElement << (MasksPerElement - 1u) * uint32(EFlag::Num);
219 Data[FirstElementIndex] = CurrentElementResult;
220 }
221 // 2. Do the middle, whole, elements...
222 for (int32 ElementIndex = FirstElementIndex + 1; ElementIndex < LastElementIndex; ++ElementIndex)
223 {
224 CurrentElement = NextElement >> uint32(EFlag::Num);
225 NextElement = LoadCountAndMask(ElementIndex + 1, true);
226 ElementType CurrentElementResult = CurrentElement | NextElement << (MasksPerElement - 1u) * uint32(EFlag::Num);
227 Data[ElementIndex] = CurrentElementResult;
228 }
229 // 2. Do the last element...
230 if (LastElementIndex != FirstElementIndex)
231 {
232 Data[LastElementIndex] = NextElement >> uint32(EFlag::Num);
233 }
234
235 FirstChangedIndex = FMath::Min(FirstChangedIndex, Index);
236 LastChangedIndex = FMath::Max(LastChangedIndex, LastIndex);
237 }
238 ClearLastElement();
239 MaxIndex -= 1;
240
241 Validate();
242 }

246 template <EFlag Flag>
247 inline void MarkIndex(int32 Index, int32 MaxInstanceIndex)
248 {
249 Validate();
250
251 // Lazy allocate
252 SetNum(FMath::Max(Index, MaxInstanceIndex));
253
254 check(Index < MaxIndex);
255
256 FirstChangedIndex = FMath::Min(FirstChangedIndex, Index);
257 LastChangedIndex = FMath::Max(LastChangedIndex, Index);
258
259 uint32 ElementSubShift = (Index % MasksPerElement) * uint32(EFlag::Num);
260
261 ElementType SetMask = FToBit<Flag>::Bit << ElementSubShift;
262
263 int32 ElementIndex = Index / MasksPerElement;
264
265 ElementType PrevElement = Data[ElementIndex];
266
267 if ((((FToBit<Flag>::Bit | FToBit<EFlag::Added>::Bit) << ElementSubShift) & PrevElement) == 0u)
268 {
269 check(NumChanged[uint32(Flag)] >= 0);
270 NumChanged[uint32(Flag)] += 1;
271 }
272
273 Data[ElementIndex] = PrevElement | SetMask;
274
275 Validate();
276 }

318 class FAnyValidIterator
319 {
320 public:
321 inline FAnyValidIterator() = default;
322 inline FAnyValidIterator(const FInstanceAttributeTracker *InTracker, int32 StartIndex) : Tracker(InTracker)
323 {
324 ElementIndex = StartIndex / MasksPerElement;
325 ElementOffset = StartIndex % MasksPerElement;
326 CurrentElement = Tracker->Data.IsValidIndex(ElementIndex) ? (Tracker->Data[ElementIndex] & ElementMask) : 0;
327 CurrentElement >>= uint32(EFlag::Num) * ElementOffset;
328 AdvanceUntilAny();
329 }
330
331 inline void AdvanceUntilAny()
332 {
333 while(ElementIndex < Tracker->Data.Num())
334 {
335 // Skip the whole word if nothing is set.
336 if (CurrentElement)
337 {
338 uint32 ZeroCount = FBitManip::CountTrailingZeros(CurrentElement);
339 uint32 Steps = ZeroCount / uint32(EFlag::Num);
340 CurrentElement >>= uint32(EFlag::Num) * Steps;
341 ElementOffset += Steps;
342 return;
343 }
344 ++ElementIndex;
345 ElementOffset = 0;
346 if (ElementIndex >= Tracker->Data.Num())
347 {
348 CurrentElement = 0u;
349 return;
350 }
351 CurrentElement = Tracker->Data[ElementIndex] & ElementMask;
352 }
353 }
354
355 inline void AdvanceToNext()
356 {
357 // Clear the current value (which will cause AdvanceUntilAny to move forwards)
358 CurrentElement &= ~ElementType(Mask);
359 AdvanceUntilAny();
360 }
361
362 inline FAnyValidIterator &operator++()
363 {
364 AdvanceToNext();
365 return *this;
366 }
367
368 inline explicit operator bool() const
369 {
370 return GetIndex() < FMath::Min(Tracker->MaxIndex, Tracker->LastChangedIndex + 1);
371 }
372
373 inline int32 GetIndex() const
374 {
375 return ElementIndex * MasksPerElement + ElementOffset;
376 }

378 /*
379 * Returns the mask for the current item
380 * NOTE: does not clear any bits outside the search mask.
381 */
382 inline uint32 GetMask() const

384 return CurrentElement;
385 }

387 template <EFlag Flag>
388 inline bool TestFlag() const

394 int32 ElementIndex = 0;
395 ElementType CurrentElement;

552 inline void SetMaskTo(int32 Index, ElementType Mask)
553 {
554 check(Index < MaxIndex);
555 check(Mask <= AnyFlagMask);
556
557 const int32 ElementIndex = Index / MasksPerElement;
558 const uint32 ElementSubShift = (Index % MasksPerElement) * uint32(EFlag::Num);

619 inline void CountMaskChanges(ElementType Element, uint32 Offset, uint32 NumMasksInElement)
620 {
621 // If the offset is equal to the number of masks then there can't be any changes
622 if (Offset < MasksPerElement)
623 {
624 // 1. Element & IndexChangedElementMask - 1 if it already had index changed
625 ElementType HasIndexChanged = (Element & IndexChangedElementMask);
626 // 2. Element & AddedElementMask - 1 if already had added << to align
627 ElementType HasAdded = (Element & AddedElementMask) << (uint32(EFlag::IndexChanged) - uint32(EFlag::Added));
628 ElementType CombinedMask = HasIndexChanged | HasAdded;
629
630 int32 Delta = NumMasksInElement - Offset - FMath::CountBits(CombinedMask >> (Offset * uint32(EFlag::Num)));
631 check(Delta >= 0);
632 NumChanged[uint32(EFlag::IndexChanged)] += Delta;
633 }
634 };
638 inline ElementType LoadCountAndMask(int32 ElementIndex, bool bHasNextElement)
639 {
640 if (bHasNextElement)
641 {
642 uint32 NumMasksInElement = FMath::Min(MasksPerElement, MaxIndex - ElementIndex * MasksPerElement);
643 ElementType Element = Data[ElementIndex];
644 CountMaskChanges(Element, 0, NumMasksInElement);
645 return Element | IndexChangedElementMask;
646 }
647 return ElementType(0);
