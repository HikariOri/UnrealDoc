571template <typename TInputType = FSimCallbackNoInput, typename TOutputType = FSimCallbackNoOutput, ESimCallbackOptions TOptions = ESimCallbackOptions::Presimulate>
572class TSimCallbackObject : public ISimCallbackObject
573{
574public:
575
576 TSimCallbackObject()
577 : ISimCallbackObject(TOptions)
578 , CurrentOutput_External(nullptr)
579 { }
580
581 UE_DEPRECATED(5.1, "Use default constructor instead and specify RunFrozenOnGameThread using TOptions template parameter.")
582 TSimCallbackObject(bool InRunOnFrozenGameThread)
583 : ISimCallbackObject(InRunOnFrozenGameThread ? (TOptions | ESimCallbackOptions::RunOnFrozenGameThread) : TOptions)
584 , CurrentOutput_External(nullptr)
585 { }
586
587 virtual void FreeOutputData_External(FSimCallbackOutput* Output) override
588 {
589 auto Concrete = static_cast<TOutputType*>(Output);
590 Concrete->Reset();
591 OutputPool.Enqueue(Concrete);
592 }
593
597 TInputType* GetProducerInputData_External()
598 {
599 return static_cast<TInputType*>(ISimCallbackObject::GetProducerInputData_External());
600 }
601
605 const TInputType* GetConsumerInput_Internal() const
606 {
607 return static_cast<const TInputType*>(GetCurrentInput_Internal());
608 }
609
614 TSimCallbackOutputHandle<TOutputType> PopOutputData_External()
615 {
616 const FReal ResultsTime = Utilities::GetSolverPhysicsResultsTime(GetSolver());
617 if(!CurrentOutput_External)
618 {
619 OutputQueue.Dequeue(CurrentOutput_External);
620 }
621
622 if(CurrentOutput_External && CurrentOutput_External->InternalTime <= ResultsTime)
623 {
624 TOutputType* Output = CurrentOutput_External;
625 CurrentOutput_External = nullptr;
626 return TSimCallbackOutputHandle<TOutputType>(Output, this);
627 }
628 else
629 {
630 return TSimCallbackOutputHandle<TOutputType>();
631 }
632 }
633
643 TSimCallbackOutputHandle<TOutputType> PopFutureOutputData_External()
644 {
645 if (!CurrentOutput_External)
646 {
647 OutputQueue.Dequeue(CurrentOutput_External);
648 }
649
650 TOutputType* Output = CurrentOutput_External;
651 CurrentOutput_External = nullptr;
652 return TSimCallbackOutputHandle<TOutputType>(Output, this);
653 }
654
659 bool IsOutputQueueEmpty_External() const
660 {
661 return OutputQueue.IsEmpty();
662 }
663
667
668 TOutputType& GetProducerOutputData_Internal()
669 {
670 if(!CurrentOutput_Internal)
671 {
672 CurrentOutput_Internal = NewOutputData_Internal(GetSimTime_Internal());
673 }
674
675 return static_cast<TOutputType&>(*CurrentOutput_Internal);
676 }
677
678private:
679
680 TOutputType* NewOutputData_Internal(const FReal InternalTime)
681 {
682 auto NewOutput = NewDataHelper(OutputBacking, OutputPool);
683 NewOutput->InternalTime = InternalTime;
684 return NewOutput;
685 }
686
687 template <typename T>
688 T* NewDataHelper(TArray<TUniquePtr<T>>& Backing, TTransactionallySafeSpscQueue<T*>& Queue)
689 {
690 T* Result;
691 if (!Queue.Dequeue(Result))
692 {
693 Backing.Emplace(new T());
694 Result = Backing.Last().Get();
695 }
696
697 return Result;
698 }
699
700 virtual void FreeInputData_Internal(FSimCallbackInput* Input) override
701 {
702 auto Concrete = static_cast<TInputType*>(Input);
703 Concrete->Reset();
704 InputPool.Enqueue(Concrete);
705 }
706
707 TInputType* NewInputData_External()
708 {
709 return NewDataHelper(InputBacking, InputPool);
710 }
711
712 virtual FSimCallbackInput* AllocateInputData_External() override
713 {
714 return NewInputData_External();
715 }
716
717 void OnFinalizeOutputData_Internal(FSimCallbackOutput* BaseOutput) override
718 {
719 OutputQueue.Enqueue(static_cast<TOutputType*>(BaseOutput));
720 }
721
722 TTransactionallySafeSpscQueue<TInputType*> InputPool;
723 TArray<TUniquePtr<TInputType>> InputBacking;
724
725 TTransactionallySafeSpscQueue<TOutputType*> OutputPool;
726 TArray<TUniquePtr<TOutputType>> OutputBacking;
727 TTransactionallySafeSpscQueue<TOutputType*> OutputQueue; //holds the outputs in order
728
729 TOutputType* CurrentOutput_External; //the earliest output we can consume
