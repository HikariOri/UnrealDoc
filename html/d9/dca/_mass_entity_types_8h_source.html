497 void ReplaceSharedFragments(TArrayView<TSharedStruct> Fragments)
498 {
499 using FDecayedSharedStruct = TDecay<TSharedStruct>::Type;
500
501 DirtyHashCache();
502 for (const TSharedStruct& NewFragment : Fragments)
503 {
504 const UScriptStruct* NewFragScriptStruct = NewFragment.GetScriptStruct();
505 check(NewFragScriptStruct);
506
507 bool bEntryFound = false;
508 for (FDecayedSharedStruct& MyFragment : GetMutableFragmentsContainer<FDecayedSharedStruct>())
509 {
510 if (MyFragment.GetScriptStruct() == NewFragScriptStruct)
511 {
512 MyFragment = NewFragment;
513 bEntryFound = true;
514 break;
515 }
516 }
517 ensureMsgf(bEntryFound, TEXT("Existing fragment of type %s could not be found"), *GetNameSafe(NewFragScriptStruct));
518 }
