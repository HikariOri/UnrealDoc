28// Use this to efficiently allocate a temporary array using MemStack, instead
29// of using TArray and going through the global allocator. Only declare local variable
30// of this struct (i.e. do not "new TTemporaryArray")
31// Remark: The allocated memory lifespan is the same as the TTemporaryArray local variable
32// (it is deallocated when TTemporaryArray goes out of scope).
33template <typename T>
34struct TTemporaryArray : public TArrayView<T>
35{
36 FMemMark MemMark;
37
38 TTemporaryArray(int32 Num)
39 : MemMark{ FMemStack::Get() }
40 {
41 *static_cast<TArrayView<T>*>(this) = { (T*)FMemStack::Get().Alloc(sizeof(T) * Num, alignof(T)), Num };
42 }
43
44 TTemporaryArray(TConstArrayView<T> Source)
45 : TTemporaryArray { Source.Num() }
46 {
47 static_assert(std::is_trivially_copyable_v<T>);
48 FMemory::Memcpy(TArrayView<T>::GetData(), Source.GetData(), Source.NumBytes());
49 }
50
51 operator TConstArrayView<T>() const
52 {
53 return { TArrayView<T>::GetData(), TArrayView<T>::Num() };
54 }
55
56 void Zero()
57 {
58 FMemory::Memzero(TArrayView<T>::GetData(), sizeof(T) * TArrayView<T>::Num());
59 }
60};

82// Returns whether two array views are equal by comparing their elements.
83// Requires operator== to be defined for T.
84template <typename T>
85bool ArrayViewEquals(TArrayView<T> A, TArrayView<T> B)
86{
87 if (A.Num() != B.Num())
88 {
89 return false;
90 }

92 {
93 if (!(A[i] == B[i]))

105// Computes the hash of an array view by combining the hashes of its elements.
106// Requires GetTypeHash to be defined for T.
107template <typename T>
108uint32 ArrayViewHash(TArrayView<T> Array)
109{
110 uint32 Hash = ::GetTypeHash(Array.Num());
111 for (const T& Element : Array)
112 {
113 Hash = HashCombine(Hash, GetTypeHash(Element));
114 }
115 return Hash;
