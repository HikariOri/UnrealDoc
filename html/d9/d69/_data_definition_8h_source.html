40// The enumerator values must match `Verse::EPersistenceExternalAccess`.
41enum struct EPersistenceExternalAccess : std::uint8_t
42{
43 Internal = 0,
44 Read = 1 << 0,
45 Write = 1 << 1,
46 ReadWrite = Read | Write
47};

49inline bool IsExternallyWritable(EPersistenceExternalAccess Arg)
50{
51 return static_cast<std::uint8_t>(Arg) & static_cast<std::uint8_t>(EPersistenceExternalAccess::Write);
52}

54inline bool IsExternallyReadable(EPersistenceExternalAccess Arg)
55{
