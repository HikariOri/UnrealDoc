794 inline void SetPackageFlagsTo( uint32 NewFlags )
795 {
796 uint32 OldFlags = GetPackageFlags();
797
798 // Fast path without atomics if already set
799 if (NewFlags == OldFlags)
800 {
801 return;
802 }
803
804 if (AutoRTFM::IsClosed())
805 {
806 UE_AUTORTFM_OPEN
807 {
808 OldFlags = PackageFlagsPrivate.exchange(NewFlags, std::memory_order_relaxed);
809 };
810
811 // If we abort we undo setting the flags we just set.
812 AutoRTFM::OnAbort([this, OldFlags]
813 {
814 PackageFlagsPrivate.store(OldFlags, std::memory_order_relaxed);
815 });
816 }
817 else
818 {
819 PackageFlagsPrivate.store(NewFlags, std::memory_order_relaxed);
820 }
821 }
822
828 inline void SetPackageFlags( uint32 FlagsToSet )
829 {
830 uint32 OldFlags = GetPackageFlags();
831 uint32 NewFlags = OldFlags | FlagsToSet;
832
833 // Fast path without atomics if already set
834 if (NewFlags == OldFlags)
835 {
836 return;
837 }
838
839 if (AutoRTFM::IsClosed())
840 {
841 UE_AUTORTFM_OPEN
842 {
843 OldFlags = PackageFlagsPrivate.fetch_or(FlagsToSet, std::memory_order_relaxed);
844 };
845
846 // If we abort we undo setting the flags we just set.
847 AutoRTFM::OnAbort([this, OldFlags, FlagsToSet]
848 {
849 uint32 MaskFlags = OldFlags;
850
851 // Now just extract out the old flags that mattered (the ones we were setting).
852 MaskFlags &= FlagsToSet;
853 // And unmask the flags we didn't mention.
854 MaskFlags |= ~FlagsToSet;
855
856 PackageFlagsPrivate.fetch_and(MaskFlags, std::memory_order_relaxed);
857 });
858 }
859 else
860 {
861 PackageFlagsPrivate.fetch_or(FlagsToSet, std::memory_order_relaxed);
862 }
863 }
864
870 inline void ClearPackageFlags( uint32 FlagsToClear )
871 {
872 uint32 OldFlags = GetPackageFlags();
873 uint32 NewFlags = OldFlags & ~FlagsToClear;
874
875 // Fast path without atomics if already cleared
876 if (NewFlags == OldFlags)
877 {
878 return;
879 }
880
881 if (AutoRTFM::IsClosed())
882 {
883 UE_AUTORTFM_OPEN
884 {
885 OldFlags = PackageFlagsPrivate.fetch_and(~FlagsToClear, std::memory_order_relaxed);
886 };
887
888 // If we abort we undo clearing the flags we just unset.
889 AutoRTFM::OnAbort([this, OldFlags, FlagsToClear]
890 {
891 int32 MaskFlags = OldFlags;
892
893 // Now just extract out the old flags that mattered (the ones we were setting).
894 MaskFlags &= FlagsToClear;
895
896 PackageFlagsPrivate.fetch_or(MaskFlags, std::memory_order_relaxed);
897 });
898 }
899 else
900 {
901 PackageFlagsPrivate.fetch_and(~FlagsToClear, std::memory_order_relaxed);
902 }
903 }
