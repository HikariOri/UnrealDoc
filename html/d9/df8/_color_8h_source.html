535//
536// FColor
537// Stores a color with 8 bits of precision per channel.
538// Note: Linear color values should always be converted to gamma space before stored in an FColor, as 8 bits of precision is not enough to store linear space colors!
539// This can be done with FLinearColor::ToFColor(true)
540//
541
542struct FColor
543{
544public:
545 // Variables.
546#if PLATFORM_LITTLE_ENDIAN
547 union { struct{ uint8 B,G,R,A; }; uint32 Bits; };
548#else // PLATFORM_LITTLE_ENDIAN
549 union { struct{ uint8 A,R,G,B; }; uint32 Bits; };
550#endif
551
552 uint32& DWColor(void) {return Bits;}
553 const uint32& DWColor(void) const {return Bits;}
554
555 // Constructors.
556 FColor() = default;
557 UE_FORCEINLINE_HINT explicit FColor(EForceInit)
558 {
559 // put these into the body for proper ordering with INTEL vs non-INTEL_BYTE_ORDER
560 R = G = B = A = 0;
561 }
562 constexpr inline FColor( uint8 InR, uint8 InG, uint8 InB, uint8 InA = 255 )
563 // put these into the body for proper ordering with INTEL vs non-INTEL_BYTE_ORDER
564#if PLATFORM_LITTLE_ENDIAN
565 : B(InB), G(InG), R(InR), A(InA)
566#else
567 : A(InA), R(InR), G(InG), B(InB)
568#endif
569 {}
