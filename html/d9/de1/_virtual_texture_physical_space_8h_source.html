30inline bool operator==(const FVTPhysicalSpaceDescription& Lhs, const FVTPhysicalSpaceDescription& Rhs)
31{
32 if (Lhs.TileSize != Rhs.TileSize ||
33 Lhs.NumLayers != Rhs.NumLayers ||
34 Lhs.Dimensions != Rhs.Dimensions ||
35 Lhs.bCanSplit != Rhs.bCanSplit)
36 {
37 return false;
38 }
39 for (int32 Layer = 0; Layer < Lhs.NumLayers; ++Layer)
40 {
41 if (Lhs.Format[Layer] != Rhs.Format[Layer])
42 {
43 return false;
44 }
45
46 if (Lhs.bHasLayerSrgbView[Layer] != Rhs.bHasLayerSrgbView[Layer])
47 {
48 return false;
49 }
50
51 }
52 return true;
53}
54inline bool operator!=(const FVTPhysicalSpaceDescription& Lhs, const FVTPhysicalSpaceDescription& Rhs)
55{
56 return !operator==(Lhs, Rhs);
57}

59inline uint32 GetTypeHash(const FVTPhysicalSpaceDescription& Desc)
60{
61 uint32 Hash = Desc.TileSize;
62 Hash = HashCombine(Hash, GetTypeHash(Desc.Dimensions));
63 Hash = HashCombine(Hash, GetTypeHash(Desc.NumLayers));
64 Hash = HashCombine(Hash, GetTypeHash(Desc.bCanSplit));
65 for (int32 Layer = 0; Layer < Desc.NumLayers; ++Layer)
66 {
67 Hash = HashCombine(Hash, GetTypeHash(Desc.Format[Layer]));
68 Hash = HashCombine(Hash, GetTypeHash(Desc.bHasLayerSrgbView[Layer]));
69 }
70 return Hash;
71}
