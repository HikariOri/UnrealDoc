403 UE_TCOMPACT_SET& operator=(const UE_TCOMPACT_SET& Copy)
404 {
405 if (this != &Copy)
406 {
407 // This could either take the full memory size of the Copy and prevent a rehash or
408 // only allocate the required set but have to hash everything again (i.e. perf vs memory)
409 // Could try a middle ground of allowing extra memory if it's within slack margins.
410 // Going for memory savings for now
411
412 // Not using Empty(NumElements) to avoid clearing the hash memory since we'll rebuild it anyway
413 // We need to make sure the relevant parts are cleared so ResizeAllocation can run safely and with minimal cost
414 DestructItems(GetData(), this->NumElements);
415 this->NumElements = 0;
416
417 ResizeAllocation(Copy.NumElements);
418
419 this->NumElements = Copy.NumElements;
420 ConstructItems<ElementType>(GetData(), Copy.GetData(), this->NumElements);
421
422 Rehash();
423 }
424 return *this;
425 }

512 void Empty(int32 ExpectedNumElements = 0)
513 {
514 DestructItems(GetData(), this->NumElements);
515 this->NumElements = 0;
516
517 ResizeAllocation(ExpectedNumElements);
518
519 if (this->MaxElements > 0)
520 {
521 GetHashTableView().Reset();
522 }
523 }

526 void Reset()
527 {
528 if (this->NumElements > 0)
529 {
530 FCompactHashTableView HashTable = GetHashTableView();
531
532 #if UE_ENABLE_COMPACTHASHTABLE_KEYSTORAGE
533 if (this->NumElements < (SizeType)(HashTable.GetHashCount() / UE_SET_UNHASH_THRESHOLD_STORED_KEY))
534 {
535 for (int32 Index = 0; Index < this->NumElements; ++Index)
536 {
537 HashTable.RemoveHashElement(HashTable.GetStoredKey(Index, this->NumElements));
538 }
539 }
540 #else
541 if (this->NumElements < (SizeType)(HashTable.GetHashCount() / UE_SET_UNHASH_THRESHOLD_CALCULATE_KEY))
542 {
543 // if we've only used a portion of the full allocated size then it can be faster
544 // to only clear out the used hash elements instead of memsetting the whole of the memory
545 // The exact number can be hard to determine since we have to recalculate the hash value to clear it
546
547 for (const ElementType& Element : *this)
548 {
549 const KeyInitType Key = KeyFuncs::GetSetKey(Element);
550 const uint32 KeyHash = KeyFuncs::GetKeyHash(Key);
551
552 HashTable.RemoveHashElement(KeyHash);
553 }
554 }
555 #endif
556 else
557 {
558 HashTable.Reset();
559 }
560
561 DestructItems(GetData(), this->NumElements);
562 this->NumElements = 0;
563 }
564 }

569 if (this->NumElements != this->MaxElements)

571 if (ResizeAllocationPreserveData(this->NumElements))
572 {

1383 {
1384 checkf(!Writer.Is32BitTarget(), TEXT("UE_TCOMPACT_SET does not currently support freezing for 32bits"));
1385 if constexpr (TAllocatorTraits<Allocator>::SupportsFreezeMemoryImage && THasTypeLayout<InElementType>::Value)
1386 {

1391 FMemoryImageWriter ArrayWriter = Writer.WritePointer(StaticGetTypeLayoutDesc<ElementType>());
1392 ArrayWriter.WriteAlignment(GetAlignment());
1393

1395 ArrayWriter.WriteObjectArray(Data, StaticGetTypeLayoutDesc<ElementType>(), this->NumElements);
1396 ArrayWriter.WritePaddingToSize(this->MaxElements * sizeof(ElementType));
1397
1398 // Write remaining byte and hash table data

1401 const uint8* HashTableDataEnd = (const uint8*)Data + Super::GetTotalMemoryRequiredInBytes(this->MaxElements, *HashTable, Layout);
1402 ArrayWriter.WriteBytes(HashTable, HashTableDataEnd - (const uint8*)HashTable);
1403
1404 Writer.WriteBytes(this->NumElements);
1405 Writer.WriteBytes(this->MaxElements);
1406 }
1407 else
1408 {
1409 Writer.WriteBytes(UE_TCOMPACT_SET());
1410 }
1411 }
1412 else
1413 {
1414 Writer.WriteBytes(UE_TCOMPACT_SET());

1535 template<typename ComparableKey>
1536 [[nodiscard]] SizeType FindIndexByHash(uint32 KeyHash, const ComparableKey& Key) const
1537 {
1538 if (this->NumElements == 0)
1539 {
1540 return INDEX_NONE;
1541 }
1542
1543 const ElementType* ElementData = GetData();
1544 const HashCountType* HashTable = this->GetHashTableMemory(GetSetLayout());
1545 const uint8* NextIndicesData = (const uint8*)(HashTable + 1);
1546 const uint32 HashCount = *HashTable;
1547
1548 // Inlining this can save up to 40% perf compared to GetHashTableView().Find()
1549 #define UE_COMPACTHASHTABLE_EXECUTEBYTYPE(Type) \
1550 const Type* NextIndices = (const Type *)NextIndicesData; \
1551 const Type* HashIndices = NextIndices + this->MaxElements * this->KeyStorageMultiplier; \
1552 for (Type Index = HashIndices[KeyHash & (HashCount-1)]; Index != (Type)INDEX_NONE; Index = NextIndices[Index]) \
1553 { \
1554 checkSlow((SizeType)Index < this->NumElements); \
1555 if (KeyFuncs::Matches(KeyFuncs::GetSetKey(ElementData[Index]), Key)) \
1556 { \
1557 return Index; \
1558 } \
1559 }
1560
1561 UE_COMPACTHASHTABLE_CALLBYTYPE(this->MaxElements)
1562 #undef UE_COMPACTHASHTABLE_EXECUTEBYTYPE
1563
1564 return INDEX_NONE;
1565 }

1582 template <bool IsStable = false>
1583 SizeType RemoveByIndexAndHash(const SizeType ElementIndex, const uint32 KeyHash, ElementType* ElementsData, FCompactHashTableView& HashTable)
1584 {
1585 const SizeType LastElementIndex = this->NumElements - 1;
1586 SizeType NextIndex;
1587
1588 // Note: Destructing/moving elements for better debugability
1589 if (ElementIndex == LastElementIndex)
1590 {
1591 NextIndex = HashTable.Remove(ElementIndex, KeyHash, ElementIndex, KeyHash);
1592
1593 ElementsData[LastElementIndex].~ElementType();
1594 }
1595 else if constexpr (IsStable)
1596 {
1597 NextIndex = HashTable.RemoveStable(ElementIndex, KeyHash);
1598
1599 ElementsData[ElementIndex].~ElementType();
1600 RelocateConstructItems<ElementType>(ElementsData + ElementIndex, ElementsData + ElementIndex + 1, LastElementIndex - ElementIndex);
1601 }
1602 else
1603 {
1604 #if UE_ENABLE_COMPACTHASHTABLE_KEYSTORAGE
1605 const uint32 LastElementHash = HashTable.GetStoredKey(LastElementIndex, this->NumElements);
1606 #else
1607 const uint32 LastElementHash = KeyFuncs::GetKeyHash(KeyFuncs::GetSetKey(ElementsData[LastElementIndex]));
1608 #endif
1609 NextIndex = HashTable.Remove(ElementIndex, KeyHash, LastElementIndex, LastElementHash);
1610
1611 MoveByRelocate(ElementsData[ElementIndex], ElementsData[LastElementIndex]);
1612 }
1613
1614 --this->NumElements;
1615
1616 return NextIndex;
1617 }

1619 template<typename ComparableKey, bool IsStable = false>
1620 int32 RemoveImpl(uint32 KeyHash, const ComparableKey& Key)
1621 {
1622 checkSlow(this->NumElements > 0);
1623 int32 NumRemovedElements = 0;
1624
1625 ElementType* ElementsData = GetData();
1626 FCompactHashTableView HashTable = GetHashTableView();
1627
1628 SizeType ElementIndex = HashTable.GetFirst(KeyHash);
1629
1630 while (ElementIndex != INDEX_NONE)
1631 {
1632 checkf(ElementIndex >= 0 && ElementIndex < this->NumElements, TEXT("Invalid ElementIndex (%d) passed to UE_TCOMPACT_SET::RemoveByIndexAndHash. NumElements=%d"), ElementIndex, this->NumElements);
1633 if (KeyFuncs::Matches(KeyFuncs::GetSetKey(ElementsData[ElementIndex]), Key))
1634 {
1635 if constexpr (!KeyFuncs::bAllowDuplicateKeys)
1636 {
1637 RemoveByIndexAndHash<IsStable>(ElementIndex, KeyHash, ElementsData, HashTable);
1638 // If the hash disallows duplicate keys, we're done removing after the first matched key.
1639 return 1;
1640 }
1641 else
1642 {
1643 ElementIndex = RemoveByIndexAndHash<IsStable>(ElementIndex, KeyHash, ElementsData, HashTable);
1644 NumRemovedElements++;
1645 }
1646 }
1647 else
1648 {
1649 ElementIndex = HashTable.GetNext(ElementIndex, this->NumElements);
1650 }
1651 }
1652
1653 return NumRemovedElements;
1654 }
