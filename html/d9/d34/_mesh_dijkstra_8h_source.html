105 void ComputeToMaxDistance(const TArray<FSeedPoint>& SeedPointsIn, double ComputeToMaxDistanceIn)
106 {
107 MaxGraphDistance = 0.0f;
108 MaxGraphDistancePointID = -1;
109
110 SeedPoints.Reset();
111 for (const FSeedPoint& SeedPoint : SeedPointsIn)
112 {
113 int32 NewIndex = SeedPoints.Num();
114 SeedPoints.Add(SeedPoint);
115
116 int32 PointID = SeedPoint.PointID;
117 if (ensure(Queue.Contains(PointID) == false))
118 {
119 FGraphNode* Node = GetNodeForPointSetID(PointID, true);
120 Node->GraphDistance = SeedPoint.StartDistance;
121 Node->bFrozen = true;
122 Node->SeedPointID = NewIndex;
123 Queue.Insert(PointID, float(Node->GraphDistance));
124 }
125 }
126
127 while (Queue.GetCount() > 0)
128 {
129 int32 NextID = Queue.Dequeue();
130 FGraphNode* Node = GetNodeForPointSetID(NextID, false);
131 check(Node != nullptr);
132
133 MaxGraphDistance = TMathUtil<double>::Max(Node->GraphDistance, MaxGraphDistance);
134 if (MaxGraphDistance > ComputeToMaxDistanceIn)
135 {
136 return;
137 }
138
139 Node->bFrozen = true;
140 MaxGraphDistancePointID = Node->PointID;
141 UpdateNeighboursSparse(Node);
142 }
143 }

283 bool FindPathToNearestSeed(int32 PointID, TArray<int32>& PathToSeedOut, int32 MaxLength = 100000)
284 {
285 const FGraphNode* CurNode = GetNodeForPointSetID(PointID);
286 if (CurNode == nullptr || CurNode->bFrozen == false)
287 {
288 return false;
289 }
290
291 PathToSeedOut.Reset();
292 PathToSeedOut.Add(PointID);
293
294 int32 IterCount = 0;
295 while (IterCount++ < MaxLength)
296 {
297 if (CurNode->ParentPointID == -1)
298 {
299 return true;
300 }
301
302 PathToSeedOut.Add(CurNode->ParentPointID);
303
304 CurNode = GetNodeForPointSetID(CurNode->ParentPointID);
305 if (CurNode == nullptr || CurNode->bFrozen == false)
306 {
307 PathToSeedOut.Reset();
308 return false;
309 }
310 }
311 return true;
312 }

399 // given new Distance/UV at Parent, check if any of its neighbours are in the queue,
400 // and if they are, and the new graph distance is shorter, update their queue position
401 // (this is basically the update step of Disjktras algorithm)
402 void UpdateNeighboursSparse(FGraphNode* Parent)
403 {
404 FVector3d ParentPos(GetPositionFunc(Parent->PointID));
405 double ParentDist = Parent->GraphDistance;
406
407 for (int32 NbrPointID : PointSet->VtxVerticesItr(Parent->PointID))
408 {
409 FGraphNode* NbrNode = GetNodeForPointSetID(NbrPointID, true);
410 if (NbrNode->bFrozen)
411 {
412 continue;
413 }
414
415 double LocalDist = Distance(ParentPos, GetPositionFunc(NbrPointID));
416 if (bEnableDistanceWeighting)
417 {
418 int SeedPointID = SeedPoints[Parent->SeedPointID].PointID;
419 LocalDist = GetWeightedDistanceFunc(Parent->PointID, NbrPointID, SeedPointID, LocalDist);
420 }
421 double NbrDist = ParentDist + LocalDist;
422
423 if (Queue.Contains(NbrPointID))
424 {
425 if (NbrDist < NbrNode->GraphDistance)
426 {
427 NbrNode->ParentPointID = Parent->PointID;
428 NbrNode->GraphDistance = NbrDist;
429 NbrNode->SeedPointID = Parent->SeedPointID;
430 Queue.Update(NbrPointID, float(NbrNode->GraphDistance));
431 }
432 }
433 else
434 {
435 NbrNode->ParentPointID = Parent->PointID;
436 NbrNode->GraphDistance = NbrDist;
437 NbrNode->SeedPointID = Parent->SeedPointID;
438 Queue.Insert(NbrPointID, float(NbrNode->GraphDistance));
439 }
440 }
441 }
