592 inline void AudioMixerPlatformLogOnce(const FString& LogDetails, const FString& FileName, int32 LineNumber, ELogVerbosity::Type InVerbosity = ELogVerbosity::Error)
593 {
594#if !NO_LOGGING
595 // Log once to avoid Spam.
596 static FCriticalSection Cs;
597 static TSet<uint32> LogHistory;
598
599 FScopeLock Lock(&Cs);
600 FString Message = FString::Printf(TEXT("Audio Platform Device: %s (File %s, Line %d)"), *LogDetails, *FileName, LineNumber);
601
602 if ((ELogVerbosity::Error == InVerbosity) || (ELogVerbosity::Fatal == InVerbosity))
603 {
604 // Save last error if it was at the error level.
605 LastError = Message;
606 }
607
608 uint32 Hash = GetTypeHash(Message);
609 if (!LogHistory.Contains(Hash))
610 {
611 switch (InVerbosity)
612 {
613 case ELogVerbosity::Fatal:
614 UE_LOGF(LogAudioMixer, Fatal, "%ls", *Message);
615 break;
616
617 case ELogVerbosity::Error:
618 UE_LOGF(LogAudioMixer, Error, "%ls", *Message);
619 break;
620
621 case ELogVerbosity::Warning:
622 UE_LOGF(LogAudioMixer, Warning, "%ls", *Message);
623 break;
624
625 case ELogVerbosity::Display:
626 UE_LOGF(LogAudioMixer, Display, "%ls", *Message);
627 break;
628
629 case ELogVerbosity::Log:
630 UE_LOGF(LogAudioMixer, Log, "%ls", *Message);
631 break;
632
633 case ELogVerbosity::Verbose:
634 UE_LOGF(LogAudioMixer, Verbose, "%ls", *Message);
635 break;
636
637 case ELogVerbosity::VeryVerbose:
638 UE_LOGF(LogAudioMixer, VeryVerbose, "%ls", *Message);
639 break;
640
641 default:
642 UE_LOGF(LogAudioMixer, Error, "%ls", *Message);
643 {
644 static_assert(static_cast<uint8>(ELogVerbosity::NumVerbosity) == 8, "Missing ELogVerbosity case coverage");
645 }
646 break;
647 }
648
649 LogHistory.Add(Hash);
650 }
651#endif
652 }
