14enum class EPSOPrecacheProxyCreationStrategy : uint8
15{
16 // Always create the render proxy regardless of whether the PSO has finished precaching or not.
17 // This will introduce a blocking wait when the proxy is rendered if the PSO is not ready.
18 AlwaysCreate = 0,
19
20 // Delay the creation of the render proxy until the PSO has finished precaching.
21 // This effectively skips drawing components until the PSO is ready, when the proxy will be created.
22 DelayUntilPSOPrecached = 1,
23
24 // Create a render proxy that uses the fallback material if the PSO has not finished precaching by creation time.
25 // The proxy will be re-created with the actual materials once the PSO is ready.
26 // Currently implemented only for static and skinned mesh components, while Niagara components will skip render proxy creation altogether.
27 UseFallbackMaterialUntilPSOPrecached = 2
28};

164template<typename ComponentType, typename PrecachedFinishedCallback>
165void FPSOPrecacheComponentData::SetMaterialPSOPrecacheData(
166 ComponentType* Component,
167 TArray<FMaterialPSOPrecacheRequestID>& InMaterialPSOPrecacheRequestIDs,
168 const FGraphEventArray& PSOPrecacheCompileEvents,
169 PrecachedFinishedCallback InCallback)
170{
171#if UE_WITH_PSO_PRECACHING
172 MaterialPSOPrecacheRequestIDs = MoveTemp(InMaterialPSOPrecacheRequestIDs);
173 bPSOPrecacheCalled = true;
174
175 // If the proxy creation strategy relies on knowing when the precached PSO has been compiled,
176 // schedule a task to mark the render state dirty when all PSOs are compiled so the proxy gets recreated.
177 if (GetPSOPrecacheProxyCreationStrategy() != EPSOPrecacheProxyCreationStrategy::AlwaysCreate)
178 {
179 int32 NewJobID = LatestPSOPrecacheJobSet + 1;
180 LatestPSOPrecacheJobSet = NewJobID;
181 if(!PSOPrecacheCompileEvents.IsEmpty())
182 {
183 TGraphTask<FPSOPrecacheFinishedTask<ComponentType, PrecachedFinishedCallback>>::CreateTask(&PSOPrecacheCompileEvents).ConstructAndDispatchWhenReady(Component, NewJobID, InCallback);
184 }
185 else
186 {
187 // No graph events to wait on, the job set can be considered complete.
188 LatestPSOPrecacheJobSetCompleted = NewJobID;
189 }
190 }
191#endif //UE_WITH_PSO_PRECACHING
