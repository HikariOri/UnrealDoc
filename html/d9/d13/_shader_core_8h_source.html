184struct FShaderTarget
185{
186 // The rest of uint32 holding the bitfields can be left unitialized. Union with a uint32 serves to prevent that to be able to set the whole uint32 value
187 union
188 {
189 uint32 Packed;
190 struct
191 {
192 uint32 Frequency : SF_NumBits;
193 uint32 Platform : SP_NumBits;
194 };
195 };
196
197 FShaderTarget()
198 : Packed(0)
199 {}
200
201 FShaderTarget(EShaderFrequency InFrequency,EShaderPlatform InPlatform)
202 : Packed(0)
203 {
204 Frequency = InFrequency;
205 Platform = InPlatform;
206 }
207
208 friend bool operator==(const FShaderTarget& X, const FShaderTarget& Y)
209 {
210 return X.Frequency == Y.Frequency && X.Platform == Y.Platform;
211 }
212
213 friend FArchive& operator<<(FArchive& Ar,FShaderTarget& Target)
214 {
215 uint32 TargetFrequency = Target.Frequency;
216 uint32 TargetPlatform = Target.Platform;
217 Ar << TargetFrequency << TargetPlatform;
218 if (Ar.IsLoading())
219 {
220 Target.Packed = 0;
221 Target.Frequency = TargetFrequency;
222 Target.Platform = TargetPlatform;
223 }
224 return Ar;
225 }
226
227 EShaderPlatform GetPlatform() const
228 {
229 return (EShaderPlatform)Platform;
230 }
231
232 EShaderFrequency GetFrequency() const
233 {
234 return (EShaderFrequency)Frequency;
235 }
236
237 friend inline uint32 GetTypeHash(FShaderTarget Target)
238 {
239 return ((Target.Frequency << SP_NumBits) | Target.Platform);
240 }
241};

414// Simple wrapper for a uint64 bitfield; doesn't use TBitArray as it is fixed size and doesn't need dynamic memory allocations
415class FShaderCompilerFlags
416{
417public:
418 FShaderCompilerFlags(uint64 InData = 0)
419 : Data(InData)
420 {
421 }
422
423 inline void Append(const FShaderCompilerFlags& In)
424 {
425 Data |= In.Data;
426 }
427
428 inline void Add(uint32 InFlag)
429 {
430 const uint64 FlagBit = (uint64)1 << (uint64)InFlag;
431 Data = Data | FlagBit;
432 }
433
434 inline void Remove(uint32 InFlag)
435 {
436 const uint64 FlagBit = (uint64)1 << (uint64)InFlag;
437 Data = Data & ~FlagBit;
438 }
439
440 inline bool Contains(uint32 InFlag) const
441 {
442 const uint64 FlagBit = (uint64)1 << (uint64)InFlag;
443 return (Data & FlagBit) == FlagBit;
444 }
445
446 inline void Iterate(TFunction<void(uint32)> Callback) const
447 {
448 uint64 Remaining = Data;
449 uint32 Index = 0;
450 while (Remaining)
451 {
452 if (Remaining & (uint64)1)
453 {
454 Callback(Index);
455 }
456 ++Index;
457 Remaining = Remaining >> (uint64)1;
458 }
459 }
460
461 friend inline FArchive& operator << (FArchive& Ar, FShaderCompilerFlags& F)
462 {
463 Ar << F.Data;
464 return Ar;
465 }
466
467 inline uint64 GetData() const
468 {
469 return Data;
470 }
471
472private:
473 uint64 Data;
474};
