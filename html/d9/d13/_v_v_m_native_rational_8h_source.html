12USTRUCT()
13struct FVerseRational
14{
15 GENERATED_BODY()

18 int64 Numerator{0};

20 UPROPERTY(EditAnywhere, Category = Rational)

23 FVerseRational ReduceAndNormalizeSigns() const
24 {
25 int64 A = Numerator;
26 int64 B = Denominator;
27 while (B)
28 {
29 int64 Remainder = A % B;
30 A = B;
31 B = Remainder;
32 }
33
34 // Avoid overflow dividing INT64_MIN / -1.
35 if ((Numerator == INT64_MIN || Denominator == INT64_MIN) && A == -1)
36 {
37 A = 1;
38 }
39
40 FVerseRational Result;
41 Result.Numerator = Numerator / A;
42 Result.Denominator = Denominator / A;
43
44 if (Result.Denominator < 0 && Result.Numerator != INT64_MIN && Result.Denominator != INT64_MIN)
45 {
46 Result.Numerator = -Result.Numerator;
47 Result.Denominator = -Result.Denominator;
48 }
49
50 return Result;
51 }
52
53 friend bool operator==(const FVerseRational& A, const FVerseRational& B)
54 {
55 FVerseRational ReducedA = A.ReduceAndNormalizeSigns();
56 FVerseRational ReducedB = B.ReduceAndNormalizeSigns();
57 return ReducedA.Numerator == ReducedB.Numerator
58 && ReducedA.Denominator == ReducedB.Denominator;
59 }
60
61 friend uint32 GetTypeHash(const FVerseRational& R)
62 {
63 FVerseRational ReducedR = R.ReduceAndNormalizeSigns();
64 // Make sure that integers hash the same whether represented as an int64 or a FVerseRational.
65 const uint32 NumeratorHash = GetTypeHash(ReducedR.Numerator);
66 return ReducedR.Denominator == 1
67 ? NumeratorHash
68 : HashCombineFast(NumeratorHash, GetTypeHash(ReducedR.Denominator));
69 }
70

77 DenominatorField << R.Denominator;
78 }
