274 [[nodiscard]] static UE_FORCEINLINE_HINT int32 RandHelper(int32 A)
275 {
276 // Note that on some platforms RAND_MAX is a large number so we cannot do ((rand()/(RAND_MAX+1)) * A)
277 // or else we may include the upper bound results, which should be excluded.
278 return A > 0 ? Min(TruncToInt(FRand() * (float)A), A - 1) : 0;
279 }
280
281 [[nodiscard]] static UE_FORCEINLINE_HINT int64 RandHelper64(int64 A)
282 {
283 // Note that on some platforms RAND_MAX is a large number so we cannot do ((rand()/(RAND_MAX+1)) * A)
284 // or else we may include the upper bound results, which should be excluded.
285 return A > 0 ? Min<int64>(TruncToInt(FRand() * (float)A), A - 1) : 0;
286 }
287
289 [[nodiscard]] static inline int32 RandRange(int32 Min, int32 Max)
290 {
291 const int32 Range = (Max - Min) + 1;
292 return Min + RandHelper(Range);
293 }
294
295 [[nodiscard]] static inline int64 RandRange(int64 Min, int64 Max)
296 {
297 const int64 Range = (Max - Min) + 1;
298 return Min + RandHelper64(Range);
299 }
300
302 [[nodiscard]] static UE_FORCEINLINE_HINT float RandRange(float InMin, float InMax)
303 {
304 return FRandRange(InMin, InMax);
305 }

426 template<typename FloatType, typename IntegralType, IntegralType SignedBit>
427 static inline bool TIsNearlyEqualByULP(FloatType A, FloatType B, int32 MaxUlps)
428 {
429 // Any comparison with NaN always fails.
430 if (FMath::IsNaN(A) || FMath::IsNaN(B))
431 {
432 return false;
433 }
434
435 // If either number is infinite, then ignore ULP and do a simple equality test.
436 // The rationale being that two infinities, of the same sign, should compare the same
437 // no matter the ULP, but FLT_MAX and Inf should not, even if they're neighbors in
438 // their bit representation.
439 if (!FMath::IsFinite(A) || !FMath::IsFinite(B))
440 {
441 return A == B;
442 }
443
444 // Convert the integer representation of the float from sign + magnitude to
445 // a signed number representation where 0 is 1 << 31. This allows us to compare
446 // ULP differences around zero values.
447 auto FloatToSignedNumber = [](IntegralType V) {
448 if (V & SignedBit)
449 {
450 return ~V + 1;
451 }
452 else
453 {
454 return SignedBit | V;
455 }
456 };
457
458 IntegralType SNA = FloatToSignedNumber(FMath::AsUInt(A));
459 IntegralType SNB = FloatToSignedNumber(FMath::AsUInt(B));
460 IntegralType Distance = (SNA >= SNB) ? (SNA - SNB) : (SNB - SNA);
461 return Distance <= IntegralType(MaxUlps);
462 }

609 template <UE::CFloatingPoint T>
610 [[nodiscard]] static constexpr UE_FORCEINLINE_HINT T Wrap(const T X, const T Min, const T Max)
611 {
612 // Our asserts are not constexpr-friendly yet
613 // checkSlow(Min <= Max);
614
615 T Size = Max - Min;
616 if (Size == 0)
617 {
618 // Guard against zero-sized ranges causing division by zero.
619 return Max;
620 }
621
622 T EndVal = X;
623 if (EndVal < Min)
624 {
625 T Mod = FMath::Modulo(Min - EndVal, Size);
626 EndVal = (Mod != (T)0) ? Max - Mod : Min;
627 }
628 else if (EndVal > Max)
629 {
630 T Mod = FMath::Modulo(EndVal - Max, Size);
631 EndVal = (Mod != (T)0) ? Min + Mod : Max;
632 }
633 return EndVal;
634 }

637 template <UE::CIntegral T>
638 [[nodiscard]] static constexpr inline T WrapExclusive(const T X, const T Min, const T Max)
639 {
640 // Use unsigned type allow for large ranges which don't overflow on subtraction.
641 using SizeType = std::make_unsigned_t<T>;
642
643 // Our asserts are not constexpr-friendly yet
644 // checkSlow(Min <= Max);
645
646 SizeType Size = (SizeType)((SizeType)Max - (SizeType)Min);
647 if (Size == 0)
648 {
649 // Guard against zero-sized ranges causing division by zero.
650 return Max;
651 }
652
653 SizeType Mod;
654 if (X < Min)
655 {
656 Mod = (SizeType)FMath::Modulo((SizeType)((SizeType)Min - (SizeType)X), Size);
657 if (Mod > 0)
658 {
659 Mod = (SizeType)(Size - Mod);
660 }
661 }
662 else
663 {
664 Mod = (SizeType)FMath::Modulo((SizeType)((SizeType)X - (SizeType)Min), Size);
665 }

759 template <UE::CFloatingPoint T>
760 static constexpr inline void SinCos(std::decay_t<T>* ScalarSin, std::decay_t<T>* ScalarCos, T Value )
761 {
762 // Map Value to y in [-pi,pi], x = 2*pi*quotient + remainder.
763 T quotient = (UE_INV_PI*0.5f)*Value;
764 if (Value >= 0.0f)
765 {
766 quotient = (T)((int64)(quotient + 0.5f));
767 }
768 else
769 {
770 quotient = (T)((int64)(quotient - 0.5f));
771 }
772 T y = Value - UE_TWO_PI * quotient;
773
774 // Map y to [-pi/2,pi/2] with sin(y) = sin(Value).
775 T sign;
776 if (y > UE_HALF_PI)
777 {
778 y = UE_PI - y;
779 sign = -1.0f;
780 }
781 else if (y < -UE_HALF_PI)
782 {
783 y = -UE_PI - y;
784 sign = -1.0f;
785 }
786 else
787 {
788 sign = +1.0f;
789 }
790
791 T y2 = y * y;
792
793 // 11-degree minimax approximation
794 *ScalarSin = ( ( ( ( (-2.3889859e-08f * y2 + 2.7525562e-06f) * y2 - 0.00019840874f ) * y2 + 0.0083333310f ) * y2 - 0.16666667f ) * y2 + 1.0f ) * y;
795
796 // 10-degree minimax approximation
797 T p = ( ( ( ( -2.6051615e-07f * y2 + 2.4760495e-05f ) * y2 - 0.0013888378f ) * y2 + 0.041666638f ) * y2 - 0.5f ) * y2 + 1.0f;
798 *ScalarCos = sign*p;
799 }

820 // It is important to use exactly the same value in all places inside this function to ensure that FastASin(0.0f) == 0.0f.
821 // For comparison:
822 // HALF_PI == 1.57079632679f == 0x3fC90FDB

832 {
833 // Clamp input to [-1,1].
834 const bool nonnegative = (Value >= 0.0f);
835 const float x = FMath::Abs(Value);
836 float omx = 1.0f - x;
837 if (omx < 0.0f)
838 {
839 omx = 0.0f;

844 result *= root; // acos(|x|)
845 // acos(x) = pi - acos(-x) when x < 0, asin(x) = pi/2 - acos(x)
846 return (nonnegative ? FASTASIN_HALF_PI - result : result - FASTASIN_HALF_PI);
847 }

1402 /*
1403 * Cubic Catmull-Rom Spline interpolation. Based on http://www.cemyuksel.com/research/catmullrom_param/catmullrom.pdf
1404 * Curves are guaranteed to pass through the control points and are easily chained together.
1405 * Equation supports abitrary parameterization. eg. Uniform=0,1,2,3 ; chordal= |Pn - Pn-1| ; centripetal = |Pn - Pn-1|^0.5
1406 * P0 - The control point preceding the interpolation range.
1407 * P1 - The control point starting the interpolation range.
1408 * P2 - The control point ending the interpolation range.
1409 * P3 - The control point following the interpolation range.
1410 * T0-3 - The interpolation parameters for the corresponding control points.
1411 * T - The interpolation factor in the range 0 to 1. 0 returns P1. 1 returns P2.
1412 */
1413 template< class U >
1414 [[nodiscard]] static constexpr inline U CubicCRSplineInterp(const U& P0, const U& P1, const U& P2, const U& P3, const float T0, const float T1, const float T2, const float T3, const float T)
1415 {
1416 //Based on http://www.cemyuksel.com/research/catmullrom_param/catmullrom.pdf
1417 float InvT1MinusT0 = 1.0f / (T1 - T0);
1418 U L01 = ( P0 * ((T1 - T) * InvT1MinusT0) ) + ( P1 * ((T - T0) * InvT1MinusT0) );
1419 float InvT2MinusT1 = 1.0f / (T2 - T1);
1420 U L12 = ( P1 * ((T2 - T) * InvT2MinusT1) ) + ( P2 * ((T - T1) * InvT2MinusT1) );
1421 float InvT3MinusT2 = 1.0f / (T3 - T2);
1422 U L23 = ( P2 * ((T3 - T) * InvT3MinusT2) ) + ( P3 * ((T - T2) * InvT3MinusT2) );
1423
1424 float InvT2MinusT0 = 1.0f / (T2 - T0);
1425 U L012 = ( L01 * ((T2 - T) * InvT2MinusT0) ) + ( L12 * ((T - T0) * InvT2MinusT0) );
1426 float InvT3MinusT1 = 1.0f / (T3 - T1);
1427 U L123 = ( L12 * ((T3 - T) * InvT3MinusT1) ) + ( L23 * ((T - T1) * InvT3MinusT1) );
1428
1429 return ( ( L012 * ((T2 - T) * InvT2MinusT1) ) + ( L123 * ((T - T1) * InvT2MinusT1) ) );
1430 }

1608 static constexpr void CriticallyDampedSmoothing(
1609 T& InOutValue,
1610 T& InOutValueRate,
1611 const T& InTargetValue,
1612 const T& InTargetValueRate,
1613 const float InDeltaTime,
1614 const float InSmoothingTime)
1615 {
1616 if (InSmoothingTime > UE_KINDA_SMALL_NUMBER)
1617 {
1618 // The closed form solution for critically damped motion towards zero is:
1619 //
1620 // x = ( x0 + t (v0 + w x0) ) exp(-w t)
1621 //
1622 // where w is the natural frequency, x0, v0 are x and dx/dt at time 0 (e.g. Brian Stone - Chatter and Machine Tools pdf)
1623 //
1624 // Differentiate to get velocity (remember d(v0)/dt = 0):
1625 //
1626 // v = ( v0 - w t (v0 + w x0) ) exp(-w t)
1627 //
1628 // We're damping towards a target, so convert - i.e. x = value - target
1629 //
1630 // The target velocity turns into an offset to the position
1631 T AdjustedTarget = InTargetValue + InTargetValueRate * InSmoothingTime;
1632
1633 const float W = 2.0f / InSmoothingTime;
1634 const float A = W * InDeltaTime;
1635 // Approximation requires A < 1, so DeltaTime < SmoothingTime / 2
1636 const float Exp = InvExpApprox(A);
1637 T X0 = InOutValue - AdjustedTarget;
1638 const T B = (InOutValueRate + X0 * W) * InDeltaTime;
1639 InOutValue = AdjustedTarget + (X0 + B) * Exp;
1640 InOutValueRate = (InOutValueRate - B * W) * Exp;
1641 }
1642 else if (InDeltaTime > 0.0f)
1643 {
1644 InOutValueRate = (InTargetValue - InOutValue) / InDeltaTime;
1645 InOutValue = InTargetValue;
1646 }
1647 else
1648 {
1649 InOutValue = InTargetValue;
1650 InOutValueRate = T(0);
1651 }
1652 }

1669 template< class T >
1670 static void SpringDamper(
1671 T& InOutValue,
1672 T& InOutValueRate,
1673 const T& InTargetValue,
1674 const T& InTargetValueRate,
1675 const float InDeltaTime,
1676 const float InUndampedFrequency,
1677 const float InDampingRatio)
1678 {
1679 if (InDeltaTime <= 0.0f)
1680 {
1681 return;
1682 }
1683
1684 float W = InUndampedFrequency * UE_TWO_PI;
1685 // Handle special cases
1686 if (W < UE_SMALL_NUMBER) // no strength which means no damping either
1687 {
1688 InOutValue += InOutValueRate * InDeltaTime;
1689 return;
1690 }
1691 else if (InDampingRatio < UE_SMALL_NUMBER) // No damping at all
1692 {
1693 T Err = InOutValue - InTargetValue;
1694 const T B = InOutValueRate / W;
1695 float S, C;
1696 FMath::SinCos(&S, &C, W * InDeltaTime);
1697 InOutValue = InTargetValue + Err * C + B * S;
1698 InOutValueRate = InOutValueRate * C - Err * (W * S);
1699 return;
1700 }
1701
1702 // Target velocity turns into an offset to the position
1703 float SmoothingTime = 2.0f / W;
1704 T AdjustedTarget = InTargetValue + InTargetValueRate * (InDampingRatio * SmoothingTime);
1705 T Err = InOutValue - AdjustedTarget;
1706
1707 // Handle the cases separately
1708 if (InDampingRatio > 1.0f) // Overdamped
1709 {
1710 const float WD = W * FMath::Sqrt(FMath::Square(InDampingRatio) - 1.0f);
1711 const T C2 = -(InOutValueRate + (W * InDampingRatio - WD) * Err) / (2.0f * WD);
1712 const T C1 = Err - C2;
1713 const float A1 = (WD - InDampingRatio * W);
1714 const float A2 = -(WD + InDampingRatio * W);
1715 // Note that A1 and A2 will always be negative. We will use an approximation for 1/Exp(-A * DeltaTime).
1716 const float A1_DT = -A1 * InDeltaTime;
1717 const float A2_DT = -A2 * InDeltaTime;
1718 // This approximation in practice will be good for all DampingRatios
1719 const float E1 = InvExpApprox(A1_DT);
1720 // As DampingRatio gets big, this approximation gets worse, but mere inaccuracy for overdamped motion is
1721 // not likely to be important, since we end up with 1 / BigNumber
1722 const float E2 = InvExpApprox(A2_DT);
1723 InOutValue = AdjustedTarget + E1 * C1 + E2 * C2;
1724 InOutValueRate = E1 * C1 * A1 + E2 * C2 * A2;
1725 }
1726 else if (InDampingRatio < 1.0f) // Underdamped
1727 {
1728 const float WD = W * FMath::Sqrt(1.0f - FMath::Square(InDampingRatio));
1729 const T A = Err;
1730 const T B = (InOutValueRate + Err * (InDampingRatio * W)) / WD;
1731 float S, C;
1732 FMath::SinCos(&S, &C, WD * InDeltaTime);
1733 const float E0 = InDampingRatio * W * InDeltaTime;
1734 // Needs E0 < 1 so DeltaTime < SmoothingTime / (2 * DampingRatio * Sqrt(1 - DampingRatio^2))
1735 const float E = InvExpApprox(E0);
1736 InOutValue = E * (A * C + B * S);
1737 InOutValueRate = -InOutValue * InDampingRatio * W;
1738 InOutValueRate += E * (B * (WD * C) - A * (WD * S));
1739 InOutValue += AdjustedTarget;
1740 }
1741 else // Critical damping
1742 {
1743 const T& C1 = Err;
1744 T C2 = InOutValueRate + Err * W;
1745 const float E0 = W * InDeltaTime;
1746 // Needs E0 < 1 so InDeltaTime < SmoothingTime / 2
1747 float E = InvExpApprox(E0);
1748 InOutValue = AdjustedTarget + (C1 + C2 * InDeltaTime) * E;
1749 InOutValueRate = (C2 - C1 * W - C2 * (W * InDeltaTime)) * E;
1750 }
1751 }
