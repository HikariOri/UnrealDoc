699{
700 TStaticArray<FRenderTargetBinding, MaxSimultaneousRenderTargets> Output;

720 {
721 return Output[Index];

724 template <typename TFunction>
725 void Enumerate(TFunction Function)
726 {
727 for (int32 Index = 0; Index < Output.Num(); Index++)
728 {
729 if (Output[Index].GetTexture())
730 {
731 Function(Output[Index]);
732 }
733 else
734 {
735 break;
736 }
737 }
738 }

740 template <typename TFunction>
741 void Enumerate(TFunction Function) const
742 {
743 for (int32 Index = 0; Index < Output.Num(); Index++)
744 {
745 if (Output[Index].GetTexture())
746 {
747 Function(Output[Index]);
748 }
749 else
750 {
751 break;
752 }
753 }
754 }
755
756 uint32 GetActiveCount() const
757 {
758 uint32 Count = 0;
759 for (; Output[Count].GetTexture() != nullptr; ++Count) {};
760 return Count;
761 }
762
763 bool CanMergeBefore(const FRenderTargetBindingSlots& Other) const
764 {
765 for (int32 Index = 0; Index < MaxSimultaneousRenderTargets; ++Index)
766 {
767 if (!Output[Index].CanMergeBefore(Other.Output[Index]))
768 {
769 return false;
770 }
771 }
772
773 if (!DepthStencil.CanMergeBefore(Other.DepthStencil))
774 {
775 return false;
776 }
777
778 if (ResolveRect != Other.ResolveRect ||
779 (NumOcclusionQueries != Other.NumOcclusionQueries && Other.NumOcclusionQueries != 0) ||
780 SubpassHint != Other.SubpassHint ||
781 MultiViewCount != Other.MultiViewCount ||
782 ShadingRateTexture != Other.ShadingRateTexture)
783 {
784 return false;
785 }
786
787 return true;
788 }
