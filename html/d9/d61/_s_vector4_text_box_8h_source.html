155 NumericType GetValueType(const FString& InString) const
156 {
157 static_assert(std::is_floating_point_v<NumericType>);
158
159 if constexpr (std::is_same_v<float, NumericType>)
160 {
161 return FCString::Atof(*InString);
162 }
163 else if constexpr (std::is_same_v<double, NumericType>)
164 {
165 return FCString::Atod(*InString);
166 }
167 else
168 {
169 return NumericType{};
170 }
171 }
