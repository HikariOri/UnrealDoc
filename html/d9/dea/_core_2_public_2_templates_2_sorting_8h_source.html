96template<class T, class PREDICATE_CLASS>
97void Merge(T* Out, T* In, const int32 Mid, const int32 Num, const PREDICATE_CLASS& Predicate)
98{
99 int32 Merged = 0;
100 int32 Picked;
101 int32 A = 0, B = Mid;
102
103 while (Merged < Num)
104 {
105 if (Merged != B && (B >= Num || !Predicate(In[B], In[A])))
106 {
107 Picked = A++;
108 }
109 else
110 {
111 Picked = B++;
112 }
113
114 Out[Merged] = In[Picked];
115
116 ++Merged;
117 }
118}

152template <class TGCDPolicy>
153class TJugglingRotation
154{
155public:
164 template <class T>
165 static void Rotate(T* First, const int32 From, const int32 To, const int32 Amount)
166 {
167 if (Amount == 0)
168 {
169 return;
170 }
171
172 auto Num = To - From;
173 auto GCD = TGCDPolicy::GCD(Num, Amount);
174 auto CycleSize = Num / GCD;
175
176 for (int32 Index = 0; Index < GCD; ++Index)
177 {
178 T BufferObject = MoveTemp(First[From + Index]);
179 int32 IndexToFill = Index;
180
181 for (int32 InCycleIndex = 0; InCycleIndex < CycleSize; ++InCycleIndex)
182 {
183 IndexToFill = (IndexToFill + Amount) % Num;
184 Exchange(First[From + IndexToFill], BufferObject);
185 }
186 }
187 }
188};

195template <class TRotationPolicy>
196class TRotationInPlaceMerge
197{
198public:
207 template <class T, class PREDICATE_CLASS>
208 static void Merge(T* First, const int32 Mid, const int32 Num, const PREDICATE_CLASS& Predicate)
209 {
210 int32 AStart = 0;
211 int32 BStart = Mid;
212
213 while (AStart < BStart && BStart < Num)
214 {
215 // Index after the last value == First[BStart]
216 int32 NewAOffset = (int32)AlgoImpl::UpperBoundInternal(First + AStart, BStart - AStart, First[BStart], FIdentityFunctor(), Predicate);
217 AStart += NewAOffset;
218
219 if (AStart >= BStart) // done
220 break;
221
222 // Index of the first value == First[AStart]
223 int32 NewBOffset = (int32)AlgoImpl::LowerBoundInternal(First + BStart, Num - BStart, First[AStart], FIdentityFunctor(), Predicate);
224 TRotationPolicy::Rotate(First, AStart, BStart + NewBOffset, NewBOffset);
225 BStart += NewBOffset;
226 AStart += NewBOffset + 1;
227 }
228 }
229};

237template <class TMergePolicy, int32 MinMergeSubgroupSize = 2>
238class TMergeSort
239{
240public:
248 template<class T, class PREDICATE_CLASS>
249 static void Sort(T* First, const int32 Num, const PREDICATE_CLASS& Predicate)
250 {
251 int32 SubgroupStart = 0;
252
253 if (MinMergeSubgroupSize > 1)
254 {
255 if (MinMergeSubgroupSize > 2)
256 {
257 // First pass with simple bubble-sort.
258 do
259 {
260 int32 GroupEnd = FPlatformMath::Min(SubgroupStart + MinMergeSubgroupSize, Num);
261 do
262 {
263 for (int32 It = SubgroupStart; It < GroupEnd - 1; ++It)
264 {
265 if (Predicate(First[It + 1], First[It]))
266 {
267 Exchange(First[It], First[It + 1]);
268 }
269 }
270 GroupEnd--;
271 } while (GroupEnd - SubgroupStart > 1);
272
273 SubgroupStart += MinMergeSubgroupSize;
274 } while (SubgroupStart < Num);
275 }
276 else
277 {
278 for (int32 Subgroup = 0; Subgroup < Num; Subgroup += 2)
279 {
280 if (Subgroup + 1 < Num && Predicate(First[Subgroup + 1], First[Subgroup]))
281 {
282 Exchange(First[Subgroup], First[Subgroup + 1]);
283 }
284 }
285 }
286 }
287
288 int32 SubgroupSize = MinMergeSubgroupSize;
289 while (SubgroupSize < Num)
290 {
291 SubgroupStart = 0;
292 do
293 {
294 TMergePolicy::Merge(
295 First + SubgroupStart,
296 SubgroupSize,
297 FPlatformMath::Min(SubgroupSize << 1, Num - SubgroupStart),
298 Predicate);
299 SubgroupStart += SubgroupSize << 1;
300 } while (SubgroupStart < Num);
301
302 SubgroupSize <<= 1;
303 }
304 }
305};

330template< typename ValueType, typename CountType, class SortKeyClass >
331void RadixSort32( ValueType* RESTRICT Dst, ValueType* RESTRICT Src, CountType Num, const SortKeyClass& SortKey )
332{
333 CountType Histograms[ 1024 + 2048 + 2048 ];
334 CountType* RESTRICT Histogram0 = Histograms + 0;
335 CountType* RESTRICT Histogram1 = Histogram0 + 1024;
336 CountType* RESTRICT Histogram2 = Histogram1 + 2048;
337
338 FMemory::Memzero( Histograms, sizeof( Histograms ) );
339
340 {
341 // Parallel histogram generation pass
342 const ValueType* RESTRICT s = (const ValueType* RESTRICT)Src;
343 for( CountType i = 0; i < Num; i++ )
344 {
345 uint32 Key = SortKey( s[i] );
346 Histogram0[ ( Key >> 0 ) & 1023 ]++;
347 Histogram1[ ( Key >> 10 ) & 2047 ]++;
348 Histogram2[ ( Key >> 21 ) & 2047 ]++;
349 }
350 }
351 {
352 // Prefix sum
353 // Set each histogram entry to the sum of entries preceding it
354 CountType Sum0 = 0;
355 CountType Sum1 = 0;
356 CountType Sum2 = 0;
357 for( CountType i = 0; i < 1024; i++ )
358 {
359 CountType t;
360 t = Histogram0[i] + Sum0; Histogram0[i] = Sum0 - 1; Sum0 = t;
361 t = Histogram1[i] + Sum1; Histogram1[i] = Sum1 - 1; Sum1 = t;
362 t = Histogram2[i] + Sum2; Histogram2[i] = Sum2 - 1; Sum2 = t;
363 }
364 for( CountType i = 1024; i < 2048; i++ )
365 {
366 CountType t;
367 t = Histogram1[i] + Sum1; Histogram1[i] = Sum1 - 1; Sum1 = t;
368 t = Histogram2[i] + Sum2; Histogram2[i] = Sum2 - 1; Sum2 = t;
369 }
370 }
371 {
372 // Sort pass 1
373 const ValueType* RESTRICT s = (const ValueType* RESTRICT)Src;
374 ValueType* RESTRICT d = Dst;
375 for( CountType i = 0; i < Num; i++ )
376 {
377 ValueType Value = s[i];
378 uint32 Key = SortKey( Value );
379 d[ ++Histogram0[ ( (Key >> 0) & 1023 ) ] ] = Value;
380 }
381 }
382 {
383 // Sort pass 2
384 const ValueType* RESTRICT s = (const ValueType* RESTRICT)Dst;
385 ValueType* RESTRICT d = Src;
386 for( CountType i = 0; i < Num; i++ )
387 {
388 ValueType Value = s[i];
389 uint32 Key = SortKey( Value );
390 d[ ++Histogram1[ ( (Key >> 10) & 2047 ) ] ] = Value;
391 }
392 }
393 {
394 // Sort pass 3
395 const ValueType* RESTRICT s = (const ValueType* RESTRICT)Src;
396 ValueType* RESTRICT d = Dst;
397 for( CountType i = 0; i < Num; i++ )
398 {
399 ValueType Value = s[i];
400 uint32 Key = SortKey( Value );
401 d[ ++Histogram2[ ( (Key >> 21) & 2047 ) ] ] = Value;
402 }
403 }

457{

534 }
