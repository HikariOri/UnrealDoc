16/*
17 * Simple dynamic array that can only grow, avoids allocation code
18 * directly calls mmap this is to avoid recursive calls when collecting memory address
19 */
20template <typename T>
21class UnixLowLevelDynamicArray
22{
23public:
24 UnixLowLevelDynamicArray()
25 {
26 Elements = static_cast<T*>(mmap(nullptr, Capacity * sizeof(T), PROT_READ | PROT_WRITE, MAP_PRIVATE | MAP_ANON, -1, 0));
27 }
28
29 ~UnixLowLevelDynamicArray()
30 {
31 munmap(Elements, Capacity);
32 }
33
34 void Emplace(T&& Elem)
35 {
36 // MMap most likely failed in the ctor, cant do much with out memory
37 if (Elements == nullptr)
38 {
39 return;
40 }
41
42 if (Size + 1 > Capacity)
43 {
44 SIZE_T OldCapacity = Capacity;
45 Capacity *= 2;
46
47 T* NewElements = static_cast<T*>(mmap(nullptr, Capacity * sizeof(T), PROT_READ | PROT_WRITE, MAP_PRIVATE | MAP_ANON, -1, 0));
48 T* OldElements = Elements;
49
50 // MMap has failed
51 if (NewElements == nullptr)
52 {
53 return;
54 }
55
56 FMemory::Memcpy(NewElements, Elements, Size * sizeof(T));
57 Elements = MoveTemp(NewElements);
58
59 munmap(OldElements, OldCapacity);
60 }
61
62 Elements[Size] = MoveTemp(Elem);
63 Size++;
64 }
65
66 T* begin() const
67 {
68 return Elements;
69 }
70
71 T* end() const
72 {
73 return Elements + Size;
74 }
75
76 T& Get(SIZE_T Index)
77 {
78 return *Elements[Index];
79 }
80
81 SIZE_T GetSize()
82 {
83 return Size;
84 }
85
86private:
87 SIZE_T Capacity = 1000U;
88 SIZE_T Size = 0U;
89 T* Elements = nullptr;
90};
