158 // ============================================================================
159 // FQueryExpression — composite expression of column conditions and relation predicates.
160 //
161 // Build one with operator&& between:
162 // - TColumn<T>() or FConditions (column conditions, evaluated at archetype level)
163 // - IsDescendantOf / IsSubjectOf / etc. free functions (relation predicates, per-entity)
164 // All conditions are AND-composed.
165 //
166 // Example:
167 // using namespace UE::Editor::DataStorage::Queries;
168 // Select(...).Where(TColumn<FMyColumn>() && IsDescendantOf(MyRelation, RootRow)).Compile()
169 // ============================================================================
170
176 struct FQueryExpression
177 {
179 TOptional<FConditions> ColumnConditions;
181 TArray<FQueryDescription::FRelationPredicate, TInlineAllocator<2>> RelationAtoms;
182 };
