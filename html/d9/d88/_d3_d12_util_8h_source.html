92struct FShaderRegisterCounts
93{
94 uint8 SamplerCount;
95 uint8 ConstantBufferCount;
96 uint8 ShaderResourceCount;
97 uint8 UnorderedAccessCount;
98
99 inline bool operator==(const FShaderRegisterCounts& RHS) const
100 {
101 return SamplerCount == RHS.SamplerCount &&
102 ConstantBufferCount == RHS.ConstantBufferCount &&
103 ShaderResourceCount == RHS.ShaderResourceCount &&
104 UnorderedAccessCount == RHS.UnorderedAccessCount;
105 }
106
107 friend uint32 GetTypeHash(const FShaderRegisterCounts& Counts)
108 {
109 uint32 Hash = GetTypeHash(Counts.SamplerCount);
110 Hash = HashCombineFast(Hash, GetTypeHash(Counts.ConstantBufferCount));
111 Hash = HashCombineFast(Hash, GetTypeHash(Counts.ShaderResourceCount));
112 Hash = HashCombineFast(Hash, GetTypeHash(Counts.UnorderedAccessCount));
113 return Hash;
114 }
115};

139 {

146 friend uint32 GetTypeHash(const FD3D12QuantizedBoundShaderState& State)
147 {
148 uint32 Hash = GetTypeHash(State.ShaderBindingLayout);
149 for (int32 Index = 0; Index < SV_ShaderVisibilityCount; ++Index)
150 {
151 Hash = HashCombineFast(Hash, GetTypeHash(State.RegisterCounts[Index]));
152 }
153 Hash = HashCombineFast(Hash, GetTypeHash(State.RootSignatureType));
154 Hash = HashCombineFast(Hash, GetTypeHash(State.Flags));
155 return Hash;
156 }

274template <class Type>
275struct FThreadsafeQueue
276{
277private:
278 mutable FCriticalSection SynchronizationObject; // made this mutable so this class can have const functions and still be thread safe
279 TQueue<Type> Items;
280 uint32 Size = 0;
281public:
282
283 inline const uint32 GetSize() const { return Size; }
284
285 void Enqueue(const Type& Item)
286 {
287 FScopeLock ScopeLock(&SynchronizationObject);
288 Items.Enqueue(Item);
289 Size++;
290 }
291
292 bool Dequeue(Type& Result)
293 {
294 FScopeLock ScopeLock(&SynchronizationObject);
295 Size--;
296 return Items.Dequeue(Result);
297 }
298
299 template <typename CompareFunc>
300 bool Dequeue(Type& Result, const CompareFunc& Func)
301 {
302 FScopeLock ScopeLock(&SynchronizationObject);
303
304 if (Items.Peek(Result))
305 {
306 if (Func(Result))
307 {
308 Items.Dequeue(Result);
309 Size--;
310
311 return true;
312 }
313 }
314
315 return false;
316 }
317
318 template <typename ResultType, typename CompareFunc>
319 bool BatchDequeue(TArray<ResultType>& Result, const CompareFunc& Func, uint32 MaxItems)
320 {
321 FScopeLock ScopeLock(&SynchronizationObject);
322
323 uint32 i = 0;
324 Type Item;
325 while (Items.Peek(Item) && i <= MaxItems)
326 {
327 if (Func(Item))
328 {
329 Items.Dequeue(Item);
330 Size--;
331 Result.Emplace(Item);
332
333 i++;
334 }
335 else
336 {
337 break;
338 }
339 }
340
341 return i > 0;
342 }
343
344 bool Peek(Type& Result)
345 {
346 FScopeLock ScopeLock(&SynchronizationObject);
347 return Items.Peek(Result);
348 }
349
350 bool IsEmpty()
351 {
352 FScopeLock ScopeLock(&SynchronizationObject);
353 return Items.IsEmpty();
354 }
355
356 void Empty()
357 {
358 FScopeLock ScopeLock(&SynchronizationObject);
359
360 Type Result;
361 while (Items.Dequeue(Result)) {}
362 }
363};

365inline bool IsCPUWritable(D3D12_HEAP_TYPE HeapType, const D3D12_HEAP_PROPERTIES *pCustomHeapProperties = nullptr)
366{
367 check(HeapType == D3D12_HEAP_TYPE_CUSTOM ? pCustomHeapProperties != nullptr : true);
368 return HeapType == D3D12_HEAP_TYPE_UPLOAD ||
369 (HeapType == D3D12_HEAP_TYPE_CUSTOM &&
370 (pCustomHeapProperties->CPUPageProperty == D3D12_CPU_PAGE_PROPERTY_WRITE_COMBINE || pCustomHeapProperties->CPUPageProperty == D3D12_CPU_PAGE_PROPERTY_WRITE_BACK));
371}
372
373inline bool IsGPUOnly(D3D12_HEAP_TYPE HeapType, const D3D12_HEAP_PROPERTIES *pCustomHeapProperties = nullptr)
374{
375 check(HeapType == D3D12_HEAP_TYPE_CUSTOM ? pCustomHeapProperties != nullptr : true);
376 return HeapType == D3D12_HEAP_TYPE_DEFAULT ||
377 (HeapType == D3D12_HEAP_TYPE_CUSTOM &&
378 (pCustomHeapProperties->CPUPageProperty == D3D12_CPU_PAGE_PROPERTY_NOT_AVAILABLE));
379}
380
381inline bool IsCPUAccessible(D3D12_HEAP_TYPE HeapType, const D3D12_HEAP_PROPERTIES* pCustomHeapProperties = nullptr)
382{
383 return !IsGPUOnly(HeapType, pCustomHeapProperties);
384}

406static void Get4KTileShape(D3D12_TILE_SHAPE* pTileShape, DXGI_FORMAT DXGIFormat, EPixelFormat UEFormat, D3D12_RESOURCE_DIMENSION Dimension, uint32 SampleCount)
407{
408 //Bits per unit
409 uint32 BPU = GPixelFormats[UEFormat].BlockBytes * 8;
410
411 switch (Dimension)
412 {
413 case D3D12_RESOURCE_DIMENSION_BUFFER:
414 case D3D12_RESOURCE_DIMENSION_TEXTURE1D:
415 {
416 check(!UE::DXGIUtilities::IsBlockCompressedFormat(DXGIFormat));
417 pTileShape->WidthInTexels = (BPU == 0) ? 4096 : 4096 * 8 / BPU;
418 pTileShape->HeightInTexels = 1;
419 pTileShape->DepthInTexels = 1;
420 }
421 break;
422 case D3D12_RESOURCE_DIMENSION_TEXTURE2D:
423 {
424 pTileShape->DepthInTexels = 1;
425 if (UE::DXGIUtilities::IsBlockCompressedFormat(DXGIFormat))
426 {
427 // Currently only supported block sizes are 64 and 128.
428 // These equations calculate the size in texels for a tile. It relies on the fact that 16*16*16 blocks fit in a tile if the block size is 128 bits.
429 check(BPU == 64 || BPU == 128);
430 pTileShape->WidthInTexels = 16 * UE::DXGIUtilities::GetWidthAlignment(DXGIFormat);
431 pTileShape->HeightInTexels = 16 * UE::DXGIUtilities::GetHeightAlignment(DXGIFormat);
432 if (BPU == 64)
433 {
434 // If bits per block are 64 we double width so it takes up the full tile size.
435 // This is only true for BC1 and BC4
436 check((DXGIFormat >= DXGI_FORMAT_BC1_TYPELESS && DXGIFormat <= DXGI_FORMAT_BC1_UNORM_SRGB) ||
437 (DXGIFormat >= DXGI_FORMAT_BC4_TYPELESS && DXGIFormat <= DXGI_FORMAT_BC4_SNORM));
438 pTileShape->WidthInTexels *= 2;
439 }
440 }
441 else
442 {
443 if (BPU <= 8)
444 {
445 pTileShape->WidthInTexels = 64;
446 pTileShape->HeightInTexels = 64;
447 }
448 else if (BPU <= 16)
449 {
450 pTileShape->WidthInTexels = 64;
451 pTileShape->HeightInTexels = 32;
452 }
453 else if (BPU <= 32)
454 {
455 pTileShape->WidthInTexels = 32;
456 pTileShape->HeightInTexels = 32;
457 }
458 else if (BPU <= 64)
459 {
460 pTileShape->WidthInTexels = 32;
461 pTileShape->HeightInTexels = 16;
462 }
463 else if (BPU <= 128)
464 {
465 pTileShape->WidthInTexels = 16;
466 pTileShape->HeightInTexels = 16;
467 }
468 else
469 {
470 check(false);
471 }
472
473 if (SampleCount <= 1)
474 { /* Do nothing */
475 }
476 else if (SampleCount <= 2)
477 {
478 pTileShape->WidthInTexels /= 2;
479 pTileShape->HeightInTexels /= 1;
480 }
481 else if (SampleCount <= 4)
482 {
483 pTileShape->WidthInTexels /= 2;
484 pTileShape->HeightInTexels /= 2;
485 }
486 else if (SampleCount <= 8)
487 {
488 pTileShape->WidthInTexels /= 4;
489 pTileShape->HeightInTexels /= 2;
490 }
491 else if (SampleCount <= 16)
492 {
493 pTileShape->WidthInTexels /= 4;
494 pTileShape->HeightInTexels /= 4;
495 }
496 else
497 {
498 check(false);
499 }
500
501 check(UE::DXGIUtilities::GetWidthAlignment(DXGIFormat) == 1);
502 check(UE::DXGIUtilities::GetHeightAlignment(DXGIFormat) == 1);
503 }
504
505 break;
506 }
507 case D3D12_RESOURCE_DIMENSION_TEXTURE3D:
508 {
509 if (UE::DXGIUtilities::IsBlockCompressedFormat(DXGIFormat))
510 {
511 // Currently only supported block sizes are 64 and 128.
512 // These equations calculate the size in texels for a tile. It relies on the fact that 16*16*16 blocks fit in a tile if the block size is 128 bits.
513 check(BPU == 64 || BPU == 128);
514 pTileShape->WidthInTexels = 8 * UE::DXGIUtilities::GetWidthAlignment(DXGIFormat);
515 pTileShape->HeightInTexels = 8 * UE::DXGIUtilities::GetHeightAlignment(DXGIFormat);
516 pTileShape->DepthInTexels = 4;
517 if (BPU == 64)
518 {
519 // If bits per block are 64 we double width so it takes up the full tile size.
520 // This is only true for BC1 and BC4
521 check((DXGIFormat >= DXGI_FORMAT_BC1_TYPELESS && DXGIFormat <= DXGI_FORMAT_BC1_UNORM_SRGB) ||
522 (DXGIFormat >= DXGI_FORMAT_BC4_TYPELESS && DXGIFormat <= DXGI_FORMAT_BC4_SNORM));
523 pTileShape->DepthInTexels *= 2;
524 }
525 }
526 else
527 {
528 if (BPU <= 8)
529 {
530 pTileShape->WidthInTexels = 16;
531 pTileShape->HeightInTexels = 16;
532 pTileShape->DepthInTexels = 16;
533 }
534 else if (BPU <= 16)
535 {
536 pTileShape->WidthInTexels = 16;
537 pTileShape->HeightInTexels = 16;
538 pTileShape->DepthInTexels = 8;
539 }
540 else if (BPU <= 32)
541 {
542 pTileShape->WidthInTexels = 16;
543 pTileShape->HeightInTexels = 8;
544 pTileShape->DepthInTexels = 8;
545 }
546 else if (BPU <= 64)
547 {
548 pTileShape->WidthInTexels = 8;
549 pTileShape->HeightInTexels = 8;
550 pTileShape->DepthInTexels = 8;
551 }
552 else if (BPU <= 128)
553 {
554 pTileShape->WidthInTexels = 8;
555 pTileShape->HeightInTexels = 8;
556 pTileShape->DepthInTexels = 4;
557 }
558 else
559 {
560 check(false);
561 }
562
563 check(UE::DXGIUtilities::GetWidthAlignment(DXGIFormat) == 1);
564 check(UE::DXGIUtilities::GetHeightAlignment(DXGIFormat) == 1);
565 }
566 }
567 break;
568 }

682FORCEINLINE_DEBUGGABLE bool CheckResourceStateCompatibility(D3D12_RESOURCE_STATES State, D3D12_RESOURCE_FLAGS Flags, FString& IncompatibilityReason)
683{
684 // Check for ALLOW_RENDER_TARGET compatibility
685 if (State & D3D12_RESOURCE_STATE_RENDER_TARGET)
686 {
687 if (!(Flags & D3D12_RESOURCE_FLAG_ALLOW_RENDER_TARGET))
688 {
689 IncompatibilityReason.Append(TEXT("D3D12_RESOURCE_STATE_RENDER_TARGET requires D3D12_RESOURCE_FLAG_ALLOW_RENDER_TARGET.\n"));
690 return false;
691 }
692 }
693
694 // Check for ALLOW_DEPTH_STENCIL compatibility
695 if (State & D3D12_RESOURCE_STATE_DEPTH_WRITE || State & D3D12_RESOURCE_STATE_DEPTH_READ)
696 {
697 if (!(Flags & D3D12_RESOURCE_FLAG_ALLOW_DEPTH_STENCIL))
698 {
699 IncompatibilityReason.Append(TEXT("D3D12_RESOURCE_STATE_DEPTH_WRITE or D3D12_RESOURCE_STATE_DEPTH_READ requires D3D12_RESOURCE_FLAG_ALLOW_DEPTH_STENCIL.\n"));
