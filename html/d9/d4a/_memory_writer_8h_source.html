39 virtual void Serialize(void* Data, int64 Num) override
40 {
41 const int64 NumBytesToAdd = Offset + Num - Bytes.Num();
42 if( NumBytesToAdd > 0 )
43 {
44 const int64 NewArrayCount = Bytes.Num() + NumBytesToAdd;

55 }

63 }
