167 FSkinningHeader Pack() const
168 {
169 Validate();
170
171 FSkinningHeader Output;
172
173 if (bRefPoseFallback)
174 {
175 FMemory::Memzero(&Output, sizeof(Output));
176 Output.HierarchyBufferOffset = SKINNING_BUFFER_HIERARCHY_OFFSET_INVALID;
177 Output.TransformBufferOffset = SKINNING_BUFFER_TRANSFORM_OFFSET_INVALID;
178 Output.ObjectSpaceBufferOffset = SKINNING_BUFFER_OBJECT_SPACE_OFFSET_INVALID;
179 Output.bIsRefPose = true;
180 return Output;
181 }
182
183 Output.HierarchyBufferOffset = HierarchyBufferOffset;
184 Output.TransformBufferOffset = TransformBufferOffset;
185 Output.ObjectSpaceBufferOffset = ObjectSpaceBufferOffset != INDEX_NONE ? ObjectSpaceBufferOffset : 0;
186 Output.MaxTransformCount = MaxTransformCount;
187 Output.MaxInfluenceCount = MaxInfluenceCount;
188 Output.bHasScale = bHasScale;
189 Output.bIsRefPose = bIsRefPose;
190 Output.CurrentTransformSlot = CurrentTransformSlot;
191 return Output;
192 }
193
194 void Validate() const
195 {
196 if (bRefPoseFallback)
197 {
198 // Offsets were released; Pack emits a zeroed header, so there is nothing to validate.
199 return;
200 }
201
202 check(TransformBufferOffset <= SKINNING_BUFFER_TRANSFORM_OFFSET_MAX);
203 check(HierarchyBufferOffset == INDEX_NONE || HierarchyBufferOffset <= SKINNING_BUFFER_HIERARCHY_OFFSET_MAX);
204 check(ObjectSpaceBufferOffset == INDEX_NONE || ObjectSpaceBufferOffset <= SKINNING_BUFFER_OBJECT_SPACE_OFFSET_MAX);
205 check(MaxInfluenceCount <= SKINNING_BUFFER_INFLUENCE_MAX);
206 }
