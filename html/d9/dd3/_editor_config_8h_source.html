114template <typename T>
115bool FEditorConfig::TryGetStruct(FStringView Key, T& OutValue, EPropertyFilter Filter) const
116{
117 if (!IsValid())
118 {
119 return false;
120 }
121
122 TSharedPtr<FJsonObject> StructData;
123 UE::FJsonPath Path(Key);
124 if (!JsonConfig->TryGetJsonObject(Path, StructData))
125 {
126 return false;
127 }
128
129 if (!StructData.IsValid())
130 {
131 return false;
132 }
133
134 const UStruct* Struct = T::StaticStruct();
135 ReadStruct(StructData, Struct, &OutValue, nullptr, Filter);
136
137 return true;
138}
