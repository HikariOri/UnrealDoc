16// Vulkan ParameterMap:
17// Buffer Index = EBufferIndex
18// Base Offset = Index into the subtype
19// Size = Ignored for non-globals
20struct FVulkanShaderHeader
21{
22 // Includes all bindings, the index in this array is the binding slot
23 struct FBindingInfo
24 {
25 // VkDescriptorType
26 uint32 DescriptorType;
27#if VULKAN_ENABLE_BINDING_DEBUG_NAMES
28 FString DebugName;
29#endif
30 };
31 TArray<FBindingInfo> Bindings;
32
33 // FBindingInfo with type VK_DESCRIPTOR_TYPE_UNIFORM_BUFFER have a corresponding entry in this table (at the same index)
34 struct FUniformBufferInfo
35 {
36 uint32 LayoutHash;
37 uint8 bHasResources;
38 uint8 BindlessCBIndex;
39 };
40 TArray<FUniformBufferInfo> UniformBufferInfos;

58 // Used to determine the EAttachmentType of a FBindingInfo with type VK_DESCRIPTOR_TYPE_INPUT_ATTACHMENT

66 // Types of Global Samplers (see Common.ush for types)

74 BilinearWrappedSampler,

76 TrilinearWrappedSampler,
77
78 Count,
79 Invalid,
80 };
81 struct FGlobalSamplerInfo
82 {
83 uint8 BindingIndex;
84 EGlobalSamplerType Type;
85 };
86 TArray<FGlobalSamplerInfo> GlobalSamplerInfos;
87
88 // The number of uniform buffers containing constants and requiring bindings
89 // Uniform buffers beyond this index do not have bindings (resource only UB)
90 uint32 NumBoundUniformBuffers = 0;
91
92 // Size of the uniform buffer containing packed globals
93 // If present (not zero), it will always be at binding 0 of the stage
94 uint32 PackedGlobalsSize = 0;
95
96 // Mask of input attachments being used (the index of the bit corresponds to EAttachmentType value)
97 uint32 InputAttachmentsMask = 0;
98
99 // Mostly relevant for Vertex Shaders
100 uint32 InOutMask;
101
102 // Relevant for Ray Tracing Shaders
103 uint32 RayTracingPayloadType = 0;
104 uint32 RayTracingPayloadSize = 0;
105
106 // Specialization constant info
107 FSpecializationConstantData SpecializationConstantData;
108
109 FShaderHash CodeHash;
110 uint32 SpirvCRC = 0;
111 uint8 WaveSize = 0;
112
113 // For RayHitGroup shaders
114 enum class ERayHitGroupEntrypoint : uint8
115 {
116 NotPresent = 0,
117
118 // Hit group types are all stored in a single spirv blob
119 // and each have different entry point names
120 // NOTE: Not used yet because of compiler issues
121 CommonBlob,
122
123 // Hit group types are each stored in a different spirv blob
124 // to circumvent DXC compilation issues
125 SeparateBlob
126 };
127 ERayHitGroupEntrypoint RayGroupAnyHit = ERayHitGroupEntrypoint::NotPresent;
128 ERayHitGroupEntrypoint RayGroupIntersection = ERayHitGroupEntrypoint::NotPresent;
