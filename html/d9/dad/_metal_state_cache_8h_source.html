374 MTL::RenderPassDescriptor* CreateDescriptor()
375 {
376 MTL::RenderPassDescriptor* Desc = Cache.Pop();
377 if (!Desc)
378 {
379 Desc = MTL::RenderPassDescriptor::alloc()->init();
380 check(Desc);
381 }
382 return Desc;
383 }
384
385 void ReleaseDescriptor(MTL::RenderPassDescriptor* Desc)
386 {
387 MTL::RenderPassColorAttachmentDescriptorArray* Attachments = Desc->colorAttachments();
388 for (uint32 i = 0; i < MaxSimultaneousRenderTargets; i++)
389 {
390 MTL::RenderPassColorAttachmentDescriptor* Color = Attachments->object(i);
391 Color->setTexture(nullptr);
392 Color->setResolveTexture(nullptr);
393 Color->setStoreAction(MTL::StoreActionStore);
394 }
395
396 MTL::RenderPassDepthAttachmentDescriptor* Depth = Desc->depthAttachment();
397 Depth->setTexture(nullptr);
398 Depth->setResolveTexture(nullptr);
399 Depth->setStoreAction(MTL::StoreActionStore);
400
401 MTL::RenderPassStencilAttachmentDescriptor* Stencil = Desc->stencilAttachment();
402 Stencil->setTexture(nullptr);
403 Stencil->setResolveTexture(nullptr);
404 Stencil->setStoreAction(MTL::StoreActionStore);
405
406 Desc->setVisibilityResultBuffer(nullptr);
407
408#if PLATFORM_MAC
409 Desc->setRenderTargetArrayLength(1);
410#endif
411
412 Cache.Push(Desc);
413 }
