41 int32_t ByteLength = Value.ByteLen();
42 Ar << ByteLength;
43 if (Ar.IsLoading())

45 Value = CUTF8String(ByteLength, [&Ar, ByteLength](UTF8Char* Memory) { Ar.Serialize(Memory, ByteLength); });
46 }

50 }
51 return Ar;
52 }

56 friend CArchive& operator<<(CArchive& Ar, TOptional<T>& Value)
57 {
58 EResult Result = Value.GetResult();
59 Ar << (int8_t&)Result;
60 if (Result == EResult::OK)
61 {
62 if (Ar.IsLoading())
63 {
64 Value.Emplace();
65 }
66 Ar << *Value;
67 }
68 return Ar;
69 }
70
71 // Convenience operator for serializing an array
72 template<class ElementType>
73 friend CArchive& operator<<(CArchive& Ar, TArray<ElementType>& Value)
74 {
75 int32_t NumElements = Value.Num();
76 Ar << NumElements;
77 if (Ar.IsLoading())
78 {
79 Value.SetNum(NumElements);
80 }
81 for (ElementType& Element : Value)

83 Ar << Element;
84 }
