112// Simple acceleration structure used to return previously searched results
113template <typename PayloadType, int32 Size = 3>
114struct TTimingEventSearchCache
115{
116public:
117 TTimingEventSearchCache()
118 : WriteIndex(0)
119 {}
120
121 // Write to the cache
122 // @return a handle to the search
123 FTimingEventSearchHandle Write(double InStartTime, double InEndTime, uint32 InDepth, const PayloadType& InPayload)
124 {
125 TPair<FTimingEventSearchHandle, FResultData>& WritePair = CachedValues[WriteIndex];
126
127 WritePair.Key = FTimingEventSearchHandle::GenerateHandle();
128 WritePair.Value.Payload = InPayload;
129 WritePair.Value.StartTime = InStartTime;
130 WritePair.Value.EndTime = InEndTime;
131 WritePair.Value.Depth = InDepth;
132
133 WriteIndex = (WriteIndex + 1) % Size;
134
135 return WritePair.Key;
136 }
137
138 // Attempt to read from the cache.
139 // @return false if the value was not found
140 bool Read(const FTimingEventSearchHandle& InHandle, double& OutStartTime, double& OutEndTime, uint32& OutDepth, PayloadType& OutPayload)
141 {
142 static_assert(Size > 1, "TTimingEventSearchCache only works with caches that have multiple values");
143
144 if (InHandle.IsValid())
145 {
146 // Start read at the previous write, to catch most recently written values
147 for (int32 ReadIndex = (WriteIndex + (Size - 1)) % Size, ReadCount = 0; ReadCount < Size; ReadCount++, ReadIndex = (ReadIndex + 1) % Size)
148 {
149 check(ReadIndex >= 0 && ReadIndex < Size);
150
151 const TPair<FTimingEventSearchHandle, FResultData>& CachedPair = CachedValues[ReadIndex];
152 if (InHandle == CachedPair.Key)
153 {
154 OutStartTime = CachedPair.Value.StartTime;
155 OutEndTime = CachedPair.Value.EndTime;
156 OutDepth = CachedPair.Value.Depth;
157 OutPayload = CachedPair.Value.Payload;
158 return true;
159 }
160 }
161 }
162
163 return false;
164 }
165
166 // Reset the cache. Any reads after this will return false
167 void Reset()
168 {
169 for (TPair<FTimingEventSearchHandle, FResultData>& CachedPair : CachedValues)
170 {
171 CachedPair.Key.Reset();
172 }
173 }
174
175private:
176 // Current write index
177 int32 WriteIndex;
178
179 // A cached result
180 struct FResultData
181 {
182 PayloadType Payload;
183 double StartTime;
184 double EndTime;
185 uint32 Depth;
186 };
187
188 // Array of cached searches
189 TPair<FTimingEventSearchHandle, FResultData> CachedValues[Size];
190};

193// Helper used to orchestrate a search of a timing event track's events
194// Example of usage:
195//
196// FMyStruct MatchedPayload;
197//
198// TTimingEventSearch<FMyStruct>::Search(
199// Parameters,
200// [](const TTimingEventSearch<FMyStruct>::FContext& InContext)
201// {
202// for (FMyStruct& Payload : Payloads)
203// {
204// InContext.Check(InEventStartTime, InEventEndTime, 0, Payload);
205// }
206// },
207// [&MatchedPayload](double InStartTime, double InEndTime, uint32 InDepth, const FMyStruct& InEvent)
208// {
209// MatchedPayload = InEvent;
210// },
211// [&MatchedPayload](double InStartTime, double InEndTime, uint32 InDepth)
212// {
213// // Do something with MatchedPayload, e.g. call a captured lambda.
214// });
215//
216template <typename PayloadType>
217struct TTimingEventSearch
218{
219public:
220 struct FContext;
221
222 // Predicate called when a match has been made to the search parameters
223 typedef TFunctionRef<void(double /*InStartTime*/, double /*InEndTime*/, uint32 /*InDepth*/, const PayloadType& /*InPayload*/)> PayloadMatchedPredicate;
224
225 // Predicate called to filter by payload contents
226 // Return true to pass the filer
227 typedef TFunctionRef<bool(double /*InStartTime*/, double /*InEndTime*/, uint32 /*InDepth*/, const PayloadType& /*InPayload*/)> PayloadFilterPredicate;
228
229 // Predicate called when a match has been found
230 typedef TFunctionRef<void(double /*InStartTime*/, double /*InEndTime*/, uint32 /*InDepth*/, const PayloadType& /*InPayload*/)> FoundPredicate;
231
232 // Predicate called to run the search, e.g. iterate over an array of events
233 // It is expected to call FContext::Check on each valid searched event.
234 typedef TFunctionRef<void(FContext& /*Context*/)> SearchPredicate;
235
236 // Context used to operate a search.
237 struct FContext
238 {
239 public:
240 FContext(const FTimingEventSearchParameters& InParameters, PayloadFilterPredicate InPayloadFilterPredicate, PayloadMatchedPredicate InPayloadMatchedPredicate)
241 : Parameters(InParameters)
242 , PayloadMatched(InPayloadMatchedPredicate)
243 , PayloadFilter(InPayloadFilterPredicate)
244 , FoundStartTime(-1.0)
245 , FoundEndTime(-1.0)
246 , FoundDepth(0)
247 , bFound(false)
248 , bContinueSearching(true)
249 {
250 }
251
252 // Function called to check and potentially match an event
253 void Check(double InEventStartTime, double InEventEndTime, uint32 InEventDepth, const PayloadType& InEvent)
254 {
255 if (bContinueSearching && Parameters.EventFilter(InEventStartTime, InEventEndTime, InEventDepth) && PayloadFilter(InEventStartTime, InEventEndTime, InEventDepth, InEvent))
256 {
257 Parameters.EventMatched(InEventStartTime, InEventEndTime, InEventDepth);
258 PayloadMatched(InEventStartTime, InEventEndTime, InEventDepth, InEvent);
259
260 FoundPayload = InEvent;
261 FoundDepth = InEventDepth;
262 FoundStartTime = InEventStartTime;
263 FoundEndTime = InEventEndTime;
264
265 bFound = true;
266 bContinueSearching = (Parameters.Flags & ETimingEventSearchFlags::StopAtFirstMatch) == ETimingEventSearchFlags::None;
267 }
268 }
269
270 // Access the search parameters
271 const FTimingEventSearchParameters& GetParameters() const { return Parameters; }
272
273 // Accessors for read-only results
274 const PayloadType& GetPayloadFound() const { return FoundPayload; }
275 double GetStartTimeFound() const { return FoundStartTime; }
276 double GetEndTimeFound() const { return FoundEndTime; }
277 uint32 GetDepthFound() const { return FoundDepth; }
278 bool IsMatchFound() const { return bFound; }
279 bool ShouldContinueSearching() const { return bContinueSearching; }
280
281 // Allows search to be aborted by predicates
282 void AbortSearch() { bContinueSearching = false; }
283
284 private:
285 // Search parameters
286 FTimingEventSearchParameters Parameters;
287
288 // Predicate called when an event was matched
289 PayloadMatchedPredicate PayloadMatched;
290
291 // Filter applied to payloads
292 PayloadFilterPredicate PayloadFilter;
293
294 // The payload we have found
295 PayloadType FoundPayload;
296
297 // The start time of the event that was found
298 double FoundStartTime;
299
300 // The end time of the event that was found
301 double FoundEndTime;
302
303 // The depth of the event that was found
304 uint32 FoundDepth;
305
306 // Whether a match was found
307 bool bFound;
308
309 // Internal flag to skip work if a match was found
310 bool bContinueSearching;
311 };
312
313public:
314 // Default predicates
315 static bool NoFilter(double, double, uint32, const PayloadType&) { return true; }
316 static void NoMatch(double, double, uint32, const PayloadType&) {}
317
318 // Search using only the event filter and no match predicate
319 static bool Search(const FTimingEventSearchParameters& InParameters, SearchPredicate InSearchPredicate, FoundPredicate InFoundPredicate)
320 {
321 return Search(InParameters, InSearchPredicate, NoFilter, InFoundPredicate, NoMatch);
322 }
323
324 // Search using only the event filter, no match predicate and a cache
325 static bool Search(const FTimingEventSearchParameters& InParameters, SearchPredicate InSearchPredicate, FoundPredicate InFoundPredicate, TTimingEventSearchCache<PayloadType>& InCache)
326 {
327 return Search(InParameters, InSearchPredicate, NoFilter, InFoundPredicate, NoMatch, &InCache);
328 }
329
330 // Search using a specific payload filter
331 static bool Search(const FTimingEventSearchParameters& InParameters, SearchPredicate InSearchPredicate, PayloadFilterPredicate InFilterPredicate, FoundPredicate InFoundPredicate, PayloadMatchedPredicate InPayloadMatchedPredicate, TTimingEventSearchCache<PayloadType>* InCache = nullptr)
332 {
333 if (InCache != nullptr && InParameters.SearchHandle != nullptr)
334 {
335 double StartTime;
336 double EndTime;
337 uint32 Depth;
338 PayloadType Payload;
339 if (InCache->Read(*InParameters.SearchHandle, StartTime, EndTime, Depth, Payload))
340 {
341 InParameters.EventMatched(StartTime, EndTime, Depth);
342 InPayloadMatchedPredicate(StartTime, EndTime, Depth, Payload);
343 InFoundPredicate(StartTime, EndTime, Depth, Payload);
344 return true;
345 }
346 }
347
348 FContext Context(InParameters, InFilterPredicate, InPayloadMatchedPredicate);
349
350 InSearchPredicate(Context);
351
352 if (Context.IsMatchFound())
353 {
354 InFoundPredicate(Context.GetStartTimeFound(), Context.GetEndTimeFound(), Context.GetDepthFound(), Context.GetPayloadFound());
355
356 if (InCache != nullptr && InParameters.SearchHandle != nullptr)
357 {
358 *InParameters.SearchHandle = InCache->Write(Context.GetStartTimeFound(), Context.GetEndTimeFound(), Context.GetDepthFound(), Context.GetPayloadFound());
359 }
360 }
361
362 return Context.IsMatchFound();
363 }
364};
