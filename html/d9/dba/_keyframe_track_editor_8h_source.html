566 /*
567 * Adds keys to the specified object. This may also add tracks and sections depending on the options specified.
568 *
569 * @param ObjectsToKey An array of objects to add keyframes to.
570 * @param KeyTime The time to add keys.
571 * @param KeyedChannels Aggregate of optionally enabled channel values for which to create new keys
572 * @param ChannelDefaults Aggregate of optionally enabled channel default values that should be applied to the section
573 * @param KeyParams The parameters to control keyframing behavior.
574 * @param TrackClass The class of track which should be created if specified in the parameters.
575 * @param PropertyName The name of the property to add keys for.
576 * @param OnInitializeNewTrack A delegate which allows for custom initialization for new tracks. This is called after the
577 * track is created, but before any sections or keys have been added.
578 * @return Whether or not a handle guid or track was created. Note this does not return true if keys were added or modified.
579 */
580 FKeyPropertyResult AddKeysToObjects(
581 TArrayView<UObject* const> ObjectsToKey, FFrameNumber KeyTime,
582 ESequencerKeyMode KeyMode, TSubclassOf<UMovieSceneTrack> TrackClass, FName PropertyName,
583 TFunction<void(TrackType*)> OnInitializeNewTrack,
584 const GenerateKeysCallback& OnGenerateKeys)
585 {
586 FKeyPropertyResult KeyPropertyResult;
587
588 EAutoChangeMode AutoChangeMode = GetSequencer()->GetAutoChangeMode();
589 EAllowEditsMode AllowEditsMode = GetSequencer()->GetAllowEditsMode();
590
591 bool bCreateHandle =
592 (KeyMode == ESequencerKeyMode::AutoKey && (AutoChangeMode == EAutoChangeMode::AutoTrack || AutoChangeMode == EAutoChangeMode::All)) ||
593 KeyMode == ESequencerKeyMode::ManualKey ||
594 KeyMode == ESequencerKeyMode::ManualKeyForced ||
595 AllowEditsMode == EAllowEditsMode::AllowSequencerEditsOnly;
596
597 bool bShouldPlaceInChildFolder = KeyMode == ESequencerKeyMode::AutoKey;
598 const FName CreatedFolderName = bShouldPlaceInChildFolder ? FName(TEXT("Autotracked Changes")) : NAME_None;
599
600 for ( UObject* Object : ObjectsToKey )
601 {
602 FFindOrCreateHandleResult HandleResult = FindOrCreateHandleToObject( Object, bCreateHandle, CreatedFolderName);
603 FGuid ObjectHandle = HandleResult.Handle;
604 KeyPropertyResult.bHandleCreated = HandleResult.bWasCreated;
605
606 if ( ObjectHandle.IsValid() )
607 {
608 KeyPropertyResult |= AddKeysToHandle( Object, ObjectHandle, KeyTime, KeyMode, TrackClass, PropertyName, OnInitializeNewTrack, OnGenerateKeys );
609 }
610 }
611 return KeyPropertyResult;
612 }

622

624 for (const FMovieSceneChannelEntry& Entry : Section->GetChannelProxy().GetAllEntries())

629 }

642 NAME_None,
643 EUserInterfaceActionType::Button );

646 /*
647 * Adds keys to the specified guid. This may also add tracks and sections depending on the options specified.
648 *
649 * @param Object Object to add keyframes to.
650 * @param ObjectHandle Object Handle to add keyframes to.
651 * @param KeyTime The time to add keys.
652 * @param GeneratedKeys Array of keys to set
653 * @param KeyParams The parameters to control keyframing behavior.
654 * @param TrackClass The class of track which should be created if specified in the parameters.
655 * @param PropertyName The name of the property to add keys for.
656 * @param OnInitializeNewTrack A delegate which allows for custom initialization for new tracks. This is called after the
657 * track is created, but before any sections or keys have been added.
658 * @return Whether or not a track was created. Note this does not return true if keys were added or modified.
659 */
660 FKeyPropertyResult AddKeysToHandle(UObject *Object,
661 FGuid ObjectHandle, FFrameNumber KeyTime,
662 ESequencerKeyMode KeyMode, TSubclassOf<UMovieSceneTrack> TrackClass, FName PropertyName,
663 TFunction<void(TrackType*)> OnInitializeNewTrack,
664 const GenerateKeysCallback& OnGenerateKeys)
665 {
666 bool bTrackCreated = false;
667
668 EAutoChangeMode AutoChangeMode = GetSequencer()->GetAutoChangeMode();
669 EAllowEditsMode AllowEditsMode = GetSequencer()->GetAllowEditsMode();
670
671 bool bCreateTrack =
672 (KeyMode == ESequencerKeyMode::AutoKey && (AutoChangeMode == EAutoChangeMode::AutoTrack || AutoChangeMode == EAutoChangeMode::All)) ||
673 KeyMode == ESequencerKeyMode::ManualKey ||
674 KeyMode == ESequencerKeyMode::ManualKeyForced ||
675 AllowEditsMode == EAllowEditsMode::AllowSequencerEditsOnly;
676
677 bool bCreateSection = bCreateTrack ||
678 (KeyMode == ESequencerKeyMode::AutoKey && (AutoChangeMode != EAutoChangeMode::None));
679
680 // Try to find an existing Track, and if one doesn't exist check the key params and create one if requested.
681 FFindOrCreateTrackResult TrackResult = FindOrCreateTrackForObject( ObjectHandle, TrackClass, PropertyName, bCreateTrack );
682 TrackType* Track = CastChecked<TrackType>( TrackResult.Track, ECastCheckedType::NullAllowed );
683 bTrackCreated = TrackResult.bWasCreated;
684
685 if ( bTrackCreated )
686 {
687 if (OnInitializeNewTrack)
688 {
689 OnInitializeNewTrack(Track);
690 }
691 }
692
693 bool bSectionCreated = false;
694
695 FKeyPropertyResult KeyPropertyResult;
696
697 if ( Track )
698 {
699 float Weight = 1.0f;
700 UMovieSceneSection* SectionToKey = bCreateSection ? Track->FindOrExtendSection(KeyTime, Weight) : Track->FindSection(KeyTime);
701
702 // If there's no overlapping section to key, create one only if a track was newly created. Otherwise, skip keying altogether
703 // so that the user is forced to create a section to key on.
704 if (bTrackCreated && !SectionToKey)
705 {
706 Track->Modify();
707 SectionToKey = Track->FindOrAddSection(KeyTime, bSectionCreated);
708 if (bSectionCreated && GetSequencer()->GetInfiniteKeyAreas())
709 {
710 SectionToKey->SetRange(TRange<FFrameNumber>::All());
711 KeyPropertyResult.SectionsCreated.Add(SectionToKey);
712 }
713 }
714
715 if (SectionToKey && CanAutoKeySection(SectionToKey, KeyTime) && OnGenerateKeys)
716 {
717 FGeneratedTrackKeys GeneratedKeys;
718 OnGenerateKeys(SectionToKey, GeneratedKeys);
719 KeyPropertyResult |= AddKeysToSection( SectionToKey, KeyTime, GeneratedKeys, KeyMode );
720 }
721 }
722
723 KeyPropertyResult.bTrackCreated |= bTrackCreated || bSectionCreated;
724
725 return KeyPropertyResult;
726 }

730 FKeyPropertyResult AddKeysToSection(UMovieSceneSection* Section, FFrameNumber KeyTime, const FGeneratedTrackKeys& Keys, ESequencerKeyMode KeyMode, EKeyFrameTrackEditorSetDefault SetDefault = EKeyFrameTrackEditorSetDefault::SetDefault)
731 {
732 FKeyPropertyResult KeyPropertyResult;
733
734 EAutoChangeMode AutoChangeMode = GetSequencer()->GetAutoChangeMode();
735
736 FMovieSceneChannelProxy& Proxy = Section->GetChannelProxy();
737
738 const bool bSetDefaults = GetSequencer()->GetAutoSetTrackDefaults() && (SetDefault != EKeyFrameTrackEditorSetDefault::DoNotSetDefault);
739
740 // The default value is a value for the channel when there are no keyframes. For example, if you add keys and
741 // then delete them all, the default value is the value of the channel. In the implementation of ApplyDefault,
742 // all the setters check that the default value is only set when there are NO keyframes. So, ApplyDefault needs
743 // to be called here in AddKeysToSection BEFORE any keys are added.
744 if (bSetDefaults)
745 {
746 for (const FMovieSceneChannelValueSetter& GeneratedKey : Keys)
747 {
748 GeneratedKey->ApplyDefault(Section, Proxy, SetDefault);
749 }
750 }
751
752 if ( KeyMode != ESequencerKeyMode::AutoKey || AutoChangeMode == EAutoChangeMode::AutoKey || AutoChangeMode == EAutoChangeMode::All)
753 {
754 EMovieSceneKeyInterpolation InterpolationMode = GetSequencer()->GetKeyInterpolation();
755
756 const bool bKeyEvenIfUnchanged =
757 KeyMode == ESequencerKeyMode::ManualKeyForced ||
758 GetSequencer()->GetKeyGroupMode() == EKeyGroupMode::KeyAll ||
759 GetSequencer()->GetKeyGroupMode() == EKeyGroupMode::KeyGroup;
760
761 const bool bKeyEvenIfEmpty =
762 (KeyMode == ESequencerKeyMode::AutoKey && AutoChangeMode == EAutoChangeMode::All) ||
763 KeyMode == ESequencerKeyMode::ManualKeyForced;
764
765 for (const FMovieSceneChannelValueSetter& GeneratedKey : Keys)
766 {
767 KeyPropertyResult.bKeyCreated |= GeneratedKey->Apply(Section, Proxy, KeyTime, InterpolationMode, bKeyEvenIfUnchanged, bKeyEvenIfEmpty);
768 if (KeyPropertyResult.bKeyCreated)
769 {
770 KeyPropertyResult.SectionsKeyed.Add(Section);
771 }
772 }
773 }
774
775 return KeyPropertyResult;
776 }
