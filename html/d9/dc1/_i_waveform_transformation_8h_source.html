146// Base class to hold editor configurable properties for an arbitrary transformation of audio waveform data
147UCLASS(Abstract, EditInlineNew, MinimalAPI)
148class UWaveformTransformationBase : public UObject
149{
150 GENERATED_BODY()
151
152public:
153 virtual Audio::FTransformationPtr CreateTransformation() const { return nullptr; }
154 virtual void UpdateConfiguration(FWaveTransformUObjectConfiguration& InOutConfiguration) {};
155 virtual void OverwriteTransformation() {};
156 virtual constexpr Audio::ETransformationPriority GetTransformationPriority() const { return Audio::ETransformationPriority::None; }
157
158 // Sort to ensure proper order of operation for audio processing
159 static void SortTransformationsArray(TArray<TObjectPtr<UWaveformTransformationBase>>& InOutTransformations)
160 {
161 Algo::StableSort(InOutTransformations, [](const TObjectPtr<UWaveformTransformationBase>& TransformationA, const TObjectPtr<UWaveformTransformationBase>& TransformationB)
162 {
163 if ((!TransformationA && TransformationB) || (!TransformationA && !TransformationB))
164 {
165 return false;
166 }
167 else if (TransformationA && !TransformationB)
168 {
169 return true;
170 }
171
172 return TransformationA->GetTransformationPriority() > TransformationB->GetTransformationPriority();
173 });
174 }
175
176#if WITH_EDITOR
177 virtual void OverwriteSoundWaveData(USoundWave& InOutSoundWave) {};
178 virtual void GetTransformationInfo(FWaveformTransformationInfo& InOutTransformationInfo) const {};
179 virtual FString GetTransformationHash() const PURE_VIRTUAL(UWaveformTransformationBase::GetTransformationHash, return FString(););
180
181 AUDIOEXTENSIONS_API virtual void PostEditUndo() override;
182 AUDIOEXTENSIONS_API virtual void PostEditChangeProperty(FPropertyChangedEvent& PropertyChangedEvent) override;
183
184 void NotifyPropertyChange(FProperty* Property);
185#endif //WITH_EDITOR
186
187 virtual bool IsEditorOnly() const override { return true; }
188
189
190 // Execute when the soundwave needs a recook to apply new transformation changes, bool flag if it should mark the file dirty
191 DECLARE_DELEGATE_OneParam(FOnTransformationChanged, bool);
192 FOnTransformationChanged OnTransformationChanged;
193
194 // Execute when only the rendering needs to be updated, but can wait for a recook
195 DECLARE_DELEGATE(FOnTransformationRenderChanged);
