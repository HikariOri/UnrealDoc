44 static constexpr IdType GenerateTypeId(
45 const TCHAR* ImplName,
46 const TCHAR* ValueTypeName)
47 {
48 // Compute hash values from the strings (16 bits each)
49 uint16 ImplHash = CompileTimeHash(ImplName) & 0xFFFF;
50 uint16 ValueHash = CompileTimeHash(ValueTypeName) & 0xFFFF;
51
52 // Combine: [implHash(16) | valueHash(16)]
53 return ((IdType)ImplHash << 16) | ValueHash;
54 }
55
60 static constexpr uint32 CompileTimeHash(const TCHAR* Str)
61 {
62 uint32 Result = 2166136261u; // FNV offset basis
63
64 while (*Str)
65 {
66 // Process one TCHAR at a time
67 TCHAR C = *Str++;
68
69 // FNV-1a hash
70 Result ^= (uint32)C;
71 Result *= 16777619u; // FNV prime
72 }
73
74 return Result;
75 }

80 static constexpr uint16 GetImplHash(IdType TypeId)
81 {
82 return (TypeId >> 16) & 0xFFFF; // Upper 16 bits
83 }
84
88 static constexpr uint16 GetValueHash(IdType TypeId)
89 {
90 return TypeId & 0xFFFF; // Lower 16 bits

97 static bool CheckTypeIdConflict(IdType TypeId, const FString& ImplName, const FString& ValueTypeName,
98 const FString& ExistingImplName, const FString& ExistingValueTypeName)
99 {
100 // If the names don't match but the TypeId does, we have a hash collision
101 bool bImplNameMismatch = (ImplName != ExistingImplName);
102 bool bValueNameMismatch = (ValueTypeName != ExistingValueTypeName);
103
104 if (bImplNameMismatch || bValueNameMismatch)
105 {
106 UE_LOGF(LogSpline, Error, " HASH COLLISION in SplineTypeId!! ️");
107 UE_LOGF(LogSpline, Error, " TypeId: 0x%08X", TypeId);
108 UE_LOGF(LogSpline, Error, " Attempted registration: Impl=%ls, Value=%ls",
109 *ImplName, *ValueTypeName);
110 UE_LOGF(LogSpline, Error, " Existing registration: Impl=%ls, Value=%ls",
111 *ExistingImplName, *ExistingValueTypeName);
112 return true;
113 }
114
115 return false;
116 }
