114 // Non-template dedup function shared across all AuditMaterials instantiations.
115 // Uses 64-bit XXHash to avoid storing full strings while keeping collision probability negligible.
116 // The set is capped to bound memory in long-running processes; resetting may re-emit some warnings.
117 inline bool ShouldLogMaterialWarning(const FString& MaterialName, const FString& MeshName)
118 {
119 constexpr uint32 MaxEntries = 4096;
120 static FRWLock RWLock;
121 static TSet<uint64> WarnedCombinations;
122
123 // Hash outside the lock — pure computation, no shared state.
124 FXxHash64Builder HashBuilder;
125 HashBuilder.Update(MaterialName.GetCharArray().GetData(), MaterialName.GetCharArray().Num() * sizeof(FString::ElementType));
126 HashBuilder.Update(MeshName.GetCharArray().GetData(), MeshName.GetCharArray().Num() * sizeof(FString::ElementType));
127 const uint64 Hash = HashBuilder.Finalize().Hash;
128
129 // Fast path: read lock to check if we already warned (common case).
130 {
131 FReadScopeLock ReadLock(RWLock);
132 if (WarnedCombinations.Contains(Hash))
133 {
134 return false;
135 }
136 }
137
138 // Slow path: write lock to insert.
139 {
140 FWriteScopeLock WriteLock(RWLock);
141
142 // Re-check under write lock — another thread may have inserted between our read and write.
143 bool bAlreadyInSet = false;
144 if (WarnedCombinations.Num() >= MaxEntries)
145 {
146 WarnedCombinations.Reset();
147 }
148 WarnedCombinations.Add(Hash, &bAlreadyInSet);
149 return !bAlreadyInSet;
150 }
151 }

154 template<class T>
155 bool FNaniteResourcesHelper::ShouldCreateNaniteProxy(const T& Component, FMaterialAudit* OutNaniteMaterials)
156 {
157 // Whether or not to allow Nanite for this component
158#if WITH_EDITORONLY_DATA
159 const bool bForceFallback = Component.IsDisplayNaniteFallbackMesh();
160#else
161 const bool bForceFallback = false;
162#endif
163
164 if (bForceFallback || Component.IsDisallowNanite() || Component.IsForceDisableNanite())
165 {
166 // Regardless of the static mesh asset supporting Nanite, this component does not want Nanite to be used
167 return false;
168 }
169
170 const EShaderPlatform ShaderPlatform = Component.GetScene() ? Component.GetScene()->GetShaderPlatform() : GMaxRHIShaderPlatform;
171
172 if (!UseNanite(ShaderPlatform) || !Component.HasValidNaniteData())
173 {
174 return false;
175 }
176
177 {
178 FMaterialAudit NaniteMaterials{};
179 AuditMaterials(&Component, NaniteMaterials, OutNaniteMaterials != nullptr);
180
181 const bool bIsMaskingAllowed = Nanite::IsMaskingAllowed(Component.GetWorld(), Component.IsForceNaniteForMasked());
182 if (!NaniteMaterials.IsValid(bIsMaskingAllowed))
183 {
184 return false;
185 }
186
187 if (OutNaniteMaterials)
188 {
189 *OutNaniteMaterials = MoveTemp(NaniteMaterials);
190 }
191 }
192
193 return true;
194 }

210 template<class T>
211 FMaterialAudit& FNaniteResourcesHelper::AuditMaterials(const T* Component, FMaterialAudit& Audit, bool bSetMaterialUsage)
212 {
213 static const auto NaniteForceEnableMeshesCvar = IConsoleManager::Get().FindTConsoleVariableDataInt(TEXT("r.Nanite.ForceEnableMeshes"));
214 static const bool bNaniteForceEnableMeshes = NaniteForceEnableMeshesCvar && NaniteForceEnableMeshesCvar->GetValueOnAnyThread() != 0;
215
216 static const auto CVarLumenDetectCardSharingCompatibility = IConsoleManager::Get().FindTConsoleVariableDataInt(TEXT("r.LumenScene.SurfaceCache.DetectCardSharingCompatibility"));
217 const bool bLumenDetectCardSharingCompatibility = CVarLumenDetectCardSharingCompatibility && CVarLumenDetectCardSharingCompatibility->GetValueOnAnyThread() != 0;
218
219 Audit.bHasAnyError = false;
220 Audit.Entries.Reset();
221
222 if (Component != nullptr)
223 {
224 TArray<Private::FAuditMaterialSlotInfo, TInlineAllocator<32>> Slots = Nanite::Private::GetMaterialSlotInfos(*Component);
225
226 Audit.bCompatibleWithLumenCardSharing = Slots.Num() > 0;
227
228 uint32 Index = 0;
229 for (const Private::FAuditMaterialSlotInfo& SlotInfo : Slots)
230 {
231 FMaterialAuditEntry& Entry = Audit.Entries.AddDefaulted_GetRef();
232 Entry.MaterialSlotName = SlotInfo.SlotName;
233 Entry.MaterialIndex = Index;
234 Index++;
235 Entry.Material = SlotInfo.Material;
236 Entry.bHasNullMaterial = Entry.Material == nullptr;
237 Entry.LocalUVDensities = FVector4f(
238 SlotInfo.UVChannelData.LocalUVDensities[0],
239 SlotInfo.UVChannelData.LocalUVDensities[1],
240 SlotInfo.UVChannelData.LocalUVDensities[2],
241 SlotInfo.UVChannelData.LocalUVDensities[3]
242 );
243
244 if (Entry.bHasNullMaterial)
245 {
246 // Never allow null materials, assign default instead
247 Entry.Material = UMaterial::GetDefaultMaterial(MD_Surface);
248 }
249
250 const UMaterial* Material = Entry.Material->GetMaterial_Concurrent();
251 check(Material != nullptr); // Should always be valid here
252
253 const EBlendMode BlendMode = Entry.Material->GetBlendMode();
254
255 bool bUsingCookedEditorData = false;
256#if WITH_EDITORONLY_DATA
257 bUsingCookedEditorData = Material->GetOutermost()->bIsCookedForEditor;
258#endif
259 bool bUsageSetSuccessfully = false;
260
261 const FMaterialCachedExpressionData& CachedMaterialData = Material->GetCachedExpressionData();
262 Entry.bHasVertexInterpolator = CachedMaterialData.bHasVertexInterpolator;
263 Entry.bHasPerInstanceRandomID = CachedMaterialData.bHasPerInstanceRandom;
264 Entry.bHasPerInstanceCustomData = CachedMaterialData.bHasPerInstanceCustomData;
265 Entry.bHasVertexUVs = CachedMaterialData.bHasCustomizedUVs;
266 Entry.bHasPixelDepthOffset = Material->HasPixelDepthOffsetConnected();
267 Entry.bHasWorldPositionOffset = Material->HasVertexPositionOffsetConnected();
268 Entry.bHasTessellationEnabled = Material->IsTessellationEnabled();
269 Entry.bHasUnsupportedBlendMode = !IsSupportedBlendMode(BlendMode);
270 Entry.bHasUnsupportedShadingModel = !IsSupportedShadingModel(Material->GetShadingModels());
271 Entry.bHasInvalidUsage = (bUsingCookedEditorData || !bSetMaterialUsage) ? Entry.Material->NeedsSetMaterialUsage_Concurrent(bUsageSetSuccessfully, MATUSAGE_Nanite) : !Entry.Material->CheckMaterialUsage_Concurrent(MATUSAGE_Nanite);
272
273 if (Private::IsMaterialSkeletalMesh(*Component))
274 {
275 Entry.bHasInvalidUsage |= (bUsingCookedEditorData || !bSetMaterialUsage) ? Entry.Material->NeedsSetMaterialUsage_Concurrent(bUsageSetSuccessfully, MATUSAGE_SkeletalMesh) : !Entry.Material->CheckMaterialUsage_Concurrent(MATUSAGE_SkeletalMesh);
276 }
277
278 Audit.bHasMasked |= (BlendMode == BLEND_Masked);
279 // Only treat the section as Nanite-translucent when the blend mode is actually supported (which is gated
280 // by r.Nanite.AllowTranslucency). Otherwise the translucent material falls through as an unsupported
281 // blend mode (warned above) instead of being routed into the Nanite translucency pipeline.
282 Entry.bHasTranslucency = !Entry.bHasUnsupportedBlendMode && IsTranslucentOnlyBlendMode(BlendMode);
283
284 if (Material->bIsSky)
285 {
286 // Sky material is a special case we want to skip
287 Audit.bHasSky = true;
288 }
289
290 auto IsCompatibleWithLumenCardSharing = [&]()
291 {
292 if (Entry.Material->IsCompatibleWithLumenCardSharing())
293 {
294 // Material is explicitly marked as compatible
295 return true;
296 }
297
298 return bLumenDetectCardSharingCompatibility
299 && !CachedMaterialData.bHasPerInstanceRandom
300 && !CachedMaterialData.bHasPerInstanceCustomData
301 && !CachedMaterialData.bHasWorldPosition;
302 };
303
304 if (!IsCompatibleWithLumenCardSharing())
305 {
306 Audit.bCompatibleWithLumenCardSharing = false;
307 }
308
309 Entry.bHasAnyError =
310 Entry.bHasUnsupportedBlendMode |
311 Entry.bHasUnsupportedShadingModel |
312 Entry.bHasInvalidUsage;
313
314 if (!bUsingCookedEditorData && Entry.bHasAnyError && !Audit.bHasAnyError)
315 {
316 // Only populate on error for performance/memory reasons
317 Audit.AssetName = Private::GetMaterialMeshName(*Component);
318 Audit.FallbackMaterial = UMaterial::GetDefaultMaterial(MD_Surface);
319 }
320
321 Audit.bHasAnyError |= Entry.bHasAnyError;
322
323#if !(UE_BUILD_SHIPPING) || WITH_EDITOR
324 if (!bUsingCookedEditorData && !bNaniteForceEnableMeshes)
325 {
326 if (Entry.bHasUnsupportedBlendMode && ShouldLogMaterialWarning(Entry.Material->GetName(), Audit.AssetName))
327 {
328 const FString BlendModeName = GetBlendModeString(Entry.Material->GetBlendMode());
329 if (Private::IsMaterialSkeletalMesh(*Component))
330 {
331 UE_LOGF(
332 LogSkeletalMesh, Warning,
333 "Invalid material [%ls] used on Nanite skeletal mesh [%ls]. Only opaque or masked blend modes are currently supported, [%ls] blend mode was specified.",
334 *Entry.Material->GetName(),
335 *Audit.AssetName,
336 *BlendModeName
337 );
338 }
339 else
340 {
341 UE_LOGF(
342 LogStaticMesh, Warning,
343 "Invalid material [%ls] used on Nanite static mesh [%ls]. Only opaque or masked blend modes are currently supported, [%ls] blend mode was specified. (NOTE: \"Disallow Nanite\" on static mesh components can be used to suppress this warning and forcibly render the object as non-Nanite.)",
344 *Entry.Material->GetName(),
345 *Audit.AssetName,
346 *BlendModeName
347 );
348 }
349 }
350 if (Entry.bHasUnsupportedShadingModel && ShouldLogMaterialWarning(Entry.Material->GetName(), Audit.AssetName))
351 {
352 const FString ShadingModelString = GetShadingModelFieldString(Entry.Material->GetShadingModels());
353 if (Private::IsMaterialSkeletalMesh(*Component))
354 {
355 UE_LOGF(
356 LogSkeletalMesh, Warning,
357 "Invalid material [%ls] used on Nanite skeletal mesh [%ls]. The SingleLayerWater shading model is currently not supported, [%ls] shading model was specified.",
358 *Entry.Material->GetName(),
359 *Audit.AssetName,
360 *ShadingModelString
361 );
362 }
363 else
364 {
365 UE_LOGF(
366 LogStaticMesh, Warning,
367 "Invalid material [%ls] used on Nanite static mesh [%ls]. The SingleLayerWater shading model is currently not supported, [%ls] shading model was specified. (NOTE: \"Disallow Nanite\" on static mesh components can be used to suppress this warning and forcibly render the object as non-Nanite.)",
368 *Entry.Material->GetName(),
369 *Audit.AssetName,
370 *ShadingModelString
371 );
372 }
373 }
374 }
375#endif
376 }
377 }
378
379 return Audit;
380 }
