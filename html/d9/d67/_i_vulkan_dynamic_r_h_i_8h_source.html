35// Base class used for external code (such as plugins) to enable extensions in the Vulkan RHI. The lifetime is managed through the use of TSharedRef.
36// Basic usage: Using this as a base class, pass in the ExtensionName and overload GetFeaturesStruct() to return the extension specific features struct.
37// IMPORTANT: Make sure the feature struct has its sType correctly set and pNext to NULL.
38//
39// Example to enable VK_EXT_memory_decompression:
40// class FDeviceExtensionMemoryDecompression : public FVulkanRHIExternalDeviceExtensionBase
41// {
42// FDeviceExtensionMemoryDecompression() : FVulkanRHIExternalDeviceExtensionBase(VK_EXT_MEMORY_DECOMPRESSION_EXTENSION_NAME)
43// {
44// ZeroVulkanStruct(Features, VK_STRUCTURE_TYPE_PHYSICAL_DEVICE_MEMORY_DECOMPRESSION_FEATURES_EXT);
45// }
46// virtual VkBaseOutStructure* GetFeaturesStruct() final override
47// {
48// return reinterpret_cast<VkBaseOutStructure*>(&Features);
49// }
50// VkPhysicalDeviceMemoryDecompressionFeaturesEXT Features;
51// }
52//
53// - If properties are needed, overload GetPropertiesStruct() to return the extension specific properties struct (with sType and pNext initialized).
54// - FeaturesCallback() / PropertiesCallback() are called after those property structures are filled.
55// - If inspection of the features and/or properties reveals the extension should not be loaded, the callbacks should return false.
56// - If the extension is already handled by the engine or loaded by another plugin, DuplicateCallback() will be called.
57class FVulkanRHIExternalDeviceExtensionBase : public TSharedFromThis<FVulkanRHIExternalDeviceExtensionBase>
58{
59public:
60 FVulkanRHIExternalDeviceExtensionBase(const ANSICHAR* InExtensionName) : ExtensionName(InExtensionName) {}
61 virtual ~FVulkanRHIExternalDeviceExtensionBase() {}
62
63 // Returns a pointer to the extension's feature structure
64 virtual VkBaseOutStructure* GetFeaturesStruct() { return nullptr; }
65
66 // Callback when the features structure has been filled. Return true if the features meet your requirements.
67 virtual bool FeaturesCallback() { return true; }
68
69 // Pointer to the extension's properties structure
70 virtual VkBaseOutStructure* GetPropertiesStruct() { return nullptr; }
71
72 // Callback when the properties structure has been filled. Return true if the properties meet your requirements.
73 virtual bool PropertiesCallback() { return true; }
74
75 // Callback if the extension is found to be a duplicate (already loaded by the engine or other plugin).
76 virtual void DuplicateCallback() {}
77
78 // Name of the extension to be loaded
79 const FAnsiString ExtensionName;
80};
