57 { TCommand::CheckBreakpoints(Forward<Args>(args)...) }

406 template<typename T>
407 T& CreateOrAddCommand()
408 {
409 ASSERT_EXECUTE_DEPRECATION(T);
410
411 static_assert(!UE::Mass::Command::TCommandTraits<T>::RequiresUniqueHandling, "This command type needs to be added via PushUniqueCommand");
412 const int32 Index = FMassBatchedCommand::GetCommandIndex<T>();
413
414 if (CommandInstances.IsValidIndex(Index) == false)
415 {
416 CommandInstances.AddZeroed(Index - CommandInstances.Num() + 1);
417 }
418 else if (CommandInstances[Index])
419 {
420 return static_cast<T&>(*CommandInstances[Index].Get());
421 }
422
423 CommandInstances[Index] = MakeUnique<T>();
424 return static_cast<T&>(*CommandInstances[Index].Get());
425 }
