36 struct FChangeInfo
37 {
38 enum class EChangeFlags
39 {
40 ChangeFlagsNone = 0,
41 ChangeFlagsDelete = (1 << 0),
42 ChangeFlagsEdit = (1 << 1),
43 ChangeFlagsAdd = (1 << 2),
44 };
45
46 int64 Adds, AddedBytes;
47 int64 Changes, ChangedBytes;
48 int64 Deletes, DeletedBytes;
49 int64 Unchanged, UnchangedBytes;
50
51 FChangeInfo()
52 {
53 Adds = AddedBytes = 0;
54 Changes = ChangedBytes = 0;
55 Deletes = DeletedBytes = 0;
56 Unchanged = UnchangedBytes = 0;
57 }
58
59 float GetChangePercentage() const
60 {
61 return GetTotalChangeSize() / (float)GetTotalSize();
62 }
63
64 int64 GetTotalChangeCount() const
65 {
66 return Adds + Changes;
67 }
68
69 int64 GetTotalChangeSize() const
70 {
71 return AddedBytes + ChangedBytes;
72 }
73
74 int64 GetTotalSize() const
75 {
76 return GetTotalChangeSize() + UnchangedBytes;
77 }
78
79 int32 GetChangeFlags() const
80 {
81 int32 Flags = int32(EChangeFlags::ChangeFlagsNone);
82
83 if (Adds > 0)
84 {
85 Flags |= int32(EChangeFlags::ChangeFlagsAdd);
86 }
87
88 if (Changes > 0)
89 {
90 Flags |= int32(EChangeFlags::ChangeFlagsEdit);
91 }
92
93 if (Deletes > 0)
94 {
95 Flags |= int32(EChangeFlags::ChangeFlagsDelete);
96 }
97
98 return Flags;
99 }
100
101 const FChangeInfo& operator+=(const FChangeInfo& Rhs)
102 {
103 Adds += Rhs.Adds;
104 AddedBytes += Rhs.AddedBytes;
105 Changes += Rhs.Changes;
106 ChangedBytes += Rhs.ChangedBytes;
107 Deletes += Rhs.Deletes;
108 DeletedBytes += Rhs.DeletedBytes;
109 Unchanged += Rhs.Unchanged;
110 UnchangedBytes += Rhs.UnchangedBytes;
111
112 return *this;
113 }
114 };
