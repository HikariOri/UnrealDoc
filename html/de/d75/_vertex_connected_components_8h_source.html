80 template<typename TriangleMeshType>
81 void ConnectVerticesToNearestDifferentComponent(const TMeshAABBTree3<TriangleMeshType>& MeshSpatial, double DistanceThreshold, TFunctionRef<bool(int32)> VertexIDFilter = [](int32) {return true;}, int32 KeepSizeThreshold = 0)
82 {
83 const TriangleMeshType* Mesh = MeshSpatial.GetMesh();
84
85 for (int32 VID = 0; VID < Mesh->MaxVertexID(); VID++)
86 {
87 if (!Mesh->IsVertex(VID) || !VertexIDFilter(VID))
88 {
89 continue;
90 }
91 int32 SetID = DisjointSet.Find(VID);
92 if (KeepSizeThreshold > 0 && DisjointSet.Sizes[SetID] < KeepSizeThreshold)
93 {
94 continue;
95 }
96
97 FVector3d Pt = Mesh->GetVertex(VID);
98 double NearestDistSqr;
99 IMeshSpatial::FQueryOptions Options;
100 Options.MaxDistance = DistanceThreshold;
101 Options.TriangleFilterF = [this, Mesh, SetID](int32 TriID) -> bool
102 {
103 FIndex3i Triangle = Mesh->GetTriangle(TriID);
104 // Accept only triangles where at least one vertex is in a different component
105 return DisjointSet.Find(Triangle[0]) != SetID
106 || DisjointSet.Find(Triangle[1]) != SetID
107 || DisjointSet.Find(Triangle[2]) != SetID;
108 };
109 int32 NearestTriID = MeshSpatial.FindNearestTriangle(Pt, NearestDistSqr, Options);
110 if (NearestTriID >= 0)
111 {
112 FIndex3i Triangle = Mesh->GetTriangle(NearestTriID);
113 DisjointSet.Union(VID, Triangle[0]);
114 DisjointSet.Union(VID, Triangle[1]);
115 DisjointSet.Union(VID, Triangle[2]);
116 }
117 }

126 template<typename TriangleMeshType>
127 void ConnectCloseFilteredVertices(const TriangleMeshType& Mesh, double CloseVertexThreshold, TFunctionRef<bool(int32)> VertexIDFilter, int32 KeepSizeThreshold = 0)
128 {
129 TPointHashGrid3d<int32> VertexHash(CloseVertexThreshold * 3, -1);
130 for (int32 VID = 0; VID < Mesh.MaxVertexID(); VID++)
131 {
132 if (!Mesh.IsVertex(VID) || !VertexIDFilter(VID))
133 {
134 continue;
135 }
136 int32 SetID = DisjointSet.Find(VID);
137 if (KeepSizeThreshold > 0 && DisjointSet.Sizes[SetID] < KeepSizeThreshold)
138 {
139 continue;
140 }
141
142 int32 MaxSetSize = Mesh.VertexCount();
143 FVector3d Pt = Mesh.GetVertex(VID);
144 VertexHash.EnumeratePointsInBall(Pt, CloseVertexThreshold, [&Mesh, Pt](int32 OtherVID)
145 {
146 return DistanceSquared(Pt, Mesh.GetVertex(OtherVID));
147 }, [this, SetID, MaxSetSize](const int32& NbrVID, double DistSq)
148 {
149 int32 UnionSetID = DisjointSet.Union(SetID, NbrVID);
150 return DisjointSet.Sizes[UnionSetID] < MaxSetSize; // stop iterating if all vertices are in the same component
151 });
152 VertexHash.InsertPointUnsafe(VID, Pt);
153 }

167 // TODO: support more overlap strategies, currently just uses AABB
168 // Note this merges components based on overlap of their bounding boxes as computed *before* any merges; multiple passes may merge additional components
169 template<typename TriangleMeshType>
170 void ConnectOverlappingComponents(const TriangleMeshType& Mesh, int32 KeepSizeThreshold = 0)
171 {
172 TMap<int32, FAxisAlignedBox3d> SetIDToBounds;
173 for (int32 VID = 0; VID < Mesh.MaxVertexID(); VID++)
174 {
175 if (!Mesh.IsVertex(VID))
176 {
177 continue;
178 }
179 int32 SetID = DisjointSet.Find(VID);
180 if (KeepSizeThreshold > 0 && DisjointSet.Sizes[SetID] < KeepSizeThreshold)
181 {
182 continue;
183 }
184 FVector3d Pt = Mesh.GetVertex(VID);
185 FAxisAlignedBox3d& Bounds = SetIDToBounds.FindOrAdd(SetID, FAxisAlignedBox3d::Empty());
186 Bounds.Contain(Pt);
187 }
188 // For each component's bounding box
189 for (auto It = SetIDToBounds.CreateConstIterator(); It; ++It)
190 {
191 auto NextIt = It;
192 // Iterate over the subsequent component bounding boxes
193 ++NextIt;
194 for (; NextIt; ++NextIt)
195 {
196 // Union the two components if their bounds overlap
197 if (It.Value().Intersects(NextIt.Value()))
198 {
199 DisjointSet.Union(It.Key(), NextIt.Key());
200 }
201 }
202 }

298 // Return an ordering of the vertex indices so that each connected component is in a contiguous block
299 TArray<int32> MakeContiguousComponentsArray(int32 MaxVID)
300 {
301 TMap<int32, int32> ComponentLoc;
302 TArray<int32> Contiguous;
303 Contiguous.SetNum(MaxVID);
304 int32 LastSingleEntryIdx = MaxVID;
305 int32 FirstUnusedIdx = 0;
306 for (int32 VID = 0; VID < MaxVID; VID++)
307 {
308 int32 SetID = DisjointSet.Find(VID);
309 int32 SetSize = DisjointSet.Sizes[SetID];
310 if (SetSize == 1) // just place the single-element groups at the end, no need to track in map
311 {
312 LastSingleEntryIdx--;
313 Contiguous[LastSingleEntryIdx] = VID;
314 continue;
315 }
316 int32* Loc = ComponentLoc.Find(SetID);
317 if (Loc)
318 {
319 Contiguous[(*Loc)++] = VID;
320 }
321 else
322 {
323 Contiguous[FirstUnusedIdx] = VID;
324 ComponentLoc.Add(SetID, FirstUnusedIdx + 1);
325 FirstUnusedIdx += SetSize;
326 }
327 }
328 return Contiguous;
329 }

335 bool EnumerateContiguousComponentsFromArray(const TArray<int32>& ContiguousComponentsArray, TFunctionRef<bool(int32, TArrayView<const int32>)> ProcessComponentFn)
336 {
337 for (int32 ContigStart = 0, NextStart = -1; ContigStart < ContiguousComponentsArray.Num(); ContigStart = NextStart)
338 {
339 int32 ComponentID = GetComponent(ContiguousComponentsArray[ContigStart]);
340 int32 ComponentSize = GetComponentSize(ComponentID);
341 NextStart = ContigStart + ComponentSize;
342 if (!ensure(NextStart <= ContiguousComponentsArray.Num()))
343 {
344 return false;
345 }
346 TArrayView<const int32> ComponentView(ContiguousComponentsArray.GetData() + ContigStart, ComponentSize);
347 bool bContinue = ProcessComponentFn(ComponentID, ComponentView);
348 if (!bContinue)
349 {
350 return false;
351 }
352 }
353 return true;
