82 void RemoveAt(int32 Index, const FScriptCompactSetLayout& Layout, TFunctionRef<uint32 (const void*)> GetKeyHash, TFunctionRef<void (void*)> DestructItem)
83 {
84 check(IsValidIndex(Index));
85
86 FCompactHashTableView HashTable = this->GetHashTableView(Layout);
87
88 void* Dst = GetData(Index, Layout);
89#if UE_ENABLE_COMPACTHASHTABLE_KEYSTORAGE
90 const uint32 DstKeyHash = HashTable.GetStoredKey(Index, this->NumElements);
91#else
92 const uint32 DstKeyHash = GetKeyHash(Dst);
93#endif
94
95 const int32 LastElementIndex = this->NumElements - 1;
96 if (Index == LastElementIndex)
97 {
98 HashTable.Remove(Index, DstKeyHash, Index, 0);
99 DestructItem(Dst);
100 }
101 else
102 {
103 void* Src = GetData(LastElementIndex, Layout);
104#if UE_ENABLE_COMPACTHASHTABLE_KEYSTORAGE
105 const uint32 SrcKeyHash = HashTable.GetStoredKey(LastElementIndex, this->NumElements);
106#else
107 const uint32 SrcKeyHash = GetKeyHash(Src);
108#endif
109 HashTable.Remove(Index, DstKeyHash, LastElementIndex, SrcKeyHash);
110 DestructItem(Dst);
111
112 // Memmove is fine here as our containers already only work if data is trivially relocatable
113 FMemory::Memmove(Dst, Src, Layout.Size);
114 }
115
116 --this->NumElements;
117 }

154 void Rehash(const FScriptCompactSetLayout& Layout, TFunctionRef<uint32 (const void*)> GetKeyHash)
155 {
156 if (this->MaxElements > 0)
157 {
158 const uint8* ElementData = this->Elements.GetAllocation();
159 FCompactHashTableView HashTable = this->GetHashTableView(Layout);
160
161 HashTable.Reset();
162
163 for (int32 Index = 0; Index < this->NumElements; ++Index)
164 {
165 HashTable.Add(Index, GetKeyHash(ElementData + Layout.Size * Index));
166 }
167 }

202 void Add(const void* Element, const FScriptCompactSetLayout& Layout, TFunctionRef<uint32(const void*)> GetKeyHash, TFunctionRef<bool(const void*, const void*)> EqualityFn, TFunctionRef<void(void*)> ConstructFn, TFunctionRef<void(void*)> DestructFn)
203 {
204 uint32 KeyHash = GetKeyHash(Element);
205 int32 OldElementIndex = FindIndexByHash(Element, Layout, KeyHash, EqualityFn);
206 if (OldElementIndex != INDEX_NONE)
207 {
208 void* ElementPtr = GetData(OldElementIndex, Layout);
209
210 DestructFn(ElementPtr);
211 ConstructFn(ElementPtr);
212
213 // We don't update the hash because we don't need to - the new element
214 // should have the same hash, but let's just check.
215 PRAGMA_DISABLE_DEPRECATION_WARNINGS
216 // Disable deprecations warnings to stop warnings being thrown by our check macro.
217 checkSlow(KeyHash == GetKeyHash(ElementPtr));
218 PRAGMA_ENABLE_DEPRECATION_WARNINGS
219 }
220 else
221 {
222 AddNewElement(Layout, GetKeyHash, KeyHash, ConstructFn);
223 }

228 {

232 int32 AddNewElement(const FScriptCompactSetLayout& Layout, TFunctionRef<uint32(const void*)> GetKeyHash, uint32 KeyHash, TFunctionRef<void(void*)> ConstructFn)
233 {
234 checkSlow(this->NumElements >= 0 && this->MaxElements >= 0);
235 if (this->NumElements == this->MaxElements)
236 {
237 if (this->ResizeAllocationPreserveData(this->AllocatorCalculateSlackGrow(this->NumElements + 1, Layout), Layout))
238 {
239 Rehash(Layout, GetKeyHash);
240 }
241 }
242
243 this->GetHashTableView(Layout).Add(this->NumElements, KeyHash);
244
245 ConstructFn(GetData(this->NumElements, Layout));
246 return this->NumElements++;
247 }
