189{

213template<class Functor>
214void FNetPacketNotify::ProcessReceivedAcks(const FNotificationHeader& NotificationData, Functor&& InFunc)
215{
216 if (NotificationData.AckedSeq > OutAckSeq)
217 {
218 // Total ack count including implicit naks
219 SequenceNumberT::DifferenceT AckCount = SequenceNumberT::Diff(NotificationData.AckedSeq, OutAckSeq);
220 UE_LOG_PACKET_NOTIFY(TEXT("Notification::ProcessReceivedAcks - AckedSeq: %u, OutAckSeq: %u AckCount: %u RemoteWaitingForFlush: %u"), NotificationData.AckedSeq.Get(), OutAckSeq.Get(), AckCount, NotificationData.bRemoteIsWaitingForHistoryReset ? 1U : 0U);
221
222 // Update InAckSeqAck used to track the needed number of bits to transmit our ack history
223 // Note: As we might reset sequence history we need to check if we already have advanced the InAckSeqAck
224 {
225 const SequenceNumberT NewInAckSeqAck = UpdateInAckSeqAck(AckCount, NotificationData.AckedSeq);
226 if (NewInAckSeqAck > InAckSeqAck)
227 {
228 UE_LOG_PACKET_NOTIFY(TEXT("FNetPacketNotify::ProcessReceivedAcks - Advance InAckSeqAck: %u"), NewInAckSeqAck.Get());
229 InAckSeqAck = NewInAckSeqAck;
230 }
231 }
232
233 // ExpectedAck = OutAckSeq + 1
234 SequenceNumberT CurrentAck(OutAckSeq);
235 ++CurrentAck;
236
237 // Make sure that we only look at the sequence history bit included in the notification data as the sequence history might have been reset,
238 // in which case we might not receive the max size history even though the ack-count is bigger than the history
239 const SequenceNumberT::DifferenceT HistoryBits = static_cast<SequenceNumberT::DifferenceT>(NotificationData.HistoryWordCount * SequenceHistoryT::BitsPerWord);
240
241 // Warn if the received sequence number is greater than our history buffer, since if that is the case we have to treat the data as lost
242 // Note: This should not be a problem as we have a mechanism to re-sync history
243 // If this occurs with no hitches on server or client, there might be reason to investigate if too much data is being sent in which case the the size sequence history might have to be increased.
244 SequenceNumberT::DifferenceT MissingAckCount = 0U;
245 if (AckCount > HistoryBits)
246 {
247 MissingAckCount = AckCount - HistoryBits;
248 UE_LOG_PACKET_NOTIFY(TEXT("FNetPacketNotify::ProcessReceivedAcks - Missed Acks: AckedSeq: %u, OutAckSeq: %u, FirstMissingSeq: %u Count: %u"), NotificationData.AckedSeq.Get(), OutAckSeq.Get(), CurrentAck.Get(), MissingAckCount);
249 }
250
251 if (NotificationData.bRemoteIsWaitingForHistoryReset)
252 {
253 // Re-sync sequence history
254 // Extra care has to be taken, we might have to synthesize naks both before and after the acks carried in the sequence history.
255 if (NotificationData.AckedHistorySeq > OutAckSeq)
256 {
257 SequenceNumberT::DifferenceT AckCountInHistory = SequenceNumberT::Diff(NotificationData.AckedHistorySeq, OutAckSeq);
258
259 // Synthesize missing naks before start of sequence history, this can occur if we get burst packet drops and then overflow history
260 while (AckCountInHistory > HistoryBits)
261 {
262 --AckCountInHistory;
263 --AckCount;
264 InFunc(CurrentAck, false);
265 ++CurrentAck;
266 }
267
268 // Acknowledge packets based on information from sequence history
269 while (AckCountInHistory > 0)
270 {
271 --AckCountInHistory;
272 --AckCount;
273
274 const uint32 AckIndex = AckCountInHistory;
275
276 UE_LOG_PACKET_NOTIFY(TEXT("Notification::ProcessReceivedAcks Seq: %u - IsAck: %u HistoryIndex: %u"), CurrentAck.Get(), NotificationData.History.IsDelivered(AckIndex) ? 1u : 0u, AckIndex);
277 InFunc(CurrentAck, NotificationData.History.IsDelivered(AckIndex));
278 ++CurrentAck;
279 }
280 }
281
282 // Synthesize missing naks after history window as the packets are discarded during re-sync
283 while (AckCount > 0)
284 {
285 --AckCount;
286 InFunc(CurrentAck, false);
287 ++CurrentAck;
288 }
289 }
290 else
291 {
292 // Normal path, missing naks are at the beginning
293
294 // Synthesize missing naks before start of history window, this can occur if we get burst packet drops with no pending acks
295 while (AckCount > HistoryBits)
296 {
297 --AckCount;
298 InFunc(CurrentAck, false);
299 ++CurrentAck;
300 }
301
302 // For sequence numbers contained in the history we lookup the delivery status from the history
303 while (AckCount > 0)
304 {
305 --AckCount;
306 UE_LOG_PACKET_NOTIFY(TEXT("Notification::ProcessReceivedAcks Seq: %u - IsAck: %u HistoryIndex: %u"), CurrentAck.Get(), NotificationData.History.IsDelivered(AckCount) ? 1u : 0u, AckCount);
307 InFunc(CurrentAck, NotificationData.History.IsDelivered(AckCount));
308 ++CurrentAck;
309 }
310 }
311
312 // Reset history if we are done with reset of sequence history.
313 if (IsWaitingForSequenceHistoryFlush() && NotificationData.AckedSeq >= WaitingForFlushSeqAck)
314 {
315 UE_LOG_PACKET_NOTIFY(TEXT("ClearIsWaitingForSequenceHistoryFlush %u"), NotificationData.AckedSeq.Get());
316 InSeqHistory.Reset();
317 WaitingForFlushSeqAck = NotificationData.AckedSeq;
318 }
319
320 OutAckSeq = NotificationData.AckedSeq;
321
322 // Deal with sequence wraparound.
323 if (OutAckSeq > WaitingForFlushSeqAck)
324 {
325 WaitingForFlushSeqAck = OutAckSeq;
326 }
327 }
