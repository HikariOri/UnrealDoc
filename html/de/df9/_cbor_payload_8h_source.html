204inline bool FResponse::GetStringArray(const char* Key, TArray<FString>& OutArray) const
205{
206 FMemoryReader MemoryReader(Buffer);
207 FCborReader CborReader(&MemoryReader, ECborEndianness::StandardCompliant);
208 FCborContext Context;
209 while (CborReader.ReadNext(Context))
210 {
211 if (Context.IsString())
212 {
213 uint32 Length = uint32(Context.AsLength());
214 uint32 Offset = uint32(MemoryReader.Tell());
215 auto* String = (const char*)(Buffer.GetData() + Offset - Length);
216 if (FCStringAnsi::Strncmp(Key, String, Length) == 0)
217 {
218 if (CborReader.ReadNext(Context) && Context.IsContainer() && Context.MajorType() == ECborCode::Array)
219 {
220 while (CborReader.ReadNext(Context) && Context.IsString())
221 {
222 OutArray.Emplace(Context.AsString());
223 }
224 return true;
225 }
226 }
227 }
228 }
229 return false;
230}

252template <typename Type, typename LambdaType>
253inline Type FResponse::GetValue(const char* Key, Type Default, LambdaType&& Lambda) const
254{
255 FMemoryReader MemoryReader(Buffer);
256 FCborReader CborReader(&MemoryReader, ECborEndianness::StandardCompliant);
257 FCborContext Context;
258
259 if (!CborReader.ReadNext(Context) || Context.MajorType() != ECborCode::Map)
260 {
261 return Default;
262 }
263
264 while (true)
265 {
266 // Read key
267 if (!CborReader.ReadNext(Context) || !Context.IsString())
268 {
269 return Default;
270 }
271
272 uint32 Length = static_cast<uint32>(Context.AsLength());
273 uint32 Offset = static_cast<uint32>(MemoryReader.Tell());
274 auto* String = (const char*)(Buffer.GetData() + Offset - Length);
275 bool bIsTarget = (FCStringAnsi::Strncmp(Key, String, Length) == 0);
276
277 // Read value
278 if (!CborReader.ReadNext(Context))
279 {
280 return Default;
281 }
282
283 if (bIsTarget)
284 {
285 return Lambda(Context, static_cast<uint32>(MemoryReader.Tell()));
286 }
287 }
288}
