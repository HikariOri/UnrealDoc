217 // This gets called if all registered handlers return false to HandleRequest. If there are
218 // _no_ handlers then this is entirely sidestepped in the cloud chunk source.
219 virtual bool HandleRequest(const FChunkUriRequest& Request, TFunction<void(FChunkUriResponse)> OnResponse) override
220 {
221 FChunkUriResponse Response;
222 Response.Uri = Request.CloudDirectory / Request.ChunkCloudSubdirectory / Request.ChunkPath;
223 OnResponse(MoveTemp(Response));
224 return true;
225 }

227 virtual bool HandleRequest(const FDeltaUriRequest& Request, TFunction<void(FExpiratedUrisResponse)> OnResponse) override

230 // Expiration time is not needed in the default case, setting this value explicitly empty

232 OnResponse(MoveTemp(Response));
233 return true;
234 }
