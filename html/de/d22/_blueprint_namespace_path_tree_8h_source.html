112 void RemovePath(const FString& InPath)
113 {
114 TArray<FString> PathSegments;
115 InPath.ParseIntoArray(PathSegments, PathSeparator);
116
117 using FPathTraceNode = TTuple<FName, TSharedPtr<FNode>>;
118 TArray<FPathTraceNode> PathTrace;
119 PathTrace.Reserve(PathSegments.Num() + 1);
120
121 TSharedPtr<FNode> Node = GetRootNode();
122 PathTrace.Push(MakeTuple(NAME_None, Node));
123 for (const FString& PathSegment : PathSegments)
124 {
125 FName SubPathName = FName(*PathSegment);
126 if (const TSharedPtr<FNode>* NodePtr = Node->Children.Find(SubPathName))
127 {
128 Node = *NodePtr;
129 PathTrace.Push(MakeTuple(SubPathName, Node));
130 }
131 }
132
133 FName LastKey = NAME_None;
134 TSharedPtr<FNode> LastNode;
135 while (PathTrace.Num() > 0)
136 {
137 const FPathTraceNode& PathTraceNode = PathTrace.Pop();
138
139 TSharedPtr<FNode> CurrNode = PathTraceNode.Get<1>();
140 check(CurrNode.IsValid());
141
142 // On the first pass, LastNode will not be valid, so we clear the added flag on the leaf node. On the
143 // next pass, the current node equates to the parent of the leaf node, and we remove the leaf node only
144 // if it has an empty subtree. All subsequent passes will iterate from there back up to the root node,
145 // where at each level we examine the last child node to see if it can be removed (no children and
146 // not explicitly added; i.e. - not a wildcard path (e.g. X.Y.*)). This means that we will not be left
147 // with a subtree that contains only non-explicitly added paths (i.e. we're also minimizing the tree).
148 //
149 // Example: Remove path "X.Y.Z" from the following tree:
150 //
151 // Root <-- The root node is never removed.
152 // |
153 // +- X <-- Node "X" will not be removed even if "Y" is removed, b/c its subtree is still not empty (i.e. "X.W" would still be a valid path).
154 // |
155 // +- Y <-- Node "Y" will be removed (*unless* "X.Y" was explicitly added) only if it's not left with any children after node "Z" is removed.
156 // | |
157 // | +- Z <-- Node "Z" will not be removed if its subtree is not empty. Since this is the leaf node, we clear the "added" flag and shift up.
158 // | |
159 // | +- T
160 // +- W
161 //
162 if (!LastNode.IsValid())
163 {
164 CurrNode->bIsAddedPath = false;
165 }
166 else if(!LastNode->bIsAddedPath && LastNode->Children.Num() == 0)
167 {
168 TSharedPtr<FNode> RemovedNode = CurrNode->Children.FindAndRemoveChecked(LastKey);
169 check(RemovedNode == LastNode);
170 }
171
172 LastKey = PathTraceNode.Get<0>();
173 LastNode = CurrNode;
174 }
175 }
