47 template<int NumSmallPools>
48 struct TGlobalRecycler
49 {
50 bool PushBundle(uint32 InPoolIndex, FMallocBinnedCommonBase::FBundleNode* InBundle)
51 {
52 const uint32 NumCachedBundles = FMath::Min<uint32>(GMallocBinnedMaxBundlesBeforeRecycle, UE_DEFAULT_GMallocBinnedMaxBundlesBeforeRecycle);
53 for (uint32 Slot = 0; Slot < NumCachedBundles; Slot++)
54 {
55 if (!Bundles[InPoolIndex].FreeBundles[Slot])
56 {
57 if (!FPlatformAtomics::InterlockedCompareExchangePointer((void**)&Bundles[InPoolIndex].FreeBundles[Slot], InBundle, nullptr))
58 {
59 return true;
60 }
61 }
62 }
63 return false;
64 }
65
66 FMallocBinnedCommonBase::FBundleNode* PopBundle(uint32 InPoolIndex)
67 {
68 const uint32 NumCachedBundles = FMath::Min<uint32>(GMallocBinnedMaxBundlesBeforeRecycle, UE_DEFAULT_GMallocBinnedMaxBundlesBeforeRecycle);
69 for (uint32 Slot = 0; Slot < NumCachedBundles; Slot++)
70 {
71 FMallocBinnedCommonBase::FBundleNode* Result = Bundles[InPoolIndex].FreeBundles[Slot];
72 if (Result)
73 {
74 if (FPlatformAtomics::InterlockedCompareExchangePointer((void**)&Bundles[InPoolIndex].FreeBundles[Slot], nullptr, Result) == Result)
75 {
76 return Result;
77 }
78 }
79 }
80 return nullptr;
81 }
82
83 private:
84 struct FBundlePointer
85 {
86 alignas(PLATFORM_CACHE_LINE_SIZE) FMallocBinnedCommonBase::FBundleNode* FreeBundles[UE_DEFAULT_GMallocBinnedMaxBundlesBeforeRecycle];
87
88 FBundlePointer()
89 {
90 DefaultConstructItems<FMallocBinnedCommonBase::FBundleNode*>(FreeBundles, UE_DEFAULT_GMallocBinnedMaxBundlesBeforeRecycle);
91 }
92 };
93 static_assert(sizeof(FBundlePointer) == PLATFORM_CACHE_LINE_SIZE, "FBundlePointer should be the same size as a cache line");
94 alignas(PLATFORM_CACHE_LINE_SIZE) FBundlePointer Bundles[NumSmallPools];
95 };
96}

102 template <class AllocType>
103 static void TrimThreadFreeBlockLists(AllocType& Allocator, typename AllocType::FPerThreadFreeBlockLists* FreeBlockLists)
104 {
105 if (FreeBlockLists)
106 {
107 TRACE_CPUPROFILER_EVENT_SCOPE(FMallocBinnedCommonUtils::TrimThreadFreeBlockLists);
108
109 for (int32 PoolIndex = 0; PoolIndex != AllocType::NUM_SMALL_POOLS; ++PoolIndex)
110 {
111 typename AllocType::FBundleNode* Bundles = FreeBlockLists->PopBundles(PoolIndex);
112 if (Bundles)
113 {
114 Allocator.FreeBundles(Bundles, PoolIndex);
115 }
116 }
117 }
118 }
119
120 template <class AllocType>
121 static void FlushCurrentThreadCache(AllocType& Allocator, bool bNewEpochOnly = false)
122 {
123 if (typename AllocType::FPerThreadFreeBlockLists* Lists = AllocType::FPerThreadFreeBlockLists::Get())
124 {
125 if (Lists->UpdateEpoch(Allocator.MemoryTrimEpoch.load(std::memory_order_relaxed)) || !bNewEpochOnly)
126 {
127 TRACE_CPUPROFILER_EVENT_SCOPE(FMallocBinnedCommonUtils::FlushCurrentThreadCache);
128 QUICK_SCOPE_CYCLE_COUNTER(STAT_FMallocBinnedCommonUtils_FlushCurrentThreadCache);
129
130 const double StartTimeInner = FPlatformTime::Seconds();
131 TrimThreadFreeBlockLists(Allocator, Lists);
132 const double WaitForTrimTime = FPlatformTime::Seconds() - StartTimeInner;
133
134 if (WaitForTrimTime > GMallocBinnedFlushThreadCacheMaxWaitTime)
135 {
136 UE_LOGF(LogMemory, Warning, "FMalloc%ls took %6.2fms to wait for mutex AND trim.", Allocator.GetDescriptiveName(), WaitForTrimTime * 1000.0f);
137 }
138 }
139 }
140 }

142 template <class AllocType>
143 static void Trim(AllocType& Allocator)
144 {
145 // Update the trim epoch so that threads cleanup their thread-local memory when going to sleep.
146 Allocator.MemoryTrimEpoch.fetch_add(1, std::memory_order_relaxed);
147
148 QUICK_SCOPE_CYCLE_COUNTER(STAT_FMallocBinnedCommonUtils_Trim);
149
150 // Process thread-local memory caches from as many threads as possible without waking them up.
151 // Skip on desktop as we may have too many threads and this could cause some hitches.
152 if (!PLATFORM_DESKTOP && GMallocBinnedFlushRegisteredThreadCachesOnOneThread != 0)
153 {
154 UE::TUniqueLock FreeBlockLock(AllocType::GetFreeBlockListsRegistrationMutex());
155 for (typename AllocType::FPerThreadFreeBlockLists* BlockList : AllocType::GetRegisteredFreeBlockLists())
156 {
157 // If we're unable to lock, it's because the thread is currently active so it
158 // will do the flush itself when going back to sleep because we incremented the Epoch.
159 if (BlockList->TryLock())
160 {
161 // Only trim if the epoch has been updated, otherwise the thread already
162 // did the trimming when it went to sleep.
163 if (BlockList->UpdateEpoch(Allocator.MemoryTrimEpoch.load(std::memory_order_relaxed)))
164 {
165 TrimThreadFreeBlockLists(Allocator, BlockList);
166 }
167 BlockList->Unlock();
168 }
169 }
170 }
171
172 TFunction<void(ENamedThreads::Type CurrentThread)> Broadcast =
173 [&Allocator](ENamedThreads::Type MyThread)
174 {
175 // We might already have updated the Epoch so we can skip doing anything costly (i.e. Mutex) in that case.
176 const bool bNewEpochOnly = true;
177 FlushCurrentThreadCache(Allocator, bNewEpochOnly);
178 };
179
180 // Skip task threads on desktop platforms as it is too slow and they don't have much memory
181 if (PLATFORM_DESKTOP)
182 {
183 FTaskGraphInterface::BroadcastSlow_OnlyUseForSpecialPurposes(false, false, Broadcast);
184 }
185 else
186 {
187 FTaskGraphInterface::BroadcastSlow_OnlyUseForSpecialPurposes(FPlatformProcess::SupportsMultithreading() && FApp::ShouldUseThreadingForPerformance(), false, Broadcast);
188 }
189 }

194#if UE_MBC_ALLOCATOR_STATS && CSV_PROFILER_STATS
