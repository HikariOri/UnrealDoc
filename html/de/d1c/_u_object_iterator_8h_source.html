52enum class EObjectIteratorThreadSafetyOptions : uint8
53{
54 None = 0,
55 ThreadSafe = 1, // Use only with function local iterators. Persistent iterators may prevent (hang) UObject creation on worker threads otherwise.
56 ThreadSafeAdvance = 2 // Can be used for global iterators but locks the global UObjectArray each time the iterator advances to the next object (can be slow but does not result in deadlocks).
57};

74template <EObjectIteratorThreadSafetyOptions IteratorThreadSafety>
75class TObjectIteratorBase : public FUObjectArray::TIterator
76{
77protected:
78
80 inline bool AdvanceIterator()
81 {
82 bool bResult = false;
83 if (IteratorThreadSafety != EObjectIteratorThreadSafetyOptions::ThreadSafeAdvance)
84 {
85 bResult = Advance();
86 }
87 else
88 {
89 GetIteratedArray().LockInternalArray();
90 bResult = Advance();
91 GetIteratedArray().UnlockInternalArray();
92 }
93 return bResult;
94 }
95
96public:
97
106 explicit TObjectIteratorBase(UClass* InClass = UObject::StaticClass(), bool bOnlyGCedObjects = false, EObjectFlags AdditionalExclusionFlags = RF_NoFlags, EInternalObjectFlags InInternalExclusionFlags = EInternalObjectFlags::None)
107 : FUObjectArray::TIterator(GUObjectArray, bOnlyGCedObjects)
108 , Class(InClass)
109 , ExclusionFlags(AdditionalExclusionFlags)
110 , InternalExclusionFlags(GetObjectIteratorDefaultInternalExclusionFlags(InInternalExclusionFlags))
111 {
112 check(Class);
113
114 if (IteratorThreadSafety == EObjectIteratorThreadSafetyOptions::ThreadSafe)
115 {
116 GetIteratedArray().LockInternalArray();
117 }
118
119 do
120 {
121 UObject* Object = **this;
122 if (!(Object->HasAnyFlags(ExclusionFlags) || Object->HasAnyInternalFlags(InternalExclusionFlags) || (Class != UObject::StaticClass() && !Object->IsA(Class))))
123 {
124 break;
125 }
126 } while (AdvanceIterator());
127 }
128
134 TObjectIteratorBase(FUObjectArray::TIterator::EEndTagType, const TObjectIteratorBase& Begin)
135 : FUObjectArray::TIterator(FUObjectArray::TIterator::EndTag, Begin)
136 , Class(Begin.Class)
137 , ExclusionFlags(Begin.ExclusionFlags)
138 , InternalExclusionFlags(Begin.InternalExclusionFlags)
139 {
140 if (IteratorThreadSafety == EObjectIteratorThreadSafetyOptions::ThreadSafe)
141 {
142 GetIteratedArray().LockInternalArray();
143 }
144 }
145
149 TObjectIteratorBase(const TObjectIteratorBase& Other)
150 : FUObjectArray::TIterator(Other)
151 , Class(Other.Class)
152 , ExclusionFlags(Other.ExclusionFlags)
153 , InternalExclusionFlags(Other.InternalExclusionFlags)
154 {
155 if (IteratorThreadSafety == EObjectIteratorThreadSafetyOptions::ThreadSafe)
156 {
157 GetIteratedArray().LockInternalArray();
158 }
159 }
160
164 ~TObjectIteratorBase()
165 {
166 if (IteratorThreadSafety == EObjectIteratorThreadSafetyOptions::ThreadSafe)
167 {
168 GetIteratedArray().UnlockInternalArray();
169 }
170 }
171
175 void operator++()
176 {
177 //@warning: behavior is partially mirrored in UnObjGC.cpp. Make sure to adapt code there as well if you make changes below.
178 // verify that the async loading exclusion flag still matches (i.e. we didn't start/stop async loading within the scope of the iterator)
179 checkSlow(IsInAsyncLoadingThread() || int32(InternalExclusionFlags & EInternalObjectFlags_AsyncLoading));
180
181 while (AdvanceIterator())
182 {
183 UObject* Object = **this;
184 if (!(Object->HasAnyFlags(ExclusionFlags) || (Class != UObject::StaticClass() && !Object->IsA(Class)) || Object->HasAnyInternalFlags(InternalExclusionFlags)))
185 {
186 break;
187 }
188 }
189 }
190
194 inline UObject* operator*() const
195 {
196 // casting UObjectBase to UObject for clients
197 FUObjectItem* ObjectItem = GetObject();
198 return (UObject*)(ObjectItem ? ObjectItem->GetObject() : nullptr);
199 }
200
204 inline UObject* operator->() const
205 {
206 FUObjectItem* ObjectItem = GetObject();
207 return (UObject*)(ObjectItem ? ObjectItem->GetObject() : nullptr);
208 }
209private:
211 UClass* Class;
212protected:
214 EObjectFlags ExclusionFlags;
216 EInternalObjectFlags InternalExclusionFlags;
217};
