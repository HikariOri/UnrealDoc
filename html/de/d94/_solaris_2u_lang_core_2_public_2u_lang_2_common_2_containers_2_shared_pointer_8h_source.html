257 void EnableRelease()
258 {
259 _ReleaseFunc = [](ObjectType * Object, const AllocatorType & Allocator)
260 {
261 if (Object->Dereference())
262 {
263 // No references left: Delete the object
264 Object->~ObjectType();
265 Allocator.Deallocate(Object);
266 }
267 };
