131 enum class ETaskState : int8
132 {
133 ReadyState = 0,
134 CanceledFlag = 1 << 0,
135 ScheduledFlag = 1 << 1,
136 RunningFlag = 1 << 2,
137 ExpeditingFlag = 1 << 3,
138 ExpeditedFlag = 1 << 4,
139 CompletedFlag = 1 << 5, //the default state when we create a handle
140 Count = (1 << 6) - 1,
141
142 Ready = ReadyState, //means the Task is ready to be launched
143 CanceledAndReady = Ready | CanceledFlag, //means the task was canceled and is ready to be launched (it still is required to be launched)
144 Scheduled = Ready | ScheduledFlag, //means the task is launched and therefore queued for execution by a worker
145 Canceled = CanceledAndReady | ScheduledFlag, //means the task was canceled and launched and therefore queued for execution by a worker (which already might be executing it's continuation)
146 Running = Scheduled | RunningFlag, //means the task is executing it's runnable and continuation by a worker
147 CanceledAndRunning = Canceled | RunningFlag, //means the task is executing it's continuation but the runnable was cancelled
148 Expediting = Running | ExpeditingFlag, //means the task is expediting and the scheduler has released it's reference to the expediting thread before that was finished
149 Expedited = Expediting | ExpeditedFlag, //means the task was expedited
150 Completed = Running | CompletedFlag, //means the task is completed with execution
151 ExpeditedAndCompleted = Expedited | CompletedFlag, //means the task is completed with execution and the runnable was expedited
152 CanceledAndCompleted = CanceledAndRunning | CompletedFlag, //means the task is completed with execution of it's continuation but the runnable was cancelled
153 };
154 ENUM_CLASS_FLAGS(ETaskState)

520 inline bool FTask::TryCancel(ECancellationFlags CancellationFlags)
521 {
522 bool bPrelaunchCancellation = EnumHasAnyFlags(CancellationFlags, ECancellationFlags::PrelaunchCancellation);
523 bool bTryLaunchOnSuccess = EnumHasAllFlags(CancellationFlags, ECancellationFlags::PrelaunchCancellation | ECancellationFlags::TryLaunchOnSuccess);
524
525 FPackedData LocalPackedData = PackedData.load(std::memory_order_relaxed);
526 FPackedData ReadyState(LocalPackedData, ETaskState::Ready);
527 FPackedData ScheduledState(LocalPackedData, ETaskState::Scheduled);
528 //to launch a canceled task it has to go though TryPrepareLaunch which is doing the memory_order_release
529 bool WasCanceled = EnumHasAnyFlags(LocalPackedData.GetFlags(), ETaskFlags::AllowCancellation)
530 && ((bPrelaunchCancellation && PackedData.compare_exchange_strong(ReadyState, FPackedData(LocalPackedData, ETaskState::CanceledAndReady), std::memory_order_acquire))
531 || PackedData.compare_exchange_strong(ScheduledState, FPackedData(LocalPackedData, ETaskState::Canceled), std::memory_order_acquire));
532
533 if(bTryLaunchOnSuccess && WasCanceled && TryPrepareLaunch())
534 {
535 verifySlow(ExecuteTask() == nullptr);
536 return true;
537 }
538 return WasCanceled;
539 }

574 template<bool bIsExpeditingThread>
575 inline void FTask::TryFinish()
576 {
577 const ETaskState NextState = bIsExpeditingThread ? ETaskState::ExpeditedFlag | ETaskState::ExpeditingFlag : ETaskState::ExpeditingFlag;
578 ETaskState PreviousState = PackedData.fetch_or(NextState, std::memory_order_acq_rel);
579 if constexpr (bIsExpeditingThread)
580 {
581 checkSlow(PreviousState == ETaskState::Running || PreviousState == ETaskState::Expediting);
582 }
583 if(EnumHasAnyFlags(PreviousState, ETaskState::ExpeditingFlag))
584 {
585 FTaskDelegate LocalRunnable = MoveTemp(Runnable);
586 //do not access the task again after this call
587 //as by defitition the task can be considered dead
588 PreviousState = PackedData.fetch_or(ETaskState::CompletedFlag, std::memory_order_seq_cst);
589 checkSlow(PreviousState == ETaskState::Expedited);
590 }

614 inline FTask* FTask::ExecuteTask()
615 {
616 ETaskState PreviousState = PackedData.fetch_or(ETaskState::RunningFlag, std::memory_order_acquire);
617 checkSlow(EnumHasAnyFlags(PreviousState, ETaskState::ScheduledFlag));
618
619 FTask* Continuation = nullptr;
620 if(!EnumHasAnyFlags(PreviousState, ETaskState::RunningFlag)) //we are running or canceled
621 {
622 FTaskDelegate LocalRunnable;
623 Continuation = Runnable.CallAndMove(LocalRunnable, !EnumHasAnyFlags(PreviousState, ETaskState::CanceledFlag));
624 //do not access the task again after this call
625 //as by defitition the task can be considered dead
626 PreviousState = PackedData.fetch_or(ETaskState::CompletedFlag, std::memory_order_seq_cst);
627 checkSlow(PreviousState == ETaskState::Running || PreviousState == ETaskState::CanceledAndRunning);
628 }
629 else // we are expedited
630 {
631 checkSlow(PreviousState == ETaskState::Running || PreviousState == ETaskState::Expediting || PreviousState == ETaskState::Expedited);
632 TryFinish<false>();
633 }
634
635 return Continuation;
