11 enum class EFeatureLevel : int32
12 {
13 // The original version.
14 Original = 0,
15 // Support for custom fields.
16 CustomFields,
17 // Started storing the version number.
18 StartStoringVersion,
19 // Made after data files where renamed to include the hash value, these chunks now go to ChunksV2.
20 DataFileRenames,
21 // Manifest stores whether build was constructed with chunk or file data.
22 StoresIfChunkOrFileData,
23 // Manifest stores group number for each chunk/file data for reference so that external readers don't need to know how to calculate them.
24 StoresDataGroupNumbers,
25 // Added support for chunk compression, these chunks now go to ChunksV3. NB: Not File Data Compression yet.
26 ChunkCompressionSupport,
27 // Manifest stores product prerequisites info.
28 StoresPrerequisitesInfo,
29 // Manifest stores chunk download sizes.
30 StoresChunkFileSizes,
31 // Manifest can optionally be stored using UObject serialization and compressed.
32 StoredAsCompressedUClass,
33 // These two features were removed and never used.
34 UNUSED_0,
35 UNUSED_1,
36 // Manifest stores chunk data SHA1 hash to use in place of data compare, for faster generation.
37 StoresChunkDataShaHashes,
38 // Manifest stores Prerequisite Ids.
39 StoresPrerequisiteIds,
40 // The first minimal binary format was added. UObject classes will no longer be saved out when binary selected.
41 StoredAsBinaryData,
42 // Temporary level where manifest can reference chunks with dynamic window size, but did not serialize them. Chunks from here onwards are stored in ChunksV4.
43 VariableSizeChunksWithoutWindowSizeChunkInfo,
44 // Manifest can reference chunks with dynamic window size, and also serializes them.
45 VariableSizeChunks,
46 // Manifest uses a build id generated from its metadata.
47 UsesRuntimeGeneratedBuildId,
48 // Manifest uses a build id generated unique at build time, and stored in manifest.
49 UsesBuildTimeGeneratedBuildId,
50 // Manifests generated with this feature level onwards will store the MD5 hash and the calculated MIME type of each file.
51 StoresFileMD5HashesAndMIMEType,
52 // Manifests generated with this feature level onwards will store the SHA256 hash of each file.
53 StoresFileSHA256Hashes,
54 // Added support for Uninstall Actions.
55 StoresUninstallActions,
56 // Added full support for chunk encryption, additionally stores encyption secrets, as well as the CompressedDataSize and AuthTag
57 // for each chunk. Chunks from here onwards are stored in ChunksV5.
58 ChunkEncryptionSupport,
59 // Added the secretID as part of the chunk pathing within a CloudDir
60 ChunksStoredBySecret,
61 // Completed full support for BPS data encryption, if a manifest got encrypted, it will additionally store encryption data
62 // and most fields will have been replaced by empty or functional but not accurate data.
63 ManifestEncryptionSupport,
64
65 // !! Always after the latest version entry, signifies the latest version plus 1 to allow the following Latest alias.
66 LatestPlusOne,
67 // An alias for the actual latest version value.
68 Latest = (LatestPlusOne - 1),
69 // An alias to provide the latest version of a manifest supported by file data (nochunks).
70 LatestNoChunks = StoresChunkFileSizes,
71 // An alias to provide the latest version of a manifest supported by a json serialized format.
72 LatestJson = StoresPrerequisiteIds,
73 // An alias to provide the latest version of a manifest supported by generation runs on platforms without OpenSSL module support.
74 LatestNoOpenSSL = StoresFileMD5HashesAndMIMEType,
75 // An alias to provide the latest version of a manifest with unencrypted chunks.
76 LatestUnencryptedChunks = StoresUninstallActions,
77 // An alias to provide the first available version of optimised delta manifest saving.
78 FirstOptimisedDelta = UsesRuntimeGeneratedBuildId,
79
80 // More aliases, but this time for values that have been renamed
81 StoresUniqueBuildId = UsesRuntimeGeneratedBuildId,
82
83 // JSON manifests were stored with a version of 255 during a certain CL range due to a bug.
84 // We will treat this as being StoresChunkFileSizes in code.
85 BrokenJsonVersion = 255,
86 // This is for UObject default, so that we always serialize it.
87 Invalid = -1
88 };
89}
