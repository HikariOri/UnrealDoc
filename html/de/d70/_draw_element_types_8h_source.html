440 void SetBrush(const FSlateBrush* InBrush, UE::Slate::FDeprecateVector2DParameter InLocalSize, float DrawScale)
441 {
442 check(InBrush);
443 ensureMsgf(InBrush->GetDrawType() != ESlateBrushDrawType::NoDrawType, TEXT("This should have been filtered out earlier in the Make... call."));
444
445 // Note: Do not store the brush. It is possible brushes are destroyed after an element is enqueued for rendering
446 Margin = InBrush->GetMargin();
447 UVRegion = InBrush->GetUVRegion();
448 Tiling = InBrush->GetTiling();
449 Mirroring = InBrush->GetMirroring();
450 DrawType = InBrush->GetDrawType();
451 const FSlateResourceHandle& Handle = InBrush->GetRenderingResource(InLocalSize, DrawScale);
452 if (Handle.IsValid())
453 {
454 ResourceProxy = Handle.GetResourceProxy();
455 }
456 else
457 {
458 ResourceProxy = nullptr;
459 }
460 }

502 FontInfo = InFontInfo;
