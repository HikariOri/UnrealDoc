435 void Add(FFileIoStoreReadRequest* Request)
436 {
437#if CHECK_IO_STORE_READ_REQUEST_LIST_MEMBERSHIP
438 check(Request->ListCookie == 0);
439 Request->ListCookie = ListCookie;
440#endif
441
442 if (Tail)
443 {
444 Tail->Next = Request;
445 Request->Previous = Tail;
446 }
447 else

449 Head = Request;
450 Request->Previous = nullptr;

457 // Remove all FFileIoStoreReadRequests from List and add them to this list
458 void AppendSteal(FFileIoStoreReadRequestList& List)
459 {
460 if (List.Head)
461 {
462 FFileIoStoreReadRequest *ListHead = List.Head, *ListTail = List.Tail;
463 List.Clear();
464 AppendSteal(ListHead, ListTail);
465 }

468 void Remove(FFileIoStoreReadRequest* Request)
469 {
470#if CHECK_IO_STORE_READ_REQUEST_LIST_MEMBERSHIP
471 check(Request);
472 check(Request->ListCookie == ListCookie);
473 Request->ListCookie = 0;
474#endif
475
476 if (Head == Request && Tail == Request)
477 {
478 check(Request->Next == nullptr);
479 check(Request->Previous == nullptr);
480
481 Head = Tail = nullptr;
482 }
483 else if (Head == Request)
484 {
485 check(Request->Previous == nullptr);
486
487 Head = Request->Next;
488 Head->Previous = nullptr;
489 Request->Next = nullptr;
490 }
491 else if (Tail == Request)
492 {
493 check(Request->Next == nullptr);
494
495 Tail = Request->Previous;
496 Tail->Next = nullptr;
497 Request->Previous = nullptr;
498 }
499 else
500 {
501 check(Request->Next != nullptr && Request->Previous != nullptr); // Neither head nor tail should mean both links are live
502
503 Request->Next->Previous = Request->Previous;
504 Request->Previous->Next = Request->Next;
505
506 Request->Next = Request->Previous = nullptr;
507 }
508 }

511 {
512#if CHECK_IO_STORE_READ_REQUEST_LIST_MEMBERSHIP
513 for (FFileIoStoreReadRequest* Cursor = Head; Cursor; Cursor = Cursor->Next)
514 {
515 check(Cursor->ListCookie == ListCookie);
516 Cursor->ListCookie = 0;
517 }
518#endif
519 Head = Tail = nullptr;
520 }

524 FFileIoStoreReadRequest* Tail = nullptr;
525
526#if CHECK_IO_STORE_READ_REQUEST_LIST_MEMBERSHIP
527 uint32 ListCookie;
528 static PAKFILE_API uint32 NextListCookie;
529#endif
530
531 void AppendSteal(FFileIoStoreReadRequest* ListHead, FFileIoStoreReadRequest* ListTail)
532 {
533 check(ListHead);
534 check(ListTail);
535 check(!ListTail->Next);
536 check(!ListHead->Previous);
537 check(ListTail == ListHead || ListTail->Previous != nullptr);
538 check(ListTail == ListHead || ListHead->Next != nullptr);
539
540#if CHECK_IO_STORE_READ_REQUEST_LIST_MEMBERSHIP
541 for (FFileIoStoreReadRequest* Cursor = ListHead; Cursor; Cursor = Cursor->Next)
542 {
543 check(Cursor->ListCookie == 0);
544 Cursor->ListCookie = ListCookie;
545 }
546#endif
547
548 if (Tail)
549 {
550 Tail->Next = ListHead;
551 ListHead->Previous = Tail;
552 }
553 else
554 {
555 Head = ListHead;
556 ListHead->Previous = nullptr;
557 }
558 Tail = ListTail;
559 }
