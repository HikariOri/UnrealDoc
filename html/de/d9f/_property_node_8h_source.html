828 static TSharedRef< FPropertyPath > CreatePropertyPath(const TSharedRef< FPropertyNode >& PropertyNode)
829 {
830 TArray< FPropertyInfo > Properties;
831 FPropertyNode* CurrentNode = &PropertyNode.Get();
832
833 if (CurrentNode != nullptr && CurrentNode->AsCategoryNode() != nullptr)
834 {
835 TSharedRef< FPropertyPath > NewPath = MakeShared<FPropertyPath>();
836 return NewPath;
837 }
838
839 while (CurrentNode != nullptr)
840 {
841 if (CurrentNode->AsItemPropertyNode() != nullptr)
842 {
843 FPropertyInfo NewPropInfo;
844 NewPropInfo.Property = CurrentNode->GetProperty();
845 if (!NewPropInfo.Property.IsValid())
846 {
847 return MakeShared<FPropertyPath>();
848 }
849 NewPropInfo.ArrayIndex = CurrentNode->GetArrayIndex();
850
851 Properties.Add(NewPropInfo);
852 }
853
854 CurrentNode = CurrentNode->GetParentNode();
855 }
856
857 TSharedRef< FPropertyPath > NewPath = MakeShared<FPropertyPath>();
858
859 for (int PropertyIndex = Properties.Num() - 1; PropertyIndex >= 0; --PropertyIndex)
860 {
861 NewPath->AddProperty(Properties[PropertyIndex]);
862 }
863
864 return NewPath;
865 }

867 static TSharedPtr< FPropertyNode > FindPropertyNodeByPath(const TSharedPtr< FPropertyPath > Path, const TSharedRef< FPropertyNode >& StartingNode)
868 {
869 if (!Path.IsValid() || Path->GetNumProperties() == 0)
870 {
871 return StartingNode;
872 }
873
874 bool FailedToFindProperty = false;
875 TSharedPtr< FPropertyNode > PropertyNode = StartingNode;
876 for (int PropertyIndex = 0; PropertyIndex < Path->GetNumProperties() && !FailedToFindProperty; PropertyIndex++)
877 {
878 FailedToFindProperty = true;
879 const FPropertyInfo& PropInfo = Path->GetPropertyInfo(PropertyIndex);
880
881 TArray< TSharedRef< FPropertyNode > > ChildrenStack;
882 ChildrenStack.Push(PropertyNode.ToSharedRef());
883 while (ChildrenStack.Num() > 0)
884 {
885 const TSharedRef< FPropertyNode > CurrentNode = ChildrenStack.Pop();
886
887 for (int32 ChildIndex = 0; ChildIndex < CurrentNode->GetNumChildNodes(); ++ChildIndex)
888 {
889 const TSharedPtr< FPropertyNode > ChildNode = CurrentNode->GetChildNode(ChildIndex);
890
891 if (ChildNode->AsItemPropertyNode() == nullptr)
892 {
893 ChildrenStack.Add(ChildNode.ToSharedRef());
894 }
895 else if (ChildNode.IsValid() &&
896 ChildNode->GetProperty() == PropInfo.Property.Get() &&
897 ChildNode->GetArrayIndex() == PropInfo.ArrayIndex)
898 {
899 PropertyNode = ChildNode;
900 FailedToFindProperty = false;
901 break;
902 }
903 }
904 }
905 }
906
907 if (FailedToFindProperty)
908 {
909 PropertyNode = nullptr;
910 }
911
912 return PropertyNode;
913 }

916 static TArray< FPropertyInfo > GetPossibleExtensionsForPath(const TSharedPtr< FPropertyPath > Path, const TSharedRef< FPropertyNode >& StartingNode)
917 {
918 TArray< FPropertyInfo > PossibleExtensions;
919 TSharedPtr< FPropertyNode > PropertyNode = FindPropertyNodeByPath(Path, StartingNode);
920
921 if (!PropertyNode.IsValid())
922 {
923 return PossibleExtensions;
924 }
925
926 for (int32 ChildIndex = 0; ChildIndex < PropertyNode->GetNumChildNodes(); ++ChildIndex)
927 {
928 TSharedPtr< FPropertyNode > CurrentNode = PropertyNode->GetChildNode(ChildIndex);
929
930 if (CurrentNode.IsValid() && CurrentNode->AsItemPropertyNode() != nullptr)
931 {
932 FPropertyInfo NewPropInfo;
933 NewPropInfo.Property = CurrentNode->GetProperty();
934 NewPropInfo.ArrayIndex = CurrentNode->GetArrayIndex();
935
936 bool AlreadyExists = false;
937 for (auto ExtensionIter = PossibleExtensions.CreateConstIterator(); ExtensionIter; ++ExtensionIter)
938 {
939 if (*ExtensionIter == NewPropInfo)
940 {
941 AlreadyExists = true;
942 break;
943 }
944 }
945
946 if (!AlreadyExists)
947 {
948 PossibleExtensions.Add(NewPropInfo);
949 }
950 }
951 }
952
953 return PossibleExtensions;
954 }
