22void FObjectThumbnail::SetImage(const FImageView & Image)
23{
24 // Image must be converted to BGRA8 to store in Thumbnail
25 // if Image is already BGRA8-SRGB then this is just a memcpy
26 // which is what we need anyway to copy the bytes into a new array
27 // so there is no inefficiency in just always using the FImage copy here

29 Image.CopyTo(ImageBGRA8,ERawImageFormat::BGRA8,EGammaSpace::sRGB);
30 SetImage( MoveTemp(ImageBGRA8) );
