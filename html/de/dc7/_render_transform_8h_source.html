127 inline void To3x4MatrixTranspose(float* Result) const
128 {
129 float* RESTRICT Dest = Result;
130
131 #if PLATFORM_ENABLE_VECTORINTRINSICS
132 VectorRegister4Float InRow0 = VectorLoad(&TransformRows[0].X); // Intentionally loads W from the next row (discarded)
133 VectorRegister4Float InRow1 = VectorLoad(&TransformRows[1].X); // Intentionally loads W from the next row (discarded)
134 VectorRegister4Float InRow2 = VectorLoad(&TransformRows[2].X); // Intentionally loads W from the next row (discarded)
135 VectorRegister4Float InRow3 = MakeVectorRegisterFloat(Origin.X, Origin.Y, Origin.Z, 0.0f); // Do not read past Origin
136
137 VectorRegister4Float Temp0 = VectorCombineLow( InRow0, InRow1);
138 VectorRegister4Float Temp1 = VectorCombineLow( InRow2, InRow3);
139 VectorRegister4Float Temp2 = VectorCombineHigh(InRow0, InRow1);
140 VectorRegister4Float Temp3 = VectorCombineHigh(InRow2, InRow3);
141
142 VectorRegister4Float Row0, Row1, Row2, Row3;
143 VectorDeinterleave(Row0, Row1, Temp0, Temp1);
144 VectorDeinterleave(Row2, Row3, Temp2, Temp3);
145
146 VectorStore(Row0, &Dest[0]);
147 VectorStore(Row1, &Dest[4]);
148 VectorStore(Row2, &Dest[8]);
149 #else
150 Dest[ 0] = TransformRows[0].X; // [0][0]
151 Dest[ 1] = TransformRows[1].X; // [1][0]
152 Dest[ 2] = TransformRows[2].X; // [2][0]
153 Dest[ 3] = Origin.X; // [3][0]

223 inline void Orthogonalize()
224 {
225 FVector3f X = TransformRows[0];
226 FVector3f Y = TransformRows[1];
227 FVector3f Z = TransformRows[2];
228
229 if (X.IsZero() || Y.IsZero())
230 {
231 return;
232 }
233
234 // Modified Gram-Schmidt orthogonalization
235 Y -= (Y | X) / (X | X) * X;
236 Z -= (Z | X) / (X | X) * X;
237 Z -= (Z | Y) / (Y | Y) * Y;
238
239 TransformRows[0] = X;
240 TransformRows[1] = Y;
241 TransformRows[2] = Z;
242 }

417// [Frisvad 2012, "Building an Orthonormal Basis from a 3D Unit Vector Without Normalization"]
418inline void GetHemiOrthoBasis( FVector3f& BasisX, FVector3f& BasisY, const FVector3f& BasisZ )
419{
420 float A = 1.0f / ( 1.0f + BasisZ.Z );
421 float B = -BasisZ.X * BasisZ.Y * A;
422 BasisX = FVector3f( 1.0f - BasisZ.X * BasisZ.X * A, B, -BasisZ.X );
423 BasisY = FVector3f( B, 1.0f - BasisZ.Y * BasisZ.Y * A, -BasisZ.Y );
424}

426inline FVector2f UnitVectorToHemiOctahedron( const FVector3f& N )
427{
428 float Sum = FMath::Abs( N.X ) + FMath::Abs( N.Y ) + FMath::Abs( N.Z );
429 return FVector2f( N.X + N.Y, N.X - N.Y ) / Sum;
430}

432inline FVector3f HemiOctahedronToUnitVector( const FVector2f& Oct )
433{
434 FVector3f N;
435 N.X = Oct.X + Oct.Y;
436 N.Y = Oct.X - Oct.Y;
437 N.Z = 2.0f - FMath::Abs( N.X ) - FMath::Abs( N.Y );
438 return N.GetUnsafeNormal();
439}
