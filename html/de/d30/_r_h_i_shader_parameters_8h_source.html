171 template <typename ElementType, typename... ArgsType>
172 void Emplace(FMemStackBase& MemStack, TArrayView<ElementType>& InOutArray, ArgsType&& ...Args)
173 {
174 static_assert(sizeof(ElementType) % alignof(ElementType) == 0, "Element size must be a multiple of its alignment");
175
176 const size_t ElementSize = sizeof(ElementType);
177 const size_t Alignment = alignof(ElementType);
178 const int32 NumElements = InOutArray.Num() + 1;
179 ElementType* Elements = InOutArray.GetData();
180
181 if (InOutArray.IsEmpty())
182 {
183 Elements = new (MemStack.Alloc(ElementSize, Alignment)) ElementType(Forward<ArgsType&>(Args)...);
184 }
185 else
186 {
187 // Sanity check that the top of the stack contains the last element that was allocated.
188 check(MemStack.GetTop() == (uint8*)(InOutArray.GetData() + InOutArray.Num()));
189
190 // Try to extend the size of the current array without resizing.
191 if (MemStack.CanFitInPage(ElementSize, 1))
192 {
193 new (MemStack.Alloc(ElementSize, 1)) ElementType(Forward<ArgsType&>(Args)...);
194 }
195 // Reached the end of the page. Reallocate the entire array into a new page.
196 else
197 {
198 Elements = reinterpret_cast<ElementType*>(MemStack.Alloc(NumElements * ElementSize, Alignment));
199 ElementType* LastElement = Elements;
200 for (int32 Index = 0; Index < InOutArray.Num(); ++Index, ++LastElement)
201 {
202 new (LastElement) ElementType(MoveTemp(InOutArray[Index]));
203 }
204 new (LastElement) ElementType(Forward<ArgsType&>(Args)...);
205 }
206 }
207
208 InOutArray = TArrayView<ElementType>(Elements, NumElements);
209 }
210
211 void AppendParametersData(TArrayView<uint8>& InOutArray, uint32 NumBytes, const uint8* Bytes)
212 {
213 constexpr size_t Alignment = 1;
214 const int32 NumArrayBytes = InOutArray.Num() + NumBytes;
215 uint8* ArrayBytes = InOutArray.GetData();
216
217 if (InOutArray.IsEmpty())
218 {
219 ArrayBytes = (uint8*)ParametersData.Alloc(NumBytes, Alignment);
220 FMemory::Memcpy(ArrayBytes, Bytes, NumBytes);
221 }
222 else
223 {
224 // Sanity check that the top of the stack contains the last element that was allocated.
225 check(ParametersData.GetTop() == InOutArray.GetData() + InOutArray.Num());
226
227 // Try to extend the size of the current array without resizing.
228 if (ParametersData.CanFitInPage(NumBytes, Alignment))
229 {
230 FMemory::Memcpy(ParametersData.Alloc(NumBytes, Alignment), Bytes, NumBytes);
231 }
232 // Reached the end of the page. Reallocate the entire array into a new page.
233 else
234 {
235 ArrayBytes = (uint8*)ParametersData.Alloc(NumArrayBytes, Alignment);
236 FMemory::Memcpy(ArrayBytes, InOutArray.GetData(), InOutArray.Num());
237 FMemory::Memcpy(ArrayBytes + InOutArray.Num(), Bytes, NumBytes);
238 }
239 }
240
241 InOutArray = TArrayView<uint8>(ArrayBytes, NumArrayBytes);
242 }
