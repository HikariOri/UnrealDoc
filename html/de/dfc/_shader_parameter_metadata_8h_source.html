276 template<typename FHasherType, typename FHashType>
277 void HashLayout(TMemoryHasher<FHasherType, FHashType>& Hasher)
278 {
279 Hasher << Offset;
280 Hasher << reinterpret_cast<uint8&>(BaseType);
281
282 Hasher.Serialize(const_cast<TCHAR*>(Name), FCString::Strlen(Name));
283 Hasher << NumElements;
284
285 const bool bIsRHIResource = IsShaderParameterTypeReadOnlyRHIResource(BaseType);
286 const bool bIsRDGResource = IsRDGResourceReferenceShaderParameterType(BaseType);
287
288 if (BaseType == UBMT_INT32 ||
289 BaseType == UBMT_UINT32 ||
290 BaseType == UBMT_FLOAT32)
291 {
292 Hasher << reinterpret_cast<uint8&>(Precision);
293 Hasher << NumRows;
294 Hasher << NumColumns;
295 }
296 else if (BaseType == UBMT_INCLUDED_STRUCT || BaseType == UBMT_NESTED_STRUCT)
297 {
298 const_cast<FShaderParametersMetadata*>(Struct)->HashLayout(Hasher);
299 }
300 else if (bIsRHIResource || bIsRDGResource)
301 {
302 Hasher.Serialize(const_cast<TCHAR*>(ShaderType), FCString::Strlen(ShaderType));
303 }
304 }
