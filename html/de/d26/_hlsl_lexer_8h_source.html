312 enum class ELiteralType
313 {
314 Unknown = -1,
315
316 Float, // All variations
317 FloatSuffix, // All variations, ends with 'f'
318 Integer, // Just digits
319 IntegerSuffix, // has a 'u|U|l|L' suffix
320 Hex, // has '0x' or '0X' prefix
321 HexSuffix, // Hex + has a 'u|U|l|L' suffix
322 Octal, // 0[0..7]+
323 Bool, // true or false
324 };
325
326 static inline bool IsIntegerType(ELiteralType Type)
327 {
328 switch (Type)
329 {
330 case ELiteralType::Integer:
331 case ELiteralType::IntegerSuffix:
332 case ELiteralType::Hex:
333 case ELiteralType::HexSuffix:
334 case ELiteralType::Octal:
335 return true;
336 default:
337 break;
338 }
339
340 return false;
341 }
