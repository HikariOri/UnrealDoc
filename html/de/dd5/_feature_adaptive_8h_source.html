11// [ Baensch et al. "A Finite Element Method for Surface Diffusion", 2004] Sec 5.4
12// Feature adaptive refinement criterion
13template <class T>
14class TFeatureSensitivityRefinementCheck
15{
16public:
17
18 // @param InFeatureSensitivity sensitivity of feature check, between 0 and 1
19 // @param InFeatureSize the grid size parameter (h_T in the paper)
20 // @param V0, V1, V2 triangle vertices
21 TFeatureSensitivityRefinementCheck(const T InFeatureSensitivity, const T InFeatureSize,
22 const Math::TVector<T>& V0,
23 const Math::TVector<T>& V1,
24 const Math::TVector<T>& V2)
25 : Threshold(FMath::Max(T(1.) / (InFeatureSensitivity*T(0.5) + T(0.5)) - T(0.95), T(0.05)))
26 , TriVertices{ V0, V1, V2 }
27 , TriNormal( (V2-V0) ^ (V1-V0) )
28 , LSqN(TriNormal.SquaredLength())
29 , IncircleRadius(CalculateInRadius(V0, V1, V2))
30 , FeatureSize(InFeatureSize)
31 {
32 }
33
34 [[nodiscard]] FORCEINLINE bool NeedsCheck() const
35 {
36 return (Threshold * FeatureSize < IncircleRadius * UE_PI); // Early-out (also if IncircleRadius is NaN)
37 }
38
39 // check whether the edge should be refined refined
40 // @param LocalEdge local edge index with respect to the triangle vertices ordering
41 // @param FarVertex vertex on the other side of that edge
42 //
43 [[nodiscard]] FORCEINLINE bool CheckCriterion(const int32 LocalEdge, const UE::Math::TVector<T>& FarVertex ) const
44 {
45 const Math::TVector<T> NT = (TriVertices[(LocalEdge + 1) % 3] - TriVertices[LocalEdge]) ^ (FarVertex - TriVertices[LocalEdge]);
46 const T LSqNT = NT.SquaredLength();
47
48 return ((TriNormal | NT) < FMath::Sqrt(LSqN * LSqNT) * FMath::Cos(Threshold * FeatureSize / IncircleRadius));
49 }
50
51private:
52
53 static T CalculateInRadius( const UE::Math::TVector<T>& V0, const UE::Math::TVector<T>& V1, const UE::Math::TVector<T>& V2 )
54 {
55 const T EdgeLengths_D[] = { (V1 - V0).Size(), (V2 - V1).Size(), (V0 - V2).Size() };
56 const T SemiPerimeter = T(0.5) * (EdgeLengths_D[0] + EdgeLengths_D[1] + EdgeLengths_D[2]);
57 const T TriArea = FMath::Sqrt(FMath::Max(T(0.), SemiPerimeter * (SemiPerimeter - EdgeLengths_D[0]) * (SemiPerimeter - EdgeLengths_D[1]) * (SemiPerimeter - EdgeLengths_D[2]))); // Heron
58 return TriArea / SemiPerimeter;
59 }
60
61 const T Threshold;
62 const Math::TVector<T> TriVertices[3];
63 const Math::TVector<T> TriNormal;
64 const T LSqN; // SquaredLength of Normal vector
65 const T IncircleRadius;
66 const T FeatureSize;
67};
