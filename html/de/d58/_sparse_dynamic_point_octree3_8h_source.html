370 inline FVector3i PointToIndex(uint32 Level, const FVector3d& Position) const
371 {
372 double CellWidth = GetCellWidth(Level);
373
374 // Converts to an int32 in such a way that values outside of the representable
375 // range ends up at the min and max values of the representable range.
376 auto SafeDoubleToInt32Floor = [](double Value)
377 {
378 return static_cast<int32>(FMath::Clamp(
379 FMathd::Floor(Value),
380 static_cast<double>(TNumericLimits<int32>::Lowest()),
381 static_cast<double>(TNumericLimits<int32>::Max())));
382 };
383
384 int32 i = SafeDoubleToInt32Floor(Position.X / CellWidth);
385 int32 j = SafeDoubleToInt32Floor(Position.Y / CellWidth);
386 int32 k = SafeDoubleToInt32Floor(Position.Z / CellWidth);
387 return FVector3i(i, j, k);
388 }

391 {
392 FVector3d Center = GetCellCenter(Level, Index);
393 int ChildIndex =
394 ((Position.X < Center.X) ? 0 : 1) +
395 ((Position.Y < Center.Y) ? 0 : 2) +
396 ((Position.Z < Center.Z) ? 0 : 4);
397 return ChildIndex;
398 }

483void FSparseDynamicPointOctree3::InsertPoint(int32 PointID, const FVector3d& Position)
484{
485 if (ContainsPoint(PointID))
486 {
487 checkSlow(false);
488 return;
489 }
490
491 FSparsePointOctreeCell CurrentCell = FindCurrentContainingCell(Position);
492 checkSlow(CurrentCell.Level != FSparsePointOctreeCell::InvalidLevel);
493
494 // if we found a containing root cell but it doesn't exist, create it and insert
495 if (CurrentCell.Level == 0 && CurrentCell.IsExistingCell() == false)
496 {
497 Insert_NewRoot(PointID, Position, CurrentCell);
498 return;
499 }
500
501 int PotentialChildIdx = ToChildCellIndex(CurrentCell, Position);
502 // if current cell does not have this child we might fit there so try it
503 if (CurrentCell.HasChild(PotentialChildIdx) == false)
504 {
505 // todo can we do a fast check based on level and dimensions??
506 FSparsePointOctreeCell NewChildCell = CurrentCell.MakeChildCell(PotentialChildIdx);
507 if (NewChildCell.Level <= MaxTreeDepth && CellContains(NewChildCell, Position))
508 {
509 Insert_NewChildCell(PointID, Position, CurrentCell.CellID, NewChildCell, PotentialChildIdx);
510 return;
511 }
512 }
513
514 // insert into current cell if
515 // 1) child cell exists (in which case we didn't fit or FindCurrentContainingCell would have returned)
516 // 2) we tried to fit in child cell and failed
517 Insert_ToCell(PointID, Position, CurrentCell);
518}

521void FSparseDynamicPointOctree3::Insert_NewChildCell(int32 PointID, const FVector3d& Position,
522 int ParentCellID, FSparsePointOctreeCell NewChildCell, int ChildIdx)
523{
524 FSparsePointOctreeCell& OrigParentCell = Cells[ParentCellID];
525 checkSlow(OrigParentCell.HasChild(ChildIdx) == false);
526
527 NewChildCell.CellID = CellRefCounts.Allocate();
528 Cells.InsertAt(NewChildCell, NewChildCell.CellID);
529
530 PointIDToCellMap.InsertAt(NewChildCell.CellID, PointID, InvalidCellID);
531
532 CellPointLists.AllocateAt(NewChildCell.CellID);
533 CellPointLists.Insert(NewChildCell.CellID, PointID);
534
535 OrigParentCell.SetChild(ChildIdx, NewChildCell);
536
537 checkSlow(CellContains(NewChildCell, Position));
538 checkSlow(PointToIndex(NewChildCell.Level, Position) == NewChildCell.Index);
539}

584void FSparseDynamicPointOctree3::ParallelInsertDensePointSet(int32 MaxPointID, TFunctionRef<FVector3d(int)> GetPositionFunc)
585{
586 // TODO: this current implementation could be adapted to handle non-dense point sets
587
588 // ParallelInsertDensePointSet can currently only be called if the tree is empty
589 if (ensure(Cells.IsEmpty()) == false)
590 {
591 return;
592 }
593
594 // information for an input point
595 struct FPointInfo
596 {
597 int32 PointID; // ID of this point
598 FVector3i RootCell; // index of Root cell this point is in
599 uint32 LeafCellIndex; // index of Leaf Cell this point should go into. This is an index into the RootCell-specific LevelCells array used inside the ParallelFor below.
600 };
601
602 // populate initial list of points
603 TArray<FPointInfo> PointInfos;
604 PointInfos.SetNumUninitialized(MaxPointID);
605 ParallelFor(MaxPointID, [&](int32 k)
606 {
607 PointInfos[k] = FPointInfo{ k, PointToIndex(0, GetPositionFunc(k)), InvalidCellID };
608 });
609
610 // build a linear list of root cells, and sort all points into a list for each root cell
611 TMap<FVector3i, int> RootCellLinearIndices;
612 TArray<FVector3i> RootCellIndexes;
613 TArray< TUniquePtr<TDynamicVector<int>> > RootCellPointIndices;
614 {
615 //TRACE_CPUPROFILER_EVENT_SCOPE(InitializeRootCellLists);
616 int LinearIndexCounter = 0;
617 for (int32 k = 0; k < MaxPointID; ++k)
618 {
619 int UseLinearIndex = 0;
620 const int* FoundLinearIndex = RootCellLinearIndices.Find(PointInfos[k].RootCell);
621 if (FoundLinearIndex == nullptr)
622 {
623 UseLinearIndex = LinearIndexCounter++;
624 RootCellLinearIndices.Add(PointInfos[k].RootCell, UseLinearIndex);
625 RootCellPointIndices.Add( MakeUnique<TDynamicVector<int>>() );
626 RootCellIndexes.Add(PointInfos[k].RootCell);
627 }
628 else
629 {
630 UseLinearIndex = *FoundLinearIndex;
631 }
632
633 // try doing these Adds in a second per-root-cell ParallelFor? (parallel-scan seems wasteful but might still be faster...)
634 RootCellPointIndices[UseLinearIndex]->Add(k);
635 }
636 }
637
638 // create all the Root Cells
639 TArray<FSparsePointOctreeCell*> RootCellPtrs;
640 {
641 //TRACE_CPUPROFILER_EVENT_SCOPE(CreateRootCells);
642 // Note RootCellPtrs holds pointers to elements of Cells; this should be safe
643 // thanks to the block storage of the TDynamicVector used for Cells
644 RootCellPtrs.SetNumUninitialized(RootCellIndexes.Num());
645 for (int32 Index = 0; Index < RootCellIndexes.Num(); ++Index)
646 {
647 FVector3i RootIndex = RootCellIndexes[Index];
648 FSparsePointOctreeCell NewRootCell(0, RootIndex);
649 int32 CellsIndex = CreateNewRootCell(NewRootCell, false);
650 RootCellPtrs[Index] = &Cells[CellsIndex];
651 }
652 RootCellIndexes.Empty(); // discard this index array; we will only use the pointer array below
653 }
654
655 // make sure the PointIDToCellMap is large enough for all the PointIDs, then we never have to insert/resize
656 PointIDToCellMap.SetNum(MaxPointID);
657
658 // FParentChildCell represents a child-cell-of-a-parent that needs to be created. We will make lists of these below.
659 // This strategy is maybe over-complicated, and possibly we can just create all the leaf cells and then figure out
660 // the necessary parents that need to exist above it?
661 struct FParentChildCell
662 {
663 FVector3i ParentCellIndex; // XYZ index of parent cell
664 FVector3i ChildCellIndex; // XYZ index of child cell that needs to exist (in it's layer)
665 int ChildNum; // index of child cell in parent cell's 8-children list
666 int NumPoints; // number of points that will be inserted in this cell (only set for leaf cells)
667 uint32 NewCellID; // ID of cell after it is created. This is set in the code below
668 bool operator==(const FParentChildCell& OtherCell) const { return ChildCellIndex == OtherCell.ChildCellIndex; }
669 };
670
671 // Parallel iteration over the Root cells. For each Root cell, we iterate over its points, and for each point,
672 // descend down to the max depth to figure out which cells need to be created. Then we create those cells, and
673 // then we insert the points into the leaf cells.
674 FCriticalSection SharedDataLock, CellPointListsLock;
675 ParallelFor(RootCellPtrs.Num(), [&](int32 Index)
676 {
677 FSparsePointOctreeCell& RootCell = *RootCellPtrs[Index];
678
679 // array of new child cells at each level
680 TArray<TArray<FParentChildCell>> LevelCells;
681 LevelCells.SetNum(MaxTreeDepth);
682
683 // descend the non-existent octree cells mathematically, to determine which cells need to be instantiated
684 TDynamicVector<int>& PointIndices = *RootCellPointIndices[Index];
685 for ( int32 PointIndex : PointIndices)
686 {
687 int32 PointID = PointInfos[PointIndex].PointID;
688 FVector3d Position = GetPositionFunc(PointID);
689
690 int32 CurLevel = 0;
691 int32 LastInsertIndex = 0;
692 FSparsePointOctreeCell CurrentCell = RootCell;
693 while (CurLevel < MaxTreeDepth)
694 {
695 int ChildIndex = ToChildCellIndex(CurrentCell, Position);
696 FSparsePointOctreeCell NextLevelCell = CurrentCell.MakeChildCell(ChildIndex);
697 LastInsertIndex = LevelCells[CurLevel].AddUnique( FParentChildCell{ CurrentCell.Index, NextLevelCell.Index, ChildIndex, 0, InvalidCellID } );
698 CurLevel++;
699 CurrentCell = NextLevelCell;
700 }
701 // keep track of which leaf cell this point needs to be inserted into, since we computed it in the last iteration of the above loop
702 PointInfos[PointIndex].LeafCellIndex = LastInsertIndex;
703 LevelCells[CurLevel-1][LastInsertIndex].NumPoints++;
704 }
705
706 // sorted CellIDs of all the Leaf Cells we create in the loop below
707 TArray<uint32> LeafCellIDs;
708
709 // instantiate all the cells at each level. This is a bit messy because at level/depth N+1 we
710 // do not know our parent Cells/CellIDs, only their indices, and that our parent must be one
711 // of the cells that were created at level N. So that takes a linear search. Unclear how to
712 // track that mapping, except maybe a TMap? (note that in profiling this is not remotely a bottleneck...)
713 TArray<FSparsePointOctreeCell*> LastParents, NextParents;
714 LastParents.Add(&RootCell);
715 for (int32 Level = 0; Level < MaxTreeDepth; ++Level)
716 {
717 for (FParentChildCell& ParentChild : LevelCells[Level])
718 {
719 FSparsePointOctreeCell** ParentCell = LastParents.FindByPredicate( [&](FSparsePointOctreeCell* Cell) { return Cell->Index == ParentChild.ParentCellIndex; } );
720 checkSlow(ParentCell != nullptr);
721 checkSlow( (*ParentCell)->HasChild(ParentChild.ChildNum) == false );
722 FSparsePointOctreeCell NewChildCell = (*ParentCell)->MakeChildCell( ParentChild.ChildNum );
723 checkSlow( NewChildCell.Index == ParentChild.ChildCellIndex );
724
725 // only this block accesses data structures shared between root cells
726 SharedDataLock.Lock();
727 NewChildCell.CellID = CellRefCounts.Allocate();
728 Cells.InsertAt(NewChildCell, NewChildCell.CellID);
729 if ( Level == MaxTreeDepth-1 )
730 {
731 LeafCellIDs.Add(NewChildCell.CellID);
732 }
733 FSparsePointOctreeCell* NewCellPtr = &Cells[NewChildCell.CellID];
734 SharedDataLock.Unlock();
735
736 (*ParentCell)->SetChild(ParentChild.ChildNum, NewChildCell);
737 ParentChild.NewCellID = NewChildCell.CellID;
738
739 NextParents.Add(NewCellPtr);
740 }
741 Swap(LastParents, NextParents);
742 NextParents.Reset();
743 }
744
745 TArray<FParentChildCell>& LeafCells = LevelCells[MaxTreeDepth-1];
746 int NumLeafCells = LeafCells.Num();
747 TArray<TSparseListSet<int32>::FListHandle> LeafCellLists;
748 LeafCellLists.SetNum(NumLeafCells);
749
750 // initialize the cell-point-lists for each new leaf cell. This has to be thread-safe, but
751 // it will return handles to the lists that can be used in parallel below
752 // (remember we are in a big ParallelFor over root cells here, that are accessing CellPointLists simultaneously)
753 CellPointListsLock.Lock();
754 for (int32 k = 0; k < NumLeafCells; ++k)
755 {
756 LeafCellLists[k] = CellPointLists.AllocateAt(LeafCellIDs[k]);
757 }
758 CellPointListsLock.Unlock();
759
760 // It is faster to batch-insert points into the leaf cell lists, even though we have
761 // to build up new arrays for it. So first we collect per-leaf-cell lists,
762 // then we send those lists to the leaf cells
763 TArray<TArray<int32>> LeafCellPointLists; // todo could convert to a single array with offsets?
764 LeafCellPointLists.SetNum(NumLeafCells);

787void FSparseDynamicPointOctree3::InsertPoint_DynamicExpand(
788 int32 PointID,
789 TFunctionRef<FVector3d(int)> GetPositionFunc)
790{

792 {
793 checkSlow(false);
794 return;
795 }
796
797 FVector3d Position = GetPositionFunc(PointID);
798
799 FSparsePointOctreeCell CurrentCell = FindCurrentContainingCell(Position);
800 checkSlow(CurrentCell.Level != FSparsePointOctreeCell::InvalidLevel);
801
802 // if we found a containing root cell but it doesn't exist, create it and insert
803 if (CurrentCell.Level == 0 && CurrentCell.IsExistingCell() == false)
804 {
805 Insert_NewRoot(PointID, Position, CurrentCell);
806 return;
807 }
808
809 // insert into current cell if there is space, or if we hit max depth
810 int NumPointsInCell = CellPointLists.GetCount(CurrentCell.CellID);
811 if (NumPointsInCell + 1 < MaxPointsPerCell || CurrentCell.Level == MaxTreeDepth)

813 Insert_ToCell(PointID, Position, CurrentCell);
814 return;
815 }
816
817 const FSparsePointOctreeCell* ParentCell = &Cells[CurrentCell.CellID];
818
819 // this function inserts a point into one of the child cells of CurrentCell
820 auto InsertToChildLocalFunc = [this,&GetPositionFunc,ParentCell](int InsertPointID)
821 {
822 FVector3d MovePosition = GetPositionFunc(InsertPointID);
823 int ChildIdx = this->ToChildCellIndex(*ParentCell, MovePosition);
824 if (ParentCell->HasChild(ChildIdx))

826 uint32 ChildCellID = ParentCell->GetChildCellID(ChildIdx);
827 const FSparsePointOctreeCell* ChildCell = &Cells[ChildCellID];
828 this->Insert_ToCell(InsertPointID, MovePosition, *ChildCell);
829 }
830 else
831 {
832 FSparsePointOctreeCell NewChildCell = ParentCell->MakeChildCell(ChildIdx);
833 this->Insert_NewChildCell(InsertPointID, MovePosition, ParentCell->CellID, NewChildCell, ChildIdx);
834 }
835 };
836
837 // otherwise current cell got too big, so move points to child cells
838 CellPointLists.Enumerate(ParentCell->CellID, [&](int MovePointID)
839 {
840 InsertToChildLocalFunc(MovePointID);
841 });
842
843 // we removed all the points from ParentCell
844 CellPointLists.Clear(ParentCell->CellID);
845
846 // now insert ourself into one of these child cells
847 InsertToChildLocalFunc(PointID);
