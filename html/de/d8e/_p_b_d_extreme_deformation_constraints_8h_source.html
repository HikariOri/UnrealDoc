131 template<int32 Valence, typename Predicate UE_REQUIRES(Valence >= 2 && Valence <= 4)>
132 static TArray<TVector<int32, 2>> TrimConstraints(const TArray<TVector<int32, Valence>>& InConstraints, Predicate TrimPredicate)
133 {
134 TSet<TVec2<int32>> TrimmedConstraints;
135 TrimmedConstraints.Reserve(Valence == 2 ? InConstraints.Num() : InConstraints.Num() * Chaos::Utilities::NChooseR(Valence, 2));
136
137 for (const TVector<int32, Valence>& ConstraintV : InConstraints)
138 {
139 for (int32 i = 0; i < Valence - 1; ++i)
140 {
141 for (int32 j = i + 1; j < Valence; ++j)
142 {
143 const int32 IndexI = ConstraintV[i];
144 const int32 IndexJ = ConstraintV[j];
145
146 if (!TrimPredicate(IndexI, IndexJ))
147 {
148 TrimmedConstraints.Add(IndexI <= IndexJ ? TVec2<int32>(IndexI, IndexJ) : TVec2<int32>(IndexJ, IndexI));
149 }
150 }
151 }
152 }
153 return TrimmedConstraints.Array();
154 }
