93 [[nodiscard]] inline TStaticBitArray(bool, uint32 InBitIndex)
94 {
95 /***********************************************************************
96
97 If this line fails to compile you are attempting to construct a bit
98 array with an out-of bounds bit index. Follow the compiler errors to
99 the initialization point
100
101 ***********************************************************************/
102// static_assert(InBitIndex >= 0 && InBitIndex < NumBits, "Invalid bit value.");
103
104 check((NumBits > 0) ? (InBitIndex<NumBits):1);
105
106 uint32 DestWordIndex = InBitIndex / NumBitsPerWord;
107 WordType Word = (WordType)1 << (InBitIndex & (NumBitsPerWord - 1));
108
109 for(int32 WordIndex = 0; WordIndex < NumWords; ++WordIndex)
110 {
111 Words[WordIndex] = WordIndex == DestWordIndex ? Word : (WordType)0;
112 }
113 }

274 [[nodiscard]] int32 FindFirstClearBit() const
275 {
276 const int32 LocalNumBits = NumBits;
277
278 int32 WordIndex = 0;
279 // Iterate over the array until we see a word with a unset bit.
280 while (WordIndex < NumWords && Words[WordIndex] == WordType(-1))
281 {
282 ++WordIndex;
283 }
284
285 if (WordIndex < NumWords)
286 {
287 // Flip the bits, then we only need to find the first one bit -- easy.
288 const WordType Bits = ~(Words[WordIndex]);
289 UE_ASSUME(Bits != 0);
290
291 const int32 LowestBitIndex = (int32)FMath::CountTrailingZeros64(Bits) + (WordIndex << NumBitsPerWordLog2);
292 if (LowestBitIndex < LocalNumBits)
293 {
294 return LowestBitIndex;
295 }
296 }
297
298 return INDEX_NONE;
299 }

309 int32 WordIndex = 0;

311 while (WordIndex < NumWords && Words[WordIndex] == WordType(0))

322 if (LowestBitIndex < LocalNumBits)
