719// Helper function to calculate Z params for froxel grids
720// Used in shaders with ComputeDepthFromZSlice(...)
721inline FVector CalculateGridZParams(float NearPlane, float FarPlane, float DepthDistributionScale, int32 GridSizeZ)
722{
723 // S = distribution scale
724 // B, O are solved for given the z distances of the first+last slice, and the # of slices.
725 //
726 // slice = log2(z*B + O) * S
727
728 // Don't spend lots of resolution right in front of the near plane
729 double NearOffset = .095 * 100;
730 // Space out the slices so they aren't all clustered at the near plane
731 double S = DepthDistributionScale;
732
733 double N = NearPlane + NearOffset;
734 double F = FarPlane;
735
736 double O = (F - N * FMath::Exp2(GridSizeZ / S)) / (F - N);
737 double B = (1 - O) / N;
738
739 return FVector(B, O, S);
740}

742// LuminanceMax is the amount of light that will cause the sensor to saturate at EV100.
743// See also https://en.wikipedia.org/wiki/Film_speed and https://en.wikipedia.org/wiki/Exposure_value for more info.
744inline float EV100ToLuminance(float LuminanceMax, float EV100)
