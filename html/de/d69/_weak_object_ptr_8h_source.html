414 inline bool SerialNumbersMatch() const
415 {
416#if UE_WITH_REMOTE_OBJECT_HANDLE
417 checkSlow(ObjectIndex >= 0); // otherwise this is a corrupted weak pointer
418 FRemoteObjectId ActualId = GUObjectArray.GetRemoteId(ObjectIndex);
419 return ActualId == ObjectRemoteId;
420#else
421 checkSlow(ObjectSerialNumber > FUObjectArray::START_SERIAL_NUMBER && ObjectIndex >= 0); // otherwise this is a corrupted weak pointer
422 int32 ActualSerialNumber = GUObjectArray.GetSerialNumber(ObjectIndex);
423 checkSlow(!ActualSerialNumber || ActualSerialNumber >= ObjectSerialNumber); // serial numbers should never shrink
424 return ActualSerialNumber == ObjectSerialNumber;
425#endif

442 inline FUObjectItem* Internal_GetObjectItem() const
443 {
444 using namespace UE::Core::Private;
445
446#if UE_WITH_REMOTE_OBJECT_HANDLE
447 if (IsExplicitlyNull())
448 {
449 return nullptr;
450 }
451 if (ObjectIndex < 0)
452 {
453 return nullptr;
454 }
455
456 FUObjectItem* ObjectItem = GUObjectArray.IndexToObject(ObjectIndex);
457 bool bResolved = false;
458 // if an existing object is marked as Remote it needs to be resolved before its ObjectItem is returned
459 if (!ObjectItem || ObjectItem->HasAnyFlags(EInternalObjectFlags::Remote))
460 {
461 ObjectItem = TryResolveRemoteObject();
462 if (!ObjectItem)
463 {
464 return nullptr;
465 }
466 bResolved = true;
467 }
468 // TryResolveRemoteObject already validated the serial number
469 if (!bResolved && !SerialNumbersMatch(ObjectItem))
470 {
471 ObjectItem = TryResolveRemoteObject();
472 if (!ObjectItem)
473 {
474 return nullptr;
475 }
476 }
477 return ObjectItem;
478#else
479 if (ObjectSerialNumber == 0)
480 {
481#if UE_WEAKOBJECTPTR_ZEROINIT_FIX
482 checkSlow(ObjectIndex == InvalidWeakObjectIndex); // otherwise this is a corrupted weak pointer
483#else
484 checkSlow(ObjectIndex == 0 || ObjectIndex == -1); // otherwise this is a corrupted weak pointer
485#endif
486
487 return nullptr;
488 }
489
490 if (ObjectIndex < 0)
491 {
492 return nullptr;
493 }
494 FUObjectItem* const ObjectItem = GUObjectArray.IndexToObject(ObjectIndex);
495 if (!ObjectItem)
496 {
497 return nullptr;
498 }
499 if (!SerialNumbersMatch(ObjectItem))
500 {
501 return nullptr;
502 }
503 return ObjectItem;
504#endif // UE_WITH_REMOTE_OBJECT_HANDLE
505 }

507#if UE_WITH_REMOTE_OBJECT_HANDLE
512 inline bool Internal_IsValid(bool bEvenIfGarbage, bool bThreadsafeTest) const
513 {
514 using namespace UE::Core::Private;
515
516 if (IsExplicitlyNull())
517 {
518 return false;
519 }
520 if (ObjectIndex < 0)
521 {
522 return false;
523 }
524
525 FUObjectItem* ObjectItem = GUObjectArray.IndexToObject(ObjectIndex);
526 if (!ObjectItem || !SerialNumbersMatch(ObjectItem))
527 {
528 if (bEvenIfGarbage)
529 {
530 return CanBeResolved(UE::RemoteObject::Handle::ECanResolveObjectBehavior::Default);
531 }
532 else
533 {
534 return CanBeResolved(UE::RemoteObject::Handle::ECanResolveObjectBehavior::CannotResolveGarbageObjects);
535 }
536 }
537 else if (bThreadsafeTest)
538 {
539 return true;
540 }
541 return GUObjectArray.IsValid(ObjectItem, bEvenIfGarbage);
542 }
543#else
545 inline bool Internal_IsValid(bool bEvenIfGarbage, bool bThreadsafeTest) const
546 {
547 FUObjectItem* const ObjectItem = Internal_GetObjectItem();
548 if (bThreadsafeTest)
549 {
550 return (ObjectItem != nullptr);
551 }
552 else
553 {
554 return (ObjectItem != nullptr) && GUObjectArray.IsValid(ObjectItem, bEvenIfGarbage);
555 }
556 }
557#endif
