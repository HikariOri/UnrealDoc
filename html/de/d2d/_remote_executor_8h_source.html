60struct FRemoteWorkPriority
61{
62 uint64 PackedData = 0;
63
64 // the data is packed in a uint64 to allow passing by value in a register
65 // this allows comparison of priority to be a single uint64 compare-less-than
66 //
67 // the WorkDepth is stored subtracted from 0xFF so that less-than means higher priority
68 // 0xFF is the root depth
69 // 0xFE is one deeper
70 //
71 using FPriorityPacking = TIntegerSequence<int32,
72 1, // 0 if this priority represents (higher priority) transactional migration work, 1 otherwise
73 REMOTE_OBJECT_SERVER_ID_BIT_SIZE, // FRemoteServerId RootServerId
74 8, // uint8 WorkDepth
75 32>; // unique id bits to ensure this priority is unique
76
77 ERemoteWorkType GetWorkType() const { return !(UnpackPriorityElement(PackedData, 0)) ? ERemoteWorkType::Migration : ERemoteWorkType::Normal; }
78 FRemoteServerId GetRootServerId() const { return FRemoteServerId::FromIdNumber((uint32)UnpackPriorityElement(PackedData, 1)); }
79 uint32 GetWorkDepth() const { return (uint32)UnpackPriorityElement(PackedData, 2); }
80 uint32 GetUniqueId() const { return (uint32)UnpackPriorityElement(PackedData, 3); }
81 bool IsValid() const { return GetRootServerId().IsValid(); }
82
83 FRemoteWorkPriority()
84 {
85 PackedData = PackPriority(ERemoteWorkType::Normal, 0, 0, 0);
86 }
87
88 COREUOBJECT_API static FRemoteWorkPriority CreateRootWorkPriority(FRemoteServerId ServerId, ERemoteWorkType WorkType = ERemoteWorkType::Normal);
89
90 // creates a copy of this priority with the WorkDepth decreased by 1
91 COREUOBJECT_API FRemoteWorkPriority CreateDependentWorkPriority() const;
92
93 COREUOBJECT_API FString ToString() const;
94
95private:
96 template<int32... Bits, typename... ArgTypes>
97 static uint64 PackPriorityImpl(TIntegerSequence<int32, Bits...>, ArgTypes... InValues)
98 {
99 static_assert(sizeof...(Bits) == sizeof...(ArgTypes));
100 static_assert((Bits + ...) <= 64, "sum of the bit sizes must be <= 64");
101
102 int32 BitWidths[] = { Bits... };
103 uint64 Values[] = { static_cast<uint64>(InValues)... };
104
105 uint64 Result = 0;
106 int32 Offset = 64; // pack the values in decreasing significance
107
108 for(int32 Index = 0; Index < sizeof...(Bits); Index++)
109 {
110 Offset -= BitWidths[Index];
111 uint64 Mask = (1ULL << BitWidths[Index]) - 1;
112 Result |= (Values[Index] & Mask) << Offset;
113 }
114
115 return Result;
116 }
117
118 template<int32... Bits>
119 static uint64 UnpackPriorityElementImpl(TIntegerSequence<int32, Bits...>, uint64 PackedPriority, int32 OutIndex)
120 {
121 static_assert((Bits + ...) <= 64, "sum of the bit sizes must be <= 64");
122
123 int32 BitWidths[] = { Bits... };
124
125 int32 Offset = 64;
126 uint64 Result = 0;
127 for(int32 Index = 0; Index < sizeof...(Bits); Index++)
128 {
129 Offset -= BitWidths[Index];
130
131 uint64 Mask = (1ULL << BitWidths[Index]) - 1;
132
133 if (Index == OutIndex)
134 {
135 Result = (PackedPriority >> Offset) & Mask;
136 break;
137 }
138 }
139
140 return Result;
141 }
142
143 template<typename... ArgTypes>
144 static uint64 PackPriority(ArgTypes... Values)
145 {
146 return PackPriorityImpl(FPriorityPacking{}, Values...);
147 }
148
149 static uint64 UnpackPriorityElement(uint64 PackedPriority, int32 OutIndex)
150 {
151 return UnpackPriorityElementImpl(FPriorityPacking{}, PackedPriority, OutIndex);
152 }

227 RequestType* ActiveRequest = nullptr;

229 RequestType* GetRequestByIndex(int32 RequestIndex) const
230 {
231 const TTuple<FRemoteTransactionId, void*> RequestData = Requests[RequestIndex];
232 return static_cast<RequestType*>(RequestData.Get<1>());
233 }
234
235 RequestType* TryGetRequest(FRemoteTransactionId RequestId) const
236 {
237 for (int32 RequestIndex = 0; RequestIndex < Requests.Num(); RequestIndex++)
238 {
239 const TTuple<FRemoteTransactionId, void*>& RequestData = Requests[RequestIndex];
240 if (RequestData.Get<0>() == RequestId)
241 return static_cast<RequestType*>(RequestData.Get<1>());
242 }
243
244 return nullptr;
245 }
246
247 int32 TryGetRequestIndex(FRemoteTransactionId RequestId) const
248 {
249 int32 Result = INDEX_NONE;
250
251 for (int32 RequestIndex = 0; RequestIndex < Requests.Num(); RequestIndex++)
252 {
253 const TTuple<FRemoteTransactionId, void*>& RequestData = Requests[RequestIndex];
254 if (RequestData.Get<0>() == RequestId)
255 {
256 Result = RequestIndex;
257 break;
258 }
259 }
260
261 return Result;
262 }
263
264 void* CreateRequest(FRemoteTransactionId RequestId, FRemoteWorkPriority Priority) final
265 {
266 RequestType* Request = new RequestType{};
267 Request->RequestId = RequestId;
268 Request->Priority = Priority;
269 Requests.Add(TTuple<FRemoteTransactionId, void*>(RequestId, Request));
270 return Request;
271 }
272
273 void DestroyRequest(FRemoteTransactionId RequestId) final
274 {
275 int32 RequestIndex = TryGetRequestIndex(RequestId);
276 check(RequestIndex != INDEX_NONE);
277 RequestType* Request = GetRequestByIndex(RequestIndex);
278 Requests.RemoveAt(RequestIndex, EAllowShrinking::No);
279 delete Request;
280 }
281
282 void SetActiveRequest(FRemoteTransactionId RequestId) final
283 {
284 ActiveRequest = TryGetRequest(RequestId);
285 check(ActiveRequest);
286 }

288 void ClearActiveRequest() final

290 ActiveRequest = nullptr;
