49template <typename GetEntryByIndex>
50inline VMapBase::VMapBase(FAllocationContext Context, uint32 MaxNumEntries, const GetEntryByIndex& GetEntry, VEmergentType* Type)
51 : VHeapValue(Context, Type)
52 , NumElements(0)
53 , Capacity(0)
54{
55 AutoRTFM::UnreachableIfClosed("#jira SOL-8415");
56
57 SetIsDeeplyMutable();
58 Reserve(Context, MaxNumEntries * 2);
59
60 // Constructing a map in Verse has these semantics:
61 // - If the same key appears more than once, it's as if only the last key was provided.
62 // - The order of the map is based on the textual order a map is written in.
63 // - E.g, map{K1=>V1, K2=>V2} has the order (K1, V1) then (K2, V2).
64 // And map{K1=>V1, K2=>V2, K1=>V3} has the order (K2, V2) then (K1, V3).
65 for (uint32 Index = 0; Index < MaxNumEntries; ++Index)
66 {
67 TPair<VValue, VValue> Pair = GetEntry(Index);
68
69 if (Pair.Key.IsUninitialized() || Pair.Value.IsUninitialized())
70 {
71 // If GetEntry returns an uninitialized VValue, immediately stop processing and jettison all data.
72 // The returned map will have no elements or capacity, and should be discarded.
73 NumElements = 0;
74 Capacity = 0;
75 Data.Set(Context, TAux<PairType>());
76 SequenceData.Set(Context, TAux<SequenceType>());
77 break;
78 }
79
80 uint32 KeyHash = GetTypeHash(Pair.Key);
81 TPair<uint32, bool> Res = AddWithoutLocking(Context, KeyHash, Pair.Key, Pair.Value); // We don't need to lock because we can't be visited by the GC until after the next handshake.
82 uint32 Slot = Res.Get<0>();
83 bool SlotOverwritten = Res.Get<1>();
84 if (SlotOverwritten)
85 {
86 SequenceType* SequenceTable = GetSequenceTable();
87 uint32 SeqIdx = 0;
88 while (SeqIdx < NumElements)
89 {
90 if (SequenceTable[SeqIdx] == Slot)
91 {
92 // if we've overwritten a value, we need to change the sequence table so the new slot is at the end.
93 FMemory::Memmove(SequenceTable + SeqIdx, SequenceTable + SeqIdx + 1, sizeof(SequenceTable[0]) * (NumElements - SeqIdx - 1));
94 SequenceTable[NumElements - 1] = Slot;
95 break;
96 }
97 ++SeqIdx;
98 }
99 }
100 }
101}
