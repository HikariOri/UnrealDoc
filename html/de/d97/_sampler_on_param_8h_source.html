59 virtual int32 CheckSamplingError(int32 FirstIndex, int32 EndIndex) override
60 {
61 const FVector& APoint = this->Sampling.GetPointAt(this->StartSamplingSegmentIndex);
62 double ACoordinate = this->Sampling.Coordinates[this->StartSamplingSegmentIndex];
63
64 const FVector& BPoint = this->EndStudySegment.Polyline->GetPointAt(this->EndStudySegment.Index);
65 double BCoordinate = this->EndStudySegment.Polyline->Coordinates[this->EndStudySegment.Index];
66
67 double ABCoordinate = BCoordinate - ACoordinate;
68 FVector ABVector = BPoint - APoint;
69
70 double MaxChord = 0;
71 double MaxParametrizationError = 0;
72
73 for (int32 Index = FirstIndex; Index < EndIndex; ++Index)
74 {
75 const FVector& PointM = this->CandidatePoints.GetPointAt(Index);
76
77 double CurvilinearAbscise;
78 FVector ProjectedPoint = ProjectPointOnSegment<FVector>(PointM, APoint, BPoint, CurvilinearAbscise, /*bRestrictCoodinateToInside*/ true);
79
80 double Chord = FVector::Distance(ProjectedPoint, PointM);
81 double AMCoordinate = this->CandidatePoints.Coordinates[Index] - ACoordinate;
82 FVector MApproximate = APoint + (AMCoordinate / ABCoordinate) * ABVector;
83 double ParametrizationError = FVector::Distance(MApproximate, ProjectedPoint);
84
85 if (Chord > MaxChord)
86 {
87 MaxChord = Chord;
88 }
89
90 if (ParametrizationError > MaxParametrizationError)
91 {
92 MaxParametrizationError = ParametrizationError;
93 }
94 }
95
96 int32 SegmentNumToRespectParametrizationError = (int32)(MaxParametrizationError / DesiredParameterizationError);
97 if (SegmentNumToRespectParametrizationError > 2)
98 {
99 SegmentNumToRespectParametrizationError = (int32)((SegmentNumToRespectParametrizationError - 1) / 2) + 2;
100 }
101
102 int32 SegmentNumToRespectChordError = 0;
103 if (MaxChord < this->DesiredChordError / 4)
104 {
105 // S2 = S1 * sqare(L2 / L1)
106 // if the length is doubled, the chord error is still smaller than the desired chord error.
107 // the end point is not needed
108 return 0;
109 }
110 else if (MaxChord < this->DesiredChordError)
111 {
112 // the end point is needed because the Chord is smaller but nearly equal to the desired chord error.
113 SegmentNumToRespectChordError = 1;
114 }
115 else if (MaxChord < 4 * this->DesiredChordError)
116 {
117 // if the length is divided by two, the chord error is divided by 4 and becomes smaller than the desired chord error.
118 // Add First index and Index
119 SegmentNumToRespectChordError = this->CheckTangentError(APoint, ACoordinate, BPoint, BCoordinate, FirstIndex, EndIndex, this->StartSamplingSegmentIndex);
120 }
121 else
122 {
123 SegmentNumToRespectChordError = this->CountOfNeededPointsToRespectChordError(APoint, BPoint, MaxChord);
124 }
125
126 return FMath::Max(SegmentNumToRespectChordError, SegmentNumToRespectParametrizationError);
127 };

288 virtual void SamplingInitializing() override
289 {
290
291#ifdef DEBUG_CURVE_SAMPLING
292 ++CurveIndex;
293#endif
294 IsOptimalSegments.Empty(100);
295 Sampling.Empty(100);
296
297 CandidatePoints.Empty(100);
298
299 // Initialization of the algorithm with the not derivable coordinates of the curve and at least 5 temporary points
300 int32 ComplementaryPointOffset = 0;
301 {
302 TArray<double> LocalNotDerivableCoordinates;
303 GetNotDerivableCoordinates(LocalNotDerivableCoordinates);
304 LocalNotDerivableCoordinates.Insert(Boundary.Min, 0);
305 LocalNotDerivableCoordinates.Add(Boundary.Max);
306
307 ComplementaryPointOffset = LocalNotDerivableCoordinates.Num() < 5 ? 5 : 1;
308
309 NextCoordinates.Empty(LocalNotDerivableCoordinates.Num() * (ComplementaryPointOffset + 1));
310 NextCoordinates.Add(Boundary.Min);
311
312 for (int32 Index = 0; Index < LocalNotDerivableCoordinates.Num() - 1; ++Index)
313 {
314 if (AddIntermediateCoordinates(LocalNotDerivableCoordinates[Index], LocalNotDerivableCoordinates[Index + 1], ComplementaryPointOffset))
315 {
316 NextCoordinates.Add(LocalNotDerivableCoordinates[Index + 1]);
317 }
318 }
319 }
320
321 CandidatePoints.SwapCoordinates(NextCoordinates);
322 EvaluatesNewCandidatePoints();
323
324 // Not Derivable point initialize the sampling
325 ComplementaryPointOffset++;
326 for (int32 Index = 0, ISampling = 0; Index < CandidatePoints.Size(); Index += ComplementaryPointOffset, ++ISampling)
327 {
328 Sampling.EmplaceAt(ISampling, CandidatePoints, Index);
329 IsOptimalSegments.Add(false);
330 }
331 IsOptimalSegments.Pop();
332
333 CandidatePoints.RemoveComplementaryPoints(ComplementaryPointOffset);
334 }
