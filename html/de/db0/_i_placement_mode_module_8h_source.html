224 void AutoSetNativeAndDisplayName()
225 {
226 UClass* Class = AssetData.GetClass() == UClass::StaticClass() ? Cast<UClass>(AssetData.GetAsset()) : nullptr;
227
228 if (Class)
229 {
230 Class->GetName(NativeName);
231 DisplayName = Class->GetDisplayNameText();
232 }
233 else
234 {
235 NativeName = AssetData.AssetName.ToString();
236 if (AssetData.AssetClassPath == UBlueprintGeneratedClass::StaticClass()->GetClassPathName())
237 {
238 // For BPGC, we do not want to display the "_C" suffix, as the name of the loaded class will not have it.
239 NativeName.RemoveFromEnd(TEXT("_C"));
240 }
241 DisplayName = FText::FromString(NativeName);
242 }
243 }
