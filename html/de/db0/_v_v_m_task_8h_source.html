168 void Park(FAllocationContext Context, TWriteBarrier<VTask>& LastTask)
169 {
170 V_DIE_IF(PrevTask || NextTask);
171 if (LastTask)
172 {
173 PrevTask.SetTransactionally(Context, LastTask.Get());
174 LastTask->NextTask.SetTransactionally(Context, this);
175 }
176 LastTask.SetTransactionally(Context, this);
177 }
178
179 void Unpark(FAllocationContext Context, TWriteBarrier<VTask>& LastTask)
180 {
181 if (LastTask.Get() == this)
182 {
183 V_DIE_IF(NextTask);
184 LastTask.SetTransactionally(Context, PrevTask.Get());
185 }
186 if (PrevTask)
187 {
188 V_DIE_UNLESS(PrevTask->NextTask.Get() == this);
189 PrevTask->NextTask.SetTransactionally(Context, NextTask.Get());
190 }
191 if (NextTask)
192 {
193 V_DIE_UNLESS(NextTask->PrevTask.Get() == this);
194 NextTask->PrevTask.SetTransactionally(Context, PrevTask.Get());
195 }
196 PrevTask.ResetTransactionally();
197 NextTask.ResetTransactionally();
198 }
