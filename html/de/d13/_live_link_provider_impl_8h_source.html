119 // Send message (only to connected addresses)

122 {
123 if (!Message)

125 return;

128 TArray<FMessageAddress> Addresses;
129 GetConnectedAddresses(Addresses);
130 if (Addresses.Num() != 0)

132 MessageEndpoint->Send(Message, Flags, {}, nullptr, Addresses, FTimespan::Zero(), FDateTime::MaxValue());
133 }
