51template<typename DerivedType, typename EnumeratedType>
52struct TPopulatableIteratorMixin
53{
54
65 template<typename ContainerType>
66 void Populate(ContainerType& OutContainer) const
67 {
68 for (const EnumeratedType& Item : static_cast<const DerivedType&>(*this))
69 {
70 OutContainer.Add(Item);
71 }
72 }
73
83 template<template<typename T, typename ...Rest> class ContainerType>
84 ContainerType<EnumeratedType> Populate() const
85 {
86 ContainerType<EnumeratedType> Temp;
87 Populate(Temp);
88 return Temp;
89 }
90
95 void ToArray(TArray<EnumeratedType>& OutArray) const
96 {
97 Populate(OutArray);
98 }
99
104 TArray<EnumeratedType> ToArray() const
105 {
106 return Populate<TArray>();
107 }
108
115 template<typename ValueType, typename ProjectionType>
116 EnumeratedType FindBy(const ValueType& Value, ProjectionType Proj) const
117 {
118 for (const EnumeratedType& Item : static_cast<const DerivedType&>(*this))
119 {
120 if (Invoke(Proj, *Item) == Value)
121 {
122 return Item;
123 }
124 }
125 return nullptr;
126 }

173struct FViewModelListIterator : TPopulatableIteratorMixin<FViewModelListIterator, FViewModelPtr>

177 , Filter(InFilter)
178 {}
179
180 explicit FViewModelListIterator(const TSharedPtr<FViewModel>& InModel, EViewModelListType InFilter = EViewModelListType::Everything)
181 : NextList(InModel ? InModel->FirstChildListHead : nullptr)
182 , Filter(InFilter)
183 {
184 FindNextValidIterator();
185 }
186
187 explicit FViewModelListIterator(const FViewModelListHead* InListHead, EViewModelListType InFilter = EViewModelListType::Everything)
188 : NextList(InListHead)
189 , Filter(InFilter)
190 {
191 FindNextValidIterator();
192 }
193
194 FORCEINLINE friend bool operator!=(const FViewModelListIterator& A, const FViewModelListIterator& B)
195 {
196 return A.ListIterator != B.ListIterator || A.NextList != B.NextList;
197 }
198 FORCEINLINE explicit operator bool() const { return (bool)ListIterator; }
199 FORCEINLINE FViewModel* operator->() const { return ListIterator.operator->(); }
200 FORCEINLINE const FViewModelPtr& operator*() const { return *ListIterator; }
201 FORCEINLINE FViewModelPtr GetCurrentItem() const { return ListIterator.GetCurrentItem(); }
202
203 void operator++()
204 {
205 ++ListIterator;
206 FindNextValidIterator();
207 }
208
209 FORCEINLINE FViewModelListIterator begin() const { return *this; }
210 FORCEINLINE FViewModelListIterator end() const { return FViewModelListIterator{ nullptr }; }
211
212private:
213
214 void FindNextValidIterator()
215 {
216 while (!ListIterator && NextList)
217 {
218 // Skip lists that do not match the filter
219 while (NextList && !EnumHasAnyFlags(NextList->Type, Filter))
220 {
221 NextList = NextList->NextListHead;
222 }
223
224 if (NextList)
225 {
226 ListIterator = FViewModelSubListIterator(NextList);
227 NextList = NextList->NextListHead;
228 }
229 }
230 }
231
232private:
233
234 const FViewModelListHead* NextList;
235 FViewModelSubListIterator ListIterator;
236 EViewModelListType Filter;
237};

498template<typename T, typename IteratorType>
499struct TTypedIteratorBase : public IteratorType, public TPopulatableIteratorMixin<TTypedIteratorBase<T, IteratorType>, TViewModelPtr<T>>
500{
501 template<typename ...ArgTypes>
502 TTypedIteratorBase(ArgTypes&&... InArgs)
503 : IteratorType(Forward<ArgTypes>(InArgs)...)
504 {
505 GetCastResultOrContinue();
506 }
507
508 FORCEINLINE T* operator->() const { check(TypedValue); return TypedValue.operator->(); }
509 FORCEINLINE const TViewModelPtr<T>& operator*() const { check(TypedValue); return TypedValue; }
510
511 void operator++()
512 {
513 IteratorType::operator++();
514 GetCastResultOrContinue();
515 }
516
517 FORCEINLINE TTypedIterator<T, IteratorType> begin() const { return IteratorType::begin(); }
518 FORCEINLINE TTypedIterator<T, IteratorType> end() const { return IteratorType::end(); }
519
520 using TPopulatableIteratorMixin<TTypedIteratorBase<T, IteratorType>, TViewModelPtr<T>>::ToArray;
521 using TPopulatableIteratorMixin<TTypedIteratorBase<T, IteratorType>, TViewModelPtr<T>>::Populate;
522 using TPopulatableIteratorMixin<TTypedIteratorBase<T, IteratorType>, TViewModelPtr<T>>::FindBy;
523
524protected:
525
526 void GetCastResultOrContinue()
527 {
528 while (*this)
529 {
530 TypedValue = IteratorType::GetCurrentItem()->template CastThisShared<T>();
531 if (TypedValue)
532 {
533 break;
534 }
535
536 // operator++ must not be called if the break; clause is hit above
537 IteratorType::operator++();
538 }
539 }
540
541 TViewModelPtr<T> TypedValue;
542};
