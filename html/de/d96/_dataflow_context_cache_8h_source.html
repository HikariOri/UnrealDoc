66 // Resolves to TBaseStructure<T>::Get() when available, otherwise to TVariantStructure<T>::Get()
67 // (covers LWC float/double variants like FVector3f, FQuat4f, FTransform3d, ...).
68 template <typename T>
69 UScriptStruct* GetReflectedScriptStruct()
70 {
71 using FStruct = typename TDecay<T>::Type;
72 if constexpr (THasTBaseStructureGet<FStruct>::Value)
73 {
74 return TBaseStructure<FStruct>::Get();
75 }
76 else if constexpr (TIsReflectedCoreVariant<FStruct>::Value)
77 {
78 return TVariantStructure<FStruct>::Get();
79 }
80 else
81 {
82 static_assert(sizeof(FStruct) == 0, "T is not reflected via TBaseStructure or TVariantStructure");
83 return nullptr;
84 }
85 }

735 {

742 }
743 }

747 }
748 if (Type == EType::CacheElementNull)
749 {

751 }

759 }

763 {
764 if constexpr (TIsUObjectPtrElement<T>::Value)
765 {
766 return static_cast<const TContextCacheElementUObject<T>&>(*this).GetData(Context, PropertyIn, Default);
767 }
768 }
769 if (Type == EType::CacheElementUStructArray)
770 {
771 if constexpr (TIsTArray<T>::Value)
772 {
773 if constexpr (TIsReflectedStruct<typename T::ElementType>::Value)
774 {
775 return static_cast<const FContextCacheElementUStructArray&>(*this).GetData<typename T::ElementType>(Context, PropertyIn, Default);

784 }
785 }
786 check(false); // should never happen
