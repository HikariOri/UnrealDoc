34struct FLocalKismetCallbacks
35{
36 static FText GetObjectName(UObject* Object)
37 {
38 return (Object != NULL) ? FText::FromString( Object->GetName() ) : LOCTEXT("UnknownObjectName", "UNKNOWN");
39 }
40
41 static FText GetGraphDisplayName(const UEdGraph* Graph)
42 {
43 if (Graph)
44 {
45 if (const UEdGraphSchema* Schema = Graph->GetSchema())
46 {
47 FGraphDisplayInfo Info;
48 Schema->GetGraphDisplayInformation(*Graph, /*out*/ Info);
49
50 return Info.DisplayName;
51 }
52 else
53 {
54 // if we don't have a schema, we're dealing with a malformed (or incomplete graph)...
55 // possibly in the midst of some transaction - here we return the object's outer path
56 // so we can at least get some context as to which graph we're referring
57 return FText::FromString(Graph->GetPathName());
58 }
59 }
60
61 return LOCTEXT("UnknownGraphName", "UNKNOWN");
62 }
63};
