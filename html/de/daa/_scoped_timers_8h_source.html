317inline double AtomicDoubleFetchAdd(std::atomic<double>& Value, double Delta,
318 std::memory_order MemoryOrder)
319{
320 double Expected = Value.load(std::memory_order_relaxed);
321 double Desired;
322 std::memory_order ReadModifyWriteMemoryOrder =
323 MemoryOrder == std::memory_order_relaxed ? std::memory_order_relaxed
324 : MemoryOrder == std::memory_order_acq_rel ? std::memory_order_acq_rel
325 : std::memory_order_seq_cst;
326
327 bool bExchangeSucceeded = false;
328 do
329 {
330 Desired = Expected + Delta;
331 bExchangeSucceeded = Value.compare_exchange_weak(Expected, Desired,
332 /* success order */ ReadModifyWriteMemoryOrder,
333 /* failure order */ std::memory_order_relaxed);
334 } while (!bExchangeSucceeded);
335
336 return Expected;
337}
