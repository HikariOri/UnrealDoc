124inline void FCallstack::Init(const FStackFrame* InFirstFrame, uint8 InFrameCount)
125{
126 check((uint64(InFirstFrame) & EntryLenMask) == 0);
127 CallstackLenIndex = (uint64(InFrameCount) << EntryLenShift) | (~EntryLenMask & uint64(InFirstFrame));
