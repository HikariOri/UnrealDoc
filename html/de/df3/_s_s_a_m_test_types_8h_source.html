23// ---------------------------------------------------------------------------
24// USSAMTestStreamableAsset
25//
26// A minimal UStreamableRenderAsset subclass that SSAM treats as a real
27// streamable texture. No GPU resources, no render data — just the shared
28// index pointer that SSAM needs for tracking.
29//
30// Note: UCLASS cannot be inside #if WITH_DEV_AUTOMATION_TESTS (UHT restriction).
31// The class is harmless in non-test builds — it's Transient and never instantiated.
32// ---------------------------------------------------------------------------
33UCLASS(Transient)
34class USSAMTestStreamableAsset : public UStreamableRenderAsset
35{
36 GENERATED_BODY()
37
38public:
39 virtual EStreamableRenderAssetType GetRenderAssetType() const override
40 {
41 return EStreamableRenderAssetType::Texture;
42 }
43
44 // StreamingIndex is protected in the base class. Setting it to a
45 // non-INDEX_NONE value makes IsStreamable() return true, without
46 // calling LinkStreaming() (which would register with the real
47 // streaming manager and cause side effects).
48 void SetStreamingIndexForTest(int32 InIndex)
49 {
50 StreamingIndex = InIndex;
51 }
52
53 // Override BeginDestroy to route asset cleanup through SSAM's UnregisterAsset
54 // directly, bypassing UnlinkStreaming which calls into the real
55 // FRenderAssetStreamingManager (our test assets were never registered there).
56 virtual void BeginDestroy() override;
57
58 virtual int32 CalcCumulativeLODSize(int32 NumLODs) const override
59 {
60 // Return a fake size: 1KB per mip, doubling each level.
61 int32 Size = 0;
62 for (int32 i = 0; i < NumLODs; ++i)
63 {
64 Size += 1024 << i;
65 }
66 return Size;
67 }
68
69 // Set up minimal CachedSRRState so the streaming manager can create a
70 // valid FStreamingRenderAsset for this test asset (needs non-zero mip counts).
71 void SetupFakeStreamingState(int32 NumMips = 10, int32 NumNonStreaming = 1)
72 {
73 CachedSRRState.bSupportsStreaming = true;
74 CachedSRRState.MaxNumLODs = NumMips;
75 CachedSRRState.NumNonStreamingLODs = NumNonStreaming;
76 CachedSRRState.NumNonOptionalLODs = NumMips;
77 CachedSRRState.NumResidentLODs = NumMips;
78 CachedSRRState.NumRequestedLODs = NumMips;
79 CachedSRRState.AssetLODBias = 0;
80 }
81};

83// ---------------------------------------------------------------------------
84// USSAMTestComponent
85//
86// Primitive component that creates an FSSAMTestSceneProxy (defined below,
87// guarded by WITH_DEV_AUTOMATION_TESTS). Configure TestAssets and TestBounds
88// before registering.
89// ---------------------------------------------------------------------------
90UCLASS(Transient)
91class USSAMTestComponent : public UPrimitiveComponent
92{
93 GENERATED_BODY()
94
95public:
96 struct FAssetEntry
97 {
98 UStreamableRenderAsset* Asset = nullptr;
99 float TexelFactor = 1.0f;
100 bool bAffectedByComponentScale = true;
101 };
102
103 TArray<FAssetEntry> TestAssets;
104 FBoxSphereBounds TestBounds = FBoxSphereBounds(FVector::ZeroVector, FVector(100.f), 100.f);
105 bool bUseFallbackProxy = false;
106 bool bSupportScaleFactor = false;
107
108 void AddTestAsset(USSAMTestStreamableAsset* Asset, float TexelFactor = 1.0f, bool bAffectedByComponentScale = true)
109 {
110 TestAssets.Add(FAssetEntry{ Asset, TexelFactor, bAffectedByComponentScale });
111 }
112
113 virtual FPrimitiveSceneProxy* CreateSceneProxy() override;
114
115 virtual FBoxSphereBounds CalcBounds(const FTransform& LocalToWorld) const override
116 {
117 return TestBounds.TransformBy(LocalToWorld);
118 }
119
120 // Mirrors the proxy-side CanApplyStreamableRenderAssetScaleFactor opt-in: when
121 // bSupportScaleFactor is true, the fallback FRegister picks up the transform's
122 // max axis scale through this knob, matching the primary path's behavior.
123 virtual float GetStreamingScale() const override
124 {
125 return bSupportScaleFactor ? GetComponentTransform().GetMaximumAxisScale() : 1.f;
126 }
127
128 // Override the streaming asset gathering used by the FALLBACK path.
129 // When using the fallback proxy, ProcessFallbackRegistrations_GameThread
130 // calls FStreamingTextureLevelContext::GetStreamingRenderAssetInfoWithNULLRemoval
131 // which calls Component->GetStreamingRenderAssetInfo. This override
132 // returns our test assets directly, bypassing the material system.
133 virtual void GetStreamingRenderAssetInfo(FStreamingTextureLevelContext& LevelContext, TArray<FStreamingRenderAssetPrimitiveInfo>& OutStreamingRenderAssets) const override;
134};
