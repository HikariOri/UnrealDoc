30struct FSolFingerprint
31{
32 alignas(uint32_t) uint8_t Bytes[20]{};
33 static const FSolFingerprint Zero;
34
35 void Reset()
36 {
37 memset(Bytes, 0, sizeof(Bytes));
38 }
39
40 friend inline bool operator==(const FSolFingerprint& A, const FSolFingerprint& B)
41 {
42 return memcmp(A.Bytes, B.Bytes, sizeof(A.Bytes)) == 0;
43 }
44
45 friend inline bool operator!=(const FSolFingerprint& A, const FSolFingerprint& B)
46 {
47 return memcmp(A.Bytes, B.Bytes, sizeof(A.Bytes)) != 0;
48 }

50 CUTF8String ToCUTF8String() const
51 {
52 const auto NibbleToHex = [](uint8_t Value) { return char(Value + (Value > 9 ? 'A' - 10 : '0')); };
53
54 CUTF8String Output;
55 const uint64_t CharLength = sizeof(Bytes) * 2;
56 Output.Resize(CharLength + 1);
57
58 const uint8_t* Data = Bytes;
59 char* WriteHead = (char*) Output.begin().CurrentByte();
60 for (const uint8_t* DataEnd = Data + sizeof(Bytes); Data != DataEnd; ++Data)
61 {
62 *WriteHead++ = NibbleToHex(*Data >> 4);
63 *WriteHead++ = NibbleToHex(*Data & 15);
64 }
65 *WriteHead++ = 0;
66 return Output;
67 }
68};
