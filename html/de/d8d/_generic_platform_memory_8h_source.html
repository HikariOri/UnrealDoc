725 static CORE_API void MemswapGreaterThan8( void* Ptr1, void* Ptr2, SIZE_T Size );
726
727public:
728 static inline void Memswap( void* Ptr1, void* Ptr2, SIZE_T Size )
729 {
730 switch (Size)
731 {
732 case 0:
733 break;
734
735 case 1:
736 Valswap(*(uint8*)Ptr1, *(uint8*)Ptr2);
737 break;
738
739 case 2:
740 Valswap(*(uint16*)Ptr1, *(uint16*)Ptr2);
741 break;
742
743 case 3:
744 Valswap(*((uint16*&)Ptr1)++, *((uint16*&)Ptr2)++);
745 Valswap(*(uint8*)Ptr1, *(uint8*)Ptr2);
746 break;
747
748 case 4:
749 Valswap(*(uint32*)Ptr1, *(uint32*)Ptr2);
750 break;
751
752 case 5:
753 Valswap(*((uint32*&)Ptr1)++, *((uint32*&)Ptr2)++);
754 Valswap(*(uint8*)Ptr1, *(uint8*)Ptr2);
755 break;
756
757 case 6:
758 Valswap(*((uint32*&)Ptr1)++, *((uint32*&)Ptr2)++);
759 Valswap(*(uint16*)Ptr1, *(uint16*)Ptr2);
760 break;
761
762 case 7:
763 Valswap(*((uint32*&)Ptr1)++, *((uint32*&)Ptr2)++);
764 Valswap(*((uint16*&)Ptr1)++, *((uint16*&)Ptr2)++);
765 Valswap(*(uint8*)Ptr1, *(uint8*)Ptr2);
766 break;
767
768 case 8:
769 Valswap(*(uint64*)Ptr1, *(uint64*)Ptr2);
770 break;
771
772 case 16:
773 Valswap(((uint64*)Ptr1)[0], ((uint64*)Ptr2)[0]);
774 Valswap(((uint64*)Ptr1)[1], ((uint64*)Ptr2)[1]);
775 break;
776
777 default:
778 MemswapGreaterThan8(Ptr1, Ptr2, Size);
779 break;
780 }
