64 virtual bool SupportsFeature( ETargetPlatformFeatures Feature ) const override
65 {
66 // we currently do not have a build target for WindowsServer
67 if (Feature == ETargetPlatformFeatures::Packaging)
68 {
69 return (TProperties::HasEditorOnlyData() || !TProperties::IsServerOnly());
70 }
71
72 if ( Feature == ETargetPlatformFeatures::ShouldSplitPaksIntoSmallerSizes )
73 {
74 return TProperties::IsClientOnly();
75 }
76
77 if (Feature == ETargetPlatformFeatures::CanCookPackages)
78 {
79 return TProperties::HasEditorOnlyData();
80 }
81
82 if (Feature == ETargetPlatformFeatures::MobileRendering)
83 {
84 static bool bCachedSupportsMobileRendering = false;
85#if WITH_ENGINE
86 static bool bHasCachedValue = false;
87 if (!bHasCachedValue)

139 TArray<FName> GetAllTargetedShaderFormatsInternal(TConstArrayView<const TCHAR*> RelevantSettings) const
140 {
141 // no shaders needed for dedicated server target
142 if (TProperties::IsServerOnly())
143 {
144 return TArray<FName>();
145 }
146
147 FConfigCacheIni* ConfigSystem = this->GetConfigSystem();
148
149 TArray<FString> TargetedShaderFormats;
150
151 for (const TCHAR* Name : RelevantSettings)
152 {
153 TArray<FString> NewTargetedShaderFormats;
154 ConfigSystem->GetArray(WIN_SECTION_TEXT, Name, NewTargetedShaderFormats, GEngineIni);
155
156 for (const FString& NewShaderFormat : NewTargetedShaderFormats)
157 {
158 TargetedShaderFormats.AddUnique(NewShaderFormat);
159 }
160 }
161
162 // Gather the list of Target RHIs and filter out any that may be invalid.
163 TArray<FName> PossibleShaderFormats;
164 GetAllPossibleShaderFormats(PossibleShaderFormats);
165
166 TArray<FName> OutFormats;
167 for (const FString& ShaderFormat : TargetedShaderFormats)
168 {
169 const FName ShaderFormatName(*ShaderFormat, FNAME_Find);
170
171 if (!ShaderFormatName.IsNone() && PossibleShaderFormats.Contains(ShaderFormatName))
172 {
173 OutFormats.Emplace(ShaderFormatName);
174 }
175 }
176
177 return OutFormats;
178 }
