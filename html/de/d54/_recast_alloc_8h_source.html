139template<class T>
140bool rcScopedDelete<T>::resizeGrow(int n)
141{
142 if (n > size)
143 {
144 T* newData = (T*)rcAlloc(sizeof(T) * n, RC_ALLOC_TEMP);
145 if (n && newData) rcMemCpy(newData, ptr, sizeof(T) * size);
146 rcFree(ptr);
147 ptr = newData;
148 size = n;
149 }
150
151 return (ptr != 0);
152}
