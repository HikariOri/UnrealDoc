7syms_spatial_map_from_line_table(SYMS_Arena *arena, SYMS_LineTable *line_table){
8 SYMS_ProfBegin("syms_spatial_map_from_line_table");
9
10 SYMS_U64 count = line_table->sequence_count;
11 SYMS_SpatialMap1DRange *ranges = syms_push_array(arena, SYMS_SpatialMap1DRange, count);
12 SYMS_SpatialMap1DRange *range_ptr = ranges;
13 SYMS_U64 *seq_index_ptr = line_table->sequence_index_array;
14 SYMS_Line *line_array = line_table->line_array;
15 for (SYMS_U64 n = 1; n <= count; n += 1, range_ptr += 1){
16 // get sequence range & inc
17 SYMS_U64 first = *seq_index_ptr;
18 seq_index_ptr += 1;
19 SYMS_U64 opl = *seq_index_ptr;
20
21 // extract sequence range
22 SYMS_U64 last = opl - 1;
23 SYMS_U64 first_voff = line_array[first].voff;
24 SYMS_U64 opl_voff = line_array[last].voff;
25
26 // fill range ptr
27 range_ptr->range.min = first_voff;
28 range_ptr->range.max = opl_voff;
29 range_ptr->val = n;
30 }
31
32 //- sort
33 if (!syms_spatial_map_1d_array_check_sorted(ranges, count)){
34 syms_spatial_map_1d_array_sort(ranges, count);
35 }
36
37 //- assemble map
38 SYMS_SpatialMap1D result = {ranges, count};
39
40 SYMS_ProfEnd();
41 return(result);
42}
