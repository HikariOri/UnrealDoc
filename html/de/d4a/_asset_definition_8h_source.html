79 template<typename ExpectedObjectType>
80 TArray<ExpectedObjectType*> LoadObjects(const TSet<FName>& LoadTags = {}, TArray<FAssetData>* OutAssetsThatFailedToLoad = nullptr) const
81 {
82 FScopedSlowTask SlowTask((float)Assets.Num());
83
84 TArray<ExpectedObjectType*> LoadedObjects;
85 LoadedObjects.Reserve(Assets.Num());
86
87 for (const FAssetData& Asset : Assets)
88 {
89 SlowTask.EnterProgressFrame(1, FText::FromString(Asset.GetObjectPathString()));
90
91 if (Asset.IsInstanceOf(ExpectedObjectType::StaticClass()))
92 {
93 if (ExpectedObjectType* ExpectedType = Cast<ExpectedObjectType>(Asset.GetAsset(LoadTags)))
94 {
95 LoadedObjects.Add(ExpectedType);
96 continue;
97 }
98 }
99
100 // If we get here than we failed to load the asset for some reason or another.
101 if (OutAssetsThatFailedToLoad)
102 {
103 OutAssetsThatFailedToLoad->Add(Asset);
104 }
105 }
106
107 return LoadedObjects;
108 }
109
110 template<typename ExpectedObjectType>
111 ExpectedObjectType* LoadFirstValid(const TSet<FName>& LoadTags = {}) const
112 {
113 for (const FAssetData& Asset : Assets)
114 {
115 if (Asset.IsInstanceOf(ExpectedObjectType::StaticClass()))
116 {
117 if (ExpectedObjectType* ExpectedType = Cast<ExpectedObjectType>(Asset.GetAsset(LoadTags)))
118 {
119 return ExpectedType;
120 }
121 }
122 }
123
124 return nullptr;
125 }

628 // DEVELOPER NOTE:
629 // Originally this class was based on the IAssetTypeActions implementation. Several of the functions on there
630 // were created organically and added without a larger discussion about if such a thing belonged on those classes.
631 //
632 // For example, IAssetTypeActions::ShouldForceWorldCentric was needed for a single asset, but we didn't instead
633 // implement GetAssetOpenSupport, which merges the needs of ShouldForceWorldCentric, and SupportsOpenedMethod.
634 //
635 // Another example, is IAssetTypeActions::SetSupported and IAssetTypeActions::IsSupported. These were concepts
636 // that could have lived in a map on the registry and never needed to be stored on the actual IAssetTypeActions.
637 //
638 // So, please do not add new functions to this class if it can be helped. The AssetDefinitions are intended to be
639 // a basic low level representation of top level assets for the Content Browser and other editor tools to do
640 // some basic interaction with them, or learn some basic common details about them.
641 //
642 // If you must add a new function to this class, some requests,
643 // 1. Can it be added as a parameter to an existing Argument struct for an existing function? If so, please do that.
644 // 2. Can it be added as part of the return structure of an existing function? If so, please do that.
645 // 3. If you add a new function, please create a struct for the Args. We'll be able to upgrade things easier.
646 // Please continue to use EAssetCommandResult and FAssetSupportResponse, for those kinds of commands.
647
