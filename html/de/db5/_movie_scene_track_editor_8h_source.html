150 template<typename TrackClass>
151 FFindOrCreateRootTrackResult<TrackClass> FindOrCreateRootTrack()
152 {
153 FFindOrCreateRootTrackResult<TrackClass> Result;
154 bool bTrackExisted;
155
156 UMovieScene* MovieScene = GetSequencer()->GetFocusedMovieSceneSequence()->GetMovieScene();
157 Result.Track = MovieScene->template FindTrack<TrackClass>();
158 bTrackExisted = Result.Track != nullptr;
159
160 if (Result.Track == nullptr)
161 {
162 Result.Track = MovieScene->template AddTrack<TrackClass>();
163 }
164
165 Result.bWasCreated = bTrackExisted == false && Result.Track != nullptr;
166 return Result;
167 }
