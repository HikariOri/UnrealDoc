94 enum class ECompilerResult : uint32_t
95 {
96 Compile_NoOp = (0x00),
97
98 Compile_RanSyntaxPass = (1<<0),
99 Compile_RanSemanticPass = (1<<1),
100 Compile_RanLocalizationPass = (1<<2),
101 Compile_RanIrPass = (1<<3),
102 Compile_RanCodeGenPass = (1<<4),
103
104 Compile_SkippedByInjection = (1<<5),
105 Compile_SkippedByEmptyPass = (1<<6),
106
107 Compile_SyntaxError = (1<<7),
108 Compile_SemanticError = (1<<8),
109 Compile_IrError = (1<<9),
110 Compile_LocalizationError = (1<<10),
111 Compile_CodeGenError = (1<<11),
112
113 CompileMask_Failure = (Compile_SyntaxError | Compile_SemanticError | Compile_IrError | Compile_LocalizationError | Compile_CodeGenError),
114 CompileMask_Skipped = (Compile_SkippedByInjection | Compile_SkippedByEmptyPass),
115 CompileMask_Aborted = (CompileMask_Failure | CompileMask_Skipped),
116 };
117 ULANG_ENUM_BIT_FLAGS(ECompilerResult, inline);
118
119 inline bool IsCompileFailure(ECompilerResult E) { return Enum_HasAnyFlags(E, ECompilerResult::CompileMask_Failure); }
120 inline bool IsAbortedCompile(ECompilerResult E) { return Enum_HasAnyFlags(E, ECompilerResult::CompileMask_Aborted); }
121 inline bool IsCompileIncomplete(ECompilerResult E) { return !E || IsAbortedCompile(E); }
122 inline bool IsCompileComplete(ECompilerResult E) { return !IsCompileIncomplete(E); }
