46 virtual int32 TranslateThreadPriority(EThreadPriority Priority)
47 {
48 // these are some default priorities
49 switch (Priority)
50 {
51 // 0 is the lowest, 31 is the highest possible priority for pthread
52 case TPri_Highest: case TPri_TimeCritical: return 30;
53 case TPri_AboveNormal: return 25;
54 case TPri_Normal: return 15;
55 case TPri_BelowNormal: return 5;
56 case TPri_Lowest: return 1;
57 case TPri_SlightlyBelowNormal: return 14;
58 case TPri_Num:
59 default:
60 UE_LOGF(LogHAL, Fatal, "Unknown Priority passed to FRunnableThreadPThread::TranslateThreadPriority()");
61 return 15;
62 }

119 virtual bool SetThreadAttr(pthread_attr_t *ThreadAttr, uint32 InStackSize, EThreadCreateFlags InCreateFlags)
120 {
121 // allow platform to adjust stack size
122 InStackSize = AdjustStackSize(InStackSize);
123
124 if (InStackSize != 0)
125 {
126 if (pthread_attr_init(ThreadAttr) == 0)
127 {
128 // we'll use this the attribute if this succeeds, otherwise, we'll wing it without it.
129 const size_t StackSize = (size_t)InStackSize;
130 if (pthread_attr_setstacksize(ThreadAttr, StackSize) != 0)
131 {
132 UE_LOGF(LogHAL, Log, "Failed to change pthread stack size to %d bytes", (int)InStackSize);
133 }
134 return true;
135 }
136 else
137 {
138 UE_LOGF(LogHAL, Log, "Failed to init thread attr");
139 }
140 }
141
142 return false;

145 bool SpinPthread(pthread_t* HandlePtr, bool* OutThreadCreated, PthreadEntryPoint Proc, uint32 InStackSize, EThreadCreateFlags InCreateFlags, void *Arg)
146 {
147 *OutThreadCreated = false;
148 pthread_attr_t ThreadAttr;
149 pthread_attr_t *ThreadAttrPtr = SetThreadAttr(&ThreadAttr, InStackSize, InCreateFlags) ? &ThreadAttr : nullptr;
150
151 const int ThreadErrno = CreateThreadWithName(HandlePtr, ThreadAttrPtr, Proc, Arg, TCHAR_TO_ANSI(*ThreadName));
152 *OutThreadCreated = (ThreadErrno == 0);
153 if (ThreadAttrPtr)
154 {
155 pthread_attr_destroy(ThreadAttrPtr);
156 }
157
158 // Move the thread to the specified processors if requested
159 if (!*OutThreadCreated)
160 {
161 UE_LOGF(LogHAL, Log, "Failed to create thread! (err=%d, %ls)", ThreadErrno, UTF8_TO_TCHAR(strerror(ThreadErrno)));
162 }
163
164 return *OutThreadCreated;
165 }

171 static void *STDCALL _ThreadProc(void *pThis)
172 {
173 check(pThis);
174
175 FRunnableThreadPThread* ThisThread = (FRunnableThreadPThread*)pThis;
176
177 // cache the thread ID for this thread (defined by the platform)
178 ThisThread->ThreadID = FPlatformTLS::GetCurrentThreadId();
179
180 if (ThisThread->ThreadID == ~uint32(0)) [[unlikely]]
181 {
182 LowLevelFatalError(TEXT("Thread ID is -1. Reconsider implementation of FThread::InvalidThreadId"));
183 }
184
185 FThreadManager::Get().AddThread(ThisThread->ThreadID, ThisThread);
186
187 // set the affinity. This function sets affinity on the current thread, so don't call in the Create function which will trash the main thread affinity.
188 FPlatformProcess::SetThreadAffinityMask(ThisThread->ThreadAffinityMask);
189
190 // run the thread!
191 ThisThread->PreRun();
192 ThisThread->Run();
193 ThisThread->PostRun();
194
195 pthread_exit(NULL);
196 return NULL;
197 }

271 virtual bool Kill(bool bShouldWait) override
272 {
273 check(Thread && "Did you forget to call Create()?");
274 bool bDidExitOK = true;
275 // Let the runnable have a chance to stop without brute force killing
276 if (Runnable)
277 {
278 Runnable->Stop();
279 }
280 // If waiting was specified, wait the amount of time. If that fails,
281 // brute force kill that thread. Very bad as that might leak.
282 if (bShouldWait && bThreadStartedAndNotCleanedUp)
283 {
284 int JoinResult = pthread_join(Thread, nullptr);
285
286 UE_CLOGF(JoinResult != 0, LogHAL, Warning, "Failed to join thread %u (%ls). (err=%d, %ls)", ThreadID, *ThreadName, JoinResult, UTF8_TO_TCHAR(strerror(JoinResult)));
287
288 bDidExitOK = (JoinResult == 0);
289 bThreadStartedAndNotCleanedUp = false;
290 }
291
292 // It's not really safe to kill a pthread. So don't do it.
293 return bDidExitOK;
294 }
295
296 virtual void WaitForCompletion() override
297 {
298 // Block until this thread exits
299 if (bThreadStartedAndNotCleanedUp)
300 {
301 int JoinResult = pthread_join(Thread, nullptr);
302
303 UE_CLOGF(JoinResult != 0, LogHAL, Warning, "Failed to join thread %u (%ls). (err=%d, %ls)", ThreadID, *ThreadName, JoinResult, UTF8_TO_TCHAR(strerror(JoinResult)));
304
305 bThreadStartedAndNotCleanedUp = false;
306 }

311 virtual bool CreateInternal(FRunnable* InRunnable, const TCHAR* InThreadName,
312 uint32 InStackSize = 0,
313 EThreadPriority InThreadPri = TPri_Normal, uint64 InThreadAffinityMask = 0,
314 EThreadCreateFlags InCreateFlags = EThreadCreateFlags::None) override
315 {
316 check(InRunnable);
317 Runnable = InRunnable;
318
319 // Create a sync event to guarantee the Init() function is called first
320 ThreadInitSyncEvent = FPlatformProcess::GetSynchEventFromPool(true);
321 // A name for the thread in for debug purposes. _ThreadProc will set it.
322 ThreadName = InThreadName ? InThreadName : TEXT("Unnamed UE4");
323 ThreadPriority = InThreadPri; // Make sure the thread priority is correct during init
324 ThreadAffinityMask = InThreadAffinityMask;
325
326 // Create the new thread
327 const bool ThreadCreated = SpinPthread(&Thread, &bThreadStartedAndNotCleanedUp, GetThreadEntryPoint(), InStackSize, InCreateFlags, this);
328 // If it fails, clear all the vars
329 if (ThreadCreated)
330 {
331 // Let the thread start up, then set the name for debug purposes.
332 ThreadInitSyncEvent->Wait((uint32)-1); // infinite wait
333
334 // set the priority
335 ThreadPriority = TPri_Num; // set to Num (invalid) to ensure that the priority will be set below
336 SetThreadPriority(InThreadPri);
337 }
338 else // If it fails, clear all the vars
339 {
340 Runnable = NULL;
341 }
342
343 // Cleanup the sync event
344 FPlatformProcess::ReturnSynchEventToPool( ThreadInitSyncEvent );
345 ThreadInitSyncEvent = nullptr;
346 return bThreadStartedAndNotCleanedUp;
347 }
