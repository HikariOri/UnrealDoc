46const inline TCHAR* GetDataLayerRuntimeStateName(EDataLayerRuntimeState State)
47{
48 switch (State)
49 {
50 case EDataLayerRuntimeState::Unloaded: return TEXT("Unloaded");
51 case EDataLayerRuntimeState::Loaded: return TEXT("Loaded");
52 case EDataLayerRuntimeState::Activated: return TEXT("Activated");
53 default: check(0);
54 }
55 return TEXT("Invalid");
56}
