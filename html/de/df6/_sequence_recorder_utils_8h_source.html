28static AssetType* MakeNewAsset(const FString& BaseAssetPath, const FString& BaseAssetName)
29{
30 const FString Dot(TEXT("."));
31 FString AssetPath = BaseAssetPath;
32 FString AssetName = BaseAssetName;
33
34 AssetPath /= AssetName;
35 AssetPath += Dot + AssetName;
36
37 FAssetRegistryModule& AssetRegistryModule = FModuleManager::LoadModuleChecked<FAssetRegistryModule>(TEXT("AssetRegistry"));
38 FAssetData AssetData = AssetRegistryModule.Get().GetAssetByObjectPath(FSoftObjectPath(AssetPath));
39
40 // if object with same name exists, try a different name until we don't find one
41 int32 ExtensionIndex = 0;
42 while(AssetData.IsValid() && AssetData.GetClass() == AssetType::StaticClass())
43 {
44 AssetName = FString::Printf(TEXT("%s_%d"), *BaseAssetName, ExtensionIndex);
45 AssetPath = (BaseAssetPath / AssetName) + Dot + AssetName;
46 AssetData = AssetRegistryModule.Get().GetAssetByObjectPath(FSoftObjectPath(AssetPath));
47
48 ExtensionIndex++;
49 }
50
51 // Create the new asset in the package we just made
52 AssetPath = (BaseAssetPath / AssetName);
53
54 FString FileName;
55 if(FPackageName::TryConvertLongPackageNameToFilename(AssetPath, FileName))
56 {
57 UObject* Package = CreatePackage( *AssetPath);
58 return NewObject<AssetType>(Package, *AssetName, RF_Public | RF_Standalone);
59 }
60
61 UE_LOGF(LogAnimation, Error, "Couldnt create file for package %ls", *AssetPath);
62
63 return nullptr;
64}

66static UObject* DuplicateAsset(const FString& BaseAssetPath, const FString& BaseAssetName, UObject* ObjectToDuplicate)
67{
68 // Verify the source object
69 if ( !ObjectToDuplicate )
70 {
71 return nullptr;
72 }
73
74 const FString Dot(TEXT("."));
75 FString AssetPath = BaseAssetPath;
76 FString AssetName = BaseAssetName;
77
78 AssetPath /= AssetName;
79 AssetPath += Dot + AssetName;
80
81 FAssetRegistryModule& AssetRegistryModule = FModuleManager::LoadModuleChecked<FAssetRegistryModule>(TEXT("AssetRegistry"));
82 FAssetData AssetData = AssetRegistryModule.Get().GetAssetByObjectPath(FSoftObjectPath(AssetPath));
83
84 // if object with same name exists, try a different name until we don't find one
85 int32 ExtensionIndex = 0;
86 while(AssetData.IsValid() && AssetData.GetClass() == ObjectToDuplicate->GetClass())
87 {
88 AssetName = FString::Printf(TEXT("%s_%d"), *BaseAssetName, ExtensionIndex);
89 AssetPath = (BaseAssetPath / AssetName) + Dot + AssetName;
90 AssetData = AssetRegistryModule.Get().GetAssetByObjectPath(FSoftObjectPath(AssetPath));
91
92 ExtensionIndex++;
93 }
94
95 const FString PackageName = BaseAssetPath + TEXT("/") + AssetName;
96
97 ObjectTools::FPackageGroupName PGN;
98 PGN.PackageName = PackageName;
99 PGN.GroupName = TEXT("");
100 PGN.ObjectName = AssetName;
101
102 TSet<UPackage*> ObjectsUserRefusedToFullyLoad;
103 UObject* NewObject = ObjectTools::DuplicateSingleObject(ObjectToDuplicate, PGN, ObjectsUserRefusedToFullyLoad);
104
105 return NewObject;
106}

144static bool DoesTakeExist(const FString& AssetPath, const FString& ActorName, const FString& SessionName, uint32 TakeNumber)
145{
146 FString TakeName = MakeTakeName(ActorName, SessionName, TakeNumber);
147 FString TakePath = AssetPath / TakeName;
148
149 FAssetRegistryModule& AssetRegistryModule = FModuleManager::LoadModuleChecked<FAssetRegistryModule>(TEXT("AssetRegistry"));
150
151 const FString Dot(TEXT("."));
152 FString FullPath = TakePath + Dot + TakeName;
153 FAssetData AssetData = AssetRegistryModule.Get().GetAssetByObjectPath(FSoftObjectPath(FullPath));
154
155 return AssetData.IsValid();
156}
