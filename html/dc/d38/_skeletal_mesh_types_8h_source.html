94struct FMeshToMeshVertData
95{
96 // Barycentric coords and distance along normal for the position of the final vert
97 FVector4f PositionBaryCoordsAndDist;
98
99 // Barycentric coords and distance along normal for the location of the unit normal endpoint
100 // Actual normal = ResolvedNormalPosition - ResolvedPosition
101 FVector4f NormalBaryCoordsAndDist;
102
103 // Barycentric coords and distance along normal for the location of the unit Tangent endpoint
104 // Actual normal = ResolvedNormalPosition - ResolvedPosition
105 FVector4f TangentBaryCoordsAndDist;
106
107 // Contains the 3 indices for verts in the source mesh forming a triangle, the last element
108 // is a flag to decide how the skinning works, 0xffff uses no simulation, and just normal
109 // skinning, anything else uses the source mesh and the above skin data to get the final position
110 uint16 SourceMeshVertIndices[4];
111
112 // For weighted averaging of multiple triangle influences
113 float Weight = 0.0f;
114
115 // Dummy for alignment
116 uint32 Padding = 0;
117
118 friend ENGINE_API FArchive& operator<<(FArchive& Ar, FMeshToMeshVertData& V);
119
120 bool operator==(const FMeshToMeshVertData& Other) const
121 {
122 return PositionBaryCoordsAndDist == Other.PositionBaryCoordsAndDist &&
123 NormalBaryCoordsAndDist == Other.NormalBaryCoordsAndDist &&
124 TangentBaryCoordsAndDist == Other.TangentBaryCoordsAndDist &&
125 SourceMeshVertIndices[0] == Other.SourceMeshVertIndices[0] &&
126 SourceMeshVertIndices[1] == Other.SourceMeshVertIndices[1] &&
127 SourceMeshVertIndices[2] == Other.SourceMeshVertIndices[2] &&
128 SourceMeshVertIndices[3] == Other.SourceMeshVertIndices[3] &&
129 Weight == Other.Weight;
130 }
131};

133class FSkeletalMeshUnifiedBoneIndices
134{
135public:
136 FSkeletalMeshUnifiedBoneIndices(TConstArrayView<uint16> InUnifiedToSkeletonIndex)
137 : UnifiedToSkeletonIndex(InUnifiedToSkeletonIndex)
138 {
139 int32 MaxSkeletonIndex = -1;
140
141 for (int32 SkeletonIndex : UnifiedToSkeletonIndex)
142 {
143 MaxSkeletonIndex = FMath::Max(MaxSkeletonIndex, SkeletonIndex);
144 }
145
146 SkeletonToUnifiedBoneIndex.Init(INDEX_NONE, MaxSkeletonIndex + 1);
147
148 for (int32 UnifiedBoneIndex = 0; UnifiedBoneIndex < UnifiedToSkeletonIndex.Num(); ++UnifiedBoneIndex)
149 {
150 SkeletonToUnifiedBoneIndex[UnifiedToSkeletonIndex[UnifiedBoneIndex]] = (uint16)UnifiedBoneIndex;
151 }
152 }
153
154 bool IsValid() const
155 {
156 return !UnifiedToSkeletonIndex.IsEmpty();
157 }
158
159 int32 GetNumUnifiedBones() const
160 {
161 return UnifiedToSkeletonIndex.Num();
162 }
163
164 uint16 GetSkeletonToUnifiedIndex(int32 SkeletonIndex) const
165 {
166 check(IsValid());
167 return SkeletonToUnifiedBoneIndex.IsValidIndex(SkeletonIndex) ? SkeletonToUnifiedBoneIndex[SkeletonIndex] : INDEX_NONE;
168 }
169
170 int32 GetUnifiedToSkeletonIndex(int32 UnifiedIndex) const
171 {
172 check(IsValid());
173 return UnifiedToSkeletonIndex[UnifiedIndex];
174 }
175
176 TConstArrayView<uint16> GetBoneMap() const
177 {
178 return UnifiedToSkeletonIndex;
179 }
180
181 ENGINE_API void RemapVertex(TConstArrayView<uint16> BoneMap, FSoftSkinVertex& Vertex) const;
182
183private:
184 TConstArrayView<uint16> UnifiedToSkeletonIndex;
185 TArray<uint16> SkeletonToUnifiedBoneIndex;
186};
