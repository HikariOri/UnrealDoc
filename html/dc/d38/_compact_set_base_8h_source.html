142 /* The total amount of memory required by the has set to store N elements with a hash table size of Y */
143 [[nodiscard]] static constexpr SizeType GetMaxElementsForAvailableSpace(size_t TotalBytes, HashCountType HashCount, uint32 MinElementCount, const FCompactSetLayout Layout)
144 {
145 // Given some space in memory and a requested size for hash table, figure out how many elements we can fit in the remaining space
146 const uint32 TypeSize = UE::Core::CompactHashTable::GetTypeSize(MinElementCount);
147 const uint32 TypeShift = UE::Core::CompactHashTable::GetTypeShift(MinElementCount);
148 const size_t AvailableBytes = TotalBytes - sizeof(HashCountType) - ((size_t)HashCount << TypeShift); // Remove hashtable and HashCount data
149 const size_t MaxElements = AvailableBytes / (Layout.Size + TypeSize * KeyStorageMultiplier); // Calculate the max available ignoring the fact that the hash data has to be aligned
150 const size_t RealAvailableBytes = AlignDown(AvailableBytes - (MaxElements << TypeShift) * KeyStorageMultiplier, UE::Core::CompactHashTable::GetMemoryAlignment()); // Remove the max required indexes and align down
151 return (SizeType)FMath::Min<size_t>(MaxElements, RealAvailableBytes / Layout.Size); // Now we can get the true number of elements we could potentially use within the aligned space
152 }

154 [[nodiscard]] SizeType AllocatorCalculateSlackGrow(SizeType NewMaxElements, const FCompactSetLayout& Layout) const
155 {
156 const HashCountType OldHashCount = MaxElements > 0 ? *(HashCountType*)(Elements.GetAllocation() + GetElementsSizeInBytes(MaxElements, Layout)) : 0;
157 const size_t OldSize = GetTotalMemoryRequiredInBytes(MaxElements, OldHashCount, Layout);
158
159 const HashCountType NewHashCount = GetHashCount(NewMaxElements);
160 const size_t NewSize = GetTotalMemoryRequiredInBytes(NewMaxElements, NewHashCount, Layout);
161 size_t NewSlackSize = 0;
162
163 checkf(NewSize > 0 && NewSize <= MAX_int32, TEXT("Invalid size for TSet[%zd]: NewMaxElements[%d] ElementSize[%d] HashCount[%d]"), NewSize, NewMaxElements, Layout.Size, NewHashCount);
164
165 if constexpr (TAllocatorTraits<AllocatorType>::SupportsElementAlignment)
166 {
167 NewSlackSize = Elements.CalculateSlackGrow((SizeType)NewSize, (SizeType)OldSize, 1, Layout.Alignment);
168 }
169 else
170 {
171 NewSlackSize = Elements.CalculateSlackGrow((SizeType)NewSize, (SizeType)OldSize, 1);
172 }
173
174 if (NewSlackSize == NewSize)
175 {
176 // No need for expensive slack calculations if there is none
177 return NewMaxElements;
178 }
179
180 // Calculate the number of items we can fit in the allotted space
181 SizeType SlackNumElements = GetMaxElementsForAvailableSpace(NewSlackSize, NewHashCount, NewMaxElements, Layout);
182 if (SlackNumElements <= NewMaxElements)
183 {
184 // This can happen if we're basically at the slack limit, when we got to calculate the results, even adding one element tracking index can
185 // cause us to lose space due to alignment changes, in this case we just take the user input as the source of truth
186 return NewMaxElements;
187 }
188
189 HashCountType SlackHashCount = GetHashCount(SlackNumElements);
190 if (SlackHashCount > NewHashCount)
191 {
192 // New Size required to much space. Cut the new hash count in half and restrict the elements to the max for that hash count
193 SlackNumElements = SlackHashCount - 1;
194 SlackHashCount /= 2;
195 }
196
197 checkSlow(SlackNumElements >= NewMaxElements);
198 checkSlow(GetTotalMemoryRequiredInBytes(SlackNumElements, SlackHashCount, Layout) <= NewSlackSize);
199
200 return SlackNumElements;
201 }

208 // Use this is you'll be keeping the element data
209 [[nodiscard]] bool ResizeAllocationPreserveData(const SizeType NewMaxElements, const FCompactSetLayout& Layout, bool bPreserve = true)
210 {
211 bool bRequiresRehash = false;
212
213 if (NewMaxElements != MaxElements)
214 {
215 const HashCountType OldHashCount = MaxElements > 0 ? *(HashCountType*)(Elements.GetAllocation() + GetElementsSizeInBytes(MaxElements, Layout)) : 0;
216 const size_t OldSize = MaxElements > 0 ? GetTotalMemoryRequiredInBytes(MaxElements, OldHashCount, Layout) : 0;
217
218 const HashCountType NewHashCount = NewMaxElements > 0 ? UE::Core::CompactHashTable::GetHashCount(NewMaxElements) : 0;
219 const size_t NewSize = NewMaxElements > 0 ? GetTotalMemoryRequiredInBytes(NewMaxElements, NewHashCount, Layout) : 0;
220
221 checkf(NewSize >= 0 && NewSize <= MAX_int32, TEXT("Invalid size for TSet[%lld]: NewMaxElements[%d] ElementSize[%d] HashCount[%d]"), NewSize, NewMaxElements, Layout.Size, NewHashCount);
222
223 if (bPreserve && NewMaxElements > MaxElements && OldHashCount == NewHashCount)
224 {
225 // If preserving then we copy over all the data
226 if constexpr (TAllocatorTraits<AllocatorType>::SupportsElementAlignment)
227 {
228 Elements.ResizeAllocation((SizeType)OldSize, (SizeType)NewSize, 1, Layout.Alignment);
229 }
230 else
231 {
232 Elements.ResizeAllocation((SizeType)OldSize, (SizeType)NewSize, 1);
233 }
234
235 const uint32 NewTypeShift = UE::Core::CompactHashTable::GetTypeShift(NewMaxElements);
236
237 // This should always be true since our type size will change on Powerof2 barriers (256, 65536, etc)
238 check(NewTypeShift == UE::Core::CompactHashTable::GetTypeShift(MaxElements));
239
240 const uint8* OldHashTable = Elements.GetAllocation() + GetElementsSizeInBytes(MaxElements, Layout);
241 uint8* NewHashTable = Elements.GetAllocation() + GetElementsSizeInBytes(NewMaxElements, Layout);
242
243 const uint8* OldHashLocation = OldHashTable + ((size_t)MaxElements << NewTypeShift) * KeyStorageMultiplier + HashCountSize;
244 uint8* NewHashLocation = NewHashTable + ((size_t)NewMaxElements << NewTypeShift) * KeyStorageMultiplier + HashCountSize;
245
246 // WARNING: Order matters here since we're moving data around in the same buffer. Make sure we're moving the tail data first
247 // since we're expanding in size. So HashTable, then (optional) key storage, then hashcount and next indexes
248
249 // Copy hash lookup table (first so to free up the space for the other data to move)
250 FMemory::Memmove(NewHashLocation, OldHashLocation, ((size_t)NewHashCount << NewTypeShift));
251
252#if UE_ENABLE_COMPACTHASHTABLE_KEYSTORAGE
253 // Copy key storage if enabled
254 const uint8* OldKeyStorage = OldHashTable + ((size_t)MaxElements << NewTypeShift) + HashCountSize;
255 uint8* NewKeyStorage = NewHashTable + ((size_t)NewMaxElements << NewTypeShift) + HashCountSize;
256 FMemory::Memmove(NewKeyStorage, OldKeyStorage, ((size_t)NumElements << NewTypeShift));
257#endif
258
259 // Copy hash size + next index table
260 FMemory::Memmove(NewHashTable, OldHashTable, ((size_t)NumElements << NewTypeShift) + HashCountSize);
261 }
262 else
263 {
264 // If not preserving (or shrinking) then we only need to copy the array data
265 if constexpr (TAllocatorTraits<AllocatorType>::SupportsElementAlignment)
266 {
267 Elements.ResizeAllocation(NumElements * Layout.Size, (SizeType)NewSize, 1, Layout.Alignment);
268 }
269 else
270 {
271 Elements.ResizeAllocation(NumElements * Layout.Size, (SizeType)NewSize, 1);
272 }
273
274 // Update hash size
275 if (NewMaxElements > 0)
276 {
277 HashCountType* NewHashTable = (HashCountType*)(Elements.GetAllocation() + GetElementsSizeInBytes(NewMaxElements, Layout));
278
279 *NewHashTable = NewHashCount;
280 bRequiresRehash = true;
281 }
282 }
283
284 MaxElements = NewMaxElements;
285 }
286
287 return bRequiresRehash;
288 }
