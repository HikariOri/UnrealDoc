45 void FreeVirtualByBlock(void* Ptr, FPerBlockSize& Block, size_t AlignedSize)
46 {
47 // already locked
48 if (!RecycledLinks)
49 {
50 void *Alloc;
51 // If we ARE malloc, then we know we can (and must) commit part of our range for free links. Otherwise, we can just use malloc (and can't use our VM space anyway)
52 if (bBacksMalloc)
53 {
54 Alloc = AllocNewVM(MaximumAlignment);
55 uint32 Pages = MaximumAlignment / FPlatformMemory::FPlatformVirtualMemoryBlock::GetVirtualSizeAlignment();
56 check(Pages && MaximumAlignment % FPlatformMemory::FPlatformVirtualMemoryBlock::GetVirtualSizeAlignment() == 0);
57 FPlatformMemory::FPlatformVirtualMemoryBlock VMBlock(Alloc, Pages);
58 VMBlock.Commit();
59 }
60 else
61 {
62 Alloc = FMemory::Malloc(MaximumAlignment);
63 }
64 for (int32 Index = 0; (Index + 1) * sizeof(FFreeLink) <= MaximumAlignment; Index++)
65 {
66 FFreeLink* NewLink = new ((void*)(((uint8*)Alloc + Index * sizeof(FFreeLink)))) FFreeLink;
67 NewLink->Next = RecycledLinks;
68 RecycledLinks = NewLink;
69 }
70 LinkSize += MaximumAlignment;
71 checkSlow(RecycledLinks); // To suppress a static analysis warning about RecycledLinks being null below.
72 }
73
74 FFreeLink* Link = RecycledLinks;
75 RecycledLinks = Link->Next;
76 check(!Link->Ptr && Ptr);
77 Link->Ptr = Ptr;
78 Link->Next = Block.FirstFree;
79 Block.FirstFree = Link;
80
81 Block.FreeBlocksSize += AlignedSize;
82 }

87 virtual uint8* AllocNewVM(size_t AlignedSize)
88 {
89 uint8* Result = NextAlloc;
90 check(IsAligned(Result, MaximumAlignment) && IsAligned(AlignedSize, MaximumAlignment));
91 NextAlloc = Result + AlignedSize;
92 SpaceConsumed = NextAlloc - LowAddress;
93 if (SpaceConsumed > SpaceConsumedPeak)
94 {
95 SpaceConsumedPeak = SpaceConsumed;
96 }
97 return Result;
98 }

120 uint32 GetPagesForSizeAndAlignment(size_t Size, size_t Alignment = 1) const
121 {
122 check(Alignment <= MaximumAlignment && Alignment > 0);
123 size_t SizeAndAlignment = FMath::Max(Align(Size, Alignment), PageSize);
124 if (SizeAndAlignment * 2 >= TotalSize)
125 {
126 // this is hack, MB3 will ask for tons of virtual and never free it, so we won't round up to power of two
127 size_t Pages = SizeAndAlignment / PageSize;
128 check(Pages == uint32(Pages)); // overflow of uint32
129 return uint32(Pages);
130 }
131 size_t BlockIndex = FMath::CeilLogTwo64(FMath::Max(SizeAndAlignment, PageSize));
132 size_t AlignedSize = size_t(1) << BlockIndex;
133 check(AlignedSize % PageSize == 0);
134 size_t Pages = AlignedSize / PageSize;
135 check(Pages == uint32(Pages)); // overflow of uint32
136 return uint32(Pages);
137 }

139 void* AllocateVirtualPages(uint32 NumPages, size_t AlignmentForCheck = 1)
140 {
141 check(AlignmentForCheck <= MaximumAlignment && AlignmentForCheck > 0 && NumPages);
142
143
144 size_t BlockIndex = FMath::CeilLogTwo64(NumPages * PageSize);
145 size_t AlignedSize = size_t(1) << BlockIndex;
146 bool bHackForHugeBlock = false;
147 if (size_t(NumPages) * PageSize * 2 >= TotalSize)
148 {
149 // this is hack, MB3 will ask for tons of virtual and never free it, so we won't round up to power of two
150 AlignedSize = size_t(NumPages) * PageSize;
151 bHackForHugeBlock = true;
152 }
153 FPerBlockSize& Block = Blocks[BlockIndex];
154
155 UE::TUniqueLock Lock(Mutex);
156
157
158 uint8* Result;
159 if (Block.FirstFree)
160 {
161 check(!bHackForHugeBlock);
162 Result = (uint8*)Block.FirstFree->Ptr;
163 check(Result);
164 Block.FirstFree->Ptr = nullptr;
165
166 FFreeLink* Next = RecycledLinks;
167 RecycledLinks = Block.FirstFree;
168 Block.FirstFree = Block.FirstFree->Next;
169 check(!Block.FirstFree || Block.FirstFree->Ptr);
170 RecycledLinks->Next = Next;
171
172 check(IsAligned(Result, FMath::Min(AlignedSize, MaximumAlignment)));
173 Block.FreeBlocksSize -= AlignedSize;
174 }
175 else
176 {
177 size_t AllocSize = FMath::Max(AlignedSize, MaximumAlignment);
178 uint8* LocalNextAlloc = AllocNewVM(AllocSize);
179 check(IsAligned(LocalNextAlloc, MaximumAlignment));
180
181 uint8* NewNextAlloc = LocalNextAlloc + AllocSize;
182
183 if (NewNextAlloc > HighAddress)
184 {
185 FPlatformMemory::OnOutOfMemory(HighAddress - LowAddress, 0);
186 }
187
188 Block.AllocBlocksSize += AllocSize;
189 Result = LocalNextAlloc;
190 check(Result);
191 LocalNextAlloc += AlignedSize;
192 while (LocalNextAlloc < NewNextAlloc && !bHackForHugeBlock)
193 {
194 FreeVirtualByBlock(LocalNextAlloc, Block, AlignedSize);
195 LocalNextAlloc += AlignedSize;
196 }
197 }
198 check(IsAligned(Result, AlignmentForCheck));
199 return Result;
200 }

202 void FreeVirtual(void* Ptr, uint32 NumPages)
203 {
204 if (size_t(NumPages) * PageSize * 2 >= TotalSize)
205 {
206 // this is hack, MB3 will ask for tons of virtual and never free it, so we won't round up to power of two
207 check(!"Huge vm blocks may not be freed.");
208 return;
209 }
210 size_t BlockIndex = FMath::CeilLogTwo64(NumPages * PageSize);
211 size_t AlignedSize = size_t(1) << BlockIndex;
212
213 FPerBlockSize& Block = Blocks[BlockIndex];
214 UE::TUniqueLock Lock(Mutex);
215 FreeVirtualByBlock(Ptr, Block, AlignedSize);
216 }
