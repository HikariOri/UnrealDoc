103struct FSceneOutlinerItemSelection
104{
106 mutable TArray<TWeakPtr<ISceneOutlinerTreeItem>> SelectedItems;
107
108 FSceneOutlinerItemSelection() {}
109
110 FSceneOutlinerItemSelection(const TArray<FSceneOutlinerTreeItemPtr>& InSelectedItems)
111 : SelectedItems(InSelectedItems) {}
112
113 FSceneOutlinerItemSelection(SSceneOutlinerTreeView& Tree)
114 : FSceneOutlinerItemSelection(Tree.GetSelectedItems()) {}
115
117 template <typename TreeType>
118 bool Has() const
119 {
120 for (const TWeakPtr<ISceneOutlinerTreeItem>& Item : SelectedItems)
121 {
122 if (const auto ItemPtr = Item.Pin())
123 {
124 if (ItemPtr->IsA<TreeType>())
125 {
126 return true;
127 }
128 }
129 }
130 return false;
131 }
132
134 uint32 Num() const
135 {
136 return SelectedItems.Num();
137 }
138
140 template <typename ...TreeTypes>
141 uint32 Num() const
142 {
143 uint32 Result = 0;
144 for (const TWeakPtr<ISceneOutlinerTreeItem>& Item : SelectedItems)
145 {
146 if (const auto ItemPtr = Item.Pin())
147 {
148 if ((ItemPtr->IsA<TreeTypes>() || ...))
149 {
150 ++Result;
151 }
152 }
153 }
154 return Result;
155 }
156
158 void Add(FSceneOutlinerTreeItemPtr NewItem)
159 {
160 SelectedItems.Add(NewItem);
161 }
162
164 template <typename TreeType>
165 void Get(TArray<TreeType*>& OutArray) const
166 {
167 for (const TWeakPtr<ISceneOutlinerTreeItem>& Item : SelectedItems)
168 {
169 if (const auto ItemPtr = Item.Pin())
170 {
171 if (TreeType* CastedItem = ItemPtr->CastTo<TreeType>())
172 {
173 OutArray.Add(CastedItem);
174 }
175 }
176 }
177 }
178
179 void Get(TArray<FSceneOutlinerTreeItemPtr>& OutArray) const
180 {
181 for (const TWeakPtr<ISceneOutlinerTreeItem>& Item : SelectedItems)
182 {
183 if (const auto ItemPtr = Item.Pin())
184 {
185 OutArray.Add(ItemPtr);
186 }
187 }
188 }
189
191 template <typename TreeType>
192 void ForEachItem(TFunctionRef<void(TreeType&)> Func) const
193 {
194 for (const TWeakPtr<ISceneOutlinerTreeItem>& Item : SelectedItems)
195 {
196 if (const auto ItemPtr = Item.Pin())
197 {
198 if (TreeType* CastedItem = ItemPtr->CastTo<TreeType>())
199 {
200 Func(*CastedItem);
201 }
202 }
203 }
204 }
205
207 void ForEachItem(TFunctionRef<void(FSceneOutlinerTreeItemPtr&)> Func) const
208 {
209 for (const TWeakPtr<ISceneOutlinerTreeItem>& Item : SelectedItems)
210 {
211 if (auto ItemPtr = Item.Pin())
212 {
213 Func(ItemPtr);
214 }
215 }
216 }
217
219 template <typename DataType>
220 TArray<DataType> GetData(TFunctionRef<bool(const TWeakPtr<ISceneOutlinerTreeItem>&, DataType&)> Selector) const
221 {
222 TArray<DataType> Result;
223 for (TWeakPtr<ISceneOutlinerTreeItem>& Item : SelectedItems)
224 {
225 DataType Data;
226 if (Selector(Item, Data))
227 {
228 Result.Add(Data);
229 }
230 }
231 return Result;
232 }

614 // Test the filters using stack-allocated data to prevent unnecessary heap allocations
615 template <typename TreeItemType, typename TreeItemData>
616 FSceneOutlinerTreeItemPtr CreateItemFor(const TreeItemData& Data, TFunctionRef<void(const TreeItemType&)> OnItemPassesFilters, bool bForce = false)
617 {
618 const TreeItemType Temporary(Data);
619 bool bPassesFilters = Filters->PassesAllFilters(Temporary);
620
621 if(FilterCollection)
622 {
623 bPassesFilters &= FilterCollection->PassesAllFilters(Temporary);
624 }
625
626 if (bPassesFilters)
627 {
628 OnItemPassesFilters(Temporary);
629 }
630
631 if (SearchBoxFilter)
632 {
633 bPassesFilters &= SearchBoxFilter->PassesFilter(Temporary);
634 }
635
636 if (bForce || bPassesFilters)
637 {
638 FSceneOutlinerTreeItemPtr Result = MakeShareable(new TreeItemType(Data));
639 Result->WeakSceneOutliner = StaticCastSharedRef<ISceneOutliner>(AsShared());
640 Result->Flags.bIsFilteredOut = !bPassesFilters;
641 Result->Flags.bInteractive = Filters->GetInteractiveState(*Result) && InteractiveFilters->GetInteractiveState(*Result);
642 return Result;
643 }
644
645 return nullptr;
646 }
