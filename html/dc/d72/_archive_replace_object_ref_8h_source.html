38 template <typename ContainerType, typename ElementToObjectType>
39 bool ShouldSkipReplacementCheckForObjectPtr(FObjectPtr& Obj, const ContainerType& ReplacementContainer, const ElementToObjectType& ElementToObject)
40 {
41 if (Obj.IsResolved())
42 {
43 return false;
44 }
45
46 if (!CanIgnoreUnresolvedImports.IsSet())
47 {
48 bool bCanIgnoreUnresolvedImportsLocal = ReplacementContainer.Num() == 0;
49 if (!bCanIgnoreUnresolvedImportsLocal)
50 {
51 bCanIgnoreUnresolvedImportsLocal = true; // Will be set to false if any of the criteria in the loop below are met.
52 for (const auto& ReplacementElem : ReplacementContainer)
53 {
54 if (const UObject* ReplacementObject = ElementToObject(ReplacementElem))
55 {
56 if (ReplacementObject->GetOutermost() != GetTransientPackage())
57 {
58 bCanIgnoreUnresolvedImportsLocal = false;
59 break;
60 }
61 }
62 }
63 }
64 CanIgnoreUnresolvedImports = bCanIgnoreUnresolvedImportsLocal;
65 }
66
67 if (CanIgnoreUnresolvedImports.GetValue())
68 {
69 return true;
70 }
71
72 if (UClass* ReferenceClass = Obj.GetClass())
73 {
74 bool bReferenceTypeIsRelatedToReplacementType = false;
75 for (const auto& ReplacementElem : ReplacementContainer)
76 {
77 if (const UObject* ReplacementObject = ElementToObject(ReplacementElem))
78 {
79 if (ReplacementObject->IsA(ReferenceClass) || Obj.IsA(ReplacementObject->GetClass()))
80 {
81 bReferenceTypeIsRelatedToReplacementType = true;
82 break;
83 }
84 }
85 }
86 if (!bReferenceTypeIsRelatedToReplacementType)
87 {
88 return true;
89 }
90 }
91
92 return false;
93 }

199 ArIgnoreOuterRef = !!(Flags & EArchiveReplaceObjectFlags::IgnoreOuterRef);
200 ArIgnoreClassGeneratedByRef = !(Flags & EArchiveReplaceObjectFlags::IncludeClassGeneratedByRef);

211 void SerializeSearchObject()
212 {
213 ReplacedReferences.Reset();
214
215 if (SearchObject != NULL && !SerializedObjects.Find(SearchObject)
216 && (ReplacementMap.Num() > 0 || bNullPrivateReferences))
217 {
218 // start the initial serialization
219 SerializedObjects.Add(SearchObject);
220 SerializingObject = SearchObject;
221 SerializeObject(SearchObject);
222 for (int32 Iter = 0; Iter < PendingSerializationObjects.Num(); Iter++)
223 {
224 SerializingObject = PendingSerializationObjects[Iter];
225 SerializeObject(SerializingObject);
226 }
227 PendingSerializationObjects.Reset();
228 }
229 }

234 virtual FArchive& operator<<( UObject*& Obj ) override
235 {
236 if (Obj != NULL)
237 {
238 // If these match, replace the reference
239 if (T* const* ReplaceWith = (T* const*)((const TMap<UObject*, UObject*>*)&ReplacementMap)->Find(Obj))
240 {
241 Obj = *ReplaceWith;
242 if (bTrackReplacedReferences)
243 {
244 ReplacedReferences.FindOrAdd(SerializingObject).AddUnique(GetSerializedProperty());
245 }
246 Count++;
247 }
248 // A->IsIn(A) returns false, but we don't want to NULL that reference out, so extra check here.
249 else if ( Obj == SearchObject || Obj->IsIn(SearchObject) )
250 {
251#if 0
252 // DEBUG: Log when we are using the A->IsIn(A) path here.
253 if(Obj == SearchObject)
254 {
255 FString ObjName = Obj->GetPathName();
256 UE_LOGF(LogSerialization, Log, "FArchiveReplaceObjectRef: Obj == SearchObject : '%ls'", *ObjName );
257 }
258#endif
259 bool bAlreadyAdded = false;
260 SerializedObjects.Add(Obj, &bAlreadyAdded);
261 if (!bAlreadyAdded)
262 {
263 // No recursion
264 PendingSerializationObjects.Add(Obj);
265 }
266 }
267 else if ( bNullPrivateReferences && !Obj->HasAnyFlags(RF_Public) )
268 {
269 Obj = NULL;
270 }
271 }
272 return *this;
273 }
