788 inline void RecomputeDerivedMatrices()
789 {
790 // Compute the view projection matrix and its inverse.
791 WorldToClip = GetWorldToView() * GetViewToClip();
792 ClipToWorld = GetClipToView() * GetViewToWorld();
793
794 // Compute a transform from view origin centered world-space to clip space.

796 ClipToTranslatedWorld = GetClipToView() * GetInvTranslatedViewMatrix();

799 static const FMatrix InvertProjectionMatrix( const FMatrix& M )
800 {
801 if( M.M[1][0] == 0.0f &&
802 M.M[3][0] == 0.0f &&
803 M.M[0][1] == 0.0f &&
804 M.M[3][1] == 0.0f &&
805 M.M[0][2] == 0.0f &&
806 M.M[1][2] == 0.0f &&
807 M.M[0][3] == 0.0f &&
808 M.M[1][3] == 0.0f &&
809 M.M[2][3] == 1.0f &&
810 M.M[3][3] == 0.0f )
811 {
812 // Solve the common case directly with very high precision.
813 /*
814 M =
815 | a | 0 | 0 | 0 |
816 | 0 | b | 0 | 0 |
817 | s | t | c | 1 |
818 | 0 | 0 | d | 0 |
819 */
820
821 double a = M.M[0][0];
822 double b = M.M[1][1];
823 double c = M.M[2][2];
824 double d = M.M[3][2];
825 double s = M.M[2][0];
826 double t = M.M[2][1];
827
828 return FMatrix(
829 FPlane( 1.0 / a, 0.0f, 0.0f, 0.0f ),
830 FPlane( 0.0f, 1.0 / b, 0.0f, 0.0f ),
831 FPlane( 0.0f, 0.0f, 0.0f, 1.0 / d ),
832 FPlane( -s/a, -t/b, 1.0f, -c/d )
833 );
834 }
835 else
836 {
837 return M.Inverse();
838 }
839 }
