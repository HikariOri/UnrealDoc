100 // Fully qualified path to a process-specific temporary directory on the local machine.
101 // This directory will be destroyed when the toolchain is destroyed
102 const FString& GetLocalTempDir() const
103 {
104 // nothing like dispatch_once on windows?
105 // this should always happen on the same thread anyway
106 static bool bLocalTempDirCreated = false;
107 if (!bLocalTempDirCreated)
108 {
109 LocalTempFolder = FPaths::Combine(FPlatformProcess::UserTempDir(), TEXT("MetalShaderCompilation"), FString::Printf(TEXT("%u"), FPlatformProcess::GetCurrentProcessId()));
110 if (!FPaths::DirectoryExists(LocalTempFolder))
111 {
112 bool bSuccess = IFileManager::Get().MakeDirectory(*LocalTempFolder, true);
113 if (!bSuccess)
114 {
115 UE_LOGF(LogMetalCompilerSetup, Fatal, "Attempting to create temporary directory at %ls but failed.", *LocalTempFolder);
116 }
117 }
118 bLocalTempDirCreated = true;
119 }
120
121 return LocalTempFolder;
122 }
