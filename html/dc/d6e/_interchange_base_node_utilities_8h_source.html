22 static FAttributeKey BuildArrayKeyFromIndex(const FString& BaseKeyName, uint32 Index)
23 {

28 static FString BuildDisplayNameForArrayKey(const FString& BaseDisplayName, const FString& ArrayAttributeKey)
29 {
30 static FString IndexKeyString = ArrayIndexKey();
31 const int32 IndexPosition = ArrayAttributeKey.Find(IndexKeyString);
32 if (IndexPosition != INDEX_NONE)
33 {
34 const int32 ChopIndex = IndexPosition + IndexKeyString.Len();
35 if (ChopIndex < ArrayAttributeKey.Len())
36 {
37 return BaseDisplayName + ArrayAttributeKey.RightChop(ChopIndex);
38 }
39 }
40
41 return BaseDisplayName;

122 void GetItems(TArray<ItemType>& OutItems) const
123 {
124 //The Class must be initialize properly before we can use it
125 TSharedPtr<FAttributeStorage, ESPMode::ThreadSafe> AttributePtr = Attributes.Pin();
126 if (!ensure(AttributePtr.IsValid()))
127 {
128 OutItems.Empty();

156 HandleItem.Get(OutItem);

160 bool AddItem(const ItemType& Item)
161 {
162 //The Class must be initialize properly before we can use it
163 TSharedPtr<FAttributeStorage, ESPMode::ThreadSafe> AttributePtr = Attributes.Pin();
164 if (!ensure(AttributePtr.IsValid()))
165 {
166 return false;
167 }
168
169 if (!AttributePtr->ContainAttribute(GetKeyCount()))
170 {
171 const int32 DependencyCount = 0;
172 EAttributeStorageResult Result = AttributePtr->RegisterAttribute<int32>(GetKeyCount(), DependencyCount);
173 if (!IsAttributeStorageResultSuccess(Result))
174 {
175 LogAttributeStorageErrors(Result, TEXT("TArrayAttributeHelper.AddName"), GetKeyCount());
176 return false;
177 }
178 }
179 FAttributeStorage::TAttributeHandle<int32> Handle = AttributePtr->GetAttributeHandle<int32>(GetKeyCount());
180 if (!ensure(Handle.IsValid()))
181 {
182 return false;
183 }
184 int32 ItemIndex = 0;
185 Handle.Get(ItemIndex);
186 FAttributeKey ItemIndexKey = GetIndexKey(ItemIndex);
187 EAttributeStorageResult AddItemResult = AttributePtr->RegisterAttribute<ItemType>(ItemIndexKey, Item);
188 if (!IsAttributeStorageResultSuccess(AddItemResult))
189 {
190 LogAttributeStorageErrors(AddItemResult, TEXT("TArrayAttributeHelper.AddName"), ItemIndexKey);
191 return false;
192 }
193
194 //Success, increment the item counter
195 ItemIndex++;
196 Handle.Set(ItemIndex);
197 return true;
198 }
199
200 bool RemoveItem(const ItemType& ItemToDelete)
201 {
202 //The Class must be initialize properly before we can use it
203 TSharedPtr<FAttributeStorage, ESPMode::ThreadSafe> AttributePtr = Attributes.Pin();
204 if (!ensure(AttributePtr.IsValid()))
205 {
206 return false;
207 }
208
209 int32 ItemCount = 0;
210 if (!AttributePtr->ContainAttribute(GetKeyCount()))
211 {
212 return false;
213 }
214 FAttributeStorage::TAttributeHandle<int32> Handle = AttributePtr->GetAttributeHandle<int32>(GetKeyCount());
215 if (!Handle.IsValid())
216 {
217 return false;
218 }
219 Handle.Get(ItemCount);
220 int32 DecrementKey = 0;
221 for (int32 ItemIndex = 0; ItemIndex < ItemCount; ++ItemIndex)
222 {
223 FAttributeKey DepIndexKey = GetIndexKey(ItemIndex);
224 FAttributeStorage::TAttributeHandle<ItemType> HandleItem = AttributePtr->GetAttributeHandle<ItemType>(DepIndexKey);
225 if (!HandleItem.IsValid())
226 {
227 continue;
228 }
229 ItemType Item;
230 HandleItem.Get(Item);
231 //Remove at most one item
232 if (DecrementKey == 0 && Item == ItemToDelete)
233 {
234 //Remove this entry
235 AttributePtr->UnregisterAttribute(DepIndexKey);
236 Handle.Set(ItemCount - 1);
237 //We have to rename the key for all the next item
238 DecrementKey++;
239 }
240 else if (DecrementKey > 0)
241 {
242 FAttributeKey NewDepIndexKey = GetIndexKey(ItemIndex - DecrementKey);
243 EAttributeStorageResult UnregisterResult = AttributePtr->UnregisterAttribute(DepIndexKey);
244 if (IsAttributeStorageResultSuccess(UnregisterResult))
245 {
246 EAttributeStorageResult RegisterResult = AttributePtr->RegisterAttribute<ItemType>(NewDepIndexKey, Item);
247 if (!IsAttributeStorageResultSuccess(RegisterResult))
248 {
249 LogAttributeStorageErrors(RegisterResult, TEXT("TArrayAttributeHelper.RemoveItem"), NewDepIndexKey);
250 }
251 }
252 else
253 {
254 LogAttributeStorageErrors(UnregisterResult, TEXT("TArrayAttributeHelper.RemoveItem"), DepIndexKey);
255 }
256
257 //Avoid doing more code in the for since the HandleItem is now invalid
258 continue;
259 }
260 }
261 return true;
262 }
263
264 bool RemoveAllItems()
265 {
266 //The Class must be initialize properly before we can use it
267 TSharedPtr<FAttributeStorage, ESPMode::ThreadSafe> AttributePtr = Attributes.Pin();
268 if (!ensure(AttributePtr.IsValid()))
269 {
270 return false;
271 }
272
273 int32 ItemCount = 0;
274 if (!AttributePtr->ContainAttribute(GetKeyCount()))
275 {
276 return false;
277 }
278 FAttributeStorage::TAttributeHandle<int32> HandleCount = AttributePtr->GetAttributeHandle<int32>(GetKeyCount());
279 if (!HandleCount.IsValid())
280 {
281 return false;
282 }
283 HandleCount.Get(ItemCount);
284 //Remove all attribute one by one
285 for (int32 NameIndex = 0; NameIndex < ItemCount; ++NameIndex)
286 {
287 FAttributeKey DepIndexKey = GetIndexKey(NameIndex);
288 AttributePtr->UnregisterAttribute(DepIndexKey);
289 }
290 //Make sure Count is zero
291 ItemCount = 0;
292 HandleCount.Set(ItemCount);
293 return true;
294 }

302 }

394 Hash

397 AttributesPtr->GetAttributeHandle<KeyType>(IndexKey)

513 void RebuildCache()
514 {
515 LLM_SCOPE_BYNAME(TEXT("Interchange"));
516
517 TSharedPtr<FAttributeStorage, ESPMode::ThreadSafe> AttributesPtr = Attributes.Pin();
518 if (AttributesPtr.IsValid() && KeyCountHandle.IsValid())
519 {
520 int32 KeyCount;
521 if (KeyCountHandle.Get(KeyCount) == EAttributeStorageResult::Operation_Success)
522 {
523 CachedKeysAndValues.Empty(KeyCount);
524
525 for (int32 Index = 0; Index < KeyCount; Index++)
526 {
527 TAttributeHandle<KeyType> KeyAttribute = AttributesPtr->GetAttributeHandle<KeyType>(GetKeyAttribute(Index));
528 if (!ensure(KeyAttribute.IsValid()))
529 {
530 continue;
531 }
532
533 KeyType Key;
534 KeyAttribute.Get(Key);
535
536 TAttributeHandle<ValueType> ValueAttribute = AttributesPtr->GetAttributeHandle<ValueType>(GetValueAttribute(Key));
537 if (!ensure(ValueAttribute.IsValid()))
538 {
539 continue;
540 }
541
542 CachedKeysAndValues.Add(Key, TPair<TAttributeHandle<KeyType>, TAttributeHandle<ValueType>>(KeyAttribute, ValueAttribute));
543 }
544 }
545 }

548private:
549 FAttributeKey GetKeyAttribute(int32 Index) const
550 {
551 static const FString KeyIndex = "_KeyIndex_";
552 FString IndexedKey = KeyCountHandle.GetKey().ToString();
553 IndexedKey.Reserve(KeyIndex.Len() + 16 /*Max size for a int32*/);
554 IndexedKey.Append(KeyIndex);
555 IndexedKey.AppendInt(Index);
556 return FAttributeKey(MoveTemp(IndexedKey));
557 }
558
559 FAttributeKey GetValueAttribute(const KeyType& InKey) const
560 {
561 static const FString KeyChars = "_Key_";
562 FString ValueAttribute = KeyCountHandle.GetKey().ToString();
563 FString Key = TTypeToString<KeyType>::ToString(InKey);
564 ValueAttribute.Reserve(KeyChars.Len() + Key.Len());
565 ValueAttribute.Append(KeyChars);
566 ValueAttribute.Append(Key);
567 return FAttributeKey(MoveTemp(ValueAttribute));
568 }

570 TAttributeHandle<KeyType> GetLastKeyAttributeHandle() const
571 {
572 TSharedPtr<FAttributeStorage, ESPMode::ThreadSafe> AttributesPtr = Attributes.Pin();
573 if (!ensure(AttributesPtr.IsValid()))
574 {
575 return {};
576 }
577 return AttributesPtr->GetAttributeHandle<KeyType>(GetKeyAttribute(CachedKeysAndValues.Num() - 1));
578 }
579
580 bool RemoveBySwap(uint32 Hash, const KeyType& InKey, TPair<TAttributeHandle<KeyType>, TAttributeHandle<ValueType>>& CachedPair)
581 {
582 TSharedPtr<FAttributeStorage, ESPMode::ThreadSafe> AttributesPtr = Attributes.Pin();
583 if (!ensure(AttributesPtr.IsValid()))
584 {
585 return false;
586 }
587
588 TAttributeHandle<KeyType> LastKeyIndex = GetLastKeyAttributeHandle();
589 KeyType LastKey;
590 LastKeyIndex.Get(LastKey);
591 CachedKeysAndValues[LastKey].Key = CachedPair.Key;
592 CachedPair.Key.Set(MoveTemp(LastKey));
593
594 AttributesPtr->UnregisterAttribute(LastKeyIndex.GetKey());
595 AttributesPtr->UnregisterAttribute(CachedPair.Value.GetKey());
596 CachedKeysAndValues.RemoveByHash(Hash, InKey);
597 KeyCountHandle.Set(CachedKeysAndValues.Num());
598
599 return true;
600 }
601
602 void EmptyInternal(int32 NumOfExpectedElements)
603 {
604 TSharedPtr<FAttributeStorage, ESPMode::ThreadSafe> AttributesPtr = Attributes.Pin();
605 if (!ensure(AttributesPtr.IsValid()))
606 {
607 return;
608 }
609
610 for (const TPair<KeyType, TPair<TAttributeHandle<KeyType>, TAttributeHandle<ValueType>>>& Pair : CachedKeysAndValues)
611 {
612 const FAttributeKey& KeyAttribute = Pair.Value.Key.GetKey();
613 AttributesPtr->UnregisterAttribute(KeyAttribute);
614
615 const FAttributeKey& ValueAttribute = Pair.Value.Value.GetKey();
616 AttributesPtr->UnregisterAttribute(ValueAttribute);
617 }
618 CachedKeysAndValues.Empty(NumOfExpectedElements);
619 KeyCountHandle.Set(0);
620 }
621
622 TMap<KeyType, TPair<TAttributeHandle<KeyType>, TAttributeHandle<ValueType>>> CachedKeysAndValues;
623 TAttributeHandle<int32> KeyCountHandle; //Assign in Initialize function, it will ensure if its the default value (NAME_None) when using the class
624 TWeakPtr<FAttributeStorage, ESPMode::ThreadSafe> Attributes = nullptr;
625};

633class TSetAttributeHelper
634{
635 template<typename T>
636 using TAttributeHandle = FAttributeStorage::TAttributeHandle<T>;
637
638public:
639 void Initialize(const TSharedPtr<FAttributeStorage, ESPMode::ThreadSafe>& InAttributes, const FString& BaseKeyName)
640 {
641 LLM_SCOPE_BYNAME(TEXT("Interchange"));
642
643 Attributes = InAttributes;
644 KeyCount = BaseKeyName;
645
646 if (!InAttributes->ContainAttribute(KeyCount))
647 {
648 EAttributeStorageResult Result = InAttributes->RegisterAttribute<int32>(KeyCount, 0);
649 check(Result == EAttributeStorageResult::Operation_Success);
650 KeyCountHandle = InAttributes->GetAttributeHandle<int32>(KeyCount);
651 }
652 else
653 {
654 KeyCountHandle = InAttributes->GetAttributeHandle<int32>(KeyCount);
655 RebuildCache();
656 }
657 }
658
659 TSetAttributeHelper() = default;
660 TSetAttributeHelper(const TSetAttributeHelper&) = default;
661 TSetAttributeHelper(TSetAttributeHelper&&) = default;
662 TSetAttributeHelper& operator=(const TSetAttributeHelper&) = default;
663 TSetAttributeHelper& operator=(TSetAttributeHelper&&) = default;
664
665 ~TSetAttributeHelper()
666 {
667 Attributes.Reset();
668 }
669
670 /*
671 * Adds an item. Returns true if the item is added successfully.
672 * Returns false if the item is already present or if Initialize was not properly called before adding an item.
673 */
674 bool AddItem(const ValueType& InValue)
675 {
676 if (!CachedValues.Contains(InValue))
677 {
678 TSharedPtr<FAttributeStorage, ESPMode::ThreadSafe> AttributesPtr = Attributes.Pin();
679 if (!ensure(AttributesPtr.IsValid()))
680 {
681 return false;
682 }
683
684 FAttributeKey AttributeKey = GetIndexKey(CachedValues.Num());
685 ensure(AttributesPtr->RegisterAttribute<ValueType>(AttributeKey, InValue) == EAttributeStorageResult::Operation_Success);
686 CachedValues.Add(InValue, AttributeKey);
687 KeyCountHandle.Set(CachedValues.Num());
688
689 return true;
690 }
691
692 return false;
693 }
694
695 bool HasItem(const ValueType& InValue) const
696 {
697 TSharedPtr<FAttributeStorage, ESPMode::ThreadSafe> AttributesPtr = Attributes.Pin();
698 if (!ensure(AttributesPtr.IsValid()))
699 {
700 return false;
701 }
702
703 return CachedValues.Contains(InValue);
704 }
705
706 /*
707 * Remove an item if it exists. Will realign attributes by indices in the storage if needed (0(N)).
708 */
709 bool RemoveItem(const ValueType& InValue)
710 {
711 TSharedPtr<FAttributeStorage, ESPMode::ThreadSafe> AttributesPtr = Attributes.Pin();
712 if (!ensure(AttributesPtr.IsValid()))
713 {
714 return false;
715 }
716
717 {
718 FAttributeKey* KeyToRemove = CachedValues.Find(InValue);
719 if (!KeyToRemove)
720 {
721 return false;
722 }
723
724 AttributesPtr->UnregisterAttribute(*KeyToRemove);
725 CachedValues.Remove(InValue);
726 KeyCountHandle.Set(CachedValues.Num());
727 }
728
729 //When removing an item, we need to re-align the keys by indices without having gaps so it can be loaded properly later.
730 uint32 Index = 0;
731 for (const TPair<ValueType, FAttributeKey>& Pair : CachedValues)
732 {
733 const FAttributeKey& AttributeKey = Pair.Value;
734 const FAttributeKey NewAttributeKey = GetIndexKey(Index++);
735 if (NewAttributeKey == AttributeKey)
736 {
737 //This item is already assigned the correct index.
738 continue;
739 }
740
741 AttributesPtr->UnregisterAttribute(AttributeKey);
742 ensure(AttributesPtr->RegisterAttribute<ValueType>(NewAttributeKey, Pair.Key) == EAttributeStorageResult::Operation_Success);
743
744 CachedValues[Pair.Key] = NewAttributeKey;
745 }
746
747 return true;
748 }

803 void RebuildCache()
804 {
805 LLM_SCOPE_BYNAME(TEXT("Interchange"));
806
807 TSharedPtr<FAttributeStorage, ESPMode::ThreadSafe> AttributesPtr = Attributes.Pin();
808 if (AttributesPtr.IsValid() && KeyCountHandle.IsValid())
809 {
810 int32 ItemCount;
811 if (KeyCountHandle.Get(ItemCount) == EAttributeStorageResult::Operation_Success)
812 {
813 CachedValues.Empty(ItemCount);
814
815 for (int32 Index = 0; Index < ItemCount; Index++)
