201 EPreAnimatedStorageRequirement RestorePreAnimatedStateStorage(FPreAnimatedStorageIndex StorageIndex, EPreAnimatedStorageRequirement SourceRequirement, EPreAnimatedStorageRequirement TargetRequirement, const FRestoreStateParams& Params) override
202 {
203 if (RestoreMask)
204 {
205 if (!RestoreMask->CanRestore(PreAnimatedStorage[StorageIndex].Key))
206 {
207 return EPreAnimatedStorageRequirement::NoChange;
208 }
209 }
210
211 if (SourceRequirement == EPreAnimatedStorageRequirement::Persistent)
212 {
213 // Restoring global state
214 if (TargetRequirement == EPreAnimatedStorageRequirement::None)
215 {
216 FCachedData CachedData = MoveTemp(PreAnimatedStorage[StorageIndex]);
217
218 KeyToStorageIndex.Remove(CachedData.Key);
219 PreAnimatedStorage.RemoveAt(StorageIndex, 1);
220 TransientPreAnimatedStorage.Remove(StorageIndex);
221
222 if (CachedData.bInitialized)
223 {
224 Traits.RestorePreAnimatedValue(CachedData.Key, CachedData.Value, Params);
225 }
226 return EPreAnimatedStorageRequirement::None;
227 }
228 else
229 {
230 ensure(TargetRequirement == EPreAnimatedStorageRequirement::NoChange);
231
232 FCachedData& CachedData = PreAnimatedStorage[StorageIndex];
233 if (CachedData.bInitialized)
234 {
235 Traits.RestorePreAnimatedValue(CachedData.Key, CachedData.Value, Params);
236 }
237
238 return EPreAnimatedStorageRequirement::NoChange;
239 }
240 }
241
242 ensure(SourceRequirement == EPreAnimatedStorageRequirement::Transient);
243
244 // Always restore from the transient storage if available
245 if (StorageType* CachedData = TransientPreAnimatedStorage.Find(StorageIndex))
246 {
247 Traits.RestorePreAnimatedValue(PreAnimatedStorage[StorageIndex].Key, *CachedData, Params);
248
249 TransientPreAnimatedStorage.Remove(StorageIndex);
250
251 return EPreAnimatedStorageRequirement::Persistent;
252 }
253
254 if (TargetRequirement == EPreAnimatedStorageRequirement::None)
255 {
256 FCachedData& ActualValue = PreAnimatedStorage[StorageIndex];
257 if (ActualValue.bPersistent)
258 {
259 if (ActualValue.bInitialized)
260 {
261 Traits.RestorePreAnimatedValue(ActualValue.Key, ActualValue.Value, Params);
262 }
263 return EPreAnimatedStorageRequirement::Persistent;
264 }
265
266 FCachedData Tmp = MoveTemp(PreAnimatedStorage[StorageIndex]);
267
268 KeyToStorageIndex.Remove(Tmp.Key);
269 PreAnimatedStorage.RemoveAt(StorageIndex, 1);
270 TransientPreAnimatedStorage.Remove(StorageIndex);
271
272 if (Tmp.bInitialized)
273 {
274 Traits.RestorePreAnimatedValue(Tmp.Key, Tmp.Value, Params);
275 }
276
277 return EPreAnimatedStorageRequirement::None;
278 }
279
280 if (TargetRequirement == EPreAnimatedStorageRequirement::Persistent)
281 {
282 ensureMsgf(
283 this->ParentExtension->IsCapturingGlobalState(),
284 TEXT("We shouldn't move cached values to the persistent storage if we are not capturing global state!"));
285
286 // Restore the value but keep the value cached
287 FCachedData& PersistentData = PreAnimatedStorage[StorageIndex];
288
289 if (PersistentData.bInitialized)
290 {
291 PersistentData.bPersistent = true;
292 Traits.RestorePreAnimatedValue(PersistentData.Key, PersistentData.Value, Params);
293 }
294 }
295
296 return EPreAnimatedStorageRequirement::Persistent;
297 }

307 EPreAnimatedStorageRequirement DiscardPreAnimatedStateStorage(FPreAnimatedStorageIndex StorageIndex, EPreAnimatedStorageRequirement SourceRequirement) override
308 {
309 if (RestoreMask)
310 {
311 if (!RestoreMask->CanRestore(PreAnimatedStorage[StorageIndex].Key))
312 {
313 return EPreAnimatedStorageRequirement::NoChange;
314 }
315 }
316
317 if (SourceRequirement == EPreAnimatedStorageRequirement::Persistent)
318 {
319 KeyType Key = PreAnimatedStorage[StorageIndex].Key;
320
321 KeyToStorageIndex.Remove(Key);
322 PreAnimatedStorage.RemoveAt(StorageIndex, 1);
323 TransientPreAnimatedStorage.Remove(StorageIndex);
324
325 return EPreAnimatedStorageRequirement::None;
326 }
327 else
328 {
329 ensure(SourceRequirement == EPreAnimatedStorageRequirement::Transient);
330 const int32 NumTransients = TransientPreAnimatedStorage.Remove(StorageIndex);
331 if (NumTransients == 0)
332 {
333 PreAnimatedStorage[StorageIndex].bPersistent = true;
334 }
335 return EPreAnimatedStorageRequirement::Persistent;
336 }

407 void AssignPreAnimatedValue(FPreAnimatedStorageIndex StorageIndex, EPreAnimatedStorageRequirement StorageRequirement, StorageType&& InNewValue)
408 {
409 check(StorageIndex);
410
411 FCachedData& CachedData = PreAnimatedStorage[StorageIndex.Value];
412
413 if (StorageRequirement == EPreAnimatedStorageRequirement::Persistent)
414 {
415 ensure(!CachedData.bInitialized);
416 CachedData.Value = MoveTemp(InNewValue);
417 CachedData.bPersistent = true;
418 CachedData.bInitialized = true;
419 }
420 else if (StorageRequirement == EPreAnimatedStorageRequirement::Transient)
421 {
422 ensure(!CachedData.bInitialized || !TransientPreAnimatedStorage.Contains(StorageIndex));
423
424 // Assign the transient value
425 if (!CachedData.bInitialized)
426 {
427 CachedData.Value = MoveTemp(InNewValue);
428 CachedData.bInitialized = true;
429 }
430 else
431 {
432 TransientPreAnimatedStorage.Add(StorageIndex, MoveTemp(InNewValue));
433 }
434 }
435 }

440 bool IsStorageRequirementSatisfied(FPreAnimatedStorageIndex StorageIndex, EPreAnimatedStorageRequirement StorageRequirement) const
441 {
442 check(StorageIndex);
443
444 const FCachedData& CachedData = PreAnimatedStorage[StorageIndex.Value];
445
446 if (StorageRequirement == EPreAnimatedStorageRequirement::Persistent)
447 {
448 return CachedData.bInitialized;
449 }
450 else if (StorageRequirement == EPreAnimatedStorageRequirement::Transient)
451 {
452 return (CachedData.bInitialized && CachedData.bPersistent == false)
453 || TransientPreAnimatedStorage.Contains(StorageIndex);
454 }
455
456 return true;
457 }

825 template<typename... KeyArgs>
826 bool ShouldTrackCaptureSource(EPreAnimatedCaptureSourceTracking TrackingMode, KeyArgs&&... InKeyArgs)
827 {
828 switch (TrackingMode)
829 {
830 case EPreAnimatedCaptureSourceTracking::CacheIfTracked:
831 {
832 // If we're capturing global state we always track changes
833 if (this->ParentExtension->IsCapturingGlobalState())
834 {
835 return true;
836 }
837
838 // Only cache the value if an entry already exists, and if metadata already exists.
839 // (ie, something is actively animating this and wants restore state)
840 if (FPreAnimatedStateEntry Entry = FindEntry(InKeyArgs...))
841 {
842 return ensureMsgf(this->ParentExtension->MetaDataExists(Entry), TEXT("PreAnimatedStateEntry has allocated storage but no metadata exists to track it."));
843 }
844 return false;
845 }
846 break;
847
848 case EPreAnimatedCaptureSourceTracking::AlwaysCache:
849 {
850 // Always cache the source meta data. If there is no capture source this calls EnsureMetaData as above
851 return true;
852 }
853 break;
854
855 default:
856 ensureMsgf(false, TEXT("Unsupported tracking mode, no pre-animated state caching will occur!"));
857 return false;
858 }
859 }
860
865 void TrackCaptureSource(const FPreAnimatedStateEntry& Entry, EPreAnimatedCaptureSourceTracking TrackingMode)
866 {
867 switch (TrackingMode)
868 {
869 case EPreAnimatedCaptureSourceTracking::CacheIfTracked:
870 {
871 // If we're capturing global state we always track changes
872 if (this->ParentExtension->IsCapturingGlobalState())
873 {
874 this->ParentExtension->EnsureMetaData(Entry);
875 return;
876 }
877
878 // Only cache the value if tracking meta-data exists for this entry
879 // (ie, something is actively animating this and wants restore state)
880 ensureMsgf(Entry.IsValid() && this->ParentExtension->MetaDataExists(Entry), TEXT("PreAnimatedStateEntry has allocated storage but no metadata exists to track it."));
881 }
882 break;
883
884 case EPreAnimatedCaptureSourceTracking::AlwaysCache:
885 {
886 // Always cache the source meta data. If there is no capture source this calls EnsureMetaData as above
887 this->ParentExtension->AddSourceMetaData(Entry);
888 return;
889 }
890 break;
891
892 default:
893 ensureMsgf(false, TEXT("Unsupported tracking mode, no pre-animated state caching will occur!"));
894 }
