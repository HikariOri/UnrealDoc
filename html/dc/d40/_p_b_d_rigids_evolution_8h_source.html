502 template <bool bPersistent>
503 FORCEINLINE_DEBUGGABLE bool IsADisableCluster(TGeometryParticleHandleImp<FReal, 3, bPersistent>& Particle)
504 {
505 const TPBDRigidParticleHandleImp<FReal, 3, bPersistent>* AsRigid = Particle.CastToRigidParticle();
506 if (AsRigid && AsRigid->Disabled())
507 {
508 TPBDRigidClusteredParticleHandleImp<FReal, 3, bPersistent>* AsClustered = Particle.CastToClustered();
509
510 if (AsClustered)
511 {
512 // For clustered particles, they may appear disabled but they're being driven by an internal (solver-owned) cluster parent.
513 // If this is the case we let the spatial data update with those particles, otherwise skip.
514 // Alternatively, the particle may be being driven by a cluster union. Disabled children of a cluster union should not be added
515 // to the SQ.
516 // #BGTODO consider converting MDisabled into a bitfield for multiple disable types (Disabled, DisabledDriven, etc.)
517 if (FPBDRigidParticleHandle* ClusterParentBase = AsClustered->ClusterIds().Id)
518 {
519 if (Chaos::TPBDRigidClusteredParticleHandle<FReal, 3>*ClusterParent = ClusterParentBase->CastToClustered())
520 {
521 if (!ClusterParent->InternalCluster())
522 {
523 return true;
524 }
525
526 // We know we're an internal cluster now. If this was a GC internal proxy, we'd expect the
527 // parent's proxy to be the same as the input particle's. If this is not the case, we know
528 // we're in a cluster union.
529 if (ClusterParent->PhysicsProxy() != Particle.PhysicsProxy())
530 {
531 return true;
532 }
533 }
534 else
535 {
536 // There's probably no way we get here since if we're clustering the parent should always be a clustered.
537 ensure(false);
538 return true;
539 }
540 }
541 else
542 {
543 // Disabled cluster particle that doesn't have a parent cluster. MUST BE IGNORED.
544 return true;
545 }
546 }
547 else
548 {
549 // Disabled particles take no immediate part in sim or query so shouldn't be added to the acceleration
550 return true;
551 }
552 }
553 return false;
554 }
