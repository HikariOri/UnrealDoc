14// Key used to uniquely look up already decoded chunks of PCM data from wave proxies
15struct FDecodedSoundWaveChunkKey
16{
17 // The name of the wave proxy (useful for debugging)
18 FName WaveName;
19
20 // The guid of the wave is used for life-time independent cache key
21 FGuid WaveGuid;
22
23 // The chunk index of the decoded chunk, based on the current frame index and chunk size
24 uint32 ChunkIndex = INDEX_NONE;
25
26 // Prevents invalid cache hits in editor in the case of re-imported or other format changes
27#if WITH_EDITOR
28 uint32 ChunkRevision = 0;
29 uint32 NumChannels = 0;
30 uint32 SampleRate = 0;
31#endif
32
33 bool operator==(const FDecodedSoundWaveChunkKey& Other) const
34 {
35 return WaveName == Other.WaveName
36 && WaveGuid == Other.WaveGuid
37 && ChunkIndex == Other.ChunkIndex
38#if WITH_EDITOR
39 && ChunkRevision == Other.ChunkRevision
40 && NumChannels == Other.NumChannels
41 && SampleRate == Other.SampleRate
42#endif
43 ;
44 }
45
46 friend uint32 GetTypeHash(const FDecodedSoundWaveChunkKey& Key)
47 {
48 uint32 Hash = HashCombineFast(GetTypeHash(Key.WaveName), Key.ChunkIndex);
49 Hash = HashCombineFast(Hash, GetTypeHash(Key.WaveGuid));
50#if WITH_EDITOR
51 Hash = HashCombineFast(Hash, Key.ChunkRevision);
52 Hash = HashCombineFast(Hash, Key.NumChannels);
53 Hash = HashCombineFast(Hash, Key.SampleRate);
54#endif
55 return Hash;
56 }
