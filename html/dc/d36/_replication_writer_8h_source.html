78 // State
79 enum class EReplicatedObjectState : uint8
80 {
81 Invalid = 0,
82
83 // Special states
84 AttachmentToObjectNotInScope, // Special state for object index used for sending attachments to remote owned objects
85 HugeObject, // Special state for object index used for sending parts of huge object payloads
86 DebugObject, // Special state for object index used for sending parts of debug object payloads
87
88 // Normal states
89 PendingCreate, // Not yet created, no data in flight
90 WaitOnCreateConfirmation, // Waiting for confirmation from remote, we can send state data, but if we do we must also include creation info until the object is Created
91 Created, // Confirmed by remote, this is the normal replicating state
92 WaitOnFlush, // Pending flush, we are waiting for all in-flight state data to be acknowledged
93 PendingTearOff, // TearOff should be sent
94 SubObjectPendingDestroy, // SubObject destroy should be sent
95 CancelPendingDestroy, // Destroy was sent but object wants to start replicating again
96 PendingDestroy, // Object is set to be destroyed
97 WaitOnDestroyConfirmation, // Destroy has been sent, waiting for response from client
98 Destroyed, // Confirmed as Destroyed,
99 PermanentlyDestroyed, // DestructionInfo has been confirmed as received
100
101 Count
102 };
103
104 static_assert(UE::Net::GetBitsNeeded(EReplicatedObjectState::Count) <= 5, "EReplicatedObjectState must fit in 5 bits. See FReplicationInfo::State and FReplicationRecord::FRecordInfo::ReplicatedObjectState members.");

108 enum EFlushFlags : uint32
109 {
110 FlushFlags_None = 0U,
111 FlushFlags_FlushState = 1U << 0U, // Make sure that all current state data is acknowledged before we stop replicating the object
112 FlushFlags_FlushReliable = FlushFlags_FlushState << 1U, // Make sure that all enqueued Reliable RPCs are delivered before we stop replicating the object
113 FlushFlags_FlushTornOffSubObjects = FlushFlags_FlushReliable << 1U, // Make sure that we flush TearOff and replicated destroy properly
114 FlushFlags_All = FlushFlags_FlushState | FlushFlags_FlushReliable | FlushFlags_FlushTornOffSubObjects,
115 FlushFlags_Default = FlushFlags_FlushReliable,
116 };

118 // Bookkeeping info required for a replicated object
119 // Keep as small as possible since there is one per replicated object per connection
120 // Changemask can and will most likely be replaced by a smaller index into a pool to reduce overhead
121 struct FReplicationInfo
122 {
123 inline FReplicationInfo();
124
125 FChangeMaskStorageOrPointer ChangeMaskOrPtr; // Changemask storage or pointer to storage
126 union
127 {
128 uint64 Value;
129 struct
130 {
131 uint64 ChangeMaskBitCount : 16; // This is cached to avoid having to look it up in the protocol
132 uint64 State : 5; // Current state
133 uint64 HasDirtySubObjects : 1; // Set if this object has dirty subobjects
134 uint64 IsSubObject : 1; // Set if this object is a subobject
135 uint64 HasDirtyChangeMask : 1; // Set if the ChangeMask might be dirty, if not set the changemask should be zero!
136 uint64 HasAttachments : 1; // Set if there are attachments, such as RPCs, waiting to be sent
137 uint64 HasChangemaskFilter : 1; // Do we need to filter our changemask or not
138 uint64 IsDestructionInfo : 1; // If this is a destruction info
139 uint64 IsCreationConfirmed : 1; // We know that this object has been created on the receiving end
140 uint64 TearOff : 1; // This object should be torn off
141 uint64 SubObjectPendingDestroy : 1; // This object is a subobject that should be destroyed when we replicate owner
142 uint64 IsDeltaCompressionEnabled : 1; // Set to 1 if deltacompression is enabled for this object
143 uint64 LastAckedBaselineIndex : 2; // Last acknowledged baseline index which we can use for deltacompresion
144 uint64 PendingBaselineIndex : 2; // Baseline index pending acknowledgment from client
145 uint64 FlushFlags : 3; // Flags indicating what we are waiting for when flushing
146 uint64 HasDirtyConditionals : 1; // If this flag is set, we must update conditionals.
147 mutable uint64 HasCannotSendInfo : 1; // If this flag is set, the object has been prevented from being sent at least once.
148 uint64 IsDebugObject : 1; // If this flag is set, the object is a debugobject and is always sent using DebugObject attachments.
149 uint64 IsFilteredOutSubObjectPendingCreate : 1; // If this flag is set, the object is a subobject that was filtered out while sending pending create which requires some extra attention
150 uint64 IsFilteredOutSubObjectTearOff : 1; // If this flag is set, the object is a subobject confirmed as created that was filtered out while marked as tearoff which requires some extra attention
151 };
152 };
