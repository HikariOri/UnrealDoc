8// Bitfield identifying which category of motion trail a hierarchy provides.
9// Each trail hierarchy sets its category; the trail tool only updates/renders
10// hierarchies whose category intersects the user-visible mask.
11enum class ETrailCategory : uint8
12{
13 None = 0,
14 Transform = 1 << 0, // Component/actor 3D transform trails
15 ControlRig = 1 << 1, // Control rig bone/control trails
16 AnimMixer = 1 << 2, // Animation mixer root motion trails
17 All = Transform | ControlRig | AnimMixer
18};
19ENUM_CLASS_FLAGS(ETrailCategory);
