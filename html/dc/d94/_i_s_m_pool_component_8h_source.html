22struct FISMPoolInstanceGroups
23{
24 using FInstanceGroupId = int32;
25
27 struct FInstanceGroupRange
28 {
29 FInstanceGroupRange(int32 InStart, int32 InCount)
30 : Start(InStart)
31 , Count(InCount)
32 {
33 }
34
35 int32 Start = 0;
36 int32 Count = 0;
37 };
38
40 void Reset()
41 {
42 TotalInstanceCount = 0;
43 TotalFreeInstanceCount = 0;
44 GroupRanges.Empty();
45 FreeList.Empty();
46 }
47
49 bool IsEmpty() const
50 {
51 return GroupRanges.Num() == FreeList.Num();
52 }
53
55 int32 GetMaxInstanceIndex() const
56 {
57 return TotalInstanceCount;
58 }
59
61 FInstanceGroupId AddGroup(int32 Count)
62 {
63 // First check to see if we can recycle a group from the free list.
64 for (int32 Index = 0; Index < FreeList.Num(); ++Index)
65 {
66 const FInstanceGroupId GroupId = FreeList[Index];
67 // todo: Could also allow allocating a subrange if Count is less than the group range count.
68 if (Count == GroupRanges[GroupId].Count)
69 {
70 TotalFreeInstanceCount -= Count;
71 FreeList.RemoveAtSwap(Index, EAllowShrinking::No);
72 return GroupId;
73 }
74 }
75
76 // Create a new group.
77 const int32 StartIndex = TotalInstanceCount;
78 TotalInstanceCount += Count;
79 const FInstanceGroupId GroupId = GroupRanges.Add(FInstanceGroupRange(StartIndex, Count));
80 return GroupId;
81 }
82
84 void RemoveGroup(FInstanceGroupId GroupId)
85 {
86 // Remove the group by putting on free list for reuse.
87 // Actually removing it would require too much shuffling of the render instance index remapping.
88 TotalFreeInstanceCount += GroupRanges[GroupId].Count;
89 FreeList.Add(GroupId);
90 }
91
92 int32 TotalInstanceCount = 0;
93 int32 TotalFreeInstanceCount = 0;
94 TArray<FInstanceGroupRange> GroupRanges;
95 TArray<FInstanceGroupId> FreeList;
96};

165struct FISMPoolStaticMeshInstance
166{
167 TWeakObjectPtr<UStaticMesh> StaticMesh;
168 TArray<TWeakObjectPtr<UMaterialInterface>> MaterialsOverrides;
169 TArray<float> CustomPrimitiveData;
170 FISMPoolComponentDescription Desc;
171
172 bool operator==(const FISMPoolStaticMeshInstance& Other) const
173 {
174 if (!(Desc == Other.Desc))
175 {
176 return false;
177 }
178 if (!StaticMesh.HasSameIndexAndSerialNumber(Other.StaticMesh))
179 {
180 return false;
181 }
182 if (MaterialsOverrides.Num() != Other.MaterialsOverrides.Num())
183 {
184 return false;
185 }
186 for (int32 MatIndex = 0; MatIndex < MaterialsOverrides.Num(); MatIndex++)
187 {
188 if (!MaterialsOverrides[MatIndex].HasSameIndexAndSerialNumber(Other.MaterialsOverrides[MatIndex]))
189 {
190 return false;
191 }
192 }
193 if (CustomPrimitiveData.Num() != Other.CustomPrimitiveData.Num())
194 {
195 return false;
196 }
197 for (int32 DataIndex = 0; DataIndex < CustomPrimitiveData.Num(); DataIndex++)
198 {
199 if (CustomPrimitiveData[DataIndex] != Other.CustomPrimitiveData[DataIndex])
200 {
201 return false;
202 }
203 }
204
205 return true;
206 }
207};

209inline uint32 GetTypeHash(const FISMPoolStaticMeshInstance& MeshInstance)
210{
211 uint32 CombinedHash = GetTypeHash(MeshInstance.StaticMesh);
212 CombinedHash = HashCombineFast(CombinedHash, GetTypeHash(MeshInstance.MaterialsOverrides.Num()));
213 for (const TWeakObjectPtr<UMaterialInterface> Material: MeshInstance.MaterialsOverrides)
214 {
215 CombinedHash = HashCombineFast(CombinedHash, GetTypeHash(Material));
216 }
217 for (const float CustomFloat : MeshInstance.CustomPrimitiveData)
218 {
219 CombinedHash = HashCombineFast(CombinedHash, GetTypeHash(CustomFloat));
220 }
221 CombinedHash = HashCombineFast(CombinedHash, GetTypeHash(MeshInstance.Desc));
222 return CombinedHash;
223}
