251 Query.DependencyFlags.Reserve(sizeof...(Dependencies));
252
253 ( Query.DependencyTypes.Emplace(Dependencies::StaticClass()), ... );

255

259 template<typename... Dependencies>
260 template<typename Dependency>
261 Dependency& FCachedQueryContext<Dependencies...>::GetCachedMutableDependency()
262 {
263 // Don't allow a dependency registered as const to be found.
264 constexpr uint32 Index = Private::GetVarArgIndex<std::remove_const_t<Dependency>, Dependencies...>();
265 static_assert(Index < sizeof...(Dependencies), "Requested dependency isn't part of the query context cache.");
266 return *static_cast<Dependency*>(Description.CachedDependencies[Index].Get());
267 }
268
269 template<typename... Dependencies>
270 template<typename Dependency>
271 const Dependency& FCachedQueryContext<Dependencies...>::GetCachedDependency() const
272 {
273 // Allow access to dependencies registered with and without const.
274 constexpr uint32 Index = Private::GetVarArgIndex<Dependency, Dependencies...>();
275 if constexpr (Index < sizeof...(Dependencies))
276 {
277 return *static_cast<Dependency*>(Description.CachedDependencies[Index].Get());
278 }
279 else
280 {
281 constexpr uint32 ConstIndex = Private::GetVarArgIndex<std::add_const_t<Dependency>, Dependencies...>();
282 static_assert(ConstIndex < sizeof...(Dependencies), "Requested dependency isn't part of the query context cache.");
283 return *static_cast<Dependency*>(Description.CachedDependencies[ConstIndex].Get());
284 }

388 template<SourceQueryContext SourceContext, bool ValidateColumns, typename... ColumnTypes>
389 struct FFunctionColumnInfo
390 {
391 std::tuple<BaseColumnType<ColumnTypes>*...> Columns;
392 static constexpr bool bArePointerColumns = AreAllColumnsPointers<ColumnTypes...>();
393
394 template<typename Column>
395 bool CheckValidity(const FQueryDescription& Description)
396 {
397 const UScriptStruct* ColumnType = Column::StaticStruct();
398 if (const TWeakObjectPtr<const UScriptStruct>* Match = Description.SelectionTypes.FindByPredicate(
399 [ColumnType](const TWeakObjectPtr<const UScriptStruct>& Entry)
400 {
401 return Entry == ColumnType;
402 }); Match != nullptr)
403 {
404 size_t Index = Match - Description.SelectionTypes.GetData();
405 if (Description.SelectionAccessTypes[Index] == EQueryAccessType::ReadOnly)
406 {
407 if (std::is_const_v<Column>)
408 {
409 return true;
410 }
411 else
412 {
413 checkf(false,
414 TEXT("Provided query callback couldn't be bound because target read-only column of type '%s' is bound to a mutable reference."),
415 *ColumnType->GetName());
416 return false;
417 }
418 }
419 else
420 {
421 return true;
422 }
423 }
424 else
425 {
426 checkf(false,
427 TEXT("Provided query callback couldn't be bound because target query doesn't contain a column of type '%s'."),
428 *ColumnType->GetName());
429 return false;
430 }
431 }
432
433 FFunctionColumnInfo(const FQueryDescription& Description, SourceContext& Context)
434 {
435 if constexpr (ValidateColumns)
436 {
437 if (!(CheckValidity<BaseColumnType<ColumnTypes>>(Description) && ...))
438 {
439 return;
440 }
441 }
442
443 char* ColumnAddresses[sizeof...(ColumnTypes)];
444 Context.GetColumnsUnguarded(sizeof...(ColumnTypes), ColumnAddresses,
445 Description.SelectionTypes.GetData(), Description.SelectionAccessTypes.GetData());
446 std::apply([ColumnAddresses](auto&&... Column)
447 {
448 int Index = 0;
449 ((Column = reinterpret_cast<BaseColumnType<ColumnTypes>*>(ColumnAddresses[Index++])), ...);
450 }, Columns);
451 }
