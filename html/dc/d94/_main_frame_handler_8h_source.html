72 bool CanCloseEditor()
73 {
74 // We don't want to close the editor if we are recreating the mainframe module
75 IMainFrameModule& MainFrameModule = FModuleManager::LoadModuleChecked<IMainFrameModule>(TEXT("MainFrame"));
76 if (MainFrameModule.IsRecreatingDefaultMainFrame())
77 {
78 return false;
79 }
80
81 if ( FSlateApplication::IsInitialized() && !FSlateApplication::Get().IsNormalExecution())
82 {
83 // DEBUGGER EXIT PATH
84
85 // The debugger is running we cannot actually close now.
86 // We will stop the debugger and enque a request to close the editor on the next frame.
87
88 // Stop debugging.
89 FSlateApplication::Get().LeaveDebuggingMode();
90
91 // Defer the call RequestCloseEditor() till next tick.
92 GEngine->DeferredCommands.AddUnique( TEXT("CLOSE_SLATE_MAINFRAME") );
93
94 // Cannot exit right now.
95 return false;
96 }
97 else if (UE::IsSavingPackage() || IsGarbageCollecting() || IsLoading() || GIsSlowTask)
98 {
99 // Saving / Loading / GC / Slow Task path. It is all unsafe to do that here.
100
101 // We're currently saving or garbage collecting and can't close the editor just yet.
102 // We will have to wait and try to request to close the editor on the next frame.
103
104 // Defer the call RequestCloseEditor() till next tick.
105 GEngine->DeferredCommands.AddUnique( TEXT("CLOSE_SLATE_MAINFRAME") );
106
107 // Cannot exit right now.
108 return false;
109 }
110 else
111 {
112 // NORMAL EXIT PATH
113
114 if (UInterchangeManager::GetInterchangeManager().WarnIfInterchangeIsActive()) { return false; }
115
116 // Unattented mode can always exit.
117 if (FApp::IsUnattended())
118 {
119 return true;
120 }
121
122 // We can't close if lightmass is currently building
123 if (GUnrealEd->WarnIfLightingBuildIsCurrentlyRunning()) {return false;}
124
125 // We can't close if a modal dialog is on screen. (Clicking on the Editor window top right 'X' already prevent that, but closing from the Windows task bar preview does not)
126 bool bOkToExit = FSlateApplication::IsInitialized() ? !FSlateApplication::Get().GetActiveModalWindow().IsValid() : true;
127
128 // Check if level Mode is open this does PostEditMove processing on actors when it closes so need to do this first before save dialog
129 if( GLevelEditorModeTools().IsModeActive( FBuiltinEditorModes::EM_Level ) ||
130 GLevelEditorModeTools().IsModeActive( FBuiltinEditorModes::EM_StreamingLevel) )
131 {
132 GLevelEditorModeTools().DeactivateMode(FBuiltinEditorModes::EM_Level);
133 GLevelEditorModeTools().DeactivateMode(FBuiltinEditorModes::EM_StreamingLevel);
134 bOkToExit = false;
135 }
136
137 // Can we close all the major tabs? They have sub-editors in them that might want to not close
138 {
139 // Ignore the LevelEditor tab; it invoked this function in the first place.
140 TSet< TSharedRef<SDockTab> > TabsToIgnore;
141 if ( MainTabPtr.IsValid() )
142 {
143 TabsToIgnore.Add( MainTabPtr.Pin().ToSharedRef() );
144 }
145
146 bOkToExit = bOkToExit && FGlobalTabmanager::Get()->CanCloseManager(TabsToIgnore);
147 }
148
149 // Allow Plugins and other systems to prevent close
150 bOkToExit = bOkToExit && MainFrameModule.ExecuteCanCloseEditorDelegates();
151
152 // Prompt for save and quit only if we did not launch a gameless rocket exe or are in demo mode or we are asking for a close to recreate the Default Main Frame
153 if ( FApp::HasProjectName() && !GIsDemoMode && !MainFrameModule.IsRecreatingDefaultMainFrame())
154 {
155 // Prompt the user to save packages/maps.
156 bool bHadPackagesToSave = false;
157 {
158 const bool bPromptUserToSave = true;
159 const bool bSaveMapPackages = true;
160 const bool bSaveContentPackages = true;
161 const bool bFastSave = false;
162 const bool bNotifyNoPackagesSaved = false;
163 const bool bCanBeDeclined = true;
164 bOkToExit = bOkToExit && FEditorFileUtils::SaveDirtyPackages(bPromptUserToSave, bSaveMapPackages, bSaveContentPackages, bFastSave, bNotifyNoPackagesSaved, bCanBeDeclined, &bHadPackagesToSave);
165 }
166
167 // If there were packages to save, or switching project, then the user already had a chance to bail out of exiting.
168 const bool bPromptedForExit = bHadPackagesToSave || !FUnrealEdMisc::Get().GetPendingProjectName().IsEmpty();
169 if ( !bOkToExit && !bPromptedForExit )
170 {
171 FUnrealEdMisc::Get().ClearPendingProjectName();
172 FGlobalTabmanager::Get()->SetCanSavePersistentLayouts(true);
173 FUnrealEdMisc::Get().ForceDeletePreferences(false);
174 FUnrealEdMisc::Get().ClearConfigRestoreFilenames();
175 }
176 else if (bOkToExit && !bPromptedForExit && GetDefault<UEditorPerProjectUserSettings>()->bConfirmEditorClose)
177 {
178 const EAppReturnType::Type Response = FMessageDialog::Open(
179 EAppMsgCategory::Info,
180 EAppMsgType::YesNo,
181 LOCTEXT("ConfirmClose", "Are you sure you want to close the Unreal Editor?"),
182 LOCTEXT("ConfirmCloseTitle", "Close Editor")
183 );
184
185 bOkToExit = (Response == EAppReturnType::Yes);
186 }
187 }
188
189 return bOkToExit;
190 }

213 void OnMainFrameGenerated( const TSharedPtr<SDockTab>& MainTab, const TSharedRef<SWindow>& InRootWindow )
214 {
215 TSharedRef<FGlobalTabmanager> GlobalTabManager = FGlobalTabmanager::Get();
216
217 if (MainTab.IsValid())
218 {
219 GlobalTabManager->SetMainTab(MainTab.ToSharedRef());
220 }
221
222 // Persistent layouts should get stored using the specified method.
223 GlobalTabManager->SetOnPersistLayout(FTabManager::FOnPersistLayout::CreateRaw(this, &FMainFrameHandler::HandleTabManagerPersistLayout));
224
225 IMainFrameModule& MainFrameModule = FModuleManager::LoadModuleChecked<IMainFrameModule>(TEXT("MainFrame"));
226
227 const bool bIncludeGameName = true;
228 GlobalTabManager->SetApplicationTitle( MainFrameModule.GetApplicationTitle( bIncludeGameName ) );
229
230 InRootWindow->SetRequestDestroyWindowOverride( FRequestDestroyWindowOverride::CreateRaw( this, &FMainFrameHandler::CloseRootWindowOverride ) );
231
232 MainTabPtr = MainTab;
233 RootWindowPtr = InRootWindow;
234
235 EnableTabClosedDelegate();
236 }
