792 void Relocate(FRelocationStats& Stats, FMemoryChunk* Dest, int64 DestOffset, const void* Source, int64 Size, void* UserPayload)
793 {
794 MemoryTrace_ReallocFree((uint64)Source);
795 MemoryTrace_ReallocAlloc((uint64)Dest->Base, Size, 4);
796 LLM_IF_ENABLED(FLowLevelMemTracker::Get().OnLowLevelAllocMoved(ELLMTracker::Default, Dest->Base, Source));
797
798 uint8* DestAddr = Dest->Base + DestOffset;
799 int64 MemDistance = (int64)(DestAddr) - (int64)(Source);
800 int64 AbsDistance = FMath::Abs(MemDistance);
801 bool bOverlappedMove = AbsDistance < Size;
802
803 if (!bBenchmarkMode)
804 {
805#if VALIDATE_MEMORY_PROTECTION
806 BlocksToProtect.Emplace(DestAddr, UserPayload, Size, CurrentSyncIndex);
807 BlocksToProtect.Emplace(Source, UserPayload, Size, CurrentSyncIndex);
808#endif
809 PlatformRelocate(DestAddr, Source, Size, UserPayload);
810 }
811 int64 RelocateSize = bOverlappedMove ? (Size * Settings.MaxDefragRelocations) : Size;
812 Dest->UserPayload = UserPayload;
813 Stats.NumBytesRelocated += RelocateSize;
814 Stats.NumRelocations++;
815 }

918 void Split(FMemoryChunk* BaseChunk, int64 FirstSize)
919 {
920 check(BaseChunk);
921 check(FirstSize < BaseChunk->Size);
922 check(FirstSize > 0);
923
924 // Don't make any assumptions on the following chunk. Because Reallocate() will make the 1st chunk free
925 // and the 2nd chunk used, so it's ok to have the following chunk free. Note, this only happens when Reallocate()
926 // is splitting the very first chunk in the pool.
927 // Don't make any assumptions for the previous chunk either...
928 // check( !BaseChunk->NextChunk || !BaseChunk->NextChunk->bIsAvailable );
929 // check( !BaseChunk->PreviousChunk || !BaseChunk->PreviousChunk->bIsAvailable || !BaseChunk->bIsAvailable );
930
931 // Calculate size of second chunk...
932 int64 SecondSize = BaseChunk->Size - FirstSize;
933 // ... and create it.
934
935 //todo: fix stats
936 //ensureMsgf(BaseChunk->Stat.IsNone(), TEXT("Free chunk has stat"));
937 FMemoryChunk* NewFreeChunk = new FMemoryChunk(BaseChunk->Base + FirstSize, SecondSize, *this, BaseChunk, TStatId());
938
939 // Keep the original sync index for the new chunk if necessary.
940 if (BaseChunk->IsRelocating() && BaseChunk->SyncSize > FirstSize)
941 {
942 int64 SecondSyncSize = BaseChunk->SyncSize - FirstSize;
943 NewFreeChunk->SetSyncIndex(BaseChunk->SyncIndex, SecondSyncSize);
944 }
945 BaseChunk->SetSyncIndex(BaseChunk->SyncIndex, FMath::Min((int64)FirstSize, BaseChunk->SyncSize));
946
947 // Resize base chunk.
948 BaseChunk->Size = FirstSize;
949 } //-V773
