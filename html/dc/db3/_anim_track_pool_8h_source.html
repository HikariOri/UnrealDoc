102 int32 AllocateTrack(const DataType& Data)
103 {
104 const int32 Index = ActiveBits.FindAndSetFirstZeroBit();
105 NumActiveTracks++;
106
107 if (Index != INDEX_NONE)
108 {
109 SetDirty(Index);
110 Datas[Index] = Data;
111 return Index;
112 }
113
114 Datas.Emplace(Data);
115 ActiveBits.Add(true);
116 DirtyBits.Add(true);
117 NumDirtyTracks++;
118
119 return Datas.Num() - 1;
120 }

122 int32 AllocateTrackAt(int32 Index, const DataType& Data)
123 {
124 if (Index < 0)
125 {
126 return INDEX_NONE;
127 }
128
129 if (Index >= Datas.Num())
130 {
131 const int32 OldNum = Datas.Num();
132 const int32 NewNum = Index + 1;
133 Datas.SetNum(NewNum);
134 ActiveBits.Add(false, NewNum - OldNum);
135 DirtyBits.Add(false, NewNum - OldNum);
136 }
137
138 if (ActiveBits[Index])
139 {
140 return INDEX_NONE;
141 }
142
143 ActiveBits[Index] = true;
144 NumActiveTracks++;
145 SetDirty(Index);
146 Datas[Index] = Data;
147 return Index;
148 }

196 template <typename ResetFunctionType = FDefaultReset>
197 FPatch Finalize(ResetFunctionType ResetFunction = FDefaultReset())
198 {
199 FPatch Patch;
200 Patch.NumTracks = Datas.Num();
201 Patch.NumActiveTracks = NumActiveTracks;
202
203 // Early-out when nothing is dirty: callers still get authoritative metadata
204 // (NumTracks / NumActiveTracks) without the cost of empty Reserve allocations
205 // or a SetAllClean memset.
206 if (NumDirtyTracks == 0)
207 {
208 return Patch;
209 }
210
211 Patch.DeactivateOps.Reserve(NumDirtyTracks);
212 Patch.ActivateOps.Reserve(NumDirtyTracks);
213
214 for (TConstSetBitIterator<> BitIt(DirtyBits); BitIt; ++BitIt)
215 {
216 const uint32 Index = BitIt.GetIndex();
217 if (ActiveBits[Index])
218 {
219 Patch.ActivateOps.Emplace(Index, Datas[Index]);
220 ResetFunction(Datas[Index]);
221 }
222 else
223 {
224 Patch.DeactivateOps.Emplace(Index);
225 }
226 }
227 SetAllClean();
228
229 return Patch;
230 }

232 template <typename SourceTrackData, typename AssignFunctionType, typename DeactivateFunctionType = FDefaultDeactivate>
233 void Patch(
234 typename TAnimTrackPool<SourceTrackData>::FPatch&& Patch,
235 AssignFunctionType AssignFunction,
236 DeactivateFunctionType DeactivateFunction = FDefaultDeactivate{})
237 {
238 using FSourcePatch = typename TAnimTrackPool<SourceTrackData>::FPatch;
239
240 Datas.SetNum(Patch.NumTracks);
241 ActiveBits.SetNum(Patch.NumTracks, false);
242 DirtyBits.SetNum(Patch.NumTracks, false);
243 NumActiveTracks = Patch.NumActiveTracks;
244
245 for (typename FSourcePatch::FDeactivateOp Op : Patch.DeactivateOps)
246 {
247 DeactivateFunction(Datas[Op.Index], int32(Op.Index));
248 ActiveBits[Op.Index] = false;
249 SetDirty(Op.Index);
250 }
251
252 for (const typename FSourcePatch::FActivateOp& Op : Patch.ActivateOps)
253 {
254 auto Bit = ActiveBits[Op.Index];
255 AssignFunction(Datas[Op.Index], Op.Data, Bit, int32(Op.Index));
256 SetDirty(Op.Index);
257 Bit = true;
258 }
259 }
