86FIoHttpRange& FIoHttpRange::ExpandInline(const FIoHttpRange& Other)
87{
88 if (IsValid())
89 {
90 Min = FMath::Min(Min, Other.Min);
91 Max = FMath::Max(Max, Other.Max);
92 }
93 else
94 {
95 Min = Other.Min;
96 Max = Other.Max;
97 }
98
99 return *this;
100}

102FIoHttpRange FIoHttpRange::Expand(const FIoHttpRange& Other) const
103{
104 FIoHttpRange Expanded(*this);
105 Expanded.ExpandInline(Other);
106 return Expanded;
107}

109FIoOffsetAndLength FIoHttpRange::ToOffsetAndLength() const
110{
111 return IsValid() ? FIoOffsetAndLength(Min, Max - Min) : FIoOffsetAndLength(0, 0);

114FIoHttpRange FIoHttpRange::FromOffsetAndLength(const FIoOffsetAndLength& OffsetAndLength)
115{
116 const uint64 End = OffsetAndLength.GetOffset() + OffsetAndLength.GetLength();
117
118 if (OffsetAndLength.GetOffset() > MAX_uint32 || End > MAX_uint32)
119 {
120 return FIoHttpRange();
121 }
122
123 return FIoHttpRange(IntCastChecked<uint32>(OffsetAndLength.GetOffset()), IntCastChecked<uint32>(End));
124}
