140 template<typename TPayloadType, typename T, int d>
141 void AddToSpatialAcceleration(TArrayView<const FPhysicsObjectHandle> InObjects, ISpatialAcceleration<TPayloadType, T, d>* SpatialAcceleration)
142 {
143 if (!SpatialAcceleration)
144 {
145 return;
146 }
147
148 for (const FConstPhysicsObjectHandle Handle : InObjects)
149 {
150 const FBox WorldBounds = this->GetWorldBounds({ &Handle, 1 });
151 const FAABB3 ChaosWorldBounds{ WorldBounds.Min, WorldBounds.Max };
152 FAccelerationStructureHandle AccelerationHandle = this->CreateAccelerationStructureHandle(Handle);
153 SpatialAcceleration->UpdateElementIn(AccelerationHandle, ChaosWorldBounds, true, this->GetSpatialIndex(Handle));
154 }
155 }

157 template<typename TPayloadType, typename T, int d>
158 void RemoveFromSpatialAcceleration(TArrayView<const FPhysicsObjectHandle> InObjects, ISpatialAcceleration<TPayloadType, T, d>* SpatialAcceleration)
159 {
160 if (!SpatialAcceleration)
161 {
162 return;
163 }
164
165 for (const FConstPhysicsObjectHandle Handle : InObjects)
166 {
167 FAccelerationStructureHandle AccelerationHandle = this->CreateAccelerationStructureHandle(Handle);
168 SpatialAcceleration->RemoveElementFrom(AccelerationHandle, this->GetSpatialIndex(Handle));
169 }
