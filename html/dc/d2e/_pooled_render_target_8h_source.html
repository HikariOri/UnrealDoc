208 bool Compare(const FPooledRenderTargetDesc& rhs, bool bExact) const
209 {
210 auto LhsFlags = Flags;
211 auto RhsFlags = rhs.Flags;
212
213 if (!bExact || !FPlatformMemory::SupportsFastVRAMMemory())
214 {
215 LhsFlags &= (~TexCreate_FastVRAM);
216 RhsFlags &= (~TexCreate_FastVRAM);
217 }
218
219 return ClearValue == rhs.ClearValue
220 && LhsFlags == RhsFlags
221 && Format == rhs.Format
222 && UAVFormat == rhs.UAVFormat
223 && Extent == rhs.Extent
224 && Depth == rhs.Depth
225 && ArraySize == rhs.ArraySize
226 && NumMips == rhs.NumMips
227 && NumSamples == rhs.NumSamples
228 && PackedBits == rhs.PackedBits
229 && FastVRAMPercentage == rhs.FastVRAMPercentage
230 && AliasableFormats == rhs.AliasableFormats;
231 }

254 bool IsValid() const
255 {
256 if(NumSamples != 1)
257 {
258 if(NumSamples < 1 || NumSamples > 8)
259 {
260 // D3D11 limitations
261 return false;
262 }
263

266 return false;

268 }
269
270 return Extent.X != 0 && NumMips != 0 && NumSamples >=1 && NumSamples <=16 && Format != PF_Unknown
271 && (EnumHasAnyFlags(Flags, TexCreate_UAV) || GMaxRHIFeatureLevel >= ERHIFeatureLevel::SM5 || GMaxRHIFeatureLevel == ERHIFeatureLevel::ES3_1);

283 FString GenerateInfoString() const
284 {
285 const TCHAR* FormatString = GetPixelFormatString(Format);
286
287 FString FlagsString;
288
289 ETextureCreateFlags LocalFlags = Flags;
290
291 if(EnumHasAnyFlags(LocalFlags, TexCreate_RenderTargetable))
292 {
293 FlagsString += TEXT(" RT");
294 }
295 if(EnumHasAnyFlags(LocalFlags, TexCreate_SRGB))
296 {
297 FlagsString += TEXT(" sRGB");
298 }
299 if(NumSamples > 1)
300 {
301 FlagsString += FString::Printf(TEXT(" %dxMSAA"), NumSamples);
302 }
303 if(EnumHasAnyFlags(LocalFlags, TexCreate_UAV))
304 {
305 FlagsString += TEXT(" UAV");
306 }
307
308 if(EnumHasAnyFlags(LocalFlags, TexCreate_FastVRAM))
309 {
310 FlagsString += TEXT(" VRam");
311 }
312
313 FString ArrayString;
314
315 if(IsArray())
316 {
317 ArrayString = FString::Printf(TEXT("[%d]"), ArraySize);
318 }
319
320 if(Is2DTexture())
321 {
322 return FString::Printf(TEXT("(2D%s %dx%d %s%s)"), *ArrayString, Extent.X, Extent.Y, FormatString, *FlagsString);
323 }
324 else if(Is3DTexture())
325 {
326 return FString::Printf(TEXT("(3D%s %dx%dx%d %s%s)"), *ArrayString, Extent.X, Extent.Y, Depth, FormatString, *FlagsString);
327 }
328 else if(IsCubemap())
329 {
330 return FString::Printf(TEXT("(Cube%s %d %s%s)"), *ArrayString, Extent.X, FormatString, *FlagsString);
331 }
332 else
333 {
334 return TEXT("(INVALID)");
335 }
