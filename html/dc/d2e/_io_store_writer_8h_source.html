155 /*
156 * Used by IIoStoreWriter to check and see if there's a reference chunk that matches the data that
157 * IoStoreWriter wants to compress and write. Validity checks must be synchronous - if a chunk can't be
158 * used for some reason (no matching chunk exists or otherwise), this function must return false and not
159 * call InCompletionCallback.
160 *
161 * Once a matching chunk is found, it is read from the source iostore container asynchronously, and upon
162 * completion InCompletionCallback is called with the raw output from FIoStoreReader::ReadCompressed (i.e.
163 * FIoStoreCompressedReadResult). Failures once the async read process has started are currently fatal due to
164 * difficulties in rekicking a read.
165 *
166 * For the moment, changes in compression method are allowed.
167 *
168 * RetrieveChunk is not currently thread safe and must be called from a single thread.
169 *
170 * Chunks provided *MUST* decompress to bits that hash to the exact value provided in InChunkKey (i.e. be exactly the same bits),
171 * and also be the same number of blocks (i.e. same CompressionBlockSize)
172 */
173 virtual UE::Tasks::FTask RetrieveChunk(const FIoContainerId& InContainerId, const FIoHash& InChunkHash, const FIoChunkId& InChunkId, TUniqueFunction<void(TIoStatusOr<FIoStoreCompressedReadResult>)> InCompletionCallback) = 0;
174
175 /*
176 * Quick synchronous existence check that returns the number of blocks for the chunk. This is used to set up
177 * the necessary structures without needing to read the source data for the chunk. This might be called from
178 * multiple threads as it has to happen after we have the source hash computed.
179 */
180 virtual bool ChunkExists(const FIoContainerId& InContainerId, const FIoHash& InChunkHash, const FIoChunkId& InChunkId, int32& OutNumChunkBlocks) = 0;
181
182 /*
183 * Returns the compression block size that was used to break up the IoChunks in the source containers. If this is different than what we want,
184 * then none of the chunks will ever match. Knowing this up front allows us to only match on hash
185 */
186 virtual uint32 GetCompressionBlockSize() const = 0;
187
188 /*
189 * Called by an iostore writer implementation to notify the ref cache it's been added
190 */
191 virtual void NotifyAddedToWriter(const FIoContainerId& InContainerId, const FString& InContainerName) = 0;
192
193 /*
194 * Returns true if ContainerId was found and OutContainerFilePath is set.
195 */
196 virtual bool GetContainerFilePath(const FIoContainerId& InContainerId, FString& OutContainerFilePath) = 0;
197};
