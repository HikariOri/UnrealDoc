12struct FAuthTicket
13{
14public:
15 FAuthTicket() = default;
16 FAuthTicket(const FString& InUsername, const FString& InPassword) : Username(InUsername), Password(InPassword) {}
17 FAuthTicket(FStringView TicketString)
18 {
19 int32 ChopIndex;
20 if(TicketString.FindChar(':', ChopIndex))
21 {
22 Username = TicketString.Left(ChopIndex);
23 Password = TicketString.RightChop(ChopIndex + 1);
24 }
25 }
26 virtual ~FAuthTicket() {}
27
28 virtual FString ToString() const
29 {
30 return TEXT("Basic ") + FBase64::Encode(Username + TEXT(":") + Password);
31 }
32
33 bool IsValid() const
34 {
35 return !Username.IsEmpty() && !Password.IsEmpty();
36 }
37 FString Username;
38private:
39 FString Password;
40};
