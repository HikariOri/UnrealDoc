108 static bool IsNumberStringInteger(const CharType* Str)
109 {
110 if (*Str == '-' || *Str == '+')
111 {
112 Str++;
113 }
114
115 while (*Str != '\0')
116 {
117 if (*Str == '.')
118 {
119 return false;
120 }
121 else if (!FChar::IsDigit(*Str))
122 {
123 return false;
124 }
125
126 ++Str;
127 }
128
129 return true;
130 }

231 {
232 if(Element->Value.IsType<FNestedVariantJson::FArrayRef>())
233 {
234 if (Element->bHasBeenProcessed)
235 {
236 Writer.WriteArrayEnd();
237 }
238 else
239 {

241 ElementStack.Push(Element);
242
243 if (bWriteValueOnly)

245 Writer.WriteArrayStart();

250 }

253

256 ElementStack.Push(MakeShared<FElement>((*Values)[Index]));

267 static bool SerializeIfObject(TArray<TSharedRef<FElement>>& ElementStack, TSharedRef<FElement>& Element, TJsonWriter<CharType, PrintPolicy>& Writer, bool bWriteValueOnly)
268 {
269 if (Element->Value.IsType<FNestedVariantJson::FMapRef>())
270 {
271 if (Element->bHasBeenProcessed)
272 {
273 Writer.WriteObjectEnd();
274 }
275 else
276 {
277 Element->bHasBeenProcessed = true;
278 ElementStack.Push(Element);
279
280 if (bWriteValueOnly)
281 {
282 Writer.WriteObjectStart();
283 }
284 else
285 {
286 Writer.WriteObjectStart(Element->Identifier);
287 }
288
289 TArray<FString> Keys;
290 TArray<FValue> Values;
291 FMapOfValues ElementMap = StaticCastSharedPtr<FNestedVariantJson::FMap>(Element->Value.Get<FNestedVariantJson::FMapRef>().ToSharedPtr());
292 for (const FNestedVariantJson::FMap::ElementType& Entry : *ElementMap)
293 {
294 Keys.Add(Entry.Key);
295
296 Values.Add(Entry.Value);
297 }
298
299 check(Keys.Num() == Values.Num());
300
301 for (int Index = Values.Num() - 1; Index >= 0; --Index)
302 {
303 ElementStack.Push(MakeShared<FElement>(Keys[Index], Values[Index]));
304 }
305 }
306
307 return true;
308 }
309
310 return false;
