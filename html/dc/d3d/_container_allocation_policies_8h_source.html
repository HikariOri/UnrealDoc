19// Array slack tracking is a debug feature to track unused space in heap allocated TArray (and TScriptArray) structures. This feature increases heap
20// memory usage, and costs perf, so it is usually disabled by default. Only works in builds where LLM is compiled in, but doesn't require -llm to be
21// active. Note that cooks will run quite slow with tracking enabled due to involving significantly more allocations than the engine, so be careful
22// about leaving this enabled when kicking off a cook (including !WITH_EDITOR ensures it will only be enabled for client builds). For more details,
23// see additional comments in LowLevelMemTracker.cpp.

35// For detailed tracking of array slack waste, we need to add a header to heap allocations. It's impossible to keep track of the TArray structure itself,
36// because it can be inside other structures, and potentially moved around in unsafe ways, while the heap allocation the TArray points to is invariant.
37// The heap allocations also have an advantage in that the contents are copied on Move or Realloc, simplifying tracking.
38struct FArraySlackTrackingHeader
39{
40 FArraySlackTrackingHeader* Next; // Linked list of tracked items
41 FArraySlackTrackingHeader** Prev;
42 uint16 AllocOffset; // Offset below the header to the start of the actual allocation, to account for alignment padding
43 uint8 Tag;
44 int8 NumStackFrames;
45 uint32 FirstAllocFrame; // Frame where array allocation first occurred
46 uint32 ReallocCount; // Number of times realloc happened
47 uint32 ArrayPeak; // Peak observed ArrayNum
48 uint64 ElemSize;
49
50 // Note that we initially set the slack tracking ArrayNum to INDEX_NONE. The container allocator is used by both arrays and
51 // other containers (Set / Map / Hash), and we don't know it's actually an array until "UpdateNumUsed" is called on it.
52 int64 ArrayNum;
53 int64 ArrayMax;
54 uint64 StackFrames[9];
55
56 CORE_API void AddAllocation();
57 CORE_API void RemoveAllocation();
58 CORE_API void UpdateNumUsed(int64 NewNumUsed);
59 CORE_API FORCENOINLINE static void* Realloc(void* Ptr, int64 Count, uint64 ElemSize, int32 Alignment);

61 static void Free(void* Ptr)
62 {
63 if (Ptr)
64 {
65 FArraySlackTrackingHeader* TrackingHeader = (FArraySlackTrackingHeader*)((uint8*)Ptr - sizeof(FArraySlackTrackingHeader));
66 TrackingHeader->RemoveAllocation();
67
68 Ptr = (uint8*)TrackingHeader - TrackingHeader->AllocOffset;
69
70 FMemory::Free(Ptr);
71 }
72 }

75 {
76 if (Ptr)
77 {
78 FArraySlackTrackingHeader* TrackingHeader = (FArraySlackTrackingHeader*)((uint8*)Ptr - sizeof(FArraySlackTrackingHeader));
79
80 TrackingHeader->UpdateNumUsed(NewNumUsed);
81 }
82 }

84 static UE_FORCEINLINE_HINT void DisableTracking(void* Ptr)
85 {
86 if (Ptr)
87 {
88 FArraySlackTrackingHeader* TrackingHeader = (FArraySlackTrackingHeader*)((uint8*)Ptr - sizeof(FArraySlackTrackingHeader));
89
90 TrackingHeader->RemoveAllocation();
91
92 // When disabling tracking, we need to also reset ArrayNum, as it's used internally as a flag specifying whether
93 // the allocation is currently tracked. We don't reset this inside RemoveAllocation, because ArrayNum needs to
94 // persist during realloc of tracked allocations, where RemoveAllocation is called, followed by AddAllocation.
95 TrackingHeader->ArrayNum = INDEX_NONE;
96 }
97 }

100 {
101 return (ArrayMax - ArrayNum) * ElemSize;

139template <typename SizeType>
140UE_FORCEINLINE_HINT SizeType DefaultCalculateSlackShrink(SizeType NewMax, SizeType CurrentMax, SIZE_T BytesPerElement, bool bAllowQuantize, uint32 Alignment = DEFAULT_ALIGNMENT)
141{
142 SizeType Retval;
143 checkSlow(NewMax < CurrentMax);
144
145 // If the container has too much slack, shrink it to exactly fit the number of elements.
146 const SizeType CurrentSlackElements = CurrentMax - NewMax;
147 const SIZE_T CurrentSlackBytes = (CurrentMax - NewMax)*BytesPerElement;
148 const bool bTooManySlackBytes = CurrentSlackBytes >= 16384;
149 const bool bTooManySlackElements = 3 * NewMax < 2 * CurrentMax;
150 if ((bTooManySlackBytes || bTooManySlackElements) && (CurrentSlackElements > 64 || !NewMax)) // hard coded 64 :-(
151 {
152 Retval = NewMax;
153 if (Retval > 0)
154 {
155 if (bAllowQuantize)
156 {
157 Retval = (SizeType)(FMemory::QuantizeSize(Retval * BytesPerElement, Alignment) / BytesPerElement);
158 }
159 }
160 }
161 else
162 {
163 Retval = CurrentMax;
164 }
165
166 return Retval;
167}

169template <typename SizeType>
170UE_FORCEINLINE_HINT SizeType DefaultCalculateSlackGrow(SizeType NewMax, SizeType CurrentMax, SIZE_T BytesPerElement, bool bAllowQuantize, uint32 Alignment = DEFAULT_ALIGNMENT)
171{
172#if !defined(AGGRESSIVE_MEMORY_SAVING)
173 #error "AGGRESSIVE_MEMORY_SAVING must be defined"
174#endif
175#if AGGRESSIVE_MEMORY_SAVING
176 const SIZE_T FirstGrow = 1;
177 const SIZE_T ConstantGrow = 0;
178#else
179 const SIZE_T FirstGrow = 4;
180 const SIZE_T ConstantGrow = 16;
181#endif
182
183 SizeType Retval;
184 checkSlow(NewMax > CurrentMax && NewMax > 0);
185
186 SIZE_T Grow = FirstGrow; // this is the amount for the first alloc
187
188#if CONTAINER_INITIAL_ALLOC_ZERO_SLACK
189 if (CurrentMax)
190 {
191 // Allocate slack for the array proportional to its size.
192 Grow = SIZE_T(NewMax) + UE_CONTAINER_SLACK_GROWTH_FACTOR_NUMERATOR * SIZE_T(NewMax) / UE_CONTAINER_SLACK_GROWTH_FACTOR_DENOMINATOR + ConstantGrow;
193 }
194 else if (SIZE_T(NewMax) > Grow)
195 {
196 Grow = SIZE_T(NewMax);
197 }
198#else
199 if (CurrentMax || SIZE_T(NewMax) > Grow)
200 {
201 // Allocate slack for the array proportional to its size.
202 Grow = SIZE_T(NewMax) + UE_CONTAINER_SLACK_GROWTH_FACTOR_NUMERATOR * SIZE_T(NewMax) / UE_CONTAINER_SLACK_GROWTH_FACTOR_DENOMINATOR + ConstantGrow;
203 }
204#endif
205
206 if (bAllowQuantize)
207 {
208 Retval = (SizeType)(FMemory::QuantizeSize(Grow * BytesPerElement, Alignment) / BytesPerElement);
209 }
210 else
211 {
212 Retval = (SizeType)Grow;
213 }
214 // NumElements and MaxElements are stored in 32 bit signed integers so we must be careful not to overflow here.
215 if (NewMax > Retval)
216 {
217 Retval = TNumericLimits<SizeType>::Max();
218 }
219
220 return Retval;
221}

223template <typename SizeType>
224UE_FORCEINLINE_HINT SizeType DefaultCalculateSlackReserve(SizeType NewMax, SIZE_T BytesPerElement, bool bAllowQuantize, uint32 Alignment = DEFAULT_ALIGNMENT)
225{
226 SizeType Retval = NewMax;
227 checkSlow(NewMax > 0);
228 if (bAllowQuantize)
229 {
230 Retval = (SizeType)(FMemory::QuantizeSize(SIZE_T(Retval) * SIZE_T(BytesPerElement), Alignment) / BytesPerElement);

232 if (NewMax > Retval)
233 {
234 Retval = TNumericLimits<SizeType>::Max();
235 }
236 }
237
238 return Retval;
239}
