181 struct FScrubRangeToScreen
182 {
183 double ViewStart;
184 float PixelsPerInput;
185
186 FScrubRangeToScreen(const TRange<double>& InViewInput, const FVector2D& InWidgetSize)
187 {
188 float ViewInputRange = InViewInput.Size<double>();
189
190 ViewStart = InViewInput.GetLowerBoundValue();
191 PixelsPerInput = ViewInputRange > 0 ? (InWidgetSize.X / ViewInputRange) : 0;
192 }
193
195 double LocalXToInput(float ScreenX) const
196 {
197 return PixelsPerInput > 0 ? (ScreenX / PixelsPerInput) + ViewStart : ViewStart;
198 }
199
201 double LocalDeltaXToDeltaInput(float ScreenDeltaX) const
202 {
203 return PixelsPerInput > 0 ? (ScreenDeltaX / PixelsPerInput) : 0;
204 }
205
207 float InputToLocalX(double Input) const
208 {
209 return (Input - ViewStart) * PixelsPerInput;
210 }
211 };
