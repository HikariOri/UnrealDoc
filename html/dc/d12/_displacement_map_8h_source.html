150FORCEINLINE float FDisplacementMap::Sample( int32 x, int32 y ) const

175FORCEINLINE float FDisplacementMap::Load( int32 x, int32 y ) const
176{
177 checkSlow(x < SizeX && x >= 0 && y < SizeY && y >= 0);
178 const uint8* PixelPtr = &SourceData[ int32( x + (int32)y * SizeX ) * BytesPerPixel ];
179
180 if( SourceFormat == TSF_BGRA8 )
181 {
182 return float( PixelPtr[2] ) / 255.0f;
183 }
184 else if( SourceFormat == TSF_RGBA16 )
185 {
186 checkSlow( BytesPerPixel == sizeof(uint16) * 4 );
187 return float( *(uint16*)PixelPtr ) / 65535.0f;
188 }
189 else if( SourceFormat == TSF_RGBA16F || SourceFormat == TSF_R16F )
190 {
191 FFloat16 HalfValue = *(FFloat16*)PixelPtr;
192 return HalfValue;
193 }
194 else if( SourceFormat == TSF_G8 )
195 {
196 return float( PixelPtr[0] ) / 255.0f;
197 }
198 else if( SourceFormat == TSF_G16 )
199 {
200 return float( *(uint16*)PixelPtr ) / 65535.0f;
201 }
202 else if( SourceFormat == TSF_RGBA32F || SourceFormat == TSF_R32F )
203 {
204 return *(float*)PixelPtr;
205 }
206 else
207 {
208 checkf( 0, TEXT("Displacement map format not supported") );
209 return 0.0f;
210 }
211}

213FORCEINLINE FVector2f FDisplacementMap::Load( int32 x, int32 y, uint32 Level ) const
214{
215 checkSlow( Level > 0 );
216
217 uint32 MipSizeX = ( ( SizeX - 1 ) >> Level ) + 1;
218 uint32 MipSizeY = ( ( SizeY - 1 ) >> Level ) + 1;
219
220 return MipData[ Level - 1 ][ x + y * MipSizeX ];
221}

224FORCEINLINE float FDisplacementMap::LoadFiltered( int32 x, int32 y, uint32 Level ) const
225{
226 checkSlow( Level >= 0 );
227
228 if( Level == 0 )
229 {
230 return Load( x, y );
231 }
232
233 uint32 MipSizeX = ( ( SizeX - 1 ) >> Level ) + 1;
234 uint32 MipSizeY = ( ( SizeY - 1 ) >> Level ) + 1;
235
236 return MipDataFiltered[ Level - 1 ][ x + y * MipSizeX ];
237}

239FORCEINLINE void FDisplacementMap::Address( int32& x, int32& y ) const
240{
241 if (AddressX == TA_Clamp)
242 {
243 x = FMath::Clamp(x, 0, SizeX - 1);
244 }
245 else
246 {
247 x = x % SizeX;
248 x += x < 0 ? SizeX : 0;
249 }
250
251 if (AddressY == TA_Clamp)
252 {
253 y = FMath::Clamp(y, 0, SizeY - 1);
254 }
255 else
256 {
257 y = y % SizeY;
258 y += y < 0 ? SizeY : 0;
259 }
260}
