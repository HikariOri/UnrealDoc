34struct FPersonaModes
35{
36 // Mode constants
37 static const FName SkeletonDisplayMode;
38 static const FName MeshEditMode;
39 static const FName PhysicsEditMode;
40 static const FName AnimationEditMode;
41 static const FName AnimBlueprintEditMode;
42 static FText GetLocalizedMode( const FName InMode )
43 {
44 static TMap< FName, FText > LocModes;
45
46 if (LocModes.Num() == 0)
47 {
48 LocModes.Add( SkeletonDisplayMode, NSLOCTEXT("PersonaModes", "SkeletonDisplayMode", "Skeleton") );
49 LocModes.Add( MeshEditMode, NSLOCTEXT("PersonaModes", "MeshEditMode", "Mesh") );
50 LocModes.Add( PhysicsEditMode, NSLOCTEXT("PersonaModes", "PhysicsEditMode", "Physics") );
51 LocModes.Add( AnimationEditMode, NSLOCTEXT("PersonaModes", "AnimationEditMode", "Animation") );
52 LocModes.Add( AnimBlueprintEditMode, NSLOCTEXT("PersonaModes", "AnimBlueprintEditMode", "Graph") );
53 }
54
55 check( InMode != NAME_None );
56 const FText* OutDesc = LocModes.Find( InMode );
57 check( OutDesc );
58 return *OutDesc;
59 }
60private:
61 FPersonaModes() {}
62};
