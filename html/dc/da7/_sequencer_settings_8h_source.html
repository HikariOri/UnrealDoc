84UCLASS(MinimalAPI)
85class USequencerSettingsContainer
86 : public UObject
87{
88public:
89 GENERATED_BODY()
90
91
92 template<class T>
93 static T* GetOrCreate(const TCHAR* InName)
94 {
95 static const TCHAR* SettingsContainerName = TEXT("SequencerSettingsContainer");
96
97 auto* Outer = FindObject<USequencerSettingsContainer>(GetTransientPackage(), SettingsContainerName);
98 if (!Outer)
99 {
100 Outer = NewObject<USequencerSettingsContainer>(GetTransientPackage(), USequencerSettingsContainer::StaticClass(), SettingsContainerName);
101 Outer->AddToRoot();
102 }
103
104 T* Inst = FindObject<T>( Outer, InName );
105 if (!Inst)
106 {
107 Inst = NewObject<T>( Outer, T::StaticClass(), InName );
108 Inst->LoadConfig();
109 }
110
111 return Inst;
112 }
113};
