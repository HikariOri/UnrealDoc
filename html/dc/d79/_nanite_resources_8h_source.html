84inline uint32 GetBits(uint32 Value, uint32 NumBits, uint32 Offset)
85{
86 uint32 Mask = (1u << NumBits) - 1u;
87 return (Value >> Offset) & Mask;
88}
89
90inline void SetBits(uint32& Value, uint32 Bits, uint32 NumBits, uint32 Offset)
91{
92 uint32 Mask = (1u << NumBits) - 1u;
93 check(Bits <= Mask);
94 Mask <<= Offset;
95 Value = (Value & ~Mask) | (Bits << Offset);
96}

259struct FPageRangeKey
260{
261 uint32 Value = NANITE_PAGE_RANGE_KEY_EMPTY_RANGE;
262
263 FORCEINLINE FPageRangeKey() {};
264 FORCEINLINE explicit FPageRangeKey(uint32 InValue) : Value(InValue) {}
265 FORCEINLINE FPageRangeKey(uint32 StartIndex, uint32 Count, bool bMultiRange, bool bHasStreamingPages)
266 {
267 Value = NANITE_PAGE_RANGE_KEY_EMPTY_RANGE;
268 if (Count > 0)
269 {
270 check(StartIndex <= NANITE_PAGE_RANGE_KEY_MAX_INDEX);
271 check(Count <= NANITE_PAGE_RANGE_KEY_MAX_COUNT);
272 Value = Count |
273 (StartIndex << NANITE_PAGE_RANGE_KEY_COUNT_BITS) |
274 (bMultiRange ? NANITE_PAGE_RANGE_KEY_FLAG_MULTI_RANGE : 0) |
275 (bHasStreamingPages ? NANITE_PAGE_RANGE_KEY_FLAG_HAS_STREAMING_PAGES : 0);
276 }
277 }
278
279
280 FORCEINLINE bool IsEmpty() const { return GetNumPagesOrRanges() == 0; }
281 FORCEINLINE uint32 GetNumPagesOrRanges() const { return Value & NANITE_PAGE_RANGE_KEY_COUNT_MASK; }
282 FORCEINLINE uint32 GetStartIndex() const { return (Value >> NANITE_PAGE_RANGE_KEY_COUNT_BITS) & NANITE_PAGE_RANGE_KEY_INDEX_MASK; }
283 FORCEINLINE bool IsMultiRange() const { return (Value & NANITE_PAGE_RANGE_KEY_FLAG_MULTI_RANGE) != 0; }
284 FORCEINLINE bool HasStreamingPages() const { return (Value & NANITE_PAGE_RANGE_KEY_FLAG_HAS_STREAMING_PAGES) != 0; }
285
286 FORCEINLINE bool operator==(FPageRangeKey Other) const { return Value == Other.Value; }
287 FORCEINLINE bool operator!=(FPageRangeKey Other) const { return Value != Other.Value; }
288
289 bool RemoveRootPages(uint32 NumRootPages)
290 {
291 if (!IsEmpty() && !IsMultiRange())
292 {
293 uint32 StartPage = GetStartIndex();
294 uint32 NumPages = GetNumPagesOrRanges();
295 if (StartPage < NumRootPages)
296 {
297 NumPages = (uint32)FMath::Max((int32)NumPages - (int32)(NumRootPages - StartPage), 0);
298 *this = FPageRangeKey(NumRootPages, NumPages, false, true);
299 return true;
300 }
301 }
302 return false;
303 }

561template<typename TFunc>
562bool FResources::TrueForAllPages(FPageRangeKey PageRangeKey, const TFunc& Func, bool bStreamingPagesOnly) const
563{
564 if (bStreamingPagesOnly)
565 {
566 if (!PageRangeKey.HasStreamingPages())
567 {
568 return true;
569 }
570 PageRangeKey.RemoveRootPages(NumRootPages);
571 }
572
573 if (PageRangeKey.IsMultiRange())
574 {
575 const uint32 NumRanges = PageRangeKey.GetNumPagesOrRanges();
576 for (uint32 i = 0; i < NumRanges; i++)
577 {
578 const FPageRangeKey PageRange = PageRangeLookup[PageRangeKey.GetStartIndex() + i];
579 check(!PageRange.IsEmpty() && !PageRange.IsMultiRange()); // sanity check valid range of pages
580 if (!TrueForAllPages(PageRange, Func, bStreamingPagesOnly))
581 {
582 return false;
583 }
584 }
585 }
586 else
587 {
588 uint32 StartPage = PageRangeKey.GetStartIndex();
589 uint32 NumPages = PageRangeKey.GetNumPagesOrRanges();
