142 inline EMessageSeverity::Type MassSeverityToMessageSeverity(EMessageSeverity::Type OriginalSeverity, EMassDebugMessageSeverity MassSeverity)
143 {
144 static constexpr EMessageSeverity::Type ConversionMap[static_cast<int32>(EMassDebugMessageSeverity::MAX)] =
145 {
146 /*EMassDebugMessageSeverity::Error=*/EMessageSeverity::Error,
147 /*EMassDebugMessageSeverity::Warning=*/EMessageSeverity::Warning,
148 /*EMassDebugMessageSeverity::Info=*/EMessageSeverity::Info

150 return MassSeverity == EMassDebugMessageSeverity::Default
151 ? OriginalSeverity
152 : ConversionMap[static_cast<int32>(MassSeverity)];
