76 struct FSourceChannelMap
77 {
78 alignas(16) float ChannelStartGains[AUDIO_MIXER_MAX_OUTPUT_CHANNELS * AUDIO_MIXER_MAX_OUTPUT_CHANNELS];
79 alignas(16) float ChannelDestinationGains[AUDIO_MIXER_MAX_OUTPUT_CHANNELS * AUDIO_MIXER_MAX_OUTPUT_CHANNELS];
80
81 // This is the number of bytes the gain array is using:
82 // (Number of input channels * number of output channels) * sizeof float.
83 int32 CopySize = 0;
84 bool bIsInit = false;
85
86 FSourceChannelMap(int32 InNumInChannels, int32 InNumOutChannels)
87 : CopySize(InNumInChannels* InNumOutChannels * sizeof(float))
88 {
89 checkSlow(InNumInChannels <= AUDIO_MIXER_MAX_OUTPUT_CHANNELS);
90 checkSlow(InNumOutChannels <= AUDIO_MIXER_MAX_OUTPUT_CHANNELS);
91 FMemory::Memzero(ChannelStartGains, CopySize);
92 }
93
94 FORCEINLINE void Reset(int32 InNumInChannels, int32 InNumOutChannels)
95 {
96 checkSlow(InNumInChannels <= AUDIO_MIXER_MAX_OUTPUT_CHANNELS);
97 checkSlow(InNumOutChannels <= AUDIO_MIXER_MAX_OUTPUT_CHANNELS);
98
99 CopySize = InNumInChannels * InNumOutChannels * sizeof(float);
100 FMemory::Memzero(ChannelStartGains, CopySize);
101 FMemory::Memzero(ChannelDestinationGains, CopySize);
102 bIsInit = false;
103 }
104
105 FORCEINLINE void CopyDestinationToStart()
106 {
107 FMemory::Memcpy(ChannelStartGains, ChannelDestinationGains, CopySize);
108 }
109
110 FORCEINLINE void SetChannelMap(const float* RESTRICT InChannelGains)
111 {
112 FMemory::Memcpy(ChannelDestinationGains, InChannelGains, CopySize);
113 if (!bIsInit)
114 {
115 FMemory::Memcpy(ChannelStartGains, InChannelGains, CopySize);
116 bIsInit = true;
117 }
118 }
