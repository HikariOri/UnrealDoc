50 bool IsVisiblePin(const UMaterial* Material, bool bIgnoreMaterialAttributes = false) const
51 {
52 if (!Material->IsPropertySupported(Property) && !bIgnoreMaterialAttributes)
53 {
54 return false;
55 }
56
57 const bool bSubstrateEnabled = Substrate::IsSubstrateEnabled();
58 const bool bSubstrateHiddenMaterialConversion = Substrate::IsHiddenMaterialAssetConversionEnabled();
59 const bool bFrontMaterialConnected = Material->HasSubstrateFrontMaterialConnected();
60 const bool bShowAsSubstrate = bSubstrateEnabled && (!bSubstrateHiddenMaterialConversion || bFrontMaterialConnected);
61
62 if (!bSubstrateEnabled && Property == MP_FrontMaterial)
63 {
64 return false;
65 }
66
67 if (Material->bUseMaterialAttributes && !bIgnoreMaterialAttributes)
68 {
69 // On the root node both MaterialAttributes and FrontMaterial are visible when Substrate is enabled.
70 // Otherwise only MaterialAttributes is visible.
71 return Property == MP_MaterialAttributes || (Property == MP_FrontMaterial && Substrate::IsSubstrateEnabled());
72 }
73 else if( Material->IsUIMaterial() )
74 {
75 if (bSubstrateEnabled && bSubstrateHiddenMaterialConversion && Property == MP_FrontMaterial)
76 {
77 return true; // If hidden conversion, always enable FrontMaterial in case we want to plug a Substrate material in there.
78 }
79 if (bShowAsSubstrate)
80 {
81 if (Property == MP_FrontMaterial || Property == MP_Opacity || Property == MP_OpacityMask || Property == MP_WorldPositionOffset)
82 {
83 return true;
84 }
85 }
86 else
87 {
88 if (Property == MP_EmissiveColor || Property == MP_Opacity || Property == MP_OpacityMask || Property == MP_WorldPositionOffset)
89 {
90 return true;
91 }
92 }
93
94 if (Property >= MP_CustomizedUVs0 && Property <= MP_CustomizedUVs7)
95 {
96 return (Property - MP_CustomizedUVs0) < Material->NumCustomizedUVs;
97 }
98
99 return false;
100
101 }
102 else
103 {
104 if (Property == MP_MaterialAttributes)
105 {
106 return false;
107 }
108
109 if (Property >= MP_CustomizedUVs0 && Property <= MP_CustomizedUVs7)
110 {
111 return (Property - MP_CustomizedUVs0) < Material->NumCustomizedUVs;
112 }
113
114 if (bShowAsSubstrate)
115 {
116 // The only properties visible when Substrate is enable and FrontMaterial pin is plugged.
117 return Property == MP_Opacity
118 || Property == MP_OpacityMask
119 || Property == MP_WorldPositionOffset
120 || Property == MP_AmbientOcclusion
121 || Property == MP_Refraction
122 || Property == MP_PixelDepthOffset
123 || Property == MP_FrontMaterial
124 || Property == MP_SurfaceThickness
125 || Property == MP_Displacement;
126 }
127
128
129 return true;
130 }
131 }
