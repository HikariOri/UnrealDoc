21 // Set the vertex, triangle and edge count.
22 //
23 // If EdgeCount is set > 0, SetTriEdgeIndices and SetEdge should be called for all triangles and edges respectively.
24 //
25 // If EdgeCount is set to 0, these methods do not need to be called, the edge topology will be built in Finalize.
26 //
27 // Providing edges explicitly, if available, is of course the faster option.
28 //
29 // SetVertex, SetTriVertexIndices should be called exactly once per vertex/triangle.
30 //
31 GEOMETRYCORE_API void Initialize(int VertexCount, int TriangleCount, int EdgeCount);

33 // per-triangle vertex indices, which correspond to the vertices provided by SetVertex.
34 // not thread-safe
35 inline void SetTriVertexIndices(const int TriangleID, const FIndex3i& VertexIndices)
36 {
37 Mesh.Triangles[TriangleID] = VertexIndices;
38 Mesh.VertexRefCounts.Increment(VertexIndices.A);
39 Mesh.VertexRefCounts.Increment(VertexIndices.B);
40 Mesh.VertexRefCounts.Increment(VertexIndices.C);
41 }

43 // per-triangle edge indices, which correspond to the edges provided by SetEdge, thread-safe
44 inline void SetTriEdgeIndices(const int TriangleID, const FIndex3i& EdgeIndices)
45 {
46 Mesh.TriangleEdges[TriangleID] = EdgeIndices;
47 }

49 // thread-safe

54
55 // thread-safe

59 }
