34// The normal GetDownscaledRect can produce overlapping rects where multiple viewports are present, because the Min rounds down, while the
35// Max rounds up. This variation rounds up for the far edges of the viewport, but rounds down for everything else, avoiding this issue.
36inline FIntRect GetDownscaledViewRect(FIntRect Rect, FIntPoint ViewExtent, FIntPoint Divisor)
37{
38 Rect.Min /= Divisor;
39 Rect.Max.X = (Rect.Max.X == ViewExtent.X) ? FMath::DivideAndRoundUp(ViewExtent.X, Divisor.X) : Rect.Max.X / Divisor.X;
40 Rect.Max.Y = (Rect.Max.Y == ViewExtent.Y) ? FMath::DivideAndRoundUp(ViewExtent.Y, Divisor.Y) : Rect.Max.Y / Divisor.Y;
41 Rect.Max = Rect.Max.ComponentMax(Rect.Min + FIntPoint(1, 1));
42 return Rect;
43}

208inline FScreenTransform FScreenTransform::Invert(const FScreenTransform& AToB)
209{
210 ensure(!FMath::IsNearlyZero(AToB.Scale.X));
211 ensure(!FMath::IsNearlyZero(AToB.Scale.Y));
212
213 float InvX = 1.0f / AToB.Scale.X;
214 float InvY = 1.0f / AToB.Scale.Y;
215
216 return FScreenTransform(
217 FVector2f(InvX, InvY),
218 FVector2f(-InvX * AToB.Bias.X, -InvY * AToB.Bias.Y));

304inline FScreenTransform FScreenTransform::ChangeTextureBasisFromTo(
305 const FIntPoint& TextureExtent,
306 const FIntRect& TextureViewport,
307 FScreenTransform::ETextureBasis SrcBasis,
308 FScreenTransform::ETextureBasis DestBasis)
309{
310 if (int32(SrcBasis) < int32(DestBasis))
311 {
312 if (int32(SrcBasis) + 1 != int32(DestBasis))
313 {
314 ETextureBasis IntermediaryBasis = ETextureBasis((int32(SrcBasis) + int32(DestBasis)) / 2);
315
316 return (
317 FScreenTransform::ChangeTextureBasisFromTo(TextureExtent, TextureViewport, SrcBasis, IntermediaryBasis) *
318 FScreenTransform::ChangeTextureBasisFromTo(TextureExtent, TextureViewport, IntermediaryBasis, DestBasis));
319 }
320 else if (DestBasis == ETextureBasis::ViewportUV)
321 {
322 return FScreenTransform::ScreenPosToViewportUV;
323 }
324 else if (DestBasis == ETextureBasis::TexelPosition)
325 {
326 return FScreenTransform(
327 FVector2f(TextureViewport.Width(), TextureViewport.Height()),
328 FVector2f(TextureViewport.Min.X, TextureViewport.Min.Y));
329 }
330 else if (DestBasis == ETextureBasis::TextureUV)
331 {
332 return FScreenTransform(FVector2f(1.0f / float(TextureExtent.X), 1.0f / float(TextureExtent.Y)), FVector2f(0.0f, 0.0f));
333 }
334 else
335 {
336 check(0);
337 }
338 }
339 else if (int32(SrcBasis) > int32(DestBasis))
340 {
341 if (int32(SrcBasis) != int32(DestBasis) + 1)
342 {
343 ETextureBasis IntermediaryBasis = ETextureBasis((int32(SrcBasis) + int32(DestBasis)) / 2);
344
345 return (
346 FScreenTransform::ChangeTextureBasisFromTo(TextureExtent, TextureViewport, SrcBasis, IntermediaryBasis) *
347 FScreenTransform::ChangeTextureBasisFromTo(TextureExtent, TextureViewport, IntermediaryBasis, DestBasis));
348 }
349 else if (DestBasis == ETextureBasis::ScreenPosition)
350 {
351 return FScreenTransform::ViewportUVToScreenPos;
352 }
353 else if (DestBasis == ETextureBasis::ViewportUV)
354 {
355 float InvWidth = 1.0f / float(TextureViewport.Width());
356 float InvHeight = 1.0f / float(TextureViewport.Height());
357
358 return FScreenTransform(
359 FVector2f(InvWidth, InvHeight),
360 FVector2f(-InvWidth * TextureViewport.Min.X, -InvHeight * TextureViewport.Min.Y));
361 }
362 else if (DestBasis == ETextureBasis::TexelPosition)
363 {
364 return FScreenTransform(FVector2f(TextureExtent.X, TextureExtent.Y), FVector2f(0.0f, 0.0f));
365 }
366 else
367 {
368 check(0);
369 }
370 }
371
372 return FScreenTransform::Identity;
373}
