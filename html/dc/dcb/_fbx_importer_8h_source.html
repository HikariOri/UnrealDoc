545class FFbxHelper
546{
547public:
551 static float NameCompareWeight(const FString& A, const FString& B)
552 {
553 FString Longer = A;
554 FString Shorter = B;
555 if (A.Len() < B.Len())
556 {
557 Longer = B;
558 Shorter = A;
559 }
560
561 if (Longer.Compare(Shorter, ESearchCase::CaseSensitive) == 0)
562 {
563 return 1.0f;
564 }
565 if (Longer.Compare(Shorter, ESearchCase::IgnoreCase) == 0)
566 {
567 return 0.98f;
568 }
569 // We do the contain so it is giving better result since often we compare thing like copy and paste string name
570 // we want to match: BackZ
571 // between: Paste_BackZ and BackX
572 // EditDistance for Paste_BackZ is 5/11 =0.45
573 // EditDistance for BackX is 4/5= 0.8
574 // The contains weight for Paste_BackZ is 0.98- (0.25*(1.0-5/11)) = 0.844
575 if (Longer.Contains(Shorter, ESearchCase::CaseSensitive))
576 {
577 return 0.98f - 0.25f*(1.0f - ((float)(Shorter.Len()) / (float)(Longer.Len())));
578 }
579 if (Longer.Contains(Shorter, ESearchCase::IgnoreCase))
580 {
581 return 0.96f - 0.25f*(1.0f - ((float)(Shorter.Len()) / (float)(Longer.Len())));
582 }
583
584
585 float LongerLength = (float)Longer.Len();
586 if (LongerLength == 0)
587 {
588 return 1.0f;
589 }
590 return (LongerLength - Algo::LevenshteinDistance(Longer, Shorter)) / LongerLength;
591 }
592};
