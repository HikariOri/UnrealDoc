88 bool Serialize(FArchive& Ar)
89 {
90 Ar << MappingRangeSpace;
91 Ar << MappingRange.Min;
92 Ar << MappingRange.Max;
93
94 // We must create the proper spline before we can serialize it.
95 if (Ar.IsLoading())
96 {
97 FSplineTypeId::IdType TypeId;
98
99 // We are about to read data from the archive that the spline itself will attempt to read,
100 // so we have to restore the archive position after reading it.
101 const int64 Pos = Ar.Tell();
102 {
103 int32 Version;
104 Ar << Version;
105 Ar << TypeId;
106 }
107 Ar.Seek(Pos);
108
109 Spline = FSplineTypeRegistry::CreateSpline(TypeId);
110
111 ensureAlways(Spline); // if this failed, most likely the spline type has not been registered.
112 }
113
114 Ar << *Spline;
115
116 return true;

422 template <typename AttrType>
423 bool CloneAttributeChannel(const FName& SourceName, const FName& DestName)
424 {
425 // Check if destination already exists
426 if (Children.Contains(DestName))
427 {
428 UE_LOGF(LogSpline, Warning, "Cannot clone channel: Destination channel '%ls' already exists.",
429 *DestName.ToString());
430 return false;
431 }
432
433 // Get source channel
434 TSplineInterface<AttrType>* SourceChannel = GetAttributeChannel<AttrType>(SourceName);
435 if (!SourceChannel)
436 {
437 UE_LOGF(LogSpline, Warning, "Cannot clone channel: Source channel '%ls' not found or type mismatch.",
438 *SourceName.ToString());
439 return false;
440 }
441
442 // Get implementation information from source
443 const FString ImplName = SourceChannel->GetImplementationName();
444
445 // Create new destination channel with same implementation
446 if (!CreateAttributeChannelInternal<AttrType>(DestName, ImplName))
447 {
448 UE_LOGF(LogSpline, Error, "Failed to create destination channel '%ls'.",
449 *DestName.ToString());
450 return false;
451 }
452
453 // Clone the source channel's data to the destination channel
454 TSplineInterface<AttrType>* DestChannel = GetAttributeChannel<AttrType>(DestName);
455 if (!DestChannel)
456 {
457 // This should not happen, but just in case
458 UE_LOGF(LogSpline, Error, "Internal error: Unable to retrieve just-created channel '%ls'.",
459 *DestName.ToString());
460 return false;
461 }
462
463 // Clone the source to the destination
464 // We need to use the ISplineInterface::Clone() approach and move the clone into our structure
465 TUniquePtr<ISplineInterface> ClonedSpline = SourceChannel->Clone();
466
467 // Find and update the destination child's spline pointer
468 FChildSpline& DestChild = Children[DestName];
469 DestChild.Spline = MoveTemp(ClonedSpline);
470
471 // Copy mapping settings from source
472 const FChildSpline& SourceChild = *Children.Find(SourceName);
473 DestChild.MappingRange = SourceChild.MappingRange;
474 DestChild.MappingRangeSpace = SourceChild.MappingRangeSpace;
475
476 return true;
477 }

533 template <typename ImplType>
534 ImplType* CreateAttributeChannel(const FName& Name)
535 {
536 // Extract the value type from the implementation type
537 using AttrType = typename ImplType::ValueType;
538
539 // Check if the type is already registered
540 uint32 TypeId = FSplineTypeRegistry::GetTypeId(ImplType::GetSplineTypeName(),
541 TSplineValueTypeTraits<AttrType>::Name);
542 // If not registered, register it manually
543 if (TypeId == 0)
544 {
545 // Get type information for registration
546 TypeId = ImplType::GetStaticTypeId();
547 FString ImplName = ImplType::GetSplineTypeName();
548 FString ValueName = TSplineValueTypeTraits<AttrType>::Name;
549
550 // Register with a factory function that creates this specific type
551 bool bSuccess = FSplineTypeRegistry::RegisterType(
552 TypeId,
553 ImplName,
554 ValueName,
555 []() { return MakeUnique<ImplType>(); }
556 );
557
558 if (!bSuccess)
559 {
560 UE_LOGF(LogSpline, Error, "Failed to register spline type '%ls' with value type '%ls'",
561 *ImplName, *ValueName);
562 return nullptr;
563 }
564 }
565
566 // Create the channel using the existing name-based method
567 CreateAttributeChannelInternal<AttrType>(Name, ImplType::GetSplineTypeName());
568
569 // Return the typed implementation
570 return GetTypedAttributeChannel<ImplType>(Name);
571 }

712 template <typename AttrType>
713 bool CreateAttributeChannelInternal(const FName& Name, const FString& ImplName)
714 {
715 // Check if channel already exists
716 if (Children.Contains(Name))
717 {
718 UE_LOGF(LogSpline, Warning, "Failed to create attribute channel '%ls': Channel already exists.", *Name.ToString());
719 return false;
720 }
721
722 // Look up the type ID
723 uint32 TypeId = FSplineTypeRegistry::GetTypeId(ImplName, TSplineValueTypeTraits<AttrType>::Name);
724 if (TypeId == 0)
725 {
726 UE_LOGF(LogSpline, Error, "Failed to find type ID for implementation '%ls' with value type '%ls'.",
727 *ImplName, *TSplineValueTypeTraits<AttrType>::Name);
728 return false;
729 }
730
731 // Create child spline by type ID
732 TUniquePtr<ISplineInterface> NewSpline = FSplineTypeRegistry::CreateSpline(TypeId);
733 if (!NewSpline)
734 {
735 UE_LOGF(LogSpline, Error, "Failed to create spline of type '%ls'/'%ls' (ID: 0x%08X) for attribute channel '%ls'.",
736 *ImplName, *TSplineValueTypeTraits<AttrType>::Name, TypeId, *Name.ToString());
737 return false;
738 }
739
740 // Verify that the created spline is of the expected value type
741 if (NewSpline->GetValueTypeName() != TSplineValueTypeTraits<AttrType>::Name)
742 {
743 UE_LOGF(LogSpline, Error, "Created spline has incorrect value type. Expected '%ls', got '%ls'.",
744 *TSplineValueTypeTraits<AttrType>::Name, *NewSpline->GetValueTypeName());
745 return false;
746 }
747
748 // Add to children map
749 FChildSpline& Child = Children.Add(Name);
750 Child.Spline = MoveTemp(NewSpline);
751 Child.MappingRangeSpace = EMappingRangeSpace::Normalized;
752 Child.MappingRange = FInterval1f(0.0f, 1.0f);
753
754 return true;
755 }
