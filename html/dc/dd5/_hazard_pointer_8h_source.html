181 //grab a hazard pointer and once hazard is set the other threads leave it alone
182 template<bool Cached>
183 inline FHazardRecord* Acquire()
184 {
185 struct FPseudo
186 {
187 static inline uint32 GetThreadId()
188 {
189 static std::atomic_uint counter{0};
190 uint32 value = counter.fetch_add(1, std::memory_order_relaxed);
191 value = ((value >> 16) ^ value) * 0x45d9f3b;
192 value = ((value >> 16) ^ value) * 0x45d9f3b;
193 value = (value >> 16) ^ value;
194 return value;
195 }
196 };
197
198 static thread_local uint32 StartIndex = FPseudo::GetThreadId();
199
200 FHazardRecordChunk* p = &Head;
201 if (Cached)
202 {
203 // acquire pairs with the release store in Grow() that publishes a new chunk;
204 // without it, the chunk constructor's writes to Records[] are not
205 // synchronized with the CAS below.
206 p = p->Next.load(std::memory_order_acquire);
207 goto TestCondition;
208 }
209
210 //search HazardPointerList for an empty Entry
211 do
212 {
213 for (uint64 idx = 0; idx < HazardChunkSize; idx++)
214 {
215 uintptr_t Nullptr = 0;
216 uintptr_t FreeEntry = FHazardRecord::FreeHazardEntry;
217 uint64 i = (StartIndex + idx) % HazardChunkSize;
218 if (p->Records[i].Hazard.compare_exchange_weak(FreeEntry, Nullptr, std::memory_order_relaxed))
219 {
220 checkSlow(p->Records[i].GetHazard() == nullptr);
221 return &p->Records[i];
222 }
223 }
224 // acquire pairs with the release store in Grow() that publishes a new chunk;
225 // without it, the chunk constructor's writes to Records[] are not
226 // synchronized with the CAS below.
227 p = p->Next.load(std::memory_order_acquire);
228 TestCondition:;
229 } while (p);
230
231 return Grow<Cached>();
232 }

285 inline THazardPointer(std::atomic<H*>& InHazard, FHazardPointerCollection& Collection)
286 : Hazard(&InHazard), Record(Collection.Acquire<Cached>())
287 {
288 checkSlow(Record->GetHazard() == nullptr);
289 }
290
291 inline ~THazardPointer()
292 {
293 if (Record)
294 {
295 Record->Release();
296 }
297 }
298
299 //retireing can be used to release the hazardpointer without reaquiring a new Hazardslot in the Collection
300 inline void Retire()
301 {
302 checkSlow(Record);
303 Record->Retire();
304 }
305
306 //use with care, because the Hazardpointer will not protect anymore and needs to be recreated
307 inline void Destroy()
308 {
309 checkSlow(Record);
310 Record->Release();
311 Record = nullptr;
312 Hazard = nullptr;
313 }
314
315 inline H* Get() const
316 {
317 checkSlow(Record);
318 H* HazardPointer;
319 do
320 {
321 HazardPointer = (H*)Record->SetHazard(Hazard->load(std::memory_order_acquire));
322 } while (HazardPointer != Hazard->load(std::memory_order_acquire));
323 return HazardPointer;
324 }
325
326 inline bool IsValid()
327 {
328 return Hazard != nullptr && Record != nullptr;
329 }
