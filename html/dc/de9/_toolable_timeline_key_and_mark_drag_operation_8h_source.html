172 static TSet<FSequencerSelectedKey> GetSelectedKeys(const FMouseInputData& InMouseInput)
173 {
174 TSet<FSequencerSelectedKey> SelectedKeys = InMouseInput.Timeline->GetKeySelection().GetSelectedKeys();

176 if (const TSharedPtr<FSequencer> Sequencer = InMouseInput.Timeline->GetSequencer())

184 const FToolableTimelineDragOperation* GetDragOperation() const
185 {
