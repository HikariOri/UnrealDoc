142 TFixedSparseBitSet& operator|=(const TFixedSparseBitSet& Other)
143 {
144 using namespace Private;
145
146 HashType One(1u);
147 HashType NewHash = Other.BucketHash | BucketHash;
148 HashType OtherHash = Other.BucketHash;
149
150 uint32 OtherBucketIndex = 0;
151 uint32 OtherBucketBitIndex = Private::CountTrailingZeros(OtherHash);
152
153 while (OtherBucketBitIndex < HashSize)
154 {
155 const HashType HashBit = HashType(1) << OtherBucketBitIndex;
156 const uint32 ThisBucketIndex = Private::CountBits(BucketHash & (HashBit-1));
157
158 if ((BucketHash & HashBit) == 0)
159 {
160 Buckets.Insert(Other.Buckets.Get(OtherBucketIndex), ThisBucketIndex);
161 }
162 else
163 {
164 Buckets.Get(ThisBucketIndex) |= Other.Buckets.Get(OtherBucketIndex);
165 }
166
167 BucketHash |= HashBit;
168
169 ++OtherBucketIndex;
170
171 // Mask out this bit and find the index of the next one
172 OtherHash &= ~(One << OtherBucketBitIndex);
173 OtherBucketBitIndex = Private::CountTrailingZeros(OtherHash);
174 }
175
176 return *this;

214 ESparseBitSetBitResult SetBit(uint32 BitIndex)
215 {
216 CheckIndex(BitIndex);
217
218 FBitOffsets Offsets(BucketHash, BitIndex);
219
220 // Do we need to add a new bucket?
221 if ( (BucketHash & Offsets.HashBit) == 0)
222 {
223 BucketHash |= Offsets.HashBit;
224 Buckets.Insert(Offsets.BitMaskWithinBucket, Offsets.BucketIndex);
225 return ESparseBitSetBitResult::NewlySet;
226 }
227 else if ((Buckets.Get(Offsets.BucketIndex) & Offsets.BitMaskWithinBucket) == 0)
228 {
229 Buckets.Get(Offsets.BucketIndex) |= Offsets.BitMaskWithinBucket;
230 return ESparseBitSetBitResult::NewlySet;
231 }

239 bool IsBitSet(uint32 BitIndex) const
240 {
241 CheckIndex(BitIndex);
242
243 const uint32 Hash = BitIndex / BucketSize;
244 const HashType HashBit = (HashType(1) << Hash);
245 if (BucketHash & HashBit)
246 {
247 const uint32 BucketIndex = Private::CountBits(BucketHash & (HashBit-1));
248 const uint32 ThisBitIndex = (BitIndex-BucketSize*Hash);
249 const BucketType ThisBitMask = BucketType(1u) << ThisBitIndex;

251 return Buckets.Get(BucketIndex) & ThisBitMask;
252 }
253 return false;

259 int32 GetSparseBucketIndex(uint32 BitIndex) const
260 {
261 CheckIndex(BitIndex);
262
263 const uint32 Hash = BitIndex / BucketSize;
264 const HashType HashBit = (HashType(1) << Hash);
265 if (BucketHash & HashBit)
266 {
267 uint32 BucketIndex = Private::CountBits(BucketHash & (HashBit-1));
268
269 const uint32 ThisBitIndex = (BitIndex-BucketSize*Hash);
270 const BucketType ThisBitMask = static_cast<BucketType>(BucketType(1u) << ThisBitIndex);
271
272 BucketType ThisBucket = Buckets.Get(BucketIndex);
273 if (ThisBucket & ThisBitMask)
274 {
275 // Compute the offset
276 int32 SparseIndex = Private::CountBits(ThisBucket & (ThisBitMask-1));
277
278 // Count all the preceding buckets to find the final sparse index
279 while (BucketIndex > 0)
280 {
281 --BucketIndex;
282 SparseIndex += Private::CountBits(Buckets.Get(BucketIndex));
283 }
284 return SparseIndex;
285 }
286 }

300 static FIterator Begin(const TFixedSparseBitSet<HashType, BucketStorage>* InBitSet)
301 {

303 It.BitSet = InBitSet;
304 It.CurrentBucket = 0;
305
306 if (InBitSet->BucketHash != 0)
307 {
308 It.BucketBitIndex = Private::CountTrailingZeros(InBitSet->BucketHash);
309 It.CurrentBucket = InBitSet->Buckets.Get(0);
310 It.IndexWithinBucket = Private::CountTrailingZeros(It.CurrentBucket);
311 }
312 else
313 {
314 It.BucketBitIndex = HashSize;
315 It.IndexWithinBucket = 0;
316 }
317 return It;
318 }
319 static FIterator End(const TFixedSparseBitSet<HashType, BucketStorage>* InBitSet)
320 {
321 FIterator It;
322 It.BitSet = InBitSet;
323 It.CurrentBucket = 0;
324 It.BucketBitIndex = HashSize;
325 It.IndexWithinBucket = 0;
326 return It;
327 }
328
329 void operator++()
330 {
331 using namespace Private;
332
333 // Clear the lowest 1 bit
334 CurrentBucket = CurrentBucket & (CurrentBucket - 1);
335
336 if (CurrentBucket != 0)
337 {
338 IndexWithinBucket = CountTrailingZeros(CurrentBucket);
339 }
340 else
341 {
342 // If this was the last bit, reset the iterator to end()
343 if (BucketBitIndex == HashSize-1)
344 {
345 IndexWithinBucket = 0;
346 BucketBitIndex = HashSize;
347 return;
348 }
349
350 HashType UnvisitedBucketBitMask = BitOffsetToHighBitMask<HashType>(BucketBitIndex+1);
351 BucketBitIndex = CountTrailingZeros(HashType(BitSet->BucketHash & UnvisitedBucketBitMask));
352
353 // Check whether we're at the end
354 if (BucketBitIndex == HashSize)
355 {
356 IndexWithinBucket = 0;
357 }
358 else
359 {
360 const uint8 NextBucketIndex = Private::CountBits(BitSet->BucketHash & BitOffsetToLowBitMask<HashType>(BucketBitIndex));
361 CurrentBucket = BitSet->Buckets.Get(NextBucketIndex);
362 IndexWithinBucket = CountTrailingZeros(CurrentBucket);
363 }
364 }
365 }
366
367 int32 operator*() const
368 {
369 return BucketSize*BucketBitIndex + IndexWithinBucket;
370 }

375 }
