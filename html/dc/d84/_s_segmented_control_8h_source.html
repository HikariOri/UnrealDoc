233 void RebuildChildren()
234 {
235 // The right padding will be applied later at the end so we dont accumulate left+right padding between all buttons
236 FMargin SlotPadding = Style->UniformPadding;
237 SlotPadding.Right = 0.0f;
238
239 TSharedPtr<SUniformGridPanel> UniformBox = SNew(SUniformGridPanel).SlotPadding(SlotPadding);
240
241 const int32 NumSlots = Children.Num();
242 for ( int32 SlotIndex = 0; SlotIndex < NumSlots; ++SlotIndex )
243 {
244 TSharedRef<SWidget> Child = Children[SlotIndex].GetWidget();
245 FSlot* ChildSlotPtr = &Children[SlotIndex];
246 const OptionType ChildValue = ChildSlotPtr->_Value;
247
248 TAttribute<FVector2D> SpacerLambda = FVector::ZeroVector;
249 if (ChildSlotPtr->_Icon.IsBound() || ChildSlotPtr->_Text.IsBound())
250 {
251 SpacerLambda = MakeAttributeLambda([ChildSlotPtr]() { return (ChildSlotPtr->_Icon.Get() != nullptr && !ChildSlotPtr->_Text.Get().IsEmpty()) ? FVector2D(8.0f, 1.0f) : FVector2D::ZeroVector; });
252 }
253 else
254 {
255 SpacerLambda = (ChildSlotPtr->_Icon.Get() != nullptr && !ChildSlotPtr->_Text.Get().IsEmpty()) ? FVector2D(8.0f, 1.0f) : FVector2D::ZeroVector;
256 }
257
258 if (Child == SNullWidget::NullWidget)
259 {
260 Child = SNew(SHorizontalBox)
261
262 +SHorizontalBox::Slot()
263 .AutoWidth()
264 .VAlign(VAlign_Center)
265 [
266 SNew(SImage)
267 .ColorAndOpacity(FSlateColor::UseForeground())
268 .Image(ChildSlotPtr->_Icon)
269 ]
270
271 +SHorizontalBox::Slot()
272 .AutoWidth()
273 [
274 SNew(SSpacer)
275 .Size(SpacerLambda)
276 ]
277
278 +SHorizontalBox::Slot()
279 .VAlign(VAlign_Center)
280 .AutoWidth()
281 [
282 SNew(STextBlock)
283 .TextStyle(TextStyle)
284 .Text(ChildSlotPtr->_Text)
285 ];
286 }
287
288 const FCheckBoxStyle* CheckBoxStyle = &Style->ControlStyle;
289 if (SlotIndex == 0)
290 {
291 CheckBoxStyle = &Style->FirstControlStyle;
292 }
293 else if (SlotIndex == NumSlots - 1)
294 {
295 CheckBoxStyle = &Style->LastControlStyle;
296 }
297
298 const int32 ColumnIndex = MaxSegmentsPerLine > 0 ? SlotIndex % MaxSegmentsPerLine : SlotIndex;
299 const int32 RowIndex = MaxSegmentsPerLine > 0 ? SlotIndex / MaxSegmentsPerLine : 0;
300
301 // Note HAlignment is applied at the check box level because if it were applied here it would make the slots look physically disconnected from each other
302 UniformBox->AddSlot(ColumnIndex, RowIndex)
303 .VAlign(ChildSlotPtr->GetVerticalAlignment())
304 [
305 SAssignNew(ChildSlotPtr->_CheckBox, SCheckBox)
306 .Clipping(EWidgetClipping::ClipToBounds)
307 .HAlign(ChildSlotPtr->GetHorizontalAlignment())
308 .ToolTipText(ChildSlotPtr->_Tooltip)
309 .ToolTip(ChildSlotPtr->_ToolTipWidget)
310 .Style(CheckBoxStyle)
311 .IsChecked(GetCheckBoxStateAttribute(ChildValue))
312 .OnCheckStateChanged(this, &SSegmentedControl::CommitValue, ChildValue)
313 .Padding(UniformPadding)
314 .Visibility(ChildSlotPtr->_Visibility)
315 [
316 Child
317 ]
318 ];
319 }
320
321 ChildSlot
322 [
323 SNew(SBorder)
324 .BorderImage(&Style->BackgroundBrush)
325 .Padding(FMargin(0,0,Style->UniformPadding.Right,0))
326 [
327 UniformBox.ToSharedRef()
328 ]
329 ];
330
331 UpdateCheckboxValuesIfNeeded();
332 }

428 static TSharedPtr<SSegmentedControl<OptionType>> Create(
429 const TArray<OptionType>& InKeys,
430 const TArray<FText>& InLabels,
431 const TArray<FText>& InTooltips,
432 const TAttribute<TArray<OptionType>>& InValues,
433 bool bSupportsMultiSelection = true,
434 FOnValuesChanged OnValuesChanged = FOnValuesChanged())
435 {
436 TSharedPtr<SSegmentedControl<OptionType>> Widget;
437
438 SAssignNew(Widget, SSegmentedControl<OptionType>)
439 .SupportsMultiSelection(bSupportsMultiSelection)
440 .Values(InValues)
441 .OnValuesChanged(OnValuesChanged);
442
443 ensure(InKeys.Num() == InLabels.Num());
444 ensure(InKeys.Num() == InTooltips.Num());
445
446 for(int32 Index = 0; Index < InKeys.Num(); Index++)
447 {
448 Widget->AddSlot(InKeys[Index], false)
449 .Text(InLabels[Index])
450 .ToolTip(InTooltips[Index]);
451 }
452
453 Widget->RebuildChildren();
454
455 return Widget;
456 }

476 void UpdateCheckboxValuesIfNeeded()
477 {
478 if (!CurrentValuesIsBound)
479 {
480 const TArray<OptionType> Values = GetValues();
481
482 for (int32 Index = 0; Index < Children.Num(); ++Index)
483 {
484 const FSlot& Slot = Children[Index];
485 if (const TSharedPtr<SCheckBox> CheckBox = Slot._CheckBox.Pin())
486 {
487 CheckBox->SetIsChecked(Values.Contains(Slot._Value) ? ECheckBoxState::Checked : ECheckBoxState::Unchecked);
488 }

492
493 void CommitValue(const ECheckBoxState InCheckState, OptionType InValue)
494 {
495 const TArray<OptionType> PreviousValues = CurrentValues.Get();
496 TArray<OptionType> Values = PreviousValues;
497
498 // don't allow to deselect the last checkbox
499 if(InCheckState != ECheckBoxState::Checked && Values.Num() == 1)
500 {
501 if(!SupportsEmptySelection)
502 {
503 UpdateCheckboxValuesIfNeeded();
504 return;
505 }
506 }
507
508 bool bModifierIsDown = false;
509 if(SupportsMultiSelection)
510 {
511 bModifierIsDown =
512 FSlateApplication::Get().GetModifierKeys().IsShiftDown() ||
513 FSlateApplication::Get().GetModifierKeys().IsControlDown();
514 }
515
516 // if the attribute is not bound update our internal state
517 if(bModifierIsDown)
518 {
519 if (InCheckState == ECheckBoxState::Checked)
520 {
521 Values.AddUnique(InValue);
522 }
523 else
524 {
525 Values.Remove(InValue);
526 }
527 }
528 else
529 {
530 if((InCheckState == ECheckBoxState::Checked) || Values.Contains(InValue))
531 {
532 Values.Reset();
533 Values.Add(InValue);
534 }
535 }
536
537 if (!CurrentValuesIsBound)
538 {
539 CurrentValues.Set(*this, Values);
540
541 UpdateCheckboxValuesIfNeeded();
542 }
543
544 if(OnValueChecked.IsBound())
545 {
546 if(!bModifierIsDown && InCheckState == ECheckBoxState::Checked)
547 {
548 for(const OptionType PreviousValue : PreviousValues)
549 {
550 if(!Values.Contains(PreviousValue))
551 {
552 OnValueChecked.Execute(PreviousValue, ECheckBoxState::Unchecked);
553 }
554 }
555 }
556
557 OnValueChecked.Execute(InValue, InCheckState);
558 }
559
560 if (InCheckState == ECheckBoxState::Checked)
561 {
562 OnValueChanged.ExecuteIfBound(InValue);
563 }
564
565 OnValuesChanged.ExecuteIfBound(Values);
566 }
