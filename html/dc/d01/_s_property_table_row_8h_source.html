62 FReply OnMouseButtonUp( const FGeometry& MyGeometry, const FPointerEvent& MouseEvent ) override
63 {
64 FReply Reply = FReply::Unhandled();
65
66 const TSharedRef< IPropertyTable > Table = Row->GetTable();
67 const EPropertyTableSelectionUnit::Type SelectionUnit = Table->GetSelectionUnit();
68
69 Table->SetCurrentCell( PreviousCurrentCell );
70 const TSharedPtr< IPropertyTableCell > CurrentCell = Table->GetCurrentCell();
71 const TSharedPtr< IPropertyTableCell > LastClickedCell = Table->GetLastClickedCell();
72
73 TSet< TSharedRef< IPropertyTableCell > > SelectedCells( Table->GetSelectedCells() );
74
75 Reply = SPropertyTableRowBase::OnMouseButtonUp( MyGeometry, MouseEvent );
76
77 if ( SelectionUnit == EPropertyTableSelectionUnit::Cell )
78 {
79 if ( LastClickedCell.IsValid() )
80 {
81 // The Row may not have any cells, so we'll detect an old LastClickedCell and clear
82 if ( !Row->HasCells() || LastClickedCell->GetRow() != Row || !LastClickedCell->GetColumn()->CanSelectCells() )
83 {
84 Table->SetLastClickedCell( nullptr );
85 Table->SetCurrentCell( nullptr );
86 }
87 else
88 {
89 if ( CurrentCell.IsValid() && LastClickedCell == CurrentCell && !MouseEvent.IsControlDown())
90 {
91 // Don't let the table enter edit mode if the ctrl key is down
92 Table->GetLastClickedCell()->EnterEditMode();
93 }
94 else if ( MouseEvent.IsShiftDown() && CurrentCell.IsValid() && LastClickedCell != CurrentCell )
95 {
96 const int32 LastClickedRowIndex = Table->GetRows().Find( LastClickedCell->GetRow() );
97 const int32 CurrentRowIndex = Table->GetRows().Find( CurrentCell->GetRow() );
98
99 const int32 TopMostRowIndex = LastClickedRowIndex < CurrentRowIndex ? LastClickedRowIndex : CurrentRowIndex;
100 const int32 BottomMostRowIndex = LastClickedRowIndex > CurrentRowIndex ? LastClickedRowIndex : CurrentRowIndex;
101
102 const int32 LastClickedColumnIndex = Table->GetColumns().Find( LastClickedCell->GetColumn() );
103 const int32 CurrentColumnIndex = Table->GetColumns().Find( CurrentCell->GetColumn() );
104
105 const int32 LeftMostColumnIndex = LastClickedColumnIndex < CurrentColumnIndex ? LastClickedColumnIndex : CurrentColumnIndex;
106 const int32 RightMostColumnIndex = LastClickedColumnIndex > CurrentColumnIndex ? LastClickedColumnIndex : CurrentColumnIndex;
107
108 const TSharedRef< IPropertyTableCell > StartingCell = Table->GetColumns()[ LeftMostColumnIndex ]->GetCell( Table->GetRows()[ TopMostRowIndex ] );
109 const TSharedRef< IPropertyTableCell > EndingCell = Table->GetColumns()[ RightMostColumnIndex ]->GetCell( Table->GetRows()[ BottomMostRowIndex ] );
110
111 Table->SelectCellRange( StartingCell, EndingCell );
112 Table->SetCurrentCell( CurrentCell );
113 }
114 else if (MouseEvent.IsControlDown())
115 {
116 if (SelectedCells.Contains(LastClickedCell.ToSharedRef()))
117 {
118 SelectedCells.Remove(LastClickedCell.ToSharedRef());
119 Table->SetCurrentCell(nullptr);
120 }
121 else
122 {
123 SelectedCells.Add(LastClickedCell.ToSharedRef());
124 Table->SetCurrentCell(LastClickedCell);
125 }
126
127 Table->SetSelectedCells(SelectedCells);
128 }
129 else
130 {
131 SelectedCells.Empty();
132 SelectedCells.Add( LastClickedCell.ToSharedRef() );
133
134 Table->SetSelectedCells( SelectedCells );
135 Table->SetCurrentCell( LastClickedCell );
136 }
137 }
138 }
139 }
140 else if ( SelectionUnit == EPropertyTableSelectionUnit::Row )
141 {
142 if ( !Table->GetSelectedRows().Contains( Row.ToSharedRef() ) )
143 {
144 Table->SetCurrentCell( nullptr );
145 }
146 }
147
148 return Reply;
149 }
