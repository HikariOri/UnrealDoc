622inline void TransitionRenderPassTargets(FRHICommandList& RHICmdList, const FRHIRenderPassInfo& RPInfo)
623{
624 FRHITransitionInfo Transitions[MaxSimultaneousRenderTargets];
625 int32 TransitionIndex = 0;
626 uint32 NumColorRenderTargets = RPInfo.GetNumColorRenderTargets();
627 for (uint32 Index = 0; Index < NumColorRenderTargets; Index++)
628 {
629 const FRHIRenderPassInfo::FColorEntry& ColorRenderTarget = RPInfo.ColorRenderTargets[Index];
630 if (ColorRenderTarget.RenderTarget != nullptr)
631 {
632 Transitions[TransitionIndex] = FRHITransitionInfo(ColorRenderTarget.RenderTarget, ERHIAccess::Unknown, ERHIAccess::RTV);
633 TransitionIndex++;
634 }
635 }
636
637 const FRHIRenderPassInfo::FDepthStencilEntry& DepthStencilTarget = RPInfo.DepthStencilRenderTarget;
638 if (DepthStencilTarget.DepthStencilTarget != nullptr && (RPInfo.DepthStencilRenderTarget.ExclusiveDepthStencil.IsAnyWrite()))
639 {
640 RHICmdList.Transition(FRHITransitionInfo(DepthStencilTarget.DepthStencilTarget, ERHIAccess::Unknown, ERHIAccess::DSVRead | ERHIAccess::DSVWrite));
641 }
642
643 RHICmdList.Transition(MakeArrayView(Transitions, TransitionIndex));
644}

657inline void TransitionAndCopyTexture(FRHICommandList& RHICmdList, FRHITexture* SrcTexture, FRHITexture* DstTexture, const FRHICopyTextureInfo& Info)
658{
659 check(SrcTexture && DstTexture);
660 check(SrcTexture->GetNumSamples() == DstTexture->GetNumSamples());
661
662 if (SrcTexture == DstTexture)
663 {
664 RHICmdList.Transition({
665 FRHITransitionInfo(SrcTexture, ERHIAccess::Unknown, ERHIAccess::SRVMask)
666 });
667 return;
668 }
669
670 RHICmdList.Transition({
671 FRHITransitionInfo(SrcTexture, ERHIAccess::Unknown, ERHIAccess::CopySrc),
672 FRHITransitionInfo(DstTexture, ERHIAccess::Unknown, ERHIAccess::CopyDest)
673 });
674
675 RHICmdList.CopyTexture(SrcTexture, DstTexture, Info);
676
677 RHICmdList.Transition({
678 FRHITransitionInfo(SrcTexture, ERHIAccess::CopySrc, ERHIAccess::SRVMask),
679 FRHITransitionInfo(DstTexture, ERHIAccess::CopyDest, ERHIAccess::SRVMask)
