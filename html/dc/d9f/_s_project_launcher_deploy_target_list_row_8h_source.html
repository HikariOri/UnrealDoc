194 // Callback for changing this row's check box state.
195 void HandleCheckBoxStateChanged(ECheckBoxState NewState)
196 {
197 ILauncherDeviceGroupPtr ActiveGroup = DeviceGroup.Get();
198
199 if (ActiveGroup.IsValid() && DeviceProxy.IsValid() && DeviceProxy->HasVariant(SelectedVariant))
200 {
201 const FString& DeviceID = DeviceProxy->GetTargetDeviceId(SelectedVariant);
202 if (NewState == ECheckBoxState::Checked)
203 {
204 ActiveGroup->AddDevice(DeviceID);
205 }
206 else
207 {
208 ActiveGroup->RemoveDevice(DeviceID);
209 }
210 }
211 }
212
213 // Callback for determining this row's check box state.
214 ECheckBoxState HandleCheckBoxIsChecked() const
215 {
216 if (IsEnabled())
217 {
218 ILauncherDeviceGroupPtr ActiveGroup = DeviceGroup.Get();
219
220 if (ActiveGroup.IsValid() && DeviceProxy.IsValid() && DeviceProxy->HasVariant(SelectedVariant))
221 {
222 const FString& DeviceID = DeviceProxy->GetTargetDeviceId(SelectedVariant);
223 return ActiveGroup->GetDeviceIDs().Contains(DeviceID) ? ECheckBoxState::Checked : ECheckBoxState::Unchecked;
224 }
225 }
226
227 return ECheckBoxState::Unchecked;
228 }

230 // Callback for getting the text of the current variant
231 FText HandleVariantSelectorText() const
232 {
233 ILauncherDeviceGroupPtr ActiveGroup = DeviceGroup.Get();
234 if (ActiveGroup.IsValid() && DeviceProxy.IsValid())
235 {
236 if (SelectedVariant == NAME_None)
237 {
238 return LOCTEXT("DefaultVariant", "Default");

240 return FText::FromName(SelectedVariant);

242
243 return FText::GetEmpty();

246 // Callback for changing the selected variant
247 void HandleVariantSelectorVariantSelected(FName InVariant)
248 {
249 if (DeviceProxy.IsValid() && DeviceProxy->HasVariant(SelectedVariant) && DeviceProxy->HasVariant(InVariant))
250 {
251 ILauncherDeviceGroupPtr ActiveGroup = DeviceGroup.Get();
252 const FString& DeviceID = DeviceProxy->GetTargetDeviceId(SelectedVariant);
253 if (ActiveGroup.IsValid() && ActiveGroup->GetDeviceIDs().Contains(DeviceID))
254 {
255 const FString& OldDeviceID = DeviceProxy->GetTargetDeviceId(SelectedVariant);
256 const FString& NewDeviceID = DeviceProxy->GetTargetDeviceId(InVariant);
257
258 if (ActiveGroup.IsValid())
259 {
260 ActiveGroup->RemoveDevice(OldDeviceID);
261 ActiveGroup->AddDevice(NewDeviceID);
262 }
263 }
264
265 SelectedVariant = InVariant;
266 }
267 }
