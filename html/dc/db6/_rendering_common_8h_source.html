380struct FShortRect
381{
382 FShortRect()
383 : Left(0)
384 , Top(0)
385 , Right(0)
386 , Bottom(0)
387 {
388 }
389
390 FShortRect(uint16 InLeft, uint16 InTop, uint16 InRight, uint16 InBottom)
391 : Left(InLeft)
392 , Top(InTop)
393 , Right(InRight)
394 , Bottom(InBottom)
395 {
396 }
397
398 explicit FShortRect(const FSlateRect& Rect)
399 : Left((uint16)FMath::Clamp(Rect.Left, 0.0f, 65535.0f))
400 , Top((uint16)FMath::Clamp(Rect.Top, 0.0f, 65535.0f))
401 , Right((uint16)FMath::Clamp(Rect.Right, 0.0f, 65535.0f))
402 , Bottom((uint16)FMath::Clamp(Rect.Bottom, 0.0f, 65535.0f))
403 {
404 }
405
406 bool operator==(const FShortRect& RHS) const { return Left == RHS.Left && Top == RHS.Top && Right == RHS.Right && Bottom == RHS.Bottom; }
407 bool operator!=(const FShortRect& RHS) const { return !(*this == RHS); }
408 bool DoesIntersect( const FShortRect& B ) const
409 {
410 const bool bDoNotOverlap =
411 B.Right < Left || Right < B.Left ||
412 B.Bottom < Top || Bottom < B.Top;
413
414 return ! bDoNotOverlap;
415 }
416
417 bool DoesIntersect(const FSlateRect& B) const
418 {
419 const bool bDoNotOverlap =
420 B.Right < Left || Right < B.Left ||
421 B.Bottom < Top || Bottom < B.Top;
422
423 return !bDoNotOverlap;
424 }
425
426 FVector2f GetTopLeft() const { return FVector2f(Left, Top); }
427 FVector2f GetBottomRight() const { return FVector2f(Right, Bottom); }
428
429 uint16 Left;
430 uint16 Top;
431 uint16 Right;
432 uint16 Bottom;
433};
