16FORCEINLINE void* FMemory_MallocInline(SIZE_T Count, uint32 Alignment)
17{
18 // AutoRTFM: For non-transactional code, all of these calls optimize away and the
19 // behavior is the same as it always has been.
20 // For transactional code, we call the allocator in the 'open' as an optimization, so that
21 // we don't end up keeping track of the writes to the allocator's internal data structures.
22 // This is because allocators are already transactional - malloc can be rolled back by
23 // calling free.
24 void* Ptr = nullptr;
25
26 UE_AUTORTFM_OPEN_NO_VALIDATION
27 {
28 if (UNLIKELY(!FMEMORY_INLINE_GMalloc))
29 {
30 Ptr = FMemory::MallocExternal(Count, Alignment);
31 }
32 else
33 {
34 DoGamethreadHook(0);
35 FScopedMallocTimer Timer(0);
36 Ptr = FMEMORY_INLINE_GMalloc->Malloc(Count, Alignment);
37 }
38 // optional tracking of every allocation
39 LLM_IF_ENABLED(FLowLevelMemTracker::Get().OnLowLevelAlloc(ELLMTracker::Default, Ptr, Count, ELLMTag::Untagged, ELLMAllocType::FMalloc));
40 };
41
42 // AutoRTFM: This is a no-op for non-transactional code.
43 // For transactional code, this defers a call to Free if the transaction aborts,
44 // so that rolling back this allocation will end up freeing the memory.
45 UE_AUTORTFM_ONABORT(Ptr)
46 {
47 // Disable the code analysis warning that complains that Free is being passed
48 // a pointer that may be null. Free explicitly handles this case already.
49 FMemory::Free(Ptr); //-V575
50 };
51
52 return AutoRTFM::DidAllocate(Ptr, Count);
53}

55FORCEINLINE void* FMemory_ReallocInline(void* Original, SIZE_T Count, uint32 Alignment)
56{
57 if (AutoRTFM::IsClosed())
58 {
59 // AutoRTFM: For transactional code, we have to do a little dance to handle Realloc
60 // properly. We turn realloc into Malloc + Memcpy + Free and never call into the
61 // underlying allocator's realloc implementation. That's required, because if we
62 // were to call into actual realloc, it could end up freeing the old memory. And
63 // if we then aborted our transaction, we would end up rolling back pointers to
64 // point to that old allocation, and it wouldn't be possible to get back that original
65 // allocation at the same address.
66 //
67 // There is an opportunity here, in that if the original pointer was allocated within
68 // this transaction, then the above doesn't apply since rolling back the transaction
69 // would free that memory and the resulting heap wouldn't point at that memory. So
70 // if we new that the Original pointer was allocated in this transaction, we could
71 // call into the underlying realloc - however we would also have to account for the
72 // malloc deferring a call to free, so we would also have to erase that call to free.
73
74 void* Ptr = nullptr;
75
76 // Depending on the underlying implementation `Malloc` here, even if `Count` is zero,
77 // could do an actual allocation (it is implementation-defined what occurs). So
78 // instead, since we are fine to return null with a `Count` of zero, we check for
79 // that case and skip the `Malloc` call entirely.
80 if (Count > 0)
81 {
82 Ptr = FMemory::Malloc(Count, Alignment);
83
84 if (!Ptr)
85 {
86 return nullptr;
87 }
88 }
89
90 if (Original)
91 {
92 if (Ptr)
93 {
94 SIZE_T OriginalCount = FMemory::GetAllocSize(Original);
95 SIZE_T CopyCount = FGenericPlatformMath::Min(Count, OriginalCount); // handle the case where the new size is smaller
96
97 FMemory::Memcpy(Ptr, Original, CopyCount);
98 }
99
100 FMemory::Free(Original);
101 }
102
103 return Ptr;
104 }
105
106 // optional tracking -- a realloc with an Original pointer of null is equivalent
107 // to malloc() so there's nothing to free
108 LLM_REALLOC_SCOPE(Original);
109 LLM_IF_ENABLED(if (Original != nullptr) FLowLevelMemTracker::Get().OnLowLevelFree(ELLMTracker::Default, Original, ELLMAllocType::FMalloc));
110
111 void* Ptr;
112 if (UNLIKELY(!FMEMORY_INLINE_GMalloc))
113 {
114 Ptr = FMemory::ReallocExternal(Original, Count, Alignment);
115 }
116 else
117 {
118 DoGamethreadHook(1);
119 FScopedMallocTimer Timer(1);
120 Ptr = FMEMORY_INLINE_GMalloc->Realloc(Original, Count, Alignment);
121 }
122
123 if (Ptr != Original)
124 {
125 // If the pointer we've got back from realloc is new memory, we are
126 // assuming the old memory was free'd.
127 AutoRTFM::DidFree(Original);
128 }
129
130 // optional tracking of every allocation - a realloc with a Count of zero is equivalent to a call
131 // to free() and will return a null pointer which does not require tracking. If realloc returns null
132 // for some other reason (like failure to allocate) there's also no reason to track it
133 LLM_IF_ENABLED(if (Ptr != nullptr) FLowLevelMemTracker::Get().OnLowLevelAlloc(ELLMTracker::Default, Ptr, Count, ELLMTag::Untagged, ELLMAllocType::FMalloc));
134
135 return Ptr;
136}

138FORCEINLINE void FMemory_FreeInline(void* Original)
139{
140 if (!Original)
141 {
142 FScopedMallocTimer Timer(3);
143 return;
144 }
145
146 // AutoRTFM: For transactional code, in order to support the transaction
147 // aborting and needing to 'roll back' the Free, we defer the actual
148 // free until commit time.
149 UE_AUTORTFM_ONCOMMIT(=)
150 {
151 // optional tracking of every allocation
152 LLM_IF_ENABLED(FLowLevelMemTracker::Get().OnLowLevelFree(ELLMTracker::Default, Original, ELLMAllocType::FMalloc));
153
154 if (UNLIKELY(!FMEMORY_INLINE_GMalloc))
155 {
156 FMemory::FreeExternal(Original);
157 return;
158 }
159 DoGamethreadHook(2);
160 FScopedMallocTimer Timer(2);
161 FMEMORY_INLINE_GMalloc->Free(Original);
162
163 AutoRTFM::DidFree(Original);
164 };
165}

167FORCEINLINE SIZE_T FMemory_GetAllocSizeInline(void* Original)
168{
169 const bool bWasClosed = AutoRTFM::IsClosed();
170
171 SIZE_T Result;
172
173 UE_AUTORTFM_OPEN_NO_VALIDATION
174 {
175 if (UNLIKELY(!FMEMORY_INLINE_GMalloc))
176 {
177 Result = FMemory::GetAllocSizeExternal(Original);
178 }
179 else
180 {
181 SIZE_T Size = 0;
182 const bool bGotSize = FMEMORY_INLINE_GMalloc->GetAllocationSize(Original, Size);
183 Result = bGotSize ? Size : 0;
184
185 // This folds away at compile time so that the check is only ever performed inside transactional
186 // code paths. The check is to ensure that the allocator used will return the correct allocation
187 // size, which is a cornerstone requirement for AutoRTFM to function.
188 if (bWasClosed)
189 {
190 checkf(bGotSize, TEXT("For AutoRTFM to function it must be able to get the size of an allocation"));
191 }
192 }
193 };
194
195 return Result;
196}
