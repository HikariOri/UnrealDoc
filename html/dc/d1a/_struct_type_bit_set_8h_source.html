22struct FConstBitSetContainer : TBitArray<TInlineAllocator<8>>

32 bool HasAll(const FBitArrayBase& Other) const
33 {
34 FConstWordIterator ThisIterator(*this);
35 FConstWordIterator OtherIterator(Other);
36
37 while (ThisIterator || OtherIterator)
38 {
39 const uint32 A = ThisIterator ? ThisIterator.GetWord() : 0;
40 const uint32 B = OtherIterator ? OtherIterator.GetWord() : 0;
41 if ((A & B) != B)
42 {
43 return false;
44 }
45
46 ++ThisIterator;
47 ++OtherIterator;
48 }
49
50 return true;
51 }
52
57 bool HasAny(const FBitArrayBase& Other) const
58 {
59 FConstWordIterator ThisIterator(*this);
60 FConstWordIterator OtherIterator(Other);
61
62 while (ThisIterator || OtherIterator)
63 {
64 const uint32 A = ThisIterator ? ThisIterator.GetWord() : 0;
65 const uint32 B = OtherIterator ? OtherIterator.GetWord() : 0;
66 if ((A & B) != 0)
67 {
68 return true;
69 }
70
71 ++ThisIterator;
72 ++OtherIterator;
73 }
74
75 return false;
76 }
77
78 FConstBitSetContainer operator&(const FBitArrayBase& Other) const
79 {
80 const int32 MinNumBits = FMath::Min(Num(), Other.Num());
81 if (MinNumBits > 0)
82 {
83 FBitArrayBase Result(false, MinNumBits);
84
85 FConstWordIterator IteratorA(*this);
86 FConstWordIterator IteratorB(Other);
87
88 FWordIterator IteratorResult(Result);
89
90 for ( ; IteratorResult; ++IteratorResult, ++IteratorA, ++IteratorB)
91 {
92 const uint32 NewValue = IteratorA.GetWord() & IteratorB.GetWord();
93 IteratorResult.SetWord(NewValue);
94 }
95
96 return FConstBitSetContainer(MoveTemp(Result));
97 }
98
99 return {};
100 }
101
102 FConstBitSetContainer operator|(const FBitArrayBase& Other) const
103 {
104 const int32 MaxNumBits = FMath::Max(Num(), Other.Num());
105 if (MaxNumBits > 0)
106 {
107 FBitArrayBase Result(false, MaxNumBits);
108
109 FConstWordIterator IteratorA(*this);
110 FConstWordIterator IteratorB(Other);
111
112 FWordIterator IteratorResult(Result);
113
114 for (; IteratorA && IteratorB; ++IteratorResult, ++IteratorA, ++IteratorB)
115 {
116 const uint32 NewValue = IteratorA.GetWord() | IteratorB.GetWord();
117 IteratorResult.SetWord(NewValue);
118 }
119 for (; IteratorA; ++IteratorResult, ++IteratorA)
120 {
121 const uint32 NewValue = IteratorA.GetWord();
122 IteratorResult.SetWord(NewValue);
123 }
124 for (; IteratorB; ++IteratorResult, ++IteratorB)
125 {
126 const uint32 NewValue = IteratorB.GetWord();
127 IteratorResult.SetWord(NewValue);
128 }
129
130 return FConstBitSetContainer(MoveTemp(Result));
131 }
132
133 return {};
134 }

136 FConstBitSetContainer AndNot(const FBitArrayBase& Other) const
137 {
138 const int32 NumResultBits = Num();
139 if (NumResultBits > 0)
140 {
141 FBitArrayBase Result(false, NumResultBits);
142
143 FConstWordIterator IteratorA(*this);
144 FConstWordIterator IteratorB(Other);
145
146 FWordIterator IteratorResult(Result);
147
148 for ( ; IteratorResult; ++IteratorResult, ++IteratorA, ++IteratorB)
149 {
150 const uint32 NewValue = IteratorB
151 ? IteratorA.GetWord() & ~IteratorB.GetWord()
152 : IteratorA.GetWord();
153 IteratorResult.SetWord(NewValue);
154 }
155
156 return FConstBitSetContainer(MoveTemp(Result));
157 }
158
159 return {};
160 }

190 friend uint32 GetTypeHash(const FConstBitSetContainer& Instance)
191 {
192 FConstWordIterator Iterator(Instance);
193 uint32 Hash = 0;
194 uint32 TrailingZeroHash = 0;
195 while (Iterator)
196 {
197 const uint32 Word = Iterator.GetWord();
198 if (Word)
199 {
200 Hash = HashCombine(TrailingZeroHash ? TrailingZeroHash : Hash, Word);
201 TrailingZeroHash = 0;
202 }
203 else // potentially a trailing 0-word
204 {
205 TrailingZeroHash = HashCombine(TrailingZeroHash ? TrailingZeroHash : Hash, Word);
206 }
207 ++Iterator;
208 }
209 return Hash;
210 }

298 {

300 FConstWordIterator OtherIterator(Other);
301

303 {
304 ThisIterator.SetWord(ThisIterator.GetWord() & ~OtherIterator.GetWord());

312 FBitSetContainer& operator+=(const FBitArrayBase& Other)
313 {
314 // make sure we have enough space for all the contents of Other
315 // not that it's fine to pass in a negative number, TBitArray::Add handles that gracefully.
316 Add(false, Other.Num() - Num());
317
318 FWordIterator ThisIterator(*this);
319 FConstWordIterator OtherIterator(Other);
320
321 while (OtherIterator)
322 {
323 ThisIterator.SetWord(ThisIterator.GetWord() | OtherIterator.GetWord());
324
325 ++ThisIterator;
326 ++OtherIterator;
327 }
328 return *this;

363 {
364 uint32& Word = GetData()[Index / NumBitsPerDWORD];

786 bool IsEquivalentFiltered(const TImplementation& Other, const TImplementation& Filter) const
787 {
788 constexpr uint32 MissingBitsFill = 0;
789
790 FBitArrayBase::FConstWordIterator ThisIterator(StructTypesBitArray);
791 FBitArrayBase::FConstWordIterator OtherIterator(Other.StructTypesBitArray);
792 FBitArrayBase::FConstWordIterator FilterIterator(Filter.StructTypesBitArray);
793
794 ThisIterator.FillMissingBits(MissingBitsFill);
795 OtherIterator.FillMissingBits(MissingBitsFill);
796 FilterIterator.FillMissingBits(MissingBitsFill);
797
798 while (FilterIterator && (ThisIterator || OtherIterator))
799 {
800 const uint32 A = ThisIterator ? (ThisIterator.GetWord() & FilterIterator.GetWord()) : MissingBitsFill;
801 const uint32 B = OtherIterator ? (OtherIterator.GetWord() & FilterIterator.GetWord()) : MissingBitsFill;
802 if (A != B)
803 {
804 return false;
805 }
806
807 ++ThisIterator;
808 ++OtherIterator;
809 ++FilterIterator;
810 }
811
812 return true;
813 }
