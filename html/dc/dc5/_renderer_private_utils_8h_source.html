291template <typename InValueType, typename InBufferTraits, int32 InNumValuesPerScatter = 1>
292class TBufferScatterUploader : public FBufferScatterUploader
293{
294public:
295 using BufferTraits = InBufferTraits;
296 using ValueType = InValueType;
297 static constexpr int32 NumBytesPerElement = BufferTraits::OverrideBytesPerElement != 0 ? BufferTraits::OverrideBytesPerElement : sizeof(ValueType);
298 static constexpr int32 NumValuesPerScatter = InNumValuesPerScatter;
299
300 static_assert((NumBytesPerElement % 4) == 0, "The struct used must be 4-byte aligned");
301
305 TBufferScatterUploader(int32 NumScatters = 0)
306 {
307 Reserve(NumScatters);
308 }
309
310 virtual ~TBufferScatterUploader() = default;
311
315 void Reserve(int32 NumScatters)
316 {
317 check(UploadDataProxy == nullptr);
318
319 UploadData.Values.Reserve(NumScatters * NumValuesPerScatter);
320 UploadData.ScatterOffsets.Reserve(NumScatters);
321 }
322
326 void Add(const ValueType &Value, int32 ScatterOffset)
327 {
328 check(1 == NumValuesPerScatter);
329 check(UploadDataProxy == nullptr);
330
331 UploadData.Values.Add(Value);
332 UploadData.ScatterOffsets.Add(ScatterOffset);
333 }
334
339 void Add(const TConstArrayView<ValueType> &InValues, int32 ScatterOffset)
340 {
341 check(InValues.Num() == NumValuesPerScatter);
342 check(UploadDataProxy == nullptr);
343
344 UploadData.Values.Append(InValues);
345 UploadData.ScatterOffsets.Add(ScatterOffset);
346 }
347
352 void AddMultiple(const TConstArrayView<ValueType> &InValues, int32 FirstScatterOffset)
353 {
354 check(InValues.Num() % NumValuesPerScatter == 0);
355 check(UploadDataProxy == nullptr);
356
357 UploadData.Values.Append(InValues);
358 for (int32 Index = 0; Index < InValues.Num() / NumValuesPerScatter; ++Index)
359 {
360 UploadData.ScatterOffsets.Add(FirstScatterOffset + Index);
361 }
362 }
363
368 TArrayView<ValueType> AddMultiple_GetRef(int32 FirstScatterOffset, int32 NumValues)
369 {
370 check(NumValues % NumValuesPerScatter == 0);
371 check(UploadDataProxy == nullptr);
372
373 int32 SrcOffset = UploadData.Values.Num();
374 UploadData.Values.AddUninitialized(NumValues);
375 for (int32 Index = 0; Index < NumValues / NumValuesPerScatter; ++Index)
376 {
377 UploadData.ScatterOffsets.Add(FirstScatterOffset + Index);
378 }
379
380 return MakeArrayView(&UploadData.Values[SrcOffset], NumValues);
381 }
382
383 int32 GetNumScatters() const { return UploadDataProxy != nullptr ? UploadDataProxy->ScatterOffsets.Num() : UploadData.ScatterOffsets.Num(); }
384
389 FRDGBuffer *ResizeAndUploadTo(FRDGBuilder& GraphBuilder, TPersistentBuffer<ValueType, BufferTraits> &DestDataBuffer, int32 DestDataMinimumSize, FScatterInfo &OutScatterInfo)
390 {
391 check(UploadDataProxy == nullptr);
392
393 FRDGBuffer *DestBufferRDG = DestDataBuffer.ResizeBufferIfNeeded(GraphBuilder, DestDataMinimumSize);
394
395 OutScatterInfo.NumScatters = UploadData.ScatterOffsets.Num();
396 OutScatterInfo.ScatterOffsetsRDG = nullptr;
397 if (OutScatterInfo.NumScatters != 0u)
398 {
399 // Move the data arrays to a proxy object owned by RDG to guarantee life-times for upload.
400 UploadDataProxy = GraphBuilder.AllocObject<FUploadData>(MoveTemp(UploadData));
401
402 // upload the values & offsets
403 OutScatterInfo.ScatterOffsetsRDG = BufferTraits::CreateOffsetUploadBuffer(GraphBuilder, TEXT("ScatterUploader.Offsets"), UploadDataProxy->ScatterOffsets);
404 FRDGBuffer *ValuesRDG = BufferTraits::CreateValueUploadBuffer(GraphBuilder, TEXT("ScatterUploader.Values"), UploadDataProxy->Values);
405
406 uint32 ElementSize = NumBytesPerElement;
407 uint32 ElementsPerScatter = NumValuesPerScatter;
408 if (BufferTraits::bAutoValuesPerScatter)
409 {
410 // Let the implementation determine the optimal way to divide up the scatter
411 ElementSize *= ElementsPerScatter;
412 ElementsPerScatter = INDEX_NONE;
413 }
414 FBufferScatterUploader::UploadTo(GraphBuilder, DestBufferRDG, OutScatterInfo.ScatterOffsetsRDG, ValuesRDG, OutScatterInfo.NumScatters, ElementSize, ElementsPerScatter, BufferTraits::PixelFormat);
415 }
416
417 return DestBufferRDG;
418 }
