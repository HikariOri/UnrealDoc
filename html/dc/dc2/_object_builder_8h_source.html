172 template <typename T>
173 bool IsCompatibleWith(const FProperty& Prop) const
174 {
175 using Raw = std::decay_t<T>;
176 if constexpr (std::is_pointer_v<T> && std::is_base_of_v<UObject, std::remove_pointer_t<Raw>>)
177 {
178 if (auto AsProperty = CastField<FObjectProperty>(&Prop))
179 {
180 return std::remove_pointer_t<Raw>::StaticClass()->IsChildOf(AsProperty->PropertyClass);
181 }
182 }
183 else if constexpr (TIsTObjectPtr<Raw>::Value)
184 {
185 if (auto AsProperty = CastField<FObjectProperty>(&Prop))
186 {
187 return Raw::ElementType::StaticClass()->IsChildOf(AsProperty->PropertyClass);
188 }
189 }
190 else if constexpr (TIsTArray<Raw>::Value)
191 {
192 if (auto AsProperty = CastField<FArrayProperty>(&Prop))
193 {
194 return IsCompatibleWith<typename Raw::ElementType>(*AsProperty->Inner);
195 }
196 }
197 else if constexpr (TIsTSet_V<Raw>)
198 {
199 if (auto AsProperty = CastField<FSetProperty>(&Prop))
200 {
201 return IsCompatibleWith<typename Raw::ElementType>(*AsProperty->ElementProp);
202 }
203 }
204 else if constexpr (TIsTMap_V<Raw>)
205 {
206 if (auto AsProperty = CastField<FMapProperty>(&Prop))
207 {
208 using Key = typename Raw::KeyInitType;
209 using Value = typename Raw::ValueInitType;
210 return IsCompatibleWith<Key>(*AsProperty->KeyProp) && IsCompatibleWith<Value>(*AsProperty->ValueProp);
211 }
212 }
213 else if constexpr (TIsEnum<Raw>::Value)
214 {
215 if (auto AsProperty = CastField<FEnumProperty>(&Prop))
216 {
217 return AsProperty->GetUnderlyingProperty()->GetSize() == sizeof(T);
218 }
219 }
220 else if constexpr (std::is_base_of_v<UObject, Raw>)
221 {
222 if (auto AsProperty = CastField<FObjectProperty>(&Prop))
223 {
224 return Raw::StaticClass()->IsChildOf(AsProperty->PropertyClass);
225 }
226 }
227 else if constexpr (std::is_same_v<FVector, Raw>)
228 {
229 if (auto AsProperty = CastField<FStructProperty>(&Prop))
230 {
231 return AsProperty->Struct->GetFName() == NAME_Vector;
232 }
233 }
234 else if constexpr (std::is_same_v<FStructProperty, typename TPropType<Raw>::Value>)
235 {
236 if (auto AsProperty = CastField<FStructProperty>(&Prop))
237 {
238 return AsProperty->Struct->GetName() == Raw::StaticStruct()->GetName();
239 }
240 }
241 else
242 {
243 using PropType = typename TPropType<Raw>::Value;
244
245 if (auto AsProperty = CastField<PropType>(&Prop))
246 {
247 return PropType::StaticClass()->IsChildOf(AsProperty->GetClass());
248 }
249 }
250
251 return false;
252 }

254 template<typename T>
255 FString GetTypeName() const
256 {
257 using Raw = std::decay_t<T>;
258 if constexpr (std::is_pointer_v<Raw> && std::is_base_of_v<UObject, std::remove_pointer_t<Raw>>)
259 {
260 return FString::Printf(TEXT("%s*"), *std::remove_pointer_t<Raw>::StaticClass()->GetName());
261 }
262 else if constexpr (TIsTObjectPtr<Raw>::Value)
263 {
264 using PtrType = typename Raw::ElementType;
265 return FString::Printf(TEXT("TObjectPtr<%s>"), *GetTypeName<PtrType>());
266 }
267 else if constexpr (TIsTArray<Raw>::Value)
268 {
269 using ElementType = typename Raw::ElementType;
270 return FString::Printf(TEXT("TArray<%s>"), *GetTypeName<ElementType>());
271 }
272 else if constexpr (TIsTSet<Raw>::Value)
273 {
274 using ElementType = typename Raw::ElementType;
275 return FString::Printf(TEXT("TSet<%s>"), *GetTypeName<ElementType>());
276 }
277 else if constexpr (TIsTMap<Raw>::Value)
278 {
279 using Key = typename Raw::KeyInitType;
280 using Value = typename Raw::ValueInitType;
281 return FString::Printf(TEXT("TMap<%s, %s>"), *GetTypeName<Key>(), *GetTypeName<Value>());
282 }
283 else if constexpr (TIsEnum<Raw>::Value)
284 {
285 return TEXT("EnumProperty");
286 }
287 else if constexpr (std::is_same_v<FVector, Raw>)
288 {
289 return TEXT("FVector");
290 }
291 else if constexpr (std::is_base_of_v<UObject, Raw>)
292 {
293 return FString::Printf(TEXT("%s"), *Raw::StaticClass()->GetName());
294 }
295 else if constexpr (std::is_same_v<FStructProperty, typename TPropType<Raw>::Value>)
296 {
297 return Raw::StaticStruct()->GetName();
298 }
299 else
300 {
301 return TNameOf<Raw>::GetName();
302 }
303 }
