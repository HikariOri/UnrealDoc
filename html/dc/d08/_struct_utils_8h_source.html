116 inline void CalcMidIndexAndCount(int32 ArrayNum, int32& InOutIndex, int32& InOutCount)
117 {
118 // Clamp minimum index at the start of the range, adjusting the length down if necessary
119 const int32 NegativeIndexOffset = (InOutIndex < 0) ? InOutIndex : 0;
120 InOutCount += NegativeIndexOffset;
121 InOutIndex -= NegativeIndexOffset;
122
123 // Clamp maximum index at the end of the range
124 InOutIndex = (InOutIndex > ArrayNum) ? ArrayNum : InOutIndex;
125
126 // Clamp count between 0 and the distance to the end of the range
127 InOutCount = FMath::Clamp(InOutCount, 0, (ArrayNum - InOutIndex));
128 }
