218template< class T >
219inline void ComputeCurveTangent( float PrevTime, const T& PrevPoint,
220 float CurTime, const T& CurPoint,
221 float NextTime, const T& NextPoint,
222 float Tension,
223 bool bWantClamping,
224 T& OutTangent )
225{
226 // NOTE: Clamping not supported for non-float vector types (bWantClamping is ignored)
227
228 AutoCalcTangent( PrevPoint, CurPoint, NextPoint, Tension, OutTangent );
229
230 const float PrevToNextTimeDiff = FMath::Max< float >( UE_KINDA_SMALL_NUMBER, NextTime - PrevTime );
231
232 OutTangent /= PrevToNextTimeDiff;
233}

240template< class T >
241inline void ComputeClampableFloatVectorCurveTangent( float PrevTime, const T& PrevPoint,
242 float CurTime, const T& CurPoint,
243 float NextTime, const T& NextPoint,
244 float Tension,
245 bool bWantClamping,
246 T& OutTangent )
247{
248 // Clamp the tangents if we need to do that
249 if( bWantClamping )
250 {
251 // NOTE: We always treat the type as an array of floats
252 float* PrevPointVal = ( float* )&PrevPoint;
253 float* CurPointVal = ( float* )&CurPoint;
254 float* NextPointVal = ( float* )&NextPoint;
255 float* OutTangentVal = ( float* )&OutTangent;
256 for( int32 CurValPos = 0; CurValPos < sizeof( T ); CurValPos += sizeof( float ) )
257 {
258 // Clamp it!
259 const float ClampedTangent =
260 ClampFloatTangent(
261 *PrevPointVal, PrevTime,
262 *CurPointVal, CurTime,
263 *NextPointVal, NextTime );
264
265 // Apply tension value
266 *OutTangentVal = ( 1.0f - Tension ) * ClampedTangent;
267
268
269 // Advance pointers
270 ++OutTangentVal;
271 ++PrevPointVal;
272 ++CurPointVal;
273 ++NextPointVal;
274 }
275 }
276 else
277 {
278 // No clamping needed
279 AutoCalcTangent( PrevPoint, CurPoint, NextPoint, Tension, OutTangent );
280
281 const float PrevToNextTimeDiff = FMath::Max< float >( UE_KINDA_SMALL_NUMBER, NextTime - PrevTime );
282
283 OutTangent /= PrevToNextTimeDiff;
284 }
285}
