22template<typename... ArgTypes>
23FString CaptureExpressionsAndValues(const FString& InExpressions, ArgTypes&&... InExpressionsValues)
24{
25 auto Args = { InExpressionsValues... };
26 auto Iter = Args.begin();
27
28 TStringBuilder<64> Result;
29
30 FString RemainingExpressions = InExpressions;
31 FString Expression;
32 while (RemainingExpressions.Split(TEXT(","), &Expression, &RemainingExpressions))
33 {
34 if (Iter == Args.end())
35 {
36 break;
37 }
38 else if (Iter != Args.begin())
39 {
40 Result << TEXT(", ");
41 }
42 Result << Expression.TrimStartAndEnd() << TEXT(" = ") << *Iter++;
43 }
44
45 if (Iter != Args.end() && !RemainingExpressions.TrimStartAndEnd().IsEmpty())
46 {
47 Result << TEXT(", ") << RemainingExpressions << TEXT(" = ") << *Iter;
48 }
49
50 Result << TEXT("\n");
51 return FString(Result.ToString());
52}
