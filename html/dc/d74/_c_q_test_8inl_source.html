176template <typename AsserterType>
177inline bool TTestRunner<AsserterType>::RunTest(const FString& RequestedTest)
178{
179 if (RequestedTest.Len() == 0)
180 {
181 return false;
182 }
183
184 CurrentTestPtr = TestInstanceFactory(*this);
185 check(CurrentTestPtr != nullptr);
186 auto& CurrentTest = *CurrentTestPtr;
187
188 TSharedRef<FRunSequence> CommandSequence = MakeShareable<FRunSequence>(new FRunSequence());
189 auto Before = MakeShared<TBeforeTestCommand<AsserterType>>(CurrentTest, CommandSequence);
190 auto Run = MakeShared<TRunTestCommand<AsserterType>>(CurrentTest, RequestedTest, *this, CommandSequence);
191 auto After = MakeShared<TAfterTestCommand<AsserterType>>(CurrentTest, CommandSequence);
192 auto TearDown = MakeShared<TTearDownRunner<AsserterType>>(*this);
193
194 auto RemainingSteps = TArray<TSharedPtr<IAutomationLatentCommand>>{ Before, Run, After, TearDown };
195
196 while (!RemainingSteps.IsEmpty())
197 {
198 RemainingSteps[0]->Update();
199 RemainingSteps.RemoveAt(0);
200 if (CurrentTestPtr != nullptr && !CommandSequence->IsEmpty())
201 {
202 CommandSequence->AppendAll(RemainingSteps);
203 FAutomationTestFramework::Get().EnqueueLatentCommand(CommandSequence);
204 return true;
205 }
206 }
207
208 return true;
209}
