23inline FVector3d xyYToXYZ(const FVector3d& xyY)
24{
25 double Divisor = FMath::Max(xyY[1], 1e-10);
26
27 return FVector3d(
28 xyY[0] * xyY[2] / Divisor,
29 xyY[2],
30 (1.0 - xyY[0] - xyY[1]) * xyY[2] / Divisor
31 );
32}

39inline FVector3d XYZToxyY(const FVector3d& XYZ)
40{

42 if (Divisor == 0.0)

44 Divisor = 1e-10;
