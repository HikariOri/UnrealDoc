46 struct RangedForIterator
47 {
48 RangedForIterator(const VMapBase* Map, uint32 Index)
49 : Map(Map)
50 , Index(Index) {}
51 TPair<VValue, VValue> operator*() const
52 {
53 const PairType& Pair = Map->GetPairTable()[Map->GetSequenceTable()[Index]];
54 return {Pair.Key.Get().Follow(), Pair.Value.Get().Follow()};
55 }
56 bool operator==(const RangedForIterator& Rhs) const { return Index == Rhs.Index; }
57 bool operator!=(const RangedForIterator& Rhs) const { return Index != Rhs.Index; }
58 RangedForIterator& operator++()
59 {
60 ++Index;
61 return *this;
62 }
63
64 const VMapBase* Map;
65 uint32 Index;
66 };
