68// Extra data for an encoded texture.
69// This is filled out by platform "child" formats that can potentially tile or otherwise
70// reorganize the mip data for use on the target platform. For platforms that use bog standard
71// "linear" texture data, this is mostly zeroes, but must still be "Valid" data returned by the
72// GetExtendedDataForTexture - notably the MipSizesInBytes.
73struct FEncodedTextureExtendedData
74{
75 // Copied from RHIDefinitions.h - we expose here to avoid the include.
76 static constexpr int32 MAX_TEXTURE_MIP_COUNT=15;
77
78 // Some platforms group the smallest mips in to a single bulk chunk of data. This is nonzero
79 // in such cases and the MipSizesInBytes array is truncated.
80 int32 NumMipsInTail = 0;
81
82 // ExtData is metadata the tiler passes through to the runtime.
83 uint32 ExtData = 0;
84
85 // If true, the mip data can't be assumed to be linear.
86 bool bIsTiled = false;
87
88 // If true, this texture might change layouts if top mips are striped (i.e. LODBias is not zero).
89 bool bSensitiveToLODBias = false;
90
91 // If bSensitiveToLODBias is set, this is the LODBias for this layout.
92 int8 LODBiasIfSensitive = 0;
93
94 // All generators of this structure are required to fill this structure out even if they are
95 // default linear sizes. For non linear textures these are not trivially calculatable.
96 // Also note that if the texture is sensitive to LODBias with respect to tiling, the indeces in this
97 // array that are for mips that would be stripped are present but should be ignored as they don't
98 // represent anything we can accurately calculate.
99 TArray<uint64, TInlineAllocator<MAX_TEXTURE_MIP_COUNT>> MipSizesInBytes;
100};

122static int32 GetNumStreamingMipsDirect(int32 InNumMips, bool bInCubeMap, bool bInVolumeTexture, bool bInTextureArray, const FEncodedTextureExtendedData* InExtendedData, const FTextureEngineParameters& InEngineParameters)
123{
124 bool bAllowStreaming = true;
125 {
126 const bool bDisableStreaming = GetStreamingDisabledForNonVirtualTextureProperties(bInCubeMap, bInVolumeTexture, bInTextureArray, InEngineParameters);
127 if (bDisableStreaming)
128 bAllowStreaming = false;
129 }
130
131 int32 NumStreamingMips = 0;
132 if (bAllowStreaming)
133 {
134 // Some platforms pack several mips in to a single entry. If this is the case,
135 // those must be non-streaming.
136 int32 NumMipsInTail = 0;
137 if (InExtendedData)
138 {
139 NumMipsInTail = InExtendedData->NumMipsInTail;
140 }
141
142 int32 NumInlineMips = FMath::Max(NumMipsInTail, InEngineParameters.NumInlineDerivedMips);
143 NumStreamingMips = FMath::Max(0, InNumMips - NumInlineMips);
144 }
145 return NumStreamingMips;
146}

157static int32 GetNumNonStreamingMipsDirect(int32 InNumMips, int32 InTopMipSizeX, int32 InTopMipSizeY, EPixelFormat InPixelFormat, int32 InNumMipsInTail, int32 InNumInlineDerivedMips)
158{
159 if (InNumMips <= 1)
160 {
161 return InNumMips;
162 }
163
164 int32 NumNonStreamingMips = 1;
165
166 // Take the min resident limit into account.
167 if (InNumMipsInTail)
168 {
169 // If we have a tail, it can't be broken up so the whole thing must be non streaming.
170 NumNonStreamingMips = FMath::Max(NumNonStreamingMips, InNumMipsInTail);
171 }
172 NumNonStreamingMips = FMath::Max(NumNonStreamingMips, InNumInlineDerivedMips);
173 NumNonStreamingMips = FMath::Min(NumNonStreamingMips, InNumMips);
174
175 if ( RequiresBlock4Alignment(InPixelFormat) )
176 {
177 // ensure the top non-streamed mip (and all streamed mips) size is >= BlockSize (and a multiple of block size!)
178
179 // note: this is not right for non pow 2; NeverStream should set !bIsStreamingPossible in that case
180 if ( FMath::IsPowerOfTwo(InTopMipSizeX) && FMath::IsPowerOfTwo(InTopMipSizeY) )
181 {
182 // This is not right for textures smaller than 4x4, however such textures are already non-streaming due to
183 // NumInlineDerivedMips (which is functionally hard coded to 7).
184 NumNonStreamingMips = FMath::Max<int32>(NumNonStreamingMips, InNumMips - FPlatformMath::FloorLog2(InTopMipSizeX / 4));
185 NumNonStreamingMips = FMath::Max<int32>(NumNonStreamingMips, InNumMips - FPlatformMath::FloorLog2(InTopMipSizeY / 4));
186 }
187 else
188 {
189 // should never get here as streaming should be disabled in this case, so we should not hit this branch, but it's not reliable
190 NumNonStreamingMips = InNumMips;
191 }
192 }
193
194 return NumNonStreamingMips;
195}

230 // This returns the SizeZ value that is expected by RHI streamable texture structures. It
231 // is only used by non-cube texture arrays and volumes, however the cubemap array and cubemap
232 // values need to be consistent as they are persisted and would cause a DDC determinism issue.
233 // (note that cubemap arrays are handled in the bTextureArray path).
234 int32 GetRHIStyleSizeZ(int32 InMipIndex) const
235 {
236 return (bVolumeTexture || bTextureArray) ? GetNumSlices_WithDepth(InMipIndex) : 1;
237 }

239 // Returns the slice count for usage cases/platform that expect slice count to include
240 // volume texture depth. InMipIndex only affects volume textures.
241 int32 GetNumSlices_WithDepth(int32 InMipIndex) const
242 {
243 if (bVolumeTexture)
244 {
245 check(bTextureArray == false && bCubeMap == false);
246 check(InMipIndex < NumMips);
247 return FMath::Max(TopMipVolumeSizeZ >> InMipIndex, 1);
248 }
249
250 check ((bTextureArray && ArraySlices >= 1) || (!bTextureArray && ArraySlices == 1));
251 int32 Slices = ArraySlices;
252 if (bCubeMap)
253 {
254 Slices *= 6;
255 }
256 return Slices;
257 }
258
259 // Returns the slice count for usage cases/platforms that expect slice count to only include
260 // cubemap/array slices.
261 int32 GetNumSlices_NoDepth() const
262 {
263 if (bVolumeTexture)
264 {
265 check(bTextureArray == false && bCubeMap == false);
266 return 1; // no such thing as a cube volume, or a volume array.
267 }
268
269 check((bTextureArray && ArraySlices >= 1) || (!bTextureArray && ArraySlices == 1));
270 int32 Slices = ArraySlices;
271 if (bCubeMap)
272 {
273 Slices *= 6;
274 }
275 return Slices;
276 }

315 // Returns the byte size of the unpacked/tiled mip. For mip chains that are packed or tiled, use FEncodedTextureExtendedData::MipSizesInBytes.
316 uint64 GetMipSizeInBytes(int32 InMipIndex) const
317 {
318 FIntVector3 MipDims = GetMipDimensions(InMipIndex);
319 uint64 SliceByteCount = GPixelFormats[PixelFormat].Get2DImageSizeInBytes(MipDims.X, MipDims.Y);
320 return SliceByteCount * GetNumSlices_WithDepth(InMipIndex);
321 }

362 // Convenience function for iterating over the encoded mips when you need to know how many mips are represented. Use as:
363 //
364 // for (int32 EncodedMipIndex = 0; EncodedMipIndex < OutMipTailIndex + 1; EncodedMipIndex++)
365 // {
366 // int32 MipsRepresentedThisIndex = EncodedMipIndex == OutMipTailIndex ? OutMipsInTail : 1;
367 // }
368 //
369 // This handles mip chains whether or not they have packed mip tails.
370 // Note GetNumEncodedMips() == OutMipTailIndex + 1
371 //
372 void GetEncodedMipIterators(const FEncodedTextureExtendedData* InExtendedData, int32& OutMipTailIndex, int32& OutMipsInTail) const
373 {
374 OutMipTailIndex = NumMips - 1;
375 OutMipsInTail = 1;
376 if (InExtendedData && InExtendedData->NumMipsInTail > 1)
377 {
378 OutMipsInTail = InExtendedData->NumMipsInTail;
379 OutMipTailIndex = NumMips - OutMipsInTail;
380 }
381 }

383 // Returns the number of mips that actually carry bulk data for this texture. Nominally the number of total mips,
384 // however some platforms have packed mip tails, which means they still have the total number of mips, but the last
385 // several are all bundled together for memory savings.
386 int32 GetNumEncodedMips(const FEncodedTextureExtendedData* InExtendedData) const
387 {
388 if (InExtendedData &&
389 InExtendedData->NumMipsInTail > 1)
390 {
391 return NumMips - InExtendedData->NumMipsInTail + 1;
392 }
393 return NumMips;
394 }

396 // Returns the description _for the single mip level_ (i.e. no further mips)
397 FEncodedTextureDescription GetDescriptionForMipLevel(const FEncodedTextureExtendedData* InExtendedData, int32 InMipIndex) const
398 {
399 check(InMipIndex < NumMips);
400
401 FEncodedTextureDescription MipTextureDescription = *this;
402 FIntVector3 TailFirstMipDims = GetMipDimensions(InMipIndex);
403 MipTextureDescription.TopMipSizeX = TailFirstMipDims.X;
404 MipTextureDescription.TopMipSizeY = TailFirstMipDims.Y;
405 MipTextureDescription.TopMipVolumeSizeZ = TailFirstMipDims.Z;
406 MipTextureDescription.NumMips = 1;
407 if (InExtendedData && InExtendedData->NumMipsInTail && InMipIndex >= NumMips - InExtendedData->NumMipsInTail)
408 {
409 // we must only ever get the first mip tail index!
410 check(InMipIndex == NumMips - InExtendedData->NumMipsInTail);
411
412 // We want the layout for the entire tail.
413 MipTextureDescription.NumMips = IntCastChecked<uint8>(InExtendedData->NumMipsInTail);
414 }
415 return MipTextureDescription;
416 }
417
418 FEncodedTextureDescription RemoveTopMips(const FEncodedTextureExtendedData* InExtendedData, int32 InRemoveCount) const
419 {
420 check(InRemoveCount < NumMips);
421
422 FEncodedTextureDescription MipTextureDescription = *this;
423 FIntVector3 TailFirstMipDims = GetMipDimensions(InRemoveCount);
424 MipTextureDescription.TopMipSizeX = TailFirstMipDims.X;
425 MipTextureDescription.TopMipSizeY = TailFirstMipDims.Y;
426 MipTextureDescription.TopMipVolumeSizeZ = TailFirstMipDims.Z;
427 MipTextureDescription.NumMips = NumMips - static_cast<uint8>(InRemoveCount);
428 if (InExtendedData && InExtendedData->NumMipsInTail && InRemoveCount >= NumMips - InExtendedData->NumMipsInTail)
429 {
430 // we must only ever get the first mip tail index!
431 check(InRemoveCount == NumMips - InExtendedData->NumMipsInTail);
432
433 // We want the layout for the entire tail.
434 MipTextureDescription.NumMips = IntCastChecked<uint8>(InExtendedData->NumMipsInTail);
435 }
436 return MipTextureDescription;
437 }
