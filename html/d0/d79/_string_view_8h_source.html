30 template <UE::CCharType CharType>
31 constexpr int32 StringLength(const CharType* InString)
32 {
33 int32 Count = 0;
34 if (InString)
35 {
36 UE_IF_CONSTEVAL
37 {
38 while (*InString++)
39 {
40 ++Count;
41 }
42 }
43 else
44 {
45 Count = TCString<CharType>::Strlen(InString);
46 }
47 }
48 return Count;
49 }

629#define CHARTEXTVIEW(CharType, x) \
630 ( \
631 []() -> decltype(auto) \
632 { \
633 /* We expect <type_traits> to already have been included any time the CHARTEXTVIEW macro is used */ \
634 using UnqualifiedCharType = std::remove_cv_t<CharType>; \
635 if constexpr (std::is_same_v<UnqualifiedCharType, ANSICHAR>) \
636 { \
637 return ANSITEXTVIEW(x); \
638 } \
639 else if constexpr (std::is_same_v<UnqualifiedCharType, TCHAR>) \

641 return TEXTVIEW(x); \

645 return WIDETEXTVIEW(x); \

647 else if constexpr (std::is_same_v<UnqualifiedCharType, UTF8CHAR>) \

649 return UTF8TEXTVIEW(x); \

653 /* We want a compile error, so forward to a templated function because of the static_assert(false) problem */ \

656 } \

706template <UE::CCharType CharType>
707[[nodiscard]] inline TStringView<CharType> TStringView<CharType>::Mid(int32 Position, int32 CharCount) const
708{
709 const CharType* CurrentStart = GetData();
710 const int32 CurrentLength = Len();
711
712 // Clamp minimum position at the start of the string, adjusting the length down if necessary
713 const int32 NegativePositionOffset = (Position < 0) ? Position : 0;
714 CharCount += NegativePositionOffset;
715 Position -= NegativePositionOffset;
716
717 // Clamp maximum position at the end of the string
718 Position = (Position > CurrentLength) ? CurrentLength : Position;
719
720 // Clamp count between 0 and the distance to the end of the string
721 CharCount = FMath::Clamp(CharCount, 0, (CurrentLength - Position));
722
723 ViewType Result = ViewType(CurrentStart + Position, CharCount);
724 return Result;
725}

735{

759template <UE::CCharType CharType>
760template <UE::CCharType OtherCharType>
761[[nodiscard]] inline bool TStringView<CharType>::PrivateEquals(const OtherCharType* const Other, ESearchCase::Type SearchCase) const
762{
763 if (IsEmpty())
764 {
765 return !*Other;
766 }
767
768 if (SearchCase == ESearchCase::CaseSensitive)
769 {
770 return FPlatformString::Strncmp(GetData(), Other, Len()) == 0 && !Other[Len()];
771 }
772 else
773 {
774 return FPlatformString::Strnicmp(GetData(), Other, Len()) == 0 && !Other[Len()];
775 }
776}

778template <UE::CCharType CharType>
779template <UE::CCharType OtherCharType>
780[[nodiscard]] inline int32 TStringView<CharType>::PrivateCompare(TStringView<OtherCharType> OtherView, ESearchCase::Type SearchCase) const
781{
782 const int32 SelfLen = Len();
783 const int32 OtherLen = OtherView.Len();
784 const int32 MinLen = FMath::Min(SelfLen, OtherLen);
785
786 if (!SelfLen || !OtherLen)
787 {
788 return SelfLen - OtherLen;
789 }
790
791 int Result;
792 if (SearchCase == ESearchCase::CaseSensitive)
793 {
794 Result = FPlatformString::Strncmp(GetData(), OtherView.GetData(), MinLen);
795 }
796 else
797 {
798 Result = FPlatformString::Strnicmp(GetData(), OtherView.GetData(), MinLen);
799 }
800
801 if (Result != 0)
802 {
803 return Result;
804 }
805
806 return SelfLen - OtherLen;
807}

809template <UE::CCharType CharType>
810template <UE::CCharType OtherCharType>
811[[nodiscard]] inline int32 TStringView<CharType>::Compare(const OtherCharType* const Other, ESearchCase::Type SearchCase) const
812{
813 if (IsEmpty())
814 {
815 return -!!*Other;
816 }
817
818 int Result;
819 if (SearchCase == ESearchCase::CaseSensitive)
820 {
821 Result = FPlatformString::Strncmp(GetData(), Other, Len());
822 }
823 else
824 {
825 Result = FPlatformString::Strnicmp(GetData(), Other, Len());
826 }
827
828 if (Result != 0)
829 {
830 return Result;
831 }
832
833 // Equal if Other[Len()] is '\0' and less otherwise. !!Other[Len()] is 0 for '\0' and 1 otherwise.
834 return -!!Other[Len()];
835}
