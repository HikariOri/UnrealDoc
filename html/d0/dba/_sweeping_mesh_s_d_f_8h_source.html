185 bool Validate(const FAxisAlignedBox3d& Bounds)
186 {
187 if (Bounds.IsEmpty() || !FMath::IsFinite(Bounds.MaxDim()))
188 {
189 return false;
190 }
191 if (!IsValid(CellSize))

207 bool Compute(FAxisAlignedBox3d Bounds)
208 {
209 if (!ensure(Validate(Bounds)))
210 {
211 return false;
212 }
213
214 if (!ensureMsgf(ExactBandWidth*2 < ApproxMaxCellsPerDimension, TEXT("Cannot request band wider than half the max cells per dimension")))
215 {
216 ExactBandWidth = FMath::Max(ApproxMaxCellsPerDimension/2-1, 1);
217 }
218
219 FVector3d ExpandedDiagonal = (Bounds.Diagonal() + ExpandBounds * 2.0);
220 FVector3d ApproxVoxelsPerDim = (FVector3d)(ExpandedDiagonal / CellSize);
221 int32 MaxVoxelsDim = VectorUtil::Max3Index(ApproxVoxelsPerDim);
222 if (!ensureMsgf(ApproxVoxelsPerDim[MaxVoxelsDim] <= ApproxMaxCellsPerDimension - 2 * ExactBandWidth, TEXT("SDF resolution clamped to avoid excessive memory use")))
223 {
224 float CellSizeFactor = float(ExpandedDiagonal[MaxVoxelsDim] / (ApproxMaxCellsPerDimension - 2 * ExactBandWidth)) / GetDim(CellSize, MaxVoxelsDim);
225 CellSize *= CellSizeFactor;
226 if (!ensure(IsValid(CellSize)))
227 {
228 return false;
229 }
230 }
231
232 CellSizeType fBufferWidth = float(ExactBandWidth) * CellSize;
233 if (ComputeMode == EComputeModes::NarrowBand_SpatialFloodFill)
234 {
235 if constexpr (bScalarCellSize)
236 {
237 fBufferWidth = (float)FMath::Max(fBufferWidth, float(NarrowBandMaxDistance));
238 }
239 else
240 {
241 for (int32 Idx = 0; Idx < 3; ++Idx)
242 {
243 fBufferWidth = FMath::Max(fBufferWidth, float(NarrowBandMaxDistance));
244 }
245 }
246 }
247 FVector3f UseOrigin = (FVector3f)Bounds.Min - fBufferWidth * FVector3f::One() - (FVector3f)ExpandBounds;
248 FVector3f max = (FVector3f)Bounds.Max + fBufferWidth * FVector3f::One() + (FVector3f)ExpandBounds;
249 FVector3i UseDims(
250 (int)((max.X - UseOrigin.X) / GetDim(CellSize, 0)) + 1,
251 (int)((max.Y - UseOrigin.Y) / GetDim(CellSize, 1)) + 1,
252 (int)((max.Z - UseOrigin.Z) / GetDim(CellSize, 2)) + 1
253 );
254
255 return Compute(UseOrigin, UseDims);
256 }

264 bool Compute(FVector3f InGridOrigin, FVector3i InDimensions)
265 {
266 GridOrigin = InGridOrigin;
267 int NI = InDimensions.X;
268 int NJ = InDimensions.Y;
269 int NK = InDimensions.Z;
270
271 // dimensions must be positive
272 if (NI < 0 || NJ < 0 || NK < 0)
273 {
274 return false;
275 }
276
277 if (ComputeMode == EComputeModes::NarrowBand_SpatialFloodFill)
278 {
279 if (!ensureMsgf(Spatial && NarrowBandMaxDistance != 0, TEXT("SweepingMeshSDF::Compute: must set Spatial data structure and band max distance")))
280 {
281 return false;
282 }
283 make_level_set3_parallel_floodfill(GridOrigin, CellSize, NI, NJ, NK, Grid);
284 }
285 else
286 {
287 if (bUseParallel)
288 {
289 if (Spatial != nullptr) // TODO: is this path better than NarrowBand_SpatialFloodFill path sometimes? when?
290 {
291 make_level_set3_parallel_spatial(GridOrigin, CellSize, NI, NJ, NK, Grid, ExactBandWidth);
292 }
293 else
294 {
295 make_level_set3_parallel(GridOrigin, CellSize, NI, NJ, NK, Grid, ExactBandWidth);
296 }
297 }
298 else
299 {
300 make_level_set3(GridOrigin, CellSize, NI, NJ, NK, Grid, ExactBandWidth);
301 }
302 }
303
304 CleanupUnwanted();
305
306 return !CancelF();
307 }

370 void make_level_set3(FVector3f Origin, CellSizeType DX, int NI, int NJ, int NK, FDenseGrid3f& Distances, int ExactBand)
371 {
372 Distances.Resize(NI, NJ, NK);
373 Distances.Assign(upper_bound(Distances)); // upper bound on distance
374
375 // closest triangle id for each Grid cell
376 FDenseGrid3i& closest_tri = ClosestTriGrid;
377 ClosestTriGrid.Resize(NI, NJ, NK);
378 ClosestTriGrid.Assign(-1);
379
380 // intersection_count(I,J,K) is # of tri intersections in (I-1,I]x{J}x{K}
381 FDenseGrid3i& intersection_count = IntersectionsGrid;
382 IntersectionsGrid.Resize(NI, NJ, NK);
383 IntersectionsGrid.Assign(0);
384
385 // Compute narrow-band Distances. For each triangle, we find its Grid-coord-bbox,
386 // and compute exact Distances within that box. The intersection_count Grid
387 // is also filled in this computation
388 CellSizeTyped ddx = (CellSizeTyped)DX;
389 double ox = (double)Origin[0], oy = (double)Origin[1], oz = (double)Origin[2];
390 FVector3d xp, xq, xr;
391 for (int TID = 0; TID < Mesh->MaxTriangleID(); TID++)
392 {
393 if (!Mesh->IsTriangle(TID))
394 {
395 continue;
396 }
397 if (TID % 100 == 0 && CancelF())
398 {
399 break;
400 }
401 Mesh->GetTriVertices(TID, xp, xq, xr);
402
403 // real IJK coordinates of xp/xq/xr
404 double fip = (xp[0] - ox) / GetDim(ddx, 0), fjp = (xp[1] - oy) / GetDim(ddx, 1), fkp = (xp[2] - oz) / GetDim(ddx, 2);
405 double fiq = (xq[0] - ox) / GetDim(ddx, 0), fjq = (xq[1] - oy) / GetDim(ddx, 1), fkq = (xq[2] - oz) / GetDim(ddx, 2);
406 double fir = (xr[0] - ox) / GetDim(ddx, 0), fjr = (xr[1] - oy) / GetDim(ddx, 1), fkr = (xr[2] - oz) / GetDim(ddx, 2);
407
408 // clamped integer bounding box of triangle plus exact-band
409 int i0 = FMath::Clamp(((int)FMath::Min3(fip, fiq, fir)) - ExactBand, 0, NI - 1);
410 int i1 = FMath::Clamp(((int)FMath::Max3(fip, fiq, fir)) + ExactBand + 1, 0, NI - 1);
411 int j0 = FMath::Clamp(((int)FMath::Min3(fjp, fjq, fjr)) - ExactBand, 0, NJ - 1);
412 int j1 = FMath::Clamp(((int)FMath::Max3(fjp, fjq, fjr)) + ExactBand + 1, 0, NJ - 1);
413 int k0 = FMath::Clamp(((int)FMath::Min3(fkp, fkq, fkr)) - ExactBand, 0, NK - 1);
414 int k1 = FMath::Clamp(((int)FMath::Max3(fkp, fkq, fkr)) + ExactBand + 1, 0, NK - 1);
415
416 // compute distance for each tri inside this bounding box
417 // note: this can be very conservative if the triangle is large and on diagonal to Grid axes
418 for (int K = k0; K <= k1; ++K) {
419 for (int J = j0; J <= j1; ++J) {
420 for (int I = i0; I <= i1; ++I) {
421 FVector3d gx((float)I * GetDim(DX, 0) + Origin[0], (float)J * GetDim(DX, 1) + Origin[1], (float)K * GetDim(DX, 2) + Origin[2]);
422 float d = (float)PointTriangleDistance(gx, xp, xq, xr);
423 if (d < Distances.At(I, J, K)) {
424 Distances.At(I, J, K) = d;
425 closest_tri.At(I, J, K) = TID;
426 }
427 }
428 }
429 }
430 }
431 if (CancelF())
432 {
433 return;
434 }
435
436 if (bComputeSigns == true)
437 {
438 compute_intersections(Origin, DX, NI, NJ, NK, intersection_count);
439 if (CancelF())
440 {
441 return;
442 }
443
444 if (ComputeMode == EComputeModes::FullGrid)
445 {
446 // and now we fill in the rest of the Distances with fast sweeping
447 for (int pass = 0; pass < 2; ++pass)
448 {
449 sweep_pass(Origin, DX, Distances, closest_tri);
450 if (CancelF())
451 {
452 return;
453 }
454 }
455 }
456 else
457 {
458 // nothing!
459 }
460
461
462 // then figure out signs (inside/outside) from intersection counts
463 compute_signs(NI, NJ, NK, Distances, intersection_count);
464 if (CancelF())
465 {
466 return;
467 }
468
469 if (!bWantIntersectionsGrid)
470 {
471 // empty grid
472 intersection_count.Resize(0,0,0,EAllowShrinking::Yes);
473 }
474 }
475
476 if (!bWantClosestTriGrid)
477 {
478 // empty grid
479 closest_tri.Resize(0,0,0,EAllowShrinking::Yes);
480 }
481
482 } // end make_level_set_3

489 void make_level_set3_parallel(FVector3f Origin, CellSizeType DX, int NI, int NJ, int NK, FDenseGrid3f& Distances, int ExactBand)
490 {
491 Distances.Resize(NI, NJ, NK);
492 Distances.Assign(upper_bound(Grid)); // upper bound on distance
493
494 // closest triangle id for each Grid cell
495 FDenseGrid3i& closest_tri = ClosestTriGrid;
496 ClosestTriGrid.Resize(NI, NJ, NK);
497 ClosestTriGrid.Assign(-1);
498
499 // intersection_count(I,J,K) is # of tri intersections in (I-1,I]x{J}x{K}
500 FDenseGrid3i& intersection_count = IntersectionsGrid;
501 IntersectionsGrid.Resize(NI, NJ, NK);
502 IntersectionsGrid.Assign(0);
503
504 double ox = (double)Origin[0], oy = (double)Origin[1], oz = (double)Origin[2];
505 CellSizeTyped invdx = CellSizeTyped(UnitCellSize()) / (CellSizeTyped)DX;
506
507 // Compute narrow-band Distances. For each triangle, we find its Grid-coord-bbox,
508 // and compute exact Distances within that box.
509
510 // To compute in parallel, we need to safely update Grid cells. Current strategy is
511 // to use a critical section to control access to Grid. Partitioning the Grid into a few regions,
512 // each w/ a separate lock, improves performance somewhat.
513 int wi = NI / 2, wj = NJ / 2, wk = NK / 2;
514 FCriticalSection GridSections[8];
515
516 bool bAbort = false;
517 ParallelFor(Mesh->MaxTriangleID(), [this, &Origin, DX, NI, NJ, NK, &Distances, ExactBand, &closest_tri, &intersection_count, ox, oy, oz, invdx, wi, wj, wk, &GridSections, &bAbort](int TID)
518 {
519 if (!Mesh->IsTriangle(TID))
520 {
521 return;
522 }
523 if (TID % 100 == 0)
524 {
525 bAbort = CancelF();
526 }
527 if (bAbort)
528 {
529 return;
530 }
531
532 FVector3d xp = FVector3d::Zero(), xq = FVector3d::Zero(), xr = FVector3d::Zero();
533 Mesh->GetTriVertices(TID, xp, xq, xr);
534
535 // real IJK coordinates of xp/xq/xr
536 double fip = (xp[0] - ox) * GetDim(invdx, 0), fjp = (xp[1] - oy) * GetDim(invdx, 1), fkp = (xp[2] - oz) * GetDim(invdx, 2);
537 double fiq = (xq[0] - ox) * GetDim(invdx, 0), fjq = (xq[1] - oy) * GetDim(invdx, 1), fkq = (xq[2] - oz) * GetDim(invdx, 2);
538 double fir = (xr[0] - ox) * GetDim(invdx, 0), fjr = (xr[1] - oy) * GetDim(invdx, 1), fkr = (xr[2] - oz) * GetDim(invdx, 2);
539
540 // clamped integer bounding box of triangle plus exact-band
541 int i0 = FMath::Clamp(((int)FMath::Min3(fip, fiq, fir)) - ExactBand, 0, NI - 1);
542 int i1 = FMath::Clamp(((int)FMath::Max3(fip, fiq, fir)) + ExactBand + 1, 0, NI - 1);
543 int j0 = FMath::Clamp(((int)FMath::Min3(fjp, fjq, fjr)) - ExactBand, 0, NJ - 1);
544 int j1 = FMath::Clamp(((int)FMath::Max3(fjp, fjq, fjr)) + ExactBand + 1, 0, NJ - 1);
545 int k0 = FMath::Clamp(((int)FMath::Min3(fkp, fkq, fkr)) - ExactBand, 0, NK - 1);
546 int k1 = FMath::Clamp(((int)FMath::Max3(fkp, fkq, fkr)) + ExactBand + 1, 0, NK - 1);
547
548 // compute distance for each tri inside this bounding box
549 // note: this can be very conservative if the triangle is large and on diagonal to Grid axes
550 for (int K = k0; K <= k1; ++K) {
551 for (int J = j0; J <= j1; ++J) {
552 int base_idx = ((J < wj) ? 0 : 1) | ((K < wk) ? 0 : 2); // construct index into spinlocks array
553
554 for (int I = i0; I <= i1; ++I) {
555 FVector3d gx((float)I * GetDim(DX, 0) + Origin[0], (float)J * GetDim(DX, 1) + Origin[1], (float)K * GetDim(DX, 2) + Origin[2]);
556 float d = (float)PointTriangleDistance(gx, xp, xq, xr);
557 if (d < Distances.At(I, J, K)) {
558 int lock_idx = base_idx | ((I < wi) ? 0 : 4);
559
560 FScopeLock GridLock(&GridSections[lock_idx]);
561 if (d < Distances.At(I, J, K)) { // have to check again in case Grid changed in another thread...
562 Distances.At(I, J, K) = d;
563 closest_tri.At(I, J, K) = TID;
564 }
565 }
566 }
567
568 }
569 }
570 });
571 if (CancelF())
572 {
573 return;
574 }
575
576
577 if (bComputeSigns == true)
578 {
579 compute_intersections(Origin, DX, NI, NJ, NK, intersection_count);
580 if (CancelF())
581 {
582 return;
583 }
584
585 if (ComputeMode == EComputeModes::FullGrid) {
586 // and now we fill in the rest of the Distances with fast sweeping
587 for (int pass = 0; pass < 2; ++pass) {
588 sweep_pass(Origin, DX, Distances, closest_tri);
589 if (CancelF())
590 return;
591 }
592 }
593 else
594 {
595 // nothing!
596 }
597
598 // then figure out signs (inside/outside) from intersection counts
599 compute_signs(NI, NJ, NK, Distances, intersection_count);
600 if (CancelF())
601 {
602 return;
603 }
604
605 if (!bWantIntersectionsGrid)
606 {
607 // empty grid
608 intersection_count.Resize(0, 0, 0, EAllowShrinking::Yes);
609 }
610 }
611
612 if (!bWantClosestTriGrid)
613 {
614 // empty grid
615 closest_tri.Resize(0, 0, 0, EAllowShrinking::Yes);
616 }

622 {

624 float upper_bound = this->upper_bound(Distances);
625 Distances.Assign(upper_bound); // upper bound on distance
626
627 // closest triangle id for each Grid cell
628 FDenseGrid3i& closest_tri = ClosestTriGrid;

645 // each w/ a separate spinlock, improves performance somewhat. Have also tried having a
646 // separate spinlock per-row, this resulted in a few-percent performance improvement.
647 // Also tried pre-sorting triangles into disjoint regions, this did not help much except
648 // on "perfect" cases like a sphere.
649 bool bAbort = false;
650 ParallelFor(Mesh->MaxTriangleID(), [this, &Origin, DX, NI, NJ, NK, &Distances, ExactBand, upper_bound, &closest_tri, &intersection_count, ox, oy, oz, invdx, &bAbort](int TID)

654 return;
655 }

657 {
658 bAbort = CancelF();
659 }
660 if (bAbort)
661 {
662 return;
663 }
664
665 FVector3d xp = FVector3d::Zero(), xq = FVector3d::Zero(), xr = FVector3d::Zero();
666 Mesh->GetTriVertices(TID, xp, xq, xr);
667
668 // real IJK coordinates of xp/xq/xr
669 double fip = (xp[0] - ox) * GetDim(invdx, 0), fjp = (xp[1] - oy) * GetDim(invdx, 1), fkp = (xp[2] - oz) * GetDim(invdx, 2);

674 int i0 = FMath::Clamp(((int)FMath::Min3(fip, fiq, fir)) - ExactBand, 0, NI - 1);
675 int i1 = FMath::Clamp(((int)FMath::Max3(fip, fiq, fir)) + ExactBand + 1, 0, NI - 1);

681 // compute distance for each tri inside this bounding box
682 // note: this can be very conservative if the triangle is large and on diagonal to Grid axes
683 for (int K = k0; K <= k1; ++K) {
684 for (int J = j0; J <= j1; ++J) {
685 for (int I = i0; I <= i1; ++I) {
686 Distances.At(I, J, K) = 1;
687 }
688 }
689 }
690 });

692 double max_dist = ExactBand * (double)GetDiagLength(DX);
693 check(Grid.Num() <= MAX_int32);
694 ParallelFor(static_cast<int32>(Grid.Num()), [this, &Origin, DX, NI, NJ, NK, &Distances, max_dist, upper_bound, &closest_tri](int LinearIdx)

696 FVector3i Idx = Grid.GetCoords(LinearIdx);

698 int I = Idx.X, J = Idx.Y, K = Idx.Z;
699 FVector3d p((float)I * GetDim(DX, 0) + Origin[0], (float)J * GetDim(DX, 1) + Origin[1], (float)K * GetDim(DX, 2) + Origin[2]);
700 double dsqr;
701 int near_tid = Spatial->FindNearestTriangle(p, dsqr, max_dist);
702 if (near_tid == IndexConstants::InvalidID) {
703 Distances[Idx] = upper_bound;
704 return;
705 }
706 Distances[Idx] = (float)FMath::Sqrt(dsqr);
707 closest_tri[Idx] = near_tid;
708 }
709 });
710
711
712
713 if (CancelF())
714 {
715 return;
716 }
717
718 if (bComputeSigns == true)
719 {
720 compute_intersections(Origin, DX, NI, NJ, NK, intersection_count);
721 if (CancelF())
722 {
723 return;
724 }
725
726 if (ComputeMode == EComputeModes::FullGrid) {
727 // and now we fill in the rest of the Distances with fast sweeping
728 for (int pass = 0; pass < 2; ++pass) {
729 sweep_pass(Origin, DX, Distances, closest_tri);

731 return;

773 void make_level_set3_parallel_floodfill(FVector3f Origin, CellSizeType DX, int NI, int NJ, int NK, FDenseGrid3f& Distances)
774 {
775 Distances.Resize(NI, NJ, NK);
776 float upper_bound = this->upper_bound(Distances);
777 Distances.Assign(upper_bound); // upper bound on distance
778
779 // closest triangle id for each Grid cell
780 FDenseGrid3i& closest_tri = ClosestTriGrid;
781 ClosestTriGrid.Resize(NI, NJ, NK);
782 ClosestTriGrid.Assign(-1);
783
784 // intersection_count(I,J,K) is # of tri intersections in (I-1,I]x{J}x{K}
785 FDenseGrid3i& intersection_count = IntersectionsGrid;
786 IntersectionsGrid.Resize(NI, NJ, NK);
787 IntersectionsGrid.Assign(0);
788
789 double ox = (double)Origin[0], oy = (double)Origin[1], oz = (double)Origin[2];
790 CellSizeTyped invdx = CellSizeTyped(UnitCellSize()) / (CellSizeTyped)DX;
791
792 // the steps below that populate the grid will potentially touch the same cells at the
793 // same time, so we need to lock them. However locking the entire grid for each cell
794 // basically means the threads are constantly fighting. So we split the grid up into
795 // chunks for locking
796 int32 NumSections = 256; // a bit arbitrary, stopped seeing improvement on a 64-core machine at this point
797 TArray<FCriticalSection> GridSections;
798 GridSections.SetNum(NumSections);
799 int64 TotalGridCellCount = NI * NJ * NK;
800 int64 SectionSize = FMath::CeilToInt(float(TotalGridCellCount) / (float)NumSections);
801
802 // this returns the FCriticalSection to use for the given span of values
803 auto GetGridSectionLock = [this, SectionSize, &Distances, &GridSections](FVector3i CellGridIndex) -> FCriticalSection*
804 {
805 int64 CellLinearIndex = Distances.GetIndex(CellGridIndex);
806 int64 SectionIndex = CellLinearIndex / SectionSize;
807 checkSlow(SectionIndex <= MAX_int32);
808 return &GridSections[(int32)SectionIndex];
809 };
810
811 // per-grid-chunk queue, each one of these cooresponds to a GridSections lock
812 TArray<TArray<int>> SectionQueues;
813 SectionQueues.SetNum(NumSections);
814 auto AddToSectionQueue = [SectionSize, &SectionQueues](int64 CellLinearIndex)
815 {
816 int64 SectionIndex = CellLinearIndex / SectionSize;
817 checkSlow(SectionIndex <= MAX_int32);
818 checkSlow(CellLinearIndex <= MAX_int32);
819 SectionQueues[(int32)SectionIndex].Add((int32)CellLinearIndex);
820 };
821
822
823 TArray<int> PendingCellQueue;
824 TArray<bool> done;
825 check(Distances.Num() <= MAX_int32);
826 done.SetNumZeroed(static_cast<int32>(Distances.Num()));
827
828 // this lambda combines the per-chunk queues into a single queue which we can pass to a ParallelFor
829 auto FlattenSectionQueues = [&PendingCellQueue, &SectionQueues]()
830 {
831 PendingCellQueue.Reset();
832 for (TArray<int>& SectionQueue : SectionQueues)
833 {
834 for (int Value : SectionQueue)
835 {
836 PendingCellQueue.Add(Value);
837 }
838 SectionQueue.Reset();
839 }
840 };
841
842 // compute values at vertices
843 bool bAbort = false;
844 {
845 TRACE_CPUPROFILER_EVENT_SCOPE(Geometry_SweepingMeshSDF_Distances);
846 ParallelFor(Mesh->MaxVertexID(), [&](int vid)
847 {
848 if (!Mesh->IsVertex(vid))
849 {
850 return;
851 }
852 if (vid % 100 == 0)
853 {
854 bAbort = CancelF();
855 }
856 if (bAbort)
857 {
858 return;
859 }
860
861 FVector3d v = Mesh->GetVertex(vid);
862 // real IJK coordinates of v
863 double fi = (v.X - ox) * GetDim(invdx, 0), fj = (v.Y - oy) * GetDim(invdx, 1), fk = (v.Z - oz) * GetDim(invdx, 2);
864 FVector3i Idx(
865 FMath::Clamp((int)fi, 0, NI - 1),
866 FMath::Clamp((int)fj, 0, NJ - 1),
867 FMath::Clamp((int)fk, 0, NK - 1));
868
869 {
870 FScopeLock GridLock(GetGridSectionLock(Idx));
871 if (Distances[Idx] < upper_bound)
872 {
873 return;
874 }
875 FVector3d p = (FVector3d)cell_center(Idx);
876 double dsqr;
877 int near_tid = Spatial->FindNearestTriangle(p, dsqr);
878 Distances[Idx] = (float)FMathd::Sqrt(dsqr);
879 closest_tri[Idx] = near_tid;
880 int idx_linear = static_cast<int32>(Distances.GetIndex(Idx));
881 if (done[idx_linear] == false)
882 {
883 done[idx_linear] = true;
884 AddToSectionQueue(idx_linear);
885 }
886 }
887 }, !bUseParallel);
888 }
889 if (CancelF())
890 {
891 return;
892 }
893
894 FlattenSectionQueues();
895
896 // we could do this parallel w/ some kind of producer-consumer...
897 FAxisAlignedBox3i Bounds = Distances.BoundsInclusive();
898 double max_dist = NarrowBandMaxDistance;
899 double max_query_dist = max_dist + GetDiagLength(DX);
900 {
901 TRACE_CPUPROFILER_EVENT_SCOPE(Geometry_SweepingMeshSDF_FloodFill);
902 while (PendingCellQueue.Num() > 0)
903 {
904 ParallelFor(PendingCellQueue.Num(), [&](int QIdx)
905 {
906 int cur_linear_index = PendingCellQueue[QIdx];
907 FVector3i cur_idx = Distances.GetCoords(cur_linear_index);
908 for (FVector3i idx_offset : IndexUtil::GridOffsets26)
909 {
910 FVector3i nbr_idx = cur_idx + idx_offset;
911 if (Bounds.Contains(nbr_idx) == false)
912 {
913 continue;
914 }
915 int nbr_linear_idx = static_cast<int32>(Distances.GetIndex(nbr_idx));
916
917 // Note: this is technically unsafe to do because other threads might be writing to this memory location
918 // which could in theory mean that the value being read is garbage or stale. However the only possible values are true and false.
919 // If we get an incorrect false, we will just do an extra unnecessary spatial query, but because we lock before
920 // any writes, that value will just be discarded. If we get an incorrect true, we potentially skip a grid cell
921 // we need. However in this context we are doing a flood-fill and so except at the very edges of the populated cells,
922 // each cell will be considered multiple times as we touch it's neighbours. In practice we have not observed
923 // this to be a problem, however locking here dramatically slows the algorithm down (eg by 3-5x on a 64-core machine)
924 if (done[nbr_linear_idx])
925 {
926 continue;
927 }
928
929 FVector3d p = (FVector3d)cell_center(nbr_idx);
930 double dsqr;
931 int near_tid = Spatial->FindNearestTriangle(p, dsqr, max_query_dist);
932 if (near_tid == -1)
933 {
934 FScopeLock GridLock(GetGridSectionLock(nbr_idx));
935 done[nbr_linear_idx] = true;
936 continue;
937 }
938 double dist = FMathd::Sqrt(dsqr);
939
940 {
941 // locked section -- if index not already done, update the grid
942 FScopeLock GridLock(GetGridSectionLock(nbr_idx));
943 if (done[nbr_linear_idx] == false)
944 {
945 Distances[nbr_linear_idx] = (float)dist;
946 closest_tri[nbr_linear_idx] = near_tid;
947 done[nbr_linear_idx] = true;
948 if (dist < max_dist)
949 {
950 AddToSectionQueue(nbr_linear_idx);
951 }
952 }
953 }
954 }
955 }, !bUseParallel);
956
957 FlattenSectionQueues();
958 }
959 }
960 if (CancelF())
961 {
962 return;
963 }
964
965
966 if (bComputeSigns == true)
967 {
968 TRACE_CPUPROFILER_EVENT_SCOPE(Geometry_SweepingMeshSDF_Signs);
969 compute_intersections(Origin, DX, NI, NJ, NK, intersection_count);
970 if (CancelF())
971 {
972 return;
973 }
974
975 if (ComputeMode == EComputeModes::FullGrid)
976 {
977 // and now we fill in the rest of the Distances with fast sweeping
978 for (int pass = 0; pass < 2; ++pass)
979 {
980 sweep_pass(Origin, DX, Distances, closest_tri);
981 if (CancelF())
982 {
983 return;
984 }
985 }
986 }
987 else
988 {
989 // nothing!
990 }
991
992 // then figure out signs (inside/outside) from intersection counts
993 compute_signs(NI, NJ, NK, Distances, intersection_count);
994 if (CancelF())
995 {
996 return;
997 }
998
999 }

1042 {

1044 if (di > 0) { i0 = 1; i1 = phi.GetDimensions().X; }
1045 else { i0 = phi.GetDimensions().X - 2; i1 = -1; }
1046 int j0, j1;
1047 if (dj > 0) { j0 = 1; j1 = phi.GetDimensions().Y; }
1048 else { j0 = phi.GetDimensions().Y - 2; j1 = -1; }

1051 else { k0 = phi.GetDimensions().Z - 2; k1 = -1; }
1052 for (int K = k0; K != k1; K += dk)
1053 {
1054 if (CancelF())
1055 {
1056 return;
1057 }
1058 for (int J = j0; J != j1; J += dj)
1059 {
1060 for (int I = i0; I != i1; I += di)
1061 {

1063 check_neighbour(phi, closest_tri, gx, I, J, K, I - di, J, K);

1067 check_neighbour(phi, closest_tri, gx, I, J, K, I - di, J, K - dk);
1068 check_neighbour(phi, closest_tri, gx, I, J, K, I, J - dj, K - dk);

1095 void compute_intersections(FVector3f Origin, CellSizeType DX, int32 NI, int32 NJ, int32 NK, FDenseGrid3i& IntersectionCount)
1096 {

1099
1100 bool cancelled = false;
1101
1102 // this is what we will do for each triangle. There are no Grid-reads, only Grid-writes,
1103 // since we use atomic_increment, it is always thread-safe
1104 ParallelFor(Mesh->MaxTriangleID(),
1105 [this, &Origin, &NI, &NJ, &NK,
1106 &IntersectionCount, &ox, &oy, &oz, &invdx, &cancelled](int32 TID)
1107 {

1109 {
1110 return;
1111 }

1113 {
1114 cancelled = true;

1122 Mesh->GetTriVertices(TID, xp, xq, xr);
1123
1124
1125 bool neg_x = false;
1126 if (InsideMode == EInsideModes::WindingCount)
1127 {
1128 FVector3d n = VectorUtil::NormalDirection(xp, xq, xr);
1129 neg_x = n.X > 0;
1130 }

1133 double fip = (xp[0] - ox) * GetDim(invdx, 0), fjp = (xp[1] - oy) * GetDim(invdx, 1), fkp = (xp[2] - oz) * GetDim(invdx, 2);

1135 double fir = (xr[0] - ox) * GetDim(invdx, 0), fjr = (xr[1] - oy) * GetDim(invdx, 1), fkr = (xr[2] - oz) * GetDim(invdx, 2);
1136
1137 // recompute J/K integer bounds of triangle w/o exact band
1138 int32 j0 = FMath::Clamp(FMath::CeilToInt32(FMath::Min3(fjp, fjq, fjr)), 0, NJ - 1);
1139 int32 j1 = FMath::Clamp(FMath::FloorToInt32(FMath::Max3(fjp, fjq, fjr)), 0, NJ - 1);
1140 int32 k0 = FMath::Clamp(FMath::CeilToInt32(FMath::Min3(fkp, fkq, fkr)), 0, NK - 1);
1141 int32 k1 = FMath::Clamp(FMath::FloorToInt32(FMath::Max3(fkp, fkq, fkr)), 0, NK - 1);
1142

1144 for (int K = k0; K <= k1; ++K)
1145 {

1151 double fi = A * fip + B * fiq + C * fir; // intersection I coordinate
1152 int i_interval = FMath::CeilToInt32(fi); // intersection is in (i_interval-1,i_interval]

1154 {
1155 DenseGrid::AtomicIncDec(IntersectionCount, 0, J, K, neg_x);
1156 }
1157 else if (i_interval < NI)
1158 {
1159 DenseGrid::AtomicIncDec(IntersectionCount, i_interval, J, K, neg_x);
1160 }
1161 else
1162 {
1163 // we ignore intersections that are beyond the +x side of the Grid
1164 }
1165 }
1166 }
1167 }
1168 }, !bUseParallel);

1245 // calculate twice signed area of triangle (0,0)-(X1,Y1)-(X2,Y2)
1246 // return an SOS-determined sign (-1, +1, or 0 only if it's a truly degenerate triangle)
1247 static int Orientation(double X1, double Y1, double X2, double Y2, double& TwiceSignedArea)
1248 {
1249 TwiceSignedArea = Y1 * X2 - X1 * Y2;
1250 if (TwiceSignedArea > 0) return 1;
1251 else if (TwiceSignedArea < 0) return -1;
1252 else if (Y2 > Y1) return 1;
1253 else if (Y2 < Y1) return -1;
1254 else if (X1 > X2) return 1;
1255 else if (X1 < X2) return -1;
1256 else return 0; // only true when X1==X2 and Y1==Y2
1257 }
1258
1259
1260 // robust test of (X0,Y0) in the triangle (X1,Y1)-(X2,Y2)-(X3,Y3)
1261 // if true is returned, the barycentric coordinates are set in A,B,C.
1262 static bool PointInTriangle2d(double X0, double Y0, double X1, double Y1, double X2, double Y2, double X3, double Y3, double& A, double& B, double& C)
1263 {
1264 A = B = C = 0;
1265 X1 -= X0; X2 -= X0; X3 -= X0;
1266 Y1 -= Y0; Y2 -= Y0; Y3 -= Y0;

1326 static float PointTriangleDistance(const FVector3f& x0, const FVector3f& x1, const FVector3f& x2, const FVector3f& x3)
1327 {
1328 // first find barycentric coordinates of closest point on infinite plane
1329 FVector3f x13 = (x1 - x3);
1330 FVector3f x23 = (x2 - x3);
1331 FVector3f x03 = (x0 - x3);
1332 float m13 = x13.SquaredLength(), m23 = x23.SquaredLength(), d = x13.Dot(x23);
1333 float invdet = 1.0f / FMath::Max(m13 * m23 - d * d, 1e-30f);
1334 float a = x13.Dot(x03), b = x23.Dot(x03);
1335 // the barycentric coordinates themselves
1336 float w23 = invdet * (m23 * a - d * b);
1337 float w31 = invdet * (m13 * b - d * a);
