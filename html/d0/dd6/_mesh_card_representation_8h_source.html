123 template<typename TT UE_REQUIRES(sizeof(TT) >= sizeof(T))>
124 TLumenCardOBB<TT> Transform(UE::Math::TMatrix<TT> LocalToWorld, bool* bAxisXFlippedPtr = nullptr) const
125 {
126 TLumenCardOBB<TT> WorldOBB;
127 WorldOBB.Origin = LocalToWorld.TransformPosition(UE::Math::TVector<TT>(Origin));
128
129 const UE::Math::TVector<TT> ScaledXAxis(LocalToWorld.TransformVector(UE::Math::TVector<TT>(AxisX)));
130 const UE::Math::TVector<TT> ScaledYAxis(LocalToWorld.TransformVector(UE::Math::TVector<TT>(AxisY)));
131 const UE::Math::TVector<TT> ScaledZAxis(LocalToWorld.TransformVector(UE::Math::TVector<TT>(AxisZ)));
132 const TT XAxisLength = ScaledXAxis.Size();
133 const TT YAxisLength = ScaledYAxis.Size();
134 const TT ZAxisLength = ScaledZAxis.Size();
135
136 // #lumen_todo: fix axisX flip cascading into entire card code
137 WorldOBB.AxisY = FVector3f(ScaledYAxis / FMath::Max(YAxisLength, TT(UE_DELTA)));
138 WorldOBB.AxisZ = FVector3f(ScaledZAxis / FMath::Max(ZAxisLength, TT(UE_DELTA)));
139 WorldOBB.AxisX = FVector3f::CrossProduct(WorldOBB.AxisZ, WorldOBB.AxisY);
140 FVector3f::CreateOrthonormalBasis(WorldOBB.AxisX, WorldOBB.AxisY, WorldOBB.AxisZ);
141
142 if (bAxisXFlippedPtr)
143 {
144 *bAxisXFlippedPtr = FVector3f::DotProduct(FVector3f(ScaledXAxis), WorldOBB.AxisX) < 0.0f;
145 }
146
147 WorldOBB.Extent = Extent * UE::Math::TVector<float>(XAxisLength, YAxisLength, ZAxisLength);
148 WorldOBB.Extent.Z = FMath::Max(WorldOBB.Extent.Z, 1.0f);
149
150 return WorldOBB;
151 }
152
153 UE::Math::TBox<T> GetBox() const
154 {
155 UE::Math::TVector<T> BoxMin(AxisX.GetAbs() * -Extent.X + AxisY.GetAbs() * -Extent.Y + AxisZ.GetAbs() * -Extent.Z);
156 UE::Math::TVector<T> BoxMax(AxisX.GetAbs() * +Extent.X + AxisY.GetAbs() * +Extent.Y + AxisZ.GetAbs() * +Extent.Z);
157 return UE::Math::TBox<T>(BoxMin + Origin, BoxMax + Origin);
158 }
159
160 bool ContainsNaN() const
161 {
162 return Origin.ContainsNaN() || AxisX.ContainsNaN() || AxisY.ContainsNaN() || AxisZ.ContainsNaN() || Extent.ContainsNaN();
163 }
