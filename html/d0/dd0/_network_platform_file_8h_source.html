321 FNetworkFileHandle(FNetworkPlatformFile& InNetwork, const TCHAR* InFilename, int64 InFilePos, int64 InFileSize, bool bWriting)
322 : Network(InNetwork)
323 , Filename(InFilename)
324 , FilePos(InFilePos)
325 , Size(InFileSize)
326 , bWritable(bWriting)

330
331 virtual int64 Tell() override
332 {
333 return FilePos;
334 }
335 virtual bool Seek(int64 NewPosition) override
336 {
337 if (NewPosition >= 0 && NewPosition <= Size)
338 {
339 FilePos = NewPosition;
340 return true;
341 }
342 return false;
343 }
344 virtual bool SeekFromEnd(int64 NewPositionRelativeToEnd = 0) override
345 {
346 return Seek(Size + NewPositionRelativeToEnd);
347 }
348 virtual bool Read(uint8* Destination, int64 BytesToRead) override
349 {
350 bool Result = false;
351 if (bReadable && BytesToRead >= 0 && BytesToRead + FilePos <= Size)
352 {
353 if (BytesToRead == 0)
354 {
355 Result = true;
356 }
357 else
358 {
359 Result = Network.SendReadMessage(Destination, BytesToRead);
360 if (Result)
361 {
362 FilePos += BytesToRead;
363 }
364 }
365 }
366 return Result;
367 }
368 virtual bool Write(const uint8* Source, int64 BytesToWrite) override
369 {
370 bool Result = false;
371 if (bWritable && BytesToWrite >= 0)
372 {
373 if (BytesToWrite == 0)
374 {
375 Result = true;
376 }
377 else
378 {
379 Result = Network.SendWriteMessage(Source, BytesToWrite);
380 if (Result)
381 {
382 FilePos += BytesToWrite;
383 Size = FMath::Max<int64>(FilePos, Size);
384 }
385 }
386 }
387 return Result;
388 }
389 virtual bool Flush(const bool bFullFlush = false) override
390 {
391 return false;
392 }
393 virtual bool Truncate(int64 NewSize) override
394 {
395 return false;
396 }
