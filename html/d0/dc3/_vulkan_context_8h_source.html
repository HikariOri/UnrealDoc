58class FVulkanContextCommon
59{
60protected:
61 // Phases are used to track where we are at in filling the current payload.
62 // Phases always move forward in the order declared here, never backwards.
63 // Always use GetPayload(EPhase) to fetch the payload for the phase you want,
64 // it will take care of creating a new payload when necessary.
65 enum class EPhase
66 {
67 Wait,
68 UpdateReservedResources,
69 Execute,
70 Signal
71 };
72
73public:
74 FVulkanContextCommon(FVulkanDevice& InDevice, FVulkanQueue& InQueue, EVulkanCommandBufferType InCommandBufferType);
75 virtual ~FVulkanContextCommon();
76
77 FVulkanPayload& GetPayload(EPhase Phase)
78 {
79 if (Payloads.Num() == 0 || Phase < CurrentPhase)
80 {
81 NewPayload();
82 }
83
84 CurrentPhase = Phase;
85 return *Payloads.Last();
86 }
87
88 // NOTE: This call is getting phased out, use GetCommandBuffer()
89 FVulkanCommandBuffer* GetActiveCmdBuffer()
90 {
91 return &GetCommandBuffer();
92 }
93
94 FVulkanCommandBuffer& GetCommandBuffer()
95 {
96 FVulkanPayload& Payload = GetPayload(EPhase::Execute);
97 checkSlow(Payload.SignalSemaphores.Num() == 0);
98
99 if (Payload.CommandBuffers.Num() == 0)
100 {
101 PrepareNewCommandBuffer(Payload);
102 }
103
104 return *Payload.CommandBuffers.Last();
105 }
106
107 void AddWaitSemaphore(VkPipelineStageFlags InWaitFlags, FVulkanSemaphore* InWaitSemaphore)
108 {
109 AddWaitSemaphores(InWaitFlags, MakeArrayView<FVulkanSemaphore*>(&InWaitSemaphore, 1));
110 }
111 void AddWaitSemaphores(VkPipelineStageFlags InWaitFlags, TArrayView<FVulkanSemaphore*> InWaitSemaphores)
112 {
113 if (InWaitSemaphores.Num())
114 {
115 FVulkanPayload& Payload = GetPayload(EPhase::Wait);
116 checkSlow((Payload.CommandBuffers.Num() == 0) && (Payload.SignalSemaphores.Num() == 0));
117
118 Payload.WaitFlags.Reserve(Payload.WaitFlags.Num() + InWaitSemaphores.Num());
119 for (int32 Index = 0; Index < InWaitSemaphores.Num(); ++Index)
120 {
121 Payload.WaitFlags.Add(InWaitFlags);
122 }
123 Payload.WaitSemaphores.Append(InWaitSemaphores);
124 }
125 }
126
127 void AddSignalSemaphore(FVulkanSemaphore* InSignalSemaphore)
128 {
129 AddSignalSemaphores(MakeArrayView<FVulkanSemaphore*>(&InSignalSemaphore, 1));
130 }
131 void AddSignalSemaphores(TArrayView<FVulkanSemaphore*> InSignalSemaphores)
132 {
133 if (InSignalSemaphores.Num())
134 {
135 FVulkanPayload& Payload = GetPayload(EPhase::Signal);
136 Payload.SignalSemaphores.Append(InSignalSemaphores);
137 }
138 }
139
140 // Complete recording of the current command list set, and appends the resulting
141 // payloads to the given array. Resets the context so new commands can be recorded.
142 virtual void Finalize(TArray<FVulkanPayload*>& OutPayloads);
143
144 VkCommandPool GetHandle() const
145 {
146 return Pool.GetHandle();
147 }
148
149 void FreeUnusedCmdBuffers(bool bTrimMemory);
150
151 // Should only ne used when we are certain there are no other pending contexts (like UploadContext)
152 void FlushCommands(EVulkanFlushFlags FlushFlags = EVulkanFlushFlags::None);
153
154 // Public immutable references, access them directly
155 FVulkanDevice& Device;
156 FVulkanQueue& Queue;
157
158 // Add a provided sync point that will be signaled at the end of the current payload's accumulated work
159 void SignalSyncPoint(const FVulkanSyncPointRef& InSync)
160 {
161 FVulkanPayload& Payload = GetPayload(EPhase::Signal);
162 Payload.SyncPoints.Add(InSync);
163 }
164
165 // Add provided sync points that will be appended next time a payload reaches the Signal phase
166 void AddPendingSyncPoint(const FVulkanSyncPointRef& InSyncs)
167 {
168 PendingSyncPoints.Add(InSyncs);
169 }
170
171 // Add an event to be signaled with the current payload is submitted (does not alter phase)
172 void AddSubmissionEvent(const FGraphEventRef& InEvent)
173 {
174 FVulkanPayload& Payload = GetPayload(CurrentPhase);
175 Payload.SubmissionEvents.Add(InEvent);
176 }
177
178 // Force pending syncs to be sent to a payload
179 void FlushPendingSyncPoints()
180 {
181 if (PendingSyncPoints.Num())
182 {
183 FVulkanPayload& Payload = GetPayload(EPhase::Signal);
184 Payload.SyncPoints.Append(PendingSyncPoints);
185 PendingSyncPoints.Empty();
186 }
187 }
188
189 // Returns a single sync point for the context that will be inserted when it is finalized
190 FVulkanSyncPointRef GetContextSyncPoint()
191 {
192 if (!ContextSyncPoint)
193 {
194 ContextSyncPoint = FVulkanSyncPoint::Create(EVulkanSyncPointType::Context, TEXT("VulkanContextSyncPoint"));
195 }
196 return ContextSyncPoint;
197 }
198
199 TArray<FVulkanQueryPool*>& GetQueryPoolArray(EVulkanQueryPoolType Type)
200 {
201 FVulkanPayload& Payload = GetPayload(EPhase::Execute);
202 return Payload.QueryPools[(int32)Type];
203 }
204
205 FVulkanQueryPool* GetCurrentTimestampQueryPool()
206 {
207 return GetCurrentTimestampQueryPool(GetPayload(EPhase::Execute));
208 }
209
210 virtual FVulkanParallelRenderPassInfo* GetParallelRenderPassInfo()
211 {
212 return nullptr;
213 }
214
215 void HandleReservedResourceCommits(TArrayView<const FRHITransition*> Transitions);
216
217protected:
218
219 // End the current payload, insert the parallel rendering work, and leave context with a valid payload
220 void AppendParallelRenderPayload(FVulkanPayload* ParallelRenderingPayload, VkRenderingInfo* NeedsResumeInfo);
221
222private:
223
224 FVulkanQueryPool* GetCurrentTimestampQueryPool(FVulkanPayload& Payload);
225 void PrepareNewCommandBuffer(FVulkanPayload& Payload, bool bAllowTimestamp=true);
226 void NewPayload();
227 void EndPayload();
228
229 FVulkanCommandBufferPool& Pool;
230
231 TArray<FVulkanPayload*> Payloads;
232 EPhase CurrentPhase = EPhase::Wait;
233
234 // Sync points signaled at the next Signal phase (will not force a phase change) or when context is finalized
235 TArray<FVulkanSyncPointRef> PendingSyncPoints;
236
237 // Sync point signaled when the current context is finalized
238 FVulkanSyncPointRef ContextSyncPoint;
239
240 friend class FVulkanDynamicRHI;
241};
