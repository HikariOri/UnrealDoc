354template <typename CharType>
355const CharType* FindMatchingBlock(const CharType* OpeningCharPtr, char OpenChar, char CloseChar)
356{
357 const CharType* SearchPtr = OpeningCharPtr;
358 int32 Depth = 0;
359
360 while (*SearchPtr)
361 {
362 if (*SearchPtr == OpenChar)
363 {
364 Depth++;
365 }
366 else if (*SearchPtr == CloseChar)
367 {
368 if (Depth == 0)
369 {
370 return SearchPtr;
371 }
372
373 Depth--;
374 }
375 SearchPtr++;
376 }
377
378 return nullptr;
379}
