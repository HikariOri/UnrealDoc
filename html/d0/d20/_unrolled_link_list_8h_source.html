100template< typename T, uint32 blockSize >
101FORCEINLINE void TUnrolledLinkList<T, blockSize>::Add( const T& Element )
102{
103 if( num != 0 && ( num & (blockSize - 1) ) == 0 )
104 {
105 if( !freeNode->next )
106 {
107 freeNode->next = new TNode;
108 freeNode->next->prev = freeNode;
109 }
110
111 freeNode = freeNode->next;
112 }
113
114 checkSlow( freeNode );
115
116 freeNode->block[ num & (blockSize - 1) ] = Element;
117 num++;
118}

133template< typename T, uint32 blockSize >
134FORCEINLINE void TUnrolledLinkList<T, blockSize>::Remove( const TIterator& i )
135{
136 checkSlow( num > 0 );
137 checkSlow( i.index < num );
138
139 num--;
140
141 if( i.index != num )
142 {
143 // i isn't last element so copy last element into hole
144 i.node->block[ i.index & (blockSize - 1) ] = freeNode->block[ num & (blockSize - 1) ];
145 }
146
147 if( ( num & (blockSize - 1) ) == 0 )
148 {
149 // more than one fully empty node after freeNode
150 delete freeNode->next;
151 freeNode->next = NULL;
152
153 freeNode = freeNode->prev ? freeNode->prev : freeNode;
154 }
155}
