220 template <typename RealType>
221 inline void MakePerpVectors(const TVector<RealType>& Normal, TVector<RealType>& OutPerp1, TVector<RealType>& OutPerp2)
222 {
223 // Duff et al method, from https://graphics.pixar.com/library/OrthonormalB/paper.pdf
224 if (Normal.Z < (RealType)0)
225 {
226 RealType A = (RealType)1 / ((RealType)1 - Normal.Z);
227 RealType B = Normal.X * Normal.Y * A;
228 OutPerp1.X = (RealType)1 - Normal.X * Normal.X * A;
229 OutPerp1.Y = -B;
230 OutPerp1.Z = Normal.X;
231 OutPerp2.X = B;
232 OutPerp2.Y = Normal.Y * Normal.Y * A - (RealType)1;
233 OutPerp2.Z = -Normal.Y;
234 }
235 else
236 {
237 RealType A = (RealType)1 / ((RealType)1 + Normal.Z);
238 RealType B = -Normal.X * Normal.Y * A;
239 OutPerp1.X = (RealType)1 - Normal.X * Normal.X * A;
240 OutPerp1.Y = B;
241 OutPerp1.Z = -Normal.X;
242 OutPerp2.X = B;
243 OutPerp2.Y = (RealType)1 - Normal.Y * Normal.Y * A;
244 OutPerp2.Z = -Normal.Y;
245 }
246 }

251 template <typename RealType>
252 inline TVector<RealType> MakePerpVector(const TVector<RealType>& Normal)
253 {
254 // Duff et al method, from https://graphics.pixar.com/library/OrthonormalB/paper.pdf
255 TVector<RealType> OutPerp1;
256 if (Normal.Z < (RealType)0)
257 {
258 RealType A = (RealType)1 / ((RealType)1 - Normal.Z);
259 RealType B = Normal.X * Normal.Y * A;
260 OutPerp1.X = (RealType)1 - Normal.X * Normal.X * A;
261 OutPerp1.Y = -B;
262 OutPerp1.Z = Normal.X;
263 }
264 else
265 {
266 RealType A = (RealType)1 / ((RealType)1 + Normal.Z);
267 RealType B = -Normal.X * Normal.Y * A;
268 OutPerp1.X = (RealType)1 - Normal.X * Normal.X * A;
269 OutPerp1.Y = B;
270 OutPerp1.Z = -Normal.X;
271 }
272 return OutPerp1;
273 }

289 template <typename RealType>
290 inline RealType PlaneAngleSignedD(const TVector<RealType>& VFrom, const TVector<RealType>& VTo, const TVector<RealType>& PlaneN)
291 {
292 TVector<RealType> vFrom = VFrom - VFrom.Dot(PlaneN) * PlaneN;
293 TVector<RealType> vTo = VTo - VTo.Dot(PlaneN) * PlaneN;
294 Normalize(vFrom);
295 Normalize(vTo);
296 TVector<RealType> C = vFrom.Cross(vTo);
297 if (C.SquaredLength() < TMathUtil<RealType>::ZeroTolerance)
298 { // vectors are parallel
299 return vFrom.Dot(vTo) < 0 ? (RealType)180 : (RealType)0;
300 }
301 RealType fSign = C.Dot(PlaneN) < 0 ? (RealType)-1 : (RealType)1;
302 return (RealType)(fSign * AngleD(vFrom, vTo));
303 }

309 template <typename RealType>
310 inline RealType PlaneAngleSignedR(const TVector<RealType>& VFrom, const TVector<RealType>& VTo, const TVector<RealType>& PlaneN)
311 {
312 TVector<RealType> vFrom = VFrom - VFrom.Dot(PlaneN) * PlaneN;
313 TVector<RealType> vTo = VTo - VTo.Dot(PlaneN) * PlaneN;
314 Normalize(vFrom);
315 Normalize(vTo);
316 TVector<RealType> C = vFrom.Cross(vTo);
317 if (C.SquaredLength() < TMathUtil<RealType>::ZeroTolerance)
318 { // vectors are parallel
319 return vFrom.Dot(vTo) < 0 ? TMathUtil<RealType>::Pi : (RealType)0;
320 }
321 RealType fSign = C.Dot(PlaneN) < 0 ? (RealType)-1 : (RealType)1;
322 return (RealType)(fSign * AngleR(vFrom, vTo));
323 }

356 template <typename RealType>
357 RealType VectorCot(const TVector<RealType>& V1, const TVector<RealType>& V2)
358 {
359 // formula from http://www.geometry.caltech.edu/pubs/DMSB_III.pdf
360 RealType fDot = V1.Dot(V2);
361 RealType lensqr1 = V1.SquaredLength();
362 RealType lensqr2 = V2.SquaredLength();
363 RealType d = Clamp(lensqr1 * lensqr2 - fDot * fDot, (RealType)0.0, TMathUtil<RealType>::MaxReal);
364 if (d < TMathUtil<RealType>::ZeroTolerance)
365 {
366 return (RealType)0;
367 }
368 else
369 {
370 return fDot / TMathUtil<RealType>::Sqrt(d);
371 }
372 }

380 template <typename RealType>
381 TVector<RealType> BarycentricCoords(const TVector<RealType>& Point, const TVector<RealType>& V0, const TVector<RealType>& V1, const TVector<RealType>& V2)
382 {
383 TVector<RealType> kV02 = V0 - V2;
384 TVector<RealType> kV12 = V1 - V2;
385 TVector<RealType> kPV2 = Point - V2;
386 RealType fM00 = kV02.Dot(kV02);
387 RealType fM01 = kV02.Dot(kV12);
388 RealType fM11 = kV12.Dot(kV12);
389 RealType fR0 = kV02.Dot(kPV2);
390 RealType fR1 = kV12.Dot(kPV2);
391 RealType fDet = fM00 * fM11 - fM01 * fM01;
392 RealType fInvDet = 1.0 / fDet;
393 RealType fBary1 = (fM11 * fR0 - fM01 * fR1) * fInvDet;
394 RealType fBary2 = (fM00 * fR1 - fM01 * fR0) * fInvDet;
395 RealType fBary3 = 1.0 - fBary1 - fBary2;
396 return TVector<RealType>(fBary1, fBary2, fBary3);
397 }

477 template <typename RealType>
478 inline RealType TriSolidAngle(TVector<RealType> A, TVector<RealType> B, TVector<RealType> C, const TVector<RealType>& P)
479 {
480 // Formula from https://igl.ethz.ch/projects/winding-number/
481 A -= P;
482 B -= P;
483 C -= P;
484 RealType la = A.Length(), lb = B.Length(), lc = C.Length();
485 RealType top = (la * lb * lc) + A.Dot(B) * lc + B.Dot(C) * la + C.Dot(A) * lb;
486 RealType bottom = A.X * (B.Y * C.Z - C.Y * B.Z) - A.Y * (B.X * C.Z - C.X * B.Z) + A.Z * (B.X * C.Y - C.X * B.Y);
487 // -2 instead of 2 to account for UE winding
488 return RealType(-2.0) * atan2(bottom, top);
489 }

497 template <typename RealType>
498 inline TVector<RealType> TriGradient(TVector<RealType> Vi, TVector<RealType> Vj, TVector<RealType> Vk, RealType fi, RealType fj, RealType fk)
499 {
500 // recenter (better for precision)
501 TVector<RealType> Centroid = (Vi + Vj + Vk) / (RealType)3;
502 Vi -= Centroid; Vj -= Centroid; Vk -= Centroid;
503 // calculate tangent-normal frame
504 TVector<RealType> Normal = VectorUtil::Normal<RealType>(Vi, Vj, Vk);
505 TVector<RealType> Perp0, Perp1;
506 VectorUtil::MakePerpVectors<RealType>(Normal, Perp0, Perp1);
507 // project points to triangle plane coordinates
508 TVector2<RealType> vi(Vi.Dot(Perp0), Vi.Dot(Perp1));
509 TVector2<RealType> vj(Vj.Dot(Perp0), Vj.Dot(Perp1));
510 TVector2<RealType> vk(Vk.Dot(Perp0), Vk.Dot(Perp1));
511 // calculate gradient
512 TVector2<RealType> GradX = (fj-fi)*PerpCW(vi-vk) + (fk-fi)*PerpCW(vj-vi);
513 // map back to 3D vector in triangle plane
514 RealType AreaScale = (RealType)1 / ((RealType)2 * VectorUtil::Area<RealType>(Vi, Vj, Vk));
515 return AreaScale * (GradX.X * Perp0 + GradX.Y * Perp1);
516 }

569 template<typename RealType>
570 inline TVector2<RealType> Circumcenter(TVector2<RealType> A, TVector2<RealType> B, const TVector2<RealType>& C, RealType Epsilon = TMathUtilConstants<RealType>::Epsilon)
571 {
572 // Compute in offset space w/ A translated to origin
573 TVector2<RealType> AB = B - A;
574 TVector2<RealType> AC = C - A;
575
576 RealType Denom = 2 * DotPerp(AB,AC);
577 if (FMath::Abs(Denom) <= Epsilon)
578 {
579 // degenerate (flat) triangle does not have a (finite) circumcenter ...
580 // we'll just fall back to the centroid in this case
581 return A + (AB + AC) * RealType(1.0/3.0);
582 }
583 RealType ABLenSq = AB.SquaredLength();
584 RealType ACLenSq = AC.SquaredLength();
585 TVector2<RealType> Center(
586 A.X + (AC.Y * ABLenSq - AB.Y * ACLenSq) / Denom,
587 A.Y + (AB.X * ACLenSq - AC.X * ABLenSq) / Denom
588 );
589 return Center;
590 }

593 /*
594 * @return incircle / circumcircle radii ratio of triangle ABC
595 */
596 template <typename RealType>
597 RealType TriangleRegularity(const TVector<RealType>& A, const TVector<RealType>& B, const TVector<RealType>& C)
598 {
599 const RealType EAB = (A - B).Length();
600 const RealType EBC = (B - C).Length();
601 const RealType ECA = (C - A).Length();
602
603 const RealType SP = RealType(0.5) * (EAB + EBC + ECA); // Semiperimeter
604 const RealType Area = FMath::Sqrt(FMath::Max(RealType(0.), SP * (SP - EAB) * (SP - EBC) * (SP - ECA))); // Heron
605 const RealType IncircleRadius = Area / SP;
606 const RealType CircumcircleRadius = EAB * EBC * ECA / (RealType(4.) * Area);
607
608 return IncircleRadius / CircumcircleRadius;
609 }

615 inline RealType BitangentSign(const TVector<RealType>& NormalIn, const TVector<RealType>& TangentIn, const TVector<RealType>& BitangentIn)

688template<typename RealType>
689RealType SnapToIncrement(RealType Value, RealType Increment, RealType Offset = 0)
690{
691 if (!FMath::IsFinite(Value))
692 {
693 return (RealType)0;
694 }
695 Value -= Offset;
