21 // Consolidated filter settings for a single bus/submix send.
22 // Passed between the game thread and render thread as a simple value type.
23 struct FSendFilterSettings
24 {
25 uint8 bEnableLPF : 1 = 0;
26 uint8 bEnableHPF : 1 = 0;
27 float LpfCutoff = 20000.0f;
28 float HpfCutoff = 20.0f;
29
30 FSendFilterSettings() = default;
31 FSendFilterSettings(bool bInEnableLPF, bool bInEnableHPF, float InLpfCutoff, float InHpfCutoff)
32 : LpfCutoff(InLpfCutoff), HpfCutoff(InHpfCutoff)
33 {
34 bEnableLPF = bInEnableLPF;
35 bEnableHPF = bInEnableHPF;
36 }
37
38 bool IsFiltering() const { return bEnableLPF || bEnableHPF; }
39
40 // Returns true if any filter parameter differs significantly from Other.
41 // Use to determine whether a render-thread update is needed.
42 bool HasChanged(const FSendFilterSettings& Other) const
43 {
44 return bEnableLPF != Other.bEnableLPF
45 || bEnableHPF != Other.bEnableHPF
46 || (bEnableLPF && !FMath::IsNearlyEqual(LpfCutoff, Other.LpfCutoff))
47 || (bEnableHPF && !FMath::IsNearlyEqual(HpfCutoff, Other.HpfCutoff));
48 }
49 };
