56 static inline void ValidateIndirectArgsBuffer(FRHIBuffer* ArgumentBuffer, uint32 ArgumentOffset, uint32 ArgumentSize, uint32 ArgumentsBoundarySize)
57 {
58 auto GetBufferDesc = [&]() -> FString { return FString::Printf(TEXT("Buffer: %s, Size: %u, Stride: %u, Offset: %u, ArgSize: %u"), ArgumentBuffer->GetDebugName(), ArgumentBuffer->GetSize(), ArgumentBuffer->GetStride(), ArgumentOffset, ArgumentSize); };
59 RHI_VALIDATION_CHECK(EnumHasAnyFlags(ArgumentBuffer->GetUsage(), EBufferUsageFlags::VertexBuffer | EBufferUsageFlags::ByteAddressBuffer), *FString::Printf(TEXT("Indirect argument buffer must be a vertex or byte address buffer to be used as an indirect dispatch parameter. %s"), *GetBufferDesc()));
60 RHI_VALIDATION_CHECK(EnumHasAnyFlags(ArgumentBuffer->GetUsage(), EBufferUsageFlags::DrawIndirect), *FString::Printf(TEXT("Indirect dispatch parameter buffer was not flagged with BUF_DrawIndirect. %s"), *GetBufferDesc()));
61 RHI_VALIDATION_CHECK((ArgumentOffset % 4) == 0, *FString::Printf(TEXT("Indirect argument offset must be a multiple of 4. %s"), *GetBufferDesc()));
62 RHI_VALIDATION_CHECK((ArgumentOffset + ArgumentSize) <= ArgumentBuffer->GetSize(), *FString::Printf(TEXT("Indirect argument doesn't fit in the buffer. %s"), *GetBufferDesc()));
63 if (ArgumentsBoundarySize > 0)
64 {
65 RHI_VALIDATION_CHECK(ArgumentOffset / ArgumentsBoundarySize == (ArgumentOffset + sizeof(FRHIDispatchIndirectParametersNoPadding) - 1) / ArgumentsBoundarySize,
66 *FString::Printf(TEXT("Indirect arguments cannot cross %d byte boundary. %s"), ArgumentsBoundarySize, *GetBufferDesc()));
67 }
68 }

131 // FlushType: Thread safe
132 virtual FDepthStencilStateRHIRef RHICreateDepthStencilState(const FDepthStencilStateInitializerRHI& Initializer) override final
133 {
134 FDepthStencilStateRHIRef State = RHI->RHICreateDepthStencilState(Initializer);
135
136 // @todo: remove this and use the PSO's dsmode instead?
137 // Determine the actual depth stencil mode that applies for this state
138 FExclusiveDepthStencil::Type DepthStencilMode = FExclusiveDepthStencil::DepthNop_StencilNop;
139 if (Initializer.DepthTest != CF_Always || Initializer.bEnableDepthWrite)
140 {
141 DepthStencilMode = Initializer.bEnableDepthWrite
142 ? FExclusiveDepthStencil::DepthWrite
143 : FExclusiveDepthStencil::DepthRead;
144 }
145
146 // set up stencil testing if it's enabled
147 if (Initializer.bEnableFrontFaceStencil || Initializer.bEnableBackFaceStencil)
148 {
149 bool bBackFaceStencilWriteEnabled = false;
150
151 // bEnableBackFaceStencil means to use separate settings for the Back, not if it's enabled at all
152 if (Initializer.bEnableBackFaceStencil)
153 {
154 bBackFaceStencilWriteEnabled =
155 (Initializer.BackFaceStencilFailStencilOp != SO_Keep ||
156 Initializer.BackFacePassStencilOp != SO_Keep ||
157 Initializer.BackFaceDepthFailStencilOp != SO_Keep);
158 }
159
160 if (Initializer.StencilReadMask != 0)
161 {
162 DepthStencilMode = FExclusiveDepthStencil::Type(DepthStencilMode | FExclusiveDepthStencil::StencilRead);
163 }
164 if (Initializer.StencilWriteMask != 0)
165 {
166 bool bFrontFaceStencilWriteEnabled =
167 Initializer.FrontFaceStencilFailStencilOp != SO_Keep ||
168 Initializer.FrontFacePassStencilOp != SO_Keep ||
169 Initializer.FrontFaceDepthFailStencilOp != SO_Keep;
170
171 if (bFrontFaceStencilWriteEnabled || bBackFaceStencilWriteEnabled)
172 {
173 DepthStencilMode = FExclusiveDepthStencil::Type(DepthStencilMode | FExclusiveDepthStencil::StencilWrite);
174 }
175 }
176 }
177 State->ActualDSMode = DepthStencilMode;
178 // @todo: remove this and use the PSO's dsmode instead?
179
180 DepthStencilStates.FindOrAdd(State.GetReference()) = Initializer;
181 return State;
182 }

377 virtual FUniformBufferRHIRef RHICreateUniformBuffer(const void* Contents, const FRHIUniformBufferLayout* Layout, EUniformBufferUsage Usage, EUniformBufferValidation Validation) override final
378 {
379 check(Layout);
380 check(Layout->Resources.Num() > 0 || Layout->ConstantBufferSize > 0);
381 FUniformBufferRHIRef UniformBuffer = RHI->RHICreateUniformBuffer(Contents, Layout, Usage, Validation);
382
383 // Use the render thread frame ID for any non RHI thread allocations. TODO: This is actually incorrect as command list recording on the render thread timeline
384 // can straddle EndFrame boundaries, causing a uniform buffer allocated in frame N to be recorded as being allocated in frame N+1. The solution here is to introduce
385 // a command list to RHICreateUniformBuffer so that the correct render thread frame can be propagated. Unfortunately, in the meantime, this means that the lifetime
386 // tracker can miss legitimate cases. For example, if a (single frame) uniform buffer is allocated in frame N (but straddles to frame N+1), and then is incorrectly
387 // used in frame N+1, that test will pass because they are equal. However, since the issue is timing dependent, it's still likely to catch legitimate allocation misuses.
388 UniformBuffer->InitLifetimeTracking(IsInRHIThread() ? RHIThreadFrameID : RenderThreadFrameID.load(std::memory_order_relaxed), Contents, Usage);
389 return UniformBuffer;
390 }
