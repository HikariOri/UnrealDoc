55class FVulkanAndroidPlatformWindowContext
56{
57public:
58 FVulkanAndroidPlatformWindowContext(void* InWindowHandle)
59 {
60 AndroidWindow = (FAndroidWindow*)InWindowHandle;
61 // in this case FVulkanAndroidPlatformWindowContext owns the FNativeAccessor.
62 if (InWindowHandle)
63 {
64 if(FAndroidMisc::UseNewWindowBehavior())
65 {
66 // the context should be locked at the beginning of the update process, this overload is used for GT initiated events.
67 check(IsInGameThread());
68 WindowContainer = AndroidWindow->GetANativeAccessor(false);
69 NativeWindow = WindowContainer.IsSet() ? WindowContainer->GetANativeWindow() : nullptr;
70 }
71 }
72 }
73
74 FVulkanAndroidPlatformWindowContext(TOptional<FAndroidWindow::FNativeAccessor> WindowContainerIn)
75 {
76 // in this case the FNativeAccessor has come from the event thread
77 if(WindowContainerIn.IsSet())
78 {
79 check(FAndroidMisc::UseNewWindowBehavior());
80 // the context should be locked at the beginning of the update process, this overload is used for ET initiated events.
81 check(IsInAndroidEventThread());
82 WindowContainer = MoveTemp(WindowContainerIn);
83 NativeWindow = WindowContainer->GetANativeWindow();
84 AndroidWindow = WindowContainer.GetValue().Get();
85 }
86 }
87
88 // we dont have a locked window to create swapchains during present, android will use the
89 // SetRHIOnReleaseWindowCallback / SetRHIOnReInitWindowCallback to create as required.
90 static bool CanCreateSwapchainOnDemand() { return false; }
91 bool IsValid() const
92 {
93 return FAndroidMisc::UseNewWindowBehavior() ? GetANativeWindow() != nullptr : true;
94 }
95
96 ANativeWindow* GetANativeWindow() const
97 {
98 return NativeWindow;
99 }
100
101 void* GetWindowHandle() const
102 {
103 return AndroidWindow;
104 }
105
106 // Android tracks window lifetime via the native accessor, not an SDL-style window id.
107 uint32 GetWindowId() const { return 0; }
108
109private:
110 ANativeWindow* NativeWindow = nullptr;
111 FAndroidWindow* AndroidWindow = nullptr;
112 TOptional<FAndroidWindow::FNativeAccessor> WindowContainer;
113};
