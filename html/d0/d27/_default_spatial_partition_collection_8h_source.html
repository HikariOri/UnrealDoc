81 EVisitResult Visit(const FUserDataType& UserData, QueryRuntimeDataType& QueryRuntimeData)
82 {
83 const FEntry& Entry = *Collection->GetEntryFromHandleValue(UserData);
84 const ESpatialCategoryMask EntryCategoryMask = ToCategoryMask(Entry.Category);
85 // Only visit an object if it passes the mask filter and it's aabb overlaps the query data.
86 if (EnumHasAnyFlags(EntryCategoryMask, QueryCategoryMask) && QueryRuntimeData.Test(Entry.Aabb))
87 {
88 return ExternalVisitor.Visit(Entry.UserData, QueryRuntimeData);
89 }
90 return EVisitResult::Continue;
91 }

120 template <typename QueryRuntimeDataType, typename VisitorType>
121 void QueryInternal(const ESpatialCategoryMask QueryCategoryMask, QueryRuntimeDataType& QueryRuntimeData, VisitorType& Visitor) const
122 {
123 if (EnumHasAnyFlags(QueryCategoryMask, ESpatialCategoryMask::Static))
124 {
125 if (EVisitResult::Stop == Query(StaticTree, QueryRuntimeData, Visitor))
126 {
127 return;
128 }
129 if (EVisitResult::Stop == Query(StaticDirty, QueryRuntimeData, Visitor))
130 {
131 return;
132 }
133 if (EVisitResult::Stop == Query(StaticGlobal, QueryRuntimeData, Visitor))
134 {
135 return;
136 }
137 }
138 if (EnumHasAnyFlags(QueryCategoryMask, ESpatialCategoryMask::Dynamic | ESpatialCategoryMask::Kinematic))
139 {
140 if (EVisitResult::Stop == Query(DynamicTree, QueryRuntimeData, Visitor))
141 {
142 return;
143 }
144 Query(DynamicGlobal, QueryRuntimeData, Visitor);
145 }
