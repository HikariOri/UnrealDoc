419 inline float GetTimerElapsed(FTimerHandle InHandle) const
420 {
421#if UE_AUTORTFM
422 if (AutoRTFM::IsClosed())
423 {
424 if (const FTransactionalTimer* Timer = TransactionalTimers.Find(InHandle))
425 {
426 // A transactionally created timer cannot be executed until the
427 // transaction commits, so will have an elapsed time of 0,
428 // unless it was cancelled, in which case we return -1 (same as
429 // InternalGetTimerElapsed()).
430 return Timer->Status == FTransactionalTimer::Canceled ? -1.0f : 0.0f;
431 }
432 }
433#endif // UE_AUTORTFM

444 inline float GetTimerRemaining(FTimerHandle InHandle) const
445 {
446#if UE_AUTORTFM

448 {
449 if (const FTransactionalTimer* Timer = TransactionalTimers.Find(InHandle))
450 {
451 if (Timer->NewData.IsSet())
452 {
453 return static_cast<float>(Timer->NewData->ExpireTime);
454 }

457 return -1.0f;

461#endif // UE_AUTORTFM
