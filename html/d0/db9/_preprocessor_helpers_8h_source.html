82// Expand the given macro with each following parenthesized argument or set of arguments, with an optional separator.
83// Supports up to 10 arguments at time of writing.
84// The primary use case is to write macros that can expand their variadic arguments to produce a variable number of
85// expressions or statements.
86//
87// e.g.
88// #define DECLARE_INT_VARIABLE(Name) int Name;
89// #define DECLARE_TYPE(TypeName, ...) struct TypeName { UE_FOR_EACH(DECLARE_INT_VARIABLE, ## __VA_ARGS__) };
90// DECLARE_TYPE(Foo, (One), (Two), (Three))
91// expands to
92// struct Foo { int One; int Two; int Three; };
93//
94// To expand macro arguments as a comma-separated list, use UE_COMMA as the separator
95// e.g.
96// SomeFunction(UE_FOR_EACH_WITH_SEPARATOR(PASSTHROUGH, UE_COMMA, ## __VA_ARGS__))
97// may expand to to
98// SomeFunction(One, Two, Three)
99#define UE_FOR_EACH(Macro, ...) UE_FOR_EACH_WITH_SEPARATOR(Macro, UE_EMPTY, ## __VA_ARGS__)
100#define UE_FOR_EACH_WITH_SEPARATOR(Macro, Separator, ...) UE_APPEND_VA_ARG_COUNT(UE_PRIVATE_FOR_EACH_, ## __VA_ARGS__)(Macro, UE_ADD_PARENS(Separator), ## __VA_ARGS__)

114// Performs macro expansion on the given arguments which will allow the preprocessor to then apply further macro
115// expansion to the result.
116// Used in cases where expansion needs to be performed for the preprocessor to recognize the first parameter as a
117// function macro rather than an object macro, or cases where expansion must be performed to produce the correct
118// number of arguments for a macro
119//
120// e.g.
121// #define ADD(a, b) a + b
122//
123// #define ARGS (1, 2)
124//
125// const TCHAR Str1[] = UE_STRINGIZE(ADD ARGS); // "ADD (1, 2)"
126// const TCHAR Str2[] = UE_STRINGIZE(UE_EXPAND(ADD ARGS)); // "1 + 2"
127//
128// UE_ADD_PARENS can also be used
129// UE_EXPAND(MYMACRO UE_ADD_PARENS(UE_REMOVE_OPTIONAL_PARENS(FIRST), UE_REMOVE_OPTIONAL_PARENS(SECOND)))
130#define UE_EXPAND(...) __VA_ARGS__
