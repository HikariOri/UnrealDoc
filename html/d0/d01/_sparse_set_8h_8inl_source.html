227 void Empty(int32 ExpectedNumElements = 0)
228 {
229 // Empty the elements array, and reallocate it for the expected number of elements.
230 const int32 DesiredHashSize = Allocator::GetNumberOfHashBuckets(ExpectedNumElements);
231 const bool ShouldDoRehash = ShouldRehash(ExpectedNumElements, DesiredHashSize, EAllowShrinking::Yes);
232
233 if (!ShouldDoRehash)
234 {
235 // If the hash was already the desired size, clear the references to the elements that have now been removed.
236 UnhashElements();
237 }
238
239 Elements.Empty(ExpectedNumElements);
240
241 // Resize the hash to the desired size for the expected number of elements.
242 if (ShouldDoRehash)
243 {
244 HashSize = DesiredHashSize;
245 Rehash();
246 }
247 }

290 inline void Reserve(int32 Number)
291 {
292 // makes sense only when Number > Elements.Num() since TSparseArray::Reserve
293 // does any work only if that's the case
294 if ((USizeType)Number > (USizeType)Elements.Num())
295 {
296 // Trap negative reserves
297 if (Number < 0)
298 {
299 UE::Core::Private::OnInvalidSetNum((unsigned long long)Number);
300 }
301
302 // Preallocates memory for array of elements
303 Elements.Reserve(Number);
304
305 // Calculate the corresponding hash size for the specified number of elements.
306 const int32 NewHashSize = Allocator::GetNumberOfHashBuckets(Number);
307
308 // If the hash hasn't been created yet, or is smaller than the corresponding hash size, rehash
309 // to force a preallocation of the hash table
310 if(!HashSize || HashSize < NewHashSize)
311 {
312 HashSize = NewHashSize;
313 Rehash();
314 }
315 }

469 ElementType& FindOrAddByHash(uint32 KeyHash, ElementReferenceType&& InElement, bool* bIsAlreadyInSetPtr = nullptr)
470 {
471 SizeType ExistingIndex = FindIndexByHash(KeyHash, KeyFuncs::GetSetKey(InElement));
472 bool bIsAlreadyInSet = ExistingIndex != INDEX_NONE;
473 if (bIsAlreadyInSetPtr)
474 {
475 *bIsAlreadyInSetPtr = bIsAlreadyInSet;
476 }
477 if (bIsAlreadyInSet)

484 SetElementType& Element = *new (ElementAllocation) SetElementType(Forward<ElementReferenceType>(InElement));
485 RehashOrLink(KeyHash, Element, ElementAllocation.Index);

490 bool TryReplaceExisting(uint32 KeyHash, SetElementType& Element, SizeType& InOutElementIndex, bool* bIsAlreadyInSetPtr)
491 {
492 bool bIsAlreadyInSet = false;
493 if constexpr (!KeyFuncs::bAllowDuplicateKeys)
494 {
495 // If the set doesn't allow duplicate keys, check for an existing element with the same key as the element being added.
496
497 // Don't bother searching for a duplicate if this is the first element we're adding
498 if (Elements.Num() != 1)
499 {
500 SizeType ExistingIndex = FindIndexByHash(KeyHash, KeyFuncs::GetSetKey(Element.Value));
501 bIsAlreadyInSet = ExistingIndex != INDEX_NONE;
502 if (bIsAlreadyInSet)
503 {
504 // If there's an existing element with the same key as the new element, replace the existing element with the new element.
505 MoveByRelocate(Elements[ExistingIndex].Value, Element.Value);
506
507 // Then remove the new element.
508 Elements.RemoveAtUninitialized(InOutElementIndex);
509
510 // Then point the return value at the replaced element.
511 InOutElementIndex = ExistingIndex;
512 }
513 }
514 }
515 if (bIsAlreadyInSetPtr)
516 {
517 *bIsAlreadyInSetPtr = bIsAlreadyInSet;
518 }
519 return bIsAlreadyInSet;
520 }

522 inline void RehashOrLink(uint32 KeyHash, SetElementType& Element, SizeType ElementIndex)
523 {
524 // Check if the hash needs to be resized.
525 if (!ConditionalRehash(Elements.Num(), EAllowShrinking::No))
526 {
527 // If the rehash didn't add the new element to the hash, add it.
528 LinkElement(ElementIndex, Element, KeyHash);
529 }

688 void RemoveByIndex(SizeType ElementIndex)
689 {
690 checkf(Elements.IsValidIndex(ElementIndex), TEXT("Invalid ElementIndex passed to UE_TSPARSE_SET::RemoveByIndex"));
691
692 const SetElementType& ElementBeingRemoved = Elements[ElementIndex];
693
694 // Remove the element from the hash.
695 FSetElementId* HashPtr = Hash.GetAllocation();
696 FSetElementId* NextElementIndexIter = &HashPtr[ElementBeingRemoved.HashIndex];
697 for (;;)
698 {
699 SizeType NextElementIndex = NextElementIndexIter->AsInteger();
700 checkf(NextElementIndex != INDEX_NONE, TEXT("Corrupt hash"));
701
702 if (NextElementIndex == ElementIndex)
703 {
704 *NextElementIndexIter = ElementBeingRemoved.HashNextId;
705 break;
706 }
707
708 NextElementIndexIter = &Elements[NextElementIndex].HashNextId;
709 }
710
711 // Remove the element from the elements array.
712 Elements.RemoveAt(ElementIndex);
713 }

741 template <typename ComparableKey>
742 [[nodiscard]] SizeType FindIndexByHash(uint32 KeyHash, const ComparableKey& Key) const
743 {
744 if (Elements.Num() == 0)
745 {
746 return INDEX_NONE;
747 }
748
749 FSetElementId* HashPtr = Hash.GetAllocation();
750 SizeType ElementIndex = HashPtr[KeyHash & (HashSize - 1)].AsInteger();
751 for (;;)
752 {
753 if (ElementIndex == INDEX_NONE)
754 {
755 return INDEX_NONE;
756 }
757
758 if (KeyFuncs::Matches(KeyFuncs::GetSetKey(Elements[ElementIndex].Value), Key))
759 {
760 // Return the first match, regardless of whether the set has multiple matches for the key or not.
761 return ElementIndex;
762 }
763
764 ElementIndex = Elements[ElementIndex].HashNextId.AsInteger();
765 }

866 template<typename ComparableKey>
867 inline int32 RemoveImpl(uint32 KeyHash, const ComparableKey& Key)
868 {
869 int32 NumRemovedElements = 0;
870
871 FSetElementId* NextElementId = &GetTypedHash(KeyHash);
872 while (NextElementId->IsValidId())
873 {
874 const int32 ElementIndex = NextElementId->AsInteger();
875 SetElementType& Element = Elements[ElementIndex];
876
877 if (KeyFuncs::Matches(KeyFuncs::GetSetKey(Element.Value), Key))
878 {
879 // This element matches the key, remove it from the set. Note that Remove sets *NextElementId to point to the next
880 // element after the removed element in the hash bucket.
881 RemoveByIndex(ElementIndex);
882 NumRemovedElements++;
883
884 if constexpr (!KeyFuncs::bAllowDuplicateKeys)
885 {
886 // If the hash disallows duplicate keys, we're done removing after the first matched key.
887 break;
888 }
889 }
890 else
891 {
892 NextElementId = &Element.HashNextId;
893 }
894 }
895
896 return NumRemovedElements;
897 }
