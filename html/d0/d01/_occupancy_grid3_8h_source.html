19struct FOccupancyGrid3
20{
22 enum class EDomain : int32
23 {
24 Exterior,
25 Boundary,
26 Interior
27 };
28
29 template<typename MeshType>
30 FOccupancyGrid3(
31 const MeshType& InMesh,
32 const int32 InVoxelResolution
33 )
34 {
35 // Compute a voxel grid
36 TMeshAABBTree3<MeshType> Spatial(&InMesh);
37 TFastWindingTree FastWinding(&Spatial);
38 const FAxisAlignedBox3d Bounds = Spatial.GetBoundingBox();
39 CellSize = float(Bounds.MaxDim() / InVoxelResolution);
40 CellMidPoint = {CellSize / 2.0f, CellSize / 2.0f, CellSize / 2.0f};
41
42 TMeshWindingNumberGrid WindingGrid(&InMesh, &FastWinding, CellSize);
43
44 WindingGrid.Compute();
45
46 // Our occupancy grid is computed on the winding number grid's cell centers.
47 const FVector3i WindingDims = WindingGrid.Dimensions();
48 Occupancy = {WindingDims.X - 1, WindingDims.Y - 1, WindingDims.Z - 1, EDomain::Exterior};
49
50 GridOrigin = WindingGrid.GridOrigin + CellMidPoint;
51
52 static constexpr FVector3i CornerOffsets[] = {
53 FVector3i(0, 0, 0),
54 FVector3i(0, 0, 1),
55 FVector3i(0, 1, 0),
56 FVector3i(0, 1, 1),
57 FVector3i(1, 0, 0),
58 FVector3i(1, 0, 1),
59 FVector3i(1, 1, 0),
60 FVector3i(1, 1, 1),
61 };
62
63 if (!ensure(Occupancy.Num() <= (int64)MAX_int32))
64 {
65 // TODO: support occupancy grids of larger sizes
66 return;
67 }
68 ParallelFor(static_cast<int32>(Occupancy.Num()), [&](const int32 OccupancyId)
69 {
70 const FVector3i OccupancyIndex(Occupancy.GetCoords(OccupancyId));
71 const SIZE_T Count = Algo::CountIf(CornerOffsets, [&WindingGrid, OccupancyIndex](FVector3i CornerOffset)
72 {
73 const FVector3i CornerIndex(OccupancyIndex + CornerOffset);
74 return WindingGrid.GetValue(CornerIndex) >= WindingGrid.WindingIsoValue;
75 });
76
77 if (Count == 8)
78 {
79 Occupancy[OccupancyIndex] = EDomain::Interior;
80 }
81 else if (Count > 0)
82 {
83 Occupancy[OccupancyIndex] = EDomain::Boundary;
84 }
85 });
86
87 // Make sure we include all the vertices of the mesh as a part of the boundary, if
88 // the vertex areas are marked as being exterior.
89 for (int32 VertexIdx = 0; VertexIdx < InMesh.VertexCount(); VertexIdx++)
90 {
91 const FVector3d& Pos = InMesh.GetVertex(VertexIdx);
92 const FVector3i OccupancyIndex = GetCellIndexFromPoint(FVector(Pos));
93
94 if (ensure(Occupancy.IsValidIndex(OccupancyIndex)))
95 {
96 if (Occupancy[OccupancyIndex] == EDomain::Exterior)
97 {
98 Occupancy[OccupancyIndex] = EDomain::Boundary;
99 }
100 }
101 }
102 }
103
107 FVector3i GetCellIndexFromPoint(const FVector &InPoint) const
108 {
109 FVector3f PP(InPoint);
110 PP -= GridOrigin;
111
112 return { FMath::FloorToInt(PP.X / CellSize),
113 FMath::FloorToInt(PP.Y / CellSize),
114 FMath::FloorToInt(PP.Z / CellSize) };
115 }
116
119 FVector3f GetCellCenterFromIndex(const FVector3i &Index) const
120 {
121 return {float(Index.X) * CellSize + GridOrigin.X + CellMidPoint.X,
122 float(Index.Y) * CellSize + GridOrigin.Y + CellMidPoint.Y,
123 float(Index.Z) * CellSize + GridOrigin.Z + CellMidPoint.Z};
124 }
125
128 FBox3f GetCellBoxFromIndex(const FVector3i &Index) const
129 {
130 const FVector3f P = GetCellCenterFromIndex(Index);
131 return {P - CellMidPoint, P + CellMidPoint};
132 }
133
135 const TDenseGrid3<EDomain>& GetOccupancyStateGrid() const
136 {
137 return Occupancy;
138 }
139
140 // Returns the size of the cell. The cell size is uniform on each axis.
141 float GetCellSize() const
142 {
143 return CellSize;
144 }
145
146private:
147 // The dense grid that contains the occupancy state of the mesh within each voxel.
148 TDenseGrid3<EDomain> Occupancy;
149
150 // The size of each cell, applies to all dimensions.
151 float CellSize;
152
153 // The origin of the occupancy grid. Since the occupancy of a cell is defined at the winding number grid's cell
154 // centers, this is offset by half the cell size, compared to the winding number grid's origin (same as the
155 // mesh's origin, which defined by the lowest coordinate value).
156 FVector3f GridOrigin;
157
158 // This value is the offset from the lowest coordinate of a winding number grid's cell's box, to its mid-point.
159 FVector3f CellMidPoint;
160};
