171 void SetRect(const FIntRect& Rect)
172 {
173 // Do nothing if this isn't a texture
174 if (!Texture)
175 {
176 return;
177 }
178
179 // Reset the extents to default (zero), if the rect is empty or covers the full extent of the texture. Forces a more efficient full resource copy.
180 if (Rect.IsEmpty() || (Rect.Min.X == 0 && Rect.Min.Y == 0 && Rect.Max.X == Texture->GetDesc().Extent.X && Rect.Max.Y == Texture->GetDesc().Extent.Y))
181 {
182 Min = FIntVector3(0);
183 Max = FIntVector3(0);
184 return;
185 }
186
187 // Initialize the extent
188 Min = FIntVector3(Rect.Min.X, Rect.Min.Y, 0);
189 Max = FIntVector3(Rect.Max.X, Rect.Max.Y, 1);
190 }
191
192 // The texture which must be must be allocated on both GPUs
193 FTextureRHIRef Texture;
194 // Or alternately, a buffer that's allocated on both GPUs
195 FBufferRHIRef Buffer;
196 // The min rect of the texture region to copy
197 FIntVector Min;
198 // The max rect of the texture region to copy
199 FIntVector Max;
200 // The GPU index where the data will be read from.
201 uint32 SrcGPUIndex;
202 // The GPU index where the data will be written to.
203 uint32 DestGPUIndex;
204 // Whether the data is read by the dest GPU, or written by the src GPU (not allowed if the texture is a backbuffer)

206 // Whether the GPUs must handshake before and after the transfer. Required if the texture rect is being written to in several render passes.
207 // Otherwise, minimal synchronization will be used.
