269 void PopulateCultureList()
270 {
271 switch(FilterCulturesChoice)
272 {
273 case EFilterCulturesChoices::AllAvailableCultures:
274 {
275 TArray<FString> CultureNames;
276 FInternationalization::Get().GetCultureNames(CultureNames);
277
278 CultureList.Reset();
279 for(const FString& CultureName : CultureNames)
280 {
281 CultureList.Add(FInternationalization::Get().GetCulture(CultureName));
282 }
283 }
284 break;
285
286 case EFilterCulturesChoices::OnlyLocalizedCultures:
287 {
288 const TArray<FString> LocalizedCultureNames = FTextLocalizationManager::Get().GetLocalizedCultureNames(ELocalizationLoadFlags::Game);
289 const TArray<FCultureRef> LocalizedCultureList = FInternationalization::Get().GetAvailableCultures(LocalizedCultureNames, true);
290
291 CultureList.Reset();
292 CultureList.Append(LocalizedCultureList);
293 }
294 break;
295
296 default:
297 checkf(false, TEXT("Unknown EFilterCulturesChoices"));
298 break;
299 }

513 void ImportFromLauncherProfile( const ILauncherProfilePtr LauncherProfilePtr, TSharedPtr<IPropertyUtilities> PropertyUtilities )
514 {
515 ILauncherServicesModule& LauncherServicesModule = FModuleManager::LoadModuleChecked<ILauncherServicesModule>("LauncherServices");
516
517 const ILauncherProfileBuildCookRunPtr BuildCookRunPtr = LauncherProfilePtr->GetFirstBuildCookRun();
518 if (BuildCookRunPtr == nullptr)
519 {
520 return;
521 }
522
523 // grab the project packaging settings
524 UProjectPackagingSettings* ProjectPackagingSettings = UProjectPackagingSettings::StaticClass()->GetDefaultObject<UProjectPackagingSettings>();
525 if (ProjectPackagingSettings == nullptr)
526 {
527 return;
528 }
529
530 // ensure the name is unique (Turnkey builds a dictionary using the Name)
531 FString ProfileName = LauncherProfilePtr->GetName();
532 int UniqueId = 1;
533 while (ProjectPackagingSettings->ProjectCustomBuilds.ContainsByPredicate( [ProfileName](const FProjectBuildSettings& Other) { return ProfileName == Other.Name; } ) )
534 {
535 ProfileName = FString::Printf( TEXT("%s %d"), *LauncherProfilePtr->GetName(), UniqueId++ );
536 }
537
538 // add a new item
539 FProjectBuildSettings& ProjectBuildSettings = ProjectPackagingSettings->ProjectCustomBuilds.AddDefaulted_GetRef();
540 ProjectBuildSettings.Name = ProfileName;
541 ProjectBuildSettings.HelpText = LauncherProfilePtr->GetDescription().IsEmpty() ? LauncherProfilePtr->GetName() : LauncherProfilePtr->GetDescription();
542 ProjectBuildSettings.SpecificPlatforms = BuildCookRunPtr->GetCookedPlatforms();
543 if (ProjectBuildSettings.SpecificPlatforms.Num() == 0 && !BuildCookRunPtr->GetDefaultDeployPlatform().IsNone())
544 {
545 ProjectBuildSettings.SpecificPlatforms.Add(BuildCookRunPtr->GetDefaultDeployPlatform().ToString());
546 }
547 ProjectBuildSettings.BuildCookRunParams = LauncherServicesModule.GetProfileManager()->MakeBuildCookRunParamsForProjectCustomBuild(LauncherProfilePtr.ToSharedRef(), ProjectBuildSettings.SpecificPlatforms);
548
549
550 // signal that the property has changed
551 FNotifyHook* NotifyHook = PropertyUtilities->GetNotifyHook();
552 FProperty* Property = UProjectPackagingSettings::StaticClass()->FindPropertyByName("ProjectCustomBuilds");
553 if (NotifyHook != nullptr && Property != nullptr)
554 {
555 TArray<const UObject*> NotifyTopLevelObjects;
556 NotifyTopLevelObjects.Add(ProjectPackagingSettings);
557
558 FEditPropertyChain PropertyChain;
559 PropertyChain.AddHead(Property);
560
561 FPropertyChangedEvent ChangeEvent(Property, EPropertyChangeType::ValueSet, MakeArrayView(NotifyTopLevelObjects));
562 NotifyHook->NotifyPostChange(ChangeEvent, &PropertyChain);
563 }
564 }
