75 static double EvaluateSag(const FVector& PointPoint, const FVector& PointNext, const FVector& PointMiddle, double& Length)
76 {
77 FVector ChordVec = PointNext - PointPoint;
78 FVector MiddleVec = PointMiddle - PointPoint;
79
80 double Sag = 0.0;
81 double NormSqrVec = ChordVec | ChordVec;
82 Length = sqrt(NormSqrVec);
83
84 if (NormSqrVec > DOUBLE_SMALL_NUMBER)
85 {
86 FVector VecSag = ChordVec ^ MiddleVec;
87 double NormSqrSag = VecSag | VecSag;
88 Sag = NormSqrSag / NormSqrVec;
89
90 if (Sag < SMALL_NUMBER_SQUARE)
91 {
92 Sag = 0.;
93 }
94 else
95 {
96 Sag = sqrt(Sag);
97 }
98 }
99 return Sag;
100 }
