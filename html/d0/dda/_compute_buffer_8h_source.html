14//
15// Implements a ring buffer using shared memory, with one writer and multiple readers.
16//
17// Contention between the writer and readers is resolved by splitting the buffer into a number of
18// fixed-size chunks, which can be appended to and read immediately, but only reset once consumed
19// by all readers.
20//
21// If the read position is never advanced and only one chunk is used, the buffer functions as an
22// append-only buffer.
23//
24class FComputeBuffer
25{
26public:
27 // Maximum allowed number of readers
28 static const int MaxReaders = 16;
29
30 // Maximum allowed number of chunks
31 static const int MaxChunks = 16;
32
33 // Maximum length of a compute buffer name
34 static const size_t MaxNameLength = 256;

36 // Parameters for constructing a compute buffer
37 struct FParams
38 {
39 const char* Name = nullptr; // When creating a buffer, we should not normally set the name. We can only *OPEN* buffers with a name. Usually we create buffers and attach them to channels.
40 int NumChunks = 2;
41 int ChunkLength = 512 * 1024;
42 int NumReaders = 1;
43 bool Anonymous = false; // Set this to true to use anonymous buffers. This could be used together with for example socket transports
44 };
