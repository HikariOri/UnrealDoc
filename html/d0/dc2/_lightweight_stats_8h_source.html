60 // A TStatId that we've already checked can be emitted (via GetStatIfEnabled).
61 struct FCheckedStat
62 {
63 // Validity is separate from the stat name being not-null
64 // as even if the stat shouldn't be emitted as a named event, we may need the stat name for other
65 // things such as the FHitchTrackingStatScope.
66 // However, it's worth noting the StatString can be null for cases where we do not want to emit a hitch scope
67 // which is the case for conditional scopes when the condition is false.
68 static constexpr FCheckedStat MakeValid(const PROFILER_CHAR* InStatString)
69 {
70 return FCheckedStat{ InStatString, true };
71 }
72
73 static constexpr FCheckedStat MakeInvalid(const PROFILER_CHAR* InStatString)
74 {
75 return FCheckedStat{ InStatString, false };
76 }
77
78 // Allow implicit conversions to TStatId as FCheckedStat can be assigned to TStatId via GetStatIfEnabled.
79 constexpr operator TStatId() const
80 {
81 // Systems that use TStatId directly test validity through the stat string.
82 return bIsValid ? TStatId(StatString) : TStatId{};
83 }
84
85 // Returns the stat name, even if this stat isn't valid.
86 constexpr const PROFILER_CHAR* GetName() const
87 {
88 return StatString;
89 }
90
91 // Returns true if this stat should be emitted.
92 constexpr bool IsValidStat() const
93 {
94 return bIsValid;
95 }
96
97 private:
98 constexpr explicit FCheckedStat(const PROFILER_CHAR* InStatString, bool bValidStat)
99 : StatString(InStatString)
100 , bIsValid(bValidStat)
101 {
102 }
103
104 const PROFILER_CHAR* StatString;
105 bool bIsValid;
106 };

309 // Tests if this stat can be emitted
310 template<typename StatDataType = FDefaultStatData, typename GroupDataType = typename StatDataType::TGroup>
311 inline bool IsStatEnabled(uint32 StatNameHash, uint32 GroupNameHash = 0)
312 {
313 if constexpr (!GroupDataType::IsCompileTimeEnable())
314 {
315 return false;
316 }
317 else
318 {
319 if constexpr (EnumHasAnyFlags(StatDataType::GetFlags(), EStatFlags::Verbose))
320 {
321 if (!GShouldEmitVerboseNamedEvents)
322 {
323 return false;
324 }
325 }
326
327#if UE_STATS_ALLOW_PER_THREAD_IGNORELIST
328 const uint32 GroupHash = GroupNameHash != 0 ? GroupNameHash : GroupDataType::GetNameHash();
329 if (UE::Stats::IsStatOrGroupIgnoredOnCurrentThread(StatNameHash, GroupHash))
330 {
331 return false;
332 }
333#endif // UE_STATS_ALLOW_PER_THREAD_IGNORELIST
334 return true;
335 }
336 }

345 template<typename StatDataType = FDefaultStatData, typename GroupDataType = typename StatDataType::TGroup>
346 FORCEINLINE FCheckedStat GetStatIfEnabled(TNotNull<const PROFILER_CHAR*> StatName, uint32 StatNameHash, uint32 GroupNameHash = 0)
347 {
348 // We're testing if named events are enabled here as it's cheaper than some of the checks in IsStatEnabled.
349 if (GCycleStatsShouldEmitNamedEvents > 0 &&
350 IsStatEnabled<StatDataType, GroupDataType>(StatNameHash, GroupNameHash))
351 {
352 return FCheckedStat::MakeValid(StatName);
353 }
354 return FCheckedStat::MakeInvalid(StatName);
355 }
356
357 /* Returns a valid TStatId if the condition is true and we can emit it. */
358 template<typename StatDataType = FDefaultStatData, typename GroupDataType = typename StatDataType::TGroup>
359 FORCEINLINE FCheckedStat GetStatIfEnabled(TNotNull<const PROFILER_CHAR*> StatName, bool bCondition, uint32 StatNameHash, uint32 GroupNameHash = 0)
360 {
361 if (bCondition)
362 {
363 return GetStatIfEnabled<StatDataType, GroupDataType>(StatName, StatNameHash, GroupNameHash);
364 }
365 // Preserve existing behavior of not emitting hitch scopes if the condition is false.
366 return FCheckedStat::MakeInvalid(nullptr);
367 }
