43 void MinifyInPlace()
44 {
45 // Eliminate in-order, lets us get away with a single text cursor
46 SymbolCtx.StoreSites.Sort([](const FMemberStoreSite& A, const FMemberStoreSite& B)
47 {
48 return A.AssignmentSite.Begin < B.AssignmentSite.Begin;
49 });
50
51 // Iteratively eliminate the realized loads of assignments we know are to be DCE'd
52 // i.e., If we have:
53 // Data.A = 41;
54 // Data.B = Data.A + 1;
55 //
56 // We know B is to be DCE'd, so we remove the realized loads on its RHS value span.
57 // After which A is known unrealized. And so on.
58 for (;;)
59 {
60 bool bMutatedRealizedLoad = false;
61
62 // Try to check for new unrealized loads
63 for (FMemberStoreSite& StoreSite : SymbolCtx.StoreSites)
64 {
65 check(StoreSite.AddressSymbol->bTagged);
66
67 // Still in use?
68 if (StoreSite.bVisited || StoreSite.AddressSymbol->Variable.RealizedLoadCount > 0)
69 {
70 continue;
71 }
72
73 // Mark as visited, fine before complex check since it's immutable
74 StoreSite.bVisited = true;
75
76 // Check if the symbol is considered "complex", i.e., the identifier was used in a
77 // code section that we couldn't parse or derive semantics from
78 StoreSite.bIsResolvedToComplex = SymbolCtx.ComplexBaseSet.Contains(StoreSite.AddressSymbol->ID);
79 if (StoreSite.bIsResolvedToComplex)
80 {
81 continue;
82 }
83
84 // Remove all the loads on the RHS
85 SemanticCtx.RemoveRealizedLoads(StoreSite.ValueSpan);
86 bMutatedRealizedLoad = true;
87 }
88
89 if (!bMutatedRealizedLoad)
90 {
91 break;
92 }
93 }
94
95 // Finally, eliminate!
96 for (const FMemberStoreSite& StoreSite : SymbolCtx.StoreSites)
97 {
98 check(StoreSite.AddressSymbol->bTagged);
99
100 // Ignore if the address has a realized load or complex in nature
101 if (StoreSite.AddressSymbol->Variable.RealizedLoadCount > 0 || StoreSite.bIsResolvedToComplex)
102 {
103 continue;
104 }
105
106 // Replace with spaces while maintaining line numbers
107 for (int32 i = 0; i < StoreSite.AssignmentSite.Length; i++)
108 {
109 if (StoreSite.AssignmentSite.Begin[i] != '\n')
110 {
111 StoreSite.AssignmentSite.Begin[i] = ' ';
112 }
113 }
114 }
115 }

119 void VisitStructPrologue(const FSmallStringView& ID)
120 {
121 // Push the stack
122 FSymbolScope& Scope = SymbolCtx.SymbolStack.Emplace_GetRef();
123
124 // Is this a tagged type?
125 const bool bTagged = TaggedTypes.Contains(ID);
126
127 #if UE_SHADER_SDCE_LOG_STRUCTURAL
128 UE_LOGF(LogTemp, Error, ">> ENTER %s", *FParseStringType::ConstructFromPtrSize(ID.Begin, ID.Length));
129 #endif // UE_SHADER_SDCE_LOG_STRUCTURAL
130
131#if UE_SHADER_SDCE_TAGGED_ONLY
132 if (!bTagged)
133 {
134 return;
135 }
136#endif // UE_SHADER_SDCE_TAGGED_ONLY
137
138 // Assign composite of current stack
139 FSymbol* Symbol = SymbolCtx.CreateSymbol(ESymbolKind::Composite);
140 Symbol->ID = ID;
141 Symbol->bTagged = bTagged;
142 Scope.Composite = Symbol;
143
144 // Add to parent stack
145 SymbolCtx.SymbolStack[SymbolCtx.SymbolStack.Num() - 2].Symbols.Add(ID, Symbol);
146 }
