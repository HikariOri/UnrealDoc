54 template<typename ValueType>
55 bool GetCustomAttribute(const UE::Interchange::FAttributeStorage& Attributes, const UE::Interchange::FAttributeKey& AttributeKey, const FString& OperationName, ValueType& OutAttributeValue)
56 {
57 if (!Attributes.ContainAttribute(AttributeKey))
58 {
59 return false;
60 }
61 UE::Interchange::FAttributeStorage::TAttributeHandle<ValueType> AttributeHandle = Attributes.GetAttributeHandle<ValueType>(AttributeKey);
62 if (!AttributeHandle.IsValid())
63 {
64 return false;
65 }
66 UE::Interchange::EAttributeStorageResult Result = AttributeHandle.Get(OutAttributeValue);
67 if (!UE::Interchange::IsAttributeStorageResultSuccess(Result))
68 {
69 UE::Interchange::LogAttributeStorageErrors(Result, OperationName, AttributeKey);
70 return false;
71 }
72 return true;
73 }
