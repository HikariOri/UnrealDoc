101 [[nodiscard]] friend inline bool Intersect(const FBoxCenterAndExtent& A,const FBoxCenterAndExtent& B)
102 {
103 // CenterDifference is the vector between the centers of the bounding boxes.
104 const VectorRegister CenterDifference = VectorAbs(VectorSubtract(VectorLoadAligned(&A.Center),VectorLoadAligned(&B.Center)));
105
106 // CompositeExtent is the extent of the bounding box which is the convolution of A with B.
107 const VectorRegister CompositeExtent = VectorAdd(VectorLoadAligned(&A.Extent),VectorLoadAligned(&B.Extent));
108
109 // For each axis, the boxes intersect on that axis if the projected distance between their centers is less than the sum of their
110 // extents. If the boxes don't intersect on any of the axes, they don't intersect.
111 return VectorAnyGreaterThan(CenterDifference,CompositeExtent) == false;
112 }

122 const VectorRegister CenterDifference = VectorAbs(VectorSubtract(VectorLoadFloat3_W0(&A.Origin),VectorLoadAligned(&B.Center)));

125 const VectorRegister CompositeExtent = VectorAdd(VectorLoadFloat3_W0(&A.BoxExtent),VectorLoadAligned(&B.Extent));

127 // For each axis, the boxes intersect on that axis if the projected distance between their centers is less than the sum of their
128 // extents. If the boxes don't intersect on any of the axes, they don't intersect.

317 const FReal TightChildExtent = Bounds.Extent.X * 0.5f;
318 const FReal LooseChildExtent = TightChildExtent * (1.0f + 1.0f / (FReal)LoosenessDenominator);
319
320 ChildExtent = LooseChildExtent;
321 ChildCenterOffset = Bounds.Extent.X - LooseChildExtent;
322 }

324 [[nodiscard]] inline VectorRegister GetChildOffsetVec(int i) const
325 {
326 // LWC_TODO: not sure this is correct for VectorRegister = VectorRegister4Double
327 union MaskType { VectorRegister4Float v; VectorRegister4Int i;
328 } Mask;
329
330 Mask.v = MakeVectorRegister(1u, 2u, 4u, 8u);
331 VectorRegister4Int X = VectorIntLoad1(&i);
332 VectorRegister4Int A = VectorIntAnd(X, Mask.i);
333 Mask.i = VectorIntCompareEQ(Mask.i, A);
334 return VectorSelect(Mask.v, VectorSetFloat1(ChildCenterOffset), VectorSetFloat1(-ChildCenterOffset));
335 }

338 [[nodiscard]] inline FOctreeNodeContext GetChildContext(FOctreeChildNodeRef ChildRef) const
339 {
340 FBoxCenterAndExtent LocalBounds;
341 VectorRegister ZeroW = MakeVectorRegister(1.0f, 1.0f, 1.0f, 0.0f);
342 VectorStoreAligned(VectorMultiply(ZeroW, VectorAdd(VectorLoadAligned(&Bounds.Center), GetChildOffsetVec(ChildRef.Index))), &(LocalBounds.Center.X));
343 VectorStoreAligned(VectorMultiply(ZeroW, VectorSetFloat1(ChildExtent)), &(LocalBounds.Extent.X));
344 return FOctreeNodeContext(LocalBounds);
345 }

348 inline void GetChildContext(FOctreeChildNodeRef ChildRef, FOctreeNodeContext* ChildContext) const
349 {
350 VectorRegister ZeroW = MakeVectorRegister(1.0f, 1.0f, 1.0f, 0.0f);
351 VectorStoreAligned(VectorMultiply(ZeroW, VectorAdd(VectorLoadAligned(&Bounds.Center), GetChildOffsetVec(ChildRef.Index))), &(ChildContext->Bounds.Center.X));
352 VectorStoreAligned(VectorMultiply(ZeroW, VectorSetFloat1(ChildExtent)), &(ChildContext->Bounds.Extent.X));
353
354 const FReal TightChildExtent = ChildExtent * 0.5f;
355 const FReal LooseChildExtent = TightChildExtent * (1.0f + 1.0f / (FReal)LoosenessDenominator);
356 ChildContext->ChildExtent = LooseChildExtent;
357 ChildContext->ChildCenterOffset = ChildExtent - LooseChildExtent;
358 }

366 VectorStoreAligned(VectorMultiply(ZeroW, VectorSetFloat1(ChildExtent)), &(LocalBounds.Extent.X));

429 void Push(FNodeIndex NodeIndex)
430 {
431 //find the index that points to our right side node
432 int Index = 1; //exclude the dummy
433 int Size = FreeList.Num() - 1;
434
435 //start with binary search for larger lists
436 while (Size > 32)
437 {
438 const int LeftoverSize = Size % 2;
439 Size = Size / 2;
440
441 const int CheckIndex = Index + Size;
442 const int IndexIfLess = CheckIndex + LeftoverSize;
443
444 Index = FreeList[CheckIndex].Start > NodeIndex ? IndexIfLess : Index;
445 }
446
447 //small size array optimization
448 int ArrayEnd = FMath::Min(Index + Size + 1, FreeList.Num());
449 while (Index < ArrayEnd)
450 {
451 if (FreeList[Index].Start < NodeIndex)
452 {
453 break;
454 }
455 Index++;
456 }
457
458 //can we merge with the right node
459 if (Index < FreeList.Num() && FreeList[Index].End + 1 == NodeIndex)
460 {
461 FreeList[Index].End = NodeIndex;
462
463 //are we filling the gap between the left and right node
464 if (FreeList[Index - 1].Start - 1 == NodeIndex)
465 {
466 FreeList[Index - 1].Start = FreeList[Index].Start;
467 FreeList.RemoveAt(Index);
468 }
469 return;
470 }
471
472 //can we merge with the left node
473 if (FreeList[Index - 1].Start - 1 == NodeIndex)
474 {
475 FreeList[Index - 1].Start = NodeIndex;
476 return;
477 }
478
479 //we are node that could not be merged
480 FreeList.Insert(FSpan{ NodeIndex , NodeIndex }, Index);
481 }
482
483 FNodeIndex Pop()
484 {
485 FSpan& Span = FreeList.Last();
486 FNodeIndex Index = Span.Start;
487 checkSlow(Index != INDEX_NONE);
488 if (Span.Start == Span.End)
489 {
490 FreeList.Pop();
491 return Index;
492 }
493 else
494 {
495 Span.Start++;
496 return Index;
497 }
498 }
499
500 void Reset()
501 {
502 FreeList.Reset(1);
503 //push a dummy
504 FreeList.Push(FSpan{ INDEX_NONE, INDEX_NONE });
505 }
506
507 [[nodiscard]] int Num() const
508 {
509 //includes one dummy
510 return FreeList.Num() - 1;
511 }
512 };

548 void AddElementInternal(FNodeIndex CurrentNodeIndex, const FOctreeNodeContext& NodeContext, const FBoxCenterAndExtent& ElementBounds, typename TCallTraits<ElementType>::ConstReference Element, ElementArrayType& TempElementStorage)
549 {
550 checkSlow(CurrentNodeIndex != INDEX_NONE);
551 TreeNodes[CurrentNodeIndex].InclusiveNumElements++;
552 if (TreeNodes[CurrentNodeIndex].IsLeaf())
553 {
554 if (TreeElements[CurrentNodeIndex].Num() + 1 > OctreeSemantics::MaxElementsPerLeaf && NodeContext.Bounds.Extent.X > MinLeafExtent)
555 {
556 TempElementStorage = MoveTemp(TreeElements[CurrentNodeIndex]);
557
558 FNodeIndex ChildStartIndex = AllocateEightNodes();
559 ParentLinks[(ChildStartIndex - 1) / 8] = CurrentNodeIndex;
560 TreeNodes[CurrentNodeIndex].ChildNodes = ChildStartIndex;
561 TreeNodes[CurrentNodeIndex].InclusiveNumElements = 0;
562
563 for (typename TCallTraits<ElementType>::ConstReference ChildElement : TempElementStorage)
564 {
565 const FBoxCenterAndExtent ChildElementBounds(OctreeSemantics::GetBoundingBox(ChildElement));
566 AddElementInternal(CurrentNodeIndex, NodeContext, ChildElementBounds, ChildElement, TempElementStorage);
567 }
568
569 TempElementStorage.Reset();
570 AddElementInternal(CurrentNodeIndex, NodeContext, ElementBounds, Element, TempElementStorage);
571 return;
572 }
573 else
574 {
575 int ElementIndex = TreeElements[CurrentNodeIndex].Emplace(Element);
576
577 SetElementId(Element, FOctreeElementId2(CurrentNodeIndex, ElementIndex));
578 return;
579 }
580 }
581 else
582 {
583 const FOctreeChildNodeRef ChildRef = NodeContext.GetContainingChild(ElementBounds);
584 if (ChildRef.IsNULL())
585 {
586 int ElementIndex = TreeElements[CurrentNodeIndex].Emplace(Element);
587 SetElementId(Element, FOctreeElementId2(CurrentNodeIndex, ElementIndex));
588 return;
589 }
590 else
591 {
592 FNodeIndex ChildNodeIndex = TreeNodes[CurrentNodeIndex].ChildNodes + ChildRef.Index;
593 FOctreeNodeContext ChildNodeContext = NodeContext.GetChildContext(ChildRef);
594 AddElementInternal(ChildNodeIndex, ChildNodeContext, ElementBounds, Element, TempElementStorage);
595 return;
596 }
597 }

708 template<typename IterateFunc>
709 void FindNearbyElementsInternal(FNodeIndex CurrentNodeIndex, const FOctreeNodeContext& NodeContext, const FBoxCenterAndExtent& BoxBounds, const IterateFunc& Func) const
710 {
711 if (TreeNodes[CurrentNodeIndex].InclusiveNumElements > 0)
712 {
713 for (int Index = 0; Index < TreeElements[CurrentNodeIndex].Num(); Index++)
714 {
715 Func(TreeElements[CurrentNodeIndex][Index]);
716 }
717
718 if (!TreeNodes[CurrentNodeIndex].IsLeaf())
719 {
720 // Find the child of the current node, if any, that contains the current new point
721 FOctreeChildNodeRef ChildRef = NodeContext.GetContainingChild(BoxBounds);
722 if (!ChildRef.IsNULL())
723 {
724 FNodeIndex ChildStartIndex = TreeNodes[CurrentNodeIndex].ChildNodes;
725 // If the specified child node exists and contains any match, push it than process it
726 if (TreeNodes[ChildStartIndex + ChildRef.Index].InclusiveNumElements > 0)
727 {
728 FindNearbyElementsInternal(ChildStartIndex + ChildRef.Index, NodeContext.GetChildContext(ChildRef), BoxBounds, Func);
729 }
730 // If the child node doesn't is a match, it's not worth pursuing any further. In an attempt to find
731 // anything to match vs. the new point, process all of the children of the current octree node
732 else
733 {
734 for (int8 i = 0; i < 8; i++)
735 {
736 FindNearbyElementsInternal(ChildStartIndex + i, NodeContext.GetChildContext(FOctreeChildNodeRef(i)), BoxBounds, Func);
737 }
738 }
739 }
740 }

845 {

847

849 TreeElements[ElementId.NodeIndex].RemoveAtSwap(ElementId.ElementIndex, EAllowShrinking::No);
850

852 {
853 // Update the external element id for the element that was swapped into the vacated element index.
854 SetElementId(TreeElements[ElementId.NodeIndex][ElementId.ElementIndex], ElementId);
855 }
856
857 FNodeIndex CollapseNodeIndex = INDEX_NONE;
858 {
859 // Update the inclusive element counts for the nodes between the element and the root node,
860 // and find the largest node that is small enough to collapse.
861 FNodeIndex NodeIndex = ElementId.NodeIndex;
862 while (true)

864 TreeNodes[NodeIndex].InclusiveNumElements--;
865 if (TreeNodes[NodeIndex].InclusiveNumElements < OctreeSemantics::MinInclusiveElementsPerNode)
866 {
867 CollapseNodeIndex = NodeIndex;
868 }

870 if (NodeIndex == 0)

872 break;

875 NodeIndex = ParentLinks[(NodeIndex - 1) / 8];
876 }
877 }
878

880 if (CollapseNodeIndex != INDEX_NONE && !TreeNodes[CollapseNodeIndex].IsLeaf())

1356 {

1358 const FNode& CurrentNode = NodeIt.GetCurrentNode();

1372 {
1373 check(NodeIt.HasPendingNodes()); // please don't call this after iteration has ended
1374
1375 while (1)
1376 {
1377 ElementConstIt LocalElementIt(ElementIt);
1378 if (LocalElementIt)
1379 {
1380 FPlatformMisc::Prefetch(&(*LocalElementIt));
1381 FPlatformMisc::Prefetch(&(*LocalElementIt), PLATFORM_CACHE_LINE_SIZE);
1382

1384 // Check if the current element intersects the bounding box.
1385 if (Intersect(OctreeSemantics::GetBoundingBox(*LocalElementIt), IteratorBounds))
1386 {
1387 // If it intersects, break out of the advancement loop.
1388 Move(ElementIt, LocalElementIt);

1393 while (++LocalElementIt)
1394 {

1398 if (Intersect(OctreeSemantics::GetBoundingBox(*LocalElementIt), IteratorBounds))
1399
1400 {

1436 {
1437 check(ElementId.IsValidId());
1438
1439 FNode* ElementIdNode = (FNode*)ElementId.Node;
1440
1441 // Remove the element from the node's element list.
1442 ElementIdNode->Elements.RemoveAtSwap(ElementId.ElementIndex);
1443
1444 SetOctreeMemoryUsage(this, TotalSizeBytes - sizeof(ElementType));
1445
1446 if (ElementId.ElementIndex < ElementIdNode->Elements.Num())
1447 {
1448 // Update the external element id for the element that was swapped into the vacated element index.
1449 SetElementId(ElementIdNode->Elements[ElementId.ElementIndex], ElementId);
1450 }
1451
1452 // Update the inclusive element counts for the nodes between the element and the root node,
1453 // and find the largest node that is small enough to collapse.
1454 const FNode* CollapseNode = NULL;
1455 for (const FNode* Node = ElementIdNode; Node; Node = Node->Parent)
1456 {
1457 --Node->InclusiveNumElements;
1458 if (Node->InclusiveNumElements < OctreeSemantics::MinInclusiveElementsPerNode)
1459 {
1460 CollapseNode = Node;
1461 }
1462 }
1463
1464 // Collapse the largest node that was pushed below the threshold for collapse by the removal.
1465 if (CollapseNode && !CollapseNode->bIsLeaf)
1466 {
1467 if (CollapseNode->Elements.Num() < (int32)CollapseNode->InclusiveNumElements)
1468 {
1469 CollapseNode->Elements.Reserve(CollapseNode->InclusiveNumElements);
1470
1471 // Gather the elements contained in this node and its children.
1472 for (TConstIterator<> ChildNodeIt(*CollapseNode, RootNodeContext); ChildNodeIt.HasPendingNodes(); ChildNodeIt.Advance())
1473 {
1474 const FNode& ChildNode = ChildNodeIt.GetCurrentNode();
1475
1476 if (&ChildNode != CollapseNode)
1477 {
1478 // Child node will be collapsed so move the child's elements to the collapse node element list.
1479 for (ElementType& Element : ChildNode.Elements)
1480 {
1481 const int32 NewElementIndex = CollapseNode->Elements.Add(MoveTemp(Element));
1482
1483 // Update the external element id for the element that's being collapsed.
1484 SetElementId(CollapseNode->Elements[NewElementIndex], FOctreeElementId(CollapseNode, NewElementIndex));
1485 }
1486 }
1487
1488 // Recursively visit all child nodes.
1489 FOREACH_OCTREE_CHILD_NODE(ChildRef)
1490 {
1491 if (ChildNode.HasChild(ChildRef))
1492 {
1493 ChildNodeIt.PushChild(ChildRef);
1494 }
1495 }
1496 }
1497
1498 // Free the child nodes.
1499 FOREACH_OCTREE_CHILD_NODE(ChildRef)
1500 {
1501 if (CollapseNode->Children[ChildRef.Index])
1502 {
1503 SetOctreeMemoryUsage(this, TotalSizeBytes - sizeof(*CollapseNode->Children[ChildRef.Index]));
1504 }
1505
1506 delete CollapseNode->Children[ChildRef.Index];
1507 CollapseNode->Children[ChildRef.Index] = NULL;
1508 }
1509 }
1510
1511 // Mark the node as a leaf.
1512 CollapseNode->bIsLeaf = true;
1513 }

1715 {
1716 const FBoxCenterAndExtent ElementBounds(OctreeSemantics::GetBoundingBox(Element));
1717
1718 for (TConstIterator<TInlineAllocator<1> > NodeIt(InNode, InContext); NodeIt.HasPendingNodes(); NodeIt.Advance())

1722 const bool bIsLeaf = Node.IsLeaf();
1723

1726 // Increment the number of elements included in this node and its children.
1727 Node.InclusiveNumElements++;
1728
1729 if (bIsLeaf)
1730 {
1731 // If this is a leaf, check if adding this element would turn it into a node by overflowing its element list.
1732 if (Node.Elements.Num() + 1 > OctreeSemantics::MaxElementsPerLeaf && Context.Bounds.Extent.X > MinLeafExtent)
1733 {

1735 ElementArrayType ChildElements;
1736 Exchange(ChildElements, Node.Elements);
1737 SetOctreeMemoryUsage(this, TotalSizeBytes - ChildElements.Num() * sizeof(ElementType));
1738 Node.InclusiveNumElements = 0;
1739
1740 // Allow elements to be added to children of this node.
1741 Node.bIsLeaf = false;
1742
1743 // Re-add all of the node's child elements, potentially creating children of this node for them.
1744 for (ElementConstIt ElementIt(ChildElements); ElementIt; ++ElementIt)
1745 {
1746 AddElementToNode(*ElementIt, Node, Context);
1747 }
1748
1749 // Add the element to this node.
1750 AddElementToNode(Element, Node, Context);

1752 }

1756 bAddElementToThisNode = true;
1757 }
1758 }

1763 if (ChildRef.IsNULL())

1766 bAddElementToThisNode = true;

1778 NodeIt.PushChild(ChildRef);
1779 }
1780 }
