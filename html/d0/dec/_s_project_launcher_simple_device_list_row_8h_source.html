370PRAGMA_DISABLE_DEPRECATION_WARNINGS
371 void UpdateProfile()
372 {
373 check(DeviceProxy.IsValid());
374
375 // This detects a corrupt device, it doesn't even have data for it's default variant Remove the profile.
376 // Entry will show invalid.
377 if (!DeviceProxy->HasVariant(NAME_None))
378 {
379 Model->GetProfileManager()->RemoveProfile(LaunchProfile.ToSharedRef());
380 LaunchProfile = nullptr;
381 }
382
383 if (LaunchProfile.IsValid())
384 {
385 FName Variant = SimpleProfile->GetDeviceVariant();
386
387 // If the profile refers to a variant no longer supported by this device fall back to the default.
388 if (!DeviceProxy->HasVariant(Variant))
389 {
390 Variant = NAME_None;
391 SimpleProfile->SetDeviceVariant(Variant);
392 }
393
394 // Setup the Profile
395 LaunchProfile->SetDeploymentMode(ELauncherProfileDeploymentModes::FileServer);
396
397 ILauncherDeviceGroupRef NewGroup = Model->GetProfileManager()->CreateUnmanagedDeviceGroup();
398
399 // multi-device targets (All_<platform>_devices_on_<host>) return a list of devices
400 const TSet<FString>& TargetDeviceIds = DeviceProxy->GetTargetDeviceIds(NAME_None);
401 for (TSet<FString>::TConstIterator ItDeviceID(TargetDeviceIds); ItDeviceID; ++ItDeviceID)
402 {
403 NewGroup->AddDevice(*ItDeviceID);
404 }
405 LaunchProfile->SetDeployedDeviceGroup(NewGroup);
406
407 LaunchProfile->ClearCookedPlatforms();
408 LaunchProfile->AddCookedPlatform(DeviceProxy->GetTargetPlatformName(Variant));
409
410 bool Advanced = IsAdvanced.IsBound() && IsAdvanced.Get();
411 if (Advanced)
412 {
413 LaunchProfile->SetBuildConfiguration(SimpleProfile->GetBuildConfiguration());
414 LaunchProfile->SetCookMode(SimpleProfile->GetCookMode());
415 }
416 }
417 }
