294 template <typename T, typename TEnableIf<TIsDerivedFrom<std::remove_pointer_t<T>, UObject>::Value>::Type* = nullptr>
295 static EPropertyBagResult CheckType(FPropertyBagPropertyDesc InDesc, const UObject* InValue)
296 {
297 if (!InDesc.ContainerTypes.IsEmpty())
298 {
299 return EPropertyBagResult::TypeMismatch;
300 }
301
302 if (InDesc.ValueType == EPropertyBagPropertyType::Object || InDesc.ValueType == EPropertyBagPropertyType::SoftObject)
303 {
304 if (!IsValid(InValue))
305 {
306 return EPropertyBagResult::Success;
307 }
308 else if (const FObjectPropertyBase* ObjectProperty = CastField<FObjectPropertyBase>(InDesc.CachedProperty))
309 {
310 const UClass* Class = InValue->GetClass();
311 const UClass* PropertyClass = ObjectProperty->PropertyClass;
312
313 if (Class && PropertyClass && Class->IsChildOf(PropertyClass))
314 {
315 return EPropertyBagResult::Success;
316 }
317 }
318 }
319 else if (InDesc.ValueType == EPropertyBagPropertyType::Class || InDesc.ValueType == EPropertyBagPropertyType::SoftClass)
320 {
321 if (!IsValid(InValue))
322 {
323 return EPropertyBagResult::Success;
324 }
325
326 const UClass* Class = Cast<UClass>(InValue);
327 const UClass* PropertyClass = nullptr;
328
329 if (const FClassProperty* ClassProperty = CastField<FClassProperty>(InDesc.CachedProperty))
330 {
331 PropertyClass = ClassProperty->MetaClass;
332 }
333 else if (const FSoftClassProperty* SoftClassProperty = CastField<FSoftClassProperty>(InDesc.CachedProperty))
334 {
335 PropertyClass = SoftClassProperty->MetaClass;
336 }
337
338 if (Class && PropertyClass && Class->IsChildOf(PropertyClass))
339 {
340 return EPropertyBagResult::Success;
341 }
342 }
343
344 return EPropertyBagResult::TypeMismatch;

395 {
396 if constexpr (TIsDerivedFrom<std::remove_pointer_t<T>, UObject>::Value)
397 {
398 static_assert(std::is_pointer_v<T>);

400 if (InDesc.ValueType == EPropertyBagPropertyType::SoftObject || InDesc.ValueType == EPropertyBagPropertyType::SoftClass)
