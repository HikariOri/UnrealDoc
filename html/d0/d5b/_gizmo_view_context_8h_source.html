91 // WorldToScreen -> ScreenToPixel. A result of (-1, -1) indicates invalid W.

103 bool ScreenToPixel(const FVector4& ScreenPoint, FVector2D& OutPixelLocation) const
104 {
105 // implementation copied from FSceneView::ScreenToPixel()
106
107 if (ScreenPoint.W != 0.0f)
108 {
109 //Reverse the W in the case it is negative, this allow to manipulate a manipulator in the same direction when the camera is really close to the manipulator.
110 double InvW = (ScreenPoint.W > 0.0 ? 1.0 : -1.0) / ScreenPoint.W;
111 double Y = (GProjectionSignY > 0.0) ? ScreenPoint.Y : 1.0 - ScreenPoint.Y;
112 OutPixelLocation = FVector2D(
113 UnscaledViewRect.Min.X + (0.5 + ScreenPoint.X * 0.5 * InvW) * UnscaledViewRect.Width(),
114 UnscaledViewRect.Min.Y + (0.5 - Y * 0.5 * InvW) * UnscaledViewRect.Height()
115 );
116 return true;
117 }
118 else
119 {
120 return false;
121 }
122 }

124 FRay ScreenToRay(const FVector4& ScreenPosition) const
125 {
126 // Derived from FViewportCursorLocation::FViewportCursorLocation()
127
128 const FMatrix& InvViewMatrix = ViewMatrices.GetInvViewMatrix();
129 const FMatrix& InvProjectionMatrix = ViewMatrices.GetInvProjectionMatrix();
130
131 if (IsPerspectiveProjection())
132 {
133 return FRay(
134 GetViewLocation(),
135 InvViewMatrix.TransformVector(FVector(InvProjectionMatrix.TransformFVector4(FVector4(
136 ScreenPosition.X * GNearClippingPlane,
137 ScreenPosition.Y * GNearClippingPlane,
138 0.0f,
139 GNearClippingPlane
140 )))).GetSafeNormal(),
141 true
142 );
143 }
144 else
145 {
146 FVector Direction = InvViewMatrix.TransformVector(FVector4(0, 0, 1)).GetSafeNormal();
147 return FRay(
148 InvViewMatrix.TransformFVector4(InvProjectionMatrix.TransformFVector4(FVector4(
149 ScreenPosition.X,
150 ScreenPosition.Y,
151 0.5,
152 1.0
153 ))) - UE_FLOAT_HUGE_DISTANCE * Direction, // Incorporate adjustment from UEditorInteractiveToolsContext::GetRayFromMousePos()
154 Direction,
155 true
156 );
157 }
158 }

160 FVector4 PixelToScreen(const FVector2D& PixelLocation, bool bClamped=true) const
161 {
162 double X = PixelLocation.X - UnscaledViewRect.Min.X;
163 double Y = PixelLocation.Y - UnscaledViewRect.Min.Y;
164
165 if (bClamped)
166 {
167 X = FMath::Clamp(X, 0.0, UnscaledViewRect.Width());
168 Y = FMath::Clamp(Y, 0.0, UnscaledViewRect.Height());
169 }
170
171 return FVector4(
172 -1.0 + X / UnscaledViewRect.Width() * 2.0,
173 1.0 + Y / UnscaledViewRect.Height() * -2.0,
174 0.0,
175 1.0
176 );
177 }

179 FRay PixelToScreenRay(const FVector2D& PixelLocation, bool bClamped=true) const
180 {
181 FVector4 ScreenPosition = PixelToScreen(PixelLocation, bClamped);
182 return ScreenToRay(ScreenPosition);
