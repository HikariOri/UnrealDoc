239 uint32 Update(FInternalHandleArryType* OutAddedHandles = nullptr, FInternalHandleArryType* OutRemovedHandles = nullptr)
240 {
241 QUICK_SCOPE_CYCLE_COUNTER(STAT_EventLoop_TManagedStorage_Update);
242
243 Traits::CheckIsManagerThread(ManagerThreadId);
244
245 uint32 NumChanges = 0;
246
247 // Add new queued items.
248 if (OutAddedHandles)
249 {
250 FAddRequest AddRequest;
251 while (AddRequests.Dequeue(AddRequest))
252 {
253 OutAddedHandles->Add(AddImpl(AddRequest.Handle, MoveTemp(AddRequest.Data)));
254 ++NumChanges;
255 }
256 }
257 else
258 {
259 FAddRequest AddRequest;
260 while (AddRequests.Dequeue(AddRequest))
261 {
262 AddImpl(AddRequest.Handle, MoveTemp(AddRequest.Data));
263 ++NumChanges;
264 }
265 }
266
267 // Remove queued items.
268 if (OutRemovedHandles)
269 {
270 FRemoveRequest RemoveRequest;
271 while (RemoveRequests.Dequeue(RemoveRequest))
272 {
273 OutRemovedHandles->Add(RemoveImpl(RemoveRequest.Handle, RemoveRequest.OnComplete));
274 ++NumChanges;
275 }
276 }
277 else
278 {
279 FRemoveRequest RemoveRequest;
280 while (RemoveRequests.Dequeue(RemoveRequest))
281 {
282 RemoveImpl(RemoveRequest.Handle, RemoveRequest.OnComplete);
283 ++NumChanges;
284 }
285 }
286
287 return NumChanges;
288 }

485 FInternalHandle AddImpl(const FExternalHandle Handle, ElementType&& Data)
486 {
487 FSparseArrayAllocationInfo Allocation = Storage.AddUninitialized();
488 FInternalHandle InternalHandle(FInternalHandle::GenerateNewHandle, Handle, Allocation.Index);
489 new(Allocation) FStorageEntry{ InternalHandle, MoveTemp(Data)};
490 StorageHandleIndex.Add(Handle, Allocation.Index);
491 return InternalHandle;
492 }
493
494 FInternalHandle RemoveImpl(const FExternalHandle Handle, const FManagedStorageOnRemoveComplete& OnComplete)
495 {
496 int32 FoundIndex = INDEX_NONE;
497 FInternalHandle InternalHandle;
498 if (StorageHandleIndex.RemoveAndCopyValue(Handle, FoundIndex) && Storage.IsValidIndex(FoundIndex))
499 {
500 InternalHandle = Storage[FoundIndex].InternalHandle;
501 Storage.RemoveAt(FoundIndex);
502 }
503
504 if (OnComplete)
505 {
506 OnComplete();
507 }
508
509 return InternalHandle;
510 }
