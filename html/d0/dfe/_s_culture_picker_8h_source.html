33struct FCultureEntry
34{
35 FCulturePtr Culture;
36 TArray< TSharedPtr<FCultureEntry> > Children;
37 bool IsSelectable;
38 bool AutoExpand = false;
39
40 FCultureEntry(const FCulturePtr& InCulture, const bool InIsSelectable = true)
41 : Culture(InCulture)
42 , IsSelectable(InIsSelectable)
43 {}
44
45 FCultureEntry(const FCultureEntry& Source)
46 : Culture(Source.Culture)
47 , IsSelectable(Source.IsSelectable)
48 {
49 Children.Reserve(Source.Children.Num());
50 for (const auto& Child : Source.Children)
51 {
52 Children.Add( MakeShareable( new FCultureEntry(*Child) ) );
53 }
54 }
55};
