45 template <typename QueryDataType, typename LeafCallbackType>
46 EVisitResult Query(const TArray<FAabbTreeNode>& Nodes, const int32 RootIndex, const QueryDataType& QueryData, LeafCallbackType LeafCallback)
47 {
48 if (RootIndex == INDEX_NONE)
49 {
50 return EVisitResult::Continue;
51 }
52
53 // TODO: Investigate the traversal order for rays/sweeps.
54 // The previous implementation would push onto the stack so that sooner hits were traversed first,
55 // however this is more difficult when the child aabbs are not embedded in the parent.
56 // In addition, legacy only ordered the direct children and did not do a full priority queue.
57 constexpr int32 MaxNodeStackNumOnSystemStack = 255;
58 TArray<int32, TSizedInlineAllocator<MaxNodeStackNumOnSystemStack, 32> > NodeStack;
59
60 NodeStack.Push(RootIndex);
61 while (!NodeStack.IsEmpty())
62 {
63 const int32 NodeIndex = NodeStack.Pop(EAllowShrinking::No);
64 const FAabbTreeNode& Node = Nodes[NodeIndex];
65
66 if (!QueryData.Test(Node.Aabb))
67 {
68 continue;
69 }
70
71 if (AabbTreeAlgorithm::IsLeaf(Node))
72 {
73 EVisitResult VisitResult = LeafCallback(NodeIndex, Node);
74 if (VisitResult == EVisitResult::Stop)
75 {
76 return EVisitResult::Stop;
77 }
78 }
79 else
80 {
81 NodeStack.Push(Node.Left);
82 NodeStack.Push(Node.Right);
83 }
84 }
85 return EVisitResult::Continue;
86 }
