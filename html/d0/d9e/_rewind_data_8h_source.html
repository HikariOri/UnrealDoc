235 FORCEINLINE virtual bool ExtractData(const int32 ExtractFrame, const bool bResetSolver, void* HistoryData, const bool bExactFrame = false) override
236 {
237 // Early out if we are trying to extract data later than latest frame but the latest data is more than the whole buffer size old, don't extrapolate that far.
238 if (ExtractFrame - NumFrames > GetLatestFrame())
239 {
240 return false;
241 }
242
243 const int32 ExtractIndex = GetFrameIndex(ExtractFrame);
244 if (ExtractFrame == DataHistory[ExtractIndex].LocalFrame)
245 {
246 CurrentFrame = ExtractFrame;
247 CurrentIndex = ExtractIndex;
248 *static_cast<DataType*>(HistoryData) = DataHistory[CurrentIndex];
249 return true;
250 }
251 else if(!bExactFrame)
252 {
253#if DEBUG_NETWORK_PHYSICS
254 if (bResetSolver)
255 {
256 UE_LOGF(LogChaos, Warning, " Unable to extract data at frame %d while rewinding the simulation", ExtractFrame);
257 }
258#endif
259 const int32 MinFrameIndex = ClosestData(ExtractFrame, true);
260 const int32 MaxFrameIndex = ClosestData(ExtractFrame, false);
261
262 if (MinFrameIndex != INDEX_NONE
263 && MaxFrameIndex != INDEX_NONE
264 && (MinFrameIndex != MaxFrameIndex)
265 && (DataHistory[MinFrameIndex].LocalFrame != DataHistory[MaxFrameIndex].LocalFrame))
266 {
267 DataType& ExtractedData = *static_cast<DataType*>(HistoryData);
268 ExtractedData = DataHistory[MinFrameIndex];
269
270 const int32 DeltaFrame = FMath::Abs(ExtractFrame - DataHistory[MinFrameIndex].LocalFrame);
271
272 ExtractedData.LocalFrame = ExtractFrame;
273 ExtractedData.ServerFrame += DeltaFrame;
274 ExtractedData.bDataAltered = true; // This history entry is now altered and doesn't correspond to the source entry
275
276PRAGMA_DISABLE_DEPRECATION_WARNINGS
277 ExtractedData.DoInterpolateData(DataHistory[MinFrameIndex], DataHistory[MaxFrameIndex]);
278PRAGMA_ENABLE_DEPRECATION_WARNINGS
279
280#if DEBUG_NETWORK_PHYSICS
281 UE_LOGF(LogChaos, Log, " Interpolating data between frame %d and %d - > [%d]", DataHistory[MinFrameIndex].LocalFrame, DataHistory[MaxFrameIndex].LocalFrame, ExtractedData.LocalFrame);
282#endif
283 return true;
284 }
285 else if (MinFrameIndex != INDEX_NONE)
286 {
287 DataType& ExtractedData = *static_cast<DataType*>(HistoryData);
288 ExtractedData = DataHistory[MinFrameIndex];
289
290 const int32 DeltaFrame = FMath::Abs(ExtractFrame - DataHistory[MinFrameIndex].LocalFrame);
291
292 ExtractedData.LocalFrame = ExtractFrame;
293 ExtractedData.ServerFrame += DeltaFrame;
294 ExtractedData.bDataAltered = true; // This history entry is now altered and doesn't correspond to the source entry
295
296#if DEBUG_NETWORK_PHYSICS
297 UE_LOGF(LogChaos, Log, " Setting data to frame %d", DataHistory[MinFrameIndex].LocalFrame);
298#endif
299 return true;
300 }
301 else
302 {
303#if DEBUG_NETWORK_PHYSICS
304 UE_LOGF(LogChaos, Log, " Failed to find data bounds : Min = %d | Max = %d", MinFrameIndex, MaxFrameIndex);
305#endif
306 return false;
307 }
308 }
309 return false;

312 FORCEINLINE virtual void MergeData(int32 FromFrame, void* ToData) override
313 {
314 if (!ToData)
315 {
316 return;
317 }
318
319 DataType& MergeToData = *static_cast<DataType*>(ToData);
320
321 const int32 ToFrame = MergeToData.LocalFrame;
322 const int32 ToServerFrame = MergeToData.ServerFrame;
323
324 // Save target data as a stable reference for interpolation and as the final *this.
325 // Older frames are accumulated separately so that each MergeData call receives
326 // *this = newer frame and From = older adjacent frame, matching the API contract.
327 const DataType TargetData = MergeToData;
328 DataType AccumulatedOlder;
329 bool bSeeded = false;
330
331 for (; FromFrame < ToFrame; FromFrame++)
332 {
333 const int32 FromIndex = GetFrameIndex(FromFrame);
334
335 // If the frame was never recorded, synthesize it via interpolation between the nearest
336 // earlier known frame and the target. This writes directly into DataHistory[FromIndex]
337 // as a side effect, filling the gap so the merge accumulation can pick it up below.
338 if (FromFrame != DataHistory[FromIndex].LocalFrame)
339 {
340 const int32 MinFrameIndex = ClosestData(FromFrame, true);
341 if (MinFrameIndex != INDEX_NONE)
342 {
343 DataType& NewData = DataHistory[FromIndex];
344 const int32 DeltaFrame = ToFrame - FromFrame;
345 NewData.LocalFrame = FromFrame;
346 NewData.ServerFrame = ToServerFrame - DeltaFrame;
347 NewData.bDataAltered = true;
348
349PRAGMA_DISABLE_DEPRECATION_WARNINGS
350 NewData.DoInterpolateData(DataHistory[MinFrameIndex], TargetData);
351PRAGMA_ENABLE_DEPRECATION_WARNINGS
352 }
353 }
354
355 if (FromFrame == DataHistory[FromIndex].LocalFrame)
356 {
357 if (!bSeeded)
358 {
359 // Seed the accumulator with the earliest available frame
360 AccumulatedOlder = DataHistory[FromIndex];
361 bSeeded = true;
362 }
363 else
364 {
365 // DataHistory[FromIndex] is newer than AccumulatedOlder.
366 // Call with *this = newer, From = accumulated-older so the API contract holds.
367 DataType NewerFrameCopy = DataHistory[FromIndex];
368 NewerFrameCopy.MergeData(AccumulatedOlder);
369 AccumulatedOlder = NewerFrameCopy;
370 }
371 }
372 }
373
374 // MergeToData was never written in the loop (only AccumulatedOlder was), so it
375 // still holds the original target data (newest). Merge accumulated older frames in:
376 // *this = newest (target), From = accumulated older frames
377 if (bSeeded)
378 {
379 MergeToData.MergeData(AccumulatedOlder);
380 MergeToData.bDataAltered = true;
381 }
382 }

424 FORCEINLINE virtual bool RecordDataGrowingOrdered(const void* HistoryData) override
425 {
426 const DataType* HistoryDataCast = static_cast<const DataType*>(HistoryData);
427
428 int32 InsertAtIdx = 0;
429 int32 StoredFrame = 0;
430
431 if (NumFrames > 0)
432 {
433 for (int32 Idx = CurrentIndex; Idx >= 0; Idx--)
434 {
435 StoredFrame = DataHistory[Idx].LocalFrame;
436 if (StoredFrame == HistoryDataCast->LocalFrame)
437 {
438 // Data for frame already stored
439 return false;
440 }
441 else if (StoredFrame < HistoryDataCast->LocalFrame)
442 {
443 // Found index where earlier data is stored, insert newer data on the next index
444 InsertAtIdx = Idx + 1;
445 break;
446 }
447 }
448 }
449
450 DataHistory.Insert(*HistoryDataCast, InsertAtIdx);
451
452 NumFrames++;
453 CurrentIndex = NumFrames - 1;
454 LatestFrame = FMath::Max(LatestFrame, HistoryDataCast->LocalFrame);
455
456 return true;

642struct FFrameAndPhase
643{
644 enum EParticleHistoryPhase : uint8
645 {
646 //The particle state before PushData, server state update, or any sim callbacks are processed
647 //This is the results of the previous frame before any GT modifications are made in this frame
648 //This is what the server state should be compared against
649 //This is what we rewind to before a resim
650 PrePushData = 0,
651
652 //The particle state after PushData is applied, but before any server state is applied
653 PostPushData,
654
655 //The particle state after sim callbacks are applied.
656 //This is used to detect desync of particles before simulation itself is run (these desyncs can come from server state or the sim callback itself)
657 PostCallbacks,
658
659 NumPhases
660 };
661
662 int32 Frame : 30;
663 uint32 Phase : 2;
664
665 bool operator<(const FFrameAndPhase& Other) const
666 {
667 return Frame < Other.Frame || (Frame == Other.Frame && Phase < Other.Phase);
668 }
669
670 bool operator<=(const FFrameAndPhase& Other) const
671 {
672 return Frame < Other.Frame || (Frame == Other.Frame && Phase <= Other.Phase);
673 }
674
675 bool operator==(const FFrameAndPhase& Other) const
676 {
677 return Frame == Other.Frame && Phase == Other.Phase;
678 }

838 template <typename THandle>
839 bool IsInSync(const THandle& Handle, const FFrameAndPhase FrameAndPhase, const FDirtyPropertiesPool& Pool) const
840 {
841 if (const T* Val = Read(FrameAndPhase, Pool))
842 {
843 T HeadVal;

850
851 T& Insert(const FFrameAndPhase FrameAndPhase, FDirtyPropertiesPool& Manager)
852 {
853 T* Result = nullptr;
854
855 int32 FrameIndex = FindIdx(FrameAndPhase);
856 if (FrameIndex != INDEX_NONE)
857 {
858 Result = &GetPool(Manager).GetElement(Buffer[FrameIndex].Ref);
859 }
860 else
861 {
862 FPropertyIdx ElementRef;
863 if (Next >= Buffer.Num())
864 {
865 GetPool(Manager).AddElement(ElementRef);
866 Buffer.Add({ ElementRef, FrameAndPhase });
867 }
868 else
869 {
870 ElementRef = Buffer[Next].Ref;
871 }
872 Result = &GetPool(Manager).GetElement(ElementRef);
873
874 int32 PrevFrame = Next;
875 int32 NextFrame = PrevFrame;
876 for (int32 Count = 0; Count < NumValid; ++Count)
877 {
878 NextFrame = PrevFrame;
879
880 --PrevFrame;
881 if (PrevFrame < 0) { PrevFrame = Buffer.Num() - 1; }
882
883 const FPropertyInterval& PrevInterval = Buffer[PrevFrame];
884 if (PrevInterval.FrameAndPhase < FrameAndPhase)
885 {
886 Buffer[NextFrame].FrameAndPhase = FrameAndPhase;
887 Buffer[NextFrame].Ref = ElementRef;
888 break;
889 }
890 else
891 {
892 Buffer[NextFrame] = Buffer[PrevFrame];
893
894 if (Count == NumValid - 1)
895 {
896 // If we shift back and reach the end of the buffer, insert here
897 Buffer[PrevFrame].FrameAndPhase = FrameAndPhase;
898 Buffer[PrevFrame].Ref = ElementRef;
899 }
900 }
901 }
902
903 ++Next;
904 if (Next == Capacity) { Next = 0; }
905
906 NumValid = FMath::Min(++NumValid, Capacity);
907 }
908 return *Result;
909 }
910
911private:
912
913 const int32 FindIdx(const FFrameAndPhase FrameAndPhase) const
914 {
915 int32 Cur = Next; //go in reverse order because hopefully we don't rewind too far back
916 int32 Result = INDEX_NONE;
917 for (int32 Count = 0; Count < NumValid; ++Count)
918 {
919 --Cur;
920 if (Cur < 0) { Cur = Buffer.Num() - 1; }
921
922 const FPropertyInterval& Interval = Buffer[Cur];
923
924 if (Interval.FrameAndPhase < FrameAndPhase)
925 {
926 //no reason to keep searching, frame is bigger than everything before this
927 break;
928 }
929 else
930 {
931 Result = Cur;
932 }
933 }
934
935 if (bNoEntryIsHead || Result == INDEX_NONE)
936 {
937 //in this mode we consider the entire interval as one entry
938 return Result;
939 }
940 else
941 {
942 //in this mode each interval just represents the frame the property was dirtied on
943 //so in that case we have to check for equality
944 return Buffer[Result].FrameAndPhase == FrameAndPhase ? Result : INDEX_NONE;
945 }
946 }
947
948 TPropertyPool<T>& GetPool(FDirtyPropertiesPool& Manager) { return Manager.GetPool<T, PropName>(); }
949 const TPropertyPool<T>& GetPool(const FDirtyPropertiesPool& Manager) const { return Manager.GetPool<T, PropName>(); }
950
951 //Gets access into buffer in FrameAndPhase order.
952 //It's assumed FrameAndPhase is monotonically increasing: x_{n+1} > x_n
953 //If bEnsureMonotonic is true we will always return a valid access (unless assert fires)
954 //If bEnsureMonotonic is false we will ensure x_{n+1} >= x_n. If x_{n+1} == x_n we return null to inform the user (can be useful when multiple writes happen in same phase)
955 template <bool bEnsureMonotonic>
956 T* WriteAccessImp(const FFrameAndPhase FrameAndPhase, FDirtyPropertiesPool& Manager)
957 {
958 if (NumValid)
959 {
960 const int32 Prev = Next == 0 ? Buffer.Num() - 1 : Next - 1;
961 const FFrameAndPhase& LatestFrameAndPhase = Buffer[Prev].FrameAndPhase;
962 if (bEnsureMonotonic)
963 {
964 //Must write in monotonic growing order so that x_{n+1} > x_n
965 ensureMsgf(LatestFrameAndPhase < FrameAndPhase, TEXT("WriteAccessImp<bEnsureMonotonic = true> trying to write to already written FrameAndPhase: %d/%d >= %d/%d"), LatestFrameAndPhase.Frame, LatestFrameAndPhase.Phase, FrameAndPhase.Frame, FrameAndPhase.Phase);
966 }
967 else
968 {
969 //Must write in growing order so that x_{n+1} >= x_n
970 ensureMsgf(LatestFrameAndPhase <= FrameAndPhase, TEXT("WriteAccessImp<bEnsureMonotonic = false> trying to write to already written FrameAndPhase: %d/%d > %d/%d"), LatestFrameAndPhase.Frame, LatestFrameAndPhase.Phase, FrameAndPhase.Frame, FrameAndPhase.Phase);
971 if (LatestFrameAndPhase == FrameAndPhase)
972 {
973 //Already wrote once for this FrameAndPhase so skip
974 return nullptr;
975 }
976 }
977
978 ValidateOrder();
979 }
980
981 T* Result;
982
983 if (Next < Buffer.Num())
984 {
985 //reuse
986 FPropertyInterval& Interval = Buffer[Next];
987 Interval.FrameAndPhase = FrameAndPhase;
988 Result = &GetPool(Manager).GetElement(Interval.Ref);
989 }
990 else
991 {
992 //no reuse yet so can just push
993 FPropertyIdx NewIdx;
994 Result = &GetPool(Manager).AddElement(NewIdx);
995 Buffer.Add({NewIdx, FrameAndPhase });
996 }
997
998 ++Next;
999 if (Next == Capacity) { Next = 0; }
1000
1001 NumValid = FMath::Min(++NumValid, Capacity);
1002
1003 return Result;
1004 }
1005
1006 void ValidateOrder()
1007 {
1008#if VALIDATE_REWIND_DATA
1009 int32 Val = Next;
1010 FFrameAndPhase PrevVal;
1011 for (int32 Count = 0; Count < NumValid; ++Count)
1012 {
1013 --Val;
1014 if (Val < 0) { Val = Buffer.Num() - 1; }
1015 if (Count == 0)
1016 {
1017 PrevVal = Buffer[Val].FrameAndPhase;
1018 }
1019 else
1020 {
1021 ensureMsgf(Buffer[Val].FrameAndPhase < PrevVal, TEXT("ValidateOrder Idx: %d TailFrame: %d/%d, HeadFrame: %d/%d"), Val, Buffer[Val].FrameAndPhase.Frame, Buffer[Val].FrameAndPhase.Phase, PrevVal.Frame, PrevVal.Phase);
1022 PrevVal = Buffer[Val].FrameAndPhase;
1023 }
1024 }
1025#endif
1026 }
1027
1028private:
1029 int32 Next;
1030 int32 NumValid;
1031 int32 Capacity;
1032 TArray<FPropertyInterval> Buffer;
1033};
