148void FNavigationOctreeController::RemoveChild(const FNavigationElementHandle Parent, const TSharedRef<const FNavigationElement>& Child)
149{
150 // Compare by handle rather than pointer identity so that callers can pass a
151 // newly-allocated element that represents the same navigation object.
152 const FNavigationElementHandle ChildHandle = Child->GetHandle();
153 for (auto It = OctreeParentChildNodesMap.CreateKeyIterator(Parent); It; ++It)
154 {
155 if (It.Value()->GetHandle() == ChildHandle)
156 {
157 It.RemoveCurrent();
158 return;
159 }
160 }
161}
