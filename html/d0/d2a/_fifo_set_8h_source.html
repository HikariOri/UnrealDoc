70template <UE::CPointer T>
71void TFifoSet<T>::Add(TNotNull<T> Key)
72{
73 int32* StoredIndex = &KeyToIndex.FindOrAdd(Key, PopOrder.Num());
74 if (*StoredIndex < PopOrder.Num())
75 {
76 // Already exists
77 return;
78 }
79
80 if (PopOrder.Num() == PopOrder.Max() && PopOrder.Max() > 0)
81 {
82 // We are about to increase the size of PopOrder. Check whether we should instead strip the nulls out of it
83 // and rewrite the stored indices because we have removed enough elements to make that worthwhile.
84 constexpr int32 MinimumRatioToBeWorthReIndex = 2;
85 if (KeyToIndex.Num() * MinimumRatioToBeWorthReIndex < PopOrder.Max())
86 {
87 ReIndex();
88
89 // ReIndex might have modified KeyToIndex, so refetch our pointer to its internal memory.
90 StoredIndex = KeyToIndex.Find(Key);
91 check(StoredIndex != nullptr);
92
93 // Assign the new index to end of PopOrder into the StoredIndex. ReIndex did not already do this
94 // because Key was not yet present in PopOrder.
95 *StoredIndex = PopOrder.Num();
96 }
97 }
98
99 // We already set StoredIndex above, just assign Key into that index (the end of PopOrder).
100 PopOrder.Add(Key);
101}

122template <UE::CPointer T>
123T TFifoSet<T>::Pop()
124{
125 while (NextPop < PopOrder.Num())
126 {
127 if (!PopOrder[NextPop])
128 {
129 ++NextPop;
130 continue;
131 }
132 T Result = MoveTemp(PopOrder[NextPop]);
133 PopOrder[NextPop] = nullptr;
134 int32 StoredIndex = 0;
135 bool bRemoved = KeyToIndex.RemoveAndCopyValue(Result, StoredIndex);
136 check(bRemoved && StoredIndex == NextPop);
137 ++NextPop;
138 return Result;
139 }
140 return nullptr;
141}

163template <UE::CPointer T>
164void TFifoSet<T>::ReIndex()
165{
166 // Note that ReIndex is called during Add and KeyToIndex has one element in it that is not yet
167 // present in PopOrder.
168 TArray<T> NewPopOrder;
169 NewPopOrder.Reserve(KeyToIndex.Num());
170 for (const T& Key : PopOrder)
171 {
172 if (Key)
173 {
174 int32* Index = KeyToIndex.Find(Key);
175 check(Index != nullptr);
176 *Index = NewPopOrder.Num();
177 NewPopOrder.Add(Key);
178 }
179 }
180 PopOrder = MoveTemp(NewPopOrder);
181 NextPop = 0;
182}
