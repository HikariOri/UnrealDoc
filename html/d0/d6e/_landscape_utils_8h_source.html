194 class FScopedBreadcrumbEvent final
195 {
196 public:
197 template <size_t N>
198 FScopedBreadcrumbEvent(FRDGBuilderRecorder& InRecorder, TCHAR const(&StaticName)[N], FRDGEventName&& EventName)
199 : Recorder(&InRecorder)
200 , RDGEvent(MakeShared<TOptional<TRDGEventScopeGuard<FRDGScope_RHI>>>())
201 {
202 // We use a shared ptr to create a TOptional<TRDGEventScopeGuard> immediately, then capture this shared ptr in these additional operations' lambdas (so that the object
203 // continues to live until the closing tag operation)
204 Recorder->EnqueueRDGCommand(
205 [
206 RDGEvent = RDGEvent,
207 &StaticName = StaticName,
208 EventName = MoveTemp(EventName)
209 ](FRDGBuilder& GraphBuilder) mutable
210 {
211 // Allocate the TOptional<TRDGEventScopeGuard> now in order to insert the tag :
212 RDG_EVENT_SCOPE_CONSTRUCT(*RDGEvent, GraphBuilder, true, ERDGScopeFlags::None, RHI_GPU_STAT_ARGS_NONE, StaticName, MoveTemp(EventName));
213 });
214 }
215
216 ~FScopedBreadcrumbEvent()
217 {
218 Recorder->EnqueueRDGCommand([RDGEvent = RDGEvent](FRDGBuilder& GraphBuilder)
219 {
220 // Reset the TOptional in order to delete the TRDGEventScopeGuard, which will remove the tag :
221 RDGEvent->Reset();
222 });
223 }
224
225 private:
226 FRDGBuilderRecorder* Recorder = nullptr;
227 TSharedPtr<TOptional<TRDGEventScopeGuard<FRDGScope_RHI>>> RDGEvent;
228 };
229#endif // RDG_EVENTS
