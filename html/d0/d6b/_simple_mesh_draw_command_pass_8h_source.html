68template <typename PassParametersType, typename AddMeshBatchesCallbackLambdaType, typename PassPrologueLambdaType>
69void AddSimpleMeshPass(FRDGBuilder& GraphBuilder, PassParametersType* PassParametersIn, const FScene* Scene, const FSceneView &View, FInstanceCullingManager* InstanceCullingManager, FRDGEventName&& PassName,
70 const ERDGPassFlags& PassFlags,
71 AddMeshBatchesCallbackLambdaType AddMeshBatchesCallback,
72 PassPrologueLambdaType PassPrologueCallback,
73 bool bAllowIndirectArgsOverride=true)
74{
75 TRACE_CPUPROFILER_EVENT_SCOPE(AddSimpleMeshPass);
76
77 // TODO: don't do this when the parameters are exclusive (optimization)
78 PassParametersType* PassParameters = GraphBuilder.AllocParameters<PassParametersType>();
79
80 // Copy parameters to ensure we can overwrite the InstanceCullingDrawParams
81 *PassParameters = *PassParametersIn;
82
83 FSimpleMeshDrawCommandPass* SimpleMeshDrawCommandPass = GraphBuilder.AllocObject<FSimpleMeshDrawCommandPass>(View, InstanceCullingManager);
84
85 AddMeshBatchesCallback(SimpleMeshDrawCommandPass->GetDynamicPassMeshDrawListContext());
86
87 // It is legal to render a simple mesh pass without a FScene, but then mesh draw commands must not reference primitive ID streams.
88 if (Scene != nullptr)
89 {
90 SimpleMeshDrawCommandPass->BuildRenderingCommands(GraphBuilder, View, *Scene, PassParameters->InstanceCullingDrawParams);
91 }
92
93 if (!bAllowIndirectArgsOverride)
94 {
95 PassParameters->InstanceCullingDrawParams.DrawIndirectArgsBuffer = nullptr;
96 }
97
98 GraphBuilder.AddPass(
99 MoveTemp(PassName),
100 PassParameters,
101 PassFlags,
102 [SimpleMeshDrawCommandPass, PassParameters, PassPrologueCallback](FRDGAsyncTask, FRHICommandList& RHICmdList)
103 {
104 PassPrologueCallback(RHICmdList);
105
106 SimpleMeshDrawCommandPass->SubmitDraw(RHICmdList, PassParameters->InstanceCullingDrawParams);
107 }
108 );
109}
