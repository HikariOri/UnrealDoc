46template <typename T>
47class TScenePrimitiveArray
48{
49 static const int32 NumElementsPerChunk = 1024;
50public:
51 TScenePrimitiveArray() = default;
52
53 T& Add(const T& Element)
54 {
55 return *(new (&AddUninitialized()) T(Element));
56 }
57
58 T& AddUninitialized()
59 {
60 if (NumElements % NumElementsPerChunk == 0)
61 {
62 ChunkType* Chunk = new ChunkType;
63 Chunk->Reserve(NumElementsPerChunk);
64 Chunks.Emplace(Chunk);
65 }
66
67 NumElements++;
68 ChunkType& Chunk = *Chunks.Last();
69 Chunk.AddUninitialized();
70 return Chunk.Last();
71 }
72
73 void Remove(uint32 Count, EAllowShrinking AllowShrinking)
74 {
75 check(Count <= NumElements);
76 const uint32 NumElementsNew = NumElements - Count;
77 while (NumElements != NumElementsNew)
78 {
79 --NumElements;
80
81 ChunkType& Chunk = *Chunks.Last();
82 Chunk.Pop(EAllowShrinking::No);
83
84 if (Chunk.IsEmpty())
85 {
86 Chunks.Pop(EAllowShrinking::No);
87 }
88 }
89 }
90
91 void Reserve(int32 Count)
92 {
93 Chunks.Reserve(NumChunks(Count));
94 }
95
96 T& Get(int32 ElementIndex)
97 {
98 const uint32 ChunkIndex = ElementIndex / NumElementsPerChunk;
99 const uint32 ChunkElementIndex = ElementIndex % NumElementsPerChunk;
100 return (*Chunks[ChunkIndex])[ChunkElementIndex];
101 }
102
103 const T& Get(int32 ElementIndex) const
104 {
105 const uint32 ChunkIndex = ElementIndex / NumElementsPerChunk;
106 const uint32 ChunkElementIndex = ElementIndex % NumElementsPerChunk;
107 return (*Chunks[ChunkIndex])[ChunkElementIndex];
108 }
109
110 FORCEINLINE T& operator[] (int32 Index) { return Get(Index); }
111 FORCEINLINE const T& operator[] (int32 Index) const { return Get(Index); }
112
113 bool IsValidIndex(int32 Index) const { return static_cast<uint32>(Index) < NumElements; }
114
115 int32 Num() const { return NumElements; }
116
117private:
118 static constexpr uint32 NumChunks(uint32 NumElements)
119 {
120 return (NumElements + NumElementsPerChunk - 1u) / NumElementsPerChunk;
121 }
122
123 using ChunkType = TArray<T>;
124 TArray<TUniquePtr<ChunkType>> Chunks;
125 uint32 NumElements = 0;
126};
