111 (PointU0V0 - PointU1V0) * (U0MinusU / U1MinusU0) +
112 (PointU0V0 - PointU0V1) * (V0MinusV / V1MinusV0) +
113 (PointU0V0 - PointU0V1 + PointU1V1 - PointU1V0) * ((U0MinusU * V0MinusV) / (U1MinusU0 * V1MinusV0));
