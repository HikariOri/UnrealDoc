97template <bool TIsSpecialization, typename TType, int32 TDimensionSize, int32 TFirstValue=0>
98struct TShaderPermutationInt
99{
101 using Type = TType;
102
104 static constexpr int32 PermutationCount = TDimensionSize;
105
109 static constexpr bool IsMultiDimensional = false;
110
112 static constexpr bool IsSpecialization = TIsSpecialization;
113
115 static constexpr Type MinValue = static_cast<Type>(TFirstValue);
116 static constexpr Type MaxValue = static_cast<Type>(TFirstValue + TDimensionSize - 1);
117
118
120 static int32 ToDimensionValueId(Type E)
121 {
122 const int32 PermutationId = static_cast<int32>(E) - TFirstValue;
123 checkf(PermutationId < PermutationCount && PermutationId >= 0, TEXT("Unknown shader permutation dimension value id %i."), PermutationId);
124 return PermutationId;
125 }
126
128 static int32 ToDefineValue(Type E)
129 {
130 return ToDimensionValueId(E) + TFirstValue;
131 }
132
134 static Type FromDimensionValueId(int32 PermutationId)
135 {
136 checkf(PermutationId < PermutationCount && PermutationId >= 0, TEXT("Invalid shader permutation dimension value id %i."), PermutationId);
137 return static_cast<Type>(PermutationId + TFirstValue);
138 }
139
141 static void GetPossibleValues(TArray<int32>& OutValues)
142 {
143 checkf(TIsSpecialization, TEXT("Getting possible values should only be called for specializations!"));
144 for (int32 Value = MinValue; Value <= MaxValue; ++Value)
145 {
146 OutValues.Add(Value);
147 }
148 }
149};

370 template<typename TPermutationVector, typename TDimension>
371 static void ModifyCompilationEnvironment(const TPermutationVector& PermutationVector, FShaderCompilerEnvironment& OutEnvironment)
372 {
373 if constexpr (bLogPermutations)
374 {
375 UE_LOGF(LogShaders, Verbose, " %ls = %d", TDimension::DefineName, TDimension::ToDefineValue(PermutationVector.DimensionValue));
376 }
377
378 OutEnvironment.SetDefine(TDimension::DefineName, TDimension::ToDefineValue(PermutationVector.DimensionValue));
379
380 if constexpr (TDimension::IsSpecialization)
381 {
382 const int32 SpecializationConstantIndex = OutEnvironment.SpecializationConstantValues.Num();
383 OutEnvironment.SetDefine(TDimension::SpecializationIndexName, SpecializationConstantIndex);
384
385 TArray<int32>& Values = OutEnvironment.SpecializationConstantValues.AddDefaulted_GetRef();
386 Values.Reserve(TDimension::PermutationCount);
387 TDimension::GetPossibleValues(Values);
388 }
389
390 return PermutationVector.Tail.ModifyCompilationEnvironment(OutEnvironment);
391 }

403 }
