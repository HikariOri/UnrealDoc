19 // Stable counting sort: groups Elements into contiguous blocks by partition index, preserving the original relative order of elements within each block.
20 // PrefixSum supplies the start index of each partition index block after sorting and used as the starting write position for the sort.
21 // T may be any copyable type (pointer, handle, value).
22 template <typename T>
23 void SortArrayByPartitionIndex(
24 TArray<T>& Elements,
25 const TArray<int32>& PartitionIndices,
26 const TArray<int32>& PrefixSum)
27 {
28 const int32 NumOfElements = Elements.Num();
29 check(PartitionIndices.Num() == NumOfElements);
30
31 // Copy start positions to use as per-bucket write cursors.
32 TArray<int32> WriteIdx = PrefixSum;
33
34 TArray<T> Sorted;
35 Sorted.SetNumUninitialized(NumOfElements);
36 for (int32 Index = 0; Index < NumOfElements; ++Index)
37 {
38 Sorted[WriteIdx[PartitionIndices[Index]]++] = Elements[Index];
39 }
40
41 Elements = MoveTemp(Sorted);
42 }
