128 FSinOsc2DRotation(float InStartingPhaseRadians = 0.f)

131 LastPhase = InStartingPhaseRadians;
132 QuadDxVec = VectorZeroFloat();
133 QuadDyVec = VectorZeroFloat();

145 void GenerateBuffer(float SampleRate, float ClampedFrequency, float* Buffer, int32 BufferSampleCount)
146 {
147 // Regenerate our vector rotation components if our changes.
148 const float PhasePerSample = (ClampedFrequency * (2 * UE_PI)) / (SampleRate);
149 if (LastPhasePerSample != PhasePerSample)
150 {
151 float QuadDx = FMath::Cos(PhasePerSample * 4);
152 float QuadDy = FMath::Sin(PhasePerSample * 4);
153
154 QuadDxVec = VectorLoadFloat1(&QuadDx);
155 QuadDyVec = VectorLoadFloat1(&QuadDy);
156
157 LastPhasePerSample = PhasePerSample;
158 }
159
160 float* Write = Buffer;
161
162 // The rotation matrix drifts, so we resync every so often
163 while (BufferSampleCount)
164 {
165 // The concept here is that cos/sin are points on a unit circle,
166 // so an oscilator is just a rotation of that point.
167 //
168 // To avoid drift, we resync off of an accurate sin/cos every evaluation
169 // then do a 2d rotation in the actual loop.
170 //
171 // we store 4 points at once to use SIMD, so each rotation is actually
172 // 4x the phase delta.
173 //
174 alignas(16) float PhaseSource[4];
175 PhaseSource[0] = LastPhase + 0 * PhasePerSample;
176 PhaseSource[1] = LastPhase + 1 * PhasePerSample;
177 PhaseSource[2] = LastPhase + 2 * PhasePerSample;
178 PhaseSource[3] = LastPhase + 3 * PhasePerSample;
179
180 VectorRegister4Float PhaseVec = VectorLoad(PhaseSource);
181 VectorRegister4Float XVector, YVector;
182
183 // We need an accurate representation of the delta
184 // vectors since we are integrating it
185 VectorSinCos(&YVector, &XVector, &PhaseVec);
186
187 // Copy to local (the compiler actually didn't do this!)
188 VectorRegister4Float LocalDxVec = QuadDxVec;
189 VectorRegister4Float LocalDyVec = QuadDyVec;
190
191 int32 BlockSampleCount = BufferSampleCount;
192 if (BlockSampleCount > 480)
193 BlockSampleCount = 480;
194
195 // Unrolling this didn't seem to really help perf wise.
196 int32 Block4 = BlockSampleCount >> 2;
197 while (Block4)
198 {
199 VectorStore(YVector, Write);
200
201 // 2D rotation matrix.
202 VectorRegister4Float NewX = VectorSubtract(VectorMultiply(LocalDxVec, XVector), VectorMultiply(LocalDyVec, YVector));
203 VectorRegister4Float NewY = VectorAdd(VectorMultiply(LocalDyVec, XVector), VectorMultiply(LocalDxVec, YVector));
204
205 XVector = NewX;
206 YVector = NewY;
207
208 Write += 4;
209 Block4--;
210 }
211
212 constexpr int32 SIMD_MASK = 0x00000003;
213 if (BlockSampleCount & SIMD_MASK)
214 {
215 // We've actually already calculated the next quad - it's in YVector
216 alignas(16) float YFloats[4];
217 VectorStore(YVector, YFloats);
218
219 int32 Remn = BlockSampleCount & SIMD_MASK;
220 for (int32 i = 0; i < Remn; i++)
221 {
222 Write[i] = YFloats[i];
223 }
224 }
225
226 // Advance phase, range reduce, and store.
227 float PhaseInRadians = LastPhase + (float)BlockSampleCount * PhasePerSample;
228 PhaseInRadians -= FMath::FloorToFloat(PhaseInRadians / (2 * UE_PI)) * (2 * UE_PI);
229 LastPhase = PhaseInRadians;
230
231 BufferSampleCount -= BlockSampleCount;
232 } // end while BufferSampleCount
233 } // end GenerateBuffer
234
235 private:
236 VectorRegister4Float QuadDxVec, QuadDyVec;
237 float LastPhasePerSample;
238 float LastPhase;
239 }; // end FSinOsc2DRotation

241 // Fast tanh based on pade approximation
242 inline float FastTanh(float X)
243 {
244 if (X < -3) return -1.0f;
245 if (X > 3) return 1.0f;
246 const float InputSquared = X*X;
247 return X*(27.0f + InputSquared) / (27.0f + 9.0f * InputSquared);
248 }

252 {

254 const float Den = (X + 0.5f * UE_PI) * (1.0f - FMath::Abs(X + 0.5f * UE_PI) / UE_PI);
255 return Num / Den;

270 // Converts in place a buffer into Unipolar (0..1)
271 // This is a Vector op version of the GetUnipolar function. (0.5f * X + 0.5f).
272 inline void ConvertBipolarBufferToUnipolar(float* InAlignedBuffer, int32 NumSamples)
273 {
274 // Make sure buffers are aligned and we can do a whole number of loops.
275 check(NumSamples % 4 == 0);

277 const VectorRegister4Float Half = VectorSetFloat1(0.5f);
278
279 // Process buffer 1 vector (4 floats) at a time.

289 inline bool IsAudioBufferSilent(const float* AudioBuffer, const int32 NumSamples)
290 {
291 bool bIsSilent = true;
292
293 int32 Index = 0;
294
295#if PLATFORM_ENABLE_VECTORINTRINSICS
296 const int32 SimdNum = NumSamples & 0xFFFFFFF0;
297 for (; Index < SimdNum; Index += 16)
298 {
299 const VectorRegister4x4Float Samples = VectorLoad16(&AudioBuffer[Index]);
300
301 if (VectorAnyGreaterThan(VectorAbs(Samples.val[0]), GlobalVectorConstants::SmallNumber) ||
302 VectorAnyGreaterThan(VectorAbs(Samples.val[1]), GlobalVectorConstants::SmallNumber) ||
303 VectorAnyGreaterThan(VectorAbs(Samples.val[2]), GlobalVectorConstants::SmallNumber) ||
304 VectorAnyGreaterThan(VectorAbs(Samples.val[3]), GlobalVectorConstants::SmallNumber))
305 {
306 bIsSilent = false;
307 Index = INT_MAX;
308 break;
309 }
310 }
311#endif
312
313 // Finish to the end of the buffer or check each sample if vector intrinsics are disabled
314 for (; Index < NumSamples; ++Index)
315 {
316 // As soon as we hit a non-silent sample, we're not silent
317 if (FMath::Abs(AudioBuffer[Index]) > UE_SMALL_NUMBER)
318 {
319 bIsSilent = false;
320 break;
321 }
322 }
323
324 return bIsSilent;

491 // Given three values, determine peak location and value of quadratic fitted to the data.
492 //
493 // @param InValues - An array of 3 values with the maximum value located in InValues[1].
494 // @param OutPeakLoc - The peak location relative to InValues[1].
495 // @param OutPeakValue - The value of the peak at the peak location.
496 //
497 // @returns True if a peak was found, false if the values do not represent a peak.
498 inline bool QuadraticPeakInterpolation(const float InValues[3], float& OutPeakLoc, float& OutPeakValue)
499 {
500 float Denom = InValues[0] - 2.f * InValues[1] + InValues[2];
501
502 if (Denom >= 0.f)
503 {
504 // This is not a peak.
505 return false;
506 }
507
508 float Tmp = InValues[0] - InValues[2];
509
510 OutPeakLoc = 0.5f * Tmp / Denom;
511
512 if ((OutPeakLoc > 0.5f) || (OutPeakLoc < -0.5f))
513 {
514 // This is not a peak.
515 return false;
516 }
517
518 OutPeakValue = InValues[1] - 0.25f * Tmp * OutPeakLoc;
519
520 return true;
521 }

523 // Polynomial interpolation using lagrange polynomials.
524 // https://en.wikipedia.org/wiki/Lagrange_polynomial
525 inline float LagrangianInterpolation(const TArray<FVector2D> Points, const float Alpha)
526 {
527 float Lagrangian = 1.0f;
528 float Output = 0.0f;
529
530 const int32 NumPoints = Points.Num();
531 for (int32 i = 0; i < NumPoints; ++i)
532 {
533 Lagrangian = 1.0f;
534 for (int32 j = 0; j < NumPoints; ++j)
535 {
536 if (i != j)
537 {
538 float Denom = UE_REAL_TO_FLOAT(Points[i].X - Points[j].X);
539 if (FMath::Abs(Denom) < UE_SMALL_NUMBER)
540 {
541 Denom = UE_SMALL_NUMBER;
542 }
543 Lagrangian *= (Alpha - UE_REAL_TO_FLOAT(Points[j].X)) / Denom;
544 }
545 }
546 Output += Lagrangian * UE_REAL_TO_FLOAT(Points[i].Y);
547 }
548 return Output;
549 }

552 class FExponentialEase
553 {
554 public:
555 FExponentialEase(float InInitValue = 0.0f, float InEaseFactor = 0.001f, float InThreshold = UE_KINDA_SMALL_NUMBER)
556 : CurrentValue(InInitValue)
557 , Threshold(InThreshold)
558 , TargetValue(InInitValue)
559 , EaseFactor(InEaseFactor)
560 , OneMinusEase(1.0f - InEaseFactor)
561 , EaseTimesTarget(EaseFactor * InInitValue)
562 {
563 }
564
565 void Init(float InInitValue, float InEaseFactor = 0.001f)
566 {
567 CurrentValue = InInitValue;
568 TargetValue = InInitValue;
569 EaseFactor = InEaseFactor;
570
571 OneMinusEase = 1.0f - EaseFactor;
572 EaseTimesTarget = TargetValue * EaseFactor;
573 }
574
575 bool IsDone() const
576 {
577 return FMath::Abs(TargetValue - CurrentValue) < Threshold;
578 }
579
580 float GetNextValue()
581 {
582 if (IsDone())
583 {
584 return CurrentValue;
585 }
586
587 // Micro-optimization,
588 // But since GetNextValue(NumTicksToJumpAhead) does this work in a tight loop (non-vectorizable), might as well
589 /*
590 return CurrentValue = CurrentValue + (TargetValue - CurrentValue) * EaseFactor;
591 = CurrentValue + EaseFactor*TargetValue - EaseFactor*CurrentValue
592 = (CurrentValue - EaseFactor*CurrentValue) + EaseFactor*TargetValue
593 = (1 - EaseFactor)*CurrentValue + EaseFactor*TargetValue
594 */
595 return CurrentValue = OneMinusEase * CurrentValue + EaseTimesTarget;
596 }

599 // (before getting the value)
600 float GetNextValue(uint32 NumTicksToJumpAhead)
601 {
602 while (NumTicksToJumpAhead && !IsDone())
603 {
604 CurrentValue = OneMinusEase * CurrentValue + EaseTimesTarget;
605 --NumTicksToJumpAhead;
606 }
607
608 return CurrentValue;
609 }

611 float PeekCurrentValue() const
612 {
613 return CurrentValue;
614 }
615
616 void SetEaseFactor(const float InEaseFactor)
617 {
618 EaseFactor = InEaseFactor;
619 OneMinusEase = 1.0f - EaseFactor;
620 }
621
622 void operator=(const float& InValue)
623 {
624 SetValue(InValue);
625 }
626
627 void SetValue(const float InValue, const bool bIsInit = false)
628 {
629 TargetValue = InValue;
630 EaseTimesTarget = EaseFactor * TargetValue;
631 if (bIsInit)
632 {
633 CurrentValue = TargetValue;
634 }
635 }

639 static float GetFactorForTau(float InTau, float InSampleRate)
