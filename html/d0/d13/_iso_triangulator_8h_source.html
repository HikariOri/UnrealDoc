305inline double CotangentCriteria(const FVector& APoint, const FVector& BPoint, const FVector& PPoint, FVector& OutNormal)
306{
307 const double BigValue = HUGE_VALUE;
308
309 FVector PA = APoint - PPoint;
310 FVector PB = BPoint - PPoint;
311
312 // the ratio between the scalar product PA.PB (=|PA| |PB| cos (A,P,B) )
313 // with the norm of the cross product |PA^PB| (=|PA| |PB| |sin(A,P,B)|)
314 // is compute.
315 double ScalareProduct = PA | PB;
316 OutNormal = PA ^ PB;
317 double NormOFScalarProduct = sqrt(OutNormal | OutNormal);
318
319 // PPoint is aligned with (A,B)
320 if (NormOFScalarProduct < DOUBLE_SMALL_NUMBER)
321 {
322 return BigValue;
323 }
324
325 // return Cotangent value
326 return ScalareProduct / NormOFScalarProduct;
327}

329inline double CotangentCriteria(const FVector2d& APoint, const FVector2d& BPoint, const FVector2d& PPoint)
330{
331 const double BigValue = HUGE_VALUE;
332
333 FVector2d PA = APoint - PPoint;
334 FVector2d PB = BPoint - PPoint;
335
336 // the ratio between the scalar product PA.PB (=|PA| |PB| cos (A,P,B) )
337 // with the norm of the cross product |PA^PB| (=|PA| |PB| |sin(A,P,B)|)
338 // is compute.
339 double ScalareProduct = PA | PB;
340 double OutNormal = PA ^ PB;
341 double NormOFPointProduct = FMath::Abs(OutNormal);
342
343 if (NormOFPointProduct < DOUBLE_SMALL_NUMBER)
344 {
345 // PPoint is aligned with (A,B)
346 return BigValue;
347 }
348
349 // return Cotangent value
350 return ScalareProduct / NormOFPointProduct;
351}

359inline FPairOfDouble IsoscelesCriteria(const FVector2d& APoint, const FVector2d& BPoint, const FVector2d& CPoint)
360{
361 const double SlopeAB = ComputeSlope(APoint, BPoint);
362 const double SlopeAC = ComputeSlope(APoint, CPoint);
363 const double SlopeBC = ComputeSlope(BPoint, CPoint);
364 const double SlopeBA = SwapSlopeOrientation(SlopeAB);
365
366 return { TransformIntoOrientedSlope(SlopeAC - SlopeAB), TransformIntoOrientedSlope(SlopeBA - SlopeBC) };
367}

381inline double EquilateralSlopeCriteria(const FVector2d& APoint, const FVector2d& BPoint, const FVector2d& CPoint)
382{
383 const double SlopeAB = ComputeSlope(APoint, BPoint);
384 const double SlopeAC = ComputeSlope(APoint, CPoint);
385 const double SlopeBC = ComputeSlope(BPoint, CPoint);
386 const double SlopeBA = SwapSlopeOrientation(SlopeAB);
387 const double SlopeCA = SwapSlopeOrientation(SlopeAC);
388 const double SlopeCB = SwapSlopeOrientation(SlopeBC);
389
390 const double A = TransformIntoOrientedSlope(SlopeAC - SlopeAB);
391 const double B = TransformIntoOrientedSlope(SlopeBA - SlopeBC);
392 const double C = TransformIntoOrientedSlope(SlopeCB - SlopeCA);
393
394 return FMath::Max3(A, B, C);
395}

397template<class PointType>
398inline double EquilateralCriteria(const PointType& SegmentA, const PointType& SegmentB, const PointType& Point)
399{
400 double Criteria1 = FMath::Abs(CoordinateOfProjectedPointOnSegment<PointType>(SegmentA, SegmentB, Point, false) - 0.5);
401 double Criteria2 = FMath::Abs(CoordinateOfProjectedPointOnSegment<PointType>(Point, SegmentA, SegmentB, false) - 0.5);
402 double Criteria3 = FMath::Abs(CoordinateOfProjectedPointOnSegment<PointType>(SegmentB, Point, SegmentA, false) - 0.5);
403 return Criteria1 + Criteria2 + Criteria3;
404}
