118 // In seconds, the longest any subtitle in the Subtitles array will last, including any initial offset starts.
119 float GetMaximumDuration() const
120 {
121 float Duration = 0.f;
122 for (const FSubtitleAssetData& Entry : Subtitles)
123 {
124 const float EntryDuration = Entry.Duration + Entry.StartOffset;
125 Duration = FMath::Max(Duration, EntryDuration);

127
128 return Duration;
129 };
