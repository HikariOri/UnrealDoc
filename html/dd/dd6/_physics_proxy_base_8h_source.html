409// Render Interpolation that both perform the linear error correction from FProxyInterpolationError and takes incoming velocity into account to make a smoother and more organic correction of the error.
410struct FProxyInterpolationErrorVelocity : FProxyInterpolationError
411{
412 using Super = FProxyInterpolationError;
413
414 FProxyInterpolationErrorVelocity(const int32 PullDataInterpIdx = INDEX_NONE, const int32 InterpChannel = 0) : Super(PullDataInterpIdx, InterpChannel)
415 {}
416 virtual ~FProxyInterpolationErrorVelocity() = default;
417
418 static const EProxyInterpolationType InterpolationType = EProxyInterpolationType::ErrorVelocity;
419 virtual const EProxyInterpolationType GetInterpolationType() const override { return FProxyInterpolationErrorVelocity::InterpolationType; }
420
421 virtual const bool IsErrorVelocitySmoothing() const override { return ErrorVelocitySmoothingCount > 0; }
422 virtual const Chaos::FRealSingle GetErrorVelocitySmoothingAlpha(const int32 ErrorVelocitySmoothDuration) const override { return Chaos::FRealSingle(ErrorVelocitySmoothingCount) / Chaos::FRealSingle(ErrorVelocitySmoothDuration); }
423 virtual const Chaos::FVec3 GetErrorVelocitySmoothingX(const Chaos::FRealSingle Alpha) const override { return FMath::Lerp(ErrorVelocitySmoothingXPrev, ErrorVelocitySmoothingX, Alpha); }
424
425 virtual const bool UpdateError(const int32 CurrentSimTick, const Chaos::FReal AsyncFixedTimeStep) override
426 {
427 if (Super::UpdateError(CurrentSimTick, AsyncFixedTimeStep))
428 {
429 StepErrorVelocitySmoothingData(AsyncFixedTimeStep);
430 return true;
431 }
432 return false;
433 }
434
435 virtual void SetVelocitySmoothing(const Chaos::FVec3 CurrV, const Chaos::FVec3 CurrX, const int32 ErrorVelocitySmoothDuration) override
436 {
437 // Cache pre error velocity and position to be used when smoothing out error correction
438 ErrorVelocitySmoothingV = CurrV;
439 ErrorVelocitySmoothingX = CurrX;
440 ErrorVelocitySmoothingXPrev = ErrorVelocitySmoothingX;
441 ErrorVelocitySmoothingCount = ErrorVelocitySmoothDuration;
442 }
443
444protected:
445 virtual void StepErrorVelocitySmoothingData(const Chaos::FReal AsyncFixedTimeStep)
446 {
447 // Step the error velocity smoothing position forward along the previous velocity to have a new position to base smoothing on each tick
448 if (IsErrorVelocitySmoothing())
449 {
450 const Chaos::FReal Time = AsyncFixedTimeStep * SimTicks;
451 ErrorVelocitySmoothingXPrev = ErrorVelocitySmoothingX;
452 ErrorVelocitySmoothingX += ErrorVelocitySmoothingV * Time;
453
454 ErrorVelocitySmoothingCount = FMath::Max(ErrorVelocitySmoothingCount - SimTicks, 0);
455 }
456 }
457
458 Chaos::FVec3 ErrorVelocitySmoothingV = Chaos::FVec3::Zero();
459 Chaos::FVec3 ErrorVelocitySmoothingX = Chaos::FVec3::Zero();
460 Chaos::FVec3 ErrorVelocitySmoothingXPrev = Chaos::FVec3::Zero();
461 int32 ErrorVelocitySmoothingCount = 0;
462};
