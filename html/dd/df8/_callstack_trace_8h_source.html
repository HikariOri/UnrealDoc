135 inline uint32 Compress(TConstArrayView<uint64> Addresses, TArrayView<uint8> Compressed)
136 {
137 check(Addresses.Num() * 8 <= Compressed.Num());
138 uint32 CompressedBytes = 0;
139 uint64 LastAddress = 0;
140 for(uint8 Index = 0; Index < Addresses.Num(); ++Index)
141 {
142 const uint64 Address = Addresses[Index];
143 uint64 XorAddress = LastAddress ^ Address;
144 LastAddress = Address;
145 const uint8 LeadingZeroes = uint8(FMath::CountLeadingZeros64(XorAddress));
146 Compressed[CompressedBytes++] = LeadingZeroes;
147 if (LeadingZeroes < 64)
148 {
149 int8 ValueBits = 64 - LeadingZeroes;
150 do
151 {
152 Compressed[CompressedBytes++] = uint8(0xff & XorAddress);
153 XorAddress >>= 8;
154 ValueBits -= 8;
155 } while(ValueBits > 0);
156 }
157 }
158 return CompressedBytes;
159 }

161 inline uint32 Uncompress(TConstArrayView<uint8> Compressed, TArrayView<uint64> Addresses, uint32& OutTotalAddressCount)
162 {
163 const int32 Bytes = Compressed.Num();
164 int32 ByteIndex = 0;
165 uint64 LastAddress = 0;
166 while (ByteIndex < Bytes)
167 {
168 uint64 OutAddress = 0;
169 const uint8 LeadingZeroes = Compressed[ByteIndex++];
170 if (LeadingZeroes < 64)
171 {
172 int8 ValueBits = 64 - LeadingZeroes;
173 uint8 BitsShift = 0;
174 check(ValueBits <= (Bytes - ByteIndex)*8);
175 while (ValueBits > 0)
176 {
177 OutAddress = (uint64(Compressed[ByteIndex++]) << BitsShift) | OutAddress;
178 ValueBits -= 8;
179 BitsShift += 8;
180 }
181 }
182 OutAddress = LastAddress ^ OutAddress;
183 LastAddress = OutAddress;
184
185 if (OutTotalAddressCount < (uint32)Addresses.Num())
186 {
187 Addresses[OutTotalAddressCount] = OutAddress;
188 }
189 OutTotalAddressCount++;
190 }
191 return FMath::Min(OutTotalAddressCount, (uint32)Addresses.Num());
192 }
193} // namespace FCallstackXORAndRLE

201 inline uint32 Compress(TConstArrayView<uint64> Addresses, TArrayView<uint8> Compressed)
202 {
203 check(Addresses.Num() * 10 <= Compressed.Num());
204 uint32 CompressedBytes = 0;
205 uint64 LastAddress = 0;
206 for(uint8 Index = 0; Index < Addresses.Num(); ++Index)
207 {
208 const uint64 Address = Addresses[Index];
209 const int64 Delta = int64(Address - LastAddress);
210 LastAddress = Address;
211
212 uint64 EncodedAddress = (uint64(Delta) << 1) ^ uint64(Delta >> 63);
213 do
214 {
215 uint8 Byte = EncodedAddress & 0x7F;
216 EncodedAddress >>= 7;
217 Compressed[CompressedBytes++] = Byte | uint8((EncodedAddress > 0) << 7);
218 }
219 while (EncodedAddress > 0);
220 }
221 return CompressedBytes;
222 }

224 inline uint32 Uncompress(TConstArrayView<uint8> Compressed, TArrayView<uint64> Addresses, uint32& OutTotalAddressCount)
225 {
226 const int32 Bytes = Compressed.Num();
227 int32 ByteIndex = 0;
228 uint64 LastAddress = 0;
229 while (ByteIndex < Bytes)
230 {
231 uint64 EncodedAddress = 0;
232 int Shift = 0;
233 uint8 Byte = 0;
234 do
235 {
236 Byte = Compressed[ByteIndex++];
237 EncodedAddress |= uint64(Byte & 0x7F) << Shift;
238 Shift += 7;
239 } while (Byte & 0x80);
240
241 int64 Delta = (EncodedAddress >> 1) ^ -(int64)(EncodedAddress & 1);
242 uint64 Address = LastAddress + Delta;
243 LastAddress = Address;
244
245 if (OutTotalAddressCount < (uint32)Addresses.Num())
246 {
247 Addresses[OutTotalAddressCount] = Address;
248 }
249 OutTotalAddressCount++;
250 }
251 return FMath::Min(OutTotalAddressCount, (uint32)Addresses.Num());
252 }

261 inline uint32 Compress(TConstArrayView<uint64> Addresses, TArrayView<uint8> Compressed)
262 {
263 check(Addresses.Num() * 9 <= Compressed.Num());
264 uint8* CompressedPtr = Compressed.GetData();
265 uint64 LastAddress = 0;
266 for(uint8 Index = 0; Index < Addresses.Num(); ++Index)
267 {
268 const uint64 Address = Addresses[Index];
269 int64 Delta = int64(Address - LastAddress);
270 LastAddress = Address;
271
272 uint32 BytesCompressed = WriteVarInt(Delta, CompressedPtr);
273 CompressedPtr += BytesCompressed;
274 }
275 return uint32(CompressedPtr - Compressed.GetData());
276 }
277
278 inline uint32 Uncompress(TConstArrayView<uint8> Compressed, TArrayView<uint64> Addresses, uint32& OutTotalAddressCount)
279 {
280 const uint8* CompressedPtr = Compressed.GetData();
281 const uint8* EndPtr = CompressedPtr + Compressed.Num();
282 uint64 Address = 0;
283 while (CompressedPtr < EndPtr)
284 {
285 uint32 BytesFound = 0;
286 int64 Delta = ReadVarInt(CompressedPtr, BytesFound);
287 CompressedPtr += BytesFound;
288 Address += Delta;
289
290 if (OutTotalAddressCount < (uint32)Addresses.Num())
291 {
292 Addresses[OutTotalAddressCount] = Address;
293 }
294 OutTotalAddressCount++;
295 }
296 return FMath::Min(OutTotalAddressCount, (uint32)Addresses.Num());
297 }
