142
144 bool CanPerformQuickCopy() const
145 {
146 // If there are selected actors and BSP surfaces AND all selected actors
147 // and surfaces are in the same level, we can do a quick copy.
148 bool bCanPerformQuickCopy = false;
149 if( LevelAllActorsAreIn && LevelWithSelectedSurface )
150 {
151 bCanPerformQuickCopy = ( LevelWithSelectedSurface == LevelAllActorsAreIn );
152 }
153 // Else, if either we have only selected actors all in one level OR we have
154 // only selected surfaces all in one level, then we can perform a quick copy.
155 else
156 {
157 bCanPerformQuickCopy = (LevelWithSelectedSurface != NULL && !bHasSelectedActors) || (LevelAllActorsAreIn != NULL && !bHasSelectedSurfaces);
158 }
159 return bCanPerformQuickCopy;
160 }
