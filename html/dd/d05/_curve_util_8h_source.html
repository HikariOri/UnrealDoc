31 {

132 template<typename RealType, typename VectorType>
133 RealType SignedArea2(const TArrayView<const VectorType>& Vertices)
134 {
135 RealType Area = 0;
136 int N = Vertices.Num();
137 if (N == 0)
138 {
139 return 0;
140 }
141 for (int Idx = 0, PrevIdx = N-1; Idx < N; PrevIdx=Idx++)
142 {
143 const TVector2<RealType>& V1 = Vertices[PrevIdx];
144 const TVector2<RealType>& V2 = Vertices[Idx];
145 Area += V1.X * V2.Y - V1.Y * V2.X;
146 }
147 return static_cast<RealType>(Area * 0.5);

153 template<typename RealType, typename VectorType>
154 RealType WindingIntegral2(const TArrayView<const VectorType>& Vertices, const VectorType& QueryPoint)
155 {
156 RealType Sum = 0;
157 int N = Vertices.Num();
158 VectorType A = Vertices[N-1] - QueryPoint, B = VectorType::Zero();
159 for (int Idx = 0; Idx < N; ++Idx)
160 {
161 B = Vertices[Idx] - QueryPoint;
162 // TODO: Consider computing closed curve winding w/out trig functions, i.e. see Contains2 below
163 Sum += TMathUtil<RealType>::Atan2(A.X * B.Y - A.Y * B.X, A.X * B.X + A.Y * B.Y);
164 A = B;
165 }
166 return Sum / FMathd::TwoPi;
167 }

226 auto ProjectToSegmentHelper = [&Vertices, &QueryPoint, &OutNearestSegIndex, &OutNearestPoint, &OutNearestSegmentUnitParam, OutNearestSegmentParam, &OutDistanceSq](int32 SegIdx, int32 EndIdx)
227 {
228 VectorType A = Vertices[SegIdx], B = Vertices[EndIdx];
229 RealType SegLenSq;
230 RealType LerpParam = ProjectToSegment(A, B, SegLenSq, QueryPoint);
231 VectorType ProjectedSegmentPoint = FMath::Lerp(A, B, LerpParam);
232 RealType DSq = VectorType::DistSquared(ProjectedSegmentPoint, QueryPoint);
233 if (DSq < OutDistanceSq)
234 {
235 OutDistanceSq = DSq;
236 OutNearestSegIndex = SegIdx;
237 OutNearestPoint = ProjectedSegmentPoint;
238 OutNearestSegmentUnitParam = LerpParam;
239 if (OutNearestSegmentParam)
240 {
241 // Convert LerpParam (0 to 1 range) to SegParam (+/- Extent aka Half-Length range)
242 *OutNearestSegmentParam = (LerpParam - (RealType).5) * FMath::Sqrt(SegLenSq);
243 }
244 }
245 };
246 for (int SegIdx = 0, N = Vertices.Num(); SegIdx + 1 < N; ++SegIdx)
247 {
248 ProjectToSegmentHelper(SegIdx, SegIdx + 1);
249 }
250 if (bLoop)
251 {
252 ProjectToSegmentHelper(Vertices.Num() - 1, 0);
253 }
254 return true;
255 }

261 template<typename RealType, typename VectorType>
262 bool Contains2(const TArrayView<const VectorType>& Vertices, const VectorType& QueryPoint)
263 {
264 int WindingNumber = 0;
265
266 int N = Vertices.Num();
267 if (N == 0)
268 {
269 return false;
270 }
271 VectorType A = Vertices[N - 1], B = VectorType::Zero();
272 for (int Idx = 0; Idx < N; ++Idx)
273 {
274 B = Vertices[Idx];
275
276 if (A.Y <= QueryPoint.Y) // y <= P.Y (below)
277 {
278 if (B.Y > QueryPoint.Y) // an upward crossing
279 {
280 if (Orient(A, B, QueryPoint) > 0) // P left of edge
281 {
282 ++WindingNumber; // have a valid up intersect
283 }
284 }
285 }
286 else // y > P.Y (above)
287 {
288 if (B.Y <= QueryPoint.Y) // a downward crossing
289 {
290 if (Orient(A, B, QueryPoint) < 0) // P right of edge
291 {
292 --WindingNumber; // have a valid down intersect
293 }
294 }
295 }
296 A = B;
297 }
298 return WindingNumber != 0;
299 }

342 template<typename RealType, typename VectorType, bool bLoop = true, int ClipDims = 2>
343 static void ClipConvexToBounds(TArray<VectorType>& Vertices, VectorType Min, VectorType Max)
344 {
345 TArray<VectorType> Clipped;
346 Clipped.Reserve(Vertices.Num());
347 VectorType SidePt = Min;
348 for (int32 SideIdx = 0; SideIdx < 2; ++SideIdx, SidePt = Max)
349 {
350 int32 SideSign = -SideIdx * 2 + 1;
351 for (int32 ClipDim = 0; ClipDim < ClipDims; ++ClipDim)
352 {
353 RealType ClipCoord = SidePt[ClipDim];
354 int32 VertNum = Vertices.Num();
355 int32 StartCur, StartPrev;
356 if constexpr (bLoop)
357 {
358 StartCur = 0;
359 StartPrev = VertNum - 1;
360 }
361 else
362 {
363 StartCur = 1;
364 StartPrev = 0;
365 }
366 RealType PrevDist = (Vertices[StartPrev][ClipDim] - ClipCoord) * RealType(SideSign);
367 if constexpr (!bLoop)
368 {
369 if (PrevDist >= 0)
370 {
371 Clipped.Add(Vertices[0]);
372 }
373 }
374 for (int32 CurIdx = StartCur, PrevIdx = StartPrev; CurIdx < VertNum; PrevIdx = CurIdx++)
375 {
376 RealType CurDist = (Vertices[CurIdx][ClipDim] - ClipCoord) * RealType(SideSign);
377 if (CurDist >= 0)
378 {
379 if (PrevDist < 0 && CurDist > 0)
380 {
381 RealType T = CurDist / (CurDist - PrevDist);
382 VectorType LerpVec = Lerp(Vertices[CurIdx], Vertices[PrevIdx], T);
383 LerpVec[ClipDim] = ClipCoord; // snap to exact bounds
384 Clipped.Add(LerpVec);
385 }
386 Clipped.Add(Vertices[CurIdx]);
387 }
388 else if (PrevDist > 0)
389 {
390 RealType T = CurDist / (CurDist - PrevDist);
391 VectorType LerpVec = Lerp(Vertices[CurIdx], Vertices[PrevIdx], T);
392 LerpVec[ClipDim] = ClipCoord; // snap to exact bounds
393 Clipped.Add(LerpVec);
394 }
395 PrevDist = CurDist;
396 }
397 Swap(Vertices, Clipped);
398 if (Vertices.IsEmpty())
399 {
400 return;
401 }
402 Clipped.Reset();
403 }
404 }

417 template<typename RealType, typename VectorType, bool bLoop = true>
418 static bool ClipConvexToPlane(TArray<VectorType>& Vertices, const TPlane<RealType>& Plane, TArray<VectorType>& OutClipped, bool bKeepPositiveSide = false)
419 {
420 OutClipped.Reset(Vertices.Num());
421 bool bWasClipped = false;
422 int32 VertNum = Vertices.Num();
423 int32 StartCur, StartPrev;
424 if constexpr (bLoop)
425 {
426 StartCur = 0;
427 StartPrev = VertNum - 1;
428 }
429 else
430 {
431 StartCur = 1;
432 StartPrev = 0;
433 }
434 RealType SideSign = bKeepPositiveSide ? (RealType)1 : (RealType)-1;
435 RealType PrevDist = Plane.PlaneDot(Vertices[StartPrev]) * RealType(SideSign);
436 if constexpr (!bLoop)
437 {
438 if (PrevDist >= 0)
439 {
440 OutClipped.Add(Vertices[0]);
441 }
442 else
443 {
444 bWasClipped = true;
445 }
446 }
447 for (int32 CurIdx = StartCur, PrevIdx = StartPrev; CurIdx < VertNum; PrevIdx = CurIdx++)
448 {
449 RealType CurDist = Plane.PlaneDot(Vertices[CurIdx]) * RealType(SideSign);
450 if (CurDist >= 0)
451 {
452 if (PrevDist < 0 && CurDist > 0)
453 {
454 RealType T = CurDist / (CurDist - PrevDist);
455 VectorType LerpVec = Lerp(Vertices[CurIdx], Vertices[PrevIdx], T);
456 OutClipped.Add(LerpVec);
457 bWasClipped = true;
458 }
459 OutClipped.Add(Vertices[CurIdx]);
460 }
461 else
462 {
463 bWasClipped = true;
464 if (PrevDist > 0)
465 {
466 RealType T = CurDist / (CurDist - PrevDist);
467 VectorType LerpVec = Lerp(Vertices[CurIdx], Vertices[PrevIdx], T);
468 OutClipped.Add(LerpVec);
469 }
470 }
471 PrevDist = CurDist;
472 }

484 template<typename RealType, typename VectorType>
485 bool IsConvex2(const TArrayView<const VectorType> Vertices, RealType Tolerance = TMathUtil<RealType>::ZeroTolerance,
486 bool bDegenerateIsConvex = true)
487 {
488 int32 N = Vertices.Num();
489 if (N < 3)
490 {
491 // degenerate case, less than 3 points
492 return bDegenerateIsConvex;
493 }
494 VectorType FromPrev = Vertices[N - 1] - Vertices[N - 2];
495 int32 UsedPrevIdx = N - 2;
496 while (!FromPrev.Normalize(0))
497 {
498 UsedPrevIdx--;
499 if (UsedPrevIdx < 0)
500 {
501 return bDegenerateIsConvex; // degenerate case: all points coincident
502 }
503 FromPrev = Vertices[N - 1] - Vertices[UsedPrevIdx];
504 }
505 bool bSeenNeg = false, bSeenPos = false;
506 RealType AngleSum = 0;
507 int32 NonZeroEdges = 0;
508 for (int32 Cur = N-1, Next = 0; Next < N; Next++)
509 {
510 VectorType ToNext = Vertices[Next] - Vertices[Cur];
511 if (!ToNext.Normalize(0)) // skip duplicates
512 {
513 continue;
514 }
515 RealType Angle = SignedAngleR(FromPrev, ToNext);
516 bSeenNeg |= (Angle < -Tolerance);
517 bSeenPos |= (Angle > Tolerance);
518 AngleSum += Angle;
519 FromPrev = ToNext;
520 Cur = Next;
521 NonZeroEdges++;
522 }
523 if (NonZeroEdges < 3)
524 {
525 return bDegenerateIsConvex;
526 }
527 return !(bSeenNeg && bSeenPos) // curve should only turn one way, and
528 // curve should loop only once, turning 2*Pi radians in that one direction
529 && int(TMathUtil<RealType>::Round(AngleSum / TMathUtil<RealType>::TwoPi)) == (int)bSeenPos - (int)bSeenNeg;
530 }

539 template<typename RealType>
540 bool ProjectPointInsideConvexPolygon(const TArrayView<const TVector2<RealType>> Vertices, TVector2<RealType>& ProjPt, bool bReverseOrientation = false)
541 {
542 bool bProjected = false;
543 const int32 N = Vertices.Num();
544 for (int32 Idx = 0, PrevIdx = N - 1; Idx < N; PrevIdx = Idx++)
545 {
546 const TVector2<RealType>& V1 = Vertices[PrevIdx];
547 const TVector2<RealType>& V2 = Vertices[Idx];
548 TVector2<RealType> ToPt = ProjPt - V1;
549 TVector2<RealType> Normal = PerpCW<RealType>(V2 - V1);
550 RealType Orient = ToPt.Dot(Normal);
551 bool bOutside = bReverseOrientation ? Orient < 0 : Orient > 0;
552 if (bOutside)
553 {
554 RealType SqLen = Normal.SquaredLength();
555 if (SqLen > (RealType)FLT_MIN)
556 {
557 bProjected = true;
558 ProjPt -= Normal * Orient / SqLen;
559 }
560 }
561 }
562 return bProjected;

569 template<typename RealType, typename VectorType>
570 void InPlaceSmooth(TArrayView<VectorType> Vertices, int StartIdx, int EndIdx, double Alpha, int NumIterations, bool bClosed)
571 {
572 int N = Vertices.Num();
573 if ( bClosed )
574 {
575 for (int Iter = 0; Iter < NumIterations; ++Iter)
576 {
577 for (int ii = StartIdx; ii < EndIdx; ++ii)
578 {
579 int i = (ii % N);
580 int iPrev = (ii == 0) ? N - 1 : ii - 1;
581 int iNext = (ii + 1) % N;
582 TVector<RealType> prev = Vertices[iPrev], next = Vertices[iNext];
583 TVector<RealType> c = (prev + next) * 0.5f;
584 Vertices[i] = (1 - Alpha) * Vertices[i] + (Alpha) * c;
585 }
586 }
587 }
588 else
589 {
590 for (int Iter = 0; Iter < NumIterations; ++Iter)
591 {
592 for (int i = StartIdx; i < EndIdx; ++i)
593 {
594 if (i == 0 || i >= N - 1)
595 {
596 continue;
597 }
598 TVector<RealType> prev = Vertices[i - 1], next = Vertices[i + 1];
599 TVector<RealType> c = (prev + next) * 0.5f;
600 Vertices[i] = (1 - Alpha) * Vertices[i] + (Alpha) * c;
601 }
602 }
