67 void ApplyFriction(FSolverParticles& Particles, const FSolverReal Dt, const int32 Index) const
68 {
69 check(!bFastPositionBasedFriction); // Do not call this function if this is setup to run with fast PB friction
70
71 if (!MVelocityConstraints.Contains(Index))
72 {
73 return;
74 }
75 const FSolverReal VN = FSolverVec3::DotProduct(Particles.V(Index), MVelocityConstraints[Index].Normal);
76 const FSolverReal VNBody = FSolverVec3::DotProduct(MVelocityConstraints[Index].Velocity, MVelocityConstraints[Index].Normal);
77 const FSolverVec3 VTBody = MVelocityConstraints[Index].Velocity - VNBody * MVelocityConstraints[Index].Normal;
78 const FSolverVec3 VTRelative = Particles.V(Index) - VN * MVelocityConstraints[Index].Normal - VTBody;
79 const FSolverReal VTRelativeSize = VTRelative.Size();
80 const FSolverReal VNMax = FMath::Max(VN, VNBody);
81 const FSolverReal VNDelta = VNMax - VN;
82 const FSolverReal CoefficientOfFriction = MPerGroupFriction[MDynamicGroupIds[Index]];
83 check(CoefficientOfFriction > 0);
84 const FSolverReal Friction = CoefficientOfFriction * VNDelta < VTRelativeSize ? CoefficientOfFriction * VNDelta / VTRelativeSize : 1;
85 Particles.V(Index) = VNMax * MVelocityConstraints[Index].Normal + VTBody + VTRelative * (1 - Friction);
86 }
