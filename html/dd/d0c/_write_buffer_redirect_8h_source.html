63template <uint16 BufferSize>
64FWriteBuffer* TWriteBufferQueue<BufferSize>::AllocBuffer()
65{
66 constexpr uint16 UsableBufferSize = BufferSize - sizeof(FWriteBuffer);
67 static_assert(PLATFORM_CACHE_LINE_SIZE % alignof(FWriteBuffer) == 0);
68 static_assert(UsableBufferSize % alignof(FWriteBuffer) == 0);
69 static_assert(UsableBufferSize <= UINT16_MAX);
70 static_assert(BufferSize % PLATFORM_CACHE_LINE_SIZE == 0); // _aligned_alloc requirement
71
72 uint8* AllocBase = (uint8*) Writer_MemoryAllocate(BufferSize, PLATFORM_CACHE_LINE_SIZE);
73
74 FWriteBuffer* NextBuffer = (FWriteBuffer*)(AllocBase + UsableBufferSize);
75 NextBuffer->Size = UsableBufferSize;
76 NextBuffer->CursorMaskInv = 0;
77
78 // Reserve some space for packet header
79 uint8* DataStart = AllocBase + sizeof(FTidPacketBase);
80 NextBuffer->SetCursor(DataStart);
81 NextBuffer->Committed = DataStart;
82 NextBuffer->Reaped = DataStart;
83
84 NextBuffer->EtxOffset = 0 - int32(sizeof(FWriteBuffer));
85 NextBuffer->ThreadId = 0;
86 NextBuffer->NextBuffer = nullptr;
87
88 return NextBuffer;
89}

92template <uint16 BufferSize>
93FWriteBuffer* TWriteBufferQueue<BufferSize>::FreeBuffer(FWriteBuffer* Buffer)
94{
95 FWriteBuffer* NextBuffer = Buffer->NextBuffer;
96 const uint32 AllocSize = sizeof(FWriteBuffer) + Buffer->Size;
97 uint8* AllocBase = ((uint8*)Buffer) - Buffer->Size;
98
99 Writer_MemoryFree(AllocBase, AllocSize);
100
101 return NextBuffer;
102}

105template <uint16 BufferSize>
106FWriteBuffer* TWriteBufferQueue<BufferSize>::NextBuffer(FWriteBuffer* __restrict PrevBuffer)
107{
108 FWriteBuffer* NextBuffer = AllocBuffer();
109 PrevBuffer->NextBuffer = NextBuffer;
110 GTlsState.WriteBuffer = NextBuffer;
111 return NextBuffer;
112}
