420 bool IsSameAs(const TDynamicVertexAttribute<AttribValueType, AttribDimension, ParentType>& Other, bool bIgnoreDataLayout) const
421 {
422 if (!bIgnoreDataLayout)
423 {
424 if (AttribValues.Num() != Other.AttribValues.Num())
425 {
426 return false;
427 }
428
429 for (int Idx = 0, NumValues = AttribValues.Num(); Idx < NumValues; Idx++)
430 {
431 if (AttribValues[Idx] != Other.AttribValues[Idx])
432 {
433 return false;
434 }
435 }
436 }
437 else
438 {
439 // bIgnoreDataLayout
440
441 FRefCountVector::IndexIterator ItVid = Parent->GetVerticesRefCounts().BeginIndices();
442 const FRefCountVector::IndexIterator ItVidEnd = Parent->GetVerticesRefCounts().EndIndices();
443 FRefCountVector::IndexIterator ItVidOther = Other.Parent->GetVerticesRefCounts().BeginIndices();
444 const FRefCountVector::IndexIterator ItVidEndOther = Other.Parent->GetVerticesRefCounts().EndIndices();
445
446 while (ItVid != ItVidEnd && ItVidOther != ItVidEndOther)
447 {
448 for (int32 i = 0; i < AttribDimension; ++i)
449 {
450 const AttribValueType AttribValue = AttribValues[*ItVid * AttribDimension + i];
451 const AttribValueType AttribValueOther = Other.AttribValues[*ItVidOther * AttribDimension + i];
452 if (AttribValue != AttribValueOther)
453 {
454 // Vertex attribute value is not the same.
455 return false;
456 }
457 }
458 ++ItVid;
459 ++ItVidOther;
460 }
461
462 if (ItVid != ItVidEnd || ItVidOther != ItVidEndOther)
463 {
464 // Number of vertex attribute values is not the same.
465 return false;
466 }
467 }
468
469 return true;
470 }

557 {

603 for (int VertexID : VertexIDs)

611template<typename AttribValueType, int AttribDimension, typename ParentType>
612bool TDynamicVertexAttributeChange<AttribValueType, AttribDimension, ParentType>::Apply(TDynamicAttributeBase<ParentType>* Attribute, bool bRevert) const
613{
614 const TArray<FChangeVertexAttribute> *Changes = bRevert ? &OldVertexAttributes : &NewVertexAttributes;
615 TDynamicVertexAttribute<AttribValueType, AttribDimension, ParentType>* AttribCast = static_cast<TDynamicVertexAttribute<AttribValueType, AttribDimension, ParentType>*>(Attribute);
616 for (const FChangeVertexAttribute& Change : *Changes)
617 {
618 bool bIsVertex = AttribCast->GetParent()->IsVertex(Change.VertexID);
619 check(bIsVertex);
620 if (bIsVertex)
621 {
622 AttribCast->SetValue(Change.VertexID, Change.Data);
623 }
624 }
625 return true;
626}
