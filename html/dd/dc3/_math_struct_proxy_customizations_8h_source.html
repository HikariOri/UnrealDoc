251template<typename ProxyType, typename NumericType>
252inline TSharedRef<SWidget> FMathStructProxyCustomization::MakeNumericProxyWidget(
253 TSharedRef<IPropertyHandle>& StructPropertyHandle,
254 TSharedRef< TProxyProperty<ProxyType, NumericType> >& ProxyValue,
255 const FText& Label,
256 bool bRotationInDegrees,
257 const FLinearColor& LabelBackgroundColor)
258{
259 TWeakPtr<IPropertyHandle> WeakHandlePtr = StructPropertyHandle;
260
261 TSharedPtr<INumericTypeInterface<NumericType>> TypeInterface = nullptr;
262 if (FUnitConversion::Settings().ShouldDisplayUnits() && bRotationInDegrees)
263 {
264 TypeInterface = MakeShared<TNumericUnitTypeInterface<NumericType>>(EUnit::Degrees);
265 }
266
267 return
268 SNew(SNumericEntryBox<NumericType>)
269 .IsEnabled(this, &FMathStructProxyCustomization::IsValueEnabled, WeakHandlePtr)
270 .Value(this, &FMathStructProxyCustomization::OnGetValue<ProxyType, NumericType>, WeakHandlePtr, ProxyValue)
271 .Font(IDetailLayoutBuilder::GetDetailFont())
272 .UndeterminedString(NSLOCTEXT("PropertyEditor", "MultipleValues", "Multiple Values"))
273 .OnValueCommitted(this, &FMathStructProxyCustomization::OnValueCommitted<ProxyType, NumericType>, WeakHandlePtr, ProxyValue)
274 .OnValueChanged(this, &FMathStructProxyCustomization::OnValueChanged<ProxyType, NumericType>, WeakHandlePtr, ProxyValue)
275 .OnBeginSliderMovement(this, &FMathStructProxyCustomization::OnBeginSliderMovement)
276 .OnEndSliderMovement(this, &FMathStructProxyCustomization::OnEndSliderMovement<ProxyType, NumericType>, WeakHandlePtr, ProxyValue)
277 // Only allow spin on handles with one object. Otherwise it is not clear what value to spin
278 .AllowSpin(StructPropertyHandle->GetNumOuterObjects() == 1)
279 .MinValue(TOptional<NumericType>())
280 .MaxValue(TOptional<NumericType>())
281 .MaxSliderValue(bRotationInDegrees ? 360.0f : TOptional<NumericType>())
282 .MinSliderValue(bRotationInDegrees ? 0.0f : TOptional<NumericType>())
283 .LabelPadding(FMargin(3))
284 .ToolTipTextFormat(this, &FMathStructProxyCustomization::OnGetValueToolTipTextFormat, Label)
285 .LabelLocation(SNumericEntryBox<NumericType>::ELabelLocation::Inside)
286 .TypeInterface(TypeInterface)
287 .Label()
288 [
289 SNumericEntryBox<NumericType>::BuildNarrowColorLabel(LabelBackgroundColor)
290 ];
291}
