216 union
217 {
218 struct
219 {
220 uint64 Reserved : REMOTE_OBJECT_RESERVED_BIT_SIZE; /* Reserved for static flag in NetGUID and FRemoteObjectHandle */
221 uint64 SerialNumber : REMOTE_OBJECT_SERIAL_NUMBER_BIT_SIZE;
222 uint64 ServerId : REMOTE_OBJECT_SERVER_ID_BIT_SIZE;
223 };
224 uint64 Id = 0;
225 };
226
227 static_assert((REMOTE_OBJECT_RESERVED_BIT_SIZE + REMOTE_OBJECT_SERIAL_NUMBER_BIT_SIZE + REMOTE_OBJECT_SERVER_ID_BIT_SIZE) <= (sizeof(Id) * 8), "FRemoteObjectId component bit sizes must fit within the Id member");
228
229 inline int64 GetComparableValue() const
230 {
231 return int64(SerialNumber) | (int64(FRemoteServerId::GlobalizeId(ServerId)) << REMOTE_OBJECT_SERIAL_NUMBER_BIT_SIZE);
232 }

234 inline int64 Compare(FRemoteObjectId Other) const
235 {
236 FRemoteServerId::VerifyComparableServerIds(ServerId, Other.ServerId);
237 return GetComparableValue() - Other.GetComparableValue();
