23template<typename TResourceType, typename TCallback>
24inline void EnumerateUniformBufferResources(const FRHIUniformBuffer* RESTRICT Buffer, int32 BufferIndex, const uint32* RESTRICT ResourceMap, TCallback && Callback)
25{
26 const TRefCountPtr<FRHIResource>* RESTRICT Resources = Buffer->GetResourceTable().GetData();
27
28 uint32 BufferOffset = ResourceMap[BufferIndex];
29 if (BufferOffset > 0)
30 {
31 const uint32* RESTRICT ResourceInfos = &ResourceMap[BufferOffset];
32 uint32 ResourceInfo = *ResourceInfos++;
33 do
34 {
35 checkSlow(FRHIResourceTableEntry::GetUniformBufferIndex(ResourceInfo) == BufferIndex);
36
37 const uint16 ResourceIndex = FRHIResourceTableEntry::GetResourceIndex(ResourceInfo);
38 const uint8 BindIndex = FRHIResourceTableEntry::GetBindIndex (ResourceInfo);
39
40 TResourceType* Resource = static_cast<TResourceType*>(Resources[ResourceIndex].GetReference());
41 checkf(Resource
42 , TEXT("Null %s (resource %d bind %d) on UB Layout %s")
43 , TResourceTypeStr<TResourceType>::String
44 , ResourceIndex
45 , BindIndex
46 , *Buffer->GetLayout().GetDebugName()
47 );
48
49 Callback(Resource, BindIndex);
50
51 ResourceInfo = *ResourceInfos++;
52 } while (FRHIResourceTableEntry::GetUniformBufferIndex(ResourceInfo) == BufferIndex);
53 }
54}

57// Helper to call the respective setters for each FRHIResource contained inside the provided uniform buffer.
58template<typename TBinder>
59void SetAllUniformBufferResourcesForBindless(TBinder&& Binder, const FRHIUniformBuffer* Buffer)
60{
61 double CurrentTimeForTextureTimes = FAppTime::GetApproximateCurrentTime();
62
63 for (const TRefCountPtr<FRHIResource>& Resource : Buffer->GetResourceTable())
64 {
65 if (Resource)
66 {
67 const ERHIResourceType ResourceType = Resource->GetType();
68 switch (ResourceType)
69 {
70 case RRT_Texture:
71 static_cast<FRHITexture*>(Resource.GetReference())->SetLastRenderTime(CurrentTimeForTextureTimes);
72 Binder.SetTexture(static_cast<FRHITexture*>(Resource.GetReference()), 0);
73 break;
74 case RRT_TextureReference:
75 static_cast<FRHITextureReference*>(Resource.GetReference())->SetLastRenderTime(CurrentTimeForTextureTimes);
76 Binder.SetTexture(static_cast<FRHITextureReference*>(Resource.GetReference()), 0);
77 break;
78 case RRT_ShaderResourceView:
79 Binder.SetSRV(static_cast<FRHIShaderResourceView*>(Resource.GetReference()), 0);
80 break;
81 case RRT_UnorderedAccessView:
82 Binder.SetUAV(static_cast<FRHIUnorderedAccessView*>(Resource.GetReference()), 0);
83 break;
84 case RRT_ResourceCollection:
85 Binder.SetResourceCollection(static_cast<FRHIResourceCollection*>(Resource.GetReference()), 0);
86 break;
87 case RRT_SamplerState:
88 Binder.SetSampler(static_cast<FRHISamplerState*>(Resource.GetReference()), 0);
89 break;
90 }
91 }
92 }
93}

96template<typename TBinder, typename TUniformBufferArrayType, typename TBitMaskType, bool bFullyBindless = false>
97void SetUniformBufferResourcesFromTables(TBinder && Binder, FRHIShader const& Shader, TBitMaskType& DirtyUniformBuffers, TUniformBufferArrayType const& BoundUniformBuffers
98 #if ENABLE_RHI_VALIDATION
99 , RHIValidation::FTracker* Tracker
100 #endif
101 )
102{
103 double CurrentTimeForTextureTimes = FAppTime::GetApproximateCurrentTime();
104 FShaderResourceTable const& SRT = Shader.GetShaderResourceTable();
105
106 // Mask the dirty bits by those buffers from which the shader has bound resources.
107 uint32 DirtyBits = SRT.ResourceTableBits & DirtyUniformBuffers;
108
109 #if PLATFORM_SUPPORTS_BINDLESS_RENDERING && !ENABLE_RHI_VALIDATION
110 if constexpr (bFullyBindless)
111 {
112 while (DirtyBits)
113 {
114 // Scan for the lowest set bit, compute its index, clear it in the set of dirty bits.
115 const uint32 LowestBitMask = (DirtyBits) & (-(int32)DirtyBits);
116 const int32 BufferIndex = FMath::CountTrailingZeros(LowestBitMask); // todo: This has a branch on zero, we know it could never be zero...
117 DirtyBits ^= LowestBitMask;
118
119 EnumerateUniformBufferResources<FRHITexture>(BoundUniformBuffers[BufferIndex], BufferIndex, SRT.TextureMap.GetData(),
120 [&](FRHITexture* Texture, uint8 Index)
121 {
122 Texture->SetLastRenderTime(CurrentTimeForTextureTimes);
123 });
124 }
125
126 DirtyUniformBuffers = TBitMaskType(0);
127
128 return;
129 }
130 #endif
131
132 while (DirtyBits)
133 {
134 // Scan for the lowest set bit, compute its index, clear it in the set of dirty bits.
135 const uint32 LowestBitMask = (DirtyBits) & (-(int32)DirtyBits);
136 const int32 BufferIndex = FMath::CountTrailingZeros(LowestBitMask); // todo: This has a branch on zero, we know it could never be zero...
137 DirtyBits ^= LowestBitMask;
138
139 check(BufferIndex < SRT.ResourceTableLayoutHashes.Num());
140
141 FRHIUniformBuffer* Buffer = BoundUniformBuffers[BufferIndex];
142
143 #if DO_CHECK
144
145 if (!Buffer)
146 {
147 UE_LOGF(LogRHI, Fatal, "Shader expected a uniform buffer at slot %u but got null instead (Shader='%ls' UB='%ls'). Rendering code needs to set a valid uniform buffer for this slot."
148 , BufferIndex
149 , Shader.GetShaderName()
150 , *Shader.GetUniformBufferName(BufferIndex)
151 );
152 }
153 else if (Buffer->GetLayout().GetHash() != SRT.ResourceTableLayoutHashes[BufferIndex])
154 {
155 FRHIUniformBufferLayout const& BufferLayout = Buffer->GetLayout();
156
157 FString ResourcesString;
158 for (FRHIUniformBufferResource const& Resource : BufferLayout.Resources)
159 {
160 ResourcesString += FString::Printf(TEXT("%s%d")
161 , ResourcesString.Len() ? TEXT(" ") : TEXT("")
162 , Resource.MemberType
163 );
164 }
165
166 // This might mean you are accessing a data you haven't bound e.g. GBuffer
167 UE_LOGF(LogRHI, Fatal,
168 "Uniform buffer bound to slot %u is not what the shader expected:\n" "\tBound : Uniform Buffer[%ls] with Hash[0x%08x]\n" "\tExpected : Uniform Buffer[%ls] with Hash[0x%08x]\n" "\tShader Name : %ls\n" "\tLayout CB Size : %d\n" "\tLayout Num Resources : %d\n" "\tResource Types : %ls\n"
169 , BufferIndex
170 , *BufferLayout.GetDebugName(), BufferLayout.GetHash()
171 , *Shader.GetUniformBufferName(BufferIndex), SRT.ResourceTableLayoutHashes[BufferIndex]
172 , Shader.GetShaderName()
173 , BufferLayout.ConstantBufferSize
174 , BufferLayout.Resources.Num()
175 , *ResourcesString
176 );
177 }
178
179 #endif // DO_CHECK
180
181 // Textures
182 EnumerateUniformBufferResources<FRHITexture>(Buffer, BufferIndex, SRT.TextureMap.GetData(),
183 [&](FRHITexture* Texture, uint8 Index)
184 {
185 #if ENABLE_RHI_VALIDATION
186 if (Tracker)
187 {
188 ERHIAccess Access = IsComputeShaderFrequency(Shader.GetFrequency())
189 ? ERHIAccess::SRVCompute
190 : Shader.GetFrequency() == SF_Pixel
191 ? ERHIAccess::SRVGraphicsPixel
192 : ERHIAccess::SRVGraphicsNonPixel;
193
194 // Textures bound here only have their "common" plane accessible. Stencil etc is ignored.
195 // (i.e. only access the color plane of a color texture, or depth plane of a depth texture)
196 Tracker->Assert(Texture->GetViewIdentity(0, 0, 0, 0, uint32(RHIValidation::EResourcePlane::Common), 1), Access);
197 }
198 #endif
199 Texture->SetLastRenderTime(CurrentTimeForTextureTimes);
200 Binder.SetTexture(Texture, Index);
201 });
202
203 // SRVs
204 EnumerateUniformBufferResources<FRHIShaderResourceView>(Buffer, BufferIndex, SRT.ShaderResourceViewMap.GetData(),
205 [&](FRHIShaderResourceView* SRV, uint8 Index)
206 {
207 #if ENABLE_RHI_VALIDATION
208 if (Tracker)
209 {
210 ERHIAccess Access = IsComputeShaderFrequency(Shader.GetFrequency())
211 ? ERHIAccess::SRVCompute
212 : Shader.GetFrequency() == SF_Pixel
213 ? ERHIAccess::SRVGraphicsPixel
214 : ERHIAccess::SRVGraphicsNonPixel;
215
216 Tracker->Assert(SRV->GetViewIdentity(), Access);
217 }
218 if (GRHIValidationEnabled)
219 {
220 RHIValidation::ValidateShaderResourceView(&Shader, Index, SRV);
221 }
222 #endif
223 Binder.SetSRV(SRV, Index);
224 });
225
226 #if PLATFORM_SUPPORTS_BINDLESS_RENDERING
227 EnumerateUniformBufferResources<FRHIResourceCollection>(Buffer, BufferIndex, SRT.ResourceCollectionMap.GetData(),
228 [&](FRHIResourceCollection* ResourceCollection, uint8 Index)
229 {
230 #if ENABLE_RHI_VALIDATION
231 // todo: christopher.waters - ResourceCollection validation
232 #endif
233 Binder.SetResourceCollection(ResourceCollection, Index);
234 });
235 #endif // PLATFORM_SUPPORTS_BINDLESS_RENDERING
236
237 // Samplers
238 EnumerateUniformBufferResources<FRHISamplerState>(Buffer, BufferIndex, SRT.SamplerMap.GetData(),
239 [&](FRHISamplerState* Sampler, uint8 Index)
240 {
241 Binder.SetSampler(Sampler, Index);
242 });
243
244 // UAVs
245 EnumerateUniformBufferResources<FRHIUnorderedAccessView>(Buffer, BufferIndex, SRT.UnorderedAccessViewMap.GetData(),
246 [&](FRHIUnorderedAccessView* UAV, uint8 Index)
247 {
248 #if ENABLE_RHI_VALIDATION
249 if (Tracker)
250 {
251 ERHIAccess Access = IsComputeShaderFrequency(Shader.GetFrequency())
252 ? ERHIAccess::UAVCompute
253 : ERHIAccess::UAVGraphics;
254
255 Tracker->AssertUAV(UAV, Access, Index);
256 }
257 #endif
258 Binder.SetUAV(UAV, Index);
259 });
260 }
261
262 DirtyUniformBuffers = TBitMaskType(0);
263}
