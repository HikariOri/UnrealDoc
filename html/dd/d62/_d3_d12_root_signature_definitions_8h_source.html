247 struct FPendingTable

251 int32 NumDescriptors;

262 {

278
279 void AddConstantsParameter(uint32 Num32BitValues, uint32 Register, uint32 Space) override
280 {
281 CD3DX12_ROOT_PARAMETER1& Parameter = Parameters.AddZeroed_GetRef();
282 Parameter.InitAsConstants(Num32BitValues, Register, Space);
283 }
284
285 void AddTable(ERootSignatureVisibility Visibility, ERootSignatureRangeType Type, int32 NumDescriptors, D3D12_DESCRIPTOR_RANGE_FLAGS FlagsOverride = D3D12_DESCRIPTOR_RANGE_FLAG_NONE) override
286 {
287 if (!ShouldSkipType(Type))
288 {
289 const FPendingTable NewTable{ Visibility, Type, NumDescriptors, FlagsOverride };
290 PendingTables.Emplace(NewTable);
291 }
292 }
293
294 const D3D12_VERSIONED_ROOT_SIGNATURE_DESC& Finalize()
295 {
296#if !defined(D3D12RHI_TOOLS_RAYTRACING_SHADERS_UNSUPPORTED)
297 if (RootFlags & D3D12_ROOT_SIGNATURE_FLAG_LOCAL_ROOT_SIGNATURE)
298 {
299 // if set, D3D12_ROOT_SIGNATURE_FLAG_LOCAL_ROOT_SIGNATURE should be the only flag set
300 RootFlags = D3D12_ROOT_SIGNATURE_FLAG_LOCAL_ROOT_SIGNATURE;
301 }
302#endif
303
304 DescriptorRanges.SetNum(PendingTables.Num());
305
306 for (int32 Index = 0; Index < PendingTables.Num(); Index++)
307 {
308 const FPendingTable& PendingTable = PendingTables[Index];
309
310 const D3D12_DESCRIPTOR_RANGE_FLAGS RangeFlags = PendingTable.FlagsOverride ? PendingTable.FlagsOverride : GetD3D12DescriptorRangeFlags(PendingTable.Type);
311
312 DescriptorRanges[Index].Init(GetD3D12DescriptorRangeType(PendingTable.Type), PendingTable.NumDescriptors, 0, RegisterSpace, RangeFlags);
313
314 CD3DX12_ROOT_PARAMETER1& Parameter = Parameters.AddZeroed_GetRef();
315 Parameter.InitAsDescriptorTable(1, &DescriptorRanges[Index], GetD3D12ShaderVisibility(PendingTable.Visibility));
316 }
317
318 RootDesc.Init_1_1(Parameters.Num(), Parameters.GetData(), 0, nullptr, RootFlags);
319
320 return RootDesc;
321 }

410 // Fat/Static Gfx Root Signature
411 inline void CreateGfxRootSignature(FRootSignatureCreator& Creator, EShaderBindingLayoutFlags InFlags)
412 {
413 // Ensure the creator starts in a clean state (in cases of creator reuse, etc.).
414 Creator.SetFlags(InFlags | EShaderBindingLayoutFlags::InputAssembler);
415
416 Creator.AddUnorderedAccessViewParameter(0, UE_HLSL_SPACE_DIAGNOSTIC);
417
418 if (EnumHasAnyFlags(InFlags, EShaderBindingLayoutFlags::RootConstants))
419 {
420 const uint32 NumConstants = 4u;
421 Creator.AddConstantsParameter(NumConstants, 0u, UE_HLSL_SPACE_SHADER_ROOT_CONSTANTS); // UERootConstants
422 }
423
424 // Tables must be added only after other parameters
425
426 AddAllStandardTablesForVisibility(Creator, ERootSignatureVisibility::Pixel);
427 AddAllStandardTablesForVisibility(Creator, ERootSignatureVisibility::Vertex);
428 AddAllStandardTablesForVisibility(Creator, ERootSignatureVisibility::Geometry);
429
430#if !defined(D3D12RHI_TOOLS_MESH_SHADERS_UNSUPPORTED)
431 if (EnumHasAnyFlags(InFlags, EShaderBindingLayoutFlags::AllowMeshShaders))
432 {
433 AddAllStandardTablesForVisibility(Creator, ERootSignatureVisibility::Mesh);
434 AddAllStandardTablesForVisibility(Creator, ERootSignatureVisibility::Amplification);
435 }
436#endif
437
438 Creator.AddTable(ERootSignatureVisibility::Pixel, ERootSignatureRangeType::UAV, MAX_UAVS);
439 Creator.AddTable(ERootSignatureVisibility::Vertex, ERootSignatureRangeType::UAV, MAX_UAVS);

463 inline void CreateRayTracingSignature(FRootSignatureCreator& Creator, bool bLocalRootSignature, D3D12_ROOT_SIGNATURE_FLAGS BaseRootFlags, EShaderBindingLayoutFlags InFlags)
464 {
465 Creator.SetFlags(InFlags);
466 Creator.AddRootFlag(BaseRootFlags);
467 Creator.SetRegisterSpace(bLocalRootSignature ? UE_HLSL_SPACE_RAY_TRACING_LOCAL : UE_HLSL_SPACE_RAY_TRACING_GLOBAL);
468
469 if (bLocalRootSignature)
470 {
471 if (Creator.HasFlags(EShaderBindingLayoutFlags::BindlessResources))
472 {
473 // Num constants could be 2 smaller if non-bindless data is removed (24 bytes instead of 32)
474 // (see notes in FD3D12RootSignatureDesc constructor)
475 uint32 NumConstants = sizeof(FD3D12HitGroupSystemParameters) / sizeof(uint32);
476 Creator.AddConstantsParameter(NumConstants, RAY_TRACING_SYSTEM_ROOTCONSTANT_REGISTER, UE_HLSL_SPACE_RAY_TRACING_SYSTEM);
477 }
478 else
479 {
480 uint32 NumConstants = sizeof(FHitGroupSystemRootConstants) / sizeof(uint32);
481 Creator.AddConstantsParameter(NumConstants, RAY_TRACING_SYSTEM_ROOTCONSTANT_REGISTER, UE_HLSL_SPACE_RAY_TRACING_SYSTEM);
482
483 Creator.AddShaderResourceViewParameter(RAY_TRACING_SYSTEM_INDEXBUFFER_REGISTER, UE_HLSL_SPACE_RAY_TRACING_SYSTEM);
484 Creator.AddShaderResourceViewParameter(RAY_TRACING_SYSTEM_VERTEXBUFFER_REGISTER, UE_HLSL_SPACE_RAY_TRACING_SYSTEM);
485 }
486 }
487 else if (Creator.HasFlags(EShaderBindingLayoutFlags::ShaderBindingLayoutUsed))
488 {
489 for (uint32 Index = 0; Index < FRHIShaderBindingLayout::MaxUniformBufferEntries; ++Index)
490 {
491 Creator.AddConstantBufferViewParameter(Index, UE_HLSL_SPACE_STATIC_SHADER_BINDINGS);
492 }
493 }
494
495 if (Creator.HasFlags(EShaderBindingLayoutFlags::BindlessResources))
496 {
497 for (uint32 Index = 0; Index < MAX_CBS; Index++)
498 {
499 Creator.AddConstantBufferViewParameter(Index, Creator.GetRegisterSpace());
500 }
501 }
502 else
503 {
504 AddAllStandardTablesForVisibility(Creator, ERootSignatureVisibility::All);
505 }
506
507 Creator.AddTable(ERootSignatureVisibility::All, ERootSignatureRangeType::UAV, MAX_UAVS, D3D12_DESCRIPTOR_RANGE_FLAG_DATA_STATIC_WHILE_SET_AT_EXECUTE | D3D12_DESCRIPTOR_RANGE_FLAG_DESCRIPTORS_VOLATILE);
508 }
