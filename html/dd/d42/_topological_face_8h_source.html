685 FVector Max;

687 FVector MaxPoints[3];
688 FVector MinPoints[3];
689 FVector2d MaxCoordinates[3];
690 FVector2d MinCoordinates[3];

695
696 FBBoxWithNormal()
697 : Max(-HUGE_VALUE, -HUGE_VALUE, -HUGE_VALUE)
698 , Min(HUGE_VALUE, HUGE_VALUE, HUGE_VALUE)
699 {
700 }
701
705 double Length()
706 {
707 double XLength = Max.X - Min.X;
708 double YLength = Max.Y - Min.Y;
709 double ZLength = Max.Z - Min.Z;
710 return XLength + YLength + ZLength;
711 }
712
716 bool CheckOrientation(bool bHasWrongOrientation)
717 {
718 int32 GoodOrientation = 0;
719 int32 WrongOrientation = 0;
720
721 const FVector BBoxNormals[] = { FVector(1., 0., 0.) , FVector(0., 1., 0.) , FVector(0., 0., 1.) };
722
723 for (int32 Index = 0; Index < 3; ++Index)
724 {
725 double DotProduct = MaxPointNormals[Index] | BBoxNormals[Index];
726 if (DotProduct > DOUBLE_KINDA_SMALL_NUMBER)
727 {
728 GoodOrientation++;
729 }
730 else if(DotProduct < DOUBLE_KINDA_SMALL_NUMBER)
731 {
732 WrongOrientation++;
733 }
734
735 DotProduct = MinPointNormals[Index] | BBoxNormals[Index];
736 if (DotProduct < DOUBLE_KINDA_SMALL_NUMBER)
737 {
738 GoodOrientation++;
739 }
740 else if (DotProduct > DOUBLE_KINDA_SMALL_NUMBER)
741 {
742 WrongOrientation++;
743 }
744 }
745
746 bHasWrongOrientation = (GoodOrientation < WrongOrientation);
747 return ((GoodOrientation >= 3) || (WrongOrientation >= 3)) && (GoodOrientation != WrongOrientation);
748 }
749
750 void Update(const FPolylineBBox& PolylineBBox, const EIso IsoType, double IsoCoordinate)
751 {
752 for (int32 Index = 0; Index < 3; ++Index)
753 {
754 if (PolylineBBox.Max[Index] > Max[Index])
755 {
756 Max[Index] = PolylineBBox.Max[Index];
757 MaxPoints[Index] = PolylineBBox.MaxPoints[Index];
758 MaxCoordinates[Index] = (IsoType == IsoU) ? FVector2d(IsoCoordinate, PolylineBBox.CoordinateOfMaxPoint[Index]) : FVector2d(PolylineBBox.CoordinateOfMaxPoint[Index], IsoCoordinate);
759 MaxNormalNeedUpdate[Index] = true;
760 }
761
762 if (PolylineBBox.Min[Index] < Min[Index])
763 {
764 Min[Index] = PolylineBBox.Min[Index];
765 MinPoints[Index] = PolylineBBox.MinPoints[Index];
766 MinCoordinates[Index] = (IsoType == IsoU) ? FVector2d(IsoCoordinate, PolylineBBox.CoordinateOfMinPoint[Index]) : FVector2d(PolylineBBox.CoordinateOfMinPoint[Index], IsoCoordinate);
767 MinNormalNeedUpdate[Index] = true;
768 }
769 }
770 }
771
772 void Update(const FVector& Point, const FVector2d InPoint2D)
773 {
774 for (int32 Index = 0; Index < 3; ++Index)
775 {
776 if (Point[Index] > Max[Index])
777 {
778 Max[Index] = Point[Index];
779 MaxPoints[Index] = Point;
780 MaxCoordinates[Index] = InPoint2D;
781 MaxNormalNeedUpdate[Index] = true;
782 }
783
784 if (Point[Index] < Min[Index])
785 {
786 Min[Index] = Point[Index];
787 MinPoints[Index] = Point;
788 MinCoordinates[Index] = InPoint2D;
789 MinNormalNeedUpdate[Index] = true;
790 }
791 }
792 }
793
794 void UpdateNormal(const FTopologicalFace& Face)
795 {
796 const FSurface& Surface = *Face.GetCarrierSurface();
797 bool bSwapOrientation = Face.IsBackOriented();
798
799 for (int32 Index = 0; Index < 3; ++Index)
800 {
801 if (MaxNormalNeedUpdate[Index])
802 {
803 MaxPointNormals[Index] = Surface.EvaluateNormal(MaxCoordinates[Index]);
804 if (bSwapOrientation)
805 {
806 MaxPointNormals[Index] *= -1.;
807 }
808 MaxNormalNeedUpdate[Index] = false;
809 }
810
811 if (MinNormalNeedUpdate[Index])
812 {
813 MinPointNormals[Index] = Surface.EvaluateNormal(MinCoordinates[Index]);
814 if (bSwapOrientation)
815 {
816 MinPointNormals[Index] *= -1.;
817 }
818 MinNormalNeedUpdate[Index] = false;
819 }
820 }
