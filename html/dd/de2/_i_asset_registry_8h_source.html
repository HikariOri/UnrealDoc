184struct FAssetDependency
185{
186 FAssetIdentifier AssetId;
187 UE::AssetRegistry::EDependencyCategory Category = UE::AssetRegistry::EDependencyCategory::None;
188 UE::AssetRegistry::EDependencyProperty Properties = UE::AssetRegistry::EDependencyProperty::None;
189
190 bool operator==(const FAssetDependency& Other) const
191 {
192 return AssetId == Other.AssetId && Category == Other.Category && Properties == Other.Properties;
193 }
194 bool LexicalLess(const FAssetDependency& Other) const
195 {
196 if (!(AssetId == Other.AssetId))
197 {
198 return AssetId.LexicalLess(Other.AssetId);
199 }
200 if (Category != Other.Category)
201 {
202 return Category < Other.Category;
203 }
204 return Properties < Other.Properties;
205 }
206
207 static FAssetDependency PackageDependency(FName PackageName, UE::AssetRegistry::EDependencyProperty Properties)
208 {
209 return FAssetDependency{ FAssetIdentifier(PackageName), UE::AssetRegistry::EDependencyCategory::Package,
210 Properties };
211 }
212
213 friend inline uint32 GetTypeHash(const FAssetDependency& InHash)
214 {
215 uint32 Hash = GetTypeHash(InHash.AssetId);
216 Hash = HashCombine(Hash, GetTypeHash(InHash.Category));
217 Hash = HashCombine(Hash, GetTypeHash(InHash.Properties));
218
219 return Hash;
220 }
