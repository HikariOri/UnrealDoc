11FORCEINLINE uint32 HashPosition( const FVector3f& Position )
12{
13 union { float f; uint32 i; } x;
14 union { float f; uint32 i; } y;
15 union { float f; uint32 i; } z;
16
17 x.f = Position.X;
18 y.f = Position.Y;
19 z.f = Position.Z;
20
21 return Murmur32( {
22 Position.X == 0.0f ? 0u : x.i,
23 Position.Y == 0.0f ? 0u : y.i,
24 Position.Z == 0.0f ? 0u : z.i
25 } );
26}

28FORCEINLINE uint32 Cycle3( uint32 Value )
29{
30 uint32 ValueMod3 = Value % 3;

52FORCEINLINE float TriangleArea( float a, float b, float c )
53{
54 float AreaSqrTimes16 = FMath::Max( 0.0f,
55 ( a + b + c ) *
56 (-a + b + c ) *
57 ( a - b + c ) *
58 ( a + b - c ) );
59 return FMath::Sqrt( AreaSqrTimes16 ) * 0.25f;
60}

63FORCEINLINE int32 ApproxNumTris( int32 a, int32 b, int32 c )
64{
65 // Heron's formula divided by area of unit triangle
66 float s = 0.5f * float( a + b + c );
67 float NumTris = 4.0f * FMath::Sqrt( FMath::Max( 0.0625f, s * (s - (float)a) * (s - (float)b) * (s - (float)c) / 3.0f ) );
68 int32 MaxFactor = FMath::Max3( a, b, c );
69 return FMath::Max( FMath::RoundToInt( NumTris ), MaxFactor );
70}

75 // [ Schindler and Chen 2012, "Barycentric Coordinates in Olympiad Geometry" https://web.evanchen.cc/handouts/bary/bary-full.pdf ]
76 FORCEINLINE float LengthSquared( const FVector3f& Barycentrics0, const FVector3f& Barycentrics1, const FVector3f& EdgeLengthsSqr )
77 {
78 // Barycentric displacement vector:
79 // 0 = x + y + z
80 FVector3f Disp = Barycentrics0 - Barycentrics1;
81
82 /* TODO change edge order to match ariel coords
83 v0
84 /\
85 e2 / \ e0
86 /____\
87 v2 e1 v1
88 */
89
90 // Length of displacement
91 return -Disp.X * Disp.Y * EdgeLengthsSqr[0]
92 -Disp.Y * Disp.Z * EdgeLengthsSqr[1]
93 -Disp.Z * Disp.X * EdgeLengthsSqr[2];
94 }

96 FORCEINLINE float SubtriangleArea( const FVector3f& Barycentrics0, const FVector3f& Barycentrics1, const FVector3f& Barycentrics2, float TriangleArea )
97 {
98 // Area * Determinant using triple product
99 return TriangleArea * FMath::Abs( Barycentrics0 | ( Barycentrics1 ^ Barycentrics2 ) );
100 }

108 // Angle at corner 0
109 FORCEINLINE float Cotangent(
110 const FVector3f& Barycentrics0,
111 const FVector3f& Barycentrics1,
112 const FVector3f& Barycentrics2,
113 const FVector3f& EdgeLengthsSqr,
114 float TriangleArea )
115 {
116 FVector3f LengthsSqr;
117 LengthsSqr[0] = LengthSquared( Barycentrics1, Barycentrics2, EdgeLengthsSqr );
118 LengthsSqr[1] = LengthSquared( Barycentrics2, Barycentrics0, EdgeLengthsSqr );
119 LengthsSqr[2] = LengthSquared( Barycentrics0, Barycentrics1, EdgeLengthsSqr );
120
121 float Area = SubtriangleArea( Barycentrics0, Barycentrics1, Barycentrics2, TriangleArea );
122
123 return 0.25f * ( -LengthsSqr.X + LengthsSqr.Y + LengthsSqr.Z ) / Area;
124 }

127FORCEINLINE void SubtriangleBarycentrics( uint32 TriX, uint32 TriY, uint32 FlipTri, uint32 NumSubdivisions, FVector3f Barycentrics[3] )
128{
129 /*
130 Vert order:
131 1 0__1
132 |\ \ |
133 | \ \ | <= flip triangle
134 |__\ \|
135 0 2 2
136 */
137
138 uint32 VertXY[3][2] =
139 {
140 { TriX, TriY },
141 { TriX, TriY + 1},
142 { TriX + 1, TriY },
143 };
144 VertXY[0][1] += FlipTri;
145 VertXY[1][0] += FlipTri;
146
147 for( int Corner = 0; Corner < 3; Corner++ )
148 {
149 Barycentrics[ Corner ][0] = (float)VertXY[ Corner ][0];
150 Barycentrics[ Corner ][1] = (float)VertXY[ Corner ][1];
151 Barycentrics[ Corner ][2] = float(NumSubdivisions - VertXY[ Corner ][0] - VertXY[ Corner ][1]);
152 Barycentrics[ Corner ] /= (float)NumSubdivisions;
153 }
154}

166class FEdgeHash

168public:

171 FEdgeHash( int32 Num )
172 : HashTable( 1 << FMath::FloorLog2( Num ), Num )
173 {}

175 template< typename FGetPosition >
176 void Add_Concurrent( int32 EdgeIndex, FGetPosition&& GetPosition )
177 {
178 const FVector3f& Position0 = GetPosition( EdgeIndex );
179 const FVector3f& Position1 = GetPosition( Cycle3( EdgeIndex ) );
180
181 uint32 Hash0 = HashPosition( Position0 );
182 uint32 Hash1 = HashPosition( Position1 );
183 uint32 Hash = Murmur32( { Hash0, Hash1 } );
184
185 HashTable.Add_Concurrent( Hash, EdgeIndex );
186 }
187
188 template< typename FGetPosition, typename FuncType >
189 void ForAllMatching( int32 EdgeIndex, bool bAdd, FGetPosition&& GetPosition, FuncType&& Function )
190 {
191 const FVector3f& Position0 = GetPosition( EdgeIndex );
192 const FVector3f& Position1 = GetPosition( Cycle3( EdgeIndex ) );
193
194 uint32 Hash0 = HashPosition( Position0 );
195 uint32 Hash1 = HashPosition( Position1 );
196 uint32 Hash = Murmur32( { Hash1, Hash0 } );
197
198 for( uint32 OtherEdgeIndex = HashTable.First( Hash ); HashTable.IsValid( OtherEdgeIndex ); OtherEdgeIndex = HashTable.Next( OtherEdgeIndex ) )
199 {
200 if( Position0 == GetPosition( Cycle3( OtherEdgeIndex ) ) &&
201 Position1 == GetPosition( OtherEdgeIndex ) )
202 {
203 // Found matching edge.
204 Function( EdgeIndex, OtherEdgeIndex );
205 }
206 }
207
208 if( bAdd )
209 HashTable.Add( Murmur32( { Hash0, Hash1 } ), EdgeIndex );
210 }
211};

214struct FAdjacency
215{
216 TArray< int32 > Direct;
217 TMultiMap< int32, int32 > Extended;
218
219 FAdjacency( int32 Num )
220 {
221 Direct.AddUninitialized( Num );
222 }
223
224 void Link( int32 EdgeIndex0, int32 EdgeIndex1 )
225 {
226 if( Direct[ EdgeIndex0 ] < 0 &&
227 Direct[ EdgeIndex1 ] < 0 )
228 {
229 Direct[ EdgeIndex0 ] = EdgeIndex1;
230 Direct[ EdgeIndex1 ] = EdgeIndex0;
231 }
232 else
233 {
234 Extended.AddUnique( EdgeIndex0, EdgeIndex1 );
235 Extended.AddUnique( EdgeIndex1, EdgeIndex0 );
236 }
237 }
238
239 template< typename FuncType >
240 void ForAll( int32 EdgeIndex, FuncType&& Function ) const
241 {
242 int32 AdjIndex = Direct[ EdgeIndex ];
243 if( AdjIndex >= 0 )
244 {
245 Function( EdgeIndex, AdjIndex );
246 }
247
248 for( auto Iter = Extended.CreateConstKeyIterator( EdgeIndex ); Iter; ++Iter )
249 {
250 Function( EdgeIndex, Iter.Value() );
251 }
252 }
253};
