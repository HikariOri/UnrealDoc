65 static EElementType DetermineElementType(const TNotNull<const UStruct*> ElementType)
66 {
67 if (UE::Mass::IsA<FMassFragment>(ElementType))
68 {
69 return EElementType::Fragment;
70 }
71 else if (UE::Mass::IsA<FMassTag>(ElementType))
72 {
73 return EElementType::Tag;

85 return EElementType::ChunkFragment;
