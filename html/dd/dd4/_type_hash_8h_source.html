51[[nodiscard]] inline constexpr uint32 HashCombine(uint32 A, uint32 C)
52{
53 uint32 B = 0x9e3779b9;
54 A += B;
55
56 A -= B; A -= C; A ^= (C>>13);
57 B -= C; B -= A; B ^= (A<<8);
58 C -= A; C -= B; C ^= (B>>13);
59 A -= B; A -= C; A ^= (C>>12);
60 B -= C; B -= A; B ^= (A<<16);
61 C -= A; C -= B; C ^= (B>>5);
62 A -= B; A -= C; A ^= (C>>3);
63 B -= C; B -= A; B ^= (A<<10);
64 C -= A; C -= B; C ^= (B>>15);
65
66 return C;
67}

119//
120// Hash functions for common types.
121//
122// WARNING! GetTypeHash result values are not expected to leave the running process.
123// Do not persist them to disk, send them to another running process or
124// expect them to be consistent across multiple runs.
125//
126
127template <
128 typename ScalarType,
129 std::enable_if_t<std::is_scalar_v<ScalarType> && !std::is_same_v<ScalarType, TCHAR*> && !std::is_same_v<ScalarType, const TCHAR*>>* = nullptr
130>
131[[nodiscard]] inline uint32 GetTypeHash(ScalarType Value)
132{
133 if constexpr (std::is_integral_v<ScalarType>)
134 {
135 if constexpr (sizeof(ScalarType) <= 4)
136 {
137 return Value;
138 }
139 else if constexpr (sizeof(ScalarType) == 8)
140 {
141 return UE::Private::GetTypeHash64(Value);
142 }
143 else if constexpr (sizeof(ScalarType) == 16)
144 {
145 const uint64 Low = (uint64)Value;
146 const uint64 High = (uint64)(Value >> 64);
147 return GetTypeHash(Low) ^ GetTypeHash(High);
148 }
149 else
150 {
151 static_assert(sizeof(ScalarType) == 0, "Unsupported integral type");
152 return 0;
153 }
154 }
155 else if constexpr (std::is_floating_point_v<ScalarType>)
156 {
157 if constexpr (std::is_same_v<ScalarType, float>)
158 {
159 return *(uint32*)&Value;
160 }
161 else if constexpr (std::is_same_v<ScalarType, double>)
162 {
163 return GetTypeHash(*(uint64*)&Value);
164 }
165 else
166 {
167 static_assert(sizeof(ScalarType) == 0, "Unsupported floating point type");
168 return 0;
169 }
170 }
171 else if constexpr (std::is_enum_v<ScalarType>)
172 {
173 return GetTypeHash((__underlying_type(ScalarType))Value);
174 }
175 else if constexpr (std::is_pointer_v<ScalarType>)
176 {
177 // Once the TCHAR* deprecations below are removed, we want to prevent accidental string hashing, so this static_assert should be commented back in
178 //static_assert(!TIsCharType<std::remove_pointer_t<ScalarType>>::Value, "Pointers to string types should use a PointerHash() or FCrc::Stricmp_DEPRECATED() call depending on requirements");
179
180 return PointerHash(Value);
181 }
182 else
183 {
184 static_assert(sizeof(ScalarType) == 0, "Unsupported scalar type");
185 return 0;
186 }
187}

201[[nodiscard]] inline uint32 GetArrayHash(const T* Ptr, uint64 Size, uint32 PreviousHash = 0)
202{
203 uint32 Result = PreviousHash;
204 while (Size)

206 Result = HashCombineFast(Result, GetTypeHash(*Ptr));

209 }
