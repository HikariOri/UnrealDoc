347 void SaveCustomTextFilters(FFilterBarSettings* FilterBarConfig)
348 {
349 /* Go through all the custom text filters (including unchecked ones) to save them, so that the user does not
350 * lose the text filters they created from the menu or a saved search
351 */
352 for (const TSharedRef<ICustomTextFilter<FilterType>>& CustomTextFilter : this->CustomTextFilters)
353 {
354 // Get the actual FFilterBase
355 TSharedRef<FFilterBase<FilterType>> CustomFilter = CustomTextFilter->GetFilter().ToSharedRef();
356
357 // Is the filter "checked", i.e visible in the filter bar
358 bool bIsChecked = this->IsFrontendFilterInUse(CustomFilter);
359
360 // Is the filter "active", i.e visible and enabled in the filter bar
361 bool bIsActive = this->IsFilterActive(CustomFilter);
362
363 // Get the data associated with this filter
364 FCustomTextFilterData FilterData = CustomTextFilter->CreateCustomTextFilterData();
365
366 FCustomTextFilterState FilterState;
367 FilterState.bIsChecked = bIsChecked;
368 FilterState.bIsActive = bIsActive;
369 FilterState.FilterData = FilterData;
370
371 FilterBarConfig->CustomTextFilters.Add(FilterState);
372 }

386 void SaveFilters(FFilterBarSettings* FilterBarConfig)
387 {
388 /* For the remaining (custom and type) filters, go through the currently active (i.e visible in the filter bar)
389 * ones to save their state, since they will be added to the filter bar programmatically every time
390 */
391 for ( const TSharedPtr<SFilter> Filter : this->Filters )
392 {
393 const FString FilterName = Filter->GetFilterName();
394
395 // Ignore custom text filters, since we saved them previously
396 if(FilterName == FCustomTextFilter<FilterType>::GetFilterTypeName().ToString())
397 {
398 continue;
399 }
400 // If it is a FrontendFilter
401 else if ( Filter->GetFrontendFilter().IsValid() )
402 {
403 FilterBarConfig->CustomFilters.Add(FilterName, Filter->IsEnabled());
404 }
405 // Otherwise we assume it is a type filter
406 else
407 {
408 FilterBarConfig->TypeFilters.Add(FilterName, Filter->IsEnabled());
409 }
410 }
411
412 // Add back all the unknown filters which we encountered while loading, but didn't find in our instance
413 FilterBarConfig->TypeFilters.Append(UnknownTypeFilters);
414 }

417 void LoadCustomTextFilters(const FFilterBarSettings* FilterBarConfig)
418 {
419 // Load all the custom text filters
420 for(const FCustomTextFilterState& FilterState : FilterBarConfig->CustomTextFilters)
421 {
422 // Create an ICustomTextFilter using the provided delegate
423 TSharedRef<ICustomTextFilter<FilterType>> NewTextFilter = this->CreateTextFilter.Execute().ToSharedRef();
424
425 // Get the actual FFilterBase
426 TSharedRef<FFilterBase<FilterType>> NewFilter = NewTextFilter->GetFilter().ToSharedRef();
427
428 // Set the internals of the custom text filter from what we have saved
429 NewTextFilter->SetFromCustomTextFilterData(FilterState.FilterData);
430
431 // Add this to our list of custom text filters
432 this->CustomTextFilters.Add(NewTextFilter);
433
434 // If the filter was checked previously, add it to the filter bar
435 if(FilterState.bIsChecked)
436 {
437 TSharedRef<SFilter> AddedFilter = this->AddFilterToBar(NewFilter);
438
439 // Set the filter as active if it was previously
440 AddedFilter->SetEnabled(FilterState.bIsActive, false);
441
442 this->SetFrontendFilterActive(NewFilter, FilterState.bIsActive);
443 }
444 }

448 void LoadFilters(const FFilterBarSettings* FilterBarConfig)
449 {
450 // Load all the custom filters (i.e FrontendFilters)
451 TMap<FString, TSharedRef<FFilterBase<FilterType>>> CustomFilters;
452 CustomFilters.Reserve(FilterBarConfig->CustomFilters.Num());
453 for (const TSharedRef<FFilterBase<FilterType>>& FrontendFilter : this->AllFrontendFilters)
454 {
455 if (!this->IsFrontendFilterInUse(FrontendFilter))
456 {
457 // Try to find this type filter in our list of saved filters
458 const FString& FilterName = FrontendFilter->GetName();
459 if (FilterBarConfig->CustomFilters.Contains(FilterName))
460 {
461 CustomFilters.Add(FilterName, FrontendFilter);
462 }
463 }
464 }
465
466 // We add them here to preserve the saving config order
467 for (const TPair<FString, bool>& CustomFilter : FilterBarConfig->CustomFilters)
468 {
469 if (const TSharedRef<FFilterBase<FilterType>>* Filter = CustomFilters.Find(CustomFilter.Key))
470 {
471 TSharedRef<SFilter> NewFilter = this->AddFilterToBar(*Filter);
472 NewFilter->SetEnabled(CustomFilter.Value, false);
473 this->SetFrontendFilterActive(*Filter, NewFilter->IsEnabled());
474 }
475 }
476
477 // Save a copy of all the type filters, and then remove anything we find
478 UnknownTypeFilters = FilterBarConfig->TypeFilters;
479
480 // Load all the type filters
481 TMap<FString, TSharedRef<FCustomClassFilterData>> AssetClassFilters;
482 AssetClassFilters.Reserve(FilterBarConfig->TypeFilters.Num());
483 for(const TSharedRef<FCustomClassFilterData> &CustomClassFilter : this->CustomClassFilters)
484 {
485 const FString FilterName = CustomClassFilter->GetFilterName();
486
487 if(!this->IsClassTypeInUse(CustomClassFilter))
488 {
489 // Try to find this type filter in our list of saved filters
490 if (FilterBarConfig->TypeFilters.Contains(FilterName))
491 {
492 AssetClassFilters.Add(FilterName, CustomClassFilter);
493
494 // Remove it from the Unknown list, we have encountered this type filter in the current instance
495 UnknownTypeFilters.Remove(FilterName);
496 }
497 }
498 else
499 {
500 // If the filter is already in use, we don't want to add a duplicate but it isn't unknown - so remove it from the UnknownTypeFilters
501 UnknownTypeFilters.Remove(FilterName);
502 }
503 }
504
505 // We add them here to preserve the saving config order
506 for (const TPair<FString, bool>& TypeFilter : FilterBarConfig->TypeFilters)
507 {
508 if (const TSharedRef<FCustomClassFilterData>* Filter = AssetClassFilters.Find(TypeFilter.Key))
509 {
510 TSharedRef<SFilter> NewFilter = this->AddAssetFilterToBar(*Filter);
511 NewFilter->SetEnabled(TypeFilter.Value, false);
512 }
513 }

655 FARFilter GetCombinedBackendFilter() const
656 {
657 FARFilter CombinedFilter;
658
659 // Add all selected filters
660 for (int32 FilterIdx = 0; FilterIdx < AssetFilters.Num(); ++FilterIdx)
661 {
662 const TSharedPtr<SAssetFilter> AssetFilter = AssetFilters[FilterIdx];
663
664 if ( AssetFilter.IsValid() && AssetFilter->IsEnabled())
665 {
666 CombinedFilter.Append(AssetFilter->GetBackendFilter());
667 }
668 }
669
670 // HACK: A blueprint can be shown as Blueprint or as BlueprintGeneratedClass, but we don't want to distinguish them while filtering.
671 // This should be removed, once all blueprints are shown as BlueprintGeneratedClass.
672 // Note: Adding the BlueprintGeneratedClass should occur before the below bRecursiveClasses check otherwise it will be added to the ExclusionSet
673 if (CombinedFilter.ClassPaths.Contains(FTopLevelAssetPath(TEXT("/Script/Engine"), TEXT("Blueprint"))))
674 {
675 CombinedFilter.ClassPaths.AddUnique(FTopLevelAssetPath(TEXT("/Script/Engine"), TEXT("BlueprintGeneratedClass")));
676 }
677
678 if ( CombinedFilter.bRecursiveClasses )
679 {
680 // Add exclusions for AssetTypeActions NOT in the filter.
681 // This will prevent assets from showing up that are both derived from an asset in the filter set and derived from an asset not in the filter set
682 // Get the list of all asset type actions
683 for(const TSharedRef<FCustomClassFilterData> &CustomClassFilter : CustomClassFilters)
684 {
685 const UClass* TypeClass = CustomClassFilter->GetClass();
686 if (TypeClass && !CombinedFilter.ClassPaths.Contains(TypeClass->GetClassPathName()))
687 {
688 CombinedFilter.RecursiveClassPathsExclusionSet.Add(TypeClass->GetClassPathName());
689 }
690 }
691 }
692
693 return CombinedFilter;
