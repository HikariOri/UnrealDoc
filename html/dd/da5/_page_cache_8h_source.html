132 template<typename T>
133 size_t Commit(void *InAddr, size_t Size, size_t& OutUnPending, T&& CommitFunction)
134 {
135 check(Size > 0 && IsAligned(Size, PageSize));
136 uint32 StartPage = AddrToPageIndex(InAddr);
137 uint32 EndPage = StartPage + Size / PageSize;
138
139 uint32 LastCommitPage = MAX_uint32;
140 uint32 StartCommitPage = MAX_uint32;
141 UE::TUniqueLock Lock(Mutex);
142 // this loop could be accelerated by using the hierarchical info in the bit tree
143 uint32 NumFound = 0;
144 uint32 NumUnPending = 0;
145 for (uint32 Index = StartPage; Index < EndPage; Index++)
146 {
147 if (CurrentlyCommitted.IsAllocated(Index))
148 {
149 if (!NotPendingDecommit.IsAllocated(Index))
150 {
151 check(PendingDecommittedPages);
152 PendingDecommittedPages--;
153 NumUnPending++;
154 NotPendingDecommit.AllocBit(Index);
155 }
156 }
157 else
158 {
159 NumFound++;
160 CommittedPages++;
161 check(DecommittedPages);
162 DecommittedPages--;
163 check(NotPendingDecommit.IsAllocated(Index));
164 CurrentlyCommitted.AllocBit(Index);
165 if (StartCommitPage == MAX_uint32 || LastCommitPage + 1 != Index)
166 {
167 if (StartCommitPage != MAX_uint32)
168 {
169 CommitFunction(PageIndexToAddr(StartCommitPage), (1 + LastCommitPage - StartCommitPage) * PageSize);
170 }
171 StartCommitPage = Index;
172 }
173 LastCommitPage = Index;
174 }
175 }
176 if (StartCommitPage != MAX_uint32)
177 {
178 CommitFunction(PageIndexToAddr(StartCommitPage), (1 + LastCommitPage - StartCommitPage) * PageSize);
179 }
180 CommitHits += NumUnPending;
181 CommitMisses += NumFound;
182 OutUnPending = size_t(NumUnPending) * PageSize;
183 return size_t(NumFound) * PageSize;
184 }

232 template<typename T>
233 size_t DecommitPending(size_t Size, T&& DecommitFunction)
234 {
235 if (!NumPages)
236 {
237 return 0; // this page cache was never really set up, nothing to sweep
238 }
239 check(Size > 0 && IsAligned(Size, PageSize));
240 uint32 NumNeed = Size / PageSize;
241
242 uint32 LastDecommitPage = MAX_uint32;
243 uint32 StartDecommitPage = MAX_uint32;
244 uint32 NumFound = 0;
245 UE::TUniqueLock Lock(Mutex);
246
247 while (NumFound < NumNeed)
248 {
249 check(SweepPage < NumPages);
250 uint32 Index = NotPendingDecommit.NextAllocBit(SweepPage);
251 if (Index == MAX_uint32)
252 {
253 SweepPage = 0;
254 break;
255 }
256 check(CurrentlyCommitted.IsAllocated(Index) && !NotPendingDecommit.IsAllocated(Index));
257 check(CommittedPages);
258 CommittedPages--;
259 DecommittedPages++;
260 check(PendingDecommittedPages);
261 PendingDecommittedPages--;
262 NumFound++;
263 CurrentlyCommitted.FreeBit(Index);
264 NotPendingDecommit.AllocBit(Index);
265 if (StartDecommitPage == MAX_uint32 || LastDecommitPage + 1 != Index)
266 {
267 if (StartDecommitPage != MAX_uint32)
268 {
269 DecommitFunction(PageIndexToAddr(StartDecommitPage), (1 + LastDecommitPage - StartDecommitPage) * PageSize);
270 }
271 StartDecommitPage = Index;
272 }
273 LastDecommitPage = Index;
274 SweepPage = Index + 1;
275 if (SweepPage >= NumPages)
276 {
277 SweepPage = 0;
278 break;
279 }
280 }
281 if (StartDecommitPage != MAX_uint32)
282 {
283 DecommitFunction(PageIndexToAddr(StartDecommitPage), (1 + LastDecommitPage - StartDecommitPage) * PageSize);
284 }
285 return size_t(NumFound) * PageSize;
286 }
287 template<typename T>
288 size_t TryDecommitPending(size_t Size, T&& DecommitFunction)
289 {
290 if (!NumPages)
291 {
292 return 0; // this page cache was never really set up, nothing to sweep
293 }
294 check(Size > 0 && IsAligned(Size, PageSize));
295 uint32 NumNeed = Size / PageSize;
296
297 uint32 LastDecommitPage = MAX_uint32;
298 uint32 StartDecommitPage = MAX_uint32;
299 uint32 NumFound = 0;
300 UE::TUniqueLock Lock(Mutex);
301
302 auto DecommitRange = [&]()
303 {
304 uint32 DecommitPageCount = 1 + LastDecommitPage - StartDecommitPage;
305
306 if (DecommitFunction(PageIndexToAddr(StartDecommitPage), DecommitPageCount * PageSize))
307 {
308 CommittedPages -= DecommitPageCount;
309 DecommittedPages += DecommitPageCount;
310 check(PendingDecommittedPages >= DecommitPageCount);
311 PendingDecommittedPages -= DecommitPageCount;
312 NumFound += DecommitPageCount;
313
314 uint32 EndBitIndex = StartDecommitPage + DecommitPageCount;
315 for (uint32 BitIndex = StartDecommitPage; BitIndex < EndBitIndex; BitIndex++)
316 {
317 CurrentlyCommitted.FreeBit(BitIndex);
318 NotPendingDecommit.AllocBit(BitIndex);
319 }
320 }
321 };
322 while (NumFound < NumNeed)
323 {
324 check(SweepPage < NumPages);
325 uint32 Index = NotPendingDecommit.NextAllocBit(SweepPage);
326 if (Index == MAX_uint32)
327 {
328 SweepPage = 0;
329 break;
330 }
331 check(CurrentlyCommitted.IsAllocated(Index) && !NotPendingDecommit.IsAllocated(Index));
332 check(CommittedPages);
333
334 if (StartDecommitPage == MAX_uint32 || LastDecommitPage + 1 != Index)
335 {
336 if (StartDecommitPage != MAX_uint32)
337 {
338 DecommitRange();
339 }
340 StartDecommitPage = Index;
341 }
342 LastDecommitPage = Index;
343 SweepPage = Index + 1;
344 if (SweepPage >= NumPages)
345 {
346 SweepPage = 0;
347 break;
348 }
349 }
350 if (StartDecommitPage != MAX_uint32)
351 {
352 DecommitRange();
353 }
354 return size_t(NumFound) * PageSize;
355 }
