49 template<typename RayTracingShaderType, typename ShaderElementDataType>
50 void BuildRayTracingMeshCommands(
51 const FMeshBatch& RESTRICT MeshBatch,
52 uint64 BatchElementMask,
53 const FPrimitiveSceneProxy* RESTRICT PrimitiveSceneProxy,
54 const FMaterialRenderProxy& RESTRICT MaterialRenderProxy,
55 const FMaterial& RESTRICT MaterialResource,
56 const TShaderRef<RayTracingShaderType>& RayTracingShader,
57 const ShaderElementDataType& ShaderElementData)
58 {
59 const FVertexFactory* RESTRICT VertexFactory = MeshBatch.VertexFactory;
60
61 FRayTracingMeshCommand SharedCommand;
62
63 SetupRayTracingMeshCommandMaskAndStatus(SharedCommand, MeshBatch, PrimitiveSceneProxy, MaterialResource, RayTracingType);
64
65 if (GRHISupportsRayTracingShaders)
66 {
67 SharedCommand.SetShader(RayTracingShader);
68 }
69
70 FVertexInputStreamArray VertexStreams;
71 VertexFactory->GetStreams(FeatureLevel, EVertexInputStreamType::Default, VertexStreams);
72
73 if (RayTracingShader.IsValid())
74 {
75 FMeshDrawSingleShaderBindings ShaderBindings = SharedCommand.ShaderBindings.GetSingleShaderBindings(SF_RayHitGroup);
76 RayTracingShader->GetShaderBindings(Scene, FeatureLevel, PrimitiveSceneProxy, MaterialRenderProxy, MaterialResource, ShaderElementData, ShaderBindings);
77 }
78
79 const int32 NumElements = MeshBatch.Elements.Num();
80
81 for (int32 BatchElementIndex = 0; BatchElementIndex < NumElements; BatchElementIndex++)
82 {
83 if ((1ull << BatchElementIndex) & BatchElementMask)
84 {
85 const FMeshBatchElement& BatchElement = MeshBatch.Elements[BatchElementIndex];
86 FRayTracingMeshCommand& RayTracingMeshCommand = CommandContext->AddCommand(SharedCommand);
87
88 if (RayTracingShader.IsValid())
89 {
90 FMeshDrawSingleShaderBindings RayHitGroupShaderBindings = RayTracingMeshCommand.ShaderBindings.GetSingleShaderBindings(SF_RayHitGroup);
91 FMeshMaterialShader::GetElementShaderBindings(RayTracingShader, Scene, ViewIfDynamicMeshCommand, VertexFactory, EVertexInputStreamType::Default, FeatureLevel, PrimitiveSceneProxy, MeshBatch, BatchElement, ShaderElementData, RayHitGroupShaderBindings, VertexStreams);
92 }
93
94 // Command can only be cached if no global/static uniform buffers are used - if all platforms use SBTLayout for all RT shaders then this could be a check
95 RayTracingMeshCommand.bCanBeCached = !RayTracingMeshCommand.HasGlobalUniformBufferBindings();
96
97 RayTracingMeshCommand.GeometrySegmentIndex = uint32(MeshBatch.SegmentIndex) + BatchElementIndex;
98 RayTracingMeshCommand.bIsTranslucent = MeshBatch.IsTranslucent(MaterialResource.GetFeatureLevel());
99 CommandContext->FinalizeCommand(RayTracingMeshCommand);
100 }
101 }
102 }

265 FRayTracingLocalShaderBindings* AllocateInternal()
266 {
267 if (!CurrentChunk || CurrentChunk->Num == FChunk::MaxNum)
268 {
269 FChunk* OldChunk = CurrentChunk;
270
271 static_assert(std::is_trivially_destructible_v<FChunk>, "Chunk must be trivially destructible, as no destructor will be called.");
272 CurrentChunk = (FChunk*)ParameterMemory.Alloc(sizeof(FChunk), alignof(FChunk));
273 CurrentChunk->Next = nullptr;
274 CurrentChunk->Num = 0;
275
276 if (FirstChunk == nullptr)
277 {
278 FirstChunk = CurrentChunk;
279 }
280
281 if (OldChunk)
282 {
283 OldChunk->Next = CurrentChunk;
284 }
285 }
286
287 FRayTracingLocalShaderBindings* ResultMemory = &CurrentChunk->Bindings[CurrentChunk->Num++];
288 return new(ResultMemory) FRayTracingLocalShaderBindings;
289 }

292template <typename FunctionType>
293void AddRayTracingLocalShaderBindingWriterTasks(
294 FRDGBuilder& GraphBuilder,
295 TConstArrayView<FRayTracingShaderBindingData> DirtyPersistentRayTracingShaderBindings,
296 TArray<FRayTracingLocalShaderBindingWriter*, SceneRenderingAllocator>& ShaderBindingWriters,
297 FunctionType SetupBindingsFunction)
298{
299 const uint32 NumTotalDirtyBindings = DirtyPersistentRayTracingShaderBindings.Num();
300 const uint32 TargetBindingsPerTask = 1024;
301 const uint32 NumTasks = FMath::Max(1u, FMath::DivideAndRoundUp(NumTotalDirtyBindings, TargetBindingsPerTask));
302 const uint32 BindingsPerTask = FMath::DivideAndRoundUp(NumTotalDirtyBindings, NumTasks); // Evenly divide commands between tasks (avoiding potential short last task)
303
304 ShaderBindingWriters.SetNum(NumTasks);
305
306 for (uint32 TaskIndex = 0; TaskIndex < NumTasks; ++TaskIndex)
307 {
308 const uint32 FirstTaskBindingIndex = TaskIndex * BindingsPerTask;
309 const FRayTracingShaderBindingData* RTShaderBindings = DirtyPersistentRayTracingShaderBindings.GetData() + FirstTaskBindingIndex;
310 const uint32 NumBindings = FMath::Min(BindingsPerTask, NumTotalDirtyBindings - FirstTaskBindingIndex);
311
312 FRayTracingLocalShaderBindingWriter* BindingWriter = new FRayTracingLocalShaderBindingWriter();
313 ShaderBindingWriters[TaskIndex] = BindingWriter;
314
315 GraphBuilder.AddSetupTask(
316 [BindingWriter, RTShaderBindings, NumBindings, SetupBindingsFunction]()
317 {
318 TRACE_CPUPROFILER_EVENT_SCOPE(BuildRayTracingMaterialBindingsTask);
319
320 for (uint32 BindingIndex = 0; BindingIndex < NumBindings; ++BindingIndex)
321 {
322 const FRayTracingShaderBindingData& RTShaderBindingData = RTShaderBindings[BindingIndex];
323 SetupBindingsFunction(RTShaderBindingData, BindingWriter);
324 }
325 });
326 }
327}
