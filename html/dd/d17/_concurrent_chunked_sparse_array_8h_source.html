23template <typename ElementType, uint32 ElementsPerChunk>
24class TConcurrentChunkedSparseArray
25{
26 static_assert(sizeof(ElementType) >= sizeof(uint32), "ElementType is too small to support the element free list.");
27 static_assert(FMath::IsPowerOfTwo(ElementsPerChunk), "ElementsPerChunk must be a power of two to support efficient index calculations.");
28 // ChunkArray is approximately 16 bytes per chunk, which makes overhead per element is 1 byte at 16 elements per chunk
29 // and 2/4/8/16 bytes per element at 8/4/2/1 elements per chunk. Require at least 16 to avoid massive overhead.
30 static_assert(ElementsPerChunk >= 16, "ElementsPerChunk is too low for ChunkArray to be small relative to the element data.");
31
32 // Not implemented but can be if necessary.
33 TConcurrentChunkedSparseArray(const TConcurrentChunkedSparseArray&) = delete;
34 TConcurrentChunkedSparseArray& operator=(const TConcurrentChunkedSparseArray&) = delete;

39 ~TConcurrentChunkedSparseArray()
40 {
41 FChunkArray* Array = ChunkArray.load(std::memory_order_acquire);
42 if (Array == &EmptyChunkArray)
43 {
44 return;
45 }
46
47 // Walk the free list to measure its size.
48 uint32 FreeElementCount = 0;
49 for (uint32 Index = uint32(FreeElementIndexAndCounter.load(std::memory_order_relaxed) >> 32); Index != MAX_uint32;)
50 {
51 ++FreeElementCount;
52 const FChunk* Chunk = Array->Chunks[Index / ElementsPerChunk].load(std::memory_order_acquire);
53 Index = reinterpret_cast<const uint32&>(Chunk->Elements[Index % ElementsPerChunk]);
54 }
55
56 // Walk the free list to copy it into an array.
57 TArray<uint32> FreeElements;
58 FreeElements.Reserve(FreeElementCount);
59 for (uint32 Index = uint32(FreeElementIndexAndCounter.load(std::memory_order_relaxed) >> 32); Index != MAX_uint32;)
60 {
61 FreeElements.Add(Index);
62 const FChunk* Chunk = Array->Chunks[Index / ElementsPerChunk].load(std::memory_order_acquire);
63 Index = reinterpret_cast<const uint32&>(Chunk->Elements[Index % ElementsPerChunk]);
64 }
65 FreeElements.Sort();
66
67 // Destroy elements that are not in the free list.
68 uint32 LastElementIndex = NextElementIndex.load(std::memory_order_relaxed);
69 typename TArray<uint32>::TConstIterator FreeIt = FreeElements.CreateConstIterator();
70 for (uint32 ChunkIndex = 0, ChunkCount = Array->ChunkCount; ChunkIndex < ChunkCount; ++ChunkIndex)
71 {
72 const uint32 ElementsInChunk = FMath::Min(LastElementIndex, ElementsPerChunk);
73 if (FChunk* Chunk = Array->Chunks[ChunkIndex].load(std::memory_order_acquire))
74 {
75 for (uint32 IndexInChunk = 0; IndexInChunk < ElementsInChunk; ++IndexInChunk)
76 {
77 if (FreeIt && *FreeIt == ChunkIndex * ElementsPerChunk + IndexInChunk)
78 {
79 ++FreeIt;
80 continue;
81 }
82 Chunk->Elements[IndexInChunk].DestroyUnchecked();
83 }
84 delete Chunk;
85 }
86 LastElementIndex -= ElementsInChunk;
87 }
88
89 // Destroy the chain of chunks arrays.
90 while (Array != &EmptyChunkArray)
91 {
92 FChunkArray* PrevArray = Array->Prev;
93 FMemory::Free(Array);
94 Array = PrevArray;
95 }
96 }

98 [[nodiscard]] inline ElementType& operator[](const uint32 ElementIndex) UE_LIFETIMEBOUND

104 const uint32 IndexInChunk = ElementIndex % ElementsPerChunk;
105 FChunk* Chunk = Array->Chunks[ChunkIndex].load(std::memory_order_acquire);
106 checkSlow(Chunk);
107 return Chunk->Elements[IndexInChunk].GetUnchecked();
108 }

131 [[nodiscard]] inline uint32 AddUninitialized(const uint32 ElementCount)
132 {
133 if (ElementCount == 0)
134 {
135 return NextElementIndex.load(std::memory_order_relaxed);
136 }
137 if (ElementCount == 1)
138 {
139 return AddUninitialized();
140 }
141
142 const uint32 FirstElementIndex = NextElementIndex.fetch_add(ElementCount, std::memory_order_relaxed);
143 const uint32 LastElementIndex = FirstElementIndex + ElementCount - 1;
144 FChunkArray* Array = GetOrGrowChunkArray(LastElementIndex);
145 const uint32 FirstChunkIndex = FirstElementIndex / ElementsPerChunk;
146 const uint32 LastChunkIndex = LastElementIndex / ElementsPerChunk;
147 for (uint32 ChunkIndex = FirstChunkIndex; ChunkIndex <= LastChunkIndex; ++ChunkIndex)
148 {
149 GetOrCreateChunk(Array, ChunkIndex);
150 }
151 return FirstElementIndex;
152 }

154 inline void RemoveAt(uint32 ElementIndex)
155 {
156 FChunkArray* Array = ChunkArray.load(std::memory_order_acquire);
157 const uint32 ChunkIndex = ElementIndex / ElementsPerChunk;
158 checkfSlow(Array && ChunkIndex < Array->ChunkCount && ElementIndex < NextElementIndex.load(std::memory_order_relaxed),
159 TEXT("Array index out of bounds: %u into an array of size %u."), ElementIndex, NextElementIndex.load(std::memory_order_relaxed));
160 const uint32 IndexInChunk = ElementIndex % ElementsPerChunk;
161 FChunk* Chunk = Array->Chunks[ChunkIndex].load(std::memory_order_acquire);
162 checkSlow(Chunk);
163 TTypeCompatibleBytes<ElementType>& Element = Chunk->Elements[IndexInChunk];
164 Element.DestroyUnchecked();
165
166 uint32& NextFreeElementIndex = reinterpret_cast<uint32&>(Element);
167 uint64 IndexAndCounter = FreeElementIndexAndCounter.load(std::memory_order_relaxed);
168 for (;;)
169 {
170 NextFreeElementIndex = uint32(IndexAndCounter >> 32);
171 const uint32 Counter = uint32(IndexAndCounter) + 1;
172 const uint64 NextIndexAndCounter = (uint64(ElementIndex) << 32) | Counter;
173 if (FreeElementIndexAndCounter.compare_exchange_strong(IndexAndCounter, NextIndexAndCounter, std::memory_order_release, std::memory_order_relaxed))
174 {
175 return;
176 }
177 }
178 }

205 FORCEINLINE uint32 AcquireElementIndex()
206 {
207 // Acquire an index from the free list or from the end of the array.
208 uint64 IndexAndCounter = FreeElementIndexAndCounter.load(std::memory_order_acquire);
209 for (;;)
210 {
211 uint32 ElementIndex = uint32(IndexAndCounter >> 32);
212
213 if (LIKELY(ElementIndex == MAX_uint32))
214 {
215 // Free list is empty. Return the next index at the end of the array.
216 return NextElementIndex.fetch_add(1, std::memory_order_relaxed);
217 }
218
219 // Attempt to pop the index from the free list.
220 const uint32 Counter = uint32(IndexAndCounter) + 1;
221 const FChunkArray* Array = ChunkArray.load(std::memory_order_acquire);
222 const FChunk* Chunk = Array->Chunks[ElementIndex / ElementsPerChunk].load(std::memory_order_acquire);
223 const uint32 NextFreeElementIndex = reinterpret_cast<const uint32&>(Chunk->Elements[ElementIndex % ElementsPerChunk]);
224 const uint64 NextIndexAndCounter = (uint64(NextFreeElementIndex) << 32) | Counter;
225 if (FreeElementIndexAndCounter.compare_exchange_strong(IndexAndCounter, NextIndexAndCounter, std::memory_order_acquire))
226 {
227 return ElementIndex;
228 }
229 }
230 }

232 FORCEINLINE FChunkArray* GetOrGrowChunkArray(const uint32 ElementIndex)
233 {
234 FChunkArray* Array = ChunkArray.load(std::memory_order_acquire);
235 if (ElementIndex < Array->ChunkCount * ElementsPerChunk)
236 {
237 return Array;
238 }
239 return GrowChunkArray(ElementIndex, Array);
240 }

242 FORCENOINLINE FChunkArray* GrowChunkArray(const uint32 ElementIndex, FChunkArray* Array)
243 {
244 do
245 {
246 // Grow to the next power of two past the index.
247 // Allocate a minimum of 16 chunks to avoid the overhead of frequent reallocations at small sizes.
248 const uint32 RequiredChunkCount = FMath::DivideAndRoundUp(ElementIndex + 1, ElementsPerChunk);
249 const uint32 NextChunkCount = FMath::Max(FMath::RoundUpToPowerOfTwo(RequiredChunkCount), 16u);
250 const uint32 NextArraySize = sizeof(FChunkArray) + sizeof(std::atomic<FChunk*>) * NextChunkCount;
251 FChunkArray* NextArray = new(FMemory::MallocZeroed(NextArraySize, alignof(FChunkArray))) FChunkArray;
252 NextArray->ChunkCount = NextChunkCount;
253 NextArray->Prev = Array;
254 for (uint32 ChunkIndex = 0; ChunkIndex < Array->ChunkCount; ++ChunkIndex)
255 {
256 // Force creation of every chunk in the existing array before the new array becomes visible
257 // to avoid the possibility of two threads creating the same chunk in two different arrays.
258 NextArray->Chunks[ChunkIndex].store(GetOrCreateChunk(Array, ChunkIndex), std::memory_order_relaxed);
259 }
260 if (ChunkArray.compare_exchange_strong(Array, NextArray, std::memory_order_release, std::memory_order_acquire))
261 {
262 return NextArray;
263 }
264 FMemory::Free(NextArray);
265 }
266 while (ElementIndex >= Array->ChunkCount * ElementsPerChunk);
267 return Array;
268 }

270 FORCEINLINE static FChunk* GetOrCreateChunk(FChunkArray* Array, uint32 ChunkIndex)
271 {
272 if (FChunk* Chunk = Array->Chunks[ChunkIndex].load(std::memory_order_acquire))
273 {
274 return Chunk;
275 }
276 return CreateChunk(Array, ChunkIndex);
277 }

279 FORCENOINLINE static FChunk* CreateChunk(FChunkArray* Array, uint32 ChunkIndex)
280 {
281 FChunk* ExistingChunk = nullptr;
282 FChunk* NewChunk = new FChunk;
283 if (Array->Chunks[ChunkIndex].compare_exchange_strong(ExistingChunk, NewChunk, std::memory_order_release, std::memory_order_acquire))
284 {
285 return NewChunk;
286 }
287
288 // Try to assign the chunk later in the array to minimize future conflicts and avoid wasting time reallocating.
289 for (uint32 NextIndex = ChunkIndex + 1, ChunkCount = Array->ChunkCount; NextIndex < ChunkCount; ++NextIndex)
290 {
291 // Use relaxed on failure because this does not read the existing chunk returned in the failure case.
292 FChunk* Chunk = nullptr;
293 if (Array->Chunks[NextIndex].compare_exchange_strong(Chunk, NewChunk, std::memory_order_release, std::memory_order_relaxed))
294 {
295 return ExistingChunk;
296 }
297 }
298
299 delete NewChunk;
300 return ExistingChunk;
301 }
302
303 // A shared empty chunk array that avoids requiring null checks to handle empty arrays.
304 inline static FChunkArray EmptyChunkArray;
305
306 // Pointer to the current chunk array.
307 std::atomic<FChunkArray*> ChunkArray = &EmptyChunkArray;
308 // Index of the first free element (or MAX_uint32) in the high 32 bits.
309 // Counter to avoid the ABA problem in the low 32 bits.
310 std::atomic<uint64> FreeElementIndexAndCounter = uint64(MAX_uint32) << 32;
311 // Index of the next element allocated at the end of the array.
312 // Used when there are no free elements or when allocating multiple elements at once.
313 std::atomic<uint32> NextElementIndex = 0;
314};
