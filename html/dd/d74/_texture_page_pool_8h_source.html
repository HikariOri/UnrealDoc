189 void RemoveMappingFromList(uint32 Index)
190 {
191 FPageMapping& Mapping = PageMapping[Index];
192 PageMapping[Mapping.PrevIndex].NextIndex = Mapping.NextIndex;
193 PageMapping[Mapping.NextIndex].PrevIndex = Mapping.PrevIndex;

197 void AddMappingToList(uint32 HeadIndex, uint32 Index)
198 {
199 FPageMapping& Head = PageMapping[HeadIndex];
200 FPageMapping& Mapping = PageMapping[Index];
201 check(Index > NumPages); // make sure we're not trying to add a list head to another list
202 check(Index <= PAGE_MAPPING_CAPACITY);
203
204 // make sure we're not currently in any list
205 check(Mapping.NextIndex == Index);
206 check(Mapping.PrevIndex == Index);
207
208 Mapping.NextIndex = HeadIndex;
209 Mapping.PrevIndex = Head.PrevIndex;
210 PageMapping[Head.PrevIndex].NextIndex = Index;
211 Head.PrevIndex = Index;
212 }
213
214 uint32 AcquireMapping()
215 {
216 const uint32 FreeHeadIndex = NumPages;
217 FPageMapping& FreeHead = PageMapping[FreeHeadIndex];
218 uint32 Index = FreeHead.NextIndex;
219 if (Index != FreeHeadIndex)
220 {
221 RemoveMappingFromList(Index);
222 return Index;
223 }
224
225 Index = PageMapping.AddDefaulted();
226 check(Index <= PAGE_MAPPING_CAPACITY);
227 FPageMapping& Mapping = PageMapping[Index];
228 Mapping.NextIndex = Mapping.PrevIndex = Index;
229 return Index;
230 }
231
232 void ReleaseMapping(uint32 Index)
233 {
234 const uint32 FreeHeadIndex = NumPages;
235 RemoveMappingFromList(Index);
236 AddMappingToList(FreeHeadIndex, Index);
237 }
