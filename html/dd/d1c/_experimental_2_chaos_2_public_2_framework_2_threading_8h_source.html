328 FPhysicsSceneGuard()
329 {
330 TlsSlot = FPlatformTLS::AllocTlsSlot();
331 CurrentWriterThreadId.Store(0);

334 ~FPhysicsSceneGuard()
335 {
336 if(FPlatformTLS::IsValidTlsSlot(TlsSlot))
337 {
338 // Validate the lock as it shuts down
339#if CHAOS_CHECKED

351 void ReadLock()
352 {
353 const FSceneLockTls ThreadData = ModifyTls([](FSceneLockTls& ThreadDataInner) {ThreadDataInner.ReadDepth++; });
354
355 const uint32 ThisThreadId = FPlatformTLS::GetCurrentThreadId();
356
357 // If we're already writing then don't attempt the lock, we already have exclusive access
358 if(CurrentWriterThreadId.Load() != ThisThreadId && ThreadData.ReadDepth == 1)
359 {
360 InnerLock.ReadLock();
361 }
362
363#if PHYSICS_THREAD_CONTEXT
364 //read lock means we can access game thread data, so set the right context
365 FPhysicsThreadContext::Get().IncGameThreadContext();
366#endif

369 void WriteLock()
370 {
371 ModifyTls([](FSceneLockTls& ThreadDataInner) {ThreadDataInner.WriteDepth++; });
372
373 const uint32 ThisThreadId = FPlatformTLS::GetCurrentThreadId();
374
375 if(CurrentWriterThreadId.Load() != ThisThreadId)
376 {
377 InnerLock.WriteLock();
378 CurrentWriterThreadId.Store(ThisThreadId);
379 }
380
381#if PHYSICS_THREAD_CONTEXT
382 //write lock means we can access game thread data, so set the right context
383 FPhysicsThreadContext::Get().IncGameThreadContext();
384#endif
385 }

387 void ReadUnlock()
388 {
389 const FSceneLockTls ThreadData = ModifyTls([](FSceneLockTls& ThreadDataInner)
390 {
391 if(ThreadDataInner.ReadDepth > 0)
392 {
393 ThreadDataInner.ReadDepth--;
394 }
395 else
396 {
397#if CHAOS_CHECKED
398 ensureMsgf(false, TEXT("ReadUnlock called on physics scene guard when the thread does not hold the lock"));
399#else
400 UE_LOGF(LogChaos, Warning, "ReadUnlock called on physics scene guard when the thread does not hold the lock")
401#endif
402 }
403
404 });
405
406 const uint32 ThisThreadId = FPlatformTLS::GetCurrentThreadId();
407
408 if(CurrentWriterThreadId.Load() != ThisThreadId && ThreadData.ReadDepth == 0)
409 {
410 InnerLock.ReadUnlock();
411 }
412
413#if PHYSICS_THREAD_CONTEXT
414 //read lock is released, the gamethread context is gone
415 FPhysicsThreadContext::Get().DecGameThreadContext();
416#endif
417 }
418
419 void WriteUnlock()
420 {
421 const uint32 ThisThreadId = FPlatformTLS::GetCurrentThreadId();
422
423 if(CurrentWriterThreadId.Load() == ThisThreadId)
424 {
425 const FSceneLockTls ThreadData = ModifyTls([](FSceneLockTls& ThreadDataInner) {ThreadDataInner.WriteDepth--; });
426
427 if(ThreadData.WriteDepth == 0)
428 {
429 CurrentWriterThreadId.Store(0);
430 InnerLock.WriteUnlock();
431 }
432 }
433 else
434 {
435#if CHAOS_CHECKED
436 ensureMsgf(false, TEXT("WriteUnlock called on physics scene guard when the thread does not hold the lock"));
437#else
438 UE_LOGF(LogChaos, Warning, "ReadUnlock called on physics scene guard when the thread does not hold the lock")
439#endif
440 }
441
442#if PHYSICS_THREAD_CONTEXT
443 //write lock is released, the gamethread context is gone
444 FPhysicsThreadContext::Get().DecGameThreadContext();
445#endif

450 // We use 32 bits to store our depths (16 read and 16 write) allowing a maximum
451 // recursive lock of depth 65,536. This unions to whatever the platform ptr size
452 // is so we can store this directly into TLS without allocating more storage
453 class FSceneLockTls
454 {
455 public:
456
457 FSceneLockTls()
458 : WriteDepth(0)
459 , ReadDepth(0)
460 {}

463 {
464 struct
465 {
466 uint16 WriteDepth;
467 uint16 ReadDepth;
468 };

474 // Helper for modifying the current TLS data
475 template<typename CallableType>
476 const FSceneLockTls ModifyTls(CallableType Callable)
477 {
478 checkSlow(FPlatformTLS::IsValidTlsSlot(TlsSlot));
479
480 void* ThreadData = FPlatformTLS::GetTlsValue(TlsSlot);
481
482 FSceneLockTls TlsData;
483 TlsData.PtrDummy = ThreadData;
484
485 Callable(TlsData);
486
487 FPlatformTLS::SetTlsValue(TlsSlot, TlsData.PtrDummy);
488
489 return TlsData;
490 }

494 FRWLock InnerLock;

537 template<typename MutexType>
538 class TRwFifoLock
539 {
540 public:
541
542 TRwFifoLock()
543 : NumReaders(0)
544 {
545 //ThreadingPrivate::CreateLockThreadData(this);
546 }
547
548 ~TRwFifoLock() = default;
549
550 void ReadLock()
551 {
552 TRACE_CHAOS_BEGIN_LOCK(Chaos::Insights::ELockEventType::RWLockReadLock);
553 if(ThreadingPrivate::GetThreadReadDepth(this) == 0)
554 {
555 TMutexScopeLock<MutexType> Guard(Mutex);
556
557 // We lock for this increment to halt if there's a writer waiting to enter the lock
558 // In this case we will be forced to wait till the write completes
559 ++NumReaders;
560 }
561 else
562 {
563 // Only require a lock on the first acquisition. this allows recursive reads even while a
564 // writer is holding the lock waiting to enter. The writer will be allowed to proceed when
565 // all write scopes end
566 ++NumReaders;
567 }
568
569#if PHYSICS_THREAD_CONTEXT
570 // Read lock means we can access game thread data, so set the right context
571 FPhysicsThreadContext::Get().IncGameThreadContext();
572#endif
573
574 CHAOS_RECORD_ENTER_READ_LOCK;
575
576 TRACE_CHAOS_ACQUIRE_LOCK();
577 }
578
579 void WriteLock()
580 {
581 TRACE_CHAOS_BEGIN_LOCK(Chaos::Insights::ELockEventType::RWLockWriteLock);
582
583#if CHAOS_SCENE_LOCK_CHECKS
584 if(ThreadingPrivate::GetThreadReadDepth(this) > 0)
585 {
586 ensureMsgf(false, TEXT("A thread holding a read lock on the physics scene attempted to upgrade to a write lock - this is not supported, performing an unsafe write."));
587
588 // Still need to increment the context when we hit this case or we'll just crash later
589#if PHYSICS_THREAD_CONTEXT
590 // Write lock means we can access game thread data, so set the right context
591 FPhysicsThreadContext::Get().IncGameThreadContext();
592#endif
593
594 return;
595 }
596#endif
597
598 Mutex.Lock();
599
600 // Spin until all readers are finished
601 for(;;)
602 {
603 if(NumReaders.load() == 0)
604 {
605 // All readers now finished - writer can enter the lock properly (pass back to caller)
606 break;
607 }
608
609 // Issue pause instruction - architecture dependent instruction to better handle
610 // a spin lock not interfering with other threads on this core, this doesn't
611 // actually yield the thread
612 FPlatformProcess::Yield();
613 }
614
615 CHAOS_RECORD_ENTER_WRITE_LOCK;
616
617#if PHYSICS_THREAD_CONTEXT
618 // Write lock means we can access game thread data, so set the right context
619 FPhysicsThreadContext::Get().IncGameThreadContext();
620#endif
621 TRACE_CHAOS_ACQUIRE_LOCK();
622 }
623
624 void ReadUnlock()
625 {
626 CHAOS_RECORD_LEAVE_READ_LOCK;
627
628 // No locking here, just decrement atomic reader count
629 --NumReaders;
630
631#if PHYSICS_THREAD_CONTEXT
632 // Read lock is released, the gamethread context is gone
633 FPhysicsThreadContext::Get().DecGameThreadContext();
634#endif
635
636 TRACE_CHAOS_END_LOCK();
637 }
638
639 void WriteUnlock()
640 {
641 CHAOS_RECORD_LEAVE_WRITE_LOCK;
642
643 Mutex.Unlock();
644#if PHYSICS_THREAD_CONTEXT
645 // Write lock is released, the gamethread context is gone
646 FPhysicsThreadContext::Get().DecGameThreadContext();
647#endif
648
649 TRACE_CHAOS_END_LOCK();
650 }
651
652 private:
653 MutexType Mutex;
654 std::atomic<uint32> NumReaders;
655 };

664 class FPhysSpinLock
665 {
666 public:
667 FPhysSpinLock()
668 : Next(0)
669 , Current(0)
670 , WriterId(0)
671 , Count()
672 {}
673
674 void Lock()
675 {
676 // Support recursive locking
677 if(WriterId.load() == FPlatformTLS::GetCurrentThreadId())
678 {
679 Count++;
680 return;
681 }
682
683 // Get the wait value - acquire operation so Current.load can't be reordered before this
684 uint32 WaitFor = Next.fetch_add(1, std::memory_order_acquire);
685 for(;;)
686 {
687 if(WaitFor == Current.load())
688 {
689 break;
690 }
691
692 // Issue pause instruction - architecture dependent instruction to better handle
693 // a spin lock not interfering with other threads on this core, this doesn't
694 // actually yield the thread
695 FPlatformProcess::Yield();
696 }
697
698 // Lock acquired, store the thread ID for recursive locking
699 WriterId.store(FPlatformTLS::GetCurrentThreadId());
700 Count++;
701 }
702
703 void Unlock()
704 {
705 checkf(WriterId.load() == FPlatformTLS::GetCurrentThreadId(), TEXT("A thread unlocked without owning the lock (calling Lock first)"));
706 checkf(Count > 0, TEXT("A thread unlocked a lock that had no outstanding lock scopes"));
707
708 // Once all recursive locks are dropped, increment Current to allow the next thread in
709 if(--Count == 0)
710 {
711 // Clear the lock owner
712 WriterId.store(0);
713
714 // Release the next thread - this must be the last operation as immediately
715 // the next user of the lock will be allowed to take ownership
716 ++Current;
717 }
718 }
719
720 private:
721 std::atomic<uint32> Next;
722 std::atomic<uint32> Current;
723 std::atomic<uint32> WriterId;
724 uint32 Count;
725 };

732 {
733 struct FRwLockInfo
734 {
735 FRwLockInfo(void* TlsSlotValue)
736 {
737 ThreadReadDepth = uint32(uint64(TlsSlotValue));
738 ThreadWriteDepth = uint64(TlsSlotValue) >> 32;
739 }
740 void* GetTlsSlotValue()
741 {
742 uint64 ValueOut = uint64(ThreadReadDepth) | (uint64(ThreadWriteDepth) << 32);
743 return (void*)ValueOut;
744 }
745
746 uint32 ThreadReadDepth = 0;
747 uint32 ThreadWriteDepth = 0;
748 };

761 FPlatformTLS::FreeTlsSlot(TlsSlot);

764 void ReadLock()
765 {
766 TRACE_CHAOS_BEGIN_LOCK(Chaos::Insights::ELockEventType::RWLockReadLock)
767
768 FRwLockInfo ThreadInfo(FPlatformTLS::GetTlsValue(TlsSlot));
769 ThreadInfo.ThreadReadDepth++;
770 FPlatformTLS::SetTlsValue(TlsSlot, ThreadInfo.GetTlsSlotValue());
771
772 if (ThreadInfo.ThreadReadDepth + ThreadInfo.ThreadWriteDepth == 1)
773 {
774 RwLock.ReadLock();
775 }
776
777#if PHYSICS_THREAD_CONTEXT
778 // Read lock means we can access game thread data, so set the right context
779 FPhysicsThreadContext::Get().IncGameThreadContext();
780#endif

785 TRACE_CHAOS_BEGIN_LOCK(Chaos::Insights::ELockEventType::RWLockWriteLock)

788 ThreadInfo.ThreadWriteDepth++;
789 FPlatformTLS::SetTlsValue(TlsSlot, ThreadInfo.GetTlsSlotValue());
790
791#if CHAOS_SCENE_LOCK_CHECKS
792 if (ThreadInfo.ThreadReadDepth > 0)
793 {
794 UE_LOGF(LogChaos, Warning, "Attempt to upgrade a read lock to a write lock. This is not supported. Writes will be unsafe")
795 }
796#endif

799 RwLock.WriteLock();

804#endif

806 }

830 {
831 RwLock.WriteUnlock();
832 }
833

835 // Write lock is released, the gamethread context is gone

969#if UE_AUTORTFM
970 // A transactionally safe lock that works in the following novel ways:
971 // - In the open (non-transactional):
972 // - Take the lock like before. Simple!
973 // - Free the lock like before too.
974 // - In the closed (transactional):
975 // - During locking we query `TransactionalLockCount`:
976 // - 0 means we haven't taken the lock within our transaction nest and need to acquire the lock.
977 // - Otherwise we already have the lock (and are preventing non-transactional code seeing any
978 // modifications we've made while holding the lock), so just bump `TransactionalLockCount`.
979 // - We also register an on-abort handler to release the lock should we abort (but we need to
980 // query `TransactionalLockCount` even there because we could be aborting an inner transaction
981 // and the parent transaction still wants to have the lock held!).
982 // - During unlocking we defer doing the unlock until the transaction commits.
983 //
984 // Thus with this approach we will hold this lock for the *entirety* of the transactional nest should
985 // we take the lock during the transaction, thus preventing non-transactional code from seeing any
986 // modifications we should make.
987 //
988 // If we are within a transaction, we pessimise our read-lock to a write-lock. Note: that it should
989 // potentially be possible to have read-locks work correctly, but serious care will have to be taken to
990 // ensure that we don't have:
991 // Open Thread Closed Thread
992 // ----------- ReadLock
993 // ----------- ReadUnlock
994 // WriteLock -------------
995 // WriteUnlock -------------
996 // ----------- ReadLock <- Invalid because the transaction can potentially observe side
997 // effects of the open-threads writes!
998 struct FPhysSceneLockTransactionallySafe final
999 {
1000 // Always open because the constructor arguments will create the underlying lock.
1001 UE_AUTORTFM_ALWAYS_OPEN
1002 FPhysSceneLockTransactionallySafe() : State(MakeShared<FState>())
1003 {
1004 if (AutoRTFM::IsTransactional())
1005 {
1006 const AutoRTFM::ETransactionStatus Status = AutoRTFM::Close([this]
1007 {
1008 AutoRTFM::PushOnAbortHandler(this, [this]
1009 {
1010 this->~FPhysSceneLockTransactionallySafe();
1011 });
1012 });
1013
1014 ensure(AutoRTFM::ETransactionStatus::Executing == Status);
1015 }
1016 }
1017
1018 ~FPhysSceneLockTransactionallySafe()
1019 {
1020 if (AutoRTFM::IsTransactional())
1021 {
1022 const AutoRTFM::ETransactionStatus Status = AutoRTFM::Close([this]
1023 {
1024 AutoRTFM::PopOnAbortHandler(this);
1025
1026 // We explicitly copy the state here for the case that `this` was stack
1027 // allocated and has already died before the on-commit is hit.
1028 AutoRTFM::OnCommit([State = AutoRTFM::TOpenWrapper{this->State}]
1029 {
1030 ensure(0 == State.Object->TransactionalLockCount);
1031 });
1032 });
1033
1034 ensure(AutoRTFM::ETransactionStatus::Executing == Status);
1035 }
1036
1037 // As the State was constructed in the open, it must be released in the open.
1038 AutoRTFM::Open([&] AUTORTFM_DISABLE { State = nullptr; });
1039 }
1040
1041 void ReadLock()
1042 {
1043 if (!AutoRTFM::IsIdle())
1044 {
1045 // Transactionally pessimise ReadLock -> WriteLock.
1046 WriteLock();
1047 }
1048 else
1049 {
1050 State->Lock.ReadLock();
1051 ensure(0 == State->TransactionalLockCount);
1052 }
1053 }
1054
1055 void ReadUnlock()
1056 {
1057 if (!AutoRTFM::IsIdle())
1058 {
1059 // Transactionally pessimise ReadUnlock -> WriteUnlock.
1060 WriteUnlock();
1061 }
1062 else
1063 {
1064 ensure(0 == State->TransactionalLockCount);
1065 State->Lock.ReadUnlock();
1066 }
1067 }
1068
1069 void WriteLock()
1070 {
1071 if (!AutoRTFM::IsIdle())
1072 {
1073 UE_AUTORTFM_OPEN
1074 {
1075 // The transactional system which can increment TransactionalLockCount
1076 // is always single-threaded, thus this is safe to check without atomicity.
1077 if (0 == State->TransactionalLockCount)
1078 {
1079 State->Lock.WriteLock();
1080 }
1081
1082 State->TransactionalLockCount += 1;
1083 };
1084
1085 // We explicitly copy the state here for the case that `this` was stack
1086 // allocated and has already died before the on-abort is hit.
1087 AutoRTFM::OnAbort([State = AutoRTFM::TOpenWrapper{this->State}]
1088 {
1089 ensure(0 != State.Object->TransactionalLockCount);
1090
1091 State.Object->TransactionalLockCount -= 1;
1092
1093 if (0 == State.Object->TransactionalLockCount)
1094 {
1095 State.Object->Lock.WriteUnlock();
1096 }
1097 });
1098 }
1099 else
1100 {
1101 State->Lock.WriteLock();
1102 ensure(0 == State->TransactionalLockCount);
1103 }
1104 }
1105
1106 void WriteUnlock()
1107 {
1108 if (!AutoRTFM::IsIdle())
1109 {
1110 // We explicitly copy the state here for the case that `this` was stack
1111 // allocated and has already died before the on-commit is hit.
1112 AutoRTFM::OnCommit([State = AutoRTFM::TOpenWrapper{this->State}]
1113 {
1114 ensure(0 != State.Object->TransactionalLockCount);
1115
1116 State.Object->TransactionalLockCount -= 1;
1117
1118 if (0 == State.Object->TransactionalLockCount)
1119 {
1120 State.Object->Lock.WriteUnlock();
1121 }
1122 });
1123 }
1124 else
1125 {
1126 ensure(0 == State->TransactionalLockCount);
1127 State->Lock.WriteUnlock();
1128 }
1129 }
