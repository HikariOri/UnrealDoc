831template<typename T>

833{

966template <typename T>
967bool TVector4<T>::SerializeFromVector3(FName StructTag, FArchive& Ar)
968{
969 // Upgrade Vector3 - only set X/Y/Z. The W should already have been set to the property specific default and we don't want to trash it by forcing 0 or 1.
970 if(StructTag == NAME_Vector3f)
971 {
972 FVector3f AsVec;
973 Ar << AsVec;
974 X = (T)AsVec.X;
975 Y = (T)AsVec.Y;
976 Z = (T)AsVec.Z;
977 return true;
978 }
979 else if(StructTag == NAME_Vector || StructTag == NAME_Vector3d)
980 {
981 FVector3d AsVec; // Note: Vector relies on FVector3d serializer to handle float/double based on archive version.
982 Ar << AsVec;
983 X = (T)AsVec.X;
984 Y = (T)AsVec.Y;
985 Z = (T)AsVec.Z;
986 return true;
987 }
988 return false;
989}
