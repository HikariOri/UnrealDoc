400 {
401 // simple test if Ptr is null (avoids resolving either side)
402 if (!Ptr)
403 {
404 return !Other;
405 }
406
407#if UE_WITH_OBJECT_HANDLE_LATE_RESOLVE || UE_WITH_REMOTE_OBJECT_HANDLE
408 if (Ptr.IsResolved())
409 {
410 return Ptr.GetNoResolveNoCheck() == Other;

416#endif
