172{
173 enum { TheMax = 0x7fff'fffful };
174 check(FThrottler().GetAllowance() >= TheMax);
175
176 FThrottler Throttler;
177 uint64 OneSecond = Throttler.CycleFreq;
178
179 // timing test
180 FIoBuffer RecvData;
181 for (uint32 SizeKiB : { 64, 128, 192 })
182 {
183 const uint32 ThrottleKiB = 64;
184
185 TAnsiStringBuilder<128> Url;
186 Url << TestUrl;
187 Url << (SizeKiB << 10);
188
189 FEventLoop Loop;
190 Loop.Throttle(ThrottleKiB);
191
192 FRequest Request = Loop.Request("GET", Url);
193 Request.Accept("*/*");
194 Loop.Send(MoveTemp(Request), [&] (const FTicketStatus& Status) {
195 check(Status.GetId() != FTicketStatus::EId::Error);
196 if (Status.GetId() == FTicketStatus::EId::Response)
197 {
198 Status.GetResponse().SetDestination(&RecvData);
199 }
200 });
201
202 int32 Timeout = -1;
203 if (SizeKiB < 128) Timeout = 123;
204 if (SizeKiB > 128) Timeout = 4567;
205
206 uint64 Time = FPlatformTime::Cycles64();
207 while (Loop.Tick(Timeout));
208 Time = FPlatformTime::Cycles64() - Time;
209 Time /= OneSecond;
210
211 // It's dangerous stuff testing elapsed time you know. The +1 is because
212 // throttling assumes one second has already passed when initialised.
213#if PLATFORM_WINDOWS
214 check(Time + 1 == (SizeKiB / ThrottleKiB));
215#endif

217 RecvData = FIoBuffer();
