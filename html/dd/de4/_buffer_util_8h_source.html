85 {
86 const int32 NumValues = Values.Num();

94 Indices.SetNumUninitialized(NumValues);

96 {
97 Indices[Idx] = Idx;
98 }
99
100 // sort it
101 if (!bStableSort)
102 {
103 if (bAscendingOrder)
104 {
105 Indices.Sort([&Values](int32 A, int32 B) { return Values[A] < Values[B]; });
106 }
107 else

110 }
111 }
112 else
113 {
114 if (bAscendingOrder)
115 {
116 Indices.StableSort([&Values](int32 A, int32 B) { return Values[A] < Values[B]; });
117 }
118 else
119 {
120 Indices.StableSort([&Values](int32 A, int32 B) { return Values[A] > Values[B]; });
121 }
122 }
123
124 // swap the ToSort array to follow the ordering of the reference array
125 for (int32 Idx = 0; Idx < NumValues; ++Idx)
