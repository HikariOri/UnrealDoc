54
55 int32 MinTileX = Region.Min.X / TileResolution.X;
56 int32 MinTileY = Region.Min.Y / TileResolution.Y;

59 int32 MaxTileY = Region.Max.Y / TileResolution.Y;
60
61 for (int32 TileY = MinTileY; TileY <= MaxTileY; ++TileY)
62 {
63 for (int32 TileX = MinTileX; TileX <= MaxTileX; ++TileX)
64 {
65 const FIntPoint TileMin(TileX * TileResolution.X, TileY * TileResolution.Y);
66
67 const FString* TileFilename = TileFilenames.Find(FIntPoint(TileX, TileY));
68 if (TileFilename)
69 {
70 FLandscapeImageDataRef ImageData;
71 FLandscapeImageFileCache& LandscapeImageFileCache = FModuleManager::GetModuleChecked<ILandscapeEditorModule>("LandscapeEditor").GetImageFileCache();
72 FLandscapeFileInfo TileResult = LandscapeImageFileCache.FindImage<T>(**TileFilename, ImageData);
73
74 if (TileResult.ResultCode == ELandscapeImportResult::Error)
75 {
76 return TileResult;
77 }
78
79 const uint8* TileData = ImageData.Data->GetData();
80
81 FIntRect TileRegion = Region;
82 TileRegion -= TileMin;
83 TileRegion.Clip(FIntRect(0, 0, TileResolution.X, TileResolution.Y));
84
85 for (int32 Y = 0; Y < TileRegion.Height(); ++Y)
86 {
87 for (int32 X = 0; X < TileRegion.Width(); ++X)
88 {
89 const FIntPoint TileCoord = FIntPoint(X, Y) + TileRegion.Min;
90 const FIntPoint OutCoord = TileCoord + TileMin - Region.Min;
91
92 const int32 SrcY = bFlipYAxis ? TileResolution.Y - TileCoord.Y - 1 : TileCoord.Y;
93
94 T& Dest = OutImageData[OutCoord.Y * Region.Width() + OutCoord.X];
95 const uint8* Src = &TileData[ImageData.BytesPerPixel * (SrcY * TileResolution.X + TileCoord.X)];
96
97 if (sizeof(T) == ImageData.BytesPerPixel)
98 {
99 Dest = *(reinterpret_cast<const T*>(Src));
100 }
101 else if (ImageData.BytesPerPixel == 2 && sizeof(T) == 1)
102 {

104 }

110 }
