25template <typename SkeletalMeshUpdatePacketType>
26void FSkeletalMeshUpdateChannel::Replay(FRHICommandList& RHICmdList, SkeletalMeshUpdatePacketType& UpdatePacket, TArray<FSkeletalMeshUpdateEvent, FConcurrentLinearArrayAllocator>& Events)
27{
28 check(IsInParallelRenderingThread());
29 TRACE_CPUPROFILER_EVENT_SCOPE(FSkeletalMeshUpdateChannel::Replay);
30
31 using MeshObjectType = typename SkeletalMeshUpdatePacketType::MeshObjectType;
32 using MeshDynamicDataType = typename SkeletalMeshUpdatePacketType::MeshDynamicDataType;
33
34 Events.Reserve(OpStream.NumUpdates);
35
36 for (FOp Op : OpStream.Ops)
37 {
38 switch (Op.Type)
39 {
40 case FOp::EType::Add:
41 {
42 const int32 ExpectedSize = Op.HandleIndex + 1;
43 if (SlotRegistry.Slots.Num() < ExpectedSize)
44 {
45 SlotRegistry.Slots.SetNum(ExpectedSize);
46 SlotRegistry.SlotBits.SetNum(ExpectedSize, false);
47 }
48 SlotRegistry.Slots[Op.HandleIndex].MeshObject = Op.Data_Add.MeshObject;
49 }
50 break;
51 case FOp::EType::Remove:
52 {
53 FSlot& Slot = SlotRegistry.Slots[Op.HandleIndex];
54 if (Slot.DynamicData)
55 {
56 UpdatePacket.Free(static_cast<MeshDynamicDataType*>(Slot.DynamicData));
57 }
58 Slot = {};
59 SlotRegistry.SlotBits[Op.HandleIndex] = false;
60 }
61 break;
62 case FOp::EType::Update:
63 {
64 FSlot& Slot = SlotRegistry.Slots[Op.HandleIndex];
65 check(Slot.MeshObject);
66 const int32 LODIndex = Op.Data_Update.MeshDynamicData->LODIndex;
67 Slot.DynamicData = Op.Data_Update.MeshDynamicData;
68 Slot.LODIndex = LODIndex;
69 SlotRegistry.SlotBits[Op.HandleIndex] = true;
70 }
71 break;
72 }
73 }
74
75 for (TConstSetBitIterator<> It(SlotRegistry.SlotBits); It; ++It)
76 {
77 FSlot& Slot = SlotRegistry.Slots[It.GetIndex()];
78 auto* MeshObject = static_cast<MeshObjectType*>(Slot.MeshObject);
79 UpdatePacket.Add(MeshObject, static_cast<MeshDynamicDataType*>(Slot.DynamicData));
80
81 Events.Emplace(FSkeletalMeshUpdateEvent
82 {
83 .MeshObject = Slot.MeshObject
84 , .LODIndex = Slot.LODIndex
85 });
86
87 Slot.DynamicData = nullptr;
88 }
89
90 SlotRegistry.SlotBits.Init(false, SlotRegistry.SlotBits.Num());
91 OpStream = {};
92}
