35 // get the index of the oldest query based on the current frame and number of buffered frames.
36 static uint32 GetQueryLookupIndex(int32 CurrentFrame, int32 NumBufferedFrames)
37 {
38 // queries are currently always requested earlier in the frame than they are issued.
39 // thus we can always overwrite the oldest query with the current one as we never need them
40 // to coexist. This saves us a buffer entry.

111 inline int32 ScanOldestNonStaleQueryIndex(uint32 FrameNumber, int32 NumBufferedFrames, int32 LagTolerance) const
112 {
113 uint32 OldestFrame = UINT32_MAX;
114 int32 OldestQueryIndex = -1;
115 for (int Index = 0; Index < NumBufferedFrames; ++Index)
116 {
117 const uint32 ThisFrameNumber = PendingOcclusionQueryFrames[Index];
118 const int32 LaggedFrames = FrameNumber - ThisFrameNumber;
119 // Queries older than LagTolerance are invalid. They may have already been reused and will give incorrect results if read
120 if (PendingOcclusionQuery[Index] && LaggedFrames <= LagTolerance && ThisFrameNumber < OldestFrame)
121 {
122 OldestFrame = ThisFrameNumber;
123 OldestQueryIndex = Index;
124 }
125 }
126 return OldestQueryIndex;
127 }

174 inline FRHIRenderQuery* GetQueryForReading(uint32 FrameNumber, int32 NumBufferedFrames, int32 LagTolerance, bool& bOutGrouped) const
175 {
176 const int32 OldestQueryIndex = bNeedsScanOnRead ? ScanOldestNonStaleQueryIndex(FrameNumber, NumBufferedFrames, LagTolerance)
177 : FOcclusionQueryHelpers::GetQueryLookupIndex(FrameNumber, NumBufferedFrames);
178 const int32 LaggedFrames = FrameNumber - PendingOcclusionQueryFrames[OldestQueryIndex];
179 // Nenever read from queries are older than LagTolerance. They may have already been reused and will give incorrect results
180 if (OldestQueryIndex == -1 || !PendingOcclusionQuery[OldestQueryIndex] || LaggedFrames > LagTolerance)
181 {
182 bOutGrouped = false;
183 return nullptr;
184 }
185 bOutGrouped = bGroupedQuery[OldestQueryIndex];
186 return PendingOcclusionQuery[OldestQueryIndex];
187 }
188
189 inline void SetCurrentQuery(uint32 FrameNumber, FRHIRenderQuery* NewQuery, int32 NumBufferedFrames, bool bGrouped, bool bNeedsScan)
190 {
191 // Get the current occlusion query
192 const uint32 QueryIndex = FOcclusionQueryHelpers::GetQueryIssueIndex(FrameNumber, NumBufferedFrames);
193 PendingOcclusionQuery[QueryIndex] = NewQuery;
194 PendingOcclusionQueryFrames[QueryIndex] = FrameNumber;
195 bGroupedQuery[QueryIndex] = bGrouped;
196
197 bNeedsScanOnRead = bNeedsScan;
198 }
199
200 inline uint32 LastQuerySubmitFrame() const
201 {
202 uint32 Result = 0;
203
204 for (int32 Index = 0; Index < FOcclusionQueryHelpers::MaxBufferedOcclusionFrames; Index++)
205 {
206 if (!bGroupedQuery[Index])
207 {
208 Result = FMath::Max(Result, PendingOcclusionQueryFrames[Index]);
209 }
210 }
211
212 return Result;
213 }

271 inline void ReleaseStaleQueries(uint32 FrameNumber, int32 NumBufferedFrames)
272 {
273 for (uint32 DeltaFrame = NumBufferedFrames; DeltaFrame > 0; DeltaFrame--)
274 {
275 if (FrameNumber >= (DeltaFrame - 1))
276 {
277 uint32 TestFrame = FrameNumber - (DeltaFrame - 1);
278 const uint32 QueryIndex = FOcclusionQueryHelpers::GetQueryLookupIndex(TestFrame, NumBufferedFrames);
279 if (PendingOcclusionQueryFrames[QueryIndex] != TestFrame)
280 {
281 PendingOcclusionQuery[QueryIndex].ReleaseQuery();
282 }
283 }
284 }
285 }

289 PendingOcclusionQuery[QueryIndex].ReleaseQuery();

293 {
294 // Get the oldest occlusion query
295 const uint32 QueryIndex = FOcclusionQueryHelpers::GetQueryLookupIndex(FrameNumber, NumBufferedFrames);
296 if (PendingOcclusionQuery[QueryIndex].GetQuery() && PendingOcclusionQueryFrames[QueryIndex] == FrameNumber - uint32(NumBufferedFrames))
