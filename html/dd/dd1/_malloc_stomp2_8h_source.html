61 static constexpr SIZE_T VirtualAddressSpaceToReserve = 1024LL * 1024LL * 1024LL * 64LL; // 64 GB blocks, we try to reserve that much, but will half it and retry
62 std::atomic<UPTRINT> CurrentAddress;
63 UPTRINT StartAddress;
64 UPTRINT EndAddress;
65 SIZE_T Size;
66
67 bool Init()
68 {
69 SIZE_T SizeToTryAndReserve = VirtualAddressSpaceToReserve;
70 // if we can't reserve the full size, half the size and try again as long as we are over 1GB
71 while (SizeToTryAndReserve >= 1024LL * 1024LL * 1024LL)
72 {
73 StartAddress = (UPTRINT) ::VirtualAlloc(nullptr, SizeToTryAndReserve, MEM_RESERVE, PAGE_READWRITE);
74 if (StartAddress)
75 {
76 EndAddress = StartAddress + SizeToTryAndReserve;
77 Size = SizeToTryAndReserve;
78 CurrentAddress = StartAddress;
79 return true;
80 }
81 SizeToTryAndReserve /= 2;
82 }
83 return false;
84 }
85
86 void* Malloc(SIZE_T InSizeToAllocate, SIZE_T InPageSize, bool InUnderrun = false)
87 {
88 UPTRINT Ret = CurrentAddress.fetch_add(InSizeToAllocate);
89 if (Ret + InSizeToAllocate > EndAddress)
90 {
91 return nullptr;
92 }
93 if (!InUnderrun)
94 {
95 // leave the last page without backing store
96 Ret = (UPTRINT) ::VirtualAlloc((void*)Ret, InSizeToAllocate - InPageSize, MEM_COMMIT, PAGE_READWRITE);
97 }
98 else
99 {
100 // leave the first page without backing store
101 Ret = (UPTRINT) ::VirtualAlloc((void*)(Ret+InPageSize), InSizeToAllocate, MEM_COMMIT, PAGE_READWRITE);
102 }
103 return (void*)Ret;
104 }
