632 inline void SetOneShotManifoldContacts(const TArrayView<const FContactPoint>& ContactPoints)
633 {
634 ResetActiveManifoldContacts();
635
636 FReal MinPhi = TNumericLimits<FReal>::Max();
637
638 int32 NumContacts = ContactPoints.Num();
639
640 // If we have too many manifold points, clip the array. This is considered and error but
641 // is important for avoiding OOM or massive slowdowns when something goes wrong
642 const int32 MaxManifoldPoints = Chaos_Collision_MaxManifoldPoints;
643 if ((MaxManifoldPoints >= 0) && (NumContacts > MaxManifoldPoints))
644 {
645 NumContacts = MaxManifoldPoints;
646 LogOneShotManifoldError(MaxManifoldPoints, ContactPoints);
647 }
648
649 ManifoldPoints.Reserve(NumContacts);
650
651 for (int32 ContactIndex = 0; ContactIndex < NumContacts; ++ContactIndex)
652 {
653 const FContactPoint& ContactPoint = ContactPoints[ContactIndex];
654 if (ContactPoint.Phi < CullDistance)
655 {
656 int32 ManifoldPointIndex = AddManifoldPoint(ContactPoint);
657 if (ContactPoint.Phi < MinPhi)
658 {
659 ClosestManifoldPointIndex = ManifoldPointIndex;
660 MinPhi = ContactPoint.Phi;
661 }
662 }
663 }
664
665 // If we have a new face we may need to change the material
666 // @todo(chaos): support per-manifold point materials?
667 if (NumManifoldPoints() > 0)
668 {
669 if (ManifoldPoints[0].ContactPoint.FaceIndex != Material.FaceIndex)
670 {
671 Material.FaceIndex = ManifoldPoints[0].ContactPoint.FaceIndex;
672 Flags.bMaterialSet = false;
673 }
674 }

792 FORCEINLINE void SetSolverResults(
793 const int32 ManifoldPointIndex,
794 const FVec3f& NetPushOut,
795 const FVec3f& NetImpulse,
796 const FRealSingle StaticFrictionRatio,
797 const FRealSingle Dt)
798 {
799 const FManifoldPoint& ManifoldPoint = ManifoldPoints[ManifoldPointIndex];
800 FManifoldPointResult& ManifoldPointResult = ManifoldPointResults[ManifoldPointIndex];
801
802 // Save contact data for friction
803 bool bInsideStaticFrictionCone = false;
804 if (StaticFrictionRatio >= FRealSingle(1.0f - UE_KINDA_SMALL_NUMBER))
805 {
806 // StaticFrictionRatio ~= 1: Static friction held - we keep the same contacts points as-is for use next frame
807 SavedManifoldPoints.Emplace(
808 ManifoldPoint.ShapeAnchorPoints[0],
809 ManifoldPoint.ShapeAnchorPoints[1],
810 ManifoldPoint.InitialPhi
811 );
812 bInsideStaticFrictionCone = true;
813 }
814 else if (StaticFrictionRatio < FRealSingle(UE_KINDA_SMALL_NUMBER))
815 {
816 // StaticFrictionRatio ~= 0: No friction (or no contact/impulse) - discard the friction anchors
817 // If we have a lot of manifold points, we don't store the previous position for these contacts because we assume
818 // that there will be others that actually applied an impulse and will fall into the other two branches.
819 // This is an optimization when we have to match the new contacts with the saved ones (see AssignSavedManifoldPoints)
820 const int32 SmallNumManifoldPoints = 8;
821 if (ManifoldPoints.Num() < SmallNumManifoldPoints)
822 {
823 SavedManifoldPoints.Emplace(
824 ManifoldPoint.ContactPoint.ShapeContactPoints[0],
825 ManifoldPoint.ContactPoint.ShapeContactPoints[1],
826 ManifoldPoint.InitialPhi
827 );
828 }
829 }
830 else
831 {
832 // 0 < StaticFrictionRatio < 1: We exceeded the friction cone. Slide the friction anchor
833 // toward the last-detected contact position so that it sits at the edge of the friction cone.
834 SavedManifoldPoints.Emplace(
835 FVec3f::Lerp(ManifoldPoint.ContactPoint.ShapeContactPoints[0], ManifoldPoint.ShapeAnchorPoints[0], StaticFrictionRatio),
836 FVec3f::Lerp(ManifoldPoint.ContactPoint.ShapeContactPoints[1], ManifoldPoint.ShapeAnchorPoints[1], StaticFrictionRatio),
837 ManifoldPoint.InitialPhi
838 );
839 }
840
841 AccumulatedImpulse += NetImpulse + (NetPushOut / Dt);
842
843 ManifoldPointResult.NetPushOut = NetPushOut;
844 ManifoldPointResult.NetImpulse = NetImpulse;
845 ManifoldPointResult.bIsValid = true;
846 ManifoldPointResult.bInsideStaticFrictionCone = bInsideStaticFrictionCone;
847
848 MinInitialPhi = FMath::Min(MinInitialPhi, ManifoldPoint.InitialPhi);
849 }
