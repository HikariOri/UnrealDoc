263 virtual FNavigationReply OnNavigation(const FGeometry& MyGeometry, const FNavigationEvent& InNavigationEvent) override
264 {
265 if (this->HasValidItemsSource() && this->bHandleDirectionalNavigation && (this->bHandleGamepadEvents || InNavigationEvent.GetNavigationGenesis() != ENavigationGenesis::Controller))
266 {
267 const TArrayView<const ItemType>& ItemsSourceRef = this->GetItems();
268
269 const int32 NumItemsPerLine = GetNumItemsPerLine();
270 const int32 CurSelectionIndex = (!TListTypeTraits<ItemType>::IsPtrValid(this->SelectorItem)) ? 0 : ItemsSourceRef.Find(TListTypeTraits<ItemType>::NullableItemTypeConvertToItemType(this->SelectorItem));
271 int32 AttemptSelectIndex = -1;
272
273 const EUINavigation NavType = InNavigationEvent.GetNavigationType();
274 if ((this->Orientation == Orient_Vertical && NavType == EUINavigation::Left) ||
275 (this->Orientation == Orient_Horizontal && NavType == EUINavigation::Up))
276 {
277 if (bWrapHorizontalNavigation || (CurSelectionIndex % NumItemsPerLine) > 0)
278 {
279 AttemptSelectIndex = CurSelectionIndex - 1;
280 }
281 }
282 else if ((this->Orientation == Orient_Vertical && NavType == EUINavigation::Right) ||
283 (this->Orientation == Orient_Horizontal && NavType == EUINavigation::Down))
284 {
285 if (bWrapHorizontalNavigation || (CurSelectionIndex % NumItemsPerLine) < (NumItemsPerLine - 1))
286 {
287 AttemptSelectIndex = CurSelectionIndex + 1;
288 }
289 }
290
291 // If it's valid we'll scroll it into view and return an explicit widget in the FNavigationReply
292 if (ItemsSourceRef.IsValidIndex(AttemptSelectIndex) && this->bIsGamepadScrollingEnabled)
293 {
294 this->NavigationSelect(ItemsSourceRef[AttemptSelectIndex], InNavigationEvent);
295 return FNavigationReply::Explicit(nullptr);
296 }
297 }
298
299 return SListView<ItemType>::OnNavigation(MyGeometry, InNavigationEvent);

304 virtual STableViewBase::FReGenerateResults ReGenerateItems( const FGeometry& MyGeometry ) override
305 {
306 // Clear all the items from our panel. We will re-add them in the correct order momentarily.
307 this->ClearWidgets();
308
309 const TArrayView<const ItemType> Items = this->GetItems();
310 if (Items.Num() > 0)
311 {
312 // Item width and height is constant by design.
313 FTableViewDimensions TileDimensions = GetTileDimensions();
314 FTableViewDimensions AllottedDimensions(this->Orientation, MyGeometry.GetLocalSize());
315
316 const int32 NumItems = Items.Num();
317 const int32 NumItemsPerLine = GetNumItemsPerLine();
318 const int32 NumItemsPaddedToFillLastLine = (NumItems % NumItemsPerLine != 0)
319 ? NumItems + NumItemsPerLine - NumItems % NumItemsPerLine
320 : NumItems;
321
322 const double LinesPerScreen = AllottedDimensions.ScrollAxis / TileDimensions.ScrollAxis;
323 const double EndOfListOffset = NumItemsPaddedToFillLastLine - NumItemsPerLine * LinesPerScreen;
324 const double ClampedScrollOffset = FMath::Clamp(this->CurrentScrollOffset, 0.0, EndOfListOffset);
325 const float LayoutScaleMultiplier = MyGeometry.GetAccumulatedLayoutTransform().GetScale();
326
327 // Once we run out of vertical and horizontal space, we stop generating widgets.
328 FTableViewDimensions DimensionsUsedSoFar(this->Orientation);
329
330 // Index of the item at which we start generating based on how far scrolled down we are
331 int32 StartIndex = FMath::Max( 0, FMath::FloorToInt32(ClampedScrollOffset / NumItemsPerLine) * NumItemsPerLine);
332
333 // Let the WidgetGenerator know that we are starting a pass so that it can keep track of data items and widgets.
334 this->WidgetGenerator.OnBeginGenerationPass();
335
336 // Actually generate the widgets.
337 bool bIsAtEndOfList = false;
338 bool bHasFilledAvailableArea = false;
339 bool bNewLine = true;
340 bool bFirstLine = true;
341 double NumLinesShownOnScreen = 0;
342 for( int32 ItemIndex = StartIndex; !bHasFilledAvailableArea && ItemIndex < NumItems; ++ItemIndex )
343 {
344 const ItemType& CurItem = Items[ItemIndex];
345
346 if (bNewLine)
347 {
348 bNewLine = false;
349
350 float LineFraction = 1.f;
351 if (bFirstLine)
352 {
353 bFirstLine = false;
354 LineFraction -= (float)FMath::Fractional(ClampedScrollOffset / NumItemsPerLine);
355 }
356
357 DimensionsUsedSoFar.ScrollAxis += TileDimensions.ScrollAxis * LineFraction;
358
359 if (DimensionsUsedSoFar.ScrollAxis > AllottedDimensions.ScrollAxis)
360 {
361 NumLinesShownOnScreen += FMath::Max(1.0f - ((DimensionsUsedSoFar.ScrollAxis - AllottedDimensions.ScrollAxis) / TileDimensions.ScrollAxis), 0.0f);
362 }
363 else
364 {
365 NumLinesShownOnScreen += LineFraction;
366 }
367 }
368
369 SListView<ItemType>::GenerateWidgetForItem(CurItem, ItemIndex, StartIndex, LayoutScaleMultiplier);
370
371 // The widget used up some of the available space for the current line
372 DimensionsUsedSoFar.LineAxis += TileDimensions.LineAxis;
373
374 bIsAtEndOfList = ItemIndex >= NumItems - 1;
375
376 if (DimensionsUsedSoFar.LineAxis + TileDimensions.LineAxis > AllottedDimensions.LineAxis)
377 {
378 // A new line of widgets was completed - time to start another one
379 DimensionsUsedSoFar.LineAxis = 0;
380 bNewLine = true;
381 }
382
383 if (bIsAtEndOfList || bNewLine)
384 {
385 // We've filled all the available area when we've finished a line that's partially clipped by the end of the view
386 const float FloatPrecisionOffset = 0.001f;
387 bHasFilledAvailableArea = DimensionsUsedSoFar.ScrollAxis > AllottedDimensions.ScrollAxis + FloatPrecisionOffset;
388 }
389 }
390
391 // We have completed the generation pass. The WidgetGenerator will clean up unused Widgets.
392 this->WidgetGenerator.OnEndGenerationPass();
393
394 const float TotalGeneratedLineAxisSize = (float)(FMath::CeilToFloat(NumLinesShownOnScreen) * TileDimensions.ScrollAxis);
395 return STableViewBase::FReGenerateResults(ClampedScrollOffset, TotalGeneratedLineAxisSize, NumLinesShownOnScreen, bIsAtEndOfList && !bHasFilledAvailableArea);
396 }
397
398 return STableViewBase::FReGenerateResults(0, 0, 0, false);
399

427 virtual float ScrollBy( const FGeometry& MyGeometry, float ScrollByAmountInSlateUnits, EAllowOverscroll InAllowOverscroll ) override
428 {
429 const bool bWholeListVisible = this->DesiredScrollOffset == 0 && this->bWasAtEndOfList;
430 if (InAllowOverscroll == EAllowOverscroll::Yes && this->Overscroll.ShouldApplyOverscroll(this->DesiredScrollOffset == 0, this->bWasAtEndOfList, ScrollByAmountInSlateUnits))
431 {
432 const float UnclampedScrollDelta = ScrollByAmountInSlateUnits / (float)GetNumItemsPerLine();
433 const float ActuallyScrolledBy = this->Overscroll.ScrollBy(MyGeometry, UnclampedScrollDelta);
434 if (ActuallyScrolledBy != 0.0f)
435 {
436 this->RequestListRefresh();
437 }
438 return ActuallyScrolledBy;
439 }
440 else if (!bWholeListVisible)
441 {
442 const double NewScrollOffset = this->DesiredScrollOffset + ((ScrollByAmountInSlateUnits * (float)GetNumItemsPerLine()) / GetTileDimensions().ScrollAxis);
443
444 return this->ScrollTo( (float)NewScrollOffset );
445 }
446
447 return 0.f;

464 virtual typename SListView<ItemType>::EScrollIntoViewResult ScrollIntoView(const FGeometry& ListViewGeometry) override
465 {
466 if (TListTypeTraits<ItemType>::IsPtrValid(this->ItemToScrollIntoView) && this->HasValidItemsSource())
467 {
468 const int32 IndexOfItem = this->GetItems().Find(TListTypeTraits<ItemType>::NullableItemTypeConvertToItemType(this->ItemToScrollIntoView));
469 if (IndexOfItem != INDEX_NONE)
470 {
471 const float NumLinesInView = FTableViewDimensions(this->Orientation, ListViewGeometry.GetLocalSize()).ScrollAxis / GetTileDimensions().ScrollAxis;
472
473 double NumLiveWidgets = this->GetNumLiveWidgets();
474 if (NumLiveWidgets == 0 && this->IsPendingRefresh())
475 {
476 // Use the last number of widgets on screen to estimate if we actually need to scroll.
477 NumLiveWidgets = this->LastGenerateResults.ExactNumLinesOnScreen;
478
479 // If we still don't have any widgets, we're not in a situation where we can scroll an item into view
480 // (probably as nothing has been generated yet), so we'll defer this again until the next frame
481 if (NumLiveWidgets == 0)
482 {
483 return SListView<ItemType>::EScrollIntoViewResult::Deferred;
484 }
485 }
486
487 this->EndInertialScrolling();
488
489 // Only scroll the item into view if it's not already in the visible range
490 const int32 NumItemsPerLine = GetNumItemsPerLine();
491 const double ScrollLineOffset = this->GetTargetScrollOffset() / NumItemsPerLine;
492 const int32 LineOfItem = FMath::FloorToInt((float)IndexOfItem / (float)NumItemsPerLine);
493 const int32 NumFullLinesInView = FMath::FloorToInt32(ScrollLineOffset + NumLinesInView) - FMath::CeilToInt32(ScrollLineOffset);
494
495 const double MinDisplayedLine = this->bNavigateOnScrollIntoView ? FMath::FloorToDouble(ScrollLineOffset) : FMath::CeilToDouble(ScrollLineOffset);
496 const double MaxDisplayedLine = this->bNavigateOnScrollIntoView ? FMath::CeilToDouble(ScrollLineOffset + NumFullLinesInView) : FMath::FloorToDouble(ScrollLineOffset + NumFullLinesInView);
497
498 if (LineOfItem < MinDisplayedLine || LineOfItem > MaxDisplayedLine)
499 {
500 float NewLineOffset = 0.0f;
501 if (LineOfItem < MinDisplayedLine - 1 || LineOfItem > MaxDisplayedLine + 1)
502 {
503 // The item was completely hidden so we scroll to center it.
504
505 // Set the line with the item at the beginning of the view area
506 NewLineOffset = (float)LineOfItem;
507 // Center the line in the view area
508 NewLineOffset -= NumLinesInView * 0.5f;
509 // Center the tile itself
510 NewLineOffset += 0.5f;
511 }
512 else if (LineOfItem == MinDisplayedLine - 1)
513 {
514 // The item was just on the line above, set the line with the item at the beginning of the view area
515 // This prevent issues when navigating the list element by element triggers this code since we don't want to suddenly center an element
516 NewLineOffset = (float)LineOfItem;
517 }
518 else // LineOfItem == MaxDisplayedLine + 1
519 {
520 // The item was just on the next line, scroll it into view
521 // This prevent issues when navigating the list element by element triggers this code since we don't want to suddenly center an element
522 if (this->FixedLineScrollOffset.IsSet())
523 {
524 // We should scroll to align a line with the top of the view while putting the item at the bottom
525 const int32 FullLinesInView = FMath::FloorToInt32(NumLinesInView);
526 NewLineOffset = (float)LineOfItem - ((float)FullLinesInView - 1.f);
527 }
528 else
529 {
530 // We should scroll just enough to show the item on the last line
531 NewLineOffset = (float)LineOfItem - NumLinesInView + 1.f + this->NavigationScrollOffset;
532 }
533 }
534
535 // Convert the line offset into an item offset
536 double NewScrollOffset = NewLineOffset * (double)NumItemsPerLine;
537 // And clamp the scroll offset within the allowed limits
538 NewScrollOffset = FMath::Clamp(NewScrollOffset, 0., (double)GetNumItemsBeingObserved() - (double)NumItemsPerLine * NumLinesInView);
539
540 this->SetScrollOffset((float)NewScrollOffset);
541 }
542 else
543 {
544 // Make sure the line containing the existing entry for this item is fully in view
545 if (LineOfItem == MinDisplayedLine)
546 {
547 // This line is clipped at the top/left, so set it as the new offset
548 this->SetScrollOffset((float)(LineOfItem * NumItemsPerLine) - (this->FixedLineScrollOffset.IsSet() && LineOfItem > 0 ? 0.f : this->NavigationScrollOffset));
549 }
550 else if (LineOfItem == MaxDisplayedLine)
551 {
552 if (this->FixedLineScrollOffset.IsSet())
553 {
554 // This line is clipped at the end but since FixedLineScrollOffset is set we must align a line to the top of the list,
555 // therefore we just scroll to the next line
556 const float HiddenLinesCount = (float)LineOfItem - FMath::FloorToFloat(NumLinesInView);
557 this->SetScrollOffset((1.0f + HiddenLinesCount) * (float)NumItemsPerLine);
558 }
559 else
560 {
561 // This line is clipped at the end, so we need to advance just enough to bring it fully into view
562 // Since all tiles are required to be of the same size, this is straightforward
563 const float NewLineOffset = (float)LineOfItem - NumLinesInView + 1.f + this->NavigationScrollOffset;
564 this->SetScrollOffset(NewLineOffset * (float)NumItemsPerLine);
565 }
566 }
567 }
568
569 this->RequestListRefresh();
570
571 this->ItemToNotifyWhenInView = this->ItemToScrollIntoView;
572 }
573
574 TListTypeTraits<ItemType>::ResetPtr(this->ItemToScrollIntoView);
575 }
576
577 if (TListTypeTraits<ItemType>::IsPtrValid(this->ItemToNotifyWhenInView))
578 {
579 if (this->bEnableAnimatedScrolling)
580 {
581 // When we have a target item we're shooting for, we haven't succeeded with the scroll until a widget for it exists
582 const bool bHasWidgetForItem = this->WidgetFromItem(TListTypeTraits<ItemType>::NullableItemTypeConvertToItemType(this->ItemToNotifyWhenInView)).IsValid();
583 return bHasWidgetForItem ? SListView<ItemType>::EScrollIntoViewResult::Success : SListView<ItemType>::EScrollIntoViewResult::Deferred;
584 }
585 }
586
587 return SListView<ItemType>::EScrollIntoViewResult::Success;
588 }
