118struct FCompressedStatsData
119{
127 FCompressedStatsData( TArray<uint8>& InSrcData, TArray<uint8>& InDestData )
128 : SrcData( InSrcData )
129 , DestData( InDestData )
130 , bEndOfCompressedData( false )
131 {}
132
136 static void WriteEndOfCompressedData( FArchive& Writer )
137 {
138 int32 Marker = EStatsFileConstants::END_OF_COMPRESSED_DATA;
139 check( Writer.IsSaving() );
140 Writer << Marker << Marker;
141 }
142
143protected:
145 friend FArchive& operator << (FArchive& Ar, FCompressedStatsData& Data)
146 {
147 if( Ar.IsSaving() )
148 {
149 Data.WriteCompressed( Ar );
150 }
151 else if( Ar.IsLoading() )
152 {
153 Data.ReadCompressed( Ar );
154 }
155 else
156 {
157 check( 0 );
158 }
159 return Ar;
160 }
161
163 void WriteCompressed( FArchive& Writer )
164 {
165 int32 UncompressedSize = SrcData.Num();
166 if( UncompressedSize > EStatsFileConstants::MAX_COMPRESSED_SIZE - EStatsFileConstants::DUMMY_HEADER_SIZE )
167 {
168 int32 DisabledCompressionSize = EStatsFileConstants::NO_COMPRESSION;
169 Writer << DisabledCompressionSize << UncompressedSize;
170 Writer.Serialize( SrcData.GetData(), UncompressedSize );
171 }
172 else
173 {
174 DestData.Reserve( EStatsFileConstants::MAX_COMPRESSED_SIZE );
175 int32 CompressedSize = (int32)DestData.GetAllocatedSize();
176
177 const bool bResult = FCompression::CompressMemory( NAME_Zlib, DestData.GetData(), CompressedSize, SrcData.GetData(), UncompressedSize );
178 check( bResult );
179 Writer << CompressedSize << UncompressedSize;
180 Writer.Serialize( DestData.GetData(), CompressedSize );
181 }
182 }
183
185 void ReadCompressed( FArchive& Reader )
186 {
187 int32 CompressedSize = 0;
188 int32 UncompressedSize = 0;
189 Reader << CompressedSize << UncompressedSize;
190
191 if( CompressedSize == EStatsFileConstants::END_OF_COMPRESSED_DATA && UncompressedSize == EStatsFileConstants::END_OF_COMPRESSED_DATA )
192 {
193 bEndOfCompressedData = true;
194 }
195 // This chunk is not compressed.
196 else if( CompressedSize == 0 )
197 {
198 DestData.Reset( UncompressedSize );
199 DestData.AddUninitialized( UncompressedSize );
200 Reader.Serialize( DestData.GetData(), UncompressedSize );
201 }
202 else
203 {
204 SrcData.Reset( CompressedSize );
205 DestData.Reset( UncompressedSize );
206 SrcData.AddUninitialized( CompressedSize );
207 DestData.AddUninitialized( UncompressedSize );
208
209 Reader.Serialize( SrcData.GetData(), CompressedSize );
210 const bool bResult = FCompression::UncompressMemory(NAME_Zlib, DestData.GetData(), UncompressedSize, SrcData.GetData(), CompressedSize );
211 check( bResult );
212 }
213 }
214
215public:
219 const bool HasReachedEndOfCompressedData() const
220 {
221 return bEndOfCompressedData;
222 }
223
224protected:
226 TArray<uint8>& SrcData;
227
229 TArray<uint8>& DestData;
230
232 bool bEndOfCompressedData;
233};

377 inline void WriteFName( FArchive& Ar, FStatNameAndInfo NameAndInfo )
378 {
379 FName RawName = NameAndInfo.GetRawName();
380 bool bSendFName = !FNamesSent.Contains( RawName.GetComparisonIndex() );
381
382 int32 ComparisonInt = RawName.GetComparisonIndex().ToUnstableInt();
383 Ar << ComparisonInt;
384
385 int32 Number = NameAndInfo.GetRawNumber();
386 if (bSendFName)
387 {
388 FNamesSent.Add( RawName.GetComparisonIndex() );
389 Number |= EStatMetaFlags::SendingFName << (EStatMetaFlags::Shift + EStatAllFields::StartShift);
390 }
391 Ar << Number;
392 if (bSendFName)
393 {
394 FString Name = RawName.ToString();
395 Ar << Name;
396 }
397 }

400 inline void WriteMessage( FArchive& Ar, FStatMessage const& Item )
401 {
402 WriteFName( Ar, Item.NameAndInfo );
403 switch (Item.NameAndInfo.GetField<EStatDataType>())
404 {
405 case EStatDataType::ST_int64:
406 {
407 int64 Payload = Item.GetValue_int64();
408 Ar << Payload;
409 break;
410 }
411
412 case EStatDataType::ST_double:
413 {
414 double Payload = Item.GetValue_double();
415 Ar << Payload;
416 break;
417 }
418
419 case EStatDataType::ST_FName:
420 {
421 WriteFName( Ar, FStatNameAndInfo( Item.GetValue_FName(), false ) );
422 break;
423 }
424
425 case EStatDataType::ST_Ptr:
426 {
427 uint64 Payload = Item.GetValue_Ptr();
428 Ar << Payload;
429 break;
430 }
431 }

624struct FStatsReadStream
625{
626public:
628 FStatsStreamHeader Header;
629
631 TMap<int32, FNameEntryId> FNamesIndexMap;
632
634 TArray<FStatsFrameInfo> FramesInfo;
635
637 bool ReadHeader( FArchive& Ar )
638 {
639 bool bStatWithHeader = false;
640
641 uint32 Magic = 0;
642 Ar << Magic;
643 if( Magic == EStatMagicNoHeader::MAGIC_NO_HEADER )
644 {
645
646 }
647 else if( Magic == EStatMagicNoHeader::MAGIC_NO_HEADER_SWAPPED )
648 {
649 Ar.SetByteSwapping( true );
650 }
651 else if( Magic == EStatMagicWithHeader::MAGIC )
652 {
653 bStatWithHeader = true;
654 }
655 else if( Magic == EStatMagicWithHeader::MAGIC_SWAPPED )
656 {
657 bStatWithHeader = true;
658 Ar.SetByteSwapping( true );
659 }
660 else
661 {
662 return false;
663 }
664
665 // We detected a header for a stats file, read it.
666 if( bStatWithHeader )
667 {
668 Ar << Header;
669 }
670
671 return true;
672 }
673
675 void ReadStatPacket( FArchive& Ar, FStatPacket& StatPacked )
676 {
677 Ar << StatPacked.Frame;
678 Ar << StatPacked.ThreadId;
679 int32 MyThreadType = 0;
680 Ar << MyThreadType;
681 StatPacked.ThreadType = (EThreadType::Type)MyThreadType;
682
683 Ar << StatPacked.bBrokenCallstacks;
684 // We must handle stat messages in a different way.
685 int32 NumMessages = 0;
686 Ar << NumMessages;
687 StatPacked.StatMessages.Reserve( NumMessages );
688 for( int32 MessageIndex = 0; MessageIndex < NumMessages; ++MessageIndex )
689 {
690 new(StatPacked.StatMessages) FStatMessage( ReadMessage( Ar, true ) );
691 }
692 }
693
695 inline FStatNameAndInfo ReadFName( FArchive& Ar, bool bHasFNameMap )
696 {
697 // If we read the whole FNames translation map, we don't want to add the FName again.
698 // This is a bit tricky, even if we have the FName translation map, we still need to read the FString.
699 // CAUTION!! This is considered to be thread safe in this case.
700 int32 Index = 0;
701 Ar << Index;
702 int32 Number = 0;
703 Ar << Number;
704 FName TheFName;
705
706 if( !bHasFNameMap )
707 {
708 if( Number & (EStatMetaFlags::SendingFName << (EStatMetaFlags::Shift + EStatAllFields::StartShift)) )
709 {
710 FString Name;
711 Ar << Name;
712
713 TheFName = FName( *Name );
714 FNamesIndexMap.Add( Index, TheFName.GetComparisonIndex() );
715 Number &= ~(EStatMetaFlags::SendingFName << (EStatMetaFlags::Shift + EStatAllFields::StartShift));
716 }
717 else
718 {
719 if( FNamesIndexMap.Contains( Index ) )
720 {
721 const FNameEntryId MyIndex = FNamesIndexMap.FindChecked( Index );
722 TheFName = FName( MyIndex, MyIndex, 0 );
723 }
724 else
725 {
726 TheFName = FName( "Unknown FName" );
727 Number = 0;
728 UE_LOGF( LogTemp, Warning, "Missing FName Indexed: %d, %d", Index, Number );
729 }
730 }
731 }
732 else
733 {
734 if( Number & (EStatMetaFlags::SendingFName << (EStatMetaFlags::Shift + EStatAllFields::StartShift)) )
735 {
736 FString Name;
737 Ar << Name;
738 Number &= ~(EStatMetaFlags::SendingFName << (EStatMetaFlags::Shift + EStatAllFields::StartShift));
739 }
740 if( FNamesIndexMap.Contains( Index ) )
741 {
742 const FNameEntryId MyIndex = FNamesIndexMap.FindChecked( Index );
743 TheFName = FName( MyIndex, MyIndex, 0 );
744 }
745 else
746 {
747 TheFName = FName( "Unknown FName" );
748 Number = 0;
749 UE_LOGF( LogTemp, Warning, "Missing FName Indexed: %d, %d", Index, Number );
750 }
751 }
752
753 FStatNameAndInfo Result( TheFName, false );
754 Result.SetNumberDirect( Number );
755 return Result;
756 }
757
759 inline FStatMessage ReadMessage( FArchive& Ar, bool bHasFNameMap = false )
760 {
761 FStatMessage Result( ReadFName( Ar, bHasFNameMap ) );
762 Result.Clear();
763 switch( Result.NameAndInfo.GetField<EStatDataType>() )
764 {
765 case EStatDataType::ST_int64:
766 {
767 int64 Payload = 0;
768 Ar << Payload;
769 Result.GetValue_int64() = Payload;
770 break;
771 }
772
773 case EStatDataType::ST_double:
774 {
775 double Payload = 0;
776 Ar << Payload;
777 Result.GetValue_double() = Payload;
778 break;
779 }
780
781 case EStatDataType::ST_FName:
782 {
783 FStatNameAndInfo Payload( ReadFName( Ar, bHasFNameMap ) );
784 Result.GetValue_FMinimalName() = NameToMinimalName( Payload.GetRawName() );
785 break;
786 }
787
788 case EStatDataType::ST_Ptr:
789 {
790 uint64 Payload = 0;
791 Ar << Payload;
792 Result.GetValue_Ptr() = Payload;
793 break;
794 }
795 }
796 return Result;
797 }
798
803 void ReadFramesOffsets( FArchive& Ar )
804 {
805 Ar.Seek( Header.FrameTableOffset );
806 Ar << FramesInfo;
807 }
808
813 void ReadFNamesAndMetadataMessages(FArchive& Ar, TArray64<FStatMessage>& out_MetadataMessages)
814 {
815 // Read FNames.
816 Ar.Seek(Header.FNameTableOffset);
817 out_MetadataMessages.Reserve(Header.NumFNames);
818 for (uint64 Index = 0; Index < Header.NumFNames; Index++)
819 {
820 ReadFName(Ar, false);
821 }
822
823 // Read metadata messages.
824 Ar.Seek(Header.MetadataMessagesOffset);
825 out_MetadataMessages.Reserve(Header.NumMetadataMessages);
826 for (uint64 Index = 0; Index < Header.NumMetadataMessages; Index++)
827 {
828 out_MetadataMessages.Add(ReadMessage(Ar, false));
829 }
830 }
