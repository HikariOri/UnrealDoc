58 template <typename RangeValueType, typename ProjectionType, typename PredicateType>
59 void HeapSiftDown(RangeValueType* Heap, int32_t Index, const int32_t Count, const ProjectionType& Projection, const PredicateType& Predicate)
60 {
61 while (!HeapIsLeaf(Index, Count))
62 {
63 const int32_t LeftChildIndex = HeapGetLeftChildIndex(Index);
64 const int32_t RightChildIndex = LeftChildIndex + 1;
65
66 int32_t MinChildIndex = LeftChildIndex;
67 if (RightChildIndex < Count)
68 {
69 MinChildIndex = Predicate( uLang::Invoke(Projection, Heap[LeftChildIndex]), uLang::Invoke(Projection, Heap[RightChildIndex]) ) ? LeftChildIndex : RightChildIndex;
70 }
71
72 if (!Predicate( uLang::Invoke(Projection, Heap[MinChildIndex]), uLang::Invoke(Projection, Heap[Index]) ))
73 {
74 break;
75 }
76
77 uLang::Swap(Heap[Index], Heap[MinChildIndex]);
78 Index = MinChildIndex;
79 }
80 }

93 template <class RangeValueType, typename ProjectionType, class PredicateType>
94 int32_t HeapSiftUp(RangeValueType* Heap, int32_t RootIndex, int32_t NodeIndex, const ProjectionType& Projection, const PredicateType& Predicate)
95 {
96 while (NodeIndex > RootIndex)
97 {
98 int32_t ParentIndex = HeapGetParentIndex(NodeIndex);
99 if (!Predicate( uLang::Invoke(Projection, Heap[NodeIndex]), uLang::Invoke(Projection, Heap[ParentIndex]) ))
100 {
101 break;
102 }
103
104 uLang::Swap(Heap[NodeIndex], Heap[ParentIndex]);
105 NodeIndex = ParentIndex;
106 }
107
108 return NodeIndex;
109 }

137 template <typename RangeValueType, typename ProjectionType, class PredicateType>
138 void HeapSortInternal(RangeValueType* First, int32_t Num, ProjectionType Projection, PredicateType Predicate)
139 {
140 TReversePredicate< PredicateType > ReversePredicateWrapper(Predicate); // Reverse the predicate to build a max-heap instead of a min-heap
141 HeapifyInternal(First, Num, Projection, ReversePredicateWrapper);
142
143 for(int32_t Index = Num - 1; Index > 0; Index--)
144 {
145 uLang::Swap(First[0], First[Index]);
146
147 HeapSiftDown(First, 0, Index, Projection, ReversePredicateWrapper);
148 }
149 }
