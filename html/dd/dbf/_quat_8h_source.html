774template<typename T>
775inline TQuat<T>::TQuat(const UE::Math::TMatrix<T>& M)
776{
777 // If Matrix is NULL, return Identity quaternion. If any of them is 0, you won't be able to construct rotation
778 // if you have two plane at least, we can reconstruct the frame using cross product, but that's a bit expensive op to do here
779 // for now, if you convert to matrix from 0 scale and convert back, you'll lose rotation. Don't do that.
780 if (M.GetScaledAxis(EAxis::X).IsNearlyZero() || M.GetScaledAxis(EAxis::Y).IsNearlyZero() || M.GetScaledAxis(EAxis::Z).IsNearlyZero())
781 {
782 *this = TQuat<T>::Identity;
783 return;
784 }
785
786#if !(UE_BUILD_SHIPPING || UE_BUILD_TEST)
787 // Make sure the Rotation part of the Matrix is unit length.
788 // Changed to this (same as RemoveScaling) from RotDeterminant as using two different ways of checking unit length matrix caused inconsistency.
789 if (!ensure((FMath::Abs(1.f - M.GetScaledAxis(EAxis::X).SizeSquared()) <= UE_KINDA_SMALL_NUMBER) && (FMath::Abs(1.f - M.GetScaledAxis(EAxis::Y).SizeSquared()) <= UE_KINDA_SMALL_NUMBER) && (FMath::Abs(1.f - M.GetScaledAxis(EAxis::Z).SizeSquared()) <= UE_KINDA_SMALL_NUMBER)))
790 {
791 *this = TQuat<T>::Identity;
792 return;
793 }
794#endif
795
796 //const MeReal *const t = (MeReal *) tm;
797 T s;
798
799 // Check diagonal (trace)
800 const T tr = M.M[0][0] + M.M[1][1] + M.M[2][2];
801
802 if (tr > 0.0f)
803 {
804 T InvS = FMath::InvSqrt(tr + T(1.f));
805 this->W = T(T(0.5f) * (T(1.f) / InvS));
806 s = T(0.5f) * InvS;
807
808 this->X = ((M.M[1][2] - M.M[2][1]) * s);
809 this->Y = ((M.M[2][0] - M.M[0][2]) * s);
810 this->Z = ((M.M[0][1] - M.M[1][0]) * s);
811 }
812 else
813 {
814 // diagonal is negative
815 int32 i = 0;
816
817 if (M.M[1][1] > M.M[0][0])
818 i = 1;
819
820 if (M.M[2][2] > M.M[i][i])
821 i = 2;
822
823 static constexpr int32 nxt[3] = { 1, 2, 0 };
824 const int32 j = nxt[i];
825 const int32 k = nxt[j];
826
827 s = M.M[i][i] - M.M[j][j] - M.M[k][k] + T(1.0f);
828
829 T InvS = FMath::InvSqrt(s);
830
831 T qt[4];
832 qt[i] = T(0.5f) * (T(1.f) / InvS);
833
834 s = T(0.5f) * InvS;
835
836 qt[3] = (M.M[j][k] - M.M[k][j]) * s;
837 qt[j] = (M.M[i][j] + M.M[j][i]) * s;
838 qt[k] = (M.M[i][k] + M.M[k][i]) * s;
839
840 this->X = qt[0];
841 this->Y = qt[1];
842 this->Z = qt[2];
843 this->W = qt[3];
844
845 DiagnosticCheckNaN();
846 }
847}

982template<typename T>
983inline bool TQuat<T>::Equals(const TQuat<T>& Q, T Tolerance) const
984{
985#if PLATFORM_ENABLE_VECTORINTRINSICS
986 const QuatVectorRegister ToleranceV = VectorLoadFloat1(&Tolerance);
987 const QuatVectorRegister A = VectorLoadAligned(this);
988 const QuatVectorRegister B = VectorLoadAligned(&Q);
989
990 const QuatVectorRegister RotationSub = VectorAbs(VectorSubtract(A, B));
991 const QuatVectorRegister RotationAdd = VectorAbs(VectorAdd(A, B));
992 return !VectorAnyGreaterThan(RotationSub, ToleranceV) || !VectorAnyGreaterThan(RotationAdd, ToleranceV);
993#else
994 return (FMath::Abs(X - Q.X) <= Tolerance && FMath::Abs(Y - Q.Y) <= Tolerance && FMath::Abs(Z - Q.Z) <= Tolerance && FMath::Abs(W - Q.W) <= Tolerance)
995 || (FMath::Abs(X + Q.X) <= Tolerance && FMath::Abs(Y + Q.Y) <= Tolerance && FMath::Abs(Z + Q.Z) <= Tolerance && FMath::Abs(W + Q.W) <= Tolerance);
996#endif // PLATFORM_ENABLE_VECTORINTRINSICS
