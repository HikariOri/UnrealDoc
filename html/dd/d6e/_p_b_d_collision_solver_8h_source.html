452 FORCEINLINE_DEBUGGABLE void FPBDCollisionSolver::CalculatePositionCorrectionNormal(
453 const FSolverReal Stiffness,
454 const FSolverReal ContactDeltaNormal,
455 const FSolverReal ContactMassNormal,
456 const FSolverReal NetPushOutNormal,
457 FSolverReal& OutPushOutNormal)
458 {
459 const FSolverReal PushOutNormal = -Stiffness * ContactDeltaNormal * ContactMassNormal;
460
461 // The total pushout so far this sub-step
462 // Unilateral constraint: Net-negative impulses not allowed (negative incremental impulses are allowed as long as the net is positive)
463 if ((NetPushOutNormal + PushOutNormal) > FSolverReal(0))
464 {
465 OutPushOutNormal = PushOutNormal;
466 }
467 else
468 {
469 OutPushOutNormal = -NetPushOutNormal;
470 }
471 }

484 FORCEINLINE_DEBUGGABLE void FPBDCollisionSolver::ApplyFrictionCone(
485 const FSolverReal StaticFriction,
486 const FSolverReal DynamicFriction,
487 const FSolverReal MaxFrictionPushOut,
488 FSolverReal& InOutPushOutTangentU,
489 FSolverReal& InOutPushOutTangentV,
490 FSolverReal& InOutNetPushOutTangentU,
491 FSolverReal& InOutNetPushOutTangentV,
492 FSolverReal& OutStaticFrictionRatio)
493 {
494 FSolverReal ClampedPushOutTangentU = InOutPushOutTangentU;
495 FSolverReal ClampedPushOutTangentV = InOutPushOutTangentV;
496 FSolverReal ClampedNetPushOutTangentU = InOutNetPushOutTangentU + InOutPushOutTangentU;
497 FSolverReal ClampedNetPushOutTangentV = InOutNetPushOutTangentV + InOutPushOutTangentV;
498 FSolverReal ClampedStaticFrictionRatio = FSolverReal(1);
499
500 if (MaxFrictionPushOut < FSolverReal(UE_KINDA_SMALL_NUMBER))
501 {
502 ClampedPushOutTangentU = -InOutNetPushOutTangentU;
503 ClampedPushOutTangentV = -InOutNetPushOutTangentV;
504 ClampedNetPushOutTangentU = FSolverReal(0);
505 ClampedNetPushOutTangentV = FSolverReal(0);
506 ClampedStaticFrictionRatio = FSolverReal(0);
507 }
508 else
509 {
510 // If we exceed the static friction cone, clip to the dynamic friction cone
511 const FSolverReal MaxStaticPushOutTangentSq = FMath::Square(StaticFriction * MaxFrictionPushOut);
512 const FSolverReal NetPushOutTangentSq = FMath::Square(ClampedNetPushOutTangentU) + FMath::Square(ClampedNetPushOutTangentV);
513 if (NetPushOutTangentSq > MaxStaticPushOutTangentSq)
514 {
515 const FSolverReal MaxDynamicPushOutTangent = DynamicFriction * MaxFrictionPushOut;
516 const FSolverReal FrictionMultiplier = MaxDynamicPushOutTangent * FMath::InvSqrt(NetPushOutTangentSq);
517 ClampedNetPushOutTangentU = FrictionMultiplier * ClampedNetPushOutTangentU;
518 ClampedNetPushOutTangentV = FrictionMultiplier * ClampedNetPushOutTangentV;
519 ClampedPushOutTangentU = ClampedNetPushOutTangentU - InOutNetPushOutTangentU;
520 ClampedPushOutTangentV = ClampedNetPushOutTangentV - InOutNetPushOutTangentV;
521 ClampedStaticFrictionRatio = FrictionMultiplier;
522 }
523 }
524
525 InOutPushOutTangentU = ClampedPushOutTangentU;
526 InOutPushOutTangentV = ClampedPushOutTangentV;
527 InOutNetPushOutTangentU = ClampedNetPushOutTangentU;
528 InOutNetPushOutTangentV = ClampedNetPushOutTangentV;
529 OutStaticFrictionRatio = ClampedStaticFrictionRatio;
530 }

532 FORCEINLINE_DEBUGGABLE void FPBDCollisionSolver::ApplyPositionCorrectionTangential(
533 const FSolverReal Stiffness,
534 const FSolverReal StaticFriction,
535 const FSolverReal DynamicFriction,
536 const FSolverReal MaxFrictionPushOut,
537 const FSolverReal ContactDeltaTangentU,
538 const FSolverReal ContactDeltaTangentV,
539 FPBDCollisionSolverManifoldPoint& ManifoldPoint)

542 FSolverReal PushOutTangentV = FSolverReal(0);
543
544 CalculatePositionCorrectionTangent(
545 Stiffness,
546 ContactDeltaTangentU,
547 ManifoldPoint.ContactMassTangentU,
548 ManifoldPoint.NetPushOutTangentU,
549 PushOutTangentU); // Out
550
551 CalculatePositionCorrectionTangent(
552 Stiffness,
553 ContactDeltaTangentV,
554 ManifoldPoint.ContactMassTangentV,
555 ManifoldPoint.NetPushOutTangentV,
556 PushOutTangentV); // Out

560 StaticFriction,

564 PushOutTangentV, // InOut
565 ManifoldPoint.NetPushOutTangentU, // InOut
566 ManifoldPoint.NetPushOutTangentV, // InOut
567 ManifoldPoint.StaticFrictionRatio); // Out
568
569 // Update the particle state based on the pushout
570 const FVec3 PushOut = PushOutTangentU * ManifoldPoint.ContactTangentU + PushOutTangentV * ManifoldPoint.ContactTangentV;

580 }

584 const FSolverVec3 DR1 = ManifoldPoint.ContactTangentUAngular1 * -PushOutTangentU + ManifoldPoint.ContactTangentVAngular1 * -PushOutTangentV;
585 FConstraintSolverBody& Body1 = SolverBody1();
586
587 // NOTE: No split impulse mode for friction

589 Body1.ApplyRotationDelta(DR1);

593 FORCEINLINE_DEBUGGABLE void FPBDCollisionSolver::ApplyPositionCorrectionNormal(
594 const FSolverReal Stiffness,
595 const FSolverReal ContactDeltaNormal,
596 FPBDCollisionSolverManifoldPoint& ManifoldPoint)
597 {
598 FSolverReal PushOutNormal = FSolverReal(0);
599
600 // How much of this position correction should be momentumless?
601 // In SplitImpulse mode, only residual overlap corrections apply momentum.
602 FSolverReal SplitAlpha = ManifoldPoint.SplitImpulseAlpha;
603 FSolverReal SplitAlphaInv = FSolverReal(1) - ManifoldPoint.SplitImpulseAlpha;
604
605 CalculatePositionCorrectionNormal(
606 Stiffness,
607 ContactDeltaNormal,
608 ManifoldPoint.ContactMassNormal,
609 ManifoldPoint.NetPushOutNormal,
610 PushOutNormal); // Out
611
612
613 ManifoldPoint.NetPushOutNormal += SplitAlphaInv * PushOutNormal;
614
615 // Update the particle state based on the pushout
616 if (IsDynamic(0))
617 {
618 const FSolverVec3 DX0 = (State.InvMs[0] * PushOutNormal) * ManifoldPoint.ContactNormal;
619 const FSolverVec3 DR0 = ManifoldPoint.ContactNormalAngular0 * PushOutNormal;
620 FConstraintSolverBody& Body0 = SolverBody0();
621 if (SplitAlpha > 0)
622 {
623 Body0.ApplyPositionCorrectionDelta(SplitAlpha * DX0);
624 Body0.ApplyRotationCorrectionDelta(SplitAlpha * DR0);
625 }
626
627 if (SplitAlphaInv > 0)
628 {
629 Body0.ApplyPositionDelta(SplitAlphaInv * DX0);
630 Body0.ApplyRotationDelta(SplitAlphaInv * DR0);
631 }
632 }
633 if (IsDynamic(1))
634 {
635 const FSolverVec3 DX1 = (State.InvMs[1] * -PushOutNormal) * ManifoldPoint.ContactNormal;
636 const FSolverVec3 DR1 = ManifoldPoint.ContactNormalAngular1 * -PushOutNormal;
637 FConstraintSolverBody& Body1 = SolverBody1();
638 if (SplitAlpha > 0)
639 {
640 Body1.ApplyPositionCorrectionDelta(SplitAlpha * DX1);
641 Body1.ApplyRotationCorrectionDelta(SplitAlpha * DR1);
642 }
643
644 if (SplitAlphaInv > 0)
645 {
646 Body1.ApplyPositionDelta(SplitAlphaInv * DX1);
647 Body1.ApplyRotationDelta(SplitAlphaInv * DR1);
648 }
649 }

661 FORCEINLINE_DEBUGGABLE void FPBDCollisionSolver::ApplyVelocityCorrection(
662 const FSolverReal Stiffness,
663 const FSolverReal Dt,
664 const FSolverReal DynamicFriction,
665 const FSolverReal ContactVelocityDeltaNormal,
666 const FSolverReal ContactVelocityDeltaTangent0,
667 const FSolverReal ContactVelocityDeltaTangent1,
668 const FSolverReal MinImpulseNormal,
669 FPBDCollisionSolverManifoldPoint& ManifoldPoint)
670 {
671 FSolverReal ImpulseNormal = -Stiffness * ManifoldPoint.ContactMassNormal * ContactVelocityDeltaNormal;
672
673 // Clamp the total impulse to be positive along the normal. We can apply a net negative impulse,
674 // but only to correct the velocity that was added by pushout (in which case MinImpulseNormal will be negative).
675 if ((ManifoldPoint.NetImpulseNormal + ImpulseNormal) < MinImpulseNormal)
676 {
677 // We are trying to apply a net negative impulse larger than one to counteract the effective pushout impulse
678 // so clamp the net impulse to be equal to minus the pushout impulse along the normal.
679 ImpulseNormal = MinImpulseNormal - ManifoldPoint.NetImpulseNormal;
680 }
681
682 ManifoldPoint.NetImpulseNormal += ImpulseNormal;
683
684 FSolverVec3 Impulse = ImpulseNormal * ManifoldPoint.ContactNormal;
685
686 // Clamp the tangential impulses to the friction cone
687 FSolverReal ImpulseTangentU = 0;
688 FSolverReal ImpulseTangentV = 0;
689 if ((DynamicFriction > 0) && (Dt > 0))
690 {
691 const FSolverReal FrictionStiffness = Stiffness * CVars::Chaos_PBDCollisionSolver_Velocity_StaticFrictionStiffness;
692 ImpulseTangentU = -FrictionStiffness * ManifoldPoint.ContactMassTangentU * ContactVelocityDeltaTangent0;
693 ImpulseTangentV = -FrictionStiffness * ManifoldPoint.ContactMassTangentV * ContactVelocityDeltaTangent1;
694
695 const FSolverReal TotalImpulseNormal
696 = ManifoldPoint.NetImpulseNormal
697 + (ManifoldPoint.NetSoftPushOutNormal
698 + ManifoldPoint.NetPushOutNormal) / Dt;
699
700 // Clamp the total tangential impulse including positional contribution to the friction cone
701 const FSolverReal MinMaxFrictionImpulse = State.MinMaxFrictionPushout / Dt;
702 const FSolverReal MaxImpulseTangent = DynamicFriction * FMath::Max(MinMaxFrictionImpulse, TotalImpulseNormal);
703 const FSolverReal MaxImpulseTangentSq = FMath::Square(MaxImpulseTangent);
704 const FSolverReal NetImpulseTangentU = ManifoldPoint.NetImpulseTangentU + ManifoldPoint.NetPushOutTangentU / Dt;
705 const FSolverReal NetImpulseTangentV = ManifoldPoint.NetImpulseTangentV + ManifoldPoint.NetPushOutTangentV / Dt;
706 const FSolverReal NetImpulseTangentSq = FMath::Square(NetImpulseTangentU + ImpulseTangentU) + FMath::Square(NetImpulseTangentV + ImpulseTangentV);
707 if (NetImpulseTangentSq > (MaxImpulseTangentSq + UE_SMALL_NUMBER))
708 {
709 const FSolverReal ImpulseTangentScale = MaxImpulseTangent * FMath::InvSqrt(NetImpulseTangentSq);
710 ImpulseTangentU = (NetImpulseTangentU + ImpulseTangentU) * ImpulseTangentScale - NetImpulseTangentU;
711 ImpulseTangentV = (NetImpulseTangentV + ImpulseTangentV) * ImpulseTangentScale - NetImpulseTangentV;
712 ManifoldPoint.StaticFrictionRatio = ImpulseTangentScale;
713 }
714
715 ManifoldPoint.NetImpulseTangentU += ImpulseTangentU;
716 ManifoldPoint.NetImpulseTangentV += ImpulseTangentV;
717
718 Impulse += ImpulseTangentU * ManifoldPoint.ContactTangentU + ImpulseTangentV * ManifoldPoint.ContactTangentV;
719 }
720
721 // Apply the velocity deltas from the impulse
722 if (IsDynamic(0))
723 {
724 const FSolverVec3 DV0 = State.InvMs[0] * Impulse;
725 const FSolverVec3 DW0 = ManifoldPoint.ContactNormalAngular0 * ImpulseNormal + ManifoldPoint.ContactTangentUAngular0 * ImpulseTangentU + ManifoldPoint.ContactTangentVAngular0 * ImpulseTangentV;
726 FConstraintSolverBody& Body0 = SolverBody0();
727 Body0.ApplyVelocityDelta(DV0, DW0);
728 }
729 if (IsDynamic(1))
730 {
731 const FSolverVec3 DV1 = -State.InvMs[1] * Impulse;
732 const FSolverVec3 DW1 = ManifoldPoint.ContactNormalAngular1 * -ImpulseNormal + ManifoldPoint.ContactTangentUAngular1 * -ImpulseTangentU + ManifoldPoint.ContactTangentVAngular1 * -ImpulseTangentV;
733 FConstraintSolverBody& Body1 = SolverBody1();
734 Body1.ApplyVelocityDelta(DV1, DW1);
735 }
736 }

738 FORCEINLINE_DEBUGGABLE void FPBDCollisionSolver::ApplyVelocityCorrectionNormal(
739 const FSolverReal Stiffness,
740 const FSolverReal ContactVelocityDeltaNormal,
741 const FSolverReal MinImpulseNormal,
742 FPBDCollisionSolverManifoldPoint& ManifoldPoint)
743 {
744 FSolverReal ImpulseNormal = -(Stiffness * ManifoldPoint.ContactMassNormal) * ContactVelocityDeltaNormal;
745
746 // See comments in ApplyVelocityCorrection
747 if (ManifoldPoint.NetImpulseNormal + ImpulseNormal < MinImpulseNormal)
748 {
749 ImpulseNormal = MinImpulseNormal - ManifoldPoint.NetImpulseNormal;
750 }
751
752 ManifoldPoint.NetImpulseNormal += ImpulseNormal;
753
754 // Calculate the velocity deltas from the impulse
755 FSolverVec3 Impulse = ImpulseNormal * ManifoldPoint.ContactNormal;
756 if (IsDynamic(0))
757 {
758 const FSolverVec3 DV0 = State.InvMs[0] * Impulse;
759 const FSolverVec3 DW0 = ManifoldPoint.ContactNormalAngular0 * ImpulseNormal;
760 FConstraintSolverBody& Body0 = SolverBody0();
761 Body0.ApplyVelocityDelta(DV0, DW0);
762 }
763 if (IsDynamic(1))
764 {
765 const FSolverVec3 DV1 = -State.InvMs[1] * Impulse;
766 const FSolverVec3 DW1 = ManifoldPoint.ContactNormalAngular1 * -ImpulseNormal;
767 FConstraintSolverBody& Body1 = SolverBody1();
768 Body1.ApplyVelocityDelta(DV1, DW1);
769 }

868 FORCEINLINE_DEBUGGABLE void FPBDCollisionSolver::CalculateContactPositionCorrectionNormal(
869 const FPBDCollisionSolverManifoldPoint& ManifoldPoint,
870 const FSolverReal MaxPushOut,
871 FSolverReal& OutContactDeltaNormal) const
872 {
873 const FConstraintSolverBody& Body0 = SolverBody0();
874 const FConstraintSolverBody& Body1 = SolverBody1();
875
876 // Linear version: calculate the contact delta assuming linear motion after applying a positional impulse at the contact point. There will be an error that depends on the size of the rotation.
877 FSolverReal ContactDelta = 0;
878 if (IsDynamic(0) && IsDynamic(1))
879 {
880 ContactDelta += FSolverVec3::DotProduct(Body0.DP() - Body1.DP(), ManifoldPoint.ContactNormal);
881 ContactDelta += FSolverVec3::DotProduct(Body0.DQ(), ManifoldPoint.ContactRxNormal0);
882 ContactDelta -= FSolverVec3::DotProduct(Body1.DQ(), ManifoldPoint.ContactRxNormal1);
883 }
884 else if (IsDynamic(0))
885 {
886 ContactDelta += FSolverVec3::DotProduct(Body0.DQ(), ManifoldPoint.ContactRxNormal0);
887 ContactDelta += FSolverVec3::DotProduct(Body0.DP(), ManifoldPoint.ContactNormal);
888 }
889 else if (IsDynamic(1))
890 {
891 ContactDelta -= FSolverVec3::DotProduct(Body1.DQ(), ManifoldPoint.ContactRxNormal1);
892 ContactDelta -= FSolverVec3::DotProduct(Body1.DP(), ManifoldPoint.ContactNormal);
893 }
894
895 // NOTE: ContactDelta is negative for penetration
896 // NOTE: MaxPushOut == 0 disables the pushout limits
897 if ((MaxPushOut > 0) && (ContactDelta < -MaxPushOut))
898 {
899 ContactDelta = -MaxPushOut;
900 }
901
902 OutContactDeltaNormal = ContactDelta;
903 }

1151 FORCEINLINE_DEBUGGABLE void FPBDCollisionSolver::SolveVelocity(
1152 const FSolverReal Dt,
1153 const bool bApplyDynamicFriction)
1154 {
1155 // Apply restitution at the average contact point
1156 // This means we don't need to run as many iterations to get stable bouncing
1157 // It also helps with zero restitution to counter any velocioty added by the PBD solve
1158 const bool bSolveAverageContact = (NumManifoldPoints() > 1) && CVars::bChaos_PBDCollisionSolver_Velocity_AveragePointEnabled;
1159 if (bSolveAverageContact)
1160 {
1161 SolveVelocityAverage(Dt);
1162 }
1163
1164 const FSolverReal DynamicFriction = (bApplyDynamicFriction && (Dt > 0) && CVars::bChaos_PBDCollisionSolver_Velocity_FrictionEnabled) ? State.VelocityFriction : FSolverReal(0);
1165
1166 for (int32 PointIndex = 0; PointIndex < NumManifoldPoints(); ++PointIndex)
1167 {
1168 FPBDCollisionSolverManifoldPoint& SolverManifoldPoint = State.ManifoldPoints[PointIndex];
1169 if (!!SolverManifoldPoint.Flags.bDisableVelocitySolve)
1170 {
1171 continue;
1172 }
1173
1174 // Solver for normal velocity when...
1175 // - we applied a positional impulse
1176 // - we skipped the position solve, but not the velocity solve
1177 // - we are in split impulse mode and have an overlap
1178 const bool bShouldSolveNormalVelocity =
1179 (SolverManifoldPoint.NetPushOutNormal > FSolverReal(0))
1180 || !!SolverManifoldPoint.Flags.bDisablePositionSolve
1181 || ((SolverManifoldPoint.SplitImpulseAlpha > 0) && (SolverManifoldPoint.ContactDeltaNormal < 0.0f));
1182
1183 // Solver for tangential velocity if we have non-zero friction and a position correction
1184 // (unless the position solve step is disabled, in which case we assume we want to run the velocity solve)
1185 const bool bShouldSolveTangentVelocity =
1186 (DynamicFriction > 0)
1187 && (
1188 bShouldSolveNormalVelocity
1189 || (SolverManifoldPoint.ContactDeltaNormal < State.SoftPhi)
1190 || (SolverManifoldPoint.NetSoftPushOutNormal > FSolverReal(0))
1191 )
1192 && !SolverManifoldPoint.Flags.bDisableFriction;
1193
1194 if (bShouldSolveNormalVelocity || bShouldSolveTangentVelocity)
1195 {
1196 const FSolverReal MinImpulseNormal = FMath::Min(FSolverReal(0), -(SolverManifoldPoint.NetPushOutNormal + SolverManifoldPoint.NetSoftPushOutNormal) / Dt);
1197
1198 // @todo(chaos): clean this up - friction and no-friction versions should share solve-normal code
1199 if (bShouldSolveTangentVelocity)
1200 {
1201 FSolverReal ContactVelocityDeltaNormal, ContactVelocityDeltaTangentU, ContactVelocityDeltaTangentV;
1202 CalculateContactVelocityError(SolverManifoldPoint, DynamicFriction, Dt, ContactVelocityDeltaNormal, ContactVelocityDeltaTangentU, ContactVelocityDeltaTangentV);
