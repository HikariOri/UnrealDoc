444 template<typename T>
445 void Notify(EAnimDataModelNotifyType NotifyType, const T& PayloadData)
446 {
447 UScriptStruct* TypeScriptStruct = T::StaticStruct();
448
449 const FAnimDataModelNotifPayload Payload((int8*)&PayloadData, TypeScriptStruct);
450 DataModel->OnNotify(NotifyType, Payload);
451
452 ModifiedEvent.Broadcast(NotifyType, DataModel.GetInterface(), Payload);
453
454 if (ModifiedEventDynamic.IsBound())
455 {
456 ModifiedEventDynamic.Broadcast(NotifyType, DataModel, Payload);
457 }
458
459 // Only regenerate transient data when not in a bracket, or at the end of one
460 {
461 if (NotifyType == EAnimDataModelNotifyType::BracketOpened)
462 {
463 ++BracketCounter;
464 }
465 if (NotifyType == EAnimDataModelNotifyType::BracketClosed)
466 {
467 --BracketCounter;
468 }
469
470 check(BracketCounter >= 0);
471 }
472 }

482 Notify<FEmptyPayload>(NotifyType, EmptyPayload);
483 }
