37 template <typename CompareType>
38 bool MatchesWildcardRecursive(const UE_STRING_CHARTYPE* Target, int32 TargetLength, const UE_STRING_CHARTYPE* Wildcard, int32 WildcardLength, CompareType Compare)
39 {
40 // Skip over common initial non-wildcard-char sequence of Target and Wildcard
41 for (;;)
42 {
43 if (WildcardLength == 0)
44 {
45 return TargetLength == 0;
46 }
47
48 UE_STRING_CHARTYPE WCh = *Wildcard;
49 if (WCh == CHARTEXT(UE_STRING_CHARTYPE, '*') || WCh == CHARTEXT(UE_STRING_CHARTYPE, '?'))
50 {
51 break;
52 }
53
54 if (TargetLength == 0 || !Compare(*Target, WCh))
55 {
56 return false;
57 }
58
59 ++Target;
60 ++Wildcard;
61 --TargetLength;
62 --WildcardLength;
63 }
64
65 // Test for common suffix. If we make it here, WildcardLength != 0, and *Wildcard
66 // is an actual wildcard character. (Postcondition of previous loop.)
67 const UE_STRING_CHARTYPE* TPtr = Target + TargetLength;
68 const UE_STRING_CHARTYPE* WPtr = Wildcard + WildcardLength;
69 for (;;)
70 {
71 if (TargetLength == 0)
72 {
73 break;
74 }
75
76 --TPtr;
77 --WPtr;
78
79 // Since *Wildcard is a wildcard character, which we will dereference here when
80 // WildcardLength is 1, we know we won't decrement WildcardLength past 1.
81 UE_STRING_CHARTYPE WCh = *WPtr;
82 if (WCh == CHARTEXT(UE_STRING_CHARTYPE, '*') || WCh == CHARTEXT(UE_STRING_CHARTYPE, '?'))
83 {
84 break;
85 }
86
87 if (!Compare(*TPtr, WCh))
88 {
89 return false;
90 }
91
92 --TargetLength;
93 --WildcardLength;
94 }
95
96 // Match * against anything and ? against single (and zero?) chars
97 UE_STRING_CHARTYPE FirstWild = *Wildcard;
98 if (WildcardLength == 1 && (FirstWild == CHARTEXT(UE_STRING_CHARTYPE, '*') || TargetLength < 2))
99 {
100 return true;
101 }
102 ++Wildcard;
103 --WildcardLength;
104
105 // This routine is very slow, though it does ok with one wildcard
106 int32 MaxNum = TargetLength;
107 if (FirstWild == CHARTEXT(UE_STRING_CHARTYPE, '?') && MaxNum > 1)
108 {
109 MaxNum = 1;
110 }
111
112 for (int32 Index = 0; Index <= MaxNum; ++Index)
113 {
114 if (MatchesWildcardRecursive(Target + Index, TargetLength - Index, Wildcard, WildcardLength, Compare))
115 {
116 return true;
117 }
118 }
119 return false;
120 }

122 template <typename SrcCharType>
123 void AppendCharacters(TArray<UE_STRING_CHARTYPE>& Out, const SrcCharType* Str, int32 Count)
124 {
125 check(Count >= 0);
126
127 if (!Count)
128 {
129 return;
130 }
131
132 checkSlow(Str);
133
134 int32 OldEnd = Out.Num();
135
136 // Try to reserve enough space by guessing that the new length will be the same as the input length.
137 // Include an extra gap for a null terminator if we don't already have a string allocated
138 Out.AddUninitialized(Count + (OldEnd ? 0 : 1));
139 OldEnd -= OldEnd ? 1 : 0;
140
141 UE_STRING_CHARTYPE* Dest = Out.GetData() + OldEnd;
142
143 // Try copying characters to end of string, overwriting null terminator if we already have one
144 UE_STRING_CHARTYPE* NewEnd = FPlatformString::Convert(Dest, Count, Str, Count);
145 if (!NewEnd)
146 {
147 // If that failed, it will have meant that conversion likely contained multi-code unit characters
148 // and so the buffer wasn't long enough, so calculate it properly.
149 int32 Length = FPlatformString::ConvertedLength<UE_STRING_CHARTYPE>(Str, Count);
150
151 // Add the extra bytes that we need
152 Out.AddUninitialized(Length - Count);
153
154 // Restablish destination pointer in case a realloc happened
155 Dest = Out.GetData() + OldEnd;
156
157 NewEnd = FPlatformString::Convert(Dest, Length, Str, Count);
158 checkSlow(NewEnd);
159 }
160 else
161 {
162 int32 NewEndIndex = (int32)(NewEnd - Dest);
163 if (NewEndIndex < Count)
164 {
165 Out.SetNumUninitialized(OldEnd + NewEndIndex + 1, EAllowShrinking::No);
166 }
167 }
168
169 // (Re-)establish the null terminator
170 *NewEnd = CHARTEXT(UE_STRING_CHARTYPE, '\0');
171 }

400{
401 CheckInvariants();
402
403 if (InChar != 0)
404 {
405 // position to insert the character.
406 // At the end of the string if we have existing characters, otherwise at the 0 position

414 Data[InsertIndex] = InChar;

416 }

464int32 UE_STRING_CLASS::Find(const ElementType* SubStr, int32 SubStrLen, ESearchCase::Type SearchCase, ESearchDir::Type SearchDir, int32 StartPosition) const
465{
466 checkf(SubStrLen >= 0, TEXT("Invalid SubStrLen: %d"), SubStrLen);
467
468 if (SearchDir == ESearchDir::FromStart)
469 {
470 const ElementType* Start = **this;
471 int32 RemainingLength = Len();
472 if (StartPosition != INDEX_NONE && RemainingLength > 0)
473 {
474 const ElementType* End = Start + RemainingLength;
475 Start += FMath::Clamp(StartPosition, 0, RemainingLength - 1);
476 RemainingLength = UE_PTRDIFF_TO_INT32(End - Start);
477 }
478 const ElementType* Tmp = SearchCase == ESearchCase::IgnoreCase
479 ? TCString<ElementType>::Strnistr(Start, RemainingLength, SubStr, SubStrLen)
480 : TCString<ElementType>::Strnstr(Start, RemainingLength, SubStr, SubStrLen);
481
482 return Tmp ? UE_PTRDIFF_TO_INT32(Tmp-**this) : INDEX_NONE;
483 }
484 else
485 {
486 // if ignoring, do a onetime ToUpper on both strings, to avoid ToUppering multiple
487 // times in the loop below
488 if (SearchCase == ESearchCase::IgnoreCase)
489 {
490 return ToUpper().Find(UE_STRING_CLASS::ConstructFromPtrSize(SubStr, SubStrLen).ToUpper(), ESearchCase::CaseSensitive, SearchDir, StartPosition);
491 }
492 else
493 {
494 const int32 SearchStringLength=FMath::Max(1, SubStrLen);
495
496 if (StartPosition == INDEX_NONE || StartPosition >= Len())
497 {
498 StartPosition = Len();
499 }
500
501 for (int32 i = StartPosition - SearchStringLength; i >= 0; i--)
502 {
503 int32 j;
504 for (j=0; j != SubStrLen; j++)
505 {
506 if ((*this)[i+j]!=SubStr[j])
507 {
508 break;
509 }
510 }
511
512 if (j == SubStrLen)
513 {
514 return i;
515 }
516 }
517 return INDEX_NONE;
518 }
519 }

522bool UE_STRING_CLASS::Split(const UE_STRING_CLASS& InStr, UE_STRING_CLASS* LeftStr, UE_STRING_CLASS* RightStr, ESearchCase::Type SearchCase, ESearchDir::Type SearchDir) const
523{
524 check(LeftStr != RightStr || LeftStr == nullptr);
525
526 int32 InPos = Find(InStr, SearchCase, SearchDir);
527
528 if (InPos < 0)
529 {
530 return false;
531 }
532
533 if (LeftStr)
534 {
535 if (LeftStr != this)
536 {
537 *LeftStr = Left(InPos);
538 if (RightStr)
539 {
540 *RightStr = Mid(InPos + InStr.Len());
541 }
542 }
543 else
544 {
545 // we know that RightStr can't be this so we can safely modify it before we deal with LeftStr
546 if (RightStr)
547 {
548 *RightStr = Mid(InPos + InStr.Len());
549 }
550 *LeftStr = Left(InPos);
551 }
552 }
553 else if (RightStr)
554 {
555 *RightStr = Mid(InPos + InStr.Len());
556 }
557
558 return true;
559}

613void UE_STRING_CLASS::RemoveSpacesInline()
614{
615 const int32 StringLength = Len();
616 if (StringLength == 0)
617 {
618 return;
619 }
620
621 ElementType* RawData = Data.GetData();
622 int32 CopyToIndex = 0;
623 for (int32 CopyFromIndex = 0; CopyFromIndex < StringLength; ++CopyFromIndex)
624 {
625 if (RawData[CopyFromIndex] != ' ')
626 { // Copy any character OTHER than space.
627 RawData[CopyToIndex] = RawData[CopyFromIndex];
628 ++CopyToIndex;
629 }
630 }
631
632 // Copy null-terminating character.
633 if (CopyToIndex <= StringLength)

1073void UE_STRING_CLASS::ReverseString()
1074{
1075 if (Len() > 0)
1076 {
1077 ElementType* StartChar = &(*this)[0];
1078 ElementType* EndChar = &(*this)[Len()-1];
1079 ElementType TempChar;
1080 do
1081 {
1082 TempChar = *StartChar; // store the current value of StartChar
1083 *StartChar = *EndChar; // change the value of StartChar to the value of EndChar
1084 *EndChar = TempChar; // change the value of EndChar to the character that was previously at StartChar
1085
1086 StartChar++;
1087 EndChar--;
1088
1089 } while (StartChar < EndChar); // repeat until we've reached the midpoint of the string
1090 }

1140{
1141 const ElementType* DigitToChar = CHARTEXT(ElementType, "9876543210123456789");
1142 constexpr int32 ZeroDigitIndex = 9;
1143 bool bIsNumberNegative = Num < 0;
1144 const int32 TempBufferSize = 16; // 16 is big enough
1145 ElementType TempNum[TempBufferSize];
1146 int32 TempAt = TempBufferSize; // fill the temp string from the top down.
1147
1148 // Convert to string assuming base ten.
1149 do
1150 {

1152 Num /= 10;
1153 } while (Num);
1154
1155 if (bIsNumberNegative)
1156 {
1157 TempNum[--TempAt] = CHARTEXT(ElementType, '-');
1158 }
1159
1160 const ElementType* CharPtr = TempNum + TempAt;
1161 const int32 NumChars = TempBufferSize - TempAt;
1162 Append(CharPtr, NumChars);

1241UE_STRING_CLASS UE_STRING_CLASS::SanitizeFloat(double InFloat, const int32 InMinFractionalDigits)
1242{
1243 // Avoids negative zero
1244 UE::Core::Private::StripNegativeZero(InFloat);
1245
1246 // First create the string
1247 UE_STRING_CLASS TempString = UE_STRING_CLASS::Printf(CHARTEXT(FmtCharType, "%f"), InFloat);
1248 if (!TempString.IsNumeric())
1249 {
1250 // String did not format as a valid decimal number so avoid messing with it
1251 return TempString;
1252 }
1253
1254 // Trim all trailing zeros (up-to and including the decimal separator) from the fractional part of the number
1255 int32 TrimIndex = INDEX_NONE;
1256 int32 DecimalSeparatorIndex = INDEX_NONE;
1257 for (int32 CharIndex = TempString.Len() - 1; CharIndex >= 0; --CharIndex)
1258 {
1259 const ElementType Char = TempString[CharIndex];
1260 if (Char == CHARTEXT(ElementType, '.'))
1261 {
1262 DecimalSeparatorIndex = CharIndex;
1263 TrimIndex = FMath::Max(TrimIndex, DecimalSeparatorIndex);
1264 break;
1265 }
1266 if (TrimIndex == INDEX_NONE && Char != CHARTEXT(ElementType, '0'))
1267 {
1268 TrimIndex = CharIndex + 1;
1269 }
1270 }
1271 check(TrimIndex != INDEX_NONE && DecimalSeparatorIndex != INDEX_NONE);
1272 TempString.RemoveAt(TrimIndex, TempString.Len() - TrimIndex, EAllowShrinking::No);
1273
1274 // Pad the number back to the minimum number of fractional digits
1275 if (InMinFractionalDigits > 0)
1276 {
1277 if (TrimIndex == DecimalSeparatorIndex)
1278 {
1279 // Re-add the decimal separator
1280 TempString.AppendChar(CHARTEXT(ElementType, '.'));
1281 }
1282
1283 const int32 NumFractionalDigits = (TempString.Len() - DecimalSeparatorIndex) - 1;
1284 const int32 FractionalDigitsToPad = InMinFractionalDigits - NumFractionalDigits;
1285 if (FractionalDigitsToPad > 0)
1286 {
1287 TempString.Reserve(TempString.Len() + FractionalDigitsToPad);
1288 for (int32 Cx = 0; Cx < FractionalDigitsToPad; ++Cx)
1289 {
1290 TempString.AppendChar(CHARTEXT(ElementType, '0'));
1291 }
1292 }
1293 }
1294
1295 return TempString;
1296}

1446int32 UE_STRING_CLASS::ParseIntoArray(TArray<UE_STRING_CLASS>& OutArray, const ElementType* const * DelimArray, int32 NumDelims, bool bInCullEmpty) const
1447{
1448 // Make sure the delimit string is not null or empty
1449 check(DelimArray);
1450 OutArray.Reset();
1451 const ElementType* Start = Data.GetData();
1452 const int32 Length = Len();
1453 int32 SubstringBeginIndex = 0;
1454
1455 // Iterate through string.
1456 for(int32 i = 0; i < Length;)
1457 {
1458 int32 SubstringEndIndex = INDEX_NONE;
1459 int32 DelimiterLength = 0;
1460
1461 // Attempt each delimiter.
1462 for(int32 DelimIndex = 0; DelimIndex < NumDelims; ++DelimIndex)
1463 {
1464 DelimiterLength = TCString<ElementType>::Strlen(DelimArray[DelimIndex]);
1465
1466 // If we found a delimiter...
1467 if (TCString<ElementType>::Strncmp(Start + i, DelimArray[DelimIndex], DelimiterLength) == 0)
1468 {
1469 // Mark the end of the substring.
1470 SubstringEndIndex = i;
1471 break;
1472 }
1473 }
1474
1475 if (SubstringEndIndex != INDEX_NONE)
1476 {
1477 const int32 SubstringLength = SubstringEndIndex - SubstringBeginIndex;
1478 // If we're not culling empty strings or if we are but the string isn't empty anyways...
1479 if(!bInCullEmpty || SubstringLength != 0)
1480 {
1481 // ... add new string from substring beginning up to the beginning of this delimiter.
1482 OutArray.Add(UE_STRING_CLASS::ConstructFromPtrSize(Start + SubstringBeginIndex, SubstringEndIndex - SubstringBeginIndex));
1483 }
1484 // Next substring begins at the end of the discovered delimiter.
1485 SubstringBeginIndex = SubstringEndIndex + DelimiterLength;
1486 i = SubstringBeginIndex;
1487 }
1488 else
1489 {
1490 ++i;
1491 }
1492 }
1493
1494 // Add any remaining characters after the last delimiter.
1495 const int32 SubstringLength = Length - SubstringBeginIndex;
1496 // If we're not culling empty strings or if we are but the string isn't empty anyways...
1497 if(!bInCullEmpty || SubstringLength != 0)
1498 {
1499 // ... add new string from substring beginning up to the beginning of this delimiter.
1500 OutArray.Emplace(TStringView<ElementType>(Start + SubstringBeginIndex, SubstringLength));
1501 }
1502
1503 return OutArray.Num();
1504}

1506UE_STRING_CLASS UE_STRING_CLASS::Replace(const ElementType* From, const ElementType* To, ESearchCase::Type SearchCase) const &
1507{
1508 // Previous code used to accidentally accept a nullptr replacement string - this is no longer accepted.
1509 check(To);
1510
1511 if (IsEmpty() || !From || !*From)
1512 {
1513 return *this;
1514 }
1515
1516 // get a pointer into the character data
1517 const ElementType* Travel = Data.GetData();
1518
1519 // precalc the lengths of the replacement strings
1520 int32 FromLength = TCString<ElementType>::Strlen(From);
1521 int32 ToLength = TCString<ElementType>::Strlen(To);
1522
1523 UE_STRING_CLASS Result;
1524 while (true)
1525 {
1526 // look for From in the remaining string
1527 const ElementType* FromLocation = SearchCase == ESearchCase::IgnoreCase ? TCString<ElementType>::Stristr(Travel, From) : TCString<ElementType>::Strstr(Travel, From);
1528 if (!FromLocation)
1529 {
1530 break;
1531 }
1532
1533 // copy everything up to FromLocation
1534 Result.AppendChars(Travel, UE_PTRDIFF_TO_INT32(FromLocation - Travel));
1535
1536 // copy over the To
1537 Result.AppendChars(To, ToLength);
1538
1539 Travel = FromLocation + FromLength;
1540 }
1541
1542 // copy anything left over
1543 Result += Travel;
1544
1545 return Result;
1546}

1554int32 UE_STRING_CLASS::ReplaceInline(const ElementType* From, const ElementType* To, ESearchCase::Type SearchCase)
1555{
1556 int32 ReplacementCount = 0;
1557
1558 if (Len() > 0
1559 && From != nullptr && *From != 0
1560 && To != nullptr && (SearchCase == ESearchCase::IgnoreCase || TCString<ElementType>::Strcmp(From, To) != 0))
1561 {
1562 const int32 NumCharsToReplace = TCString<ElementType>::Strlen(From);
1563 const int32 NumCharsToInsert = TCString<ElementType>::Strlen(To);
1564
1565 if (NumCharsToInsert == NumCharsToReplace)
1566 {
1567 ElementType* Pos = SearchCase == ESearchCase::IgnoreCase ? TCString<ElementType>::Stristr(&(*this)[0], From) : TCString<ElementType>::Strstr(&(*this)[0], From);
1568 while (Pos != nullptr)
1569 {
1570 ReplacementCount++;
1571
1572 // TCString<ElementType>::Strcpy now inserts a terminating zero so can't use that
1573 for (int32 i = 0; i < NumCharsToInsert; i++)
1574 {
1575 Pos[i] = To[i];
1576 }
1577
1578 if (Pos + NumCharsToReplace - **this < Len())
1579 {
1580 Pos = SearchCase == ESearchCase::IgnoreCase ? TCString<ElementType>::Stristr(Pos + NumCharsToReplace, From) : TCString<ElementType>::Strstr(Pos + NumCharsToReplace, From);
1581 }
1582 else
1583 {
1584 break;
1585 }
1586 }
1587 }
1588 else if (Contains(From, SearchCase))
1589 {
1590 UE_STRING_CLASS Copy(MoveTemp(*this));
1591
1592 // get a pointer into the character data
1593 ElementType* WritePosition = (ElementType*)Copy.Data.GetData();
1594 // look for From in the remaining string
1595 ElementType* SearchPosition = SearchCase == ESearchCase::IgnoreCase ? TCString<ElementType>::Stristr(WritePosition, From) : TCString<ElementType>::Strstr(WritePosition, From);
1596 while (SearchPosition != nullptr)
1597 {
1598 ReplacementCount++;
1599
1600 // replace the first letter of the From with 0 so we can do a strcpy (via operator+=)
1601 *SearchPosition = CHARTEXT(ElementType, '\0');
1602
1603 // copy everything up to the SearchPosition
1604 (*this) += WritePosition;
1605
1606 // copy over the replacement string
1607 (*this) += To;
1608
1609 // restore the letter, just so we don't have 0's in the string
1610 *SearchPosition = *From;
1611
1612 WritePosition = SearchPosition + NumCharsToReplace;
1613 SearchPosition = SearchCase == ESearchCase::IgnoreCase ? TCString<ElementType>::Stristr(WritePosition, From) : TCString<ElementType>::Strstr(WritePosition, From);
1614 }
1615
1616 // copy anything left over
1617 (*this) += WritePosition;
1618 }
1619 }
1620
1621 return ReplacementCount;
1622}

1809
1810FArchive& operator<<(FArchive& Ar, UE_STRING_CLASS& Str)
1811{
1812 using ElementType = UE_STRING_CLASS::ElementType;
1813
1814 #if UE_STRING_CHARTYPE_IS_TCHAR
1815 // > 0 for ANSICHAR, < 0 for UTF16CHAR serialization
1816 static_assert(sizeof(UTF16CHAR) == sizeof(UCS2CHAR), "UTF16CHAR and UCS2CHAR are assumed to be the same size!");
1817 #endif
1818
1819 if (Ar.IsLoading())
1820 {
1821 int32 SaveNum = 0;
1822 Ar << SaveNum;
1823
1824 // Validate loaded num to ensure the archive is in a good state
1825 #if UE_STRING_CHARTYPE_IS_TCHAR
1826 bool bLoadUnicodeChar = SaveNum < 0;
1827 if (bLoadUnicodeChar)
1828 {
1829 // If SaveNum cannot be negated due to integer overflow, Ar is corrupted.
1830 if (SaveNum == MIN_int32)
1831 {
1832 Ar.SetCriticalError();
1833 UE_LOGF(LogCore, Error, "Archive is corrupted");
1834 return Ar;
1835 }
1836
1837 SaveNum = -SaveNum;
1838 }
1839 #else
1840 if (SaveNum < 0)
1841 {
1842 Ar.SetCriticalError();
1843 UE_LOGF(LogCore, Error, "Archive is corrupted");
1844 return Ar;
1845 }
1846 #endif
1847
1848 int64 MaxSerializeSize = Ar.GetMaxSerializeSize();
1849 // Protect against network packets allocating too much memory
1850 if ((MaxSerializeSize > 0) && (SaveNum > MaxSerializeSize))
1851 {
1852 Ar.SetCriticalError();
1853 UE_LOG(LogCore, Error, TEXT("String is too large (Size: %i, Max: %" INT64_FMT ")"), SaveNum, MaxSerializeSize);
1854 return Ar;
1855 }
1856
1857 #if UE_STRING_CHARTYPE_IS_TCHAR
1858 // Resize the array only if it passes the above tests to prevent rogue packets from crashing
1859 Str.Data.Empty(SaveNum);
1860 Str.Data.AddUninitialized(SaveNum);
1861
1862 if (SaveNum)
1863 {
1864 if (bLoadUnicodeChar)
1865 {
1866 // read in the unicode string
1867 auto Passthru = StringMemoryPassthru<UCS2CHAR>(Str.Data.GetData(), SaveNum, SaveNum);
1868 Ar.Serialize(Passthru.Get(), SaveNum * sizeof(UCS2CHAR));
1869 if (Ar.IsByteSwapping())
1870 {
1871 for (int32 CharIndex = 0; CharIndex < SaveNum; ++CharIndex)
1872 {
1873 Passthru.Get()[CharIndex] = ByteSwap(Passthru.Get()[CharIndex]);
1874 }
1875 }
1876 // Ensure the string has a null terminator
1877 Passthru.Get()[SaveNum - 1] = '\0';
1878 Passthru.Apply();
1879
1880 // Since Microsoft's vsnwprintf implementation raises an invalid parameter warning
1881 // with a character of 0xffff, scan for it and terminate the string there.
1882 // 0xffff isn't an actual Unicode character anyway.
1883 int Index = 0;
1884 if (Str.FindChar(0xffff, Index))
1885 {
1886 Str[Index] = CHARTEXT(ElementType, '\0');
1887 Str.TrimToNullTerminator();
1888 }
1889 }
1890 else
1891 {
1892 auto Passthru = StringMemoryPassthru<ANSICHAR>(Str.Data.GetData(), SaveNum, SaveNum);
1893 Ar.Serialize(Passthru.Get(), SaveNum * sizeof(ANSICHAR));
1894 // Ensure the string has a null terminator
1895 Passthru.Get()[SaveNum - 1] = '\0';
1896 Passthru.Apply();
1897 }
1898
1899 // Throw away empty string.
1900 if (SaveNum == 1)
1901 {
1902 Str.Data.Empty();
1903 }
1904 }
1905 #else
1906 if (SaveNum)
1907 {
1908 Str.Data.Empty(SaveNum + 1);
1909 Str.Data.AddUninitialized(SaveNum + 1);
1910
1911 auto Passthru = StringMemoryPassthru<ElementType>(Str.Data.GetData(), SaveNum, SaveNum);
1912 Ar.Serialize(Passthru.Get(), SaveNum * sizeof(ElementType));
1913 // Ensure the string has a null terminator
1914 Passthru.Get()[SaveNum] = UTF8TEXT('\0');
1915 Passthru.Apply();
1916
1917 // We don't need to throw away empty strings here, unlike above, because we never saved a null terminator
1918 }
1919 else
1920 {
1921 Str.Empty();
1922 }
1923 #endif
1924 }
1925 else
1926 {
1927 Str.Data.CountBytes(Ar);
1928
1929 #if UE_STRING_CHARTYPE_IS_TCHAR
1930 const bool bSaveUnicodeChar = Ar.IsForcingUnicode() || !TCString<ElementType>::IsPureAnsi(*Str);
1931 if (bSaveUnicodeChar)
1932 {
1933 // This preprocessor block should not be necessary when the StrCast below understands UTF16CHAR.
1934 FTCHARToUTF16 UTF16String(*Str, Str.Len() + 1); // include the null terminator
1935 int32 Num = UTF16String.Length() + 1; // include the null terminator
1936
1937 int32 SaveNum = -Num;
1938 Ar << SaveNum;
1939
1940 if (Num)
1941 {
1942 if (!Ar.IsByteSwapping())
1943 {
1944 Ar.Serialize((void*)UTF16String.Get(), sizeof(UTF16CHAR) * Num);
1945 }
1946 else
1947 {
1948 TArray<UTF16CHAR> Swapped(UTF16String.Get(), Num);
1949 for (int32 CharIndex = 0; CharIndex < Num; ++CharIndex)
1950 {
1951 Swapped[CharIndex] = ByteSwap(Swapped[CharIndex]);
1952 }
1953 Ar.Serialize((void*)Swapped.GetData(), sizeof(UTF16CHAR) * Num);
1954 }
1955 }
1956 }
1957 else
1958 {
1959 int32 Num = Str.Data.Num();
1960 Ar << Num;
1961
1962 if (Num)
1963 {
1964 Ar.Serialize((void*)StrCast<ANSICHAR>(Str.Data.GetData(), Num).Get(), sizeof(ANSICHAR) * Num);
1965 }
1966 }
1967 #else
1968 // Unlike the TCHAR case, we don't bother to save the null terminator
1969 int32 SaveNum = Str.Len();
1970 Ar << SaveNum;
1971
1972 if (SaveNum)
1973 {
1974 auto CompactString = StrCast<UTF8CHAR>(Str.Data.GetData(), SaveNum);
1975 Ar.Serialize((void*)CompactString.Get(), sizeof(UTF8CHAR) * CompactString.Length());
1976 }
1977 #endif
1978 }
1979
1980 return Ar;
