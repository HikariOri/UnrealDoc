13// Type of a compute message
14enum class EAgentMessageType : unsigned char
15{
16 // No message was received (end of stream)
17 None = 0x00,
18
19 // No-op message sent to keep the connection alive. Remote should reply with the same message.
20 Ping = 0x01,
21
22 // Sent in place of a regular response if an error occurs on the remote
23 Exception = 0x02,
24
25 // Fork the message loop into a new channel
26 Fork = 0x03,
27
28 // Sent as the first message on a channel to notify the remote that the remote end is attached
29 Attach = 0x04,
30
31 // Extract files on the remote machine (Initiator -> Remote)
32 WriteFiles = 0x10,
33
34 // Notification that files have been extracted (Remote -> Initiator)
35 WriteFilesResponse = 0x11,
36
37 // Deletes files on the remote machine (Initiator -> Remote)
38 DeleteFiles = 0x12,
39
40 // Execute a process in a sandbox (Initiator -> Remote)
41 ExecuteV1 = 0x16,
42
43 // Execute a process in a sandbox (Initiator -> Remote)
44 ExecuteV2 = 0x22,
45
46 // Returns output from the child process to the caller (Remote -> Initiator)
47 ExecuteOutput = 0x17,
48
49 // Returns the process exit code (Remote -> Initiator)
50 ExecuteResult = 0x18,
51
52 // Reads a blob from storage
53 ReadBlob = 0x20,
54
55 // Response to a ReadBlob request.
56 ReadBlobResponse = 0x21,
57
58 // Xor a block of data with a value
59 XorRequest = 0xf0,
60
61 // Result from an XorRequest request.
62 XorResponse = 0xf1,
63};
