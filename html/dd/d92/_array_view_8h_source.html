30 // Gets the data from the passed argument and proceeds to reinterpret the resulting elements
31 template <typename T>
32 inline decltype(auto) GetReinterpretedDataHelper(T&& Arg)
33 {
34 auto NaturalPtr = GetData(Forward<T>(Arg));
35 using NaturalElementType = std::remove_pointer_t<decltype(NaturalPtr)>;
36
37 auto Size = GetNum(Arg);
38 auto EndPtr = NaturalPtr + Size;
39 TContainerElementTypeCompatibility<NaturalElementType>::ReinterpretRangeContiguous(NaturalPtr, EndPtr, Size);
40
41 return reinterpret_cast<typename TContainerElementTypeCompatibility<NaturalElementType>::ReinterpretType*>(NaturalPtr);
42 }

520 [[nodiscard]] constexpr TArrayView Mid(SizeType Index, SizeType Count = TNumericLimits<SizeType>::Max()) const
521 {
522 ElementType* const CurrentStart = GetData();
523 const SizeType CurrentLength = Num();
524
525 // Clamp minimum index at the start of the range, adjusting the length down if necessary
526 const SizeType NegativeIndexOffset = (Index < 0) ? Index : 0;
527 Count += NegativeIndexOffset;
528 Index -= NegativeIndexOffset;
529
530 // Clamp maximum index at the end of the range
531 Index = (Index > CurrentLength) ? CurrentLength : Index;
532
533 // Clamp count between 0 and the distance to the end of the range
534 Count = FMath::Clamp(Count, (SizeType)0, (CurrentLength - Index));
535
536 TArrayView Result = TArrayView(CurrentStart + Index, Count);
537 return Result;
538 }
