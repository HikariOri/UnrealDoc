990
997 void SetupPosition( const FVector2D& InPosition, const FVector2D& InRadius )
998 {
999 if( TriListItem )
1000 {
1001 delete TriListItem;
1002 }
1003 // todo: make this calculate the UVs correctly so if you pass a texture it will work as you would expect
1004 float Angle = 0.0f;
1005 int32 NumSides = TriangleList.Num();
1006 FVector2D LastPoint = InPosition + FVector2D( InRadius.X*FMath::Cos(Angle), InRadius.Y*FMath::Sin(Angle) );
1007 for(int32 i=1; i< NumSides+1; i++)
1008 {
1009 Angle = (2 * (float)UE_PI) * (float)i/(float)NumSides;
1010 TriangleList[ i - 1 ].V0_Pos = InPosition;
1011 TriangleList[ i - 1 ].V0_Color = Color;
1012 TriangleList[ i - 1 ].V1_Pos = LastPoint;
