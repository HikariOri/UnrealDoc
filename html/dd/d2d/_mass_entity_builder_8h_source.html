383template<typename T, typename... TArgs>
384T& FEntityBuilder::Add_GetRef(TArgs&&... InArgs) requires (CElement<T> && !(CTag<T> || CChunkFragment<T>))
385{
386 State = EState::ReadyToCommit;
387
388 if (ensureMsgf(Composition.GetElementsBitSet().Contains<T>() == false, TEXT("Element of type %s has already been added"), *T::StaticStruct()->GetName()))
389 {
390 Composition.GetElementsBitSet().Add<T>();
391 CachedArchetypeHandle = FMassArchetypeHandle();
392
393 return GetInstancedStructContainerInternal<T>().Add_GetRef(FInstancedStruct::Make<T>(InArgs...)).template GetMutable<T>();
394 }
395
396 return GetInstancedStructContainerInternal<T>().FindByPredicate(FStructInstanceFindingPredicate{T::StaticStruct()})
397 ->template GetMutable<T>();
398}

400template<typename T, typename... TArgs>
401T& FEntityBuilder::GetOrCreate(TArgs&&... InArgs) requires (CElement<T> && !(CTag<T> || CChunkFragment<T>))
402{
403 State = EState::ReadyToCommit;
404
405 FInstancedStruct* ElementFound = nullptr;
406 if (Composition.GetElementsBitSet().Contains<T>())
407 {
408 // replace
409 ElementFound = GetInstancedStructContainerInternal<T>().FindByPredicate(FStructInstanceFindingPredicate{T::StaticStruct()});
410
411 checkf(ElementFound, TEXT("We expect the element to be found since we already tested the Composition"));
412 ElementFound->GetMutable<T>() = T(InArgs...);
413 }
414 else
415 {
416 Composition.GetElementsBitSet().Add<T>();
417 CachedArchetypeHandle = FMassArchetypeHandle();
418
419 ElementFound = &GetInstancedStructContainerInternal<T>().Add_GetRef(FInstancedStruct::Make<T>(InArgs...));
420 }
421
422 return ElementFound->GetMutable<T>();
423}
424
425template<typename T>
426T* FEntityBuilder::Find() requires (CElement<T> && !(CTag<T> || CChunkFragment<T>))
427{
428 if (Composition.GetElementsBitSet().Contains<T>())
429 {
430 FInstancedStruct* ElementFound = GetInstancedStructContainerInternal<T>().FindByPredicate(FStructInstanceFindingPredicate{T::StaticStruct()});
431
432 checkf(ElementFound, TEXT("We expect the element to be found since we already tested the Composition"));
433 return ElementFound->GetMutablePtr<T>();
434 }
435
436 return nullptr;
437}
