51struct FPackageStoreEntry
52{
53#if WITH_EDITORONLY_DATA
54 // the corrected package name, which may be a different case than PackageName that's passed in to GetPackageStoreEntry
55 // only applies to LinkerLoad (loose on the local filesystem) zen packages are never submitted to source control
56 FName LinkerLoadCaseCorrectedPackageName;
57#endif
58
59 // for LinkerLoad entries, this will contain the extension of the package. for Zen entries, this will stay Unspecified
60 EPackageExtension PackageExtension = EPackageExtension::Unspecified;
61
62 // default to IoDispatcher, since almost all package store entries will use it
63 EPackageLoader LoaderType = EPackageLoader::Zen;
64
65 TArrayView<const FPackageId> ImportedPackageIds;
66 TArrayView<const FShaderHash> ShaderMapHashes;
67#if WITH_EDITOR
68 TArrayView<const FPackageId> OptionalSegmentImportedPackageIds;
69 bool bHasOptionalSegment = false;
70 // This field is used to indicate that the package should load the optional chunk instead of the regular chunk. This is
71 // needed by the StorageServerPackgeStore (ie ZenStore) when loading "AutoOptional" assets, but the FilePackageStore
72 // (ie .ucas files) does _not_ set this. This is because:
73 // * AutoOptional is used to "silently" replace the normal asset with an asset that still has editor-only data in it
74 // * When .ucas files are made, they create a .o.ucas file, which is mounted with a higher priority then the .ucas file
75 // * When FilePackageStore loads, it will automatically read the editor-only version of the asset, _as if it was the normal asset_
76 // * However, ZenStore has no ".o.ucas" file to silently read from instead of the regular file, so it must, at runtime,
77 // request the editor version of the asset (the .o.uasset chunk in the store).
78 // This field is how we communicate out to the AsyncLoadingThread code that it should request the optional chunk instead of the
79 // regular chunk for a given asset. It is basically making a runtime decision that is handled offline by ucas files.
80 // Note that bHasOptionalSegment is always false for AutoOptional because of the "silently read optional as regular".
81 // So to sum up for the two main PackageStores:
82 // ZenStore:
83 // Manual Optional
84 // bHasOptionalSegment = true
85 // bReplaceChunkWithOptional = false
86 // AutoOptional
87 // bHasOptionalSegment = false
88 // bReplaceChunkWithOptional = true
89 // FileStore
90 // Manual Optional(same as Zen)
91 // bHasOptionalSegment = true
92 // bReplaceChunkWithOptional = false
93 // AutoOptional (_not_ same)
94 // bHasOptionalSegment = false
95 // bReplaceChunkWithOptional = false
96 bool bReplaceChunkWithOptional = false;
97#endif
98};
