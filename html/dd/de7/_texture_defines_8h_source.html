541inline bool IsHDR(ETextureSourceFormat Format)
542{
543 return (Format == TSF_BGRE8 || Format == TSF_RGBA16F || Format == TSF_RGBA32F || Format == TSF_R16F || Format == TSF_R32F);
544}
545
546inline bool IsHDR(TextureCompressionSettings CompressionSettings)
547{
548 switch(CompressionSettings)
549 {
550 case TC_HDR:
551 case TC_HDR_F32:
552 case TC_HDR_Compressed:
553 case TC_HalfFloat:
554 case TC_SingleFloat:
555 return true;
556 default:
557 return false;
558 }
559}
560
561inline bool IsUncompressed(TextureCompressionSettings CompressionSettings)
562{
563 return (CompressionSettings == TC_Grayscale ||
564 CompressionSettings == TC_Displacementmap ||
565 CompressionSettings == TC_VectorDisplacementmap ||
566 CompressionSettings == TC_HDR ||
567 CompressionSettings == TC_HDR_F32 ||
568 CompressionSettings == TC_EditorIcon ||
569 CompressionSettings == TC_DistanceFieldFont ||
570 CompressionSettings == TC_HalfFloat ||
571 CompressionSettings == TC_SingleFloat
572 );
573}
574
575inline bool ShouldUseGreyScaleEditorVisualization(TextureCompressionSettings CompressionSettings)
576{
577 // these formats should do R -> RGB red to gray replication in Editor viz
578 return (CompressionSettings == TC_Grayscale ||
579 CompressionSettings == TC_Alpha ||
580 CompressionSettings == TC_Displacementmap ||
581 CompressionSettings == TC_DistanceFieldFont );
582
583 // ?? maybe these too ??
584 // CompressionSettings == TC_HalfFloat ||
585 // CompressionSettings == TC_SingleFloat
586}
