146 bool BlendWidget( const float& DeltaTime, const float& DesiredValue, const float& Speed, float& OutCurrentValue )
147 {
148 bool bStillBlending = true;
149
150 float DesiredBlendSpeed = Speed * DeltaTime;
151 float BlendDif = OutCurrentValue - DesiredValue;
152
153 if (bPeak)
154 {
155 if (FMath::Abs(BlendDif) > PeakDistance + (float)FLOAT_NORMAL_THRESH)
156 {
157 SlideInCurve.JumpToEnd();
158 SlideInCurve.PlayReverse(FXWidget.ToSharedRef());
159 }
160 float Lerp = FMath::Max(SlideInCurve.GetLerp(), 0.1f);
161 DesiredBlendSpeed = Speed * Lerp * DeltaTime;
162 }
163
164 if (FMath::Abs(DesiredBlendSpeed) > FMath::Abs(BlendDif))
165 {
166 OutCurrentValue = DesiredValue;
167 bStillBlending = false;
168 }
169 else
170 {
171 // Reverse Direction if sliding left
172 if ( BlendDif > 0.f )
173 {
174 DesiredBlendSpeed *= -1.f;
175 }
176
177 OutCurrentValue += DesiredBlendSpeed;
178 }
179
180 return bStillBlending;
181 }

683 void SwapBuffer(EWidgetCarouselScrollDirection::Type ScrollDirection, int32 OverrideWidgetIndex = INDEX_NONE)
684 {
685 // Register for an active update every 2.5 seconds
686 RegisterActiveTimer(0.0f, FWidgetActiveTimerDelegate::CreateSP(this, &SWidgetCarousel::RegisteredTick));
687
688 const TArray<ItemType>& ItemsSourceRef = (*ItemsSource);
689
690 TSharedRef<SWidget> LeftWidget = LeftCarouselWidget->GetWidgetContent();
691 TSharedRef<SWidget> RightWidget = RightCarouselWidget->GetWidgetContent();
692 TSharedRef<SWidget> CenterWidget = CenterCarouselWidget->GetWidgetContent();
693
694 const bool bIsScrollingLeft = ScrollDirection == EWidgetCarouselScrollDirection::Carousel_Left;
695
696 if (ItemsSourceRef.Num() > 0)
697 {
698 if (bIsScrollingLeft)
699 {
700 WidgetIndex = GetLeftWidgetIndex(WidgetIndex);
701 TSharedPtr<FCarouselDisplayItem> TempWidget = RightCarouselWidget;
702 RightCarouselWidget = CenterCarouselWidget;
703 RightWidget = (CenterWidget == SNullWidget::NullWidget) ? OnGenerateWidget.Execute(ItemsSourceRef[GetRightWidgetIndex(WidgetIndex)]) : CenterWidget;
704 CenterCarouselWidget = LeftCarouselWidget;
705 CenterWidget = (LeftWidget == SNullWidget::NullWidget) ? OnGenerateWidget.Execute(ItemsSourceRef[WidgetIndex]) : LeftWidget;
706 LeftCarouselWidget = TempWidget;
707 LeftWidget = OnGenerateWidget.Execute(ItemsSourceRef[GetLeftWidgetIndex(WidgetIndex)]);
708 LeftCarouselWidget->SetSlide(CenterCarouselWidget->GetSlideValue() + SlideValueLeftLimit, true);
709 }
710 else
711 {
712 WidgetIndex = GetRightWidgetIndex(WidgetIndex);
713 TSharedPtr<FCarouselDisplayItem> TempWidget = LeftCarouselWidget;
714 LeftCarouselWidget = CenterCarouselWidget;
715 LeftWidget = (CenterWidget == SNullWidget::NullWidget) ? OnGenerateWidget.Execute(ItemsSourceRef[GetLeftWidgetIndex(WidgetIndex)]) : CenterWidget;
716 CenterCarouselWidget = RightCarouselWidget;
717 CenterWidget = (RightWidget == SNullWidget::NullWidget) ? OnGenerateWidget.Execute(ItemsSourceRef[WidgetIndex]) : RightWidget;
718 RightCarouselWidget = TempWidget;
719 RightWidget = OnGenerateWidget.Execute(ItemsSourceRef[GetRightWidgetIndex(WidgetIndex)]);
720 RightCarouselWidget->SetSlide(CenterCarouselWidget->GetSlideValue() + SlideValueRightLimit, true);
721 }
722 }
723
724 if (ItemsSourceRef.IsValidIndex(OverrideWidgetIndex))
725 {
726 // Need to find a way to clear left and right images
727 WidgetIndex = OverrideWidgetIndex;
728 CenterWidget = OnGenerateWidget.Execute(ItemsSourceRef[OverrideWidgetIndex]);
729 }
730
731 // Ensure the 3 CarouselWidgets here have unique widgets, if that isn't the case, prioritize the 2 widgets that will display during the transition
732 const bool bIsLeftWidgetUnique = (LeftWidget != CenterWidget) && (!bIsScrollingLeft || RightWidget != LeftWidget);
733 const bool bIsRightWidgetUnique = (RightWidget != CenterWidget) && (bIsScrollingLeft || RightWidget != LeftWidget);
734 LeftCarouselWidget->SetWidgetContent(bIsLeftWidgetUnique ? LeftWidget : SNullWidget::NullWidget);
735 RightCarouselWidget->SetWidgetContent(bIsRightWidgetUnique ? RightWidget : SNullWidget::NullWidget);
736 CenterCarouselWidget->SetWidgetContent(CenterWidget);
737
738 SetSliderLimits();
739 GenerateWidgets();
740 CenterCarouselWidget->ScrollIn(ScrollDirection);
741 RightCarouselWidget->ScrollIn(ScrollDirection);
742 LeftCarouselWidget->ScrollIn(ScrollDirection);
743
744 OnPageChanged.ExecuteIfBound(WidgetIndex);
745 }
