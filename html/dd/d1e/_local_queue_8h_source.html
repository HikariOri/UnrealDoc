32template<uint32 NumItems>
33class TWorkStealingQueueBase2
34{
35 enum class ESlotState : uintptr_t
36 {
37 Free = 0, //The slot is free and items can be put there
38 Taken = 1, //The slot is in the proccess of beeing stolen
39 };
40
41protected:
42 //insert an item at the head position (this can only safe on a single thread, shared with Get)
43 inline bool Put(uintptr_t Item)
44 {
45 checkSlow(Item != uintptr_t(ESlotState::Free));
46 checkSlow(Item != uintptr_t(ESlotState::Taken));
47
48 uint32 Idx = (Head + 1) % NumItems;
49 uintptr_t Slot = ItemSlots[Idx].Value.load(std::memory_order_acquire);
50
51 if (Slot == uintptr_t(ESlotState::Free))
52 {
53 ItemSlots[Idx].Value.store(Item, std::memory_order_release);
54 Head++;
55 checkSlow(Head % NumItems == Idx);
56 return true;
57 }
58 return false;
59 }
60
61 //remove an item at the head position in FIFO order (this can only safe on a single thread, shared with Put)
62 inline bool Get(uintptr_t& Item)
63 {
64 uint32 Idx = Head % NumItems;
65 uintptr_t Slot = ItemSlots[Idx].Value.load(std::memory_order_acquire);
66
67 if (Slot > uintptr_t(ESlotState::Taken) && ItemSlots[Idx].Value.compare_exchange_strong(Slot, uintptr_t(ESlotState::Free), std::memory_order_acq_rel))
68 {
69 Head--;
70 checkSlow((Head + 1) % NumItems == Idx);
71 Item = Slot;
72 return true;
73 }
74 return false;
75 }
76
77 //remove an item at the tail position in LIFO order (this can be done from any thread including the one that accesses the head)
78 inline bool Steal(uintptr_t& Item)
79 {
80#if UE_ASYNC_WORK_STEALING_QUEUE_USES_RETRY_LIMIT
81 int32 Retries = GWorkStealingQueueRetryLimit;
82#endif
83
84 do
85 {
86 uint32 IdxVer = Tail.load(std::memory_order_acquire);
87 uint32 Idx = IdxVer % NumItems;
88 uintptr_t Slot = ItemSlots[Idx].Value.load(std::memory_order_acquire);
89
90 if (Slot == uintptr_t(ESlotState::Free))
91 {
92 // Once we find a free slot, we need to verify if it's been freed by another steal
93 // so check back the Tail value to make sure it wasn't incremented since we first read the value.
94 // If we don't do this, some threads might not see that other threads
95 // have already stolen the slot, and will wrongly return that no more tasks are available to steal.
96 if (IdxVer != Tail.load(std::memory_order_acquire))
97 {
98 continue; // Loop again since tail has changed
99 }
100 return false;
101 }
102 else if (Slot != uintptr_t(ESlotState::Taken))
103 {
104 if (ItemSlots[Idx].Value.compare_exchange_weak(Slot, uintptr_t(ESlotState::Taken), std::memory_order_acq_rel))
105 {
106 if(IdxVer == Tail.load(std::memory_order_acquire))
107 {
108 uint32 Prev = Tail.fetch_add(1, std::memory_order_release); (void)Prev;
109 checkSlow(Prev % NumItems == Idx);
110 ItemSlots[Idx].Value.store(uintptr_t(ESlotState::Free), std::memory_order_release);
111 Item = Slot;
112 return true;
113 }
114 ItemSlots[Idx].Value.store(Slot, std::memory_order_release);
115 }
116 }
117#if UE_ASYNC_WORK_STEALING_QUEUE_USES_RETRY_LIMIT
118 else if (Retries-- == 0)
119 {
120 FPlatformProcess::YieldThread();
121 Retries = GWorkStealingQueueRetryLimit;
122 }
123#endif
124 } while(true);
125 }

169/********************************************************************************************************************************************
170 * A LocalQueueRegistry is a collection of LockFree queues that store pointers to Items, there are ThreadLocal LocalQueues with LocalItems. *
171 * LocalQueues can only be Enqueued and Dequeued by the current Thread they were installed on. But Items can be stolen from any Thread *
172 * There is a global OverflowQueue than is used when a LocalQueue goes out of scope to dump all the remaining Items in *
173 * or when a Thread has no LocalQueue installed or when the LocalQueue is at capacity. A new LocalQueue is registers itself always. *
174 * A Dequeue Operation can only be done starting from a LocalQueue, than the GlobalQueue will be checked. *
175 * Finally Items might get Stolen from other LocalQueues that are registered with the LocalQueueRegistry. *
176 ********************************************************************************************************************************************/
177template<uint32 NumLocalItems = LOCALQUEUEREGISTRYDEFAULTS_MAX_ITEMCOUNT, uint32 MaxLocalQueues = LOCALQUEUEREGISTRYDEFAULTS_MAX_LOCALQUEUES>
178class TLocalQueueRegistry

181 {
182 uint32 State = FPlatformTime::Cycles();
183 State = State * 747796405u + 2891336453u;
184 State = ((State >> ((State >> 28u) + 4u)) ^ State) * 277803737u;
185 return (State >> 22u) ^ State;
186 }

353 // Add a queue to the Registry. Thread-safe.
354 void AddLocalQueue(TLocalQueue* QueueToAdd)
355 {
356 uint32 Index = NumLocalQueues.fetch_add(1, std::memory_order_relaxed);
357 UE_CLOGF(Index >= MaxLocalQueues, LowLevelTasks, Fatal, "Attempting to add more than the maximum allowed number of queues (%d)", MaxLocalQueues);
358
359 // std::memory_order_release to make sure values are all written to the TLocalQueue before publishing.
360 LocalQueues[Index].store(QueueToAdd, std::memory_order_release);
361 }

363 // StealItem tries to steal an Item from a Registered LocalQueue
364 // Thread-safe with AddLocalQueue
365 FTask* StealItem(uint32& CachedRandomIndex, uint32& CachedPriorityIndex, bool GetBackGroundTasks)
366 {
367 uint32 NumQueues = NumLocalQueues.load(std::memory_order_relaxed);
368 uint32 MaxPriority = GetBackGroundTasks ? int32(ETaskPriority::Count) : int32(ETaskPriority::ForegroundCount);
369 CachedRandomIndex = CachedRandomIndex % NumQueues;
370
371 for (uint32 Index = 0; Index < NumLocalQueues; Index++)
372 {
373 // Test for null in case we race on reading NumLocalQueues reserved index before the pointer is set
374 if (TLocalQueue* LocalQueue = LocalQueues[Index].load(std::memory_order_acquire))
375 {
376 for(uint32 PriorityIndex = 0; PriorityIndex < MaxPriority; PriorityIndex++)
377 {
378 FTask* Item;
379 if (LocalQueue->LocalQueues[PriorityIndex].Steal(Item))
380 {
381 return Item;
382 }
383 CachedPriorityIndex = ++CachedPriorityIndex < MaxPriority ? CachedPriorityIndex : 0;
384 }
385 CachedRandomIndex = ++CachedRandomIndex < NumQueues ? CachedRandomIndex : 0;
386 }
387 }
388 CachedPriorityIndex = 0;
389 CachedRandomIndex = TLocalQueue::InvalidIndex;
390 return nullptr;
391 }
