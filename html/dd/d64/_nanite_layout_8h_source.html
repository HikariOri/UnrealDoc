43template< typename StructType, typename CopyFuncType >
44IDetailPropertyRow& AddDefaultRow( IDetailCategoryBuilder& CategoryBuilder, StructType& Struct, FName PropertyName, CopyFuncType CopyFunc )
45{
46 TSharedPtr<FStructOnScope> TempStruct = MakeShared< FStructOnScope >( StructType::StaticStruct() );
47 StructType::StaticStruct()->CopyScriptStruct( TempStruct->GetStructMemory(), &Struct, 1 );
48 IDetailPropertyRow* PropertyRow = CategoryBuilder.AddExternalStructureProperty( TempStruct, PropertyName );
49 PropertyRow->GetPropertyHandle()->SetOnPropertyValueChanged( FSimpleDelegate::CreateLambda(
50 [ &Struct, TempStruct, CopyFunc ]
51 {
52 StructType* TempStruct2 = (StructType*)TempStruct->GetStructMemory();
53 CopyFunc( Struct, *TempStruct2 );
54 }
55 ));
56 PropertyRow->GetPropertyHandle()->SetOnChildPropertyValueChanged( FSimpleDelegate::CreateLambda(
57 [ &Struct, TempStruct, CopyFunc ]
58 {
59 StructType* TempStruct2 = (StructType*)TempStruct->GetStructMemory();
60 CopyFunc( Struct, *TempStruct2 );
61 }
62 ));
63 return *PropertyRow;
64}

66template< typename StructType, typename MemberType >
67IDetailPropertyRow& AddDefaultRow( IDetailCategoryBuilder& CategoryBuilder, StructType& Struct, MemberType (StructType::*MemberPointer), FName PropertyName )
68{
69 return AddDefaultRow(CategoryBuilder, Struct, PropertyName,
70 [MemberPointer]( StructType& Dst, StructType& Src )
71 {
72 Dst.*MemberPointer = Src.*MemberPointer;
73 } );
74}

138 static const int32 DisplayPositionPrecisionAuto = MIN_int32;
139 static const int32 DisplayPositionPrecisionMin = -6;
140 static const int32 DisplayPositionPrecisionMax = 13;

142 static int32 PositionPrecisionIndexToValue(int32 Index)
143 {
144 check(Index >= 0);
145
146 if (Index == 0)
147 {
148 return DisplayPositionPrecisionAuto;
149 }
150 else
151 {
152 int32 Value = DisplayPositionPrecisionMin + (Index - 1);
153 Value = FMath::Min(Value, DisplayPositionPrecisionMax);
154 return Value;
155 }
156 }
157
158 static int32 PositionPrecisionValueToIndex(int32 Value)
159 {
160 if (Value == DisplayPositionPrecisionAuto)
161 {
162 return 0;
163 }
164 else
165 {
166 Value = FMath::Clamp(Value, DisplayPositionPrecisionMin, DisplayPositionPrecisionMax);
167 return Value - DisplayPositionPrecisionMin + 1;
168 }
169 }
170
172 static FString PositionPrecisionValueToDisplayString(int32 Value)
173 {
174 check(Value != DisplayPositionPrecisionAuto);
175
176 if(Value <= 0)
177 {
178 return FString::Printf(TEXT("%dcm"), 1 << (-Value));
179 }
180 else
181 {
182 const float fValue = static_cast<float>(FMath::Exp2((double)-Value));
183 return FString::Printf(TEXT("1/%dcm (%.3gcm)"), 1 << Value, fValue);
184 }
185 }

322 static const int32 DisplayMinimumResidencyMinimalIndex = 0;
323 static const int32 DisplayMinimumResidencyExpRangeMin = 5;
324 static const int32 DisplayMinimumResidencyExpRangeMax = 15;
325 static const int32 DisplayMinimumResidencyFullIndex = DisplayMinimumResidencyExpRangeMax - DisplayMinimumResidencyExpRangeMin + 2;
326
327 static uint32 MinimumResidencyIndexToValue(int32 Index)
328 {
329 if (Index == DisplayMinimumResidencyMinimalIndex)
330 {
331 return 0;
332 }
333 else if (Index == DisplayMinimumResidencyFullIndex)
334 {
335 return MAX_uint32;
336 }
337 else
338 {
339 return 1u << (DisplayMinimumResidencyExpRangeMin + Index - 1);
340 }
341 }
342
343 static int32 MinimumResidencyValueToIndex(uint32 Value)
344 {
345 if (Value == 0)
346 {
347 return DisplayMinimumResidencyMinimalIndex;
348 }
349 else if (Value == MAX_uint32)
350 {
351 return DisplayMinimumResidencyFullIndex;
352 }
353 else
354 {
355 int32 Exp = (int32)FMath::CeilLogTwo(Value);
356 return FMath::Clamp(Exp, DisplayMinimumResidencyExpRangeMin, DisplayMinimumResidencyExpRangeMax) - DisplayMinimumResidencyExpRangeMin + 1;
357 }
358 }

361 static FString MinimumResidencyValueToDisplayString(uint32 Value)
362 {
363 if (Value < 1024)
364 {
365 return FString::Printf(TEXT("%dKB"), Value);
366 }
367 else
368 {
369 return FString::Printf(TEXT("%dMB"), Value >> 10);
370 }
