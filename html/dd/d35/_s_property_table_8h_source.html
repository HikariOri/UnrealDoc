488 FReply SelectRange( const TSharedPtr< IPropertyTableCell >& StartingCell, const TSharedPtr< IPropertyTableCell >& EndingCell, const TSharedPtr< IPropertyTableCell >& CellToFocus )
489 {
490 TSharedPtr< IPropertyTableCell > CurrentCell = Table->GetCurrentCell();
491
492 if ( !StartingCell.IsValid() || !EndingCell.IsValid() )
493 {
494 return FReply::Unhandled();
495 }
496
497 const bool InEditMode = CurrentCell.IsValid() ? CurrentCell->InEditMode() : false;
498
499 const int32 StartingCellRowIndex = Table->GetRows().Find( StartingCell->GetRow() );
500 const int32 EndingCellRowIndex = Table->GetRows().Find( EndingCell->GetRow() );
501
502 const int32 TopMostRowIndex = StartingCellRowIndex < EndingCellRowIndex ? StartingCellRowIndex : EndingCellRowIndex;
503 const int32 BottomMostRowIndex = StartingCellRowIndex > EndingCellRowIndex ? StartingCellRowIndex : EndingCellRowIndex;
504
505 const int32 StartingCellColumnIndex = Table->GetColumns().Find( StartingCell->GetColumn() );
506 const int32 EndingCellColumnIndex = Table->GetColumns().Find( EndingCell->GetColumn() );
507
508 const int32 LeftMostColumnIndex = StartingCellColumnIndex < EndingCellColumnIndex ? StartingCellColumnIndex : EndingCellColumnIndex;
509 const int32 RightMostColumnIndex = StartingCellColumnIndex > EndingCellColumnIndex ? StartingCellColumnIndex : EndingCellColumnIndex;
510
511 const TSharedRef< IPropertyTableCell > ActualStartingCell = Table->GetColumns()[ LeftMostColumnIndex ]->GetCell( Table->GetRows()[ TopMostRowIndex ] );
512 const TSharedRef< IPropertyTableCell > ActualEndingCell = Table->GetColumns()[ RightMostColumnIndex ]->GetCell( Table->GetRows()[ BottomMostRowIndex ] );
513
514 Table->SelectCellRange( ActualStartingCell, ActualEndingCell );
515
516 if ( CellToFocus.IsValid() )
517 {
518 Table->SetCurrentCell( CellToFocus );
519
520 if ( InEditMode )
521 {
522 CellToFocus->EnterEditMode();
523 }
524 }
525
526 return FReply::Handled();

529 FReply MoveToCell( const TSharedPtr< IPropertyTableCell >& CellToFocus )
530 {
531 TSharedPtr< IPropertyTableCell > CurrentCell = Table->GetCurrentCell();
532
533 if ( !CellToFocus.IsValid() || CellToFocus == CurrentCell )
534 {
535 return FReply::Unhandled();
536 }
537
538 bool InEditMode = CurrentCell.IsValid() ? CurrentCell->InEditMode() : false;
539 TSet< TSharedRef< IPropertyTableCell > > CellsToSelect;
540 CellsToSelect.Add( CellToFocus.ToSharedRef() );
541
542 Table->SetSelectedCells( CellsToSelect );
543 Table->SetCurrentCell( CellToFocus );
544
545 // If we dont have a valid item for the row we want to move to, then it is outside the
546 // scrolled area & we should scroll to view it.
547 TSharedPtr<ITableRow> WidgetForItem = WidgetGenerator.GetWidgetForItem( CellToFocus->GetRow() );
548 if ( !WidgetForItem.IsValid() )
549 {
550 RequestScrollIntoView( CellToFocus->GetRow() );
551 }
552
553 if ( InEditMode )
554 {
555 CellToFocus->EnterEditMode();
556 }
557
558 return FReply::Handled();

658 {

660 {
661 BreadcrumbTrail->ClearCrumbs( false );
662 BreadcrumbTrail->PushCrumb( LOCTEXT("BreadcrumbRootDisplayName", "Root"), -1 );
663
664 const TSharedPtr< FPropertyPath > RootPath = Table->GetRootPath();
665 if ( RootPath.IsValid() )

667 for (int Index = 0; Index < RootPath->GetNumProperties(); Index++)
668 {
669 const FPropertyInfo& PropInfo = RootPath->GetPropertyInfo( Index );
670
671 if ( PropInfo.ArrayIndex != INDEX_NONE )
672 {
673 FFormatNamedArguments Args;
674 Args.Add(TEXT("ArrayIndex"), PropInfo.ArrayIndex);
675 BreadcrumbTrail->PushCrumb(FText::Format(LOCTEXT("ArrayIndexWrapper", "[{ArrayIndex}]"), Args), Index);
676 }
677 else
678 {
679 BreadcrumbTrail->PushCrumb( FText::FromName(PropInfo.Property->GetFName()), Index );
680 }
