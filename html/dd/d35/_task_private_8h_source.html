122 // Implements internal logic of task prerequisites, nested tasks and deep task retraction.
123 // Implements intrusive ref-counting and so can be used with TRefCountPtr.
124 // It doesn't store task body, instead it expects a derived class to provide a task body as a parameter to `TryExecute` method. @see TExecutableTask

129 // `ExecutionFlag` is set at the beginning of execution as the most significant bit of `NumLocks` and indicates a switch
130 // of `NumLocks` from "execution prerequisites" (a number of uncompleted prerequisites that block task execution) to

226 // The task will be executed only when all prerequisites are completed. The task type must be a task handle that holds a pointer to
227 // FTaskBase as its `Pimpl` member (see Tasks::TTaskBase).
228 // Must not be called concurrently
229 bool AddPrerequisites(FTaskBase& Prerequisite)
230 {
231 TASKGRAPH_VERBOSE_EVENT_SCOPE(FTaskBase::AddPrerequisites_Single);
232
233 checkf(NumLocks.load(std::memory_order_relaxed) >= NumInitialLocks && NumLocks.load(std::memory_order_relaxed) < ExecutionFlag, TEXT("Prerequisites can be added only before the task is launched"));
234
235 // registering the task as a subsequent of the given prerequisite can cause its immediate launch by the prerequisite
236 // (if the prerequisite has been completed on another thread), so we need to keep the task locked by assuming that the
237 // prerequisite can be added successfully, and release the lock if it wasn't
238 uint32 PrevNumLocks = NumLocks.fetch_add(1, std::memory_order_relaxed); // relaxed because the following
239 // `AddSubsequent` provides required sync
240 checkf(PrevNumLocks + 1 < ExecutionFlag, TEXT("Max number of task prerequisites reached: %d"), ExecutionFlag);
241
242 if (!Prerequisite.AddSubsequent(*this)) // linearisation point, acq_rel semantic
243 {
244 // failed to add the prerequisite (too late), correct the number
245 NumLocks.fetch_sub(1, std::memory_order_relaxed); // relaxed because the previous `AddSubsequent` call provides required sync
246 return false;
247 }
248
249 Prerequisite.AddRef(); // keep it alive until this task's execution
250 Prerequisites.Push(&Prerequisite); // release memory order
251 return true;
252 }

271protected:
272 // The task will be executed only when all prerequisites are completed.
273 // Must not be called concurrently.
274 // @param InPrerequisites - an iterable collection of tasks
275 template<typename PrerequisiteCollectionType, decltype(std::declval<PrerequisiteCollectionType>().begin())* = nullptr>
276 void AddPrerequisites(const PrerequisiteCollectionType& InPrerequisites, bool bLockPrerequisite)
277 {
278 const int32 PrerequisiteCount = GetNum(InPrerequisites);
279 if (UNLIKELY(PrerequisiteCount <= 0))
280 {
281 return;
282 }
283
284 TASKGRAPH_VERBOSE_EVENT_SCOPE(FTaskBase::AddPrerequisites_Collection);
285
286 checkf(NumLocks.load(std::memory_order_relaxed) >= NumInitialLocks && NumLocks.load(std::memory_order_relaxed) < ExecutionFlag, TEXT("Prerequisites can be added only before the task is launched"));
287
288 // registering the task as a subsequent of the given prerequisite can cause its immediate launch by the prerequisite
289 // (if the prerequisite has been completed on another thread), so we need to keep the task locked by assuming that the
290 // prerequisite can be added successfully, and release the lock if it wasn't
291 uint32 PrevNumLocks = NumLocks.fetch_add(PrerequisiteCount, std::memory_order_relaxed); // relaxed because the following
292 // `AddSubsequent` provides required sync
293
294 bool bUnlockPrerequisites = false;
295
296 uint32 NumCompletedPrerequisites = 0;
297 for (auto& Prereq : InPrerequisites)
298 {
299 // prerequisites can be either `FTaskBase*` or its Pimpl handle
300 FTaskBase* Prerequisite;
301 using FPrerequisiteType = std::decay_t<decltype(*std::declval<PrerequisiteCollectionType>().begin())>;
302 if constexpr (std::is_same_v<FPrerequisiteType, FTaskBase*>)
303 {
304 Prerequisite = Prereq;
305 }
306 else if constexpr (std::is_same_v<FPrerequisiteType, FGraphEventRef>)
307 {
308 Prerequisite = Prereq.GetReference();
309 }
310 else if constexpr (std::is_pointer_v<FPrerequisiteType>)
311 {
312 Prerequisite = Prereq->Pimpl;
313 }
314 else
315 {
316 Prerequisite = Prereq.Pimpl;
317 }
318
319 if (Prerequisite == nullptr)
320 {
321 ++NumCompletedPrerequisites;
322 continue;
323 }
324
325 if (Prerequisite->AddSubsequent(*this)) // acq_rel memory order
326 {
327 Prerequisite->AddRef(); // keep it alive until this task's execution
328 if (bLockPrerequisite)
329 {
330 Prerequisites.Lock();
331 bLockPrerequisite = false; // don't lock again
332 bUnlockPrerequisites = true;
333 }
334
335 Prerequisites.PushNoLock(Prerequisite); // relaxed memory order
336 }
337 else
338 {
339 ++NumCompletedPrerequisites;
340 }
341 }
342
343 // This check is here to avoid the data dependency on PrevNumLocks.
344 checkf(PrevNumLocks + GetNum(InPrerequisites) < ExecutionFlag, TEXT("Max number of nested tasks reached: %d"), ExecutionFlag);
345
346 if (bUnlockPrerequisites)
347 {
348 Prerequisites.Unlock();
349 }
350
351 // unlock for prerequisites that weren't added
352 if (NumCompletedPrerequisites)
353 {
354 NumLocks.fetch_sub(NumCompletedPrerequisites, std::memory_order_release);
355 }
356 }

362 void AddPrerequisites(const PrerequisiteCollectionType& InPrerequisites)
363 {
364 AddPrerequisites(InPrerequisites, true /* bLockPrerequisites */);

394 // Tries to schedule task execution but has additional tracking to ensure a task is not scheduled more than once.
395 // This method should only be used when scheduling more than once is allowed/practical (e.g. FTaskEvent).
396 // Returns false if the task has been triggered before or has incomplete dependencies (prerequisites or is blocked by a pipe).
397 // In the latter case, the task will be automatically scheduled when all dependencies are completed.
398 bool Trigger(uint64 TaskSize)
399 {
400 if (TaskTriggered.exchange(true, std::memory_order_relaxed))
401 {
402 // Task is already launched
403 return false;
404 }
405
406 // Now that we know we have triggered the task (it might not yet be launched/running based on dependencies) it's now in the
407 // task system and can outlive external references, so we need to keep it alive by holding an internal reference.
408 // It will be released when completed.
409 AddRef();
410 return TryLaunch(TaskSize);
411 }

449 // adds a nested task that must be completed before the parent (this) is completed
450 void AddNested(FTaskBase& Nested)
451 {
452 TASKGRAPH_VERBOSE_EVENT_SCOPE(FTaskBase::AddNested);
453
454 uint32 PrevNumLocks = NumLocks.fetch_add(1, std::memory_order_relaxed); // in case we'll succeed in adding subsequent,
455 // "happens before" registering this task as a subsequent
456 checkf(PrevNumLocks + 1 < TNumericLimits<uint32>::Max(), TEXT("Max number of nested tasks reached: %d"), TNumericLimits<uint32>::Max() - ExecutionFlag);
457 checkf(PrevNumLocks > ExecutionFlag, TEXT("Internal error: nested tasks can be added only during parent's execution (%u)"), PrevNumLocks);
458
459 if (Nested.AddSubsequent(*this)) // "release" memory order
460 {
461 Nested.AddRef(); // keep it alive as we store it in `Prerequisites` and we can need it to try to retract it. it's released on closing the task
462 Prerequisites.Push(&Nested);
463 }
464 else
465 {
466 NumLocks.fetch_sub(1, std::memory_order_relaxed);
467 }
468 }

495 // tries to get execution permission and if successful, executes given task body and completes the task if there're no pending nested tasks.
496 // does all required accounting before/after task execution. the task can be deleted as a result of this call.
497 // @returns true if the task was executed by the current thread
498 bool TryExecuteTask()
499 {
500 TASKGRAPH_VERBOSE_EVENT_SCOPE(FTaskBase::TryExecuteTask);
501
502 if (!TrySetExecutionFlag())
503 {
504 return false;
505 }
506
507 AddRef(); // `LowLevelTask` will automatically release the internal reference after execution, but there can be pending nested tasks, so keep it alive
508 // it's released either later here if the task is closed, or when the last nested task is completed and unlocks its parent (in `TryUnlock`)
509
510 ReleasePrerequisites();
511
512 FTaskBase* PrevTask = ExchangeCurrentTask(this);
513 ExecutingThreadId.store(FPlatformTLS::GetCurrentThreadId(), std::memory_order_relaxed);
514
515 if (GetPipe() != nullptr)
516 {
517 StartPipeExecution();
518 }
519
520 {
521 UE::FInheritedContextScope InheritedContextScope = RestoreInheritedContext();
522 TaskTrace::FTaskTimingEventScope TaskEventScope(GetTraceId());
523 TASKGRAPH_VERBOSE_EVENT_SCOPE(FTaskBase::ExecuteTask);
524 ExecuteTask();
525 }
526
527 if (GetPipe() != nullptr)
528 {
529 FinishPipeExecution();
530 }
531
532 ExecutingThreadId.store(FThread::InvalidThreadId, std::memory_order_relaxed); // no need to sync with loads as they matter only if
533 // executed by the same thread
534 ExchangeCurrentTask(PrevTask);
535
536 // close the task if there are no pending nested tasks
537 uint32 LocalNumLocks = NumLocks.fetch_sub(1, std::memory_order_acq_rel) - 1; // "release" to make task execution "happen before" this, and "acquire" to
538 // "sync with" another thread that completed the last nested task
539 if (LocalNumLocks == ExecutionFlag) // unlocked (no pending nested tasks)
540 {
541 Close();
542 Release(); // the internal reference that kept the task alive for nested tasks
543 } // else there're non completed nested tasks, the last one will unlock, close and release the parent (this task)
544
545 return true;
546 }

554 // A task can be locked for execution (by prerequisites or if it's not launched yet) or for completion (by nested tasks).
555 // This method is called to unlock the task and so can result in its scheduling (and execution) or completion
556 bool TryUnlock(bool& bWakeUpWorker)
557 {
558 TASKGRAPH_VERBOSE_EVENT_SCOPE(FTaskBase::TryUnlock);
559
560 FPipe* LocalPipe = GetPipe(); // cache data locally so we won't need to touch the member (read below)
561
562 uint32 PrevNumLocks = NumLocks.fetch_sub(1, std::memory_order_acq_rel); // `acq_rel` to make it happen after task
563 // preparation and before launching it
564 // the task can be dead already as the prev line can remove the lock hold for this execution path, another thread(s) can unlock
565 // the task, execute, complete and delete it. thus before touching any members or calling methods we need to make sure
566 // the task can't be destroyed concurrently
567
568 uint32 LocalNumLocks = PrevNumLocks - 1;
569
570 if (PrevNumLocks < ExecutionFlag)
571 {
572 // pre-execution state, try to schedule the task
573
574 checkf(PrevNumLocks != 0, TEXT("The task is not locked"));
575
576 bool bPrerequisitesCompleted = LocalPipe == nullptr ? LocalNumLocks == 0 : LocalNumLocks <= 1; // the only remaining lock is pipe's one (if any)
577 if (!bPrerequisitesCompleted)
578 {
579 return false;
580 }
581
582 // this thread unlocked the task, no other thread can reach this point concurrently, we can touch the task again
583
584 if (LocalPipe != nullptr)
585 {
586 bool bFirstPipingAttempt = LocalNumLocks == 1;
587 if (bFirstPipingAttempt)
588 {
589 FTaskBase* PrevPipedTask = TryPushIntoPipe();
590 if (PrevPipedTask != nullptr) // the pipe is blocked
591 {
592 // the prev task in pipe's chain becomes this task's prerequisite, to enabled piped task retraction.
593 // its ref count already accounted for this ref. the ref will be released when the prereq is not needed anymore
594 Prerequisites.Push(PrevPipedTask);
595 return false;
596 }
597
598 NumLocks.store(0, std::memory_order_release); // release pipe's lock
599 }
600 }
601
602 if (ExtendedPriority == EExtendedTaskPriority::Inline)
603 {
604 // "inline" tasks are not scheduled but executed straight away
605 TryExecuteTask(); // result doesn't matter, this can fail if task retraction jumped in and got execution
606 // permission between this thread unlocked the task and tried to execute it
607 ReleaseInternalReference();
608
609 // Use-after-free territory, do not touch any of the task's properties here.
610 }
611 else if (ExtendedPriority == EExtendedTaskPriority::TaskEvent)
612 {
613 // task events have nothing to execute, try to close it. task retraction can jump in and close the task event,
614 // so this thread still needs to check execution permission
615 if (TrySetExecutionFlag())
616 {
617 // task events are used as an empty prerequisites/subsequents
618 ReleasePrerequisites();
619 Close();
620 ReleaseInternalReference();
621
622 // Use-after-free territory, do not touch any of the task's properties here.
623 }
624 }
625 else
626 {
627 Schedule(bWakeUpWorker);
628
629 // Use-after-free territory, do not touch any of the task's properties here.
630 }
631
632 return true;
633 }
634
635 // execution already started (at least), this is nested tasks unlocking their parent
636 checkf(PrevNumLocks != ExecutionFlag, TEXT("The task is not locked"));
637 if (LocalNumLocks != ExecutionFlag) // still locked
638 {
639 return false;
640 }
641
642 // this thread unlocked the task, no other thread can reach this point concurrently, we can touch the task again
643 Close();
644 Release(); // the internal reference that kept the task alive for nested tasks
645
646 // Use-after-free territory, do not touch any of the task's properties here.
647
648 return true;
649 }

656 // only one thread can successfully set execution flag, that grants task execution permission
657 // @returns false if another thread got execution permission first
658 bool TrySetExecutionFlag()
659 {
660 uint32 ExpectedUnlocked = 0;
661 // set the execution flag and simultenously lock it (+1) so a nested task completion doesn't close it before its execution is finished
662 return NumLocks.compare_exchange_strong(ExpectedUnlocked, ExecutionFlag + 1, std::memory_order_acq_rel, std::memory_order_relaxed); // on success

976 // task retraction of multiple tasks, with timeout. The timeout is rounded up to any successful task execution, which means that it can
977 // time out only in-between individual task retractions.
978 // WARNING: the function can return `true` even if some tasks are still not completed. The `true` means only that the tasks are executed
979 // and have no other pending dependencies, but can be still in the process of completion (concurrently). The caller still needs to wait
980 // for completion.
981 template<typename TaskCollectionType>
982 bool TryRetractAndExecute(const TaskCollectionType& Tasks, FTimeout Timeout)
983 {
984 bool bResult = true;
985
986 for (auto& Task : Tasks)
987 {
988 if (Task.IsValid() && !Task.Pimpl->TryRetractAndExecute(Timeout))
989 {
990 bResult = false; // do not stop here to let this thread to help in executing tasks as much as possible, as it's waiting for their completion anyway
991 }
992
993 if (Timeout.IsExpired())
994 {
995 return false;
996 }
997 }
998
999 return bResult;
1000 }
