47 class TReferenceControllerBase

49 using RefCountType = std::conditional_t<Mode == ESPMode::ThreadSafe, std::atomic<int32>, int32>;

51 public:

54 // Number of shared references to this object. When this count reaches zero, the associated object
55 // will be destroyed (even if there are still weak references!), but not the reference controller.
56 //
57 // This starts at 1 because we create reference controllers via the construction of a TSharedPtr,
58 // and that is the first reference. There is no point in starting at 0 and then incrementing it.
59 RefCountType SharedReferenceCount{1};
60
61 // Number of weak references to this object. If there are any shared references, that counts as one
62 // weak reference too. When this count reaches zero, the reference controller will be deleted.
63 //
64 // This starts at 1 because it represents the shared reference that we are also initializing
65 // SharedReferenceCount with.
66 RefCountType WeakReferenceCount{1};
67
69 virtual void DestroyObject() = 0;
70
71 virtual ~TReferenceControllerBase() = default;

74 inline int32 GetSharedReferenceCount() const
75 {
76 if constexpr (Mode == ESPMode::ThreadSafe)
77 {
78 // A 'live' shared reference count is unstable by nature and so there's no benefit
79 // to try and enforce memory ordering around the reading of it.
80 //
81 // This is equivalent to https://en.cppreference.com/w/cpp/memory/shared_ptr/use_count
82
83 int32 Count = 0;
84
85 UE_AUTORTFM_OPEN
86 {
87 // This reference count may be accessed by multiple threads
88 Count = SharedReferenceCount.load(std::memory_order_relaxed);
89 }
90 ;
91
92 return Count;
93 }
94 else
95 {
96 return SharedReferenceCount;
97 }
98 }
99
101 UE_FORCEINLINE_HINT bool IsUnique() const
102 {
103 return 1 == GetSharedReferenceCount();
104 }

107 inline void AddSharedReference()
108 {
109 if constexpr (Mode == ESPMode::ThreadSafe)
110 {
111 // Incrementing a reference count with relaxed ordering is always safe because no other action is taken
112 // in response to the increment, so there's nothing to order with.
113
114#if defined(_MSC_VER) && (defined(_M_X64) || defined(_M_IX86))
115 UE_AUTORTFM_OPEN
116 {
117 // We do a regular SC increment here because it maps to an _InterlockedIncrement (lock inc).
118 // The codegen for a relaxed fetch_add is actually much worse under MSVC (lock xadd).
119 ++SharedReferenceCount;
120 };
121#else
122 UE_AUTORTFM_OPEN
123 {
124 SharedReferenceCount.fetch_add(1, std::memory_order_relaxed);
125 };
126#endif
127
128 // If the transaction would abort, we need to undo adding the shared reference.
129 UE_AUTORTFM_ONABORT(this)
130 {
131 ReleaseSharedReference();
132 };
133 }
134 else
135 {
136 ++SharedReferenceCount;
137 }
138 }

145 bool ConditionallyAddSharedReference()
146 {
147 if constexpr (Mode == ESPMode::ThreadSafe)
148 {
149 bool bSucceeded = false;
150
151 UE_AUTORTFM_OPEN
152 {
153 // See AddSharedReference for the same reasons that std::memory_order_relaxed is used in this function.
154
155 // Peek at the current shared reference count. Remember, this value may be updated by
156 // multiple threads.
157 int32 OriginalCount = SharedReferenceCount.load(std::memory_order_relaxed);
158
159 for (; ; )
160 {
161 if (OriginalCount == 0)
162 {
163 // Never add a shared reference if the pointer has already expired
164 bSucceeded = false;
165 break;
166 }
167
168 // Attempt to increment the reference count.
169 //
170 // We need to make sure that we never revive a counter that has already expired, so if the
171 // actual value what we expected (because it was touched by another thread), then we'll try
172 // again. Note that only in very unusual cases will this actually have to loop.
173 //
174 // We do a weak read here because we require a loop and this is the recommendation:
175 //
176 // https://en.cppreference.com/w/cpp/atomic/atomic/compare_exchange
177 //
178 // > When a compare-and-exchange is in a loop, the weak version will yield better performance on some platforms.
179 // > When a weak compare-and-exchange would require a loop and a strong one would not, the strong one is preferable
180 if (SharedReferenceCount.compare_exchange_weak(OriginalCount, OriginalCount + 1, std::memory_order_relaxed))
181 {
182 bSucceeded = true;
183 break;
184 }
185 }
186 };
187
188 // If we succeeded in taking a shared reference count, we need to undo that on an abort.
189 if (bSucceeded)
190 {
191 UE_AUTORTFM_ONABORT(this)
192 {
193 ReleaseSharedReference();
194 };
195 }
196
197 return bSucceeded;
198 }
199 else
200 {
201 if( SharedReferenceCount == 0 )
202 {
203 // Never add a shared reference if the pointer has already expired
204 return false;
205 }
206
207 ++SharedReferenceCount;
208 return true;
209 }
210 }
211
213 inline void ReleaseSharedReference()
214 {
215 if constexpr (Mode == ESPMode::ThreadSafe)
216 {
217 UE_AUTORTFM_ONCOMMIT(this)
218 {
219 // std::memory_order_acq_rel is used here so that, if we do end up executing the destructor, it's not possible
220 // for side effects from executing the destructor end up being visible before we've determined that the shared
221 // reference count is actually zero.
222
223 int32 OldSharedCount = SharedReferenceCount.fetch_sub(1, std::memory_order_acq_rel);
224 checkSlow(OldSharedCount > 0);
225 if (OldSharedCount == 1)
226 {
227 // Last shared reference was released! Destroy the referenced object.
228 DestroyObject();
229
230 // No more shared referencers, so decrement the weak reference count by one. When the weak
231 // reference count reaches zero, this object will be deleted.
232 ReleaseWeakReference();
233 }
234 };
235 }
236 else
237 {
238 checkSlow( SharedReferenceCount > 0 );
239
240 if( --SharedReferenceCount == 0 )
241 {
242 // Last shared reference was released! Destroy the referenced object.
243 DestroyObject();
244
245 // No more shared referencers, so decrement the weak reference count by one. When the weak
246 // reference count reaches zero, this object will be deleted.
247 ReleaseWeakReference();
248 }
249 }
250 }

264 inline void AddWeakReference()
265 {
266 if constexpr (Mode == ESPMode::ThreadSafe)
267 {
268 // See AddSharedReference for the same reasons that std::memory_order_relaxed is used in this function.

270#if defined(_MSC_VER) && (defined(_M_X64) || defined(_M_IX86))

273 // We do a regular SC increment here because it maps to an _InterlockedIncrement (lock inc).

276 };
277#else

280 WeakReferenceCount.fetch_add(1, std::memory_order_relaxed);
281 };
282#endif

288 };

292 ++WeakReferenceCount;

297 void ReleaseWeakReference()
298 {
299 if constexpr (Mode == ESPMode::ThreadSafe)
300 {
301 UE_AUTORTFM_ONCOMMIT(this)
302 {
303 // See ReleaseSharedReference for the same reasons that std::memory_order_acq_rel is used in this function.
304
305 int32 OldWeakCount = WeakReferenceCount.fetch_sub(1, std::memory_order_acq_rel);
306 checkSlow(OldWeakCount > 0);
307 if (OldWeakCount == 1)
308 {
309 // Disable this if running clang's static analyzer. Passing shared pointers
310 // and references to functions it cannot reason about, produces false
311 // positives about use-after-free in the TSharedPtr/TSharedRef destructors.
312#if !defined(__clang_analyzer__)
313 // No more references to this reference count. Destroy it!
314 delete this;
315#endif
316 }
317 };
318 }
319 else
320 {
321 checkSlow( WeakReferenceCount > 0 );
322
323 if( --WeakReferenceCount == 0 )
324 {
325 // No more references to this reference count. Destroy it!
326#if !defined(__clang_analyzer__)
327 delete this;
328#endif
329 }
330 }

338 // A helper class that efficiently stores a custom deleter and is intended to be derived from. If the custom deleter is an empty class,
339 // TDeleterHolder derives from it exploiting empty base optimisation (https://en.cppreference.com/w/cpp/language/ebo). Otherwise
340 // it stores the custom deleter as a member to allow a function pointer to be used as a custom deleter (a function pointer can't
341 // be a base class)

343 struct TDeleterHolder : private DeleterType
344 {
345 explicit TDeleterHolder(DeleterType&& Arg)
346 : DeleterType(MoveTemp(Arg))
347 {
348 }

351 void InvokeDeleter(ObjectType * Object)
352 {
353 Invoke(*static_cast<DeleterType*>(this), Object);
354 }

416 // public:

420 // explicit FMyType(FPrivateToken, int32 Int, float Real, const TCHAR* String);
421 // };

424 // TSharedPtr<FMyType> Val = MakeShared<FMyType>(FMyType::FPrivateToken{}, 5, 3.14f, TEXT("Banana"));

574 // shared reference count
575 if (ReferenceController != nullptr)
576 {
577 // Attempt to elevate a weak reference to a shared one. For this to work, the object this
578 // weak counter is associated with must already have at least one shared reference. We'll
579 // never revive a pointer that has already expired!
580 if (!ReferenceController->ConditionallyAddSharedReference())
581 {
582 ReferenceController = nullptr;
583 }
584 }
585 }
586
589 FSharedReferencer( FWeakReferencer< Mode >&& InWeakReference )
590 : ReferenceController( InWeakReference.ReferenceController )
591 {
592 // If the incoming reference had an object associated with it, then go ahead and increment the
593 // shared reference count
594 if( ReferenceController != nullptr )
595 {
596 // Attempt to elevate a weak reference to a shared one. For this to work, the object this
597 // weak counter is associated with must already have at least one shared reference. We'll
598 // never revive a pointer that has already expired!
599 if( !ReferenceController->ConditionallyAddSharedReference() )
600 {
601 ReferenceController = nullptr;
602 }
603
604 // Tell the reference counter object that we're no longer referencing the object with
605 // this weak pointer
606 InWeakReference.ReferenceController->ReleaseWeakReference();
607 InWeakReference.ReferenceController = nullptr;
608 }
609 }
610
612 inline ~FSharedReferencer()
613 {
614 TReferenceControllerBase<Mode>::ReleaseSharedReferenceNoInline(ReferenceController);
615 }
616
619 inline FSharedReferencer& operator=( FSharedReferencer const& InSharedReference )
620 {
621 // Make sure we're not be reassigned to ourself!
622 auto NewReferenceController = InSharedReference.ReferenceController;
623 if( NewReferenceController != ReferenceController )
624 {
625 // First, add a shared reference to the new object
626 if( NewReferenceController != nullptr )
627 {
628 NewReferenceController->AddSharedReference();
629 }
630
631 // Release shared reference to the old object
632 if( ReferenceController != nullptr )
633 {
634 ReferenceController->ReleaseSharedReference();
635 }
636
637 // Assume ownership of the assigned reference counter
638 ReferenceController = NewReferenceController;
639 }
640
641 return *this;
642 }
643
646 inline FSharedReferencer& operator=( FSharedReferencer&& InSharedReference )
647 {
648 // Make sure we're not be reassigned to ourself!
649 auto NewReferenceController = InSharedReference.ReferenceController;
650 auto OldReferenceController = ReferenceController;
651 if( NewReferenceController != OldReferenceController )
652 {
653 // Assume ownership of the assigned reference counter
654 InSharedReference.ReferenceController = nullptr;
655 ReferenceController = NewReferenceController;
656
657 // Release shared reference to the old object
658 if( OldReferenceController != nullptr )
659 {
660 OldReferenceController->ReleaseSharedReference();
661 }
662 }
663
664 return *this;
665 }
