34 FORCEINLINE TUniquePtr<IAudioCaptureStream> FAudioCapture::CreateImpl()
35 {
36 IModularFeatures::Get().LockModularFeatureList();
37 TArray<IAudioCaptureFactory*> AudioCaptureStreamFactories = IModularFeatures::Get().GetModularFeatureImplementations<IAudioCaptureFactory>(IAudioCaptureFactory::GetModularFeatureName());
38 IModularFeatures::Get().UnlockModularFeatureList();
39
40 // For now, just return the first audio capture stream implemented. We can make this configurable at a later point.
41 if (AudioCaptureStreamFactories.Num() > 0 && AudioCaptureStreamFactories[0] != nullptr && FApp::CanEverRenderAudio())
42 {
43 return AudioCaptureStreamFactories[0]->CreateNewAudioCaptureStream();
44 }
45 else
46 {
47 UE_LOGF(LogAudioCaptureCore, Display, "No Audio Capture implementations found. Audio input will be silent.");
48 return TUniquePtr<IAudioCaptureStream>(new FNullAudioCaptureDevice());
49 }
50 }
