89private:
90 void UpdateValue()
91 {
92 const double* Time = TimestampsIterator.GetCurrentItem();
93 if (Time)
94 {
95 bool bIsNewFrame = false;
96 while (FrameStartTimesIterator && *FrameStartTimesIterator < *Time)
97 {
98 bIsNewFrame = true;
99 ++FrameStartTimesIterator;
100 }
101 CurrentTime = *Time;
102 CurrentOp = bIsNewFrame ? ECounterOpType::Set : *OpTypesIterator;
103 CurrentOpArgument = *OpArgumentsIterator;
104 switch (CurrentOp)
105 {
106 case ECounterOpType::Set:
107 CurrentValue = CurrentOpArgument;
108 break;
109 case ECounterOpType::Add:
110 CurrentValue += CurrentOpArgument;
111 break;
112 }
113 }
114 }
115
116 TArray64<double>::TConstIterator FrameStartTimesIterator;
117#if USE_VARIABLE_PAGED_ARRAY
118 TVariablePagedArray<double>::TIterator TimestampsIterator;
119 TVariablePagedArray<ECounterOpType>::TIterator OpTypesIterator;
120 typename TVariablePagedArray<ValueType>::TIterator OpArgumentsIterator;
121#else
122 TPagedArray<double>::TIterator TimestampsIterator;
123 TPagedArray<ECounterOpType>::TIterator OpTypesIterator;
124 typename TPagedArray<ValueType>::TIterator OpArgumentsIterator;
125#endif
126 double CurrentTime;
127 ECounterOpType CurrentOp;
128 ValueType CurrentOpArgument;
129 ValueType CurrentValue;

133class TCounterData

135public:
136 typedef TCounterDataIterator<ValueType> TIterator;
137
138 TCounterData(ILinearAllocator& Allocator)
139 : Timestamps(Allocator, 1024)
140 , OpTypes(Allocator, 1024)
141 , OpArguments(Allocator, 1024)

143
144 }
145
146 void InsertOp(double Timestamp, ECounterOpType OpType, ValueType OpArgument)
147 {
148 uint64 InsertionIndex;
149 if (Timestamps.Num() == 0 || Timestamps.Last() <= Timestamp)
150 {
151 InsertionIndex = Timestamps.Num();
152 }
153 else if (Timestamps.First() >= Timestamp)
154 {
155 InsertionIndex = 0;
156 }
157 else
158 {
159 auto TimestampIterator = Timestamps.GetIteratorFromItem(Timestamps.Num() - 1);
160 auto CurrentPage = TimestampIterator.GetCurrentPage();
161 while (*GetFirstItem(*CurrentPage) > Timestamp)
162 {
163 CurrentPage = TimestampIterator.PrevPage();
164 }
165 check(CurrentPage != nullptr);
166 uint64 PageInsertionIndex = Algo::LowerBound(*CurrentPage, Timestamp);
167#if USE_VARIABLE_PAGED_ARRAY
168 InsertionIndex = TimestampIterator.GetCurrentItemIndex() + PageInsertionIndex + 1 - CurrentPage->ItemCount;
169#else
170 InsertionIndex = TimestampIterator.GetCurrentPageIndex() * Timestamps.GetPageSize() + PageInsertionIndex;
171#endif
172 check(InsertionIndex <= Timestamps.Num());
173 }
174 Timestamps.Insert(InsertionIndex) = Timestamp;
175 OpTypes.Insert(InsertionIndex) = OpType;
176 OpArguments.Insert(InsertionIndex) = OpArgument;
177 }
178
179 uint64 Num() const
180 {
181 return Timestamps.Num();
182 }
183
184 TIterator GetIterator(const TArray64<double>& FrameStartTimes) const
185 {
186 return TIterator(*this, FrameStartTimes);
187 }
188
189 double GetFirstTimestamp() const { return Timestamps.First(); }
190 double GetLastTimestamp() const { return Timestamps.Last(); }
191
192private:
193 template<typename IteratorValueType>
194 friend class TCounterDataIterator;
195
196#if USE_VARIABLE_PAGED_ARRAY
197 TVariablePagedArray<double> Timestamps;
198 TVariablePagedArray<ECounterOpType> OpTypes;
199 TVariablePagedArray<ValueType> OpArguments;
200#else
201 TPagedArray<double> Timestamps;
202 TPagedArray<ECounterOpType> OpTypes;
203 TPagedArray<ValueType> OpArguments;
204#endif
205};
