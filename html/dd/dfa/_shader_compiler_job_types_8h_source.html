77enum class EShaderCompileJobStatus : uint8
78{
79 /* Initial status, created but not yet manipulated */
80 Unset,
81 /* Ready for execution, set after preprocessing and input hash computation is completed */
82 Ready,
83 /* Job execution was skipped due to a preprocessing failure */
84 Skipped,
85 /* Job was cancelled by owning system */
86 Cancelled,
87 /* DDC query for job has been submitted and is waiting on results */
88 PendingDDC,
89 /* Job is queued in the shader compilation system, but not yet picked up by a compilation thread (i.e. not a cache or DDC hit) */
90 Queued,
91 /* Job has been picked up by a compilation thread, has been submitted for compilation is pending execution completion */
92 Pending,
93 /* Job has been completed and results are available for processing */
94 Completed,
95 /* Job results have been read and processed into the owning shadermap, and job has been released by the shader compilation system.
96 * Note that the job may remain alive if other systems are holding references to it at this point.
97 */
98 Released
99};

101enum class EShaderCompileJobCompletionMethod
102{
103 /* Initial value, execution method not yet determined */
104 Unset,
105 /* Job was found in the in-memory job cache */
106 CacheHit,
107 /* Job was found in DDC */
108 DdcHit,
109 /* Job was a duplicate of another and shared its results */
110 Duplicate,
111 /* Job was executed by a local shader compilation mechanism (local shader compile thread or local fallback) */
112 LocalExecution,
113 /* Job was executed by a distributed shader compilation mechanism (note that this could still have executed on the local machine, but coordinated by a distributed compiler) */
114 DistributedExecution,
115};

120class FShaderCompileJobStatus
121{
122public:
123 void Reset()
124 {
125 // Unset status before the input hash since GetInputHash() uses it as a substitute for lock guards
126 Status.store(EShaderCompileJobStatus::Unset);
127 CompletionMethod.store(EShaderCompileJobCompletionMethod::Unset);
128
129 InputHash = nullptr;
130 }
131
132 /* Returns true if the job has been completed (i.e. is in "Completed" or "Released" phase) */
133 bool IsCompleted() const
134 {
135 EShaderCompileJobStatus LocalStatus = GetStatus();
136 return
137 (LocalStatus == EShaderCompileJobStatus::Completed) ||
138 (LocalStatus == EShaderCompileJobStatus::Released);
139 }
140
141 /* Returns true if the owning job is completed but the compile step was not executed. */
142 bool WasCompilationSkipped() const
143 {
144 EShaderCompileJobStatus LocalStatus = GetStatus();
145 EShaderCompileJobCompletionMethod LocalCompletionMethod = GetCompletionMethod();
146 bool bIsCompleted = IsCompleted();
147
148 return
149 (LocalStatus == EShaderCompileJobStatus::Skipped) ||
150 (bIsCompleted &&
151 (LocalCompletionMethod == EShaderCompileJobCompletionMethod::Duplicate ||
152 LocalCompletionMethod == EShaderCompileJobCompletionMethod::CacheHit ||
153 LocalCompletionMethod == EShaderCompileJobCompletionMethod::DdcHit));
154 }
155
156 void SetStatus(EShaderCompileJobStatus InStatus)
157 {
158 Status.store(InStatus);
159 }

161 void SetStatus(EShaderCompileJobStatus InStatus, EShaderCompileJobCompletionMethod InCompletionMethod)
162 {
163 SetCompletionMethod(InCompletionMethod);
164 SetStatus(InStatus);
165 }
166
167 void SetCompletionMethod(EShaderCompileJobCompletionMethod InCompletionMethod)
168 {
169 CompletionMethod.store(InCompletionMethod);
170 }
171
172 EShaderCompileJobCompletionMethod GetCompletionMethod() const
173 {
174 return CompletionMethod.load();
175 }
176
177 EShaderCompileJobStatus GetStatus() const
178 {
179 return Status.load();
180 }

183 {
