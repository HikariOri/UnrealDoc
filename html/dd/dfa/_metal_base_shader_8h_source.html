137template<typename BaseResourceType, int32 ShaderType>
138void TMetalBaseShader<BaseResourceType, ShaderType>::Init(const FRHICreateShaderDesc& CreateShaderDesc, FMetalCodeHeader& Header, MTLLibraryPtr InLibrary, bool bBinaryCompiled, bool bUseExtendedCodeSection)
139{
140 FShaderCodeReader ShaderCode(CreateShaderDesc.Code);
141
142 FMemoryReaderView Ar(CreateShaderDesc.Code, true);
143
144 Ar.SetLimitSize(ShaderCode.GetActualShaderCodeSize());
145
146 // was the shader already compiled offline?
147 uint8 OfflineCompiledFlag;
148 Ar << OfflineCompiledFlag;
149 check(OfflineCompiledFlag == 0 || OfflineCompiledFlag == 1);
150
151 // get the header
152 Header.Serialize(Ar, BaseResourceType::ShaderResourceTable);
153
154 ValidateVersion(Header.Version);
155
156 SourceLen = Header.SourceLen;
157 SourceCRC = Header.SourceCRC;
158
159 // If this triggers than a level above us has failed to provide valid shader data and the cook is probably bogus
160 UE_CLOGF(Header.SourceLen == 0 || Header.SourceCRC == 0, LogMetal, Fatal, "Invalid Shader Bytecode provided.");
161
162 // remember where the header ended and code (precompiled or source) begins
163 int32 CodeOffset = Ar.Tell();
164 uint32 BufferSize = ShaderCode.GetActualShaderCodeSize() - CodeOffset;
165 const ANSICHAR* SourceCode = (ANSICHAR*)CreateShaderDesc.Code.GetData() + CodeOffset;
166
167 if (bUseExtendedCodeSection)
168 {
169 int32 ExtendedCodeDataOffset = Header.RayTracing.ExtendedCodeDataOffset;
170 int32 ExtendedCodeDataSize = Header.RayTracing.ExtendedCodeDataSize;
171
172 ANSICHAR const* SourceCodeExtended = nullptr;
173 if (ExtendedCodeDataOffset != -1 && ExtendedCodeDataSize != -1)
174 {
175 int32 ExtendedCodeStartOffset = CodeOffset + ExtendedCodeDataOffset;
176 SourceCodeExtended = (ANSICHAR*)CreateShaderDesc.Code.GetData() + ExtendedCodeStartOffset;
177 }
178 else
179 {
180 SourceCodeExtended = (ANSICHAR*)ShaderCode.FindOptionalDataAndSize(EShaderOptionalDataKey::VendorExtension, ExtendedCodeDataSize);
181 }
182
183 check(SourceCodeExtended);
184 check(ExtendedCodeDataSize != 0);
185
186 BufferSize = (uint32)ExtendedCodeDataSize;
187 SourceCode = SourceCodeExtended;
188
189 bIsExtendedBytecode = true;
190 }
191
192 // Only archived shaders should be in here.
193 UE_CLOG(InLibrary && !(Header.CompileFlags & (1 << CFLAG_Archive)), LogMetal, Warning, TEXT("Shader being loaded wasn't marked for archiving but a MTLLibrary was provided - this is unsupported."));
194
195 if (!OfflineCompiledFlag)
196 {
197 UE_LOGF(LogMetal, Display, "Loaded a text shader (will be slower to load)");
198 }
199
200 const bool bOfflineCompile = (OfflineCompiledFlag > 0);
201 const bool bUseMetalShaderConverter = EnumHasAnyFlags(Header.Bindings.Flags, EMetalBindingsFlags::UseMetalShaderConverter);
202
203 const ANSICHAR* ShaderSource = ShaderCode.FindOptionalData(EShaderOptionalDataKey::SourceCode);
204 bool bHasShaderSource = false;
205 static bool bForceTextShaders = false;
206
207 if(!bUseMetalShaderConverter)
208 {
209 bForceTextShaders = FParse::Param(FCommandLine::Get(),TEXT("metalshaderdebug"));
210 bHasShaderSource = (ShaderSource && FCStringAnsi::Strlen(ShaderSource) > 0);
211 }
212
213 if (!bHasShaderSource)
214 {
215#if !UE_BUILD_SHIPPING
216 if(bForceTextShaders)
217 {
218 MetalCodeNSString = FMetalShaderDebugCache::Get().GetShaderCode(SourceLen, SourceCRC);
219 if(MetalCodeNSString)
220 {
221 MetalCodeNSString->retain();
222 }
223 else
224 {
225 UE_LOGF(LogMetal, Warning, "-metalshaderdebug was used but shader source are unavailable, please use r.Shaders.ExtraData=1 or enable AllowsRuntimeShaderCompiling for the platform and add CFLAG_Debug to this shader before cooking.");
226 }
227 }
228#endif
229 }
230 else if (bOfflineCompile && bHasShaderSource)
231 {
232 MetalCodeNSString = NS::String::string(ShaderSource, NS::UTF8StringEncoding);
233 check(MetalCodeNSString);
234 MetalCodeNSString->retain();
235 }
236
237 Library = InLibrary;
238
239 LibrarySeed = reinterpret_cast<uint64>(CreateShaderDesc.Library);
240
241 bool bNeedsCompiling = false;
242
243 FMetalCompiledShaderKey Key(Header.SourceLen, Header.SourceCRC, bIsExtendedBytecode, LibrarySeed);
244
245 FCachedCompiledShader CachedShader = GetMetalCompiledShaderCache().Find(Key);
246 Function = CachedShader.Function;
247 if (!Library && Function)
248 {
249 Library = CachedShader.Library;
250 }
251 else
252 {
253 bNeedsCompiling = true;
254 }
255
256 Bindings = Header.Bindings;
257 if (bNeedsCompiling || !Library)
258 {
259 if (bOfflineCompile METAL_DEBUG_OPTION(&& !(bHasShaderSource && bForceTextShaders)))
260 {
261 if (InLibrary)
262 {
263 Library = InLibrary;
264 }
265 else
266 {
267 // Archived shaders should never get in here.
268 check(!(Header.CompileFlags & (1 << CFLAG_Archive)) || BufferSize > 0);
269 dispatch_data_t GCDBuffer;
270 if (CreateShaderDesc.CodeOwner.IsValid())
271 {
272 // Hold a reference until the MTLLibrary is done using it
273 __block auto OwnerCopy = CreateShaderDesc.CodeOwner;
274 GCDBuffer = dispatch_data_create(SourceCode, BufferSize, nil, ^{
275 OwnerCopy.Reset();
276 });
277
278 // Append seed data after the metallib content to prevent Metal's
279 // newLibrary cache from returning a cached library. Metal's cache key appears
280 // to be based on the full dispatch_data bytes, so appending unique data
281 // ensures each library is treated as distinct.
282 if (LibrarySeed != 0)
283 {
284 uint64* SeedData = new uint64;
285 *SeedData = LibrarySeed;
286 dispatch_data_t SeedBuffer = dispatch_data_create(SeedData, sizeof(uint64), nil, ^{ delete SeedData; });
287 dispatch_data_t CombinedBuffer = dispatch_data_create_concat(GCDBuffer, SeedBuffer);
288 dispatch_release(GCDBuffer);
289 dispatch_release(SeedBuffer);
290 GCDBuffer = CombinedBuffer;
291 }
292 }
293 else
294 {
295 // Create a copy of the memory
296 GCDBuffer = dispatch_data_create(SourceCode, BufferSize, nil, DISPATCH_DATA_DESTRUCTOR_DEFAULT);
297 }
298
299 // load up the already compiled shader
300 NS::Error* AError;
301 Library = NS::TransferPtr(Device.GetDevice()->newLibrary(GCDBuffer, &AError));
302 dispatch_release(GCDBuffer);
303
304 if (!Library)
305 {
306 UE_LOGF(LogMetal, Display, "Failed to create library: %ls", *NSStringToFString(AError->description()));
307 }
308 }
309 }
310 else
311 {
312 MTL_SCOPED_AUTORELEASE_POOL;
313
314 NS::String* ShaderString = ((OfflineCompiledFlag == 0) ? NS::String::string(SourceCode, NS::UTF8StringEncoding) : MetalCodeNSString);
315
316 FString FinalShaderString;
317 const FString ShaderName = ShaderCode.FindOptionalData(FShaderCodeName::Key);
318 if(ShaderName.Len())
319 {
320 FinalShaderString = FString::Printf(TEXT("// %s\n"), *ShaderName);
321 }
322
323 FinalShaderString += NSStringToFString(ShaderString);
324 FinalShaderString.Replace(TEXT("#pragma once"), TEXT(""));
325 NS::String* NewShaderString = FStringToNSString(FinalShaderString);
326
327 MTL::CompileOptions* CompileOptions = MTL::CompileOptions::alloc()->init();
328 check(CompileOptions);
329
330#if DEBUG_METAL_SHADERS
331 static bool bForceFastMath = FParse::Param(FCommandLine::Get(), TEXT("metalfastmath"));
332 static bool bForceNoFastMath = FParse::Param(FCommandLine::Get(), TEXT("metalnofastmath"));
333 if (bForceNoFastMath)
334 {
335 CompileOptions->setFastMathEnabled(NO);
336 }
337 else if (bForceFastMath)
338 {
339 CompileOptions->setFastMathEnabled(YES);
340 }
341 else
342#endif
343 {
344 CompileOptions->setFastMathEnabled((BOOL)(!(Header.CompileFlags & (1 << CFLAG_NoFastMath))));
345 }
346
347#if !PLATFORM_MAC || DEBUG_METAL_SHADERS
348 NS::Dictionary* PreprocessorMacros = nullptr;;
349#if !PLATFORM_MAC // Pretty sure that as_type-casts work on macOS, but they don't for half2<->uint on older versions of the iOS runtime compiler.
350 PreprocessorMacros = NS::Dictionary::dictionary(NS::String::string("1", NS::UTF8StringEncoding),
351 NS::String::string("METAL_RUNTIME_COMPILER", NS::UTF8StringEncoding));
352#endif
353#if DEBUG_METAL_SHADERS
354 PreprocessorMacros = NS::Dictionary::dictionary(NS::String::string("1", NS::UTF8StringEncoding),
355 NS::String::string("MTLSL_ENABLE_DEBUG_INFO", NS::UTF8StringEncoding));
356#endif
357 if(PreprocessorMacros)
358 {
359 CompileOptions->setPreprocessorMacros(PreprocessorMacros);
360 PreprocessorMacros->release();
361 }
362#endif
363
364 MTL::LanguageVersion MetalVersion;
365 switch (Header.Version)
366 {
367 case 8:
368 MetalVersion = MTL::LanguageVersion3_0;
369 break;
370 case 7:
371 MetalVersion = MTL::LanguageVersion2_4;
372 break;
373#if PLATFORM_MAC
374 case 6:
375 MetalVersion = MTL::LanguageVersion2_3;
376 break;
377 case 5:
378 // Fall through
379 case 0:
380 // Fall through
381 default:
382 MetalVersion = MTL::LanguageVersion2_2;
383 break;
384#else
385 case 0:
386 MetalVersion = MTL::LanguageVersion2_4;
387 break;
388 case 9:
389 MetalVersion = MTL::LanguageVersion3_1;
390 break;
391 default:
392 UE_LOGF(LogRHI, Fatal, "Failed to create shader with unknown version %d: %ls", Header.Version, *NSStringToFString(NewShaderString));
393 MetalVersion = MTL::LanguageVersion2_4;
394 break;
395#endif
396 }
397 CompileOptions->setLanguageVersion(MetalVersion);
398
399 if(ShaderType == SF_Vertex && MetalVersion > MTL::LanguageVersion2_2)
400 {
401 CompileOptions->setPreserveInvariance(YES);
402 }
403
404 NS::Error* Error = nullptr;
405 Library = NS::TransferPtr(Device.GetDevice()->newLibrary(NewShaderString, CompileOptions, &Error));
406 if (Library.get() == nullptr)
407 {
408 UE_LOGF(LogRHI, Error, "*********** Error\n%ls", *NSStringToFString(NewShaderString));
409 UE_LOGF(LogRHI, Fatal, "Failed to create shader: %ls", *NSStringToFString(Error->description()));
410 }
411 else if (Error != nullptr)
412 {
413 // Warning...
414 UE_LOGF(LogRHI, Warning, "*********** Warning\n%ls", *NSStringToFString(NewShaderString));
415 UE_LOGF(LogRHI, Warning, "Created shader with warnings: %ls", *NSStringToFString(Error->description()));
416 }
417
418 MetalCodeNSString = NewShaderString;
419 MetalCodeNSString->retain();
420 }
421
422 GetCompiledFunction(true, -1, bBinaryCompiled);
423 }
424 SideTableBinding = Header.SideTable;
425
426 UE::RHICore::InitStaticUniformBufferSlots(this);
427
428#if RHI_INCLUDE_SHADER_DEBUG_DATA
429 this->Debug.ShaderName = GetFunctionName();
430#endif
431}

457template<typename BaseResourceType, int32 ShaderType>
458MTLFunctionPtr TMetalBaseShader<BaseResourceType, ShaderType>::GetCompiledFunction(bool const bAsync, const int32 FunctionIndex, bool bCompileToBinary)
459{
460 MTL_SCOPED_AUTORELEASE_POOL;
461
462 MTLFunctionPtr Func = Function;
463
464 bool bNeedToRecreateFunction = (LibraryFunctionIndex != FunctionIndex);
465 if (!Func || bNeedToRecreateFunction)
466 {
467 // Find the existing compiled shader in the cache.
468 FMetalCompiledShaderKey Key(SourceLen, SourceCRC, bIsExtendedBytecode, LibrarySeed);
469 Func = Function = GetMetalCompiledShaderCache().Find(Key).Function;
470
471 if (bNeedToRecreateFunction)
472 {
473 Function = MTLFunctionPtr();
474 Func = MTLFunctionPtr();
475 LibraryFunctionIndex = FunctionIndex;
476 }
477
478 if (!Func)
479 {
480 NS::String* Name = (LibraryFunctionIndex != -1) ? (NS::String*)Library->functionNames()->object(LibraryFunctionIndex) : FStringToNSString(GetFunctionName());
481
482 MTL::FunctionDescriptor* Descriptor = MTL::FunctionDescriptor::alloc()->init();
483 Descriptor->setName(Name);
484 if(bCompileToBinary)
485 {
486 Descriptor->setOptions(MTLFunctionOptionCompileToBinary);
487 }
488
489 if (!bAsync)
490 {
491 NS::Error* Error;
492 Function = NS::TransferPtr(Library->newFunction(Descriptor, &Error));
493 UE_CLOGF(Function.get() == nullptr, LogMetal, Error, "Failed to create function: %ls", *NSStringToFString(Error->description()));
494 UE_CLOGF(Function.get() == nullptr, LogMetal, Fatal, "*********** Error\n%ls", *NSStringToFString(GetSourceCode()));
495
496 check(Function);
497 GetMetalCompiledShaderCache().Add(Key, Library, Function);
498
499 Func = Function;
500
501 Descriptor->release();
502 }
503 else
504 {
505 const std::function<void(MTL::Function* pFunction, NS::Error* pError)> Handler = [Key, this](MTL::Function* NewFunction, NS::Error* Error)
506 {
507 UE_CLOGF(NewFunction == nullptr, LogMetal, Error, "Failed to create function: %ls", *NSStringToFString(Error->description()));
508 UE_CLOGF(NewFunction == nullptr, LogMetal, Fatal, "*********** Error\n%ls", *NSStringToFString(GetSourceCode()));
509
510 GetMetalCompiledShaderCache().Add(Key, Library, NS::RetainPtr(NewFunction));
511 };
512
513 NS::Array* FunctionNames = Library->functionNames();
514 check(FunctionNames->count());
515
516 Library->newFunction(Descriptor, Handler);
517
518 Descriptor->release();
519
520 return MTLFunctionPtr();
521 }
522 }
523 }
524
525 check(Func);
526 return Func;
527}
