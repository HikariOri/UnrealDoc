142 template<typename TToolType, typename = TEnableIf<TIsDerivedFrom<TToolType, IToolableTimelineToolFactory>::IsDerived>::Type>
143 bool AddToolFactory()
144 {
145 if (!HasToolFactory<TToolType>())
146 {
147 const TSharedRef<TToolType> NewFactory = MakeShared<TToolType>();
148 const int32 NumAdded = ToolFactories.Add(NewFactory);
149
150 ToolFactories.Sort([](const TSharedRef<IToolableTimelineToolFactory>& InA
151 , const TSharedRef<IToolableTimelineToolFactory>& InB)
152 {
153 return InA->GetPriority() < InB->GetPriority();
154 });
155
156 return NumAdded > 0;
157 }
158 return false;
159 }
