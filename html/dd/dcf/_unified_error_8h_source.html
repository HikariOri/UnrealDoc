215 enum class EDetailFilter : uint8
216 {
217 IncludeInSerialize = 1 << 0, // included when we serialize the details objects, almost every details should have this flag
218 IncludeInAnalytics = 1 << 1, // what details objects to include when we are creating events for analytics
219 IncludeInContextLogMessage = 1 << 2, // what details objects are included when the log message includes context objects on the string
220 IncludeInLogMessage = 1 << 3, // which details objects are included when we log a message without context objects included
221 // append here
222
223
224 // standard values
225 Default = IncludeInSerialize | IncludeInContextLogMessage,
226 None = 0x00,
227 All = 0xff
228 };

481 template<CErrorContext T>
482 const T* FError::GetErrorContext() const
483 {
484 TRefCountPtr<const IErrorDetails> Details = GetErrorDetails(T::DeclarationHash);
485 if (Details)
486 {
487 FMemoryView Data = Details->GetErrorContext();
488 if (!Data.IsEmpty())
489 {
490 checkf(Data.GetSize() == sizeof(T), TEXT("Size mismatch fetching error payload"));
491 return reinterpret_cast<const T*>(Data.GetData());
492 }
493 }
494 return nullptr;
495 }
