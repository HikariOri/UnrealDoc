30template <typename T>
31struct TEPAEntry
32{
33 int32 IdxBuffer[3];

58 FORCEINLINE_DEBUGGABLE bool Initialize(const TVec3<T>* VerticesA, const TVec3<T>* VerticesB, int32 InIdx0, int32 InIdx1, int32 InIdx2, const TVector<int32,3>& InAdjFaces, const TVector<int32,3>& InAdjEdges)
59 {
60 const TVec3<T> V0 = MinkowskiVert(VerticesA, VerticesB, InIdx0);
61 const TVec3<T> V1 = MinkowskiVert(VerticesA, VerticesB, InIdx1);
62 const TVec3<T> V2 = MinkowskiVert(VerticesA, VerticesB, InIdx2);
63
64 const TVec3<T> V0V1 = V1 - V0;
65 const TVec3<T> V0V2 = V2 - V0;
66 const TVec3<T> Norm = TVec3<T>::CrossProduct(V0V1, V0V2);
67 const T NormLenSq = Norm.SizeSquared();
68 // We have the square of the size of a cross product, so we need the distance margin to be a power of 4
69 // Verbosity emphasizes that
70 const T Eps = TEPAEntry::SelectEpsilon(1.e-4f * 1.e-4f * 1.e-4f * 1.e-4f, 1.e-8 * 1.e-8 * 1.e-8 * 1.e-8);
71 if (NormLenSq < Eps)
72 {
73 return false;
74 }
75 PlaneNormal = Norm * FMath::InvSqrt(NormLenSq);
76
77 IdxBuffer[0] = InIdx0;
78 IdxBuffer[1] = InIdx1;
79 IdxBuffer[2] = InIdx2;
80
81 AdjFaces = InAdjFaces;
82 AdjEdges = InAdjEdges;
83
84 Distance = TVec3<T>::DotProduct(PlaneNormal, V0);
85 bObsolete = false;
86
87 return true;
88 }

90 void SwapWinding(TEPAEntry* Entries)
91 {
92 //change vertex order
93 std::swap(IdxBuffer[0], IdxBuffer[1]);
94
95 //edges went from 0,1,2 to 1,0,2
96 //0th edge/face is the same (0,1 becomes 1,0)
97 //1th edge/face is now (0,2 instead of 1,2)
98 //2nd edge/face is now (2,1 instead of 2,0)
99
100 //update the adjacent face's adjacent edge first
101 auto UpdateAdjEdge = [Entries, this](int32 Old, int32 New)
102 {
103 TEPAEntry& AdjFace = Entries[AdjFaces[Old]];
104 int32& StaleAdjIdx = AdjFace.AdjEdges[AdjEdges[Old]];
105 check(StaleAdjIdx == Old);
106 StaleAdjIdx = New;
107 };
108
109 UpdateAdjEdge(1, 2);
110 UpdateAdjEdge(2, 1);
111
112 //now swap the actual edges and faces
113 std::swap(AdjFaces[1], AdjFaces[2]);
114 std::swap(AdjEdges[1], AdjEdges[2]);
115
116 PlaneNormal = -PlaneNormal;
117 Distance = -Distance;
118 }

120 FORCEINLINE_DEBUGGABLE T DistanceToPlane(const TVec3<T>& X) const
121 {
122 return TVec3<T>::DotProduct(PlaneNormal, X) - Distance;
123 }

125 bool IsOriginProjectedInside(const TVec3<T>* VertsABuffer, const TVec3<T>* VertsBBuffer, const T Epsilon) const
126 {
127 //Compare the projected point (PlaneNormal) to the triangle in the plane
128 const TVec3<T> PN = Distance * PlaneNormal;
129 const TVec3<T> PA = MinkowskiVert(VertsABuffer, VertsBBuffer, IdxBuffer[0]) - PN;
130 const TVec3<T> PB = MinkowskiVert(VertsABuffer, VertsBBuffer, IdxBuffer[1]) - PN;
131 const TVec3<T> PC = MinkowskiVert(VertsABuffer, VertsBBuffer, IdxBuffer[2]) - PN;
132
133 const TVec3<T> PACNormal = TVec3<T>::CrossProduct(PA, PC);
134 const T PACSign = TVec3<T>::DotProduct(PACNormal, PlaneNormal);
135 const TVec3<T> PCBNormal = TVec3<T>::CrossProduct(PC, PB);
136 const T PCBSign = TVec3<T>::DotProduct(PCBNormal, PlaneNormal);
137
138 if((PACSign < -Epsilon && PCBSign > Epsilon) || (PACSign > Epsilon && PCBSign < -Epsilon))
139 {
140 return false;
141 }
142
143 const TVec3<T> PBANormal = TVec3<T>::CrossProduct(PB, PA);
144 const T PBASign = TVec3<T>::DotProduct(PBANormal, PlaneNormal);
145
146 if((PACSign < -Epsilon && PBASign > Epsilon) || (PACSign > Epsilon && PBASign < -Epsilon))
147 {
148 return false;
149 }
150
151 return true;
152 }
153};

155template <typename T>
156bool InitializeEPA(TArray<TVec3<T>>& VertsA, TArray<TVec3<T>>& VertsB, TFunctionRef<TVector<T, 3>(const TVec3<T>& V)> SupportA, TFunctionRef<TVector<T, 3>(const TVec3<T>& V)> SupportB, TEPAWorkingArray<TEPAEntry<T>>& OutEntries, TVec3<T>& OutTouchNormal)
157{
158 const int32 NumVerts = VertsA.Num();
159 check(VertsB.Num() == NumVerts);
160
161 auto AddFartherPoint = [&](const TVec3<T>& Dir)
162 {
163 const TVec3<T> NegDir = -Dir;
164 const TVec3<T> A0 = SupportA(Dir); //should we have a function that does both directions at once?
165 const TVec3<T> A1 = SupportA(NegDir);
166 const TVec3<T> B0 = SupportB(NegDir);
167 const TVec3<T> B1 = SupportB(Dir);
168
169 const TVec3<T> W0 = A0 - B0;
170 const TVec3<T> W1 = A1 - B1;
171
172 const T Dist0 = TVec3<T>::DotProduct(W0, Dir);
173 const T Dist1 = TVec3<T>::DotProduct(W1, NegDir);
174
175 if (Dist1 >= Dist0)
176 {
177 VertsA.Add(A1);
178 VertsB.Add(B1);
179 }
180 else
181 {
182 VertsA.Add(A0);
183 VertsB.Add(B0);
184 }
185 };
186
187 OutEntries.AddUninitialized(4);
188 OutTouchNormal = TVec3<T>(0,0,1);
189
190 bool bValid = false;
191
192 switch(NumVerts)
193 {
194 case 1:
195 {
196 //assuming it's a touching hit at origin, but we still need to calculate a separating normal
197 AddFartherPoint(OutTouchNormal); // Use an arbitrary direction
198
199 // Now we might have a line! So fall trough to the next case
200 }
201 case 2:
202 {
203 //line, add farthest point along most orthogonal axes
204 TVec3<T> Dir = MinkowskiVert(VertsA.GetData(), VertsB.GetData(), 1) - MinkowskiVert(VertsA.GetData(), VertsB.GetData(), 0);
205
206 bValid = Dir.SizeSquared() > 1e-4;
207 if (bValid) //two verts given are distinct
208 {
209 //find most opposing axis
210 int32 BestAxis = 0;
211 T MinVal = TNumericLimits<T>::Max();
212 for (int32 Axis = 0; Axis < 3; ++Axis)
213 {
214 const T AbsVal = FMath::Abs(Dir[Axis]);
215 if (MinVal > AbsVal)
216 {
217 BestAxis = Axis;
218 MinVal = AbsVal;
219 }
220 }
221 const TVec3<T> OtherAxis = TVec3<T>::AxisVector(BestAxis);
222 const TVec3<T> Orthog = TVec3<T>::CrossProduct(Dir, OtherAxis);
223 const TVec3<T> Orthog2 = TVec3<T>::CrossProduct(Orthog, Dir);
224
225 AddFartherPoint(Orthog);
226 AddFartherPoint(Orthog2);
227
228 bValid = OutEntries[0].Initialize(VertsA.GetData(), VertsB.GetData(), 1, 2, 3, { 3,1,2 }, { 1,1,1 });
229 bValid &= OutEntries[1].Initialize(VertsA.GetData(), VertsB.GetData(), 0, 3, 2, { 2,0,3 }, { 2,1,0 });
230 bValid &= OutEntries[2].Initialize(VertsA.GetData(), VertsB.GetData(), 0, 1, 3, { 3,0,1 }, { 2,2,0 });
231 bValid &= OutEntries[3].Initialize(VertsA.GetData(), VertsB.GetData(), 0, 2, 1, { 1,0,2 }, { 2,0,0 });
232
233 if(!bValid)
234 {
235 OutTouchNormal = Orthog.GetUnsafeNormal();
236 return false;
237 }
238 }
239 else
240 {
241 // The two vertices are not distinct that may happen when the single vertex case above was hit and our CSO is very thin in that direction
242 CHAOS_ENSURE(NumVerts == 1); // If this ensure fires we were given 2 vertices that are not distinct to start with
243 return false;
244 }
245 break;
246 }
247 case 3:
248 {
249 //triangle, add farthest point along normal
250 bValid = OutEntries[3].Initialize(VertsA.GetData(), VertsB.GetData(), 0, 2, 1, { 1,0,2 }, { 2,0,0 });
251 if (CHAOS_ENSURE(bValid)) //input verts must form a valid triangle
252 {
253 const TEPAEntry<T>& Base = OutEntries[3];
254
255 AddFartherPoint(Base.PlaneNormal);
256
257 bValid = OutEntries[0].Initialize(VertsA.GetData(), VertsB.GetData(), 1, 2, 3, { 3,1,2 }, { 1,1,1 });
258 bValid &= OutEntries[1].Initialize(VertsA.GetData(), VertsB.GetData(), 0, 3, 2, { 2,0,3 }, { 2,1,0 });
259 bValid &= OutEntries[2].Initialize(VertsA.GetData(), VertsB.GetData(), 0, 1, 3, { 3,0,1 }, { 2,2,0 });
260
261 if(!bValid)
262 {
263 OutTouchNormal = Base.PlaneNormal;
264 return false;
265 }
266 }
267 break;
268 }
269 case 4:
270 {
271 bValid = OutEntries[0].Initialize(VertsA.GetData(), VertsB.GetData(), 1, 2, 3, { 3,1,2 }, { 1,1,1 });
272 bValid &= OutEntries[1].Initialize(VertsA.GetData(), VertsB.GetData(), 0, 3, 2, { 2,0,3 }, { 2,1,0 });
273 bValid &= OutEntries[2].Initialize(VertsA.GetData(), VertsB.GetData(), 0, 1, 3, { 3,0,1 }, { 2,2,0 });
274 bValid &= OutEntries[3].Initialize(VertsA.GetData(), VertsB.GetData(), 0, 2, 1, { 1,0,2 }, { 2,0,0 });
275
276 if(!bValid)
277 {
278 CHAOS_ENSURE(false); //expect user to give us valid tetrahedron
279 UE_LOGF(LogChaos, Log, "Invalid tetrahedron encountered in InitializeEPA");
280 }
281
282 break;
283 }
284
285 default:
286 {
287 CHAOS_ENSURE(false);
288 break;
289 }
290 }
291
292 if (bValid)
293 {
294 //make sure normals are pointing out of tetrahedron
295 // In the usual case the distances will either all be positive or negative,
296 // but the tetrahedron can be very close to (or touching) the origin
297 // Look for farthest plane to decide
298 T MaxSignedDistance = 0;
299 for (TEPAEntry<T>& Entry : OutEntries)
300 {
301 if (FMath::Abs(Entry.Distance) > FMath::Abs(MaxSignedDistance))
302 {
303 MaxSignedDistance = Entry.Distance;
304 }
305 }
306
307 if (MaxSignedDistance < 0.0f)
308 {
309 for (TEPAEntry<T>& Entry : OutEntries)
310 {
311 Entry.SwapWinding(OutEntries.GetData());
312 }
313 }
314 }
315
316 return bValid;
317}

334template <typename T>
335void EPAComputeVisibilityBorder(TEPAWorkingArray<TEPAEntry<T>>& Entries, int32 EntryIdx, const TVec3<T>& W, TEPAWorkingArray<FEPAFloodEntry>& OutBorderEdges, TEPAWorkingArray<FEPAFloodEntry> &ToVisitStack)
336{
337 {
338 TEPAEntry<T>& Entry = Entries[EntryIdx];
339 for (int i = 0; i < 3; ++i)
340 {
341 ToVisitStack.Add({ Entry.AdjFaces[i], Entry.AdjEdges[i] });
342 }
343 }
344
345 int32 Iteration = 0;
346 const int32 MaxIteration = 10000;
347
348 while (ToVisitStack.Num() && Iteration++ < MaxIteration)
349 {
350 const FEPAFloodEntry FloodEntry = ToVisitStack.Pop(EAllowShrinking::No);
351 TEPAEntry<T>& Entry = Entries[FloodEntry.EntryIdx];
352 if (!Entry.bObsolete)
353 {
354 if (Entry.DistanceToPlane(W) <= TEPAEntry<T>::SelectEpsilon(1.e-4f, 1.e-8))
355 {
356 //W can't see this triangle so mark the edge as a border
357 OutBorderEdges.Add(FloodEntry);
358 }
359 else
360 {
361 //W can see this triangle so continue flood fill
362 Entry.bObsolete = true; //no longer needed
363 const int32 Idx0 = FloodEntry.EdgeIdx;
364 const int32 Idx1 = (Idx0 + 1) % 3;
365 const int32 Idx2 = (Idx0 + 2) % 3;
366 ToVisitStack.Add({ Entry.AdjFaces[Idx1], Entry.AdjEdges[Idx1] });
367 ToVisitStack.Add({ Entry.AdjFaces[Idx2], Entry.AdjEdges[Idx2] });
368 }
369 }
370 }
371
372 if(Iteration >= MaxIteration)
373 {
374 UE_LOGF(LogChaos,Warning,"EPAComputeVisibilityBorder reached max iteration - something is wrong");
375 }
376}

378template <typename T>
379void ComputeEPAResults(const TVec3<T>* VertsA, const TVec3<T>* VertsB, const TEPAEntry<T>& Entry, T& OutPenetration, TVec3<T>& OutDir, TVec3<T>& OutA, TVec3<T>& OutB)
380{
381 //NOTE: We use this function as fallback when robustness breaks. So - do not assume adjacency is valid as these may be new uninitialized traingles that failed
382 FSimplex SimplexIDs({ 0,1,2 });
383 TVec3<T> As[4] = { VertsA[Entry.IdxBuffer[0]], VertsA[Entry.IdxBuffer[1]], VertsA[Entry.IdxBuffer[2]] };
384 TVec3<T> Bs[4] = { VertsB[Entry.IdxBuffer[0]], VertsB[Entry.IdxBuffer[1]], VertsB[Entry.IdxBuffer[2]] };
385 TVec3<T> Simplex[4] = { As[0] - Bs[0], As[1] - Bs[1], As[2] - Bs[2] };
386 T Barycentric[4];
387
388 OutDir = SimplexFindClosestToOrigin(Simplex, SimplexIDs, Barycentric, As, Bs);
389 OutPenetration = OutDir.Size();
390
391 // @todo(chaos): pass in epsilon? Does it need to match anything in GJK?
392 if (OutPenetration < 1e-4) //if closest point is on the origin (edge case when surface is right on the origin)
393 {
394 OutDir = Entry.PlaneNormal; //just fall back on plane normal
395 if (Entry.Distance < 0)
396 {
397 OutPenetration = -OutPenetration; // We are a bit outside of the shape so penetration is negative
398 }
399 }
400 else
401 {
402 OutDir /= OutPenetration;
403 if (Entry.Distance < 0)
404 {
405 //The origin is on the outside, so the direction is reversed
406 OutDir = -OutDir;
407 OutPenetration = -OutPenetration; // We are a bit outside of the shape so penetration is negative
408 }
409 }
410
411 OutA = TVec3<T>(0);
412 OutB = TVec3<T>(0);
413
414 for (int i = 0; i < SimplexIDs.NumVerts; ++i)
415 {
416 OutA += As[i] * Barycentric[i];
417 OutB += Bs[i] * Barycentric[i];
418 }
419}

421enum class EEPAResult
422{
423 Ok, // Successfully found the contact point to within the tolerance
424 MaxIterations, // We have a contact point, but did not reach the target tolerance before we hit the iteration limit - result accuracy is unknown
425 Degenerate, // We hit a degenerate condition in EPA which prevents a solution from being generated (result is invalid but objects may be penetrating)
426 BadInitialSimplex, // The initial setup did not provide a polytope containing the origin (objects are separated)
427 NoValidContact, // No valid contact have been found, no contacts will be returned
428};

440// Expanding Polytope Algorithm for finding the contact point for overlapping convex polyhedra.
441// See e.g., "Collision Detection in Interactive 3D Environments" (Gino van den Bergen, 2004)
442// or "Real-time Collision Detection with Implicit Objects" (Leif Olvang, 2010)
443template <typename T, typename TSupportA, typename TSupportB>
444EEPAResult EPA(TArray<TVec3<T>>& VertsABuffer, TArray<TVec3<T>>& VertsBBuffer, const TSupportA& SupportA, const TSupportB& SupportB, T& OutPenetration, TVec3<T>& OutDir, TVec3<T>& WitnessA, TVec3<T>& WitnessB, const FReal Eps = 1.e-2f)
445{
446 const TFunctionRef<TVector<T, 3>(const TVec3<T>& V)> SupportFuncA(SupportA);
447 const TFunctionRef<TVector<T, 3>(const TVec3<T>& V)> SupportFuncB(SupportB);
448 return EPA<T>(VertsABuffer, VertsBBuffer, SupportFuncA, SupportFuncB, OutPenetration, OutDir, WitnessA, WitnessB, Eps);
449}

451template <typename T>
452EEPAResult EPA(TArray<TVec3<T>>& VertsABuffer, TArray<TVec3<T>>& VertsBBuffer, const TFunctionRef<TVector<T, 3>(const TVec3<T>& V)>& SupportA,
453 const TFunctionRef<TVector<T, 3>(const TVec3<T>& V)>& SupportB, T& OutPenetration, TVec3<T>& OutDir, TVec3<T>& WitnessA, TVec3<T>& WitnessB, const FReal EpsRel = 1.e-2f)
454{
455 struct FEPAEntryWrapper
456 {
457 const TArray<TEPAEntry<T>>* Entries;
458 int32 Idx;
459
460 bool operator>(const FEPAEntryWrapper& Other) const
461 {
462 return (*Entries)[Idx] > (*Entries)[Other.Idx];
463 }
464 };
465
466 constexpr T OriginInsideEps = 0.0f;
467
468 TEPAWorkingArray<TEPAEntry<T>> Entries;
469
470 if(!InitializeEPA(VertsABuffer,VertsBBuffer,SupportA,SupportB, Entries, OutDir))
471 {
472 //either degenerate or a touching hit. Either way return penetration 0
473 OutPenetration = 0;
474 WitnessA = TVec3<T>(0);
475 WitnessB = TVec3<T>(0);
476 return EEPAResult::BadInitialSimplex;
477 }
478
479#if DEBUG_EPA
480 TArray<TVec3<T>> VertsWBuffer;
481 for (int32 Idx = 0; Idx < 4; ++Idx)
482 {
483 VertsWBuffer.Add(MinkowskiVert(VertsABuffer.GetData(), VertsBBuffer.GetData(), Idx));
484 }
485#endif
486
487 TEPAWorkingArray<int32> Queue;
488 for(int32 Idx = 0; Idx < Entries.Num(); ++Idx)
489 {
490 //ensure(Entries[Idx].Distance > -Eps);
491 // Entries[Idx].Distance <= 0.0f is true if the origin is a bit out of the polytope (we need to support this case for robustness)
492 if(Entries[Idx].Distance <= 0.0f || Entries[Idx].IsOriginProjectedInside(VertsABuffer.GetData(), VertsBBuffer.GetData(), OriginInsideEps))
493 {
494 Queue.Add(Idx);
495 }
496 }
497
498 TEPAWorkingArray<FEPAFloodEntry> VisibilityBorder;
499 TEPAWorkingArray<FEPAFloodEntry> VisibilityBorderToVisitStack;
500
501 //TEPAEntry<T> BestEntry;
502 //BestEntry.Distance = 0;
503 TEPAEntry<T> LastEntry = Queue.Num() > 0 ? Entries[Queue.Last()] : Entries[0];
504 T UpperBound = TNumericLimits<T>::Max();
505 T LowerBound = TNumericLimits<T>::Lowest();
506 bool bQueueDirty = true;
507 int32 Iteration = 0;
508 int32 constexpr MaxIterations = 128;
509 EEPAResult ResultStatus = EEPAResult::MaxIterations;
510 while (Queue.Num() && (Iteration++ < MaxIterations))
511 {
512 if (bQueueDirty)
513 {
514 // Avoiding UE's Sort here because it has a call to FMath::Loge. The std version calls insertion sort when possible
515 std::sort(Queue.GetData(), Queue.GetData() + Queue.Num(), [&Entries](const int32 L, const int32 R) { return Entries[L] > Entries[R]; });
516 bQueueDirty = false;
517 }
518
519 int32 EntryIdx = Queue.Pop(EAllowShrinking::No);
520 TEPAEntry<T>& Entry = Entries[EntryIdx];
521 //bool bBadFace = Entry.IsOriginProjectedInside(VertsABuffer.GetData(), VertsBBuffer.GetData());
522 {
523 //UE_LOGF(LogChaos, Warning, "%d BestW:%f, Distance:%f, bObsolete:%d, InTriangle:%d",
524 // Iteration, BestW.Size(), Entry.Distance, Entry.bObsolete, bBadFace);
525 }
526 if (Entry.bObsolete)
527 {
528 // @todo(chaos): should this count as an iteration? Currently it does...
529 continue;
530 }
531
532 if (Entry.Distance > UpperBound)
533 {
534 ResultStatus = EEPAResult::Ok;
535 break;
536 }
537
538 const TVec3<T> ASupport = SupportA(Entry.PlaneNormal);
539 const TVec3<T> BSupport = SupportB(-Entry.PlaneNormal);
540 const TVec3<T> W = ASupport - BSupport;
541 const T DistanceToSupportPlane = TVec3<T>::DotProduct(Entry.PlaneNormal, W);
542 if(DistanceToSupportPlane < UpperBound)
543 {
544 UpperBound = DistanceToSupportPlane;
545 //Remember the entry that gave us the lowest upper bound and use it in case we have to terminate early
546 //This can result in very deep planes. Ideally we'd just use the plane formed at W, but it's not clear how you get back points in A, B for that
547 //BestEntry = Entry;
548 }
549
550 LowerBound = Entry.Distance;
551
552 // It's possible the origin is not contained by the CSO, probably because of numerical error.
553 // In this case the upper bound will be negative, at which point we should just exit.
554 const T UpperBoundTolerance = (T(1) + EpsRel) * FMath::Abs(LowerBound);
555 if (UpperBound <= UpperBoundTolerance)
556 {
557 ResultStatus = EEPAResult::Ok;
558 LastEntry = Entry;
559 break;
560 }
561
562 if (UpperBound < LowerBound)
563 {
564 //we cannot get any better than what we saw, so just return previous face
565 ResultStatus = EEPAResult::Ok;
566 break;
567 }
568
569 LastEntry = Entry;
570
571
572 const int32 NewVertIdx = VertsABuffer.Add(ASupport);
573 VertsBBuffer.Add(BSupport);
574
575#if DEBUG_EPA
576 VertsWBuffer.Add(MinkowskiVert(VertsABuffer.GetData(), VertsBBuffer.GetData(), NewVertIdx));
577#endif
578
579 Entry.bObsolete = true;
580 VisibilityBorder.Reset();
581 VisibilityBorderToVisitStack.Reset();
582 EPAComputeVisibilityBorder(Entries, EntryIdx, W, VisibilityBorder, VisibilityBorderToVisitStack);
583 const int32 NumBorderEdges = VisibilityBorder.Num();
584 const int32 FirstIdxInBatch = Entries.Num();
585 int32 NewIdx = FirstIdxInBatch;
586 Entries.AddUninitialized(NumBorderEdges);
587 if (NumBorderEdges >= 3)
588 {
589 bool bTerminate = false;
590 for (int32 VisibilityIdx = 0; VisibilityIdx < NumBorderEdges; ++VisibilityIdx)
591 {
592 //create new entries and update adjacencies
593 const FEPAFloodEntry& BorderInfo = VisibilityBorder[VisibilityIdx];
594 TEPAEntry<T>& NewEntry = Entries[NewIdx];
595 const int32 BorderEntryIdx = BorderInfo.EntryIdx;
596 TEPAEntry<T>& BorderEntry = Entries[BorderEntryIdx];
597 const int32 BorderEdgeIdx0 = BorderInfo.EdgeIdx;
598 const int32 BorderEdgeIdx1 = (BorderEdgeIdx0 + 1) % 3;
599 const int32 NextEntryIdx = (VisibilityIdx + 1) < VisibilityBorder.Num() ? NewIdx + 1 : FirstIdxInBatch;
600 const int32 PrevEntryIdx = NewIdx > FirstIdxInBatch ? NewIdx - 1 : FirstIdxInBatch + NumBorderEdges - 1;
601 const bool bValidTri = NewEntry.Initialize(VertsABuffer.GetData(), VertsBBuffer.GetData(), BorderEntry.IdxBuffer[BorderEdgeIdx1], BorderEntry.IdxBuffer[BorderEdgeIdx0], NewVertIdx,
602 { BorderEntryIdx, PrevEntryIdx, NextEntryIdx },
603 { BorderEdgeIdx0, 2, 1 });
604 BorderEntry.AdjFaces[BorderEdgeIdx0] = NewIdx;
605 BorderEntry.AdjEdges[BorderEdgeIdx0] = 0;
606
607 if (!bValidTri)
608 {
609 //couldn't properly expand polytope, so just stop
610 ResultStatus = EEPAResult::Degenerate;
611 bTerminate = true;
612 break;
613 }
614
615 // Due to numerical inaccuracies NewEntry.Distance >= LowerBound may be false!
616 // However these Entries still have good normals, and needs to be included to prevent
617 // this exiting with very deep penetration results
618
619 if (bValidTri && NewEntry.Distance <= UpperBound)
620 {
621 // NewEntry.Distance <= 0.0f is if the origin is a bit out of the polytope
622 if (NewEntry.Distance <= 0.0f || NewEntry.IsOriginProjectedInside(VertsABuffer.GetData(), VertsBBuffer.GetData(), OriginInsideEps))
623 {
624 Queue.Add(NewIdx);
625 bQueueDirty = true;
626 }
627 }
628
629 ++NewIdx;
630 }
631
632 if (bTerminate)
633 {
634 break;
635 }
636 }
637 else
638 {
639 //couldn't properly expand polytope, just stop now
640 ResultStatus = EEPAResult::Degenerate;
641 break;
642 }
643 }
644
645 ComputeEPAResults(VertsABuffer.GetData(), VertsBBuffer.GetData(), LastEntry, OutPenetration, OutDir, WitnessA, WitnessB);
646
647 return ResultStatus;
648}
