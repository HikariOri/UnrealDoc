90template<typename TPlatformShaderSymbolData>
91inline void FShaderSymbolExport::NotifyShaderCompiled(const FCompressedBuffer& PlatformSymbolDataCompressed, const FString& DebugInfo)
92{
93 // Double-checked locking: atomic acquire load on the fast path, InitCs taken only on first call
94 if (!bInitialized.load(std::memory_order_acquire))
95 {
96 FScopeLock Lock(&InitCs);
97 if (!bInitialized.load(std::memory_order_relaxed))
98 {
99 // If we get called, we know we're compiling. Do one time initialization
100 // which will create the output directory / open the output file stream.
101 Initialize();
102 bInitialized.store(true, std::memory_order_release);
103 }
104 }
105
106 if (!bExportShaderSymbols) // read-only after init; safe via acquire above
107 {
108 return;
109 }
110
111 // Dispatch a task to decompress, deserialize, and write. All heavy work off the calling thread.
112 // The counter is incremented before launch; the task decrements it on completion and triggers
113 // AllTasksDoneEvent when it reaches zero so shutdown can drain without holding task references.
114 PendingTaskCount.fetch_add(1, std::memory_order_relaxed);
115
116 UE::Tasks::Launch(TEXT("ShaderSymbolExport::WriteSymbolData"),
117 [this, PlatformSymbolDataCompressed, DebugInfo]()
118 {
119 TPlatformShaderSymbolData FullSymbolData;
120 FSharedBuffer PlatformSymbolData = PlatformSymbolDataCompressed.Decompress();
121 FMemoryReaderView Ar(PlatformSymbolData.GetView());
122 Ar << FullSymbolData;
123
124 for (const auto& SymbolData : FullSymbolData.GetAllSymbolData())
125 {
126 WriteSymbolData(SymbolData.GetFilename(), DebugInfo, SymbolData.GetContents());
127 }
128
129 if (PendingTaskCount.fetch_sub(1, std::memory_order_acq_rel) == 1)
130 {
131 AllTasksDoneEvent->Trigger();
132 }
133 },
134 UE::Tasks::ETaskPriority::BackgroundLow);
135}
