136{

140 SerialAny_Case(ESerialMethod::Bool_Default , bool );
141 SerialAny_Case(ESerialMethod::Uint8_Default , uint8 );
142 SerialAny_Case(ESerialMethod::Int32_Default , int32 );
143 SerialAny_Case(ESerialMethod::Uint32_Default , uint32 );
144 SerialAny_Case(ESerialMethod::Uint64_Default , uint64 );
145 SerialAny_Case(ESerialMethod::Uint32_Packed , uint32 );
146 SerialAny_Case(ESerialMethod::String_Default , FString );
147 SerialAny_Case(ESerialMethod::Double_Default , double );
148 SerialAny_Case(ESerialMethod::Float_Default , float );
149 SerialAny_Case(ESerialMethod::Int32_Array , TArray<int32> );
150 SerialAny_Case(ESerialMethod::String_Array , TArray<FString> );
151 SerialAny_Case(ESerialMethod::Float_Array , TArray<float> );
152 SerialAny_Case(ESerialMethod::Vector_f32 , FVector );
153 SerialAny_Case(ESerialMethod::Vector_f64 , FVector );
154 SerialAny_Case(ESerialMethod::Quat_f32 , FQuat );
155 SerialAny_Case(ESerialMethod::Quat_f64 , FQuat );
156 SerialAny_Case(ESerialMethod::LinearColor_Default, FLinearColor );
157 SerialAny_Case(ESerialMethod::MD5Hash_Default , FMD5Hash );
158 SerialAny_Case(ESerialMethod::Transform_Default , FTransform );
159 SerialAny_Case(ESerialMethod::Transform_Array , TArray<FTransform>);
160
161 case ESerialMethod::None: return true;
162 case ESerialMethod::_NotImplementedYet:
163 default: ensure(false);
164 }
165 return false;
166#undef SerialAny_Case
