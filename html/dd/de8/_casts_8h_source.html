87// Dynamically cast an object type-safely.
88template <typename To, typename From>
89inline TCopyQualifiersFromTo_T<From, To>* Cast(From* Src)
90{
91 static_assert(sizeof(From) > 0 && sizeof(To) > 0, "Attempting to cast between incomplete types");
92
93 if (Src)
94 {
95 if constexpr (TIsIInterface<From>::Value)
96 {
97PRAGMA_DISABLE_DEPRECATION_WARNINGS
98 if (UObject* Obj = Src->_getUObject())
99PRAGMA_ENABLE_DEPRECATION_WARNINGS
100 {
101 if constexpr (TIsIInterface<To>::Value)
102 {
103 return (To*)Obj->GetInterfaceAddress(To::UClassType::StaticClass());
104 }
105 else
106 {
107 if constexpr (std::is_same_v<To, UObject>)
108 {
109 return Obj;
110 }
111 else
112 {
113 if (Obj->IsA<To>())
114 {
115 return (To*)Obj;
116 }
117 }
118 }
119 }
120 }
121 else
122 {
123 static_assert(std::is_base_of_v<UObjectBase, From>, "Attempting to use Cast<> on a type that is not a UObject or an Interface");
124
125 if constexpr (UE_USE_CAST_FLAGS && UE::CoreUObject::Private::TCastFlags_V<To> != CASTCLASS_None)
126 {
127 if constexpr (std::is_base_of_v<To, From>)
128 {
129 return (To*)Src;
130 }
131 else
132 {
133#if UE_ENABLE_UNRELATED_CAST_WARNINGS
134 UE_STATIC_ASSERT_WARN((std::is_base_of_v<From, To>), "Attempting to use Cast<> on types that are not related");
135#endif
136 if (((const UObject*)Src)->GetClass()->HasAnyCastFlag(UE::CoreUObject::Private::TCastFlags_V<To>))
137 {
138 return (To*)Src;
139 }
140 }
141 }
142 else
143 {
144 if constexpr (TIsIInterface<To>::Value)
145 {
146 return (To*)((UObject*)Src)->GetInterfaceAddress(To::UClassType::StaticClass());
147 }
148 else if constexpr (std::is_base_of_v<To, From>)
149 {
150 return Src;
151 }
152 else
153 {
154#if UE_ENABLE_UNRELATED_CAST_WARNINGS
155 UE_STATIC_ASSERT_WARN((std::is_base_of_v<From, To>), "Attempting to use Cast<> on types that are not related");
156#endif
157 if (((const UObject*)Src)->IsA<To>())
158 {
159 return (To*)Src;
160 }
161 }
162 }
163 }
164 }
165
166 return nullptr;
167}

315template <typename To, typename From>
316inline TCopyQualifiersFromTo_T<From, To>* Cast(const TObjectPtr<From>& InSrc)
317{
318 static_assert(sizeof(To) > 0 && sizeof(From) > 0, "Attempting to cast between incomplete types");
319
320 const FObjectPtr& Src = (const FObjectPtr&)InSrc;
321
322 if constexpr (UE_USE_CAST_FLAGS && UE::CoreUObject::Private::TCastFlags_V<To> != CASTCLASS_None)
323 {
324 if (Src)
325 {
326 if constexpr (std::is_base_of_v<To, From>)
327 {
328 return (To*)Src.Get();
329 }
330 else
331 {
332 #if UE_ENABLE_UNRELATED_CAST_WARNINGS
333 UE_STATIC_ASSERT_WARN((std::is_base_of_v<From, To>), "Attempting to use Cast<> on types that are not related");
334 #endif
335 if (Src.GetClass()->HasAnyCastFlag(UE::CoreUObject::Private::TCastFlags_V<To>))
336 {
337 return (To*)Src.Get();
338 }
339 }
340 }
341 }
342 else if constexpr (TIsIInterface<To>::Value)
343 {
344 const UObject* SrcObj = UE::CoreUObject::Private::ResolveObjectHandleNoRead(Src.GetHandleRef());
345 if (SrcObj)
346 {
347 UE::CoreUObject::Private::OnHandleRead(SrcObj);
348 return (To*)Src.Get()->GetInterfaceAddress(To::UClassType::StaticClass());
349 }
350 }
351 else if constexpr (std::is_base_of_v<To, From>)
352 {
353 if (Src)
354 {
355 return (To*)Src.Get();
356 }
357 }
358 else
359 {
360#if UE_ENABLE_UNRELATED_CAST_WARNINGS
361 UE_STATIC_ASSERT_WARN((std::is_base_of_v<From, To>), "Attempting to use Cast<> on types that are not related");
362#endif
363 if (Src && Src.IsA<To>())
364 {
365 return (To*)Src.Get();
366 }
367 }
368
369 return nullptr;
370}

543 template <typename To, typename From>
544 inline To DynamicCast(From* Arg)
545 {
546 using ToValueType = std::remove_pointer_t<To>;
547
548 if constexpr (!std::is_pointer_v<To> || !TIsCastable<From>::Value || !TIsCastable<ToValueType>::Value)
549 {
550 return dynamic_cast<To>(Arg);
551 }
552 else
553 {
554 // Casting away const/volatile
555 static_assert(!TLosesQualifiersFromTo_V<From, ToValueType>, "Conversion loses qualifiers");
556
557 if constexpr (std::is_void_v<ToValueType>)
558 {
559 // When casting to void, cast to UObject instead and let it implicitly cast to void
560 return Cast<UObject>(Arg);
561 }
562 else
563 {
564 return Cast<ToValueType>(Arg);
565 }
566 }
567 }
568
569 template <typename To, typename From>
570 inline To DynamicCast(From&& Arg)
571 {
572 using FromValueType = std::remove_reference_t<From>;
573 using ToValueType = std::remove_reference_t<To>;
574
575 if constexpr (!TIsCastable<FromValueType>::Value || !TIsCastable<ToValueType>::Value)
576 {
577 // This may fail when dynamic_casting rvalue references due to patchy compiler support
578 return dynamic_cast<To>(Arg);
579 }
580 else
581 {
582 // Casting away const/volatile
583 static_assert(!TLosesQualifiersFromTo_V<FromValueType, ToValueType>, "Conversion loses qualifiers");
584
585 // T&& can only be cast to U&&
586 // http://en.cppreference.com/w/cpp/language/dynamic_cast
587 static_assert(std::is_lvalue_reference_v<From> || std::is_rvalue_reference_v<To>, "Cannot dynamic_cast from an rvalue to a non-rvalue reference");
588
589 return Forward<To>(*CastChecked<ToValueType>(&Arg));
590 }
591 }
