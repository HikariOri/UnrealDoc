2073inline FArrangedWidget FGeometry::MakeChild(const TSharedRef<SWidget>& ChildWidget, const UE::Slate::FDeprecateVector2DParameter& InLocalSize, const FSlateLayoutTransform& LayoutTransform) const
2074{
2075 // If there is no render transform set, use the simpler MakeChild call that doesn't bother concatenating the render transforms.
2076 // This saves a significant amount of overhead since every widget does this, and most children don't have a render transform.
2077 const TOptional<FSlateRenderTransform> RenderTransform = ChildWidget->GetRenderTransformWithRespectToFlowDirection();
2078 if (RenderTransform.IsSet() )
2079 {
2080 const FVector2f RenderTransformPivot = UE::Slate::CastToVector2f(ChildWidget->GetRenderTransformPivotWithRespectToFlowDirection());
2081 return FArrangedWidget(ChildWidget, MakeChild(InLocalSize, LayoutTransform, RenderTransform.GetValue(), RenderTransformPivot));
2082 }
2083 else
2084 {
2085 return FArrangedWidget(ChildWidget, MakeChild(InLocalSize, LayoutTransform));
2086 }
2087}

2094inline FArrangedWidget FGeometry::MakeChild(const TSharedRef<SWidget>& ChildWidget, const UE::Slate::FDeprecateVector2DParameter& ChildOffset, const UE::Slate::FDeprecateVector2DParameter& InLocalSize, float ChildScale) const
2095{
2096 // Since ChildOffset is given as a LocalSpaceOffset, we MUST convert this offset into the space of the parent to construct a valid layout transform.
2097 // The extra TransformPoint below does this by converting the local offset to an offset in parent space.
2098 return MakeChild(ChildWidget, UE::Slate::CastToVector2f(InLocalSize), FSlateLayoutTransform(ChildScale, TransformPoint(ChildScale, UE::Slate::CastToVector2f(ChildOffset))));
