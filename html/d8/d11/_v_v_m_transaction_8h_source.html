37 FWriteLog()

40 std::memset(Table, 0, InitialCapacity * sizeof(uint64));

52 {
53 for (uint32 I = 0, Last = Num; I != Last; ++I)

56 }

65 else

71protected:
72 void AddImpl(FAllocationContext Context, T Entry)
73 {
74 checkSlow(Entry.Key() != 0);
75 if (IsInline())
76 {
77 AddToInlineHashTable(Context, Entry);
78 }
79 else
80 {
81 AddToHashTable(Context, Entry);
82 }
83 }

85private:
86 V_FORCEINLINE bool IsInline() { return Table == InlineTable; }
87
88 V_FORCEINLINE bool ShouldGrowTable()
89 {
90 return 2 * Num > TableCapacity;
91 }
92
93 static uint64& FindBucket(uint64 Entry, uint64* Table, uint32 Capacity, bool& bIsNewEntry)
94 {
95 checkSlow(Capacity && (Capacity & (Capacity - 1)) == 0);
96 // We use a simple linear probing hash table.
97 uint32 Mask = Capacity - 1;
98 uint32 Index = ::GetTypeHash(Entry) & Mask;
99 for (;;)
100 {
101 if (!Table[Index] || Table[Index] == Entry)
102 {
103 bIsNewEntry = !Table[Index];
104 return Table[Index];
105 }
106 Index = (Index + 1) & Mask;
107 }
108 }
109
110 FORCENOINLINE void GrowTable(FAllocationContext Context)
111 {
112 uint32 NewCapacity = TableCapacity * 2;
113 if (TableCapacity == InitialCapacity)
114 {
115 NewCapacity *= 2;
116 }
117
118 std::size_t AllocationSize = sizeof(uint64) * NewCapacity;
119 uint64* NewTable = BitCast<uint64*>(Context.AllocateAuxCell(AllocationSize));
120 std::memset(NewTable, 0, AllocationSize);
121
122 for (uint32 I = 0, Last = Num; I != Last; ++I)
123 {
124 bool bIsNewEntry;
125 FindBucket(Log[I].Key(), NewTable, NewCapacity, bIsNewEntry) = Log[I].Key();
126 }
127
128 TableCapacity = NewCapacity;
129 Table = NewTable;
130 }
131
132 void AddToInlineHashTable(FAllocationContext Context, T Entry)
133 {
134 for (uint32 I = 0, Last = InitialCapacity; I != Last; ++I)
135 {
136 if (!Table[I])
137 {
138 Table[I] = Entry.Key();
139 AppendToLog(Context, Entry);
140 return;
141 }
142 if (Entry.Key() == Table[I])
143 {
144 return;
145 }
146 }
147
148 GrowTable(Context);
149 AddToHashTable(Context, Entry);
150 }
151
152 void AddToHashTable(FAllocationContext Context, T Entry)
153 {
154 bool bIsNewEntry;
155 uint64& Bucket = FindBucket(Entry.Key(), Table, TableCapacity, bIsNewEntry);
156 if (bIsNewEntry)
157 {
158 Bucket = Entry.Key();
159 AppendToLog(Context, Entry);
160 if (ShouldGrowTable())
161 {
162 GrowTable(Context);
163 }
164 }
165 }
166
167 void EmptyInlineHashTable()
168 {
169 std::memset(Table, 0, InitialCapacity * sizeof(uint64));
170 Num = 0;
171 TableCapacity = InitialCapacity;
172 LogCapacity = InitialCapacity;
173 }
174
175 void EmptyHashTable()
176 {
177 Table = InlineTable;
178 Log = BitCast<T*>(static_cast<char*>(InlineLog));
179 EmptyInlineHashTable();
180 }
181
182 void AppendToLog(FAllocationContext Context, T Entry)
183 {
184 if (Num == LogCapacity)
185 {
186 uint32 NewCapacity = LogCapacity * 2;
187 T* NewLog = BitCast<T*>(Context.AllocateAuxCell(NewCapacity * sizeof(T)));
188 std::memcpy(NewLog, Log, Num * sizeof(T));
189 LogCapacity = NewCapacity;
190 Log = NewLog;
191 }
192
193 new (Log + Num) T{::MoveTemp(Entry)};
194 ++Num;
195 }
196
197private:
198 static constexpr uint32 InitialCapacity = 4;
199
200 uint64* Table = InlineTable;
201 T* Log = BitCast<T*>(static_cast<char*>(InlineLog));
202
203 uint64 InlineTable[InitialCapacity];
204 alignas(alignof(T)) char InlineLog[InitialCapacity * sizeof(T)];
205
206 uint32 Num = 0;
207 uint32 TableCapacity = InitialCapacity;
208 // TODO: It's conceivable we could make LogCapacity a function of TableCapacity.
209 // But we're just doing the simple thing for now.
210 uint32 LogCapacity = InitialCapacity;
211};

262 }
263 else
264 {
265 V_DIE("Unknown FTrailLogEntry::EType encountered!");
266 }
267 }

272 void* Ptr; // VPlaceholder::Mode() is byte-aligned, leaving no bits for TPtrVariant.
273 uint64 Value;
274 EType Type;

386struct AUTORTFM_DISABLE FTransaction
387{
388 enum class EState
389 {
390 // Initial constructed state.
391 Constructed,
392 // The transaction has begun, and is on transaction stack.
393 // Active can only transition from the Constructed state, and can
394 // only transition to the Committed or Aborted state.
395 Active,
396 // The transaction has completed without failure.
397 // Committed can only transition from the Active state, and is a
398 // terminal state.
399 Committed,
400 // The transaction has completed with failure.
401 // Aborted can only transition from the Active state, and is a terminal
402 // state.
403 Aborted,
404 };
405
406 // Convenience function for returning true iff State is Active.
407 inline bool IsActive() const { return State == EState::Active; }
408
409 // The outer transaction. The outer VM transaction will have a null parent.
410 FTransaction* Parent{nullptr};
411
412 // The current VM transaction state.
413 EState State = EState::Constructed;
414
415#if ENABLE_VVM_TRANSACTION_ID_VALIDATION
416 // The identifier of the AutoRTFM transaction created with Start()
417 AutoRTFM::TransactionID AutoRTFMTransactionID{0};
418#endif
419
420 // Begins the transaction.
421 // Creates and pushes a new AutoRTFM transaction, and sets this FTransaction
422 // as the current transaction with Context.
423 // Must be in EState::Constructed to call.
424 void Start(FRunningContext Context)
425 {
426 V_DIE_UNLESS(State == EState::Constructed);
427 DieUnlessTransactionIDIsZero();
428 AutoRTFM::StartTransaction();
429 Parent = Context.CurrentTransaction();
430 Context.SetCurrentTransaction(this);
431 SnapshotTransactionID();
432 State = EState::Active;
433 }
434
435 // Commits the transaction.
436 // Sets the parent transaction (which may be null) as the current
437 // transaction with Context.
438 // Commits the AutoRTFM transaction which will call on-commit handlers, and
439 // pop the AutoRTFM transaction created with Start() from the AutoRTFM
440 // transaction stack.
441 // Must be in EState::Active to call.
442 void Commit(FRunningContext Context)
443 {
444 V_DIE_UNLESS(State == EState::Active);
445 DieUnlessTransactionIDsMatch();
446 AutoRTFM::CommitTransaction();
447 Context.SetCurrentTransaction(Parent);
448 State = EState::Committed;
449 }
450
451 // Controls the behaviour of FTransaction::Abort()
452 enum class EAbortMode
453 {
454 // Abort() will rollback and pop the current AutoRTFM transaction.
455 Default,
456 // Abort() will not rollback and pop the current AutoRTFM transaction,
457 // as the AutoRTFM transaction was already aborted before calling.
458 AutoRTFMTransactionAlreadyAborted,
459 };
460
461 // Aborts the transaction.
462 // Sets the parent transaction (which may be null) as the current
463 // transaction with Context.
464 // If AbortMode is Default, then aborts the AutoRTFM transaction created
465 // with Start(), which will revert any memory writes, call on-abort handlers
466 // and pop the AutoRTFM transaction from the AutoRTFM transaction stack.
467 // Must be in EState::Active to call.
468 template <EAbortMode AbortMode = EAbortMode::Default>
469 void Abort(FAccessContext Context)
470 {
471 V_DIE_UNLESS(State == EState::Active);
472 State = EState::Aborted;
473 Context.SetCurrentTransaction(Parent);
474 if constexpr (AbortMode == EAbortMode::Default)
475 {
476 DieUnlessTransactionIDsMatch();
477 AutoRTFM::AbortTransaction();
478 }
479 else
480 {
481 // Identifier should not match, as the transaction should be aborted.
482 DieIfTransactionIDsMatch();
483 }
484 ClearTransactionID();
485 }
486
487 // Stores the current AutoRTFM transaction ID to AutoRTFMTransactionID
488 // No-op if !ENABLE_VVM_TRANSACTION_ID_VALIDATION
489 inline void SnapshotTransactionID()
490 {
491#if ENABLE_VVM_TRANSACTION_ID_VALIDATION
492 AutoRTFMTransactionID = AutoRTFM::CurrentTransactionID();
493#endif
494 }
495
496 // Resets the current AutoRTFM transaction ID to 0
497 // No-op if !ENABLE_VVM_TRANSACTION_ID_VALIDATION
498 inline void ClearTransactionID()
499 {
500#if ENABLE_VVM_TRANSACTION_ID_VALIDATION
501 AutoRTFMTransactionID = 0;
502#endif
503 }
504
505 // Dies if AutoRTFMTransactionID is not 0
506 // No-op if !ENABLE_VVM_TRANSACTION_ID_VALIDATION
507 inline void DieUnlessTransactionIDIsZero() const
508 {
509#if ENABLE_VVM_TRANSACTION_ID_VALIDATION
510 V_DIE_UNLESS(AutoRTFMTransactionID == 0)
511#endif
512 }
513
514 // Dies if AutoRTFMTransactionID is not equal to the current AutoRTFM transaction ID
515 // No-op if !ENABLE_VVM_TRANSACTION_ID_VALIDATION
516 inline void DieUnlessTransactionIDsMatch() const
517 {
518#if ENABLE_VVM_TRANSACTION_ID_VALIDATION
519 V_DIE_UNLESS(AutoRTFMTransactionID == AutoRTFM::CurrentTransactionID())
520#endif
521 }
522
523 // Dies if AutoRTFMTransactionID is equal to the current AutoRTFM transaction ID
524 // No-op if !ENABLE_VVM_TRANSACTION_ID_VALIDATION
525 inline void DieIfTransactionIDsMatch() const
526 {
527#if ENABLE_VVM_TRANSACTION_ID_VALIDATION
528 V_DIE_IF(AutoRTFMTransactionID == AutoRTFM::CurrentTransactionID());
529#endif
530 }
