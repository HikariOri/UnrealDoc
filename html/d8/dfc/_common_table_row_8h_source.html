169template <class TTableRow>
170TSharedRef<SWidget> CreateEditableLabelWidget(TSharedRef<TTableRow> TableRow)
171{
172 TSharedRef<SInlineEditableTextBlock> InlineEditableTextBlock = SNew(SInlineEditableTextBlock)
173 .Text(TableRow->GetText())
174 .ToolTipText(TableRow->GetTooltipText())
175 .OnVerifyTextChanged(TableRow, &TTableRow::OnLabelTextVerifyChanged)
176 .OnTextCommitted(TableRow, &TTableRow::OnLabelTextCommitted)
177 .OnEnterEditingMode(TableRow, &TTableRow::OnEnterEditingMode)
178 .OnExitEditingMode(TableRow, &TTableRow::OnExitEditingMode)
179 .IsReadOnly(TableRow, &TTableRow::IsReadOnly)
180 .IsSelected(TableRow, &TTableRow::IsSelected);
181
182 TableRow->GetRenameRequestedDelegate().BindSP(InlineEditableTextBlock, &SInlineEditableTextBlock::EnterEditingMode);
183
184 return InlineEditableTextBlock;
185}
