15// A transactionally safe mutex that works in the following novel ways:
16// - In the open (non-transactional):
17// - Take the lock like before. Simple!
18// - Free the lock like before too.
19// - In the closed (transactional):
20// - Unlock() decrements `TransactionalLockCount` and registers the on-commit handler (if not
21// already registered) to perform the deferred unlock to the underlying mutex, if not rebalanced
22// in the transaction with a lock.
23// - During locking we query `TransactionalLockCount`:
24// * -1 means the transaction performed an Unlock() before the lock, so the underlying mutex
25// must have been locked prior to the start of the transaction. `TransactionalLockCount` is
26// incremented to 0, and a lock is emulated (i.e. TryLock() will return `true` without
27// touching the underlying mutex).
28// * 0 means the underlying mutex will be locked if it hasn't been already (tracked by
29// `bTransactionallyLocked`). Once the underlying mutex is locked, it remains locked for the
30// entire duration of the transaction. This is done as a transactional abort will likely undo
31// the writes to memory that is guarded by the mutex, so the lock needs be held until the
32// transaction is either fully committed or aborted. When the underlying mutex is locked an
33// on-abort handler and on-commit handler are registered to unlock the underlying mutex if
34// necessary.
35//
36// Thus with this approach we will hold this lock for the *entirety* of the transactional nest should
37// we take the lock during the transaction, thus preventing non-transactional code from seeing any
38// modifications we should make.
39struct FTransactionallySafeMutex final
40{
41 FTransactionallySafeMutex() : State(MakeShared<FState>())
42 {
43 }

50 void Lock()
51 {
52 if (AutoRTFM::IsClosed() || AutoRTFM::IsCommittingOrAborting())
53 {
54 LockTransactionally();
55 }
56 else if (AutoRTFM::IsTransactional())
57 {
58 InTheClosed<&FTransactionallySafeMutex::LockTransactionally>();
59 }
60 else
61 {
62 // Non-transactional path. Call Lock() on the mutex directly.
63 State->MutexLock();
64 }
65 }

74 {
75 return InTheClosed<&FTransactionallySafeMutex::TryLockTransactionally>();
76 }

81 }
82 }

97 }
98 }

118 template <bool (FTransactionallySafeMutex::*Functor)()>
119 bool InTheClosed()
120 {
121 bool Result;
122 AutoRTFM::ETransactionStatus Status = AutoRTFM::Close([this, &Result]
123 {
124 Result = (this->*Functor)();
125 });
126 check(Status == AutoRTFM::ETransactionStatus::Executing);
127 return Result;
128 }

130 bool TryLockTransactionally()
131 {
132 check(AutoRTFM::IsClosed() || AutoRTFM::IsCommittingOrAborting());
133 ensure(State->TransactionalLockCount >= -1 && State->TransactionalLockCount <= 1);
134
135 if (State->TransactionalLockCount > 0)
136 {
137 return false; // Attempting double-lock within transaction.
138 }
139
140 if (State->bTransactionallyLocked)
141 {
142 // Underlying mutex has already been locked for the duration of the transaction.
143 State->TransactionalLockCount++;
144 return true;
145 }
146
147 if (State->TransactionalLockCount < 0)
148 {
149 // The mutex was locked before the transaction and unlocked inside the transaction.
150 // As unlock is deferred, bump the lock counter and return true to emulate the lock.
151 State->TransactionalLockCount++;
152 return true;
153 }
154
155 // Attempt to lock the underlying mutex.
156 if (!State->MutexTryLock())
157 {
158 return false;
159 }
160
161 // First time the mutex has been locked during the transaction.
162 // Increment the lock count, mark the transactionally locked flag and register
163 // on-commit and on-abort handlers.
164 State->TransactionalLockCount++;
165 SetTransactionallyLockedAndRegisterHandlers();
166 return true;
167 };
168
169 // Acquires the mutex in a transactionally-safe way. Assumes we are either in the closed or
170 // are mid-commit/mid-abort.
171 bool LockTransactionally()
172 {
173 check(AutoRTFM::IsClosed() || AutoRTFM::IsCommittingOrAborting());
174 ensure(State->TransactionalLockCount >= -1 && State->TransactionalLockCount <= 0);
175
176 State->TransactionalLockCount++;
177
178 if (State->bTransactionallyLocked)
179 {
180 // Underlying mutex has already been locked for the duration of the transaction.
181 return true;
182 }
183
184 if (State->TransactionalLockCount == 0)
185 {
186 // The mutex was locked before the transaction and unlocked inside the transaction.
187 return true;
188 }
189
190 // First time the mutex has been locked during the transaction.
191 // Mark the transactionally locked flag and register on-commit and on-abort handlers.
192 State->MutexLock();
193 SetTransactionallyLockedAndRegisterHandlers();
194 return true;
195 }
196
197 // Releases the mutex in a transactionally-safe way. Assumes we are either in the closed or
198 // are mid-commit/mid-abort.
199 bool UnlockTransactionally()
200 {
201 check(AutoRTFM::IsClosed() || AutoRTFM::IsCommittingOrAborting());
202 ensure(State->TransactionalLockCount >= 0 && State->TransactionalLockCount <= 1);
203
204 State->TransactionalLockCount--;
205 // Use an on-commit handler to unlock the underlying mutex if not balanced with a lock
206 // before the transaction commits.
207 MaybeRegisterCommitHandler();
208 return true;
209 }
210
211 // Called when the underlying mutex is locked for the first time within a transaction nest.
212 // Sets the bTransactionallyLocked flag, calls MaybeRegisterCommitHandler() and registers an
213 // on-abort handler to unlock the underlying mutex.
214 void SetTransactionallyLockedAndRegisterHandlers()
215 {
216 ensure(!State->bTransactionallyLocked);
217 State->bTransactionallyLocked = true;
218
219 MaybeRegisterCommitHandler();
220
221 // Capture State instead of 'this' as State is a TSharedPtr which can outlive this
222 // FTransactionallySafeMutex.
223 AutoRTFM::OnAbort([State = AutoRTFM::TOpenWrapper{this->State}]
224 {
225 State.Object->MutexUnlock();
226 State.Object->ResetTransactionState();
227 });
228 }
229
230 // Registers an on-commit handler (if it hasn't been already registered for this transaction
231 // nest) to unlock the underlying mutex if the transaction unlocked more times than locked.
232 void MaybeRegisterCommitHandler()
233 {
234 if (!State->bRegisteredCommitHandler)
235 {
236 State->bRegisteredCommitHandler = true;
237
238 // Capture State instead of 'this' as State is a TSharedPtr which can outlive this
239 // FTransactionallySafeMutex.
240 AutoRTFM::OnCommit([State = AutoRTFM::TOpenWrapper{this->State}]
241 {
242 // If the transaction's lock count is negative (more unlocks than locks), or the
243 // underlying mutex was locked and then re-balanced to an unlocked state, unlock
244 // the underlying mutex.
245 if (State.Object->TransactionalLockCount < 0 ||
246 (State.Object->bTransactionallyLocked && State.Object->TransactionalLockCount == 0))
247 {
248 State.Object->MutexUnlock();
249 }
250 State.Object->ResetTransactionState();
251 });
252 }

301 // The underlying mutex.
302 UE::FMutex Mutex;
303 // Counter for calls to unlock / lock within a transaction:
304 // -1: mutex has had one more unlock than lock. (e.g. transaction starts locked)
305 // 0: mutex has had the same number of locks as unlocks.
306 // +1: mutex has had one more lock than unlock.
307 // Other values indicate a double-lock or double-unlock within a transaction.
308 int8 TransactionalLockCount = 0;
309 // True if the underlying mutex has been locked for the duration of the transaction.
310 bool bTransactionallyLocked : 1;
311 // True if the on-commit handler has been registered for this transaction nest.
312 bool bRegisteredCommitHandler : 1;
313 };
