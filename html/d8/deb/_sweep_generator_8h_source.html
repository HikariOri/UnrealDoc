142{
143public:
144 FPolygon2d CrossSection;
145 TArray<FVector3d> Path;

147 FFrame3d InitialFrame;
148 // If PathFrames.Num == Path.Num, then PathFrames[k] is used for each step instead of the propagated InitialFrame.
149 TArray<FFrame3d> PathFrames;
150 // If PathScales.Num == Path.Num, then PathScales[k] is applied to the CrossSection at each step (this is combined with StartScale/EndScale, but ignored if bLoop=true)
151 TArray<FVector2d> PathScales;
152
153 bool bCapped = false;
154 bool bLoop = false;
155 ECapType CapType = ECapType::FlatTriangulation;
156
157 // 2D uniform scale of the CrossSection, interpolated along the Path (via arc length) from StartScale to EndScale
158 double StartScale = 1.0;
159 double EndScale = 1.0;
160
161 // Maximum factor by which mitering can expand the cross section at sharp turns, to give the appearance of a consistent cross section width. Only used if > 1.0.
162 // Note: If PathFrames are specified, then bAlignFramesToSampledTangents must be true for mitering to be performed.
163 // (Similar to the MiterLimit concept in SVG / CSS)
164 double MiterLimit = 1.0;
165
166 // Whether to align the frames to the path tangents. Only relevant if PathFrames is specified.
167 bool bAlignFramesToSampledTangents = false;
168
169 // Configure settings to support mitering -- i.e., scaling the cross sections as needed to maintain consistent cross section size through sharp corners, up to the specified scale limit
170 void EnableMitering(double InMiterLimit = 10)
171 {
172 MiterLimit = InMiterLimit;
173 bAlignFramesToSampledTangents = true;
174 }
175
176 // Set the MiterLimit based on the maximum turn angle at which a correct miter should be applied. For turns sharper than this angle, the cross section will appear to shrink at the turn.
177 // @param MiterAngleLimitInDeg Maximum turn angle for correct mitering; should be >= 0 and < 180
178 void SetMiterLimitByAngle(double MiterAngleLimitInDeg)
179 {
180 MiterAngleLimitInDeg = FMath::Clamp(MiterAngleLimitInDeg, 0, 180 - FMathd::ZeroTolerance);
181 MiterLimit = 1.0/FMath::Sin(MiterAngleLimitInDeg * FMathd::DegToRad * .5);
182 }
183
184 // When true, the generator attempts to scale UV's in a way that preserves scaling across different mesh
185 // results, aiming for 1.0 in UV space to be equal to UnitUVInWorldCoordinates in world space. This in
186 // practice means adjusting the U scale relative to the CrossSection curve length and V scale relative
187 // to the distance between vertices on the Path.
188 bool bUVScaleRelativeWorld = false;
189
190 // Only relevant if bUVScaleRelativeWorld is true (see that description)
191 float UnitUVInWorldCoordinates = 100;
192
193 // Optional custom UV values:
194
195 // -If FSweepGeneratorBase::bProfileCurveIsClosed == true and CrossSectionTexCoord.Num() >= CrossSection.VertexCount() + 1
196 // then the first CrossSection.VertexCount() + 1 values will be used as U coordinates.
197 //
198 // -If FSweepGeneratorBase::bProfileCurveIsClosed == true and CrossSectionTexCoord.Num() == CrossSection.VertexCount()
199 // then the CrossSectionTexCoord[0] will be used as the value for the last element.
200 //
201 // -If FSweepGeneratorBase::bProfileCurveIsClosed == false and CrossSectionTexCoord.Num() >= CrossSection.VertexCount()
202 // then the first CrossSection.VertexCount() values will be used as U coordinates.
203 //
204 // -Otherwise, the U coordinates will be automatically generated.
205 TArray<float> CrossSectionTexCoord;
206
207 // -If bLoop == true and PathTexCoord.Num() >= Path.Num() + 1 then the first Path.Num() + 1 values will be used as
208 // V coordinates.
209 //
210 // -If bLoop == true and PathTexCoord.Num() == Path.Num() then the PathTexCoord[0] will be used as the value for the
211 // last element.
212 //
213 // -If bLoop == false and PathTexCoord.Num() >= Path.Num() then the first Path.Num() values will be used as V
214 // coordinates.
215 //
216 // -Otherwise, the V coordinates will be automatically generated.
217 TArray<float> PathTexCoord;
218
219public:
221 GEOMETRYCORE_API virtual FMeshShapeGenerator& Generate() override;
