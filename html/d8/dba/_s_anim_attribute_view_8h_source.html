128 void DisplayNewAttributeContainerSnapshots(
129 const TArray<TTuple<FName, const UE::Anim::FMeshAttributeContainer&>>& InSnapshots,
130 const USkeletalMeshComponent* InOwningComponent)
131 {
132 if (!ensure(InSnapshots.Num() != 0))
133 {
134 ClearListView();
135 return;
136 }
137
138 // we need the SKM to look up bone names
139 if (!InOwningComponent || !InOwningComponent->GetSkeletalMeshAsset())
140 {
141 ClearListView();
142 return;
143 }
144
145 if (ShouldInvalidateListViewCache(InSnapshots, InOwningComponent))
146 {
147 CachedNumSnapshots = InSnapshots.Num();
148 CachedSnapshotNameIndexMap.Reset();
149
150 CachedAttributeIdentifierLists.Reset(InSnapshots.Num());
151 CachedAttributeSnapshotMap.Reset();
152
153 CachedAttributeIdentifierLists.AddDefaulted( InSnapshots.Num());
154
155 for (int32 SnapshotIndex = 0; SnapshotIndex < InSnapshots.Num(); SnapshotIndex++)
156 {
157 const TTuple<FName, const UE::Anim::FMeshAttributeContainer&> Snapshot = InSnapshots[SnapshotIndex];
158 CachedSnapshotNameIndexMap.FindOrAdd(Snapshot.Key) = SnapshotIndex;
159
160 TTuple<FName, TArray<FAnimationAttributeIdentifier>>& CachedIdentifierList = CachedAttributeIdentifierLists[SnapshotIndex];
161 CachedIdentifierList.Key = Snapshot.Key;
162
163 const UE::Anim::FMeshAttributeContainer& AttributeContainer = Snapshot.Value;
164 TArray<FAnimationAttributeIdentifier>& CachedIdentifiers = CachedIdentifierList.Value;
165
166 CachedIdentifiers.AddDefaulted(AttributeContainer.Num());
167
168 const TArray<TWeakObjectPtr<UScriptStruct>>& Types = AttributeContainer.GetUniqueTypes();
169
170 int32 CachedListIdentifierIndex = 0;
171 for (int32 TypeIndex = 0; TypeIndex < Types.Num(); TypeIndex++)
172 {
173 const TArray<UE::Anim::FAttributeId>& Ids = AttributeContainer.GetKeys(TypeIndex);
174
175 for (int32 IdIndex = 0; IdIndex < Ids.Num(); IdIndex++)
176 {
177 const UE::Anim::FAttributeId& Id = Ids[IdIndex];
178 const FName BoneName = InOwningComponent->GetSkeletalMeshAsset()->GetRefSkeleton().GetBoneName(Id.GetIndex());
179
180 const FAnimationAttributeIdentifier Identifier(
181 Id.GetName(), Id.GetIndex() ,BoneName, Types[TypeIndex].Get());
182
183 CachedIdentifiers[CachedListIdentifierIndex] = Identifier;
184
185 CachedAttributeSnapshotMap.FindOrAdd(Identifier).Add(Snapshot.Key);
186
187 CachedListIdentifierIndex++;
188 }
189 }
190 }
191
192 // filtered list should also be refreshed since it depends on the cache
193 RefreshFilteredAttributeEntries();
194
195 // delay value view refresh until tick since this function can be called from animation thread
196 bShouldRefreshValueView = true;
197
198 return;
199 }
200
201 if (SelectedAttribute.IsSet())
202 {
203 for (const FAttributeValueView& ValueView : SelectedAttributeSnapshotValueViews)
204 {
205 const int32* SnapshotIndex = CachedSnapshotNameIndexMap.Find(ValueView.SnapshotName);
206 if (ensure(SnapshotIndex) && ensure(InSnapshots.IsValidIndex(*SnapshotIndex)))
207 {
208 const TTuple<FName, const UE::Anim::FMeshAttributeContainer&> Snapshot = InSnapshots[*SnapshotIndex];
209 const UE::Anim::FMeshAttributeContainer& AttributeContainer = Snapshot.Value;
210
211 ValueView.UpdateValue(AttributeContainer);
212 }
213 }
214 }
215 }

223 template <typename ContainerType>
224 bool ShouldInvalidateListViewCache(
225 const TArray<TTuple<FName, const ContainerType&>>& InSnapshots,
226 const USkeletalMeshComponent* InOwningComponent)
227 {
228 // we need the SKM to look up bone names
229 // it should have been checked
230 check(InOwningComponent);
231
232 if (InSnapshots.Num() != CachedAttributeIdentifierLists.Num())
233 {
234 return true;
235 }
236
237 for (int32 SnapshotIndex = 0; SnapshotIndex < InSnapshots.Num(); SnapshotIndex++)
238 {
239 const TTuple<FName, const ContainerType&> Snapshot = InSnapshots[SnapshotIndex];
240 const TTuple<FName, TArray<FAnimationAttributeIdentifier>>& CachedIdList = CachedAttributeIdentifierLists[SnapshotIndex];
241
242 if (Snapshot.Key != CachedIdList.Key)
243 {
244 return true;
245 }
246
247 if (Snapshot.Value.Num() != CachedIdList.Value.Num())
248 {
249 return true;
250 }
251 }
252
253 for (int32 SnapshotIndex = 0; SnapshotIndex < InSnapshots.Num(); SnapshotIndex++)
254 {
255 const TTuple<FName, const ContainerType&> Snapshot = InSnapshots[SnapshotIndex];
256 const TTuple<FName, TArray<FAnimationAttributeIdentifier>> CachedIdentifierList = CachedAttributeIdentifierLists[SnapshotIndex];
257
258 const ContainerType& AttributeContainer = Snapshot.Value;
259 const TArray<FAnimationAttributeIdentifier>& CachedIdentifiers = CachedIdentifierList.Value;
260
261 const TArray<TWeakObjectPtr<UScriptStruct>>& Types = AttributeContainer.GetUniqueTypes();
262
263 int32 CachedListIdentifierIndex = 0;
264 for (int32 TypeIndex = 0; TypeIndex < Types.Num(); TypeIndex++)
265 {
266 const TArray<UE::Anim::FAttributeId>& Ids = AttributeContainer.GetKeys(TypeIndex);
267
268 for (int32 IdIndex = 0; IdIndex < Ids.Num(); IdIndex++)
269 {
270 const UE::Anim::FAttributeId& Id = Ids[IdIndex];
271 const FName BoneName = InOwningComponent->GetSkeletalMeshAsset()->GetRefSkeleton().GetBoneName(Id.GetIndex());
272
273 FAnimationAttributeIdentifier Identifier(
274 Id.GetName(), Id.GetIndex() ,BoneName, Types[TypeIndex].Get());
275
276 if (!(Identifier == CachedIdentifiers[CachedListIdentifierIndex]))
277 {
278 return true;
279 }
280
281 CachedListIdentifierIndex++;
282 }
283 }
284 }
285
286 return false;
287 }
