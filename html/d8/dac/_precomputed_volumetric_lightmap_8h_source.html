338inline static const float GPointFilteringThreshold = .001f;
339
340template<typename VoxelDataType>
341FLinearColor FilteredVolumeLookup(FVector Coordinate, FIntVector DataDimensions, const VoxelDataType* Data)
342{
343 FVector CoordinateFraction(FMath::Frac(Coordinate.X), FMath::Frac(Coordinate.Y), FMath::Frac(Coordinate.Z));
344 FIntVector FilterNeighborSize(CoordinateFraction.X > GPointFilteringThreshold ? 2 : 1, CoordinateFraction.Y > GPointFilteringThreshold ? 2 : 1, CoordinateFraction.Z > GPointFilteringThreshold ? 2 : 1);
345 FIntVector CoordinateInt000(Coordinate);
346
347 FLinearColor FilteredValue(0, 0, 0, 0);
348 FVector FilterWeight(1.0f, 1.0f, 1.0f);
349
350 for (int32 Z = 0; Z < FilterNeighborSize.Z; Z++)
351 {
352 if (FilterNeighborSize.Z > 1)
353 {
354 FilterWeight.Z = (Z == 0 ? 1.0f - CoordinateFraction.Z : CoordinateFraction.Z);
355 }
356
357 for (int32 Y = 0; Y < FilterNeighborSize.Y; Y++)
358 {
359 if (FilterNeighborSize.Y > 1)
360 {
361 FilterWeight.Y = (Y == 0 ? 1.0f - CoordinateFraction.Y : CoordinateFraction.Y);
362 }
363
364 for (int32 X = 0; X < FilterNeighborSize.X; X++)
365 {
366 if (FilterNeighborSize.X > 1)
367 {
368 FilterWeight.X = (X == 0 ? 1.0f - CoordinateFraction.X : CoordinateFraction.X);
369 }
370
371 FIntVector CoordinateInt = CoordinateInt000 + FIntVector(X, Y, Z);
372 CoordinateInt.X = FMath::Clamp(CoordinateInt.X, 0, DataDimensions.X - 1);
373 CoordinateInt.Y = FMath::Clamp(CoordinateInt.Y, 0, DataDimensions.Y - 1);
374 CoordinateInt.Z = FMath::Clamp(CoordinateInt.Z, 0, DataDimensions.Z - 1);
375 const int32 LinearIndex = ((CoordinateInt.Z * DataDimensions.Y) + CoordinateInt.Y) * DataDimensions.X + CoordinateInt.X;
376
377 FilteredValue += ConvertToLinearColor<VoxelDataType>(Data[LinearIndex]) * FilterWeight.X * FilterWeight.Y * FilterWeight.Z;
378 }
379 }
380 }
381
382 return FilteredValue;
383}

385template<typename VoxelDataType>
386VoxelDataType FilteredVolumeLookupReconverted(FVector Coordinate, FIntVector DataDimensions, const VoxelDataType* Data)
387{
388 FVector CoordinateFraction(FMath::Frac(Coordinate.X), FMath::Frac(Coordinate.Y), FMath::Frac(Coordinate.Z));
389 FIntVector FilterNeighborSize(CoordinateFraction.X > GPointFilteringThreshold ? 2 : 1, CoordinateFraction.Y > GPointFilteringThreshold ? 2 : 1, CoordinateFraction.Z > GPointFilteringThreshold ? 2 : 1);
390
391 if (FilterNeighborSize.X == 1 && FilterNeighborSize.Y == 1 && FilterNeighborSize.Z == 1)
392 {
393 FIntVector CoordinateInt000(Coordinate);
394 const int32 LinearIndex = ((CoordinateInt000.Z * DataDimensions.Y) + CoordinateInt000.Y) * DataDimensions.X + CoordinateInt000.X;
395 return Data[LinearIndex];
396 }
397 else
398 {
399 FLinearColor FilteredValue = FilteredVolumeLookup<VoxelDataType>(Coordinate, DataDimensions, Data);
400 return ConvertFromLinearColor<VoxelDataType>(FilteredValue);
401 }
402}
