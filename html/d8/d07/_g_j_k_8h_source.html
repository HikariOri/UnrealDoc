153 // Calculate the margins used for queries based on shape radius, shape margins and shape types
154 template <typename TGeometryA, typename TGeometryB, typename T>
155 void CalculateQueryMargins(const TGeometryA& A, const TGeometryB& B, T& outMarginA, T& outMarginB)
156 {
157 // Margin selection logic: we only need a small margin for sweeps since we only move the sweeping object
158 // to the point where it just touches.
159 // Spheres and Capsules: always use the core shape and full "margin" because it represents the radius
160 // Sphere/Capsule versus OtherShape: no margin on other
161 // OtherShape versus OtherShape: use margin of the smaller shape, zero margin on the other
162 const T RadiusA = A.GetRadiusf();
163 const T RadiusB = B.GetRadiusf();
164 const bool bHasRadiusA = RadiusA > 0;
165 const bool bHasRadiusB = RadiusB > 0;
166
167 // The sweep margins if required. Only one can be non-zero (we keep the smaller one)
168 const T SweepMarginScale = 0.05f;
169 const bool bAIsSmallest = A.GetMarginf() < B.GetMarginf();
170 const T SweepMarginA = (bHasRadiusA || bHasRadiusB) ? 0.0f : (bAIsSmallest ? SweepMarginScale * A.GetMarginf() : 0.0f);
171 const T SweepMarginB = (bHasRadiusA || bHasRadiusB) ? 0.0f : (bAIsSmallest ? 0.0f : SweepMarginScale * B.GetMarginf());
172
173 // Net margin (note: both SweepMargins are zero if either Radius is non-zero, and only one SweepMargin can be non-zero)
174 outMarginA = RadiusA + SweepMarginA;
175 outMarginB = RadiusB + SweepMarginB;
176 }

208 template <typename T>
209 inline bool GJKIntersectionSimd(const FGeomGJKHelperSIMD& RESTRICT A, const FGeomGJKHelperSIMD& RESTRICT B, FRealSingle EpsilonScale, const T& Translation, const T& Rotation, FRealSingle InThicknessA, const T& InitialDir)
210 {
211 T V = VectorNegate(InitialDir);
212 V = VectorNormalizeSafe(V, MakeVectorRegisterFloatConstant(-1.f, 0.f, 0.f, 0.f));
213
214 const T AToBRotation = VectorQuaternionInverse(Rotation);
215 bool bTerminate;
216 bool bNearZero = false;
217 int NumIterations = 0;
218 T PrevDist2Simd = MakeVectorRegisterFloatConstant(FLT_MAX, FLT_MAX, FLT_MAX, FLT_MAX);
219
220 T SimplexSimd[4] = { VectorZero(), VectorZero(), VectorZero(), VectorZero() };
221 T BarycentricSimd;
222 int32 NumVerts = 0;
223
224 FRealSingle ThicknessA;
225 FRealSingle ThicknessB;
226 CalculateQueryMargins(A, B, ThicknessA, ThicknessB);
227 ThicknessA += InThicknessA;
228
229 EpsilonScale = FMath::Min<FRealSingle>(EpsilonScale, 1e5f);
230
231 const FRealSingle InflationReal = ThicknessA + ThicknessB + 1e-6f * EpsilonScale;
232 const T Inflation = VectorSet1(InflationReal);
233 const T Inflation2 = VectorMultiply(Inflation, Inflation);
234
235 int32 VertexIndexA = INDEX_NONE, VertexIndexB = INDEX_NONE;
236 do
237 {
238 if (!ensure(NumIterations++ < 32)) //todo: take this out
239 {
240 break; //if taking too long just stop. This should never happen
241 }
242 const T NegVSimd = VectorNegate(V);
243 const T SupportASimd = A.SupportFunction(TMakeVectorRegisterFloatFromDouble<T>(NegVSimd), ThicknessA);
244 const T VInBSimd = VectorQuaternionRotateVector(AToBRotation, V);
245 const T SupportBLocalSimd = B.SupportFunction(TMakeVectorRegisterFloatFromDouble<T>(VInBSimd), ThicknessB);
246 const T SupportBSimd = VectorAdd(VectorQuaternionRotateVector(Rotation, SupportBLocalSimd), Translation);
247 const T WSimd = VectorSubtract(SupportASimd, SupportBSimd);
248
249 if (::VectorMaskBits(VectorCompareGT(VectorDot3(V, WSimd), Inflation)))
250 {
251 return false;
252 }
253
254 SimplexSimd[NumVerts++] = WSimd;
255
256 V = VectorSimplexFindClosestToOrigin<T, false>(SimplexSimd, NumVerts, BarycentricSimd, nullptr, nullptr);
257
258 const T NewDist2Simd = VectorDot3(V, V);
259 bNearZero = ::VectorMaskBits(VectorCompareLT(NewDist2Simd, Inflation2)) != 0;
260
261 //as simplices become degenerate we will stop making progress. This is a side-effect of precision, in that case take V as the current best approximation
262 //question: should we take previous v in case it's better?
263 bool bMadeProgress = ::VectorMaskBits(VectorCompareLT(NewDist2Simd, PrevDist2Simd)) != 0;
264 bTerminate = bNearZero || !bMadeProgress;
265 PrevDist2Simd = NewDist2Simd;
266
267 if (!bTerminate)
268 {
269 V = VectorDivide(V, VectorSqrt(NewDist2Simd));
270 }
271
272 } while (!bTerminate);
273
274 return bNearZero;
275 }

305 bool GJKIntersectionSameSpaceSimd(const TGeometryA& A, const TGeometryB& B, FRealSingle InThicknessA, const VectorRegister4Float& InitialDir)
306 {
307 VectorRegister4Float V = VectorNegate(InitialDir);
308
309 //ensure(FMath::Abs(VectorDot3Scalar(V, V) - 1.0f) < 1e-3f ); // Check that the vector is normalized, comment out for performance

313 int32 NumVerts = 0;
314 VectorRegister4Float Barycentric = VectorZero();
315
316 bool bTerminate;
317 bool bNearZero = false;
318 int NumIterations = 0;
319 VectorRegister4Float PrevDist2 = GlobalVectorConstants::BigNumber;
320 FRealSingle ThicknessA;
321 FRealSingle ThicknessB;
322 CalculateQueryMargins(A, B, ThicknessA, ThicknessB);
323 ThicknessA += InThicknessA;
324

326 const FRealSingle Inflation2 = Inflation * Inflation;
327
328 VectorRegister4Float InflationSimd = VectorSet1(Inflation);
329 VectorRegister4Float Inflation2Simd = VectorMultiply(InflationSimd, InflationSimd);
330 do
331 {
332 if (!ensure(NumIterations++ < 32)) //todo: take this out
333 {
334 break; //if taking too long just stop. This should never happen
335 }
336 const VectorRegister4Float NegV = VectorNegate(V);
337 const VectorRegister4Float SupportA = A.SupportCoreSimd(NegV, ThicknessA);
338 const VectorRegister4Float VInB = V; // same space
339 const VectorRegister4Float SupportB = B.SupportCoreSimd(VInB, ThicknessB);
340 const VectorRegister4Float W = VectorSubtract(SupportA, SupportB);
341
342 VectorRegister4Float VDotW = VectorDot3(V, W);
343
344 if (::VectorMaskBits(VectorCompareGT(VDotW, InflationSimd)))
345 {
346 return false;
347 }
348
349 Simplex[NumVerts++] = W;
350
351 V = VectorSimplexFindClosestToOrigin<VectorRegister4Float, false>(Simplex, NumVerts, Barycentric, nullptr, nullptr);
352 const VectorRegister4Float NewDist2 = VectorDot3(V, V);
353
354 bNearZero = static_cast<bool>(::VectorMaskBits(VectorCompareGT(Inflation2Simd, NewDist2)));
355
356 //as simplices become degenerate we will stop making progress. This is a side-effect of precision, in that case take V as the current best approximation
357 //question: should we take previous v in case it's better?
358 const bool bMadeProgress = static_cast<bool>(::VectorMaskBits(VectorCompareGT(PrevDist2, NewDist2)));
359 bTerminate = bNearZero || !bMadeProgress;
360
361 PrevDist2 = NewDist2;
362
363 if (!bTerminate)
364 {
365 V = VectorDivide(V, VectorSqrt(NewDist2));
366 }
367
368 } while (!bTerminate);
369
370 return bNearZero;
371 }

380 template<typename T>
381 class TGJKSimplexData
382 {
383 public:
384 TGJKSimplexData()
385 : NumVerts(0)
386 {}
387
388 // Clear the data - used to start a GJK search from the default search direction
389 void Reset()
390 {
391 NumVerts = 0;
392 }
393
394 // Save any data that was not directly updated while iterating in GJK
395 void Save(const FSimplex InSimplexIDs)
396 {
397 // We don't need to store the simplex vertex order because the indices are always
398 // sorted at the end of each iteration. We just need to know how many vertices we have.
399 NumVerts = InSimplexIDs.NumVerts;
400 }
401
402 // Recompute the Simplex and separating vector from the stored data at the current relative transform
403 // This aborts if we have no simplex data to restore or the origin is inside the simplex. Outputs must already
404 // have reasonable default values for running GJK without a warm-start.
405 void Restore(const TRigidTransform<T, 3>& BToATM, FSimplex& OutSimplexIDs, TVec3<T> OutSimplex[], TVec3<T>& OutV, T& OutDistance, const T Epsilon)
406 {
407 if (NumVerts > 0)
408 {
409 OutSimplexIDs.NumVerts = NumVerts;
410
411 for (int32 VertIndex = 0; VertIndex < NumVerts; ++VertIndex)
412 {
413 OutSimplexIDs.Idxs[VertIndex] = VertIndex;
414 OutSimplex[VertIndex] = As[VertIndex] - BToATM.TransformPositionNoScale(Bs[VertIndex]);
415 }
416
417 const TVec3<T> V = SimplexFindClosestToOrigin(OutSimplex, OutSimplexIDs, Barycentric, As, Bs);
418 const T Distance = V.Size();
419
420 // If the origin is inside the simplex at the new transform, we need to abort the restore
421 // This is necessary to cover the very-small separation case where we use the normal
422 // calculated in the previous iteration in GJKL, but we have no way to restore that.
423 // Note: we have already written to the simplex but that's ok because we reset the vert count.
424 if (Distance > Epsilon)
425 {
426 OutV = V / Distance;
427 OutDistance = Distance;
428 }
429 else
430 {
431 OutSimplexIDs.NumVerts = 0;
432 }
433 }
434 }
435
436 void Restore2(const TRigidTransform<T, 3>& BToATM, int32& OutNumVerts, TVec3<T> OutSimplex[], TVec3<T>& OutV, T& OutDistance, const T Epsilon)
437 {
438 OutNumVerts = 0;
439
440 if (NumVerts > 0)
441 {
442 for (int32 VertIndex = 0; VertIndex < NumVerts; ++VertIndex)
443 {
444 OutSimplex[VertIndex] = As[VertIndex] - BToATM.TransformPositionNoScale(Bs[VertIndex]);
445 }
446
447 const TVec3<T> V = SimplexFindClosestToOrigin2(OutSimplex, NumVerts, Barycentric, As, Bs);
448 const T DistanceSq = V.SizeSquared();
449
450 // If the origin is inside the simplex at the new transform, we need to abort the restore
451 // This is necessary to cover the very-small separation case where we use the normal
452 // calculated in the previous iteration in GJKL, but we have no way to restore that.
453 // Note: we have already written to the simplex but that's ok because we reset the vert count.
454 if (DistanceSq > FMath::Square(Epsilon))
455 {
456 const T Distance = FMath::Sqrt(DistanceSq);
457 OutNumVerts = NumVerts;
458 OutV = V / Distance;
459 OutDistance = Distance;
460 }
461 }
462 }

1899 }

1907 }
1908 }
1909 }
1910 else

1915 }

2085 template <typename T, typename GJKShapeTypeA, typename GJKShapeTypeB>
2086 EGJKDistanceResult GJKDistance(const GJKShapeTypeA& A, const GJKShapeTypeB& B, const TVec3<T>& InitialV, T& OutDistance, TVec3<T>& OutNearestA, TVec3<T>& OutNearestB, TVec3<T>& OutNormalA, const T Epsilon = (T)1e-3, const int32 MaxIts = 16)
2087 {
2088 FSimplex SimplexIDs;
2089 TVec3<T> Simplex[4], SimplexA[4], SimplexB[4];
2090 T Barycentric[4] = { -1, -1, -1, -1 };
2091
2092 const T AMargin = A.GetMargin();
2093 const T BMargin = B.GetMargin();
2094 T Mu = 0;
2095
2096 // Initial vector in Minkowski(A - B)
2097 // NOTE: If InitialV is not in the Minkowski Sum we may incorrectly early-out in the bCloseEnough chaeck below
2098 TVec3<T> V = InitialV;
2099 T VLen = V.Size();
2100 int32 VertexIndexA = INDEX_NONE, VertexIndexB = INDEX_NONE;
2101
2102 int32 It = 0;
2103 while (VLen > Epsilon)
2104 {
2105 // Find a new point in A-B that is closer to the origin
2106 // NOTE: we do not use support thickness here. Thickness is used when separating objects
2107 // so that GJK can find a solution, but that can be added in a later step.
2108 const TVec3<T> SupportA = A.SupportCore(-V, AMargin, nullptr, VertexIndexA);
2109 const TVec3<T> SupportB = B.SupportCore(V, BMargin, nullptr, VertexIndexB);
2110 const TVec3<T> W = SupportA - SupportB;
2111
2112 T D = TVec3<T>::DotProduct(V, W) / VLen;
2113 Mu = FMath::Max(Mu, D);
2114
2115 // See if we are still making progress toward the origin
2116 bool bCloseEnough = ((VLen - Mu) < Epsilon);
2117 if (bCloseEnough || (++It > MaxIts))
2118 {
2119 // We have reached the minimum to within tolerance. Or we have reached max iterations, in which
2120 // case we (probably) have a solution but with an error larger than Epsilon (technically we could be missing
2121 // the fact that we were going to eventually find the origin, but it'll be a close call so the approximation
2122 // is still good enough).
2123 if (SimplexIDs.NumVerts == 0)
2124 {
2125 // Our initial guess of V was already the minimum separating vector
2126 OutNearestA = SupportA;
2127 OutNearestB = SupportB;
2128 }
2129 else
2130 {
2131 // The simplex vertices are the nearest point/line/face
2132 OutNearestA = TVec3<T>(0, 0, 0);
2133 OutNearestB = TVec3<T>(0, 0, 0);
2134 for (int32 VertIndex = 0; VertIndex < SimplexIDs.NumVerts; ++VertIndex)
2135 {
2136 int32 WIndex = SimplexIDs[VertIndex];
2137 check(Barycentric[WIndex] >= (T)0);
2138 OutNearestA += Barycentric[WIndex] * SimplexA[WIndex];
2139 OutNearestB += Barycentric[WIndex] * SimplexB[WIndex];
2140 }
2141 }
2142 const TVec3<T> NormalA = -V / VLen;
2143 OutDistance = VLen - (AMargin + BMargin);
2144 OutNearestA += AMargin * NormalA;
2145 OutNearestB -= BMargin * NormalA;
2146 OutNormalA = NormalA;
2147
2148 // NearestB should be in B-space but is currently in A-space
2149 OutNearestB = B.InverseTransformPositionNoScale(OutNearestB);
2150
2151 return (OutDistance >= 0.0f) ? EGJKDistanceResult::Separated : EGJKDistanceResult::Contact;
2152 }
2153
2154 // Add the new vertex to the simplex
2155 SimplexIDs[SimplexIDs.NumVerts] = SimplexIDs.NumVerts;
2156 Simplex[SimplexIDs.NumVerts] = W;
2157 SimplexA[SimplexIDs.NumVerts] = SupportA;
2158 SimplexB[SimplexIDs.NumVerts] = SupportB;
