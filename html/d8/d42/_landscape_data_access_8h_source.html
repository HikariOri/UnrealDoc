54 FORCEINLINE FColor PackNormal(const FVector& InNormal)
55 {
56 const FVector Normal = InNormal.GetSafeNormal();
57
58 FColor Color(ForceInit);
59 Color.B = static_cast<uint8>(FMath::Clamp(Normal.X * 127.5 + 127.5, 0.0f, 255.0f));
60 Color.A = static_cast<uint8>(FMath::Clamp(Normal.Y * 127.5 + 127.5, 0.0f, 255.0f));
61 return MoveTemp(Color);
62 }
63
64 FORCEINLINE FVector UnpackNormal(const FColor& InHeightmapSample)
65 {
66 FVector Normal;
67 Normal.X = 2.f * static_cast<float>(InHeightmapSample.B) / 255.f - 1.f;
68 Normal.Y = 2.f * static_cast<float>(InHeightmapSample.A) / 255.f - 1.f;
69 Normal.Z = FMath::Sqrt(FMath::Max(1.0f - (FMath::Square(Normal.X) + FMath::Square(Normal.Y)), 0.0f));
70 return Normal;
71 }
72

74 {
75 const uint16 DefaultHeight = static_cast<uint16>(MidValue);
76 const uint8 DefaultNormal = static_cast<uint8>(MidNormalValue);
77 return FColor(DefaultHeight >> 8, DefaultHeight & 255, DefaultNormal, DefaultNormal);

109 void* LockMip(UTexture2D* Texture, int32 MipLevel)
110 {
111 check(Texture->Source.IsValid());
112 check(MipLevel < Texture->Source.GetNumMips());
113
114 TArray<FLockedMipDataInfo>* MipInfo = LockedMipInfoMap.Find(Texture);
115 if( MipInfo == nullptr )
116 {
117 MipInfo = &LockedMipInfoMap.Add(Texture, TArray<FLockedMipDataInfo>() );
118 for (int32 i = 0; i < Texture->Source.GetNumMips(); ++i)
119 {
120 MipInfo->Add(FLockedMipDataInfo());
121 }
122 }
123
124 if( (*MipInfo)[MipLevel].MipData.Num() == 0 )
125 {
126 verify( Texture->Source.GetMipData((*MipInfo)[MipLevel].MipData, MipLevel) );
127 }
128 (*MipInfo)[MipLevel].LockCount++;
129
130 return (*MipInfo)[MipLevel].MipData.GetData();
131 }
132
133 void UnlockMip(UTexture2D* Texture, int32 MipLevel)
134 {
135 TArray<FLockedMipDataInfo>* MipInfo = LockedMipInfoMap.Find(Texture);
136 check(MipInfo);
137
138 if ((*MipInfo)[MipLevel].LockCount <= 0)
139 return;
140 (*MipInfo)[MipLevel].LockCount--;
141 if( (*MipInfo)[MipLevel].LockCount == 0 )
142 {
143 check( (*MipInfo)[MipLevel].MipData.Num() != 0 );
144 (*MipInfo)[MipLevel].MipData.Empty();
145 }
146 }

148private:
149 TMap<UTexture2D*, TArray<FLockedMipDataInfo> > LockedMipInfoMap;

177 void ComponentXYToSubsectionXY(int32 CompX, int32 CompY, int32& SubNumX, int32& SubNumY, int32& SubX, int32& SubY ) const
178 {
179 // We do the calculation as if we're looking for the previous vertex.
180 // This allows us to pick up the last shared vertex of every subsection correctly.
181 SubNumX = (CompX-1) / (SubsectionSizeVerts - 1);
182 SubNumY = (CompY-1) / (SubsectionSizeVerts - 1);
183 SubX = (CompX-1) % (SubsectionSizeVerts - 1) + 1;
184 SubY = (CompY-1) % (SubsectionSizeVerts - 1) + 1;
185
186 // If we're asking for the first vertex, the calculation above will lead
187 // to a negative SubNumX/Y, so we need to fix that case up.
188 if( SubNumX < 0 )
189 {
190 SubNumX = 0;
191 SubX = 0;
192 }
193
194 if( SubNumY < 0 )
195 {
196 SubNumY = 0;
197 SubY = 0;
198 }
199 }
200
201 void VertexXYToTexelXY(int32 VertX, int32 VertY, int32& OutX, int32& OutY) const
202 {
203 int32 SubNumX, SubNumY, SubX, SubY;
204 ComponentXYToSubsectionXY(VertX, VertY, SubNumX, SubNumY, SubX, SubY);
205
206 OutX = SubNumX * SubsectionSizeVerts + SubX;
207 OutY = SubNumY * SubsectionSizeVerts + SubY;
208 }

417void FillCornerValues(uint8& CornerSet, T* CornerValues)
418{
419 uint8 OriginalSet = CornerSet;
420
421 if (CornerSet)
422 {
423 // Fill unset values
424 while (CornerSet != 15)
425 {
426 if (CornerSet != 15 && (OriginalSet & 1))
427 {
428 if (!(CornerSet & 1 << 1))
429 {
430 CornerValues[1] = CornerValues[0];
431 CornerSet |= 1 << 1;
432 }
433 if (!(CornerSet & 1 << 2))
434 {
435 CornerValues[2] = CornerValues[0];
436 CornerSet |= 1 << 2;
437 }
438 }
439 if (CornerSet != 15 && (OriginalSet & 1 << 1))
440 {
441 if (!(CornerSet & 1))
442 {
443 CornerValues[0] = CornerValues[1];
444 CornerSet |= 1;
445 }
446 if (!(CornerSet & 1 << 3))
447 {
448 CornerValues[3] = CornerValues[1];
449 CornerSet |= 1 << 3;
450 }
451 }
452 if (CornerSet != 15 && (OriginalSet & 1 << 2))
453 {
454 if (!(CornerSet & 1))

457 CornerSet |= 1;
458 }
459 if (!(CornerSet & 1 << 3))

462 CornerSet |= 1 << 3;

464 }
