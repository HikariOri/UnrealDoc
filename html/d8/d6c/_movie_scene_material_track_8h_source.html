16UENUM(BlueprintType)
17enum class EComponentMaterialType : uint8
18{
19 /* Empty/Uninitialized*/
20 Empty,
21 /* A material in one of the indexed slots on a primitive component*/
22 IndexedMaterial,
23 /* An overlay material on a mesh component*/
24 OverlayMaterial,
25 /* A decal material*/
26 DecalMaterial,
27 /* Volumetric Cloud Material*/
28 VolumetricCloudMaterial,
29 /* A per-slot overlay material on a mesh component */
30 SlotOverlayMaterial,
31};

36USTRUCT(BlueprintType)
37struct FComponentMaterialInfo
38{
39 GENERATED_BODY()
40
41 UPROPERTY(EditAnywhere, BlueprintReadWrite, Category="Sequencer|Material")
42 FName MaterialSlotName;
43
44 UPROPERTY(EditAnywhere, BlueprintReadWrite, Category = "Sequencer|Material")
45 int32 MaterialSlotIndex = 0;
46
47 UPROPERTY(EditAnywhere, BlueprintReadWrite, Category = "Sequencer|Material")
48 EComponentMaterialType MaterialType = EComponentMaterialType::Empty;
49
50 friend uint32 GetTypeHash(const FComponentMaterialInfo& In)
51 {
52 return GetTypeHash(In.MaterialSlotName) ^ ::GetTypeHash(In.MaterialSlotIndex) ^ ::GetTypeHash(In.MaterialType);
53 }
54 friend bool operator==(const FComponentMaterialInfo& A, const FComponentMaterialInfo& B)
55 {
56 return A.MaterialSlotIndex == B.MaterialSlotIndex && A.MaterialSlotName == B.MaterialSlotName && A.MaterialType == B.MaterialType;
57 }
58
59 FString ToString() const
60 {
61 switch (MaterialType)
62 {
63 case EComponentMaterialType::IndexedMaterial:
64 if (!MaterialSlotName.IsNone())
65 {
66 return FString::Printf(TEXT("Material Slot: %s"), *MaterialSlotName.ToString());
67 }
68 else
69 {
70 return FString::Printf(TEXT("Material Element %i"), MaterialSlotIndex);
71 }
72 break;
73 case EComponentMaterialType::OverlayMaterial:
74 return TEXT("Overlay Material");
75 break;
76 case EComponentMaterialType::DecalMaterial:
77 return TEXT("Decal Material");
78 break;
79 case EComponentMaterialType::SlotOverlayMaterial:
80 if (!MaterialSlotName.IsNone())
81 {
82 return FString::Printf(TEXT("Overlay Material Slot: %s"), *MaterialSlotName.ToString());
83 }
84 else
85 {
86 return FString::Printf(TEXT("Overlay Material Element %i"), MaterialSlotIndex);
87 }
88 break;
89 }
90 return FString();
91 }
92};
