281inline void FindPrimitiveType(uint32 InPrimitiveType, uint32 InNumPrimitives, GLenum &DrawMode, GLsizei &NumElements)
282{
283 DrawMode = GL_TRIANGLES;
284 NumElements = InNumPrimitives;
285
286 switch (InPrimitiveType)
287 {
288 case PT_TriangleList:
289 DrawMode = GL_TRIANGLES;
290 NumElements = InNumPrimitives * 3;
291 break;
292 case PT_TriangleStrip:
293 DrawMode = GL_TRIANGLE_STRIP;
294 NumElements = InNumPrimitives + 2;
295 break;
296 case PT_LineList:
297 DrawMode = GL_LINES;
298 NumElements = InNumPrimitives * 2;
299 break;
300 case PT_PointList:
301 DrawMode = GL_POINTS;
302 NumElements = InNumPrimitives;
303 break;
304 default:
305 UE_LOGF(LogRHI, Fatal,"Unsupported primitive type %u", InPrimitiveType);
306 break;
307 }
308}

310inline uint32 FindUniformElementSize(GLenum UniformType)
311{
312 switch (UniformType)
313 {
314 case GL_FLOAT:

316 case GL_FLOAT_VEC2:
317 return sizeof(float) * 2;
318 case GL_FLOAT_VEC3:

323 case GL_INT:
324 case GL_BOOL:

332 case GL_INT_VEC4:
333 case GL_BOOL_VEC4:
334 return sizeof(uint32) * 4;
335
336 case GL_FLOAT_MAT2:
337 return sizeof(float) * 4;
338 case GL_FLOAT_MAT3:
339 return sizeof(float) * 9;
340 case GL_FLOAT_MAT4:
341 return sizeof(float) * 16;
342 case GL_FLOAT_MAT2x3:
343 return sizeof(float) * 6;
344 case GL_FLOAT_MAT2x4:
345 return sizeof(float) * 8;
346 case GL_FLOAT_MAT3x2:
347 return sizeof(float) * 6;
348 case GL_FLOAT_MAT3x4:
349 return sizeof(float) * 12;
350 case GL_FLOAT_MAT4x2:
351 return sizeof(float) * 8;
352 case GL_FLOAT_MAT4x3:
353 return sizeof(float) * 12;
354
355 case GL_SAMPLER_1D:
356 case GL_SAMPLER_2D:
357 case GL_SAMPLER_3D:
358 case GL_SAMPLER_CUBE:
359 case GL_SAMPLER_1D_SHADOW:

362 return sizeof(uint32);
