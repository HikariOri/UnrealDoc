346 void Tick( const FGeometry& AllottedGeometry, const double InCurrentTime, const float InDeltaTime ) override
347 {
348 if (PendingMessages.Num() > 0)
349 {
350 FScopeLock ScopedLock(&CriticalSection);
351 for (int32 Index = 0; Index < PendingMessages.Num(); ++Index)
352 {
353 MessageList.Add(PendingMessages[Index]);
354 }
355 PendingMessages.Reset();
356 MessageListView->RequestListRefresh();
357
358 // only scroll when at the end of the listview
359 if (FMath::IsNearlyEqual(MessageListView->GetScrollDistanceRemaining().Y, 0.0f, 1.e-7f))
360 {
361 MessageListView->RequestScrollIntoView(MessageList.Last());
362 }
363 }
364 SaveButton->SetEnabled(MessageList.Num() > 0);
365 ClearButton->SetEnabled(MessageList.Num() > 0);
366 CopyButton->SetEnabled(MessageListView->GetNumItemsSelected() > 0);
367 }

371 void HandleOutputReceived(const FString& InMessage)
372 {
373 FScopeLock ScopedLock(&CriticalSection);
374 ELogVerbosity::Type Verbosity = ELogVerbosity::Log;
375
376 if (InMessage.Contains(TEXT("Automation.ParseCommandLine:"), ESearchCase::CaseSensitive))
377 {
378 Verbosity = ELogVerbosity::Display;
379 }
380 else if ( InMessage.Contains(TEXT("Error:"), ESearchCase::IgnoreCase) )
381 {
382 Verbosity = ELogVerbosity::Error;
383 }
384 else if ( InMessage.Contains(TEXT("Warning:"), ESearchCase::IgnoreCase) )
385 {
386 Verbosity = ELogVerbosity::Warning;
387 }
388 TSharedPtr<FProjectLauncherMessage> Message = MakeShareable(new FProjectLauncherMessage(FText::FromString(InMessage), Verbosity));
389 PendingMessages.Add(Message);
390 }

392 // Callback for getting the filled percentage of the progress bar.
393 TOptional<float> HandleProgressBarPercent( ) const
394 {
395 if (TaskList.Num() > 0)
396 {
397 ILauncherWorkerPtr LauncherWorkerPtr = LauncherWorker.Pin();
398
399 if (LauncherWorkerPtr.IsValid())
400 {
401 int32 NumFinished = 0;
402
403 for (int32 TaskIndex = 0; TaskIndex < TaskList.Num(); ++TaskIndex)
404 {
405 if (TaskList[TaskIndex]->IsFinished())
406 {
407 ++NumFinished;
408 }
409 }
410
411 return ((float)NumFinished / TaskList.Num());
412 }
413 }
414
415 return 0.0f;
416 }

418 // Callback for getting the text of the progress text box.
419 FText HandleProgressTextBlockText( ) const
420 {
421 ILauncherWorkerPtr LauncherWorkerPtr = LauncherWorker.Pin();
422
423 if (LauncherWorkerPtr.IsValid())
424 {
425 if ((LauncherWorkerPtr->GetStatus() == ELauncherWorkerStatus::Busy) ||
426 (LauncherWorkerPtr->GetStatus() == ELauncherWorkerStatus::Canceling))
427 {
428 return LOCTEXT("OperationInProgressText", "Operation in progress...");
429 }
430
431 int32 NumCanceled = 0;
432 int32 NumCompleted = 0;
433 int32 NumFailed = 0;
434
435 for (int32 TaskIndex = 0; TaskIndex < TaskList.Num(); ++TaskIndex)
436 {
437 ELauncherTaskStatus::Type TaskStatus = TaskList[TaskIndex]->GetStatus();
438
439 if (TaskStatus == ELauncherTaskStatus::Canceled)
440 {
441 ++NumCanceled;
442 }
443 else if (TaskStatus == ELauncherTaskStatus::Completed)
444 {
445 ++NumCompleted;
446 }
447 else if (TaskStatus == ELauncherTaskStatus::Failed)
448 {
449 ++NumFailed;
450 }
451 }
452
453 FFormatNamedArguments Args;
454 Args.Add(TEXT("NumCompleted"), NumCompleted);
455 Args.Add(TEXT("NumFailed"), NumFailed);
456 Args.Add(TEXT("NumCanceled"), NumCanceled);
457
458 return FText::Format(LOCTEXT("TasksFinishedFormatText", "Operation finished. Completed: {NumCompleted}, Failed: {NumFailed}, Canceled: {NumCanceled}"), Args);
459 }
460
461 return FText::GetEmpty();
462 }

624 void SaveLog()
625 {
626 IDesktopPlatform* DesktopPlatform = FDesktopPlatformModule::Get();
627
628 if (DesktopPlatform != NULL)
629 {
630 TArray<FString> Filenames;
631
632 TSharedPtr<SWindow> ParentWindow = FSlateApplication::Get().FindWidgetWindow(AsShared());
633 void* ParentWindowHandle = (ParentWindow.IsValid() && ParentWindow->GetNativeWindow().IsValid()) ? ParentWindow->GetNativeWindow()->GetOSWindowHandle() : nullptr;
634
635 if (DesktopPlatform->SaveFileDialog(
636 ParentWindowHandle,
637 LOCTEXT("SaveLogDialogTitle", "Save Log As...").ToString(),
638 LastLogFileSaveDirectory,
639 TEXT("ProjectLauncher.log"),
640 TEXT("Log Files (*.log)|*.log"),
641 EFileDialogFlags::None,
642 Filenames))
643 {
644 if (Filenames.Num() > 0)
645 {
646 FString Filename = Filenames[0];
647
648 // keep path as default for next time
649 LastLogFileSaveDirectory = FPaths::GetPath(Filename);
650
651 // add a file extension if none was provided
652 if (FPaths::GetExtension(Filename).IsEmpty())
653 {
654 Filename += Filename + TEXT(".log");
655 }
656
657 // save file
658 FArchive* LogFile = IFileManager::Get().CreateFileWriter(*Filename);
659
660 if (LogFile != NULL)
661 {
662 for( int32 Index = 0; Index < MessageList.Num(); ++Index )
663 {
664 FString LogEntry = MessageList[Index]->Message.ToString() + LINE_TERMINATOR;
665
666 LogFile->Serialize(TCHAR_TO_ANSI(*LogEntry), LogEntry.Len());
667 }
668
669 LogFile->Close();
670
671 delete LogFile;
672 }
673 else
674 {
675 FMessageDialog::Open(EAppMsgType::Ok, LOCTEXT("SaveLogDialogFileError", "Failed to open the specified file for saving!"));
676 }
677 }
678 }
679 }
680 else
681 {
682 FMessageDialog::Open(EAppMsgType::Ok, LOCTEXT("SaveLogDialogUnsupportedError", "Saving is not supported on this platform!"));
683 }
