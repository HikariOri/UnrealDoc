704 // Allocation flags to account for conditions not covered by vkGet*MemoryRequirements or resource usage flags
705 enum class EVulkanAllocationFlags : uint16
706 {
707 None = 0x0000,
708
709 HostVisible = 0x0001, // Will be written from CPU, will likely contain the HOST_VISIBLE + HOST_COHERENT flags
710 HostCached = 0x0002, // Will be used for readback, will likely contain the HOST_CACHED flag. Implies HostVisible.
711 PreferBAR = 0x0004, // Will be allocated from a HOST_VISIBLE + DEVICE_LOCAL heap if available and possible (HOST_VISIBLE if not)
712
713 Dedicated = 0x0008, // Will not share a memory block with other resources
714 External = 0x0010, // To be used with VK_KHR_external_memory
715 Memoryless = 0x0020, // Will use LAZILY_ALLOCATED
716 Protected = 0x0040, // Will use PROTECTED
717
718 NoError = 0x0080, // OOM is not fatal, return an invalid allocation
719 AutoBind = 0x0100, // Will automatically bind the allocation to the supplied VkBuffer/VkImage, avoids an extra lock to bind separately
720 };
