42 template <typename CharType>
43 [[nodiscard]] static uint32 StrCrc32(const CharType* Data, uint32 CRC = 0)
44 {
45 // We ensure that we never try to do a StrCrc32 with a CharType of more than 4 bytes. This is because
46 // we always want to treat every CRC as if it was based on 4 byte chars, even if it's less, because we
47 // want consistency between equivalent strings with different character types.
48 static_assert(sizeof(CharType) <= 4, "StrCrc32 only works with CharType up to 32 bits.");
49
50 CRC = ~CRC;
51 while (CharType Ch = *Data++)
52 {
53 uint32 V = CRC ^ Ch;
54 CRC =
55 CRCTablesSB8[3][ V & 0xFF] ^
56 CRCTablesSB8[2][(V >> 8) & 0xFF] ^
57 CRCTablesSB8[1][(V >> 16) & 0xFF] ^
58 CRCTablesSB8[0][ V >> 24 ];
59 }
60 return ~CRC;
61 }

99 {

103 uint32 CRC = 0xFFFFFFFF;
104 while (*Data)

106 CharType C = *Data++;
107 int32 CL = (C&255);
108 CRC = (CRC << 8) ^ CRCTable_DEPRECATED[(CRC >> 24) ^ CL];

210template <>
211inline uint32 FCrc::Strihash_DEPRECATED(const UTF8CHAR* Data)
212{
213 // We can't utilize StringConv.h here due to circular includes, so do the conversion manually
214
215 int32 Len = FPlatformString::Strlen(Data);
216 int32 ConvertedLen = FPlatformString::ConvertedLength<WIDECHAR>(Data, Len);
217
218 WIDECHAR* Temp = new WIDECHAR[ConvertedLen];
219
220 WIDECHAR* TempEnd = FPlatformString::Convert(Temp, ConvertedLen, Data, Len);
221 checkf(TempEnd!=nullptr, TEXT("String conversion unsuccessful"));
222
223 // This doesn't work for strings containing characters outside the BMP, but
224 // then neither does the WIDECHAR overload.
225 uint32 Result = FCrc::Strihash_DEPRECATED(ConvertedLen, Temp);
226
227 delete [] Temp;
228
229 return Result;
230}
