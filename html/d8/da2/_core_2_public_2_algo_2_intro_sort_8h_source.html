31 template <typename IteratorType, typename IndexType, typename ProjectionType, typename PredicateType>
32 void IntroSortInternal(IteratorType First, IndexType Num, ProjectionType Proj, PredicateType Predicate)
33 {
34 struct FStack
35 {
36 IteratorType Min;
37 IteratorType Max;
38 uint32 MaxDepth;
39 };
40
41 if( Num < 2 )
42 {
43 return;
44 }
45
46 FStack RecursionStack[32] = { {First, First + (Num - 1), (uint32)(FMath::Loge((float)Num) * 2.f)} };
47 FStack Current;
48 FStack Inner;
49 for( FStack* StackTop=RecursionStack; StackTop>=RecursionStack; --StackTop ) //-V625
50 {
51 Current = *StackTop;
52
53 Loop:
54 IndexType Count = (IndexType)(Current.Max - Current.Min + 1);
55
56 if ( Current.MaxDepth == 0 )
57 {
58 // We're too deep into quick sort, switch to heap sort
59 HeapSortInternal( Current.Min, Count, Proj, Predicate );
60 continue;
61 }
62
63 if( Count <= 8 )
64 {
65 // Use simple selection sort.
66 while( Current.Max > Current.Min )
67 {
68 IteratorType Max;
69 IteratorType Item;
70 for( Max=Current.Min, Item=Current.Min+1; Item<=Current.Max; ++Item )
71 {
72#if UE_VIRTUAL_PROJECTION_WORKAROUND
73 if( Invoke( Predicate, Invoke( Proj, *Max ), Invoke( Proj, *Item ) ) )
74#else
75 if( Predicate( Proj( *Max ), Proj( *Item ) ) )
76#endif
77 {
78 Max = Item;
79 }
80 }
81 Swap( *Max, *Current.Max-- );
82 }
83 }
84 else
85 {
86 // Grab middle element so sort doesn't exhibit worst-cast behavior with presorted lists.
87 Swap( Current.Min[Count/2], Current.Min[0] );
88
89 // Divide list into two halves, one with items <=Current.Min, the other with items >Current.Max.
90 Inner.Min = Current.Min;
91 Inner.Max = Current.Max+1;
92 for( ; ; )
93 {
94#if UE_VIRTUAL_PROJECTION_WORKAROUND
95 while( ++Inner.Min<=Current.Max && !Invoke( Predicate, Invoke( Proj, *Current.Min ), Invoke( Proj, *Inner.Min ) ) );
96 while( --Inner.Max> Current.Min && !Invoke( Predicate, Invoke( Proj, *Inner.Max ), Invoke( Proj, *Current.Min ) ) );
97#else
98 while( ++Inner.Min<=Current.Max && !Predicate( Proj( *Current.Min ), Proj( *Inner.Min ) ) );
99 while( --Inner.Max> Current.Min && !Predicate( Proj( *Inner.Max ), Proj( *Current.Min ) ) );
100#endif
101 if( Inner.Min>Inner.Max )
102 {
103 break;
104 }
105 Swap( *Inner.Min, *Inner.Max );
106 }
107 Swap( *Current.Min, *Inner.Max );
108
109 --Current.MaxDepth;
110
111 // Save big half and recurse with small half.
112 if( Inner.Max-1-Current.Min >= Current.Max-Inner.Min )
113 {
114 if( Current.Min+1 < Inner.Max )
115 {
116 StackTop->Min = Current.Min;
117 StackTop->Max = Inner.Max - 1;
118 StackTop->MaxDepth = Current.MaxDepth;
119 StackTop++;
120 }
121 if( Current.Max>Inner.Min )
122 {
123 Current.Min = Inner.Min;
124 goto Loop;
125 }
126 }
127 else
128 {
129 if( Current.Max>Inner.Min )
130 {
131 StackTop->Min = Inner .Min;
132 StackTop->Max = Current.Max;
133 StackTop->MaxDepth = Current.MaxDepth;
134 StackTop++;
135 }
136 if( Current.Min+1<Inner.Max )
137 {
138 Current.Max = Inner.Max - 1;
139 goto Loop;
140 }
141 }
142 }
143 }
144 }
