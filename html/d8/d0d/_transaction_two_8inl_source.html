72struct FFrame
73{
74 enum class EType : uint8
75 {
76 Data, // = 0x00,
77 Headers, // = 0x01,
78 Priority, // = 0x02,
79 RstStream, // = 0x03,
80 Settings, // = 0x04,
81 PushPromise, // = 0x05,
82 Ping, // = 0x06,
83 Goaway, // = 0x07,
84 WindowUpdate, // = 0x08,
85 Continuation, // = 0x09,
86 _Num,
87 };
88
89 enum : uint8
90 {
91 Flag_Padding = 0x08,
92 Flag_EndHeaders = 0x04,
93 Flag_EndStream = 0x01,
94 };
95
96 uint8 Size[3];
97 EType Type;
98 uint8 Flags;
99 uint8 Id[4];
100
101 int32 IsPertinent() const { return (Id[3] & 1) != 0; }
102 uint32 GetSize() const { return (Size[0] << 16u) | (Size[1] << 8u) | Size[2]; }
103 uint32 GetId() const { return (Id[0] << 24u) | (Id[1] << 16u) | (Id[2] << 8u) | Id[3]; }
104 uint32 GetIdFast() const { return Id[3]; }
105};
106static_assert(sizeof(FFrame) == 9);

277FOutcome FDriverNg::PumpSend(FTlsPeer& Peer)
278{
279 for (uint32 Sent = 0;;)
280 {
281 if (SendPending.IsEmpty())
282 {
283 const uint8* Data = nullptr;
284 int32 Result = int32(nghttp2_session_mem_send2(Session, &Data));
285 if (Result < 0)
286 {
287 const char* Message = nghttp2_strerror(Result);
288 return FOutcome::Error(Message);
289 }
290
291 if (Result == 0)
292 {
293 return FOutcome::Ok(Sent);
294 }
295
296 SendPending = { Data, uint32(Result) };
297 }
298
299 const char* Data = (char*)(SendPending.GetData());
300 uint32 Size = uint32(SendPending.GetSize());
301 FOutcome Outcome = Peer.Send((char*)Data, Size);
302 if (Outcome.IsError())
303 {
304 return Outcome;
305 }
306
307 if (Outcome.IsOk())
308 {
309 int32 Result = Outcome.GetResult();
310 Sent += Result;
311
312 SendPending = SendPending.Mid(Result);
313 if (SendPending.IsEmpty())
314 {
315 continue;
316 }
317 }
318
319 check(!SendPending.IsEmpty());
320 return FOutcome::Waiting();
321 }
322}

325FOutcome FDriverNg::PumpRecv(FTlsPeer& Peer, FMemoryView Data)
326{
327 int32 Result = int32(nghttp2_session_mem_recv2(Session, (uint8*)Data.GetData(), Data.GetSize()));
328
329 if (Result < 0)
330 {
331 const char* Message = nghttp2_strerror(Result);
332 return FOutcome::Error(Message);
333 }
334
335 if (Result != int32(Data.GetSize()))
336 {
337 return FOutcome::Error("Not enough data consumed");
338 }
339
340 if (nghttp2_session_want_write(Session))
341 {
342 // We'll not care about the outcome here. If an error occurred it will
343 // surface later and data-waits will solved themselves later.
344 PumpSend(Peer);
345 }
346
347 return FOutcome::Ok();
348}

390FOutcome FDriverNg::TickInner(FTlsPeer& Peer)
391{
392 FMutableMemoryView Dest(RecvBuffer, sizeof(RecvBuffer) + sizeof(RecvFrame));
393 if (RecvRemaining < 0)
394 {
395 Dest = Dest.Right(0 - RecvRemaining);
396 }
397 else if (RecvRemaining <= int32(sizeof(RecvBuffer)))
398 {
399 Dest = Dest.Right(RecvRemaining + sizeof(RecvFrame));
400 }
401 else
402 {
403 Dest = Dest.Left(sizeof(RecvBuffer));
404 }
405
406 FOutcome Outcome = Peer.Recv((char*)(Dest.GetData()), int32(Dest.GetSize()));
407 if (!Outcome.IsOk())
408 {
409 return Outcome;
410 }
411
412 int32 Result = Outcome.GetResult();
413 Dest = Dest.Left(Result);
414
415 FMutableMemoryView FrameView;
416 if (const uint8* End = (uint8*)(Dest.GetDataEnd()); End > RecvFrameData)
417 {
418 FrameView = FMutableMemoryView(RecvFrameData, End - RecvFrameData);
419 Dest = Dest.LeftChop(FrameView.GetSize());
420 }
421
422 if (!Dest.IsEmpty() && (Outcome = PumpRecv(Peer, Dest)).IsError())
423 {
424 return Outcome;
425 }
426
427 if (FrameView.IsEmpty())
428 {
429 if (RecvRemaining < int32(Dest.GetSize()))
430 {
431 return FOutcome::Error("RecvRemaining value too small");
432 }
433
434 if ((RecvRemaining -= int32(Dest.GetSize())) == 0)
435 {
436 RecvRemaining = 0 - int32(sizeof(RecvFrame));
437 }
438 return FOutcome::Waiting(Dest.GetDataEnd() == RecvFrameData);
439 }
440
441 if (Result = int32(FrameView.GetSize() - sizeof(RecvFrame)); Result < 0)
442 {
443 RecvRemaining = Result;
444 return FOutcome::Waiting();
445 }
446
447 if ((Outcome = PumpRecv(Peer, FrameView)).IsError())
448 {
449 return Outcome;
450 }
451
452 RecvRemaining = RecvFrame.GetSize();
453 if (RecvFrame.IsPertinent())
454 {
455 uint32 Id = RecvFrame.GetId();
456 return FOutcome::Ok(Id);
457 }
458
459 if (RecvRemaining == 0)
460 {
461 RecvRemaining = 0 - int32(sizeof(RecvFrame));
462 }
463
464 return FOutcome::Waiting(1);
465}

481FOutcome FDriverNg::RecvResponse(FTlsPeer& Peer, TArray<FHeaderNg>* Sink)
482{
483 if (RecvFrame.Type == FFrame::EType::Data)
484 {
485 // Sometimes streams of requests with no content can be closed with an
486 // empty data frame instead of via header frame's flags. By deferring
487 // RecvResponse() completion we avoid orphaning zero-sized data frames
488 if ((RecvFrame.Flags & FFrame::Flag_EndStream) && RecvFrame.GetSize() == 0)
489 {
490 RecvFrame = {};
491 }
492 return FOutcome::Ok();
493 }
494
495 ResponseHeaders = Sink;
496 ON_SCOPE_EXIT { ResponseHeaders = nullptr; };
497
498 while (true)
499 {
500 if (RecvFrame.Type == FFrame::EType::RstStream)
501 {
502 return FOutcome::Error("Received stream reset");
503 }
504
505 // Assume if we're filling we are expecting something
506 check(RecvRemaining > 0);
507 if (RecvFrame.Type != FFrame::EType::Headers)
508 {
509 return FOutcome::Error("Not a headers frame", int32(RecvFrame.Type));
510 }
511
512 int32 RecvSize = FMath::Min<int32>(sizeof(RecvBuffer), RecvRemaining);
513
514 FMutableMemoryView Dest(RecvBuffer, RecvSize);
515 FOutcome Outcome = Peer.Recv((char*)(Dest.GetData()), RecvSize);
516 if (!Outcome.IsOk())
517 {
518 return Outcome;
519 }
520 int32 Result = Outcome.GetResult();
521
522 Dest = Dest.Left(Outcome.GetResult());
523 if (!Dest.IsEmpty() && (Outcome = PumpRecv(Peer, Dest)).IsError())
524 {
525 return Outcome;
526 }
527
528 RecvRemaining -= Result;
529 if (RecvRemaining > 0)
530 {
531 continue;
532 }
533
534 check(RecvRemaining == 0);
535 RecvRemaining -= sizeof(RecvFrame);
536
537 int32 EndOfHeaders = (RecvFrame.Flags & FFrame::Flag_EndHeaders);
538
539 if (RecvFrame.Flags & FFrame::Flag_EndStream)
540 {
541 check(EndOfHeaders);
542 return FOutcome::Ok();
543 }
544
545 uint32 ThisId = RecvFrame.GetId();
546 if (Outcome = Tick(Peer); Outcome.IsError())
547 {
548 return Outcome;
549 }
550
551 if (Outcome.GetResult() != ThisId)
552 {
553 return FOutcome::WaitStream();
554 }
555
556 if (EndOfHeaders)
557 {
558 // We shall end headers once the first data frame arrives, which may
559 // have already happened in the Tick() call.
560 return RecvResponse(Peer, ResponseHeaders);
561 }
562 }
563}

566FOutcome FDriverNg::RecvBody(FTlsPeer& Peer, FMutableMemoryView Dest)
567{
568 for (uint32 RecvSize = 0;;)
569 {
570 if (RecvFrame.Type == FFrame::EType::RstStream)
571 {
572 return FOutcome::Error("Received stream reset");
573 }
574
575 check(RecvRemaining >= 0);
576 if (RecvFrame.Type != FFrame::EType::Data)
577 {
578 return FOutcome::Error("Not a data frame", int32(RecvFrame.Type));
579 }
580
581 int32 EndOfStream = (RecvFrame.Flags & FFrame::Flag_EndStream);
582
583 FMutableMemoryView View = Dest.Left(RecvRemaining);
584 if (View.IsEmpty())
585 {
586 if (RecvRemaining <= 0 && EndOfStream)
587 {
588 RecvFrame = {};
589 return FOutcome::Ok(RecvSize);
590 }
591
592 return FOutcome::WaitBuffer(RecvSize);
593 }
594
595 FOutcome Outcome = Peer.Recv((char*)View.GetData(), int32(View.GetSize()));
596 if (!Outcome.IsOk())
597 {
598 if (Outcome.IsError())
599 {
600 return Outcome;
601 }
602 return FOutcome::Waiting(RecvSize);
603 }
604
605 int32 Result = Outcome.GetResult();
606 RecvRemaining -= Result;
607 RecvSize += Result;
608 Dest = Dest.Mid(Result);
609
610 if ((Outcome = PumpRecv(Peer, View.Left(Result))).IsError())
611 {
612 return Outcome;
613 }
614
615 // more of this frame to go?
616 if (RecvRemaining > 0)
617 {
618 continue;
619 }
620
621 RecvRemaining = 0 - int32(sizeof(RecvFrame)) - RecvRemaining;
622
623 // End of the road?
624 if (EndOfStream)
625 {
626 RecvFrame = {};
627 return FOutcome::Ok(RecvSize);
628 }
629
630 uint32 ThisId = RecvFrame.GetId();
631 Outcome = Tick(Peer);
632 if (Outcome.IsError()) return Outcome;
633 if (ThisId != Outcome.GetResult()) return FOutcome::WaitBuffer(RecvSize);
634 }

837FOutcome FStatusHeaders::TryRecvResponse(FTlsPeer& Peer)
838{
839 FOutcome Outcome = GetDriver().RecvResponse(Peer, &ResponseHeaders);
840 if (!Outcome.IsOk())
841 {
842 return Outcome;
843 }
844
845 enum : uint32 {
846 Flag_Code = 1 << 0,
847 Flag_Length = 1 << 1,
848 Flag_All = Flag_Code | Flag_Length,
849 };
850 uint32 Flags = 0;
851 for (const FHeaderNg& Header : ResponseHeaders)
852 {
853 if (Flags == Flag_All)
854 {
855 break;
856 }
857
858 FAnsiStringView Key = Header.Key.Get();
859 FAnsiStringView Value = Header.Value.Get();
860
861 if (!(Flags & Flag_Code) && (Key == ":status"))
862 {
863 StatusCode = uint32(CrudeToInt(Value));
864 Flags |= Flag_Code;
865 }
866 else if (!(Flags & Flag_Length) && (Key == "content-length"))
867 {
868 ContentLength = CrudeToInt(Value);
869 Flags |= Flag_Length;
870 }
871 }
872
873 if ((Flags & Flag_Code) == 0 || !IsStatusCodeOk(StatusCode))
874 {
875 return FOutcome::Error("Invalid status code", StatusCode);
876 }
877
878 if (IsContentless(StatusCode))
879 {
880 ContentLength = 0;
881 }
882
883 return FOutcome::Ok();
884}
