24class TTraceSetFilter : public IFilter<const FTraceViewModel&>, public TSharedFromThis< TTraceSetFilter<TSetType> >
25{
26public:
27 TTraceSetFilter();

37 virtual bool PassesFilter(const FTraceViewModel& InTrace) const override
38 {

48protected:
49 virtual void AddDefaultValues(TArray<TSetType>& InOutDefaultValues) const { }

53protected:
55 FChangedEvent ChangedEvent;

224public:
225 enum class ESizeCategory : uint8
226 {
227 Empty, // 0 bytes
228 Small, // < 1 MiB
229 Medium, // < 1 GiB
230 Large, // >= 1 GiB
231
232
233 InvalidOrMax
234 };
235
236public:
237 FTraceFilterBySize();
238
239protected:
240 virtual void AddDefaultValues(TArray<uint8>& InOutDefaultValues) const override;
241
242 virtual uint8 GetFilterValueForTrace(const FTraceViewModel& InTrace) const override
243 {
244 const uint64 TraceSize = InTrace.GetSize();
245 if (TraceSize == 0)
246 {
247 return (uint8)ESizeCategory::Empty;
248 }
249 else if (TraceSize < 1024ull * 1024ull)
250 {
251 return (uint8)ESizeCategory::Small;
252 }
253 else if (TraceSize < 1024ull * 1024ull * 1024ull)
254 {
255 return (uint8)ESizeCategory::Medium;
256 }
257 else
258 {
259 return (uint8)ESizeCategory::Large;
260 }
261 }
