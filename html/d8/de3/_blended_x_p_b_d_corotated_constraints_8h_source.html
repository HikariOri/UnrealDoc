101 virtual TVec4<TVector<FSolverReal, 3>> GetDeterminantDelta(const FSolverParticlesRange& Particles, const FSolverReal Dt, const int32 ElementIndex, const FSolverReal Tol = (FSolverReal)1e-3) const override
102 {
103 TRACE_CPUPROFILER_EVENT_SCOPE(STAT_ChaosBlendedXPBDCorotatedApplyDet);
104
105 const PMatrix<FSolverReal, 3, 3> Fe = Base::F(ElementIndex, Particles);
106 PMatrix<FSolverReal, 3, 3> DmInvT = Base::ElementDmInv(ElementIndex).GetTransposed();
107
108 FSolverReal J = Fe.Determinant();
109 if (J - (FSolverReal)1. < Tol)
110 {
111 return TVec4<TVector<FSolverReal, 3>>(TVector<FSolverReal, 3>((FSolverReal)0.));
112 }
113
114 TVec4<TVector<FSolverReal, 3>> dC2 = Base::GetDeterminantGradient(Fe, DmInvT);
115
116 FSolverReal AlphaTilde = (FSolverReal)2. / (Dt * Dt * Base::Lambda * Measure[ElementIndex]);
117
118 FSolverReal DLambda = (1 - J) - AlphaTilde * LambdaArray[2 * ElementIndex + 1];
119
120 FSolverReal Denom = AlphaTilde;
121 for (int i = 0; i < 4; i++)
122 {
123 for (int j = 0; j < 3; j++)
124 {
125 Denom += dC2[i][j] * Particles.InvM(MeshConstraints[ElementIndex][i]) * dC2[i][j];
126 }
127 }
128 DLambda /= Denom;
129 LambdaArray[2 * ElementIndex + 1] += DLambda;
130 TVec4<TVector<FSolverReal, 3>> Delta2(TVector<FSolverReal, 3>((FSolverReal)0.));
131 for (int i = 0; i < 4; i++)
132 {
133 for (int j = 0; j < 3; j++)
134 {
135 Delta2[i][j] = Particles.InvM(MeshConstraints[ElementIndex][i]) * dC2[i][j] * DLambda;
136 }
137 }
138 TVec4<TVector<FSolverReal, 3>> Delta1(TVector<FSolverReal, 3>((FSolverReal)0.));
139 for (int i = 0; i < 4; i++)
140 {
141 for (int j = 0; j < 3; j++)
142 {
143 Delta1[i][j] = Particles.InvM(MeshConstraints[ElementIndex][i])
144 * dC2[i][j] * LambdaArray[2 * ElementIndex + 1] - C2Contribution[ElementIndex][i][j];
145 }
146 }
147 TVec4<TVector<FSolverReal, 3>> Delta(TVector<FSolverReal, 3>((FSolverReal)0.));
148 for (int i = 0; i < 4; i++)
149 {
150 Delta[i] = Zeta * Delta1[i] + ((FSolverReal)1. - Zeta) * Delta2[i];
151 }
152
153 for (int i = 0; i < 4; i++)
154 {
155 C2Contribution[ElementIndex][i] += Delta[i];
156 }
157
158 return Delta;
159 }

161 virtual TVec4<TVector<FSolverReal, 3>> GetPolarDelta(const FSolverParticlesRange& Particles, const FSolverReal Dt, const int32 ElementIndex, const FSolverReal Tol = (FSolverReal)1e-3) const override
162 {
163 TRACE_CPUPROFILER_EVENT_SCOPE(STAT_ChaosBlendedXPBDCorotatedApplyPolar);
164 SCOPE_CYCLE_COUNTER(STAT_ChaosXPBDCorotatedPolar);
165 const PMatrix<FSolverReal, 3, 3> Fe = Base::F(ElementIndex, Particles);
166
167 PMatrix<FSolverReal, 3, 3> Re((FSolverReal)0.), Se((FSolverReal)0.);
168
169 Chaos::PolarDecomposition(Fe, Re, Se);
170
171 FSolverReal C1 = (FSolverReal)0.;
172 for (int i = 0; i < 3; i++)
173 {
174 for (int j = 0; j < 3; j++)
175 {
176 C1 += FMath::Square((Fe - Re).GetAt(i, j));
177 }
178 }
179 C1 = FMath::Sqrt(C1);
180
181 if (C1 < Tol)
182 {
183 return TVec4<TVector<FSolverReal, 3>>(TVector<FSolverReal, 3>((FSolverReal)0.));
184 }
185
186 PMatrix<FSolverReal, 3, 3> DmInvT = Base::ElementDmInv(ElementIndex).GetTransposed();
187
188 TVec4<TVector<FSolverReal, 3>> dC1 = Base::GetPolarGradient(Fe, Re, DmInvT, C1);
189
190 FSolverReal AlphaTilde = (FSolverReal)1. / (Dt * Dt * Base::Mu * Measure[ElementIndex]);
191
192 FSolverReal DLambda = -C1 - AlphaTilde * LambdaArray[2 * ElementIndex + 0];
193
194 FSolverReal Denom = AlphaTilde;
195 for (int i = 0; i < 4; i++)
196 {
197 for (int j = 0; j < 3; j++)
198 {
199 Denom += dC1[i][j] * Particles.InvM(MeshConstraints[ElementIndex][i]) * dC1[i][j];
200 }
201 }
202 DLambda /= Denom;
203 LambdaArray[2 * ElementIndex + 0] += DLambda;
204
205 TVec4<TVector<FSolverReal, 3>> Delta2(TVector<FSolverReal, 3>((FSolverReal)0.));
206 for (int i = 0; i < 4; i++)
207 {
208 for (int j = 0; j < 3; j++)
209 {
210 Delta2[i][j] = Particles.InvM(MeshConstraints[ElementIndex][i]) * dC1[i][j] * DLambda;
211 }
212 }
213 TVec4<TVector<FSolverReal, 3>> Delta1(TVector<FSolverReal, 3>((FSolverReal)0.));
214 for (int i = 0; i < 4; i++)
215 {
216 for (int j = 0; j < 3; j++)
217 {
218 Delta1[i][j] = Particles.InvM(MeshConstraints[ElementIndex][i])
219 * dC1[i][j] * LambdaArray[2 * ElementIndex + 0] - C1Contribution[ElementIndex][i][j];
220 }
221 }
222 TVec4<TVector<FSolverReal, 3>> Delta(TVector<FSolverReal, 3>((FSolverReal)0.));
223 for (int i = 0; i < 4; i++)
224 {
225 Delta[i] = Zeta * Delta1[i] + ((FSolverReal)1. - Zeta) * Delta2[i];
226 }
227
228 for (int i = 0; i < 4; i++)
229 {
230 C1Contribution[ElementIndex][i] += Delta[i];
231 }
232
233 return Delta;
234
235 }
