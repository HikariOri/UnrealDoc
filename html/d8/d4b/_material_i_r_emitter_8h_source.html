59// A lightweight wrapper around FValue* that also optionally tracks the source expression input.
60// Used to carry both value and its origin in the expression graph.
61// A null reference evaluates has the FValue* set to nullptr and evaluates false. It is
62// returned by the FEmitter::TryInput() value to indicate a missing value. An invalid reference
63// is either null or has a poison value. It is usually generated as the result of an invalid
64// operation.
65// Important: The Emitter APIs can safely take both null and poison values. Operations that
66// take a single argument simply return the argument if it is invalid (null or poison),
67// whereas operations that take multiple arguments return poison if any is invalid.
68//
69// When your intention is to test for whether a value is not-null (available), leverage the
70// operator bool()
71//
72// if (Value) { /* handle case when value is available */ }
73//
74// Otherwise, if you would like the value to be available AND valid (not-poison) you should
75// test for IsValid()
76//
77// if (Value.IsValid()) {
78// /* handle case when the operations that produced the value were all succesful */
79// }
80//
81// Note: In user Expression::Build() implementations, you should avoid testing values validity
82// directly and instead leverage the UE_MIR_CHECKPOINT() macro to verify that all emitter
83// performed so far were all valid, and thus the values you have computed so far are what
84// you expect.
85struct FValueRef
86{
87 // Default constructor. Creates an invalid reference.
88 FValueRef() {}
89
90 // Constructs from a value pointer.
91 FValueRef(FValue* Value) : Value{ Value } {}
92
93 // Constructs from a value pointer and its corresponding input source.
94 FValueRef(FValue* Value, const FExpressionInput* Input) : Value{ Value }, Input{ Input } {}
95
96 // Returns whether this references a non-null value.
97 operator bool() const { return Value != nullptr; }
98
99 // Returns the referenced value, which must be non-null.
100 operator FValue*() const { return Value; }
101
102 // Returns the referenced value, which must be non-null.
103 FValue* operator->() const { return Value; }
104
105 // Checks if the reference holds a valid non-null value.
106 bool IsValid() const;
107
108 // Checks if the reference holds a poison value.
109 bool IsPoison() const;
110
111 // Creates a new reference with the same input but a different value.
112 FValueRef To(FValue* Value) const;
113
114 // Creates a poison version of the current reference (same input, but poison value).
115 FValueRef ToPoison() const;
116
117 // Equality/inequality operators
118 bool operator==(const FValueRef Other) const { return Value == Other.Value; }
119 bool operator!=(const FValueRef Other) const { return Value != Other.Value; }
120
121 // The referenced value.
122 FValue* Value{};
123
124 // The expression input being read.
125 const FExpressionInput* Input{};
126};

287// The FEmitter is a helper object responsible for emitting MIR values and instructions.
288// It is primarily used within UMaterialExpression::Build() implementations to lower the
289// high-level semantics of a material expression into corresponding lower-level MIR nodes.
290//
291// During the emission process, the emitter automatically performs various
292// simplifications and constant folding optimizations where possible.
293//
294// An instance of the emitter is created and managed by the FMaterialIRModuleBuilder.
295// This instance is then passed to the UMaterialExpression::Build() functions
296// so that they can emit their corresponding IR values.
297//
298// IMPORTANT: Users should *not* assume the exact MIR::FType of the FValue* returned
299// by EmitXXX() functions. The emitter is free to optimize operations, which may
300// result in a returned value having a different type than naively expected (e.g.,
301// folding an operation into a constant).
302//
303// The emitter handles invalid inputs gracefully. If an operation is attempted
304// with invalid arguments (e.g., acting on arguments of unexpected types), the emitter
305// automatically reports an error and returns the MIR::FPoison value. Furthermore, all
306// EmitXXX() functions are robust to receiving MIR::FPoison as input; they will simply
307// propagate the poison value without attempting the operation.
