238// Snapshot of current MeshInfo state. Created at start of a brush stroke to store the existing instance info.
239class FMeshInfoSnapshot
240{
241 FFoliageInstanceHash Hash;
242 TArray<FVector> Locations;
243public:
244 FMeshInfoSnapshot(FFoliageInfo* MeshInfo)
245 : Hash(*MeshInfo->InstanceHash)
246 {
247 int32 NumInstances = MeshInfo->Instances.Num();
248 Locations.Reserve(NumInstances);
249 Locations.AddUninitialized(NumInstances);
250 for (int32 Idx = 0; Idx < NumInstances; Idx++)
251 {
252 Locations[Idx] = MeshInfo->Instances[Idx].Location;
253 }
254 }
255
256 int32 CountInstancesInsideSphere(const FSphere& Sphere) const
257 {
258 int32 Count = 0;
259
260 auto TempInstances = Hash.GetInstancesOverlappingBox(FBox::BuildAABB(Sphere.Center, FVector(Sphere.W)));
261 for (int32 Idx : TempInstances)
262 {
263 if (FSphere(Locations[Idx], 0.f).IsInside(Sphere))
264 {
265 Count++;
266 }
267 }
268
269 return Count;
270 }
