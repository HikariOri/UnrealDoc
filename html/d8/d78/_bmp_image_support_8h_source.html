86public:
87 EBitmapHeaderVersion GetHeaderVersion() const
88 {
89 // Since there is no field indicating the header version of the bitmap in the FileHeader,
90 // the only way to know the format version is to check the header size.
91 //
92 // note that Adobe (incorrectly) writes biSize as 40 and then BMFH bfOffBits will be 52 or 56
93 // so this code does not detect those variant headers
94 switch (biSize)
95 {
96 case 40:
97 return EBitmapHeaderVersion::BHV_BITMAPINFOHEADER;
98 case 52:
99 // + RGB 32 bit masks
100 return EBitmapHeaderVersion::BHV_BITMAPV2INFOHEADER;
101 case 56:
102 // + RGBA 32 bit masks
103 return EBitmapHeaderVersion::BHV_BITMAPV3INFOHEADER;
104 case 108:
105 return EBitmapHeaderVersion::BHV_BITMAPV4HEADER;
106 case 124:
107 return EBitmapHeaderVersion::BHV_BITMAPV5HEADER;
108 default:
109 return EBitmapHeaderVersion::BHV_INVALID;
110 }
111 }
112};
