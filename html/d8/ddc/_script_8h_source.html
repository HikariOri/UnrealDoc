132enum EFunctionFlags : uint32
133{
134 // Function flags.
135 FUNC_None = 0x00000000,
136
137 FUNC_Final = 0x00000001, // Function is final (prebindable, non-overridable function).
138 FUNC_RequiredAPI = 0x00000002, // Indicates this function is DLL exported/imported.
139 FUNC_BlueprintAuthorityOnly= 0x00000004, // Function will only run if the object has network authority
140 FUNC_BlueprintCosmetic = 0x00000008, // Function is cosmetic in nature and should not be invoked on dedicated servers
141 // FUNC_ = 0x00000010, // unused.
142 // FUNC_ = 0x00000020, // unused.
143 FUNC_Net = 0x00000040, // Function is network-replicated.
144 FUNC_NetReliable = 0x00000080, // Function should be sent reliably on the network.
145 FUNC_NetRequest = 0x00000100, // Function is sent to a net service
146 FUNC_Exec = 0x00000200, // Executable from command line.
147 FUNC_Native = 0x00000400, // Native function.
148 FUNC_Event = 0x00000800, // Event function.
149 FUNC_NetResponse = 0x00001000, // Function response from a net service
150 FUNC_Static = 0x00002000, // Static function.
151 FUNC_NetMulticast = 0x00004000, // Function is networked multicast Server -> All Clients
152 FUNC_UbergraphFunction = 0x00008000, // Function is used as the merge 'ubergraph' for a blueprint, only assigned when using the persistent 'ubergraph' frame
153 FUNC_MulticastDelegate = 0x00010000, // Function is a multi-cast delegate signature (also requires FUNC_Delegate to be set!)
154 FUNC_Public = 0x00020000, // Function is accessible in all classes (if overridden, parameters must remain unchanged).
155 FUNC_Private = 0x00040000, // Function is accessible only in the class it is defined in (cannot be overridden, but function name may be reused in subclasses. IOW: if overridden, parameters don't need to match, and Super.Func() cannot be accessed since it's private.)
156 FUNC_Protected = 0x00080000, // Function is accessible only in the class it is defined in and subclasses (if overridden, parameters much remain unchanged).
157 FUNC_Delegate = 0x00100000, // Function is delegate signature (either single-cast or multi-cast, depending on whether FUNC_MulticastDelegate is set.)
158 FUNC_NetServer = 0x00200000, // Function is executed on servers (set by replication code if passes check)
159 FUNC_HasOutParms = 0x00400000, // function has out (pass by reference) parameters
160 FUNC_HasDefaults = 0x00800000, // function has structs that contain defaults
161 FUNC_NetClient = 0x01000000, // function is executed on clients
162 FUNC_DLLImport = 0x02000000, // function is imported from a DLL
163 FUNC_BlueprintCallable = 0x04000000, // function can be called from blueprint code
164 FUNC_BlueprintEvent = 0x08000000, // function can be overridden/implemented from a blueprint
165 FUNC_BlueprintPure = 0x10000000, // function can be called from blueprint code, and is also pure (produces no side effects). If you set this, you should set FUNC_BlueprintCallable as well.
166 FUNC_EditorOnly = 0x20000000, // function can only be called from an editor scrippt.
167 FUNC_Const = 0x40000000, // function can be called from blueprint code, and only reads state (never writes state)
168 FUNC_NetValidate = 0x80000000, // function must supply a _Validate implementation
169
170 // !!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!
171 // DO NOT ADD composite flags to EFunctionFlags. Debugger visualizations have trouble displaying composite flag values as text.
172 // !!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!
173};
