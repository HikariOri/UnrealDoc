213 FMovieSceneEntityID CreateEntity(FEntityManager* EntityManager, FComponentMask NewType = FComponentMask())
214 {
215 bool bLocalAddMutualComponents = false;
216 GenerateType(EntityManager, NewType, bLocalAddMutualComponents);
217
218 FMutualComponentInitializers MutualInitializers;
219 FEntityAllocationWriteContext WriteContext(*EntityManager);
220 EMutuallyInclusiveComponentType MutualTypes = bLocalAddMutualComponents ? EMutuallyInclusiveComponentType::All : EMutuallyInclusiveComponentType::Mandatory;
221
222 EntityManager->GetComponents()->Factories.ComputeMutuallyInclusiveComponents(MutualTypes, NewType, MutualInitializers);
223
224 FEntityInfo Entry = EntityManager->AllocateEntity(NewType);
225 Initialize(EntityManager, Entry);
226
227 // Run mutual initializers after the builder has actually constructed the entity
228 // otherwise mutual components would be reading garbage
229 MutualInitializers.Execute(Entry.Data.AsRange(), WriteContext);
230
231 return Entry.EntityID;
232 }

287 {

545inline FComponentMask FTypelessMutation::MutateType(const FComponentMask& Current) const
546{
547 FComponentMask NewMask;
548 if (!bRemoveAll)
549 {
550 NewMask = Current;
551 }
552
553 if (RemoveMask.Num())
554 {
555 FComponentMask MaskOut = RemoveMask;
556 MaskOut.BitwiseNOT();
557
558 NewMask.CombineWithBitwiseAND(MaskOut, EBitwiseOperatorFlags::MaintainSize | EBitwiseOperatorFlags::OneFillMissingBits);
559 }
560
561 if (AddMask.Num())
562 {
563 NewMask.CombineWithBitwiseOR(AddMask, EBitwiseOperatorFlags::MaxSize);
564 }
565 return NewMask;
566}
