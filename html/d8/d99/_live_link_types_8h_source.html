353 template<typename DataType>
354 void InitializeWith(const DataType* InData)
355 {
356 static_assert(TIsDerivedFrom<typename TRemoveReference<DataType>::Type, BaseType>::IsDerived, "'DataType' template parameter must be derived from 'BaseType' to InitializeWith.");
357 InitializeWith(DataType::StaticStruct(), InData);
358 }
359
364 void InitializeWith(const UScriptStruct* InOtherStruct, const BaseType* InData)
365 {
366 check(InOtherStruct);
367
368 //To be used after creating a base struct for a specific type. We should only have a type set.
369 WrappedStruct.Initialize(InOtherStruct);
370 if (InData)
371 {
372 GetStruct()->CopyScriptStruct(GetBaseData(), InData);
373 }
374 }
375
380 void InitializeWith(const FLiveLinkBaseDataStruct& InOther)
381 {
382 //To be used after creating a base struct for a specific type. We should only have a type set.
383 WrappedStruct.Initialize(InOther.GetStruct());
384 if (InOther.GetBaseData())
385 {
386 GetStruct()->CopyScriptStruct(GetBaseData(), InOther.GetBaseData());
387 }
388 }
389
390 BaseType* CloneData() const
391 {
392 BaseType* DataStruct = nullptr;
393 if (const UScriptStruct* ScriptStructPtr = ::Cast<UScriptStruct>(WrappedStruct.GetStruct()))
394 {
395 DataStruct = (BaseType*)FMemory::Malloc(ScriptStructPtr->GetStructureSize() ? ScriptStructPtr->GetStructureSize() : 1);
396 ScriptStructPtr->InitializeStruct(DataStruct);
397 ScriptStructPtr->CopyScriptStruct(DataStruct, GetBaseData());
398 }
399 return DataStruct;
400 }

402 template<typename Type>
403 Type* Cast()
404 {
405 return FCastImpl<Type>::Cast(GetStruct(), GetBaseData());
406 }
407

412 }
413

415 {
416 WrappedStruct = MoveTemp(InOther.WrappedStruct);
417 return *this;
418 }
419
420 bool operator==(const FLiveLinkBaseDataStruct& Other) const
421 {
422 if (Other.GetStruct() == GetStruct())
423 {
424 if (GetBaseData())
425 {
426 return GetStruct()->CompareScriptStruct(GetBaseData(), Other.GetBaseData(), PPF_None);
427 }
428 return Other.GetBaseData() == nullptr; // same struct and both uninitialized
429 }
430 return false;
431 }
432
433 bool operator!=(const FLiveLinkBaseDataStruct& Other) const
434 {
435 return !(*this == Other);
436 }

438 friend FArchive& operator<<(FArchive& Ar, FLiveLinkBaseDataStruct& InStruct)
439 {
440 if (Ar.IsLoading())
441 {
442 FString StructPath;
443 Ar << StructPath;
444 if (!StructPath.IsEmpty())
445 {
446 UScriptStruct* ScriptStructPtr = FindObject<UScriptStruct>(nullptr, *StructPath, EFindObjectFlags::None);
447 if (ScriptStructPtr == nullptr || !ScriptStructPtr->IsChildOf(TBaseStructure<BaseType>::Get()))
448 {
449 Ar.SetError();
450 return Ar;
451 }
452 InStruct.WrappedStruct.Initialize(ScriptStructPtr);
453 ScriptStructPtr->SerializeItem(Ar, InStruct.GetBaseData(), nullptr);
454 }
455 }
456 // Saving
457 else
458 {
459 FString StructPath;
460 if (UScriptStruct* ScriptStructPtr = const_cast<UScriptStruct*>(::Cast<UScriptStruct>(InStruct.GetStruct())))
461 {
462 StructPath = ScriptStructPtr->GetPathName();
463 Ar << StructPath;
464 ScriptStructPtr->SerializeItem(Ar, InStruct.GetBaseData(), nullptr);
465 }
466 else
467 {
468 Ar << StructPath;
469 }
470 }
471
472 return Ar;
473 }

476 template<typename Type>
477 struct FCastImpl
478 {
479 static Type* Cast(const UScriptStruct* ScriptStruct, BaseType* BaseData)
480 {
481 if constexpr (std::is_same_v<Type, BaseType>)
482 {
483 return StaticCast<Type*>(BaseData);
484 }
485 else
486 {
487 static_assert(TIsDerivedFrom<typename TRemoveReference<Type>::Type, BaseType>::IsDerived, "'Type' template parameter must be derived from 'BaseType' to Cast.");
488 if (ScriptStruct && ScriptStruct->IsChildOf(Type::StaticStruct()))
489 {
490 return StaticCast<Type*>(BaseData);
491 }
492 return nullptr;
493 }
494 }
495 static const Type* ConstCast(const UScriptStruct* ScriptStruct, const BaseType* BaseData)
496 {
497 if constexpr (std::is_same_v<Type, BaseType>)
498 {
499 return StaticCast<const Type*>(BaseData);
500 }
501 else
502 {
503 return const_cast<const Type*>(Cast(ScriptStruct, const_cast<BaseType*>(BaseData)));
504 }
505 }
506 };

601 // Implicit conversion to FTimecode

603 {
604 int32 TotalFrameNumber = (int32)FMath::RoundToZero(Seconds * (FrameRate.Numerator / (double)FrameRate.Denominator)) + Frames;
605 FFrameTime FrameTime = FFrameTime(TotalFrameNumber);
606 return FQualifiedFrameTime(FrameTime, FrameRate);

609 FLiveLinkTimeCode& operator=(const FQualifiedFrameTime& InFrameTime)
610 {
611 const int32 NumberOfFramesInSecond = FMath::CeilToInt32(InFrameTime.Rate.AsDecimal());
612 const int32 NumberOfFrames = (int32)(FMath::RoundToZero(InFrameTime.Time.AsDecimal()));
613
614 Seconds = (int32)FMath::RoundToZero(NumberOfFrames / (double)NumberOfFramesInSecond);
615 Frames = NumberOfFrames % NumberOfFramesInSecond;
616 FrameRate = FLiveLinkFrameRate(InFrameTime.Rate.Numerator, InFrameTime.Rate.Denominator);
617
618 return *this;
