381 virtual FReply OnKeyDown( const FGeometry& MyGeometry, const FKeyEvent& InKeyEvent ) override
382 {
383 if (this->OnKeyDownHandler.IsBound())
384 {
385 FReply Reply = this->OnKeyDownHandler.Execute(MyGeometry, InKeyEvent);
386 if (Reply.IsEventHandled())
387 {
388 return Reply;
389 }
390 }
391
392 // Check for selection/expansion toggling keys (Left, Right)
393 // SelectorItem represents the keyboard selection. If it isn't valid then we don't know what to expand.
394 // Don't respond to key-presses containing "Alt" as a modifier
395 if ( TListTypeTraits<ItemType>::IsPtrValid(this->SelectorItem) && !InKeyEvent.IsAltDown() )
396 {
397 if ( InKeyEvent.GetKey() == EKeys::Left )
398 {
399 if( TListTypeTraits<ItemType>::IsPtrValid(this->SelectorItem) )
400 {
401 ItemType RangeSelectionEndItem = TListTypeTraits<ItemType>::NullableItemTypeConvertToItemType( this->SelectorItem );
402 int32 SelectionIndex = this->LinearizedItems.Find( RangeSelectionEndItem );
403
404 const bool bDoesItemHaveChildren = Private_DoesItemHaveChildren(SelectionIndex);
405 if( bDoesItemHaveChildren && InKeyEvent.GetModifierKeys().IsShiftDown() )
406 {
407 // Collapse the selected item and all of it's children
408 Private_OnExpanderArrowShiftClicked(RangeSelectionEndItem, false);
409 }
410 else if ( bDoesItemHaveChildren && Private_IsItemExpanded( RangeSelectionEndItem ) )
411 {
412 // Collapse the selected item
413 Private_SetItemExpansion(RangeSelectionEndItem, false);
414 }
415 else
416 {
417 // Select the parent, who should be a previous item in the list whose nesting level is less than the selected one
418 int32 SelectedNestingDepth = Private_GetNestingDepth(SelectionIndex);
419 for (SelectionIndex--; SelectionIndex >= 0; --SelectionIndex)
420 {
421 if ( Private_GetNestingDepth(SelectionIndex) < SelectedNestingDepth )
422 {
423 // Found the parent
424 this->NavigationSelect(this->LinearizedItems[SelectionIndex], InKeyEvent);
425 break;
426 }
427 }
428 }
429 }
430
431 return FReply::Handled();
432 }
433 else if ( InKeyEvent.GetKey() == EKeys::Right )
434 {
435 if( TListTypeTraits<ItemType>::IsPtrValid(this->SelectorItem) )
436 {
437 ItemType RangeSelectionEndItem = TListTypeTraits<ItemType>::NullableItemTypeConvertToItemType( this->SelectorItem );
438 int32 SelectionIndex = this->LinearizedItems.Find( RangeSelectionEndItem );
439
440 // Right only applies to items with children
441 if ( Private_DoesItemHaveChildren(SelectionIndex) )
442 {
443 if( InKeyEvent.GetModifierKeys().IsShiftDown() )
444 {
445 // Expand the selected item and all of it's children
446 Private_OnExpanderArrowShiftClicked(RangeSelectionEndItem, true);
447 }
448 else if ( !Private_IsItemExpanded(RangeSelectionEndItem) )
449 {
450 // Expand the selected item
451 Private_SetItemExpansion(RangeSelectionEndItem, true);
452 }
453 else
454 {
455 // Select the first child, who should be the next item in the list
456 // Make sure we aren't the last item on the list
457 if ( SelectionIndex < this->LinearizedItems.Num() - 1 )
458 {
459 this->NavigationSelect(this->LinearizedItems[SelectionIndex + 1], InKeyEvent);
460 }
461 }
462 }
463 }
464
465 return FReply::Handled();
466 }
467 }
468
469 return SListView<ItemType>::OnKeyDown_Internal(MyGeometry, InKeyEvent);
470 }

607 virtual void Tick( const FGeometry& AllottedGeometry, const double InCurrentTime, const float InDeltaTime ) override
608 {
609 TRACE_CPUPROFILER_EVENT_SCOPE(STreeView::Tick);
610
611 if ( bTreeItemsAreDirty )
612 {
613 // Check that ItemsPanel was made; we never make it if the user failed to specify all the parameters necessary to make the tree work.
614 if ( this->ItemsPanel.IsValid() )
615 {
616 // We are about to repopulate linearized items; the ListView that TreeView is built on top of will also need to refresh.
617 bTreeItemsAreDirty = false;
618
619 if ( OnGetChildren.IsBound() && HasValidRootItemsSource() )
620 {
621 // We make copies of the old expansion and selection sets so that we can remove
622 // any items that are no longer seen by the tree.
623 TItemSet TempSelectedItemsMap;
624 TSparseItemMap TempSparseItemInfo;
625 TArray<FItemInfo> TempDenseItemInfos;
626
627 // Rebuild the linearized view of the tree data.
628 LinearizedItems.Empty();
629 PopulateLinearizedItems(GetRootItems(), LinearizedItems, TempDenseItemInfos, TBitArray<>(), TempSelectedItemsMap, TempSparseItemInfo, true, INDEX_NONE);
630
631 if( !bAllowInvisibleItemSelection &&
632 (this->SelectedItems.Num() != TempSelectedItemsMap.Num() ||
633 this->SelectedItems.Difference(TempSelectedItemsMap).Num() > 0 ||
634 TempSelectedItemsMap.Difference(this->SelectedItems).Num() > 0 ))
635 {
636 this->SelectedItems = TempSelectedItemsMap;
637
638 if ( !TListTypeTraits<ItemType>::IsPtrValid( this->RangeSelectionStart ) ||
639 !this->SelectedItems.Contains( TListTypeTraits<ItemType>::NullableItemTypeConvertToItemType( this->RangeSelectionStart ) ))
640 {
641 TListTypeTraits< ItemType >::ResetPtr( this->RangeSelectionStart );
642 TListTypeTraits< ItemType >::ResetPtr( this->SelectorItem );
643 }
644 else if ( !TListTypeTraits<ItemType>::IsPtrValid( this->SelectorItem ) ||
645 !this->SelectedItems.Contains( TListTypeTraits<ItemType>::NullableItemTypeConvertToItemType( this->SelectorItem ) ) )
646 {
647 this->SelectorItem = this->RangeSelectionStart;
648 }
649
650 this->Private_SignalSelectionChanged(ESelectInfo::Direct);
651 }
652
653 // these should come after Private_SignalSelectionChanged(); because through a
654 // series of calls, Private_SignalSelectionChanged() could end up in a child
655 // that indexes into either of these arrays (the index wouldn't be updated yet,
656 // and could be invalid)
657 SparseItemInfos = MoveTemp(TempSparseItemInfo);
658 DenseItemInfos = MoveTemp(TempDenseItemInfos);
659
660 // Once the selection changed events have gone through we can update the parent highlight statuses, which are based on your current selection.
661 if (bHighlightParentNodesForSelection)
662 {
663 this->Private_UpdateParentHighlights();
664 }
665 }
666 }
667 }
668
669 // Tick the TreeView superclass so that it can refresh.
670 // This may be due to TreeView requesting a refresh or because new items became visible due to resizing or scrolling.
671 SListView< ItemType >::Tick(AllottedGeometry, InCurrentTime, InDeltaTime);
672 }

691 bool PopulateLinearizedItems(
692 TArrayView<const ItemType> InItemsSource,
693 TArray<ItemType>& InLinearizedItems,
694 TArray<FItemInfo>& NewDenseItemInfos,
695 TBitArray<> NeedsParentWire,
696 TItemSet& OutNewSelectedItems,
697 TSparseItemMap& NewSparseItemInfo,
698 bool bAddingItems,
699 int32 ParentIndex)
700 {
701 TRACE_CPUPROFILER_EVENT_SCOPE(STreeView::PopulateLinearizedItems);
702
703 NeedsParentWire.Add(false);
704 const int32 NestingDepthIndex = NeedsParentWire.Num()-1;
705
706 bool bSawExpandedItems = false;
707 for ( int32 ItemIndex = 0; ItemIndex < InItemsSource.Num(); ++ItemIndex )
708 {
709 const ItemType& CurItem = InItemsSource[ItemIndex];
710
711 // Find this items children.
712 TArray<ItemType> ChildItems;
713 {
714 TRACE_CPUPROFILER_EVENT_SCOPE(STreeView_GetChildren);
715 OnGetChildren.Execute(InItemsSource[ItemIndex], ChildItems);
716 }
717
718 const bool bHasChildren = ChildItems.Num() > 0;
719
720 // Child items will need a parent wire at this depth if the item we are inserting now is
721 // not the last item in its immediate parent's list.
722 const bool bIsLastChild = (ItemIndex == InItemsSource.Num() - 1);
723 NeedsParentWire[NestingDepthIndex] = !bIsLastChild;
724
725 // Is this item expanded, does it have expanded children?
726 const FSparseItemInfo* CurItemInfo = SparseItemInfos.Find( CurItem );
727 const bool bIsExpanded = (CurItemInfo == nullptr) ? false : CurItemInfo->bIsExpanded;
728 bool bHasExpandedChildren = (CurItemInfo == nullptr) ? false : CurItemInfo->bHasExpandedChildren;
729
730 // Add this item to the linearized list and update the selection set.
731 if (bAddingItems)
732 {
733 InLinearizedItems.Add( CurItem );
734
735 NewDenseItemInfos.Add( FItemInfo(NeedsParentWire, bHasChildren, bIsLastChild, ParentIndex) );
736
737 const bool bIsSelected = this->IsItemSelected( CurItem );
738 if (bIsSelected)
739 {
740 OutNewSelectedItems.Add( CurItem );
741 }
742 }
743
744 if ( bIsExpanded || bHasExpandedChildren )
745 {
746 // If this item is expanded, we should process all of its children at the next indentation level.

765
766 int32 PopulatePinnedItems(const TArray<ItemType>& InItemsSource, TArray< ItemType >& InPinnedItems, const STableViewBase::FReGenerateResults& Results)
767 {
768 // The value we return, to signify if we want the hierarchy to be collapsed even if it doesn't reach the max amount
769 int32 MaxPinnedItemsOverride = -1;
770
771 if (InItemsSource.IsEmpty())
772 {
773 return MaxPinnedItemsOverride;
774 }
775
776 // Calculate the index of the first item in view
777 int32 StartIndex = FMath::Clamp((int32)(FMath::FloorToDouble(Results.NewScrollOffset)), 0, InItemsSource.Num() - 1);
778 int32 CurrentItemIndex = StartIndex;
779
780 auto GetNonVisibleParents = [this, &InItemsSource, StartIndex](TArray<ItemType>& OutParents, int32 ItemIndex) {
781 if (!DenseItemInfos.IsValidIndex(ItemIndex))
782 {
783 return;
784 }
785
786 int32 ParentIndex = ItemIndex;
787
788 // Walk through the list of parents of the current item until you reach the root
789 do
790 {
791 ParentIndex = DenseItemInfos[ItemIndex].ParentIndex;
792
793 // If the current item has a parent, and the parent is not visible, add the parent to the list of pinned items
794 if (InItemsSource.IsValidIndex(ParentIndex) && ParentIndex < StartIndex)
795 {
796 OutParents.Add(InItemsSource[ParentIndex]);
797 }
798
799 ItemIndex = ParentIndex;
800
801 } while (ParentIndex != INDEX_NONE);
802 };
803
804 /* Special Case for if we are at the end of the list. When there is no space to scroll down in a list, changing the pinned hierarchy could also change the first visible item
805 * which is used to calculate the pinned hierarchy. This leads to an infinite loop, so we solve this by finding a first visible item that has a hierachy large enough to hide
806 * itself, and then collapse the hierarchy until the item remains the first visible item (so there are no infinite loops since the first visible item doesn't change)
807 *
808 */
809 if (Results.bGeneratedPastLastItem && Results.NewScrollOffset > 0)
810 {
811 int32 LastItem = InItemsSource.Num() - 1;
812 int32 CurrentMaxPinnedItems = this->GetMaxPinnedItems();
813
814 // Could be different than reported by STableViewBase if some items are collapsed
815 int32 NumPinnedItems = (this->GetNumPinnedItems() < CurrentMaxPinnedItems) ? this->GetNumPinnedItems() : CurrentMaxPinnedItems;
816
817 // This is the first item that would be visible, if there were no pinned rows
818 int32 FirstItem = FMath::TruncToInt32(Results.NewScrollOffset - NumPinnedItems);
819
820 // We find items that have a hierarchy big enough to cover themselves, but select the smallest among them
821 int32 MinSpaceOccupied = TNumericLimits<int32>::Max();
822
823 // The index of the item we select
824 int32 MinIndex = -1;
825
826 for (int32 ItemIndex = FirstItem; ItemIndex <= LastItem; ItemIndex++)
827 {
828 // Get all parents of the current item that are not visible, to calculate the number of items in its hierarchy
829 TArray<ItemType> NonVisibleParents;
830 GetNonVisibleParents(NonVisibleParents, ItemIndex);
831
832 int32 NumParents = NonVisibleParents.Num();
833
834 // How many items would be required in the hierarchy to cover the item itself
835 int32 IndexOffset = ItemIndex - FirstItem;
836
837 // If the hierarchy is too small, ignore it
838 if (NumParents < IndexOffset)
839 {
840 continue;
841 }
842
843 // If hierarchy is the smallest we have found so far, AND the number of pinned items it will require is < the allowed max
844 if (NumParents - IndexOffset < MinSpaceOccupied && IndexOffset <= CurrentMaxPinnedItems)
845 {
846 MinSpaceOccupied = NumParents - IndexOffset;
847 MinIndex = ItemIndex;
848 }
849
850 }
851
852 // If we found no such items, we are in the middle of generating the list so pinned rows are not required
853 if (MinIndex == -1)
854 {
855 return MaxPinnedItemsOverride;
856 }
857
858 CurrentItemIndex = MinIndex;
859 MaxPinnedItemsOverride = MinIndex - FirstItem;
860 }
861
862 // Get all the parents of the item that are not visible, which is the hierarchy to stack
863 GetNonVisibleParents(InPinnedItems, CurrentItemIndex);
864
865 // Reverse the list so the root is at the front
866 Algo::Reverse(InPinnedItems);
867
868 return MaxPinnedItemsOverride;
869 }
