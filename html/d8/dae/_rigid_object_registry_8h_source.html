42 // A Registry for Rigid Objects that allows us to convert from a Handle to an Object Pointer.
43 // Each registered object is assigned an Id, which is a combination of an array index and an epoch counter.
44 // The array index and epoch counter combined are unique for every allocated object, even when array indices
45 // get reused after an object is unregistered and a new one takes its place.
46 //
47 // ObjectType must have a GetHandle() function.
48 //
49 // TODO_CHAOSAPI: ideally would not require a GetHandle() function on the object - it is only used the repack
50 // the AddedObjects array. Should be a template policy instead to get to the Id? See RemoveObject().
51 //
52 // TODO_CHAOSAPI: this container will fail to work if we reuse any ObjectsById array index enough times to
53 // wrap its Epoch. We can mitigate this by removing those indices from the free list. This will leave unusable
54 // gaps in the array but would allow for infinite growth at the cost of some wasted memory.
55 //
56 template <typename ObjectType>
57 class UE_INTERNAL TRigidObjectRegistry
58 {
59 public:
60 using FObjectType = ObjectType;
61 using FObjectPtrType = ObjectType*;
62 using FObjectRegistryItem = TRigidObjectRegistryItem<FObjectPtrType>;
63
64 public:
65 UE_INTERNAL TRigidObjectRegistry(int32 InSlack)
66 : ArraySlack(InSlack)
67 , TotalNumAdded(0)
68 , bIsVisiting(false)
69 {
70 }
71
72 UE_INTERNAL ~TRigidObjectRegistry() = default;
73
74 // The number of objects that have been added (and not yet removed)
75 UE_INTERNAL int32 GetNumObjects() const
76 {
77 return ObjectsById.Num() - ObjectsByIdFreeIndices.Num();
78 }
79
80 // True if there are no objects in the registry
81 UE_INTERNAL bool IsEmpty() const
82 {
83 return GetNumObjects() == 0;
84 }
85
86 // Add an object to the registry and assign it an Id. The Id is persistent
87 // and will never be reused as long as less that 1<<31 objects are added.
88 // After that, Ids may start to repeat.
89 UE_INTERNAL FRigidObjectId AddObject(FObjectPtrType InObject)
90 {
91 UE_RIGIDPHYSICS_CHECK(InObject != nullptr);
92
93 // Cannot be called while visiting elements
94 UE_RIGIDPHYSICS_CHECK(!bIsVisiting);
95
96 // Assert if we add enough objects such that we might wrap the Epoch counter
97 // (this is a lower bound, but is consistent and doesn't depend on add/remove order)
98 // See class comments for future solution to this issue.
99 UE_RIGIDPHYSICS_CHECK(TotalNumAdded < std::numeric_limits<int32>::max());
100 ++TotalNumAdded;
101
102 // Make sure we have free array entries
103 if (ObjectsByIdFreeIndices.IsEmpty())
104 {
105 ReserveArray(ObjectsById.Num() + ArraySlack);
106 }
107
108 // Assign an array index
109 int32 ObjectsByIdIndex = ObjectsByIdFreeIndices.Pop(EAllowShrinking::No);
110
111 // Put the item in the array
112 FObjectRegistryItem& ObjectsByIdItem = ObjectsById[ObjectsByIdIndex];
113 UE_RIGIDPHYSICS_CHECK(ObjectsByIdItem.Object == nullptr);
114 ObjectsByIdItem.Object = InObject;
115 ObjectsByIdItem.AddedObjectsIndex = AddedObjects.Add(InObject);
116
117 // Build the Id from the Index and Epoch
118 return FRigidObjectId(ObjectsByIdIndex, ObjectsByIdItem.Epoch);
119 }
120
121 UE_INTERNAL void RemoveObject(const FRigidObjectId& InObjectId)
122 {
123 // Cannot be called while visiting elements
124 UE_RIGIDPHYSICS_CHECK(!bIsVisiting);
125
126 // Call FindObject() first if presence is unknown
127 UE_RIGIDPHYSICS_CHECK(ObjectsById.IsValidIndex(InObjectId.GetArrayIndex()));
128
129 // Get the entry
130 FObjectRegistryItem& ObjectsByIdItem = ObjectsById[InObjectId.GetArrayIndex()];
131
132 // NOTE: We already know that this object was in the map before calling
133 // so the epoch should always match
134 UE_RIGIDPHYSICS_CHECK(ObjectsByIdItem.Epoch == InObjectId.GetEpoch());
135
136 // Remove from the Allocated array (and fix up the AddedObjectsIndex of the moved item)
137 int32 AddedObjectsIndex = ObjectsByIdItem.AddedObjectsIndex;
138 if (AddedObjects.IsValidIndex(AddedObjectsIndex))
139 {
140 // If this fires, we have a ObjectsByIdItem.AddedObjectsIndex mismatch
141 UE_RIGIDPHYSICS_CHECK(AddedObjects[AddedObjectsIndex] == ObjectsByIdItem.Object);
142
143 AddedObjects.RemoveAtSwap(AddedObjectsIndex, EAllowShrinking::No);
144
145 if (AddedObjects.IsValidIndex(AddedObjectsIndex))
146 {
147 // TODO_CHAOSAPI: GetId() is only required to maintain the packed allocated object array. Try to remove it.
148 FRigidObjectId RelocatedObjectId = AddedObjects[AddedObjectsIndex]->GetHandle().GetId();
149 FObjectRegistryItem& ReloctedArrayItem = ObjectsById[RelocatedObjectId.GetArrayIndex()];
150 ReloctedArrayItem.AddedObjectsIndex = AddedObjectsIndex;
151 }
152 }
153
154 // Reset the entry and increment the epoch to invalidate outstanding Ids using the same index
155 ObjectsByIdItem.Object = nullptr;
156 ObjectsByIdItem.Epoch = ObjectsByIdItem.Epoch + 1;
157 ObjectsByIdItem.AddedObjectsIndex = INDEX_NONE;
158
159 // Add the array item back into the free pool
160 ObjectsByIdFreeIndices.Push(InObjectId.GetArrayIndex());
161
162 // TODO_CHAOSAPI: we could shrink the ObjectArray here if we have too much slack, but
163 // we'd need to know what the max allocated index was, and would have to prune
164 // the free indices.
165 }
166
167 UE_INTERNAL void RemoveObjectChecked(const FRigidObjectId& InObjectId, const ObjectType* Object)
168 {
169 UE_RIGIDPHYSICS_CHECK(FindObject(InObjectId) == Object);
170 RemoveObject(InObjectId);
171 }

173 UE_INTERNAL FObjectType* FindObject(const FRigidObjectId& InId) const
174 {
175 if (ObjectsById.IsValidIndex(InId.GetArrayIndex()))
176 {
177 const FObjectRegistryItem& ArrayItem = ObjectsById[InId.GetArrayIndex()];
178 if (ArrayItem.Epoch == InId.GetEpoch())
179 {
180 return ArrayItem.Object;
181 }
182 }
183 return nullptr;
184 }

186 // Visit all of the added objects in somewhat arbitrary order that may change whenever
187 // objects are added or removed.
188 UE_INTERNAL void VisitObjects(const TFunctionRef<ERigidVisitorResponse(FObjectType*)>& Visitor)
189 {
190 bIsVisiting = true;
191 for (FObjectType* Object : AddedObjects)
192 {
193 ERigidVisitorResponse VisitorResponse = Visitor(Object);
194 if (VisitorResponse == ERigidVisitorResponse::Break)
195 {
196 break;
197 }
198 }
199 bIsVisiting = false;
200 }
201
202 private:
203 void ReserveArray(int32 MaxItems)
204 {
205 if (MaxItems > ObjectsById.Num())
206 {
207 // Resize the arrays
208 int32 PrevNumItems = ObjectsById.Num();
209 ObjectsById.SetNum(MaxItems);
210 ObjectsByIdFreeIndices.Reserve(MaxItems);
211 AddedObjects.Reserve(MaxItems);
212
213 // Add to the free indices
214 // Loop backawards so we allocate lowest indices first in AddObject
215 for (int32 ItemIndex = MaxItems - 1; ItemIndex >= PrevNumItems; --ItemIndex)
216 {
217 ObjectsByIdFreeIndices.Push(ItemIndex);
218 }
219 }
220 }
221
222 // An array with no holes containing all currently allocated objects indexed by ObjectId.AddedObjectsIndex.
223 // This exists to support iteration over items and easier debugging - we may be able remove it later.
224 // NOTE: Object position in this array is not persistent (RemoveObject uses RemoveAtSwap)
225 // TODO_CHAOSAPI: see if we really need the AddedObjects array.
226 TArray<FObjectType*> AddedObjects;
227
228 // An Id->Object map, implemented as an array indexed by ObjectId.ArrayIndex.
229 // Indices into this array are persistent as long as the object exists, but the array may contain holes.
230 // Array Entries will be reused after objects have been deleted, and the Epoch counter is used to
231 // validate handles against the current object at the array index.
232 // This array is presized and will often contain many empty (unused) elements.
233 TArray<FObjectRegistryItem> ObjectsById;
234
235 // A pool of unused entries in ObjectsById.
236 TArray<int32> ObjectsByIdFreeIndices;
237
238 // How much extra to allocate in ObjectsById when we run out of space.
239 int32 ArraySlack;
240
241 // How many objects have been added. This is only used to assert if we may reuse Ids.
242 int32 TotalNumAdded;
243
244 // Use to trap modifications while in the Visitor loop
245 bool bIsVisiting;
246 };
