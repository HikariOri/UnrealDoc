114struct FAnimPhysPose
115{
116 FVector Position;
117 FQuat Orientation;
118 float Scale;
119
120 FAnimPhysPose(const FVector& InPosition, const FQuat& InOrient)
121 : Position(InPosition)
122 , Orientation(InOrient)
123 , Scale(1.0)
124 {}
125
126 FAnimPhysPose(const FVector& InPosition, const FQuat& InOrient, const float& InScale)
127 : Position(InPosition)
128 , Orientation(InOrient)
129 , Scale(InScale)
130 {}
131
132 FAnimPhysPose()
133 : Position(FVector::ZeroVector)
134 , Orientation(FQuat::Identity)
135 , Scale(1.0)
136 {
137
138 }
139
140 FAnimPhysPose Inverse() const
141 {
142 FQuat InverseOrient = Orientation.Inverse();
143 FVector InversePosition = InverseOrient * Position;
144 return FAnimPhysPose(InversePosition, InverseOrient);
145 }
146
147 FMatrix Matrix() const
148 {
149 return FTransform(Orientation, Position).ToMatrixNoScale();
150 }
151
152 FVector operator*(const FVector& InPoint) const
153 {
154 return Position + (Orientation * InPoint);
155 }
156
157 FAnimPhysPose operator*(const FAnimPhysPose& InPose) const
158 {
159 return FAnimPhysPose(*this *InPose.Position, Orientation * InPose.Orientation);
160 }
161
162 FPlane TransformPlane(const FPlane& InPlane)
163 {
164 FVector NewNormal(InPlane.X, InPlane.Y, InPlane.Z);
165 NewNormal = Orientation * NewNormal;
166 return FPlane(NewNormal, InPlane.W - FVector::DotProduct(Position, NewNormal));
167 }
168};
