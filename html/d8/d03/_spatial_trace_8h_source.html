183 {
184 SetInternal({
185 .Position = Position,
186 .Velocity = Velocity,
187 .Direction = Direction,

192
193 void SetInternal(const FData& NewData)
194 {
195 EPointComponentFlags ChangedMask = EPointComponentFlags::None;
196 if constexpr (EnumHasAnyFlags(ComponentMask, EPointComponentFlags::Position))
197 {
198 if (Data.Position != NewData.Position)
199 {
200 Data.Position = NewData.Position;
201 ChangedMask |= EPointComponentFlags::Position;
202 }
203 }
204 if constexpr (EnumHasAnyFlags(ComponentMask, EPointComponentFlags::Velocity))
205 {
206 if (Data.Velocity != NewData.Velocity)
207 {
208 Data.Velocity = NewData.Velocity;
209 ChangedMask |= EPointComponentFlags::Velocity;
210 }
211 }
212 if constexpr (EnumHasAnyFlags(ComponentMask, EPointComponentFlags::Direction))
213 {
214 if (Data.Direction != NewData.Direction)
215 {
216 Data.Direction = NewData.Direction;
217 ChangedMask |= EPointComponentFlags::Direction;
218 }
219 }
220
221 if (!UE_TRACE_CHANNELEXPR_IS_ENABLED(SpatialChannel) || ChangedMask == EPointComponentFlags::None)
222 {
223 return;
224 }
225
226 uint32 SpecId = InvalidPointSpecId;
227 bool bSpecCreated = false;
228 Tie(SpecId, bSpecCreated) = GetOrCreatePointSpecId(Id, Name.GetView(), ComponentMask);
229 TracePointSet(SpecId, ComponentMask, bSpecCreated ? ComponentMask : ChangedMask, GetDataView());
230 }
231
232 FPointDataView GetDataView() const
233 {
234 FPointDataView DataView;
235 if constexpr (EnumHasAnyFlags(ComponentMask, EPointComponentFlags::Position))
236 {
237 DataView.Position = &Data.Position;
238 }
239 if constexpr (EnumHasAnyFlags(ComponentMask, EPointComponentFlags::Velocity))
240 {
241 DataView.Velocity = &Data.Velocity;
242 }
243 if constexpr (EnumHasAnyFlags(ComponentMask, EPointComponentFlags::Direction))
244 {
245 DataView.Direction = &Data.Direction;
246 }
247 return DataView;
248 }
