72 {
73 checkSlow(!(NewFlags & (RF_MarkAsNative | RF_MarkAsRootSet | RF_MirroredGarbage))); // These flags can't be used outside of constructors / internal code

632 template <typename OtherClassType>
633 inline bool IsA( OtherClassType SomeBase ) const
634 {
635 // We have a cyclic dependency between UObjectBaseUtility and UClass,
636 // so we use a template to allow inlining of something we haven't yet seen, because it delays compilation until the function is called.
637
638 // 'static_assert' that this thing is actually a UClass pointer or convertible to it.
639 const UClass* SomeBaseClass = SomeBase;
640 (void)SomeBaseClass;
641 checkfSlow(SomeBaseClass, TEXT("IsA(NULL) cannot yield meaningful results"));
642
643 const UClass* ThisClass = GetClass();
644
645 // Stop the compiler doing some unnecessary branching for nullptr checks
646 UE_ASSUME(SomeBaseClass);
647 UE_ASSUME(ThisClass);
648
649 return IsChildOfWorkaround(ThisClass, SomeBaseClass);
650 }

848inline bool IsPossiblyAllocatedUObjectPointer(UObject* Ptr)
849{
850 auto CountByteValues = [](UPTRINT Val, UPTRINT ByteVal) -> int32
851 {
852 int32 Result = 0;
853
854 // != changed to < because of false positive in MSVC's static analyzer: warning C6295: Ill-defined for-loop. Loop executes infinitely.
855 for (int32 I = 0; I < sizeof(UPTRINT); ++I)
856 {
857 if ((Val & 0xFF) == ByteVal)
858 {
859 ++Result;
860 }
861 Val >>= 8;
862 }
863
864 return Result;
865 };
866
867 UPTRINT PtrVal = (UPTRINT)Ptr;
868 return PtrVal >= 0x1000 && CountByteValues(PtrVal, 0xCD) < sizeof(UPTRINT) / 2;
869}
