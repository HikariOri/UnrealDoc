60 void GetNavigatableTimes(TArray<FFrameNumber>& NavigatableTimes)
61 {
62 TSharedPtr<FViewModel> DataModel = WeakModel.Pin();
63 if (!DataModel)
64 {
65 return;
66 }
67
68 TSet<TSharedPtr<IKeyArea>> KeyAreas;
69 SequencerHelpers::GetAllKeyAreas( DataModel, KeyAreas );
70
71 for ( TSharedPtr<IKeyArea> Keyarea : KeyAreas )
72 {
73 Keyarea->GetKeyTimes(NavigatableTimes);
74 }
75
76 TSet<TWeakObjectPtr<UMovieSceneSection> > Sections;
77 SequencerHelpers::GetAllSections( DataModel, Sections );
78 for ( TWeakObjectPtr<UMovieSceneSection> Section : Sections )
79 {
80 if (Section.IsValid())
81 {
82 Section->GetSnapTimes(NavigatableTimes, true);
83 }
84 }
85 }

108 FScopedTransaction Transaction(LOCTEXT("AddKeys", "Add Keys at Current Time"));
109 FAddKeyOperation::FromNode(InModel).Commit(Time.RoundToFrame(), *Sequencer);
110 }
111 }
