99inline FRHIRasterizerState* GetStaticRasterizerState(ERasterizerFillMode FillMode,ERasterizerCullMode CullMode)
100{
101 constexpr ERasterizerDepthClipMode DepthClipMode = ERasterizerDepthClipMode::DepthClip;
102 switch(FillMode)
103 {
104 default:
105 case FM_Solid:
106 switch(CullMode)
107 {
108 default:
109 case CM_CW: return TStaticRasterizerState<FM_Solid,CM_CW , DepthClipMode, bEnableMSAA>::GetRHI();
110 case CM_CCW: return TStaticRasterizerState<FM_Solid,CM_CCW , DepthClipMode, bEnableMSAA>::GetRHI();
111 case CM_None: return TStaticRasterizerState<FM_Solid,CM_None, DepthClipMode, bEnableMSAA>::GetRHI();
112 };
113 break;
114 case FM_Wireframe:
115 switch(CullMode)
116 {
117 default:
118 case CM_CW: return TStaticRasterizerState<FM_Wireframe,CM_CW , DepthClipMode, bEnableMSAA>::GetRHI();
119 case CM_CCW: return TStaticRasterizerState<FM_Wireframe,CM_CCW , DepthClipMode, bEnableMSAA>::GetRHI();
120 case CM_None: return TStaticRasterizerState<FM_Wireframe,CM_None, DepthClipMode, bEnableMSAA>::GetRHI();
121 };
122 break;
123 case FM_Point:
124 switch(CullMode)
125 {
126 default:
127 case CM_CW: return TStaticRasterizerState<FM_Point,CM_CW , DepthClipMode, bEnableMSAA>::GetRHI();
128 case CM_CCW: return TStaticRasterizerState<FM_Point,CM_CCW , DepthClipMode, bEnableMSAA>::GetRHI();
129 case CM_None: return TStaticRasterizerState<FM_Point,CM_None, DepthClipMode, bEnableMSAA>::GetRHI();
130 };
131 break;
132 }
