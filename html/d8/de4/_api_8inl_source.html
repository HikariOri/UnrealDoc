220bool FTicketStatus::IsComplete() const
221{
222 EId Id = GetId();
223 if (Id == EId::Response)
224 {
225 // If the response has no body there will be no further status updates
226 return GetResponse().GetContentStatus() == EContentStatus::NoContent;
227 }
228
229 if (Id == EId::Content)
230 {
231 const auto* Activity = (FActivity*)this;
232
233 // When the size of content is unknown then the end is signalled with
234 // an empty destination buffer. Otherwise a status update is only
235 // called when all the destination has been received.
236 int64 ContentLength = Activity->GetTransaction()->GetContentLength();
237 ContentLength = FMath::Max(ContentLength, 0ll);
238 return GetContent().GetSize() == ContentLength;
239 }
240
241 return true;
242}

292FTicketPerf::FSample FTicketPerf::GetSample() const
293{
294 const auto& Activity = *(FActivity*)this;
295
296 static uint64 Freq;
297 if (Freq == 0)
298 {
299 Freq = uint64(1.0 / FPlatformTime::GetSecondsPerCycle64());
300 }
301
302 auto Clamp = [] (auto Value) { return uint16(FMath::Min(uint32(Value), 0xffffu)); };
303 auto ToMs = [&] (uint64 Value) { return Clamp((Value * 1000ull) / Freq); };
304
305 const FStopwatch& Stopwatch = Activity.GetStopwatch();
306 FSample Sample = {
307 ToMs(Stopwatch.GetInterval(0)),
308 ToMs(Stopwatch.GetInterval(1)),
309 ToMs(Stopwatch.GetInterval(2)),
310 };
311
312 uint32 Bps = ~0u;
313 if (Sample.RecvMs)
314 {
315 Bps = uint32((Activity.GetTransaction()->GetContentLength() * 1000ull) / Sample.RecvMs);
316 }
317 Sample.RecvKiBps = Clamp(Bps >> 10);
318
319 return Sample;
320}
