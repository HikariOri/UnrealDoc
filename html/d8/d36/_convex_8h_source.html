240 // Distance to the surface
241 FReal PhiWithNormalInternal(const FVec3& X, FVec3& Normal) const
242 {
243 const int32 NumPlanes = Planes.Num();
244 if (NumPlanes == 0)
245 {
246 return FLT_MAX;
247 }
248 check(NumPlanes > 0);
249
250 FReal MaxPhi = TNumericLimits<FReal>::Lowest();
251 int32 MaxPlane = 0;
252
253 for (int32 Idx = 0; Idx < NumPlanes; ++Idx)
254 {
255 const FReal Phi = Planes[Idx].SignedDistance(X);
256 if (Phi > MaxPhi)
257 {
258 MaxPhi = Phi;
259 MaxPlane = Idx;
260 }
261 }
262
263 FReal Phi = Planes[MaxPlane].PhiWithNormal(X, Normal);
264 if (Phi <= 0)
265 {
266 return Phi;
267 }
268
269 // If x is outside the convex mesh, we should find for the nearest point to triangles on the plane
270 const int32 PlaneVerticesNum = NumPlaneVertices(MaxPlane);
271 const FVec3 XOnPlane = X - Phi * Normal;
272 FReal ClosestDistance = TNumericLimits<FReal>::Max();
273 FVec3 ClosestPoint;
274 for (int32 Index = 0; Index < PlaneVerticesNum - 2; Index++)
275 {
276 const FVec3 A(GetVertex(GetPlaneVertex(MaxPlane, 0)));
277 const FVec3 B(GetVertex(GetPlaneVertex(MaxPlane, Index + 1)));
278 const FVec3 C(GetVertex(GetPlaneVertex(MaxPlane, Index + 2)));
279
280 const FVec3 Point = FindClosestPointOnTriangle(XOnPlane, A, B, C, X);
281 if (XOnPlane == X)
282 {
283 return Phi;
284 }
285
286 const FReal Distance = (Point - XOnPlane).Size();
287 if (Distance < ClosestDistance)
288 {
289 ClosestDistance = Distance;
290 ClosestPoint = Point;
291 }
292 }
293
294 const TVector<FReal, 3> Difference = X - ClosestPoint;
295 Phi = Difference.Size();
296 if (Phi > UE_SMALL_NUMBER)
297 {
298 Normal = (Difference) / Phi;
299 }
300 return Phi;
301 }

303 // Distance from a point to the surface for use in the scaled version. When the convex
304 // is scaled, we need to bias the depth calculation to take into account the world-space scale
305 FReal PhiWithNormalScaledInternal(const FVec3& X, const FVec3& Scale, FVec3& Normal) const
306 {
307 const int32 NumPlanes = Planes.Num();
308 if (NumPlanes == 0)
309 {
310 return FLT_MAX;
311 }
312 check(NumPlanes > 0);
313
314 FReal MaxPhi = TNumericLimits<FReal>::Lowest();
315 FVec3 MaxNormal = FVec3(0,0,1);
316 int32 MaxPlane = 0;
317 for (int32 Idx = 0; Idx < NumPlanes; ++Idx)
318 {
319 FVec3 PlaneNormal = (Planes[Idx].Normal() / Scale).GetUnsafeNormal();
320 FVec3 PlanePos = Planes[Idx].X() * Scale;
321 FReal PlaneDistance = FVec3::DotProduct(X - PlanePos, PlaneNormal);
322 if (PlaneDistance > MaxPhi)
323 {
324 MaxPhi = PlaneDistance;
325 MaxNormal = PlaneNormal;
326 MaxPlane = Idx;
327 }
328 }
329
330 Normal = MaxNormal;
331
332 if (MaxPhi < 0)
333 {
334 return MaxPhi;
335 }
336
337 // If X is outside the convex mesh, we should find for the nearest point to triangles on the plane
338 const int32 PlaneVerticesNum = NumPlaneVertices(MaxPlane);
339 const FVec3 XOnPlane = X - MaxPhi * Normal;
340 FReal ClosestDistance = TNumericLimits<FReal>::Max();
341 FVec3 ClosestPoint;
342 for (int32 Index = 0; Index < PlaneVerticesNum - 2; Index++)
343 {
344 const FVec3 A(Scale * GetVertex(GetPlaneVertex(MaxPlane, 0)));
345 const FVec3 B(Scale * GetVertex(GetPlaneVertex(MaxPlane, Index + 1)));
346 const FVec3 C(Scale * GetVertex(GetPlaneVertex(MaxPlane, Index + 2)));
347
348 const FVec3 Point = FindClosestPointOnTriangle(XOnPlane, A, B, C, X);
349 if (XOnPlane == X)
350 {
351 return MaxPhi;
352 }
353
354 const FReal Distance = (Point - XOnPlane).Size();
355 if (Distance < ClosestDistance)
356 {
357 ClosestDistance = Distance;
358 ClosestPoint = Point;
359 }
360 }
361
362 const FVec3 Difference = X - ClosestPoint;
363 const FReal DifferenceLen = Difference.Size();
364 if (DifferenceLen > UE_SMALL_NUMBER)
365 {
366 Normal = Difference / DifferenceLen;
367 MaxPhi = DifferenceLen;
368 }
369 return MaxPhi;
370 }

382 virtual Pair<FVec3, bool> FindClosestIntersectionImp(const FVec3& StartPoint, const FVec3& EndPoint, const FReal Thickness) const override
383 {
384 const int32 NumPlanes = Planes.Num();
385 TArray<Pair<FReal, FVec3>> Intersections;
386 Intersections.Reserve(FMath::Min(static_cast<int32>(NumPlanes*.1), 16)); // Was NumPlanes, which seems excessive.
387 for (int32 Idx = 0; Idx < NumPlanes; ++Idx)
388 {
389 // Offset plane with thickness in the normal direction and do a 0 thickness intersection query, so you only find intersections on the +Thickness*Normal plane, not the -Thickness*Normal plane.
390 const Pair<FVec3, bool> PlaneIntersection = Thickness > 0 ?
391 FPlaneType(Planes[Idx].X() + Thickness * Planes[Idx].Normal(), Planes[Idx].Normal()).FindClosestIntersection(StartPoint, EndPoint, 0) :
392 Planes[Idx].FindClosestIntersection(StartPoint, EndPoint, 0);
393 if (PlaneIntersection.Second)
394 {
395 Intersections.Add(MakePair((FReal)(PlaneIntersection.First - StartPoint).SizeSquared(), PlaneIntersection.First));
396 }
397 }
398 Intersections.Sort([](const Pair<FReal, FVec3>& Elem1, const Pair<FReal, FVec3>& Elem2) { return Elem1.First < Elem2.First; });
399 for (const Pair<FReal, FVec3>& Elem : Intersections)
400 {
401 if (this->SignedDistance(Elem.Second) < (Thickness + 1e-4))
402 {
403 return MakePair(Elem.Second, true);
404 }
405 }
406 return MakePair(FVec3(0), false);
407 }

595 // @todo(chaos): Move to utils
596 inline bool IntersectPlanes3(const FVec3& X1, const FVec3& N1, const FVec3& X2, const FVec3& N2, const FVec3& X3, const FVec3& N3, FVec3& OutX, const FReal EpsilonSq = FReal(1.e-6)) const
597 {
598 // Compute determinant, the triple product P1|(P2^P3)==(P1^P2)|P3.
599 const FVec3 N1CrossN2 = FVec3::CrossProduct(N1, N2);
600 const FReal Det = FVec3::DotProduct(N1CrossN2, N3);
601 if (FMath::Square(Det) < EpsilonSq)
602 {
603 // Degenerate.
604 OutX = FVec3(0);
605 return false;
606 }
607 else
608 {
609 // Compute the intersection point, guaranteed valid if determinant is nonzero.
610 const FVec3 N2CrossN3 = FVec3::CrossProduct(N2, N3);
611 const FVec3 N3CrossN1 = FVec3::CrossProduct(N3, N1);
612 const FReal D1 = FVec3::DotProduct(X1, N1);
613 const FReal D2 = FVec3::DotProduct(X2, N2);
614 const FReal D3 = FVec3::DotProduct(X3, N3);
615 OutX = (D1 * N2CrossN3 + D2 * N3CrossN1 + D3 * N1CrossN2) / Det;
616 }
617 return true;
618 }

620 FVec3 GetMarginAdjustedVertex(const int32 VertexIndex, const FReal InMargin, FReal* OutSupportDelta) const
621 {
622 // @chaos(todo): moving the vertices this way based on margin is only valid for small margins. If the margin
623 // is large enough to cause a face to reduce to zero size, vertices should be merged and the path is non-linear.
624 // This can be fixed with some extra data in the convex structure, but for now we accept the fact that large
625 // margins on convexes with small faces can cause non-convex core shapes.
626
627 if (InMargin == FReal(0))
628 {
629 return GetVertex(VertexIndex);
630 }
631
632 // Get any 3 planes that contribute to this vertex
633 int32 PlaneIndex0 = INDEX_NONE;
634 int32 PlaneIndex1 = INDEX_NONE;
635 int32 PlaneIndex2 = INDEX_NONE;

713 // @todo(chaos): track down the invalid hull issue

715 if (NumVertexPlanes == 2)
716 {
717 const FVec3 NewPlaneX = Scale * GetVertex(VertexIndex);
718 const FVec3 NewPlaneN0 = (Planes[PlaneIndex0].Normal() * InvScale).GetUnsafeNormal();
