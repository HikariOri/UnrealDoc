17 static bool ElementPassesFilter(ECurveFilterMode InFilterMode, ECurveFilterFlags InFilterFlags)
18 {
19 if(EnumHasAnyFlags(InFilterFlags, ECurveFilterFlags::Disallowed))
20 {
21 return false;

27 return true;

39 // Helper function for building curves, applying any filtering
40 // Assumes (and enforces in debug builds) elements being built are in FName sorted order
41 template<typename NamePredicateType, typename ValuePredicateType, typename CurveAllocatorType, typename CurveElementType>
42 static void BuildSortedFiltered(TBaseBlendedCurve<CurveAllocatorType, CurveElementType>& OutCurve, int32 InNumElements, NamePredicateType InNamePredicate, ValuePredicateType InValuePredicate, const FCurveFilter& InFilter)
43 {
44 // Early out if we are filtering all curves
45 if(InFilter.FilterMode == ECurveFilterMode::DisallowAll)
46 {
47 OutCurve.Empty();
48 return;
49 }
50
51 InFilter.SortElementsIfRequired();
52
53 OutCurve.Empty();
54 OutCurve.Reserve(InNumElements);
55
56 const int32 NumElements0 = InNumElements;
57 const int32 NumElements1 = InFilter.Elements.Num();
58
59 int32 ElementIndex0 = 0;
60 int32 ElementIndex1 = 0;
61
62#if DO_ANIM_NAMED_VALUE_SORTING_CHECKS
63 FName LastCurveName;
64#endif
65
66 // Perform dual-iteration on the two sorted arrays
67 while(ElementIndex0 < NumElements0 || ElementIndex1 < NumElements1)
68 {
69 if(ElementIndex0 == NumElements0 && ElementIndex1 < NumElements1)
70 {
71 // Reached end of user data with remaining in filter, we can just early out
72 break;
73 }
74 else if(ElementIndex1 == NumElements1 && ElementIndex0 < NumElements0)
75 {
76 // Reached end of filter with remaining in user data, we can just run straight through the user data
77 OutCurve.Elements.Reserve(OutCurve.Elements.Num() + (NumElements0 - ElementIndex0));
78 for( ; ElementIndex0 < NumElements0; ++ElementIndex0)
79 {
80 if(ElementPassesFilter(InFilter.FilterMode, ECurveFilterFlags::None))
81 {
82 OutCurve.Elements.Emplace(InNamePredicate(ElementIndex0), InValuePredicate(ElementIndex0));
83 }
84 }
85 break;
86 }
87
88 const FName CurveName = InNamePredicate(ElementIndex0);
89 const FCurveFilterElement* RESTRICT Element1 = &InFilter.Elements[ElementIndex1];
90
91#if DO_ANIM_NAMED_VALUE_SORTING_CHECKS
92 // Check sorting invariants
93 if(ElementIndex0 > 0)
94 {
95 check(LastCurveName == CurveName || LastCurveName.FastLess(CurveName))
96 }
97 LastCurveName = CurveName;
98#endif
99
100 if(CurveName == Element1->Name)
101 {
102 // Elements match, check filter & write to curve
103 if(ElementPassesFilter(InFilter.FilterMode, Element1->Flags))
104 {
105 OutCurve.Elements.Emplace(CurveName, InValuePredicate(ElementIndex0));
106 }
107
108 ++ElementIndex0;
109 ++ElementIndex1;
110 }
111 else if(CurveName.FastLess(Element1->Name))
112 {
113 // Element exists only in user data
114 if(ElementPassesFilter(InFilter.FilterMode, ECurveFilterFlags::None))
115 {
116 OutCurve.Elements.Emplace(CurveName, InValuePredicate(ElementIndex0));
117 }
118
119 ++ElementIndex0;
120 }
121 else
122 {
123 // Element exists only in filter, skip
124 ++ElementIndex1;
125 }
126 }
127
128 OutCurve.CheckDuplicates();
129 }

289 SortedIndices.SetNumUninitialized(InNumElements);
290 for(int32 NameIndex = 0; NameIndex < InNumElements; ++NameIndex)

332 template<typename CurveAllocatorType, typename CurveElementType>
333 static void Filter(TBaseBlendedCurve<CurveAllocatorType, CurveElementType>& InOutCurve, const FCurveFilter& InFilter)
334 {
335 CURVE_PROFILE_CYCLE_COUNTER(FCurveUtils_Filter);
336
337 switch(InFilter.FilterMode)
338 {
339 case ECurveFilterMode::None:
340 // No filtering, early out
341 return;
342 case ECurveFilterMode::DisallowAll:
343 // Filtering all curves, so just clear curve
344 InOutCurve.Elements.Reset();
345 return;
346 case ECurveFilterMode::AllowOnlyFiltered:
347 if(InFilter.Num() == 0)
348 {
349 // Allow only filtered, and no filtered, so just clear curve
350 InOutCurve.Elements.Reset();
351 return;
352 }
353 break;
354 default:
355 break;
356 }
357
358 // Sort both inputs if required
359 InOutCurve.SortElementsIfRequired();
360 InFilter.SortElementsIfRequired();
361
362 int32 NumElements0 = InOutCurve.Num();
363 const int32 NumElements1 = InFilter.Num();
364
365 // Perform dual-iteration on the two sorted arrays
366 int32 ElementIndex0 = 0;
367 int32 ElementIndex1 = 0;
368
369 while(true)
370 {
371 if(ElementIndex0 == NumElements0 && ElementIndex1 < NumElements1)
372 {
373 // Reached end of curve with remaining in filter, we can just early out
374 break;
375 }
376 else if(ElementIndex1 == NumElements1 && ElementIndex0 < NumElements0)
377 {
378 // Reached end of filter with remaining in curve, we can just run straight through the curve
379 while(ElementIndex0 < NumElements0)
380 {
381 if(!ElementPassesFilter(InFilter.FilterMode, ECurveFilterFlags::None))
382 {
383 InOutCurve.Elements.RemoveAt(ElementIndex0, EAllowShrinking::No);
384 NumElements0 = InOutCurve.Num();
385 }
386 else
387 {
388 ++ElementIndex0;
389 }
390 }
391 break;
392 }
393 else if(ElementIndex0 == NumElements0 && ElementIndex1 == NumElements1)
394 {
395 // All elements exhausted, exit
396 break;
397 }
398
399 const CurveElementType* RESTRICT Element0 = &InOutCurve.Elements[ElementIndex0];
400 const FCurveFilterElement* RESTRICT Element1 = &InFilter.Elements[ElementIndex1];
401
402 if(Element0->Name == Element1->Name)
403 {
404 // Elements match so check filter flags to see if it should be removed from curve
405 if(!ElementPassesFilter(InFilter.FilterMode, Element1->Flags))
406 {
407 InOutCurve.Elements.RemoveAt(ElementIndex0, EAllowShrinking::No);
408 NumElements0 = InOutCurve.Num();
409 ++ElementIndex1;
410 }
411 else
412 {
413 ++ElementIndex0;
414 ++ElementIndex1;
415 }
416 }
417 else if(Element0->Name.FastLess(Element1->Name))
418 {
419 // Element exists only in curve, check filter
420 if(!ElementPassesFilter(InFilter.FilterMode, ECurveFilterFlags::None))
421 {
422 InOutCurve.Elements.RemoveAt(ElementIndex0, EAllowShrinking::No);
423 NumElements0 = InOutCurve.Num();
424 }
425 else
426 {
427 ++ElementIndex0;
428 }
429 }
430 else
431 {
432 // Element exists only in filter, skip
433 ++ElementIndex1;
434 }
435 }
436
437 InOutCurve.CheckSorted();
438 }
