150 inline void SetObject(class UObjectBase* InObject)
151 {
152#if UE_ENABLE_FUOBJECT_ITEM_PACKING

234 inline bool ThisThreadAtomicallyClearedFlag(EInternalObjectFlags FlagToClear)
235 {
236 checkf((int32(FlagToClear) & ~int32(EInternalObjectFlags_AllFlags)) == 0, TEXT("%d is not a valid internal flag value"), int32(FlagToClear));
237 bool bIChangedIt = false;
238
239 UE_AUTORTFM_OPEN
240 {
241 FlagToClear &= ~EInternalObjectFlags_ReachabilityFlags; // reachability flags can only be cleared by GC through *_ForGC functions
242 if (!!(FlagToClear & EInternalObjectFlags_RootFlags))
243 {
244 bIChangedIt = ClearRootFlags(FlagToClear);
245 }
246 else
247 {
248 bIChangedIt = AtomicallyClearFlag_ForGC(FlagToClear);
249 }
250 };
251
252 if (bIChangedIt)
253 {
254 // If we abort we undo clearing the flags we just set because `bIChangedIt` being true tells us that we were the ones who cleared the flag.
255 // #jira SOL-8462: this could potentially undo flag changes made on a separate thread if an abort occurs.
256 UE_AUTORTFM_ONABORT(this, FlagToClear)
257 {
258 ThisThreadAtomicallySetFlag(FlagToClear);
259 };
260 }
261
262 return bIChangedIt;
263 }

347 void AddRef()
348 {
349 UE_AUTORTFM_OPEN
350 {
351 const int64 NewRefCount = FPlatformAtomics::InterlockedIncrement(&FlagsAndRefCount);
352
353 // Only dirty root when doing a transition from no root flags to a refcount.
354 if ((NewRefCount & EInternalObjectFlags_RefCountMask) == 1 && (GetFlagsInternal(NewRefCount) & (int32)EInternalObjectFlags_RootFlags) == 0)
355 {
356 if (UE::GC::GIsIncrementalReachabilityPending)
357 {
358 // Setting any of the root flags on an object during incremental reachability requires a GC barrier
359 // to make sure an object with root flags does not get Garbage Collected
360 checkf(GetObject(), TEXT("Setting an internal object flag on a null object entry"));
361 GetObject()->MarkAsReachable();
362 }
363
364 MarkRootAsDirty();
365 }
366 };
367
368 // If the transaction is aborted we need to remember to release the reference we added in the open!
369 UE_AUTORTFM_ONABORT(this)
370 {
371 this->ReleaseRef();
372 };
373 }
374
375 void ReleaseRef()
376 {
377 UE_AUTORTFM_OPEN
378 {
379 // Only dirty root when doing a transition from a refcount to no refcounts and no other root flags present.
380 const int64 NewRefCount = FPlatformAtomics::InterlockedDecrement(&FlagsAndRefCount);
381 check((NewRefCount & EInternalObjectFlags_RefCountMask) >= 0);
382 if ((NewRefCount & EInternalObjectFlags_RefCountMask) == 0 && (GetFlagsInternal(NewRefCount) & (int32)EInternalObjectFlags_RootFlags) == 0)
383 {
384 MarkRootAsDirty();
385 }
386 };
387
388 // If the transaction is aborted we need to remember to re-add the reference we released in the open!
389 // Note: this is different to how we handle ref counts in general in UE. Elsewhere we will eagerly
390 // increment ref counts, but delay decrement. This is because `0` is normally the 'destroy the object'
391 // trigger. But for UObject's `0` just clears the `RefCounted` flag, and since GC cannot run between
392 // when we clear the flag and potentially re-add the flag below, the GC couldn't observe the state
393 // that the object wasn't being actively ref-counted and thus could be collected. We do it differently
394 // here because a bunch of the UObject systems depend on the ref count being correct after calls to
395 // add/release.
396 UE_AUTORTFM_ONABORT(this)
397 {
398 this->AddRef();
399 };
400 }

436 inline bool AtomicallySetFlag_ForGC(EInternalObjectFlags FlagToSet)
437 {
438 static_assert(sizeof(int64) == sizeof(FlagsAndRefCount), "Flags must be 64-bit for atomics.");
439 bool bIChangedIt = false;
440 while (true)
441 {
442 const int64 StartValue = FPlatformAtomics::AtomicRead_Relaxed((int64*)&FlagsAndRefCount);
443 const int32 StartFlagsValue = GetFlagsInternal(StartValue);
444 if ((StartFlagsValue & int32(FlagToSet)) == int32(FlagToSet))
445 {
446 break;
447 }
448 const int32 NewFlagsValue = StartFlagsValue | int32(FlagToSet);
449 const int64 NewValue = ((int64)NewFlagsValue << 32) | GetRefCountInternal(StartValue);
450 if (FPlatformAtomics::InterlockedCompareExchange((int64*)&FlagsAndRefCount, NewValue, StartValue) == StartValue)
451 {
452 bIChangedIt = true;
453 break;
454 }
455 }
456 return bIChangedIt;
457 }

707class FChunkedFixedUObjectArray
708{
709 enum
710 {
711 NumElementsPerChunk = 64 * 1024,
712 };
713
715 FUObjectItem** Objects;
717 TSAN_ATOMIC(int32) NumElements;
719 TSAN_ATOMIC(int32) MaxElements;
721 TSAN_ATOMIC(int32) NumChunks;
723 int32 MaxChunks;
725 FUObjectItem* PreAllocatedObjects;
726
727 static constexpr bool bFUObjectItemIsPacked = UE_ENABLE_FUOBJECT_ITEM_PACKING;
728
729
733 void ExpandChunksToIndex(int32 Index)
734 {
735 check(Index >= 0 && Index < MaxElements);
736 int32 ChunkIndex = Index / NumElementsPerChunk;
737 while (ChunkIndex >= NumChunks)
738 {
739 // add a chunk, and make sure nobody else tries
740 FUObjectItem** Chunk = &Objects[NumChunks];
741 FUObjectItem* NewChunk = new FUObjectItem[NumElementsPerChunk];
742 if (FPlatformAtomics::InterlockedCompareExchangePointer((void**)Chunk, NewChunk, nullptr))
743 {
744 // someone else beat us to the add, we don't support multiple concurrent adds
745 check(0);
746 }
747 else
748 {
749 NumChunks++;
750 check(NumChunks <= MaxChunks);
751 }
752 }
753 check(ChunkIndex < NumChunks && Objects[ChunkIndex]); // should have a valid pointer now
754 }
755
756public:
757
759 FChunkedFixedUObjectArray()
760 : Objects(nullptr)
761 , NumElements(0)
762 , MaxElements(0)
763 , NumChunks(0)
764 , MaxChunks(0)
765 , PreAllocatedObjects(nullptr)
766 {
767 }

769 ~FChunkedFixedUObjectArray()
770 {
771 if (!PreAllocatedObjects)
772 {
773 for (int32 ChunkIndex = 0; ChunkIndex < MaxChunks; ++ChunkIndex)
774 {
775 delete[] Objects[ChunkIndex];
776 }
777 }
778 else
779 {
780 delete[] PreAllocatedObjects;
781 }
782 delete[] Objects;
783 }

789 void PreAllocate(int32 InMaxElements, bool bPreAllocateChunks)
790 {
791 check(!Objects);
792 MaxChunks = InMaxElements / NumElementsPerChunk + 1;
793 MaxElements = MaxChunks * NumElementsPerChunk;
794 Objects = new FUObjectItem*[MaxChunks];
795 FMemory::Memzero(Objects, sizeof(FUObjectItem*) * MaxChunks);
796 if (bPreAllocateChunks)
797 {
798 // Fully allocate all chunks as contiguous memory
799 PreAllocatedObjects = new FUObjectItem[MaxElements];
800 for (int32 ChunkIndex = 0; ChunkIndex < MaxChunks; ++ChunkIndex)
801 {
802 Objects[ChunkIndex] = PreAllocatedObjects + ChunkIndex * NumElementsPerChunk;
803 }
804 NumChunks = MaxChunks;
805 }
806 }
