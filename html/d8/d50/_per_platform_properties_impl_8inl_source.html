22template<typename StructType, typename ValueType, EName BasePropertyName>
23FArchive& operator<<(FArchive& Ar, TPerPlatformProperty<StructType, ValueType, BasePropertyName>& Property)
24{
25 bool bStripSerializeAllProperties = true;
26#if WITH_EDITOR
27 bool bClientUsePreviewPPXData = false;
28 if(Ar.IsCooking())
29 {
30 bool bIsClientOnly = Ar.CookingTarget()->IsClientOnly();
31 bool bIsServerOnly = Ar.CookingTarget()->IsServerOnly();
32 bool bHasEditorOnlyData = Ar.CookingTarget()->HasEditorOnlyData();
33 if(bIsClientOnly || !(bIsClientOnly || bIsServerOnly || bHasEditorOnlyData))
34 {
35 bClientUsePreviewPPXData = Ar.CookingTarget()->GetConfigSystem()->GetBoolOrDefault(TEXT("CookedPreviewData"), TEXT("bClientUsePreviewPPXData"), false, GEngineIni);
36 }
37 }
38 bStripSerializeAllProperties = !bClientUsePreviewPPXData && Ar.IsCooking() && !Ar.IsSavingOptionalObject();
39#elif WITH_PREVIEW_PPX_DATA
40 bStripSerializeAllProperties = false;
41#endif
42
43 StructType* This = StaticCast<StructType*>(&Property);
44 Ar << bStripSerializeAllProperties;
45#if WITH_EDITOR
46 if (Ar.IsCooking() && !Ar.IsSavingOptionalObject())
47 {
48 // Save out platform override if it exists and Default otherwise
49 ValueType Value = Property.GetValueForPlatform(*Ar.CookingTarget()->IniPlatformName());
50 Ar << Value;
51 }
52 else
53#endif
54 {
55 Ar << This->Default;
56 }
57
58 if (!bStripSerializeAllProperties)
59 {
60#if WITH_EDITORONLY_DATA
61 if (Ar.IsCooking() && !Ar.IsSavingOptionalObject())
62 {
63 TMap<FName, ValueType> ToBeCookedPerPlatform(This->PerPlatform);
64 for (auto It = ToBeCookedPerPlatform.CreateIterator(); It; ++It)
65 {
66 if (FDataDrivenPlatformInfoRegistry::GetPlatformInfo(It.Key()).bIsConfidential)
67 {
68 It.RemoveCurrent();
69 }
70 }
71 Ar << ToBeCookedPerPlatform;
72 }
73 else
74 {
75 Ar << This->PerPlatform;
76 }
77#elif WITH_PREVIEW_PPX_DATA
78 {
79 Ar << This->PerPlatform;
80 }
81#else
82 // If we have old assets that were serialized with a PerPlatform map, but we no longer need it.
83 TMap<FName, ValueType> TempPerPlatform;
84 Ar << TempPerPlatform;
85#endif
86 }
87
88 return Ar;
89}
