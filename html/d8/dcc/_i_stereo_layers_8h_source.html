19 // The pointer to the shape of a stereo layer is wrapped in this simple class that clones and destroys the shape.
20 // This is to avoid having to implement non-default copy and assignment operations for the entire FLayerDesc struct.
21 struct FShapeWrapper
22 {
23 TUniquePtr<IStereoLayerShape> Wrapped;
24 FShapeWrapper(IStereoLayerShape* InWrapped)
25 : Wrapped(InWrapped)
26 {}
27
28 FShapeWrapper(const FShapeWrapper& In) :
29 Wrapped(In.IsValid()?In.Wrapped->Clone():nullptr)
30 {}
31
32 FShapeWrapper(FShapeWrapper&& In)
33 {
34 Wrapped = MoveTemp(In.Wrapped);
35 }
36
37 FShapeWrapper& operator= (IStereoLayerShape* InWrapped)
38 {
39 Wrapped = TUniquePtr<IStereoLayerShape>(InWrapped);
40 return *this;
41 }

44 {
45 return operator=(In.IsValid() ? In.Wrapped->Clone() : nullptr);
46 }
47
48 FShapeWrapper& operator= (FShapeWrapper&& In)
49 {
50 Wrapped = MoveTemp(In.Wrapped);

66 enum ELayerFlags
67 {
68 // Internally copies the texture on every frame for video, etc.
69 LAYER_FLAG_TEX_CONTINUOUS_UPDATE = 0x00000001,
70 // Ignore the textures alpha channel, this makes the stereo layer opaque. Flag is ignored on Steam VR.
71 LAYER_FLAG_TEX_NO_ALPHA_CHANNEL = 0x00000002,
72 // Quad Y component will be calculated based on the texture dimensions
73 LAYER_FLAG_QUAD_PRESERVE_TEX_RATIO = 0x00000004,
74 // The layer will intersect with the scene's depth. Currently only supported on Oculus platforms.
75 LAYER_FLAG_SUPPORT_DEPTH = 0x00000008,
76 // Required on some platforms to enable rendering of external textures.
77 LAYER_FLAG_TEX_EXTERNAL = 0x00000010,
78 // When set, this layer will not be rendered.
79 LAYER_FLAG_HIDDEN = 0x00000020,
80 // When this is set and the HMD implementation is compatible, the layer will be copied to the spectator screen.
81 LAYER_FLAG_DEBUG = 0x00000040,
82 // Max flag value, update this when new flags are added!
83 LAYER_FLAG_MAX_VALUE = LAYER_FLAG_DEBUG << 1,
84 };
