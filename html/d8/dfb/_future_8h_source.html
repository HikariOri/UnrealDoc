647template<typename ResultType>
648template<typename Func>
649auto TFutureBase<ResultType>::Then(Func Continuation) //-> TFuture<decltype(Continuation(MoveTemp(TFuture<ResultType>())))>
650{
651 check(IsValid());
652 using ReturnValue = decltype(Continuation(MoveTemp(TFuture<ResultType>())));
653
654 TPromise<ReturnValue> Promise;
655 TFuture<ReturnValue> FutureResult = Promise.GetFuture();
656 TUniqueFunction<void()> Callback = [PromiseCapture = MoveTemp(Promise), ContinuationCapture = MoveTemp(Continuation), StateCapture = this->State]() mutable
657 {
658 if constexpr (std::is_void_v<ReturnValue>)
659 {
660 ContinuationCapture(TFuture<ResultType>(MoveTemp(StateCapture)));
661 PromiseCapture.SetValue();
662 }
663 else
664 {
665 PromiseCapture.SetValue(ContinuationCapture(TFuture<ResultType>(MoveTemp(StateCapture))));
666 }
667 };
668
669 // This invalidate this future.
670 TSharedPtr<StateType> MovedState = MoveTemp(this->State);
671 MovedState->SetContinuation(MoveTemp(Callback));
672 return FutureResult;
673}

675// Next implementation
676template<typename ResultType>
677template<typename Func>
678auto TFutureBase<ResultType>::Next(Func Continuation) //-> TFuture<decltype(Continuation(Consume()))>
679{
680 return this->Then([Continuation = MoveTemp(Continuation)](TFuture<ResultType> Self) mutable
681 {
682 if constexpr (std::is_void_v<ResultType>)
683 {
684 Self.Consume();
685 return Continuation();
686 }
687 else
688 {
689 return Continuation(Self.Consume());
690 }
691 });
