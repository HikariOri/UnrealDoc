65 {
66 const FVector2d AB = SegmentAB.GetVector().GetSafeNormal();
67 const FVector2d CD = SegmentCD.GetVector().GetSafeNormal();

264 template<typename ExtremityType1, typename ExtremityType2>
265 const SegmentType* FindIntersectingSegment(const ExtremityType1* StartExtremity, const ExtremityType2* EndExtremity) const
266 {
267 using namespace IntersectionSegmentTool;
268 const SegmentType InSegment(Grid, Tolerance, *StartExtremity, *EndExtremity);
269
270 for (const SegmentType& Segment : Segments)
271 {
272 if (!Segment.IsValid())
273 {
274 continue;
275 }
276
277 if (bSegmentsAreSorted)
278 {
279 if (Segment.IsFullyBefore(InSegment))
280 {
281 continue;
282 }
283
284 if (Segment.IsFullyAfter(InSegment))
285 {
286 break;
287 }
288 }
289
290 switch (Segment.DoesItStartFromAndSuperimposed(StartExtremity, EndExtremity, InSegment.Segment2D))
291 {
292 case EConnectionType::SameSegment:
293 case EConnectionType::StartFrom:
294 continue;
295 case EConnectionType::SuperimposedByOrOn:
296 return &Segment;
297 case EConnectionType::DoesntStartFrom:
298 default:
299 break;
300 }
301
302 if (Segment.DoesItIntersect(InSegment))
303 {
304 return &Segment;
305 }
306 }
307
308 return nullptr;
309 }

347 case EConnectionType::SuperimposedByOrOn:
