217template <typename T, typename Allocator>
218inline bool LoadFromCompactBinary(FCbFieldView Field, TArray<T, Allocator>& OutValue)
219{
220 const uint64 Size = Field.AsArrayView().Num();
221 if (Size <= MAX_int32)
222 {
223 OutValue.Reset((int32)Size);
224 bool bOk = !Field.HasError();
225 for (const FCbFieldView& ElementField : Field)
226 {
227 bOk &= LoadFromCompactBinary(ElementField, OutValue.Emplace_GetRef());
228 }
229 return bOk;
230 }
231 else
232 {
233 UE::Serialization::Private::LogFieldTooLargeForArrayWarning(Size);
234 return false;
235 }
