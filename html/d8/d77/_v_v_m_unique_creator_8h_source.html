18// Ensure all created items are unique. If an old one exist then it's reused, otherwise a new is created.
19// Will use a hash table in the future, but for now a simple array and linear search.
20
21template <typename Type>
22class VUniqueCreator : public FGlobalHeapRoot
23{
24private:
25 UE::FMutex Mutex;
26
27 // This should be a hash table
28 static constexpr uint32 MaxSize = 4 * 1024;
29 TWriteBarrier<Type> Items[MaxSize];
30 uint32 ItemsEnd = 0;
31
32 template <typename SubType, typename... ArgumentTypes>
33 Type* LookupLocked(ArgumentTypes... Arguments)
34 {
35 for (uint32 Ix = 0; Ix < ItemsEnd; ++Ix)
36 {
37 if (SubType::Equals(*Items[Ix], Arguments...))
38 {
39 return Items[Ix].Get();
40 }
41 }
42 return nullptr;
43 }
44
45 Type* AddLocked(FAccessContext Context, Type* Item)
46 {
47 V_DIE_IF(ItemsEnd >= MaxSize);
48 Items[ItemsEnd].Set(Context, Item);
49 ItemsEnd++;
50 return Item;
51 }
52
53public:
54 VUniqueCreator() = default;
55
56 template <typename SubType, typename... ArgumentTypes>
57 AUTORTFM_DISABLE Type* GetOrCreate(FAllocationContext Context, ArgumentTypes... Arguments)
58 {
59 static_assert(std::is_base_of_v<Type, SubType>);
60 // This lock acquisition is *almost* wrong. If this UniqueCreator ever participated in census or a fixpoint,
61 // then allocation slow paths could call into some GC callback on UniqueCreator, and then that would need to
62 // grab this lock, and we'd die.
63 UE::TUniqueLock Lock(Mutex);
64 if (Type* Item = LookupLocked<SubType, ArgumentTypes...>(Arguments...))
65 {
66 return Item;
67 }
68 Type* Item = SubType::New(Context, Arguments...);
69 return AddLocked(Context, Item);
70 }
71
72 // Only call if certain that Item is unique
73 Type* Add(FAccessContext Context, Type* Item)
74 {
75 UE::TUniqueLock Lock(Mutex);
76 return AddLocked(Context, Item);
77 }
78
79 void Visit(FMarkStackVisitor& Visitor) override { VisitImpl(Visitor); }
80 void Visit(FAbstractVisitor& Visitor) override { VisitImpl(Visitor); }
81
82 template <class VisitorType>
83 void VisitImpl(VisitorType& Visitor)
84 {
85 UE::TUniqueLock Lock(Mutex);
86 Visitor.Visit(Items, ItemsEnd, TEXT("Items"));
87 }
88};
89}; // namespace Verse
