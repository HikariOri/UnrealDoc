460 static float WrapPosition(EAnimSequenceTrackLoopMode LoopMode, double Position, double SequenceLength)
461 {
462 if (LoopMode == EAnimSequenceTrackLoopMode::Loop)
463 {
464 if (SequenceLength > 0.0)
465 {
466 Position = FMath::Fmod(Position, SequenceLength);
467
468 if (Position < 0.0)
469 {
470 Position += SequenceLength;
471 }
472 }
473 else
474 {
475 Position = 0.0;
476 }
477 }
478 else
479 {
480 Position = FMath::Clamp(Position, 0.0, SequenceLength);
481 }
482
483 return static_cast<float>(Position);
484 }

488 {
489 if (FMath::Abs(PlayRate) > UE_SMALL_NUMBER)
490 {
491 return WorldTime - static_cast<double>(Position / PlayRate);
492 }
493 return WorldTime;
494 }
