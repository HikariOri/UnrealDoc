357template <typename ThumbnailSceneType, int32 MaxNumScenes>
358class TClassInstanceThumbnailScene
359{
360public:
362 TClassInstanceThumbnailScene()
363 {
364 InstancedThumbnailScenes.Reserve(MaxNumScenes);
365 MRUClassNames.Reserve(MaxNumScenes);
366 }
367
369 TSharedPtr<ThumbnailSceneType> FindThumbnailScene(const UClass* InClass) const
370 {
371 check(InClass);
372 const FName ClassName = InClass->GetFName();
373
374 TSharedPtr<ThumbnailSceneType> Scene = InstancedThumbnailScenes.FindRef(ClassName);
375 if (Scene.IsValid())
376 {
377 MRUClassNames.AddUnique(ClassName);
378 }
379
380 return Scene;
381 }
382
384 TSharedRef<ThumbnailSceneType> EnsureThumbnailScene(const UClass* InClass)
385 {
386 check(InClass);
387 const FName ClassName = InClass->GetFName();
388
389 TSharedPtr<ThumbnailSceneType> ExistingThumbnailScene = InstancedThumbnailScenes.FindRef(ClassName);
390 if (!ExistingThumbnailScene.IsValid())
391 {
392 if (InstancedThumbnailScenes.Num() >= MaxNumScenes)
393 {
394 FName ClassNameToRemove = MRUClassNames.Pop();
395 TSharedPtr<ThumbnailSceneType> SceneToDelete;
396 InstancedThumbnailScenes.RemoveAndCopyValue(ClassNameToRemove, SceneToDelete);
397 ScenesToDelete.Add(MoveTemp(SceneToDelete));
398
399 // Now that we have scenes to delete, schedule them to be destroyed in batches periodically until we have none left.
400 // Destroying many scenes at once can result in large stalls when iterating through content so we attempt to
401 // smooth that out here by deferring destruction, doing destruction in small, timed, batches and destroying the
402 // least recently used first to avoid deleting something we are likely to immediately recreate
403 const int32 MinBatchSize = 8;
404 if (!SceneDeleteTimerHandle.IsValid() && ScenesToDelete.Num() >= MinBatchSize)
405 {
406 checkf(IsInGameThread(), TEXT("Must be running on the game thread."));
407 GEditor->GetTimerManager()->SetTimer(SceneDeleteTimerHandle, FTimerDelegate::CreateLambda([this]() mutable
408 {
409 checkf(IsInGameThread(), TEXT("Must be running on the game thread."));
410 const double TimeoutSeconds = 0.25;
411
412 int32 Iteration = 0;
413 double StartTime = FPlatformTime::Seconds();
414 while (!ScenesToDelete.IsEmpty())
415 {
416 if (++Iteration > MinBatchSize)
417 {
418 if ((FPlatformTime::Seconds() - StartTime) > TimeoutSeconds)
419 {
420 break;
421 }
422 Iteration = 1;
423 }
424
425 // The destruction of these scenes can be quite expensive if the world has many objects
426 ScenesToDelete.Pop(EAllowShrinking::No);
427 }
428
429 CollectGarbage(GARBAGE_COLLECTION_KEEPFLAGS);
430
431 if (ScenesToDelete.IsEmpty())
432 {
433 ScenesToDelete.Shrink();
434 FTimerHandle LocalHandle = SceneDeleteTimerHandle;
435 SceneDeleteTimerHandle.Invalidate();
436 GEditor->GetTimerManager()->ClearTimer(LocalHandle); // TimerManager owns this lambda and clearing the timer will destroy it
437 }
438 }), 2.0f/*InRate*/, true /*bInLoop*/, 1.0f/*InFirstDelay*/);
439 }
440 }
441
442 ExistingThumbnailScene = MakeShareable(new ThumbnailSceneType());
443 InstancedThumbnailScenes.Add(ClassName, ExistingThumbnailScene);
444 MRUClassNames.Add(ClassName);
445 }
446
447 return ExistingThumbnailScene.ToSharedRef();
448 }
449
451 void RemoveThumbnailScene(const UClass* InClass)
452 {
453 check(InClass);
454 const FName ClassName = InClass->GetFName();
455 InstancedThumbnailScenes.Remove(ClassName);
456 MRUClassNames.RemoveSingle(ClassName);
457 }
458
460 void Clear()
461 {
462 InstancedThumbnailScenes.Reset();
463 MRUClassNames.Reset();
464 }
465
466private:
471 TMap<FName, TSharedPtr<ThumbnailSceneType>> InstancedThumbnailScenes;
472
474 mutable TMRUArray<FName, TFixedAllocator<MaxNumScenes>> MRUClassNames;
476 TArray<TSharedPtr<ThumbnailSceneType>> ScenesToDelete;
478 FTimerHandle SceneDeleteTimerHandle;
479};
