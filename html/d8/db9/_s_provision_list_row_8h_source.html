178 {

185 return FSlateColor(FLinearColor(1.0f, 0.0f, 0.0f));
186 }
187 else if (Provision->bSelected)
188 {
189 return FSlateColor(FLinearColor(0.0f, 1.0f, 0.0f));
190 }
191 return FSlateColor(FLinearColor(1.0f, 1.0f, 1.0f));
192 }
193

215 void HandleCheckStateChanged(ECheckBoxState InState)
216 {
217 Provision->bManuallySelected = InState == ECheckBoxState::Checked;
218
219 // update the property
220 if (OnProvisionChanged_Handler.IsBound())
221 {
222 OnProvisionChanged_Handler.Execute(Provision->bManuallySelected ? Provision->FileName : "");

224
225 // disable any other objects
226 for (int32 Idx = 0; Idx < ProvisionList->Num(); ++Idx)
227 {
228 if ((*ProvisionList)[Idx] != Provision && (*ProvisionList)[Idx]->bManuallySelected)
229 {
230 (*ProvisionList)[Idx]->bManuallySelected = false;
