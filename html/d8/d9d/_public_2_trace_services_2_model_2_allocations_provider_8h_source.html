25 // Allocation query rules.
26 // The enum uses the following naming convention:
27 // A, B, C, D = time markers
28 // a = time when "alloc" event occurs
29 // f = time when "free" event occurs (can be infinite)
30 // Ex.: "AaBf" means "all memory allocations allocated between time A and time B and freed after time B".
31 enum class EQueryRule
32 {
33 aAf, // active allocs at A
34 afA, // before
35 Aaf, // after
36 aAfB, // decline
37 AaBf, // growth
38 aAfaBf, // decline + growth (used as "growth vs. decline")
39 AfB, // free events
40 AaB, // alloc events
41 AafB, // short living allocs
42 aABf, // long living allocs
43 AaBCf, // memory leaks
44 AaBfC, // limited lifetime
45 aABfC, // decline of long living allocs
46 AaBCfD, // specific lifetime
47 AoB, // all paged out allocs between A and B
48 AiB, // all paged in allocs between A and B
49 //A_vs_B, // compare A vs. B; {aAf} vs. {aBf}
50 //A_or_B, // live at A or at B; {aAf} U {aBf}
51 //A_xor_B, // live either at A or at B; ({aAf} U {aBf}) \ {aABf}
52 };
