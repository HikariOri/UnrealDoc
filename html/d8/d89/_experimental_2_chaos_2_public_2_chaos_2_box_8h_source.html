219 // Get the nearest point on an edge and the edge vertices
220 // Used for manifold generation
221 FVec3 GetClosestEdge(int32 PlaneIndexHint, const FVec3& Position, FVec3& OutEdgePos0, FVec3& OutEdgePos1) const
222 {
223 TVector<T, d> ClosestEdgePosition = FVec3(0);
224 if (PlaneIndexHint >= 0)
225 {
226 FReal ClosestDistanceSq = FLT_MAX;
227
228 int32 PlaneVerticesNum = NumPlaneVertices(PlaneIndexHint);
229 if (PlaneVerticesNum > 0)
230 {
231 FVec3 P0 = GetVertex(GetPlaneVertex(PlaneIndexHint, PlaneVerticesNum - 1));
232 for (int32 PlaneVertexIndex = 0; PlaneVertexIndex < PlaneVerticesNum; ++PlaneVertexIndex)
233 {
234 const int32 VertexIndex = GetPlaneVertex(PlaneIndexHint, PlaneVertexIndex);
235 const TVector<T, d> P1 = GetVertex(VertexIndex);
236
237 const TVector<T, d> EdgePosition = FMath::ClosestPointOnLine(P0, P1, Position);
238 const FReal EdgeDistanceSq = (EdgePosition - Position).SizeSquared();
239
240 if (EdgeDistanceSq < ClosestDistanceSq)
241 {
242 ClosestDistanceSq = EdgeDistanceSq;
243 ClosestEdgePosition = EdgePosition;
244 OutEdgePos0 = P0;
245 OutEdgePos1 = P1;
246 }
247
248 P0 = P1;
249 }
250 }
251 }
252 else
253 {
254 // @todo(chaos)
255 check(false);
256 }
257 return ClosestEdgePosition;
258 }

430 FORCEINLINE static PMatrix<T, 3, 3> GetInertiaTensor(const T Mass, const TVector<T, 3>& Dim)
431 {
432 // https://www.wolframalpha.com/input/?i=cuboid
433 const T M = Mass / 12;
434 const T WW = Dim[0] * Dim[0];
435 const T HH = Dim[1] * Dim[1];

483 template<typename OtherType>
484 static void SerializeAsAABBs(FArchive& Ar, TArray<TAABB<OtherType, d>>& AABBs)
485 {
486 Ar.UsingCustomVersion(FExternalPhysicsCustomObjectVersion::GUID);
487 if (Ar.CustomVer(FExternalPhysicsCustomObjectVersion::GUID) < FExternalPhysicsCustomObjectVersion::TBoxReplacedWithTAABB)
488 {
489 TArray<TBox<T, d>> Tmp;
490 Ar << Tmp;
491 AABBs.Reset(Tmp.Num());
492 for (const TBox<T, d>& Box : Tmp)
493 {
494 AABBs.Emplace(TAABB<OtherType, d>(Box.AABB));
495 }
496 }
497 else if (std::is_same_v<T, OtherType>)
498 {
499 Ar << AABBs;
500 }
501 else if (Ar.IsLoading())
502 {
503 // Reload to a different floating point type
504 TArray<TAABB<T, d>> TmpAABBs;
505 Ar << TmpAABBs;
506 AABBs.Reset(TmpAABBs.Num());
507 for (const TAABB<T, d>& TmpAABB : TmpAABBs)
508 {
509 AABBs.Emplace(TAABB<OtherType, d>(TmpAABB));
510 }
511 }
512 else
513 {
514 // Serialize from a different floating point type
515 TArray<TAABB<T, d>> TmpAABBs;
516 TmpAABBs.Reserve(AABBs.Num());
517 for (const TAABB<OtherType, d>& AABB : AABBs)
518 {
519 TmpAABBs.Emplace(TAABB<T, d>(AABB));
520 }

522 }
