129inline bool FAndroidTargetDevice::GetAdbFullFilename(FString& OutFilename)
130{
131 TOptional<FString> ResultPath;
132
133 // get the SDK binaries folder
134 FString AndroidDirectory = FPlatformMisc::GetEnvironmentVariable(TEXT("ANDROID_HOME"));
135 if (AndroidDirectory.Len() == 0)
136 {
137 return false;
138 }
139
140#if PLATFORM_WINDOWS
141 OutFilename = FString::Printf(TEXT("%s\\platform-tools\\adb.exe"), *AndroidDirectory);
142#else
143 OutFilename = FString::Printf(TEXT("%s/platform-tools/adb"), *AndroidDirectory);
144#endif
145
146 return true;
147}

149inline bool FAndroidTargetDevice::ExecuteAdbCommand( const FString& CommandLine, FString* OutStdOut, FString* OutStdErr ) const
150{
151 FString Filename;
152 if (!GetAdbFullFilename(Filename))
153 {
154 return false;
155 }
156
157 // execute the command
158 int32 ReturnCode;
159 FString DefaultError;
160
161 // make sure there's a place for error output to go if the caller specified NULL
162 if (!OutStdErr)
163 {
164 OutStdErr = &DefaultError;
165 }
166
167 FString CommandLineWithDevice;
168 // the devices command should never include a specific device
169 if (CommandLine == TEXT("devices"))
170 {
171 CommandLineWithDevice = CommandLine;
172 }
173 else
174 {
175 CommandLineWithDevice = FString::Printf(TEXT("-s %s %s"), *SerialNumber, *CommandLine);
176 }
177
178 FPlatformProcess::ExecProcess(*Filename, *CommandLineWithDevice, &ReturnCode, OutStdOut, OutStdErr);
179
180 if (ReturnCode != 0)
181 {
182 FPlatformMisc::LowLevelOutputDebugStringf(TEXT("The Android SDK command '%s' failed to run. Return code: %d, Error: %s\n"), *CommandLineWithDevice, ReturnCode, **OutStdErr);
183
184 return false;
185 }
186
187 return true;
188}
