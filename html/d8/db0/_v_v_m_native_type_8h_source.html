91 FNativeType(UStruct* InType)

95 if (!InType)

98 }

102 }

104 {
105 VerseType = VerseStruct->Class.Get();
106 }
107 else
108 {
109 VerseType = Verse::GlobalProgram->LookupImport(Context, InType);
110 V_DIE_UNLESS(VerseType);
111 }
112
113 Type.Set(Context, VerseType);
114 }
