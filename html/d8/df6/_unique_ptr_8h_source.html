16// Single-ownership smart pointer in the vein of std::unique_ptr.
17// Use this when you need an object's lifetime to be strictly bound to the lifetime of a single smart pointer.
18//
19// This class is non-copyable - ownership can only be transferred via a move operation, e.g.:
20//
21// TUniquePtr<MyClass> Ptr1(new MyClass); // The MyClass object is owned by Ptr1.
22// TUniquePtr<MyClass> Ptr2(Ptr1); // Error - TUniquePtr is not copyable
23// TUniquePtr<MyClass> Ptr3(MoveTemp(Ptr1)); // Ptr3 now owns the MyClass object - Ptr1 is now nullptr.
24//
25// If you provide a custom deleter, it is up to your deleter to handle null pointers. This is a departure
26// from std::unique_ptr which will not invoke the deleter if the owned pointer is null:
27// https://en.cppreference.com/w/cpp/memory/unique_ptr/~unique_ptr
28
29template <typename T>
30struct TDefaultDelete
31{
32 DECLARE_INLINE_TYPE_LAYOUT(TDefaultDelete, NonVirtual);
33
34 TDefaultDelete() = default;
35 TDefaultDelete(const TDefaultDelete&) = default;
36 TDefaultDelete& operator=(const TDefaultDelete&) = default;
37 ~TDefaultDelete() = default;
38
39 template <UE::CPointerConvertibleTo<T> U>
40 TDefaultDelete(const TDefaultDelete<U>&)
41 {
42 }
43
44 template <UE::CPointerConvertibleTo<T> U>
45 TDefaultDelete& operator=(const TDefaultDelete<U>&)
46 {
47 return *this;
48 }
49
50 void operator()(T* Ptr) const
51 {
52 // If you get an error here when trying to use a TUniquePtr<FForwardDeclaredType> inside a UObject then:
53 //
54 // * Declare all your UObject's constructors and destructor in the .h file.
55 // * Define all of them in the .cpp file. You can use UMyObject::UMyObject() = default; to auto-generate
56 // the default constructor and destructor so that they don't have to be manually maintained.
57 // * Define a UMyObject(FVTableHelper& Helper) constructor too, otherwise it will be defined in the
58 // .gen.cpp file where your pimpl type doesn't exist. It cannot be defaulted, but it need not
59 // contain any particular implementation; the object just needs to be garbage collectable.
60 //
61 // If efficiency is less important than simplicity, you may want to consider
62 // using a TPimplPtr instead, though it is also pretty efficient.
63 delete Ptr;
64 }
65};
