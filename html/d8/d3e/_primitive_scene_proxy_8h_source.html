182 inline const FSimpleLightPerViewEntry& GetViewDependentData(int32 LightIndex, int32 ViewIndex, int32 NumViews) const
183 {
184 // If InstanceData has an equal number of elements to PerViewData then all views share the same PerViewData.
185 if (InstanceData.Num() == PerViewData.Num())
186 {
187 check(InstancePerViewDataIndices.Num() == 0);

189 }
190 else
191 {
192 check(InstancePerViewDataIndices.Num() == InstanceData.Num());
193
194 // Calculate per-view index.
195 FSimpleLightInstacePerViewIndexData PerViewIndex = InstancePerViewDataIndices[LightIndex];
196 const int32 PerViewDataIndex = PerViewIndex.PerViewIndex + ( PerViewIndex.bHasPerViewData ? ViewIndex : 0 );
197 return PerViewData[PerViewDataIndex];
198 }
199 }

228enum class ERayTracingPrimitiveFlags : uint8
229{
230 None = 0,
231
232 // Visibility flags:
233
234 // This type of geometry is not supported in ray tracing and all proxies will be excluded
235 // If a proxy decides to return UnsupportedProxyType it must be consistent across all proxies of this type
236 // This value is used for primitives that return false from FPrimitiveSceneProxy::IsRayTracingRelevant().
237 UnsupportedProxyType = 1 << 0,
238
239 // This scene proxy will be excluded, because it decides to be invisible in ray tracing (probably due to other flags)
240 // Excluded proxies will skip visibility checks
241 Exclude = 1 << 1,
242
243 // Similar to Exclude however scene proxy will still go through culling and run relevant logic when it is deemed visible
244 Skip = 1 << 2,
245
246 // Caching flags:
247
248 // Fully dynamic (the ray tracing representation of this scene proxy will be polled every frame)
249 // Ray tracing mesh commands generated from this proxy's materials can be cached
250 // (not compatible with CacheInstances or ComputeLOD)
251 Dynamic = 1 << 3,
252 // Instances from this proxy can be cached
253 // (not compatible with Dynamic or ComputeLOD)
254 CacheInstances = 1 << 4,
255
256 // Misc flags:
257
258 // Static meshes with multiple LODs will want to select a LOD index based on screen size
259 // (not compatible with Dynamic or CacheInstances)
260 ComputeLOD = 1 << 5,
261
262 // Primitive is masked as a far field object
263 FarField = 1 << 6,
264
265 // Raytracing data is streamable (TODO: support in dynamic path)
266 Streaming = 1 << 7,
267};
268ENUM_CLASS_FLAGS(ERayTracingPrimitiveFlags);
