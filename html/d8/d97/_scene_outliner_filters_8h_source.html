148template <class T>
149struct TSceneOutlinerPredicateFilter : public TSceneOutlinerFilter<T>
150{
151 using TFilterPredicate = typename T::FFilterPredicate;
152 using TInteractivePredicate = typename T::FInteractivePredicate;
153
155 mutable TFilterPredicate FilterPred;
156 mutable TInteractivePredicate InteractivePred;
157
158 TSceneOutlinerPredicateFilter(TFilterPredicate InFilterPred, FSceneOutlinerFilter::EDefaultBehaviour InDefaultBehaviour, TInteractivePredicate InInteractivePredicate = TInteractivePredicate())
159 : TSceneOutlinerFilter<T>(InDefaultBehaviour)
160 , FilterPred(InFilterPred)
161 , InteractivePred(InInteractivePredicate)
162 {}
163
164 virtual bool PassesFilterImpl(const T& InItem) const override
165 {
166 return InItem.Filter(FilterPred);
167 }
168
169 virtual bool GetInteractiveStateImpl(const T& InItem) const override
170 {
171 if (InteractivePred.IsBound())
172 {
173 return InItem.GetInteractiveState(InteractivePred);
174 }
175
176 // If not interactive state impl is provided, default to interactive if filter passes
177 return PassesFilterImpl(InItem);
178 }
179};
